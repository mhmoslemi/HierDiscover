def smoothmax_gradient(a, lam):
    n = a.shape[0]
    b = full_convolution(a) # length 2n-1
    max_b = b.max()
    # Compute exp(lam * (b - max_b)) for numerical stability.
    scaled = np.exp(lam * (b - max_b))
    Z = scaled.sum()
    p = scaled / Z
    # gradient via convolution of p with a_rev
    a_rev = a[::-1]
    # convolution using FFT maybe faster
    grad_full = np.convolve(p, a_rev) # full conv
    grad = 2 * grad_full[n-1:n-1 + n]
    return grad, max_b
```

def fft_convolve(x, y):
    n = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    result = np.fft.irfft(X * Y, n=n_fft)[:n]

"""
Propose a low‑score step‑function using a smooth‑max / exponentiated‑gradient
homotopy followed by a short projected sub‑gradient polishing pass.

The objective of evaluate_sequence is
    2·n·max(conv(a,a)) / (sum(a)**2)
which is scale‑invariant, so we keep  Σa = 1.  The algorithm works on the
probability simplex:

1. Initialise a from the previously discovered construction
   ``height_sequence_1`` (if available) or from a uniform vector.
2. Run a mirror‑descent (exponentiated‑gradient) scheme with a
   smooth approximation of the max‑convolution:
       f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k) .
   The gradient of f_λ can be written as a convolution and is computed
   with an FFT.  The smoothness parameter λ is increased slowly, turning
   the surrogate into the true max‑objective.
3. When the time budget is almost exhausted, a few iterations of the
   classical projected sub‑gradient descent from the reference are applied
   for a fine polish.
4. The best vector found (converted to a plain Python list) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σv = 1, v≥0."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution using a real FFT.
    Length of result = len(x) + len(y) - 1.
    """
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (full linear convolution)."""
    return fft_convolve(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw a random point from Dirichlet(concentration,…,concentration)."""
    if concentration <= 0:
        raise ValueError("concentration must be positive")
    y = np.random.gamma(concentration, 1.0, size=n)
    if y.sum() == 0.0:
        raise RuntimeError("Dirichlet draw gave zero sum")
    return y / y.sum()

                          eta0: float = 0.5,
                          lam0: float = 1.0,
                          lam_factor: float = 1.001,
                          lam_max: float = 200.0) -> np.ndarray:
    """
    Mirror‑descent on the smooth‑max surrogate.
    a is always kept on the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    # hyper‑parameters
    eta = eta0          # base step for EG (will decay)
    lam = lam0          # smoothness
    it = 0
    no_improve = 0

    while time.time() < time_deadline:
        it += 1

        # ---- compute convolution and smooth probabilities ----
        b = full_convolution(a)               # length 2n‑1
        max_b = b.max()
        # stable soft‑max (shift by max_b)
        exp_terms = np.exp(lam * (b - max_b))
        p = exp_terms / exp_terms.sum()       # p_k ≥ 0, Σp = 1

        # ---- gradient of the smooth‑max (vector of length n) ----
        # grad = 2 * (p * a) convolved appropriately.
        a_rev = a[::-1]                       # reversed a
        conv_full = fft_convolve(p, a_rev)    # length 3n‑2
        grad = 2.0 * conv_full[n - 1:n - 1 + n]

        # ---- exponentiated‑gradient update (mirror descent) ----
        # learning rate decays like 1/√t to keep steps reasonable
        eta_t = eta / math.sqrt(it)
        a = a * np.exp(-eta_t * grad)
        a = a / a.sum()                        # simplex projection (normalisation)

        # ---- keep best so far ----
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[EG] iter={it}  score={best_score:.6f}  λ={lam:.2f}  time={time.time():.1f}s")
        else:
            no_improve += 1

        # ---- anneal smoothness parameter ----
        if lam < lam_max:
            lam = min(lam * lam_factor, lam_max)

        # ---- occasional jitter to escape plateaus ----
        if no_improve >= 50000:
            jitter = np.random.randn(n) * 1e-3
            a = best_a + jitter
            a = np.clip(a, 1e-14, None)
            a = a / a.sum()
            no_improve = 0
            print(f"[EG] jitter after {it} iters, reset to explore")
            # a small boost in step size after a restart
            eta = max(eta, 0.2)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polish (as in the reference algorithm)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray,
                       time_deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on max(conv(a,a)).
    Uses Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # active constraints (within tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of each active entry: 2 * a shifted
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
            print(f"[sub‑grad] iter={t}  new M={best_M:.6e}")

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative list) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # 0. Prepare an initial feasible vector (Σ = 1)
    # -----------------------------------------------------------------
    try:
        init = np.array([float(x) for x in height_sequence_1], dtype=np.float64)
        print(f"[init] using height_sequence_1 of length {init.size}")
    except Exception:
        init = None

    target_n = 4096                     # trade‑off between quality and speed
    if init is None or init.size == 0:
        init = np.full(target_n, 1.0 / target_n, dtype=np.float64)
        print("[init] fallback to uniform vector")
    else:
        # embed into target_n‑dimensional simplex (tiny epsilon everywhere)
        if init.size < target_n:
            a = np.full(target_n, 1e-12, dtype=np.float64)
            a[:init.size] = init
            init = a
        elif init.size > target_n:
            init = init[:target_n]
        # ensure strictly positive entries to allow multiplicative updates
        init = init + 1e-12
    init = init / init.sum()
    best_vec = init.copy()
    best_score = compute_score(best_vec)
    print(f"[init] initial score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 1. Optional quick random Dirichlet exploration (≈ 20 s)
    # -----------------------------------------------------------------
    explore_deadline = deadline - 200.0
    while time.time() < explore_deadline and time.time() - start_time < 20.0:
        conc = 10 ** np.random.uniform(-2, 1)   # 0.01 … 10
        cand = random_dirichlet(target_n, conc)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score, best_vec = cand_score, cand.copy()
            print(f"[rand] new best {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Smooth‑max exponentiated‑gradient homotopy (major part of budget)
    # -----------------------------------------------------------------
    eg_deadline = deadline - 30.0
    if time.time() < eg_deadline:
        a_opt = smoothmax_eg_optimize(best_vec, eg_deadline,
                                      eta0=0.6, lam0=1.0,
                                      lam_factor=1.001, lam_max=150.0)
        opt_score = compute_score(a_opt)
        if opt_score < best_score:
            best_score, best_vec = opt_score, a_opt.copy()
            print(f"[EG] post‑optim score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Final projected sub‑gradient polishing (remaining time)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        a_final = subgradient_polish(best_vec, deadline, step0=0.12)
        final_score = compute_score(a_final)
        if final_score < best_score:
            best_score, best_vec = final_score, a_final.copy()
            print(f"[final] polished score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 4. Safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_vec is None or best_vec.size == 0:
        best_vec = np.full(target_n, 1.0 / target_n, dtype=np.float64)
        best_score = compute_score(best_vec)

    # -----------------------------------------------------------------
    # 5. Final official evaluation (scale‑invariant, but we keep Σ=1)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_vec.tolist()
