def propose_candidate(seed=12345,budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    n = 2048   # maybe bigger after improvement?
    best_a = None
    best_score = np.inf

    # Starting candidates: uniform + some random Dirichlet draws (maybe 5)
    candidates = [np.full(n,1/n)]
    for _ in range(5):
        candidates.append(np.random.dirichlet(np.ones(n)*0.5))  # moderate concentration

    # Define optimizer function optimizing logits using Adam scheduler.
    def logit_optimize(a0, sub_deadline):
        # Convert a0 to logits via log; avoid zeros.
        eps = 1e-12
        x = np.log(np.maximum(a0, eps))
        # Adam parameters
        lr = 0.05
        beta1=0.9
        beta2=0.999
        eps_adam=1e-8
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        t=0
        best_local_a = a0.copy()
        best_local_score = compute_score(a0)
        tau = 0.5
        tau_end = 1e-12
        tau_decay=0.9
        decay_interval = 200
        while time.time()<sub_deadline:
            # compute a
            a = np.exp(x - np.max(x))  # subtract max for stability
            a = a / a.sum()
            # conv and gradient
            conv = full_convolution(a)
            # smooth max
            shift = conv.max()
            # choose tau decayed
            if t % decay_interval == 0 and tau > tau_end:
                tau = max(tau_end, tau*tau_decay)
            # compute weights
            scaled = (conv - shift) / tau
            # cap scaled maybe for overflow
            # Use np.exp safely
            exp_scaled = np.exp(scaled - np.max(scaled)) # shift for numeric
            w = exp_scaled / exp_scaled.sum()
            # compute gradient wrt a
            grad_a = conv_with_weights(a,w)
            # gradient wrt logits x
            # grad_x_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
            dot = np.dot(a, grad_a)
            grad_x = a * (grad_a - dot)
            # Adam step
            t+=1
            m = beta1*m + (1-beta1)*grad_x
            v = beta2*v + (1-beta2)*(grad_x**2)
            m_hat = m/(1-beta1**t)
            v_hat = v/(1-beta2**t)
            step = lr * m_hat/(np.sqrt(v_hat)+eps_adam)
            x = x - step
            # evaluate true score occasionally
            if t % 200 == 0:
                cur_a = np.exp(x - np.max(x))
                cur_a = cur_a / cur_a.sum()
                cur_score = compute_score(cur_a)
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_a = cur_a.copy()
        # final evaluate
        final_a = np.exp(x - np.max(x))
        final_a = final_a / final_a.sum()
        final_score= compute_score(final_a)
        if final_score < best_local_score:
            best_local_score, best_local_a = final_score, final_a.copy()
        return best_local_a, best_local_score

    # allocate time for optimization per candidate
    remaining = deadline - time.time()
    # allocate maybe half to optimizer, half to polishing and annealing
    opt_time_budget = remaining * 0.6
    # compute per candidate time: split equally among candidates
    per_candidate_time = opt_time_budget / len(candidates)
    for idx, a0 in enumerate(candidates):
        cand_deadline = time.time() + per_candidate_time
        a_opt, score_opt = logit_optimize(a0, cand_deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
        print(f"[optimize {idx}] score {score_opt:.6f}")

    # Polishing using subgradient (like earlier)
    if best_a is not None:
        polish_deadline = deadline - 5.0
        if time.time() < polish_deadline:
            a_polished = subgradient_polish(best_a, polish_deadline)
            pol_score = compute_score(a_polished)
            print(f"[polish] {pol_score:.6f}")
            if pol_score < best_score:
                best_score = pol_score
                best_a = a_polished.copy()

    # Simulated annealing final refinement
    if best_a is not None and time.time() < deadline - 2:
        a_final = simulated_annealing(best_a, deadline - 2)
        final_score = compute_score(a_final)
        print(f"[anneal] {final_score:.6f}")
        if final_score < best_score:
            best_score = final_score
            best_a = a_final.copy()
    # final return
    if best_a is None:
        best_a = np.full(n, 1/n)
        best_score = compute_score(best_a)

    print("Final score = ", best_score)
    return best_a.tolist()

"""
Optimisation of non‑negative step heights using a logit‑Adam optimiser
on a smooth‑max surrogate of the convolution peak, followed by a
sub‑gradient polish and a short simulated‑annealing refinement.
The algorithm explores many random Dirichlet initialisations, keeps the
few best candidates and spends the majority of the 1000 s budget
optimising them on the simplex (via a softmax parametrisation).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT based)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    Returns g_i = 2·Σ_k w_k·a_{k‑i}.
    """
    n = a.shape[0]
    m = w.shape[0]
    rev_w = w[::-1]
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Softmax‑based Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def logit_adam_optimize(a0: np.ndarray,
                        deadline: float,
                        lr: float = 0.05,
                        beta1: float = 0.9,
                        beta2: float = 0.999,
                        eps: float = 1e-8,
                        tau_start: float = 0.5,
                        tau_end: float = 1e-12,
                        tau_decay: float = 0.85,
                        decay_interval: int = 200) -> (np.ndarray, float):
    """
    Optimises a probability vector a (sum = 1, a_i≥0) by optimising its
    logits x (softmax(x) = a) with Adam on the smooth‑max of the
    convolution.  Returns the best a discovered and its exact score.
    """
    # initialise logits (avoid log(0) by clipping)
    eps0 = 1e-12
    x = np.log(np.clip(a0, eps0, None))

    m_t = np.zeros_like(x)
    v_t = np.zeros_like(x)
    t = 0
    tau = tau_start

    best_a = None
    best_val = np.inf

    while time.time() < deadline:
        # softmax to obtain probability vector
        shift = np.max(x)                     # for numerical stability
        ex = np.exp(x - shift)
        a = ex / ex.sum()

        # convolution and smooth‑max weights
        conv = full_convolution(a)
        shift_c = conv.max()
        # smooth‑max (log‑sum‑exp) with current temperature
        scaled = (conv - shift_c) / tau
        # shift again to avoid overflow inside exp
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()

        # gradient w.r.t a of the weighted sum
        grad_a = conv_with_weights(a, w)

        # chain‑rule to get gradient w.r.t logits x
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update on x
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad_x
        v_t = beta2 * v_t + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # temperature annealing
        if t % decay_interval == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # occasional exact evaluation
        if t % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_val:
                best_val = cur_score
                best_a = a.copy()

    # final evaluation
    shift = np.max(x)
    ex = np.exp(x - shift)
    a_final = ex / ex.sum()
    final_score = compute_score(a_final)
    if final_score < best_val:
        best_val, best_a = final_score, a_final.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as in the reference implementation)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        # project onto simplex
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (keeps an upper bound)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑height vector that minimises
    ``evaluate_sequence``.  The routine mixes a large random
    Dirichlet exploration, a logit‑Adam optimisation on a smooth‑max
    surrogate, sub‑gradient polishing and a short simulated‑annealing
    phase.  It respects the 1000 s wall‑clock limit.
    Returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    n = 2048                                   # problem dimension
    best_seq = None
    best_score = np.inf

    # -----------------------------------------------------------------
    # 0. Optional fallback to a known construction (if present)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Random Dirichlet exploration (≈ 60 s)
    # -----------------------------------------------------------------
    explore_deadline = min(deadline - 300.0, start + 60.0)
    top_k = 8                               # keep the best few
    best_candidates = []                    # list of (score, vector)

    def consider(v):
        sc = compute_score(v)
        if len(best_candidates) < top_k:
            best_candidates.append((sc, v.copy()))
            best_candidates.sort(key=lambda x: x[0])
        elif sc < best_candidates[-1][0]:
            best_candidates[-1] = (sc, v.copy())
            best_candidates.sort(key=lambda x: x[0])

    # always include the uniform vector
    consider(np.full(n, 1.0 / n))

    # sweep a few concentration parameters
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    while time.time() < explore_deadline:
        conc = np.random.choice(concs)
        vec = np.random.dirichlet(np.full(n, conc))
        consider(vec)

    # keep the best few vectors
    best_candidates.sort(key=lambda x: x[0])
    cand_vecs = [vec for sc, vec in best_candidates[:top_k]]
    print(f"[explore] best raw score = {best_candidates[0][0]:.6f}")

    # initialise global best from this pool
    if cand_vecs:
        best_seq = cand_vecs[0].tolist()
        best_score = best_candidates[0][0]

    # -----------------------------------------------------------------
    # 2. Logit‑Adam optimisation on the elite candidates (≈ 700 s)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200.0 and cand_vecs:
        # allocate roughly 80 % of what is left to optimisation
        opt_deadline = time.time() + remaining * 0.80
        per_candidate = (opt_deadline - time.time()) / len(cand_vecs)

        for idx, a0 in enumerate(cand_vecs):
            if time.time() >= opt_deadline:
                break
            sub_dead = min(opt_deadline, time.time() + per_candidate)
            print(f"[adam {idx}] start optimisation")
            a_opt, sc_opt = logit_adam_optimize(a0, sub_dead)
            print(f"[adam {idx}] refined score = {sc_opt:.6f}")
            if sc_opt < best_score:
                best_score, best_seq = sc_opt, a_opt.tolist()

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 60.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[polish] start")
        a_polished = subgradient_polish(a0, polish_deadline)
        sc_polish = compute_score(a_polished)
        print(f"[polish] refined score = {sc_polish:.6f}")
        if sc_polish < best_score:
            best_score, best_seq = sc_polish, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing final refinement (≈ 50 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[anneal] start")
        a_final = simulated_annealing(a_start, anneal_deadline)
        sc_ann = compute_score(a_final)
        print(f"[anneal] final score = {sc_ann:.6f}")
        if sc_ann < best_score:
            best_score, best_seq = sc_ann, a_final.tolist()

    # -----------------------------------------------------------------
    # 5. Safeguard – uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
