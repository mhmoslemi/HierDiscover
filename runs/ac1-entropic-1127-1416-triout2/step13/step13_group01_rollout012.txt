```
def project_simplex(v):
    # Implements algorithm from Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def subgradient_descent(a0, deadline, init_step=0.5, decay=0.99):
    a = a0.copy()
    step = init_step
    it = 0
    best_val = compute_score(a)
    best_a = a.copy()
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        k = int(conv.argmax())
        # gradient
        grad = np.zeros_like(a)
        if k < len(a):
            # overlapping part i=0..k
            grad[:k+1] = 2.0 * a[k::-1]
        else:
            start = k - len(a) + 1
            grad[start:] = 2.0 * a[:len(a)-start][::-1]
        # take step
        a = a - step * grad
        # projection onto simplex
        a = project_simplex(a)
        # compute objective occasionally
        if it % 100 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                # maybe log

                # adapt step? could reduce if not improving
        # decay step size
        step = step * decay
        # ensure step not too small
        if step < 1e-8:
            step = 1e-8
    return best_a

"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

The algorithm consists of four stages:
1. **Broad baseline sweep** – Explore many parametric families
   (power‑law, exponential, beta, Gaussian, linear‑decrease,
   truncated exponential) together with sorted Dirichlet draws.
   The best few vectors are kept.

2. **Smooth optimisation** – minimise a soft‑max approximation of the
   max‑convolution using Adam on a soft‑max (log‑softmax) parametrisation.
   Several restarts from the retained candidates are performed.

3. **Projected sub‑gradient descent** – directly minimises the true
   non‑smooth objective (the max of the convolution) by taking a step
   opposite to the sub‑gradient of the active convolution entry and
   projecting back onto the probability simplex.

4. **Final simulated‑annealing polish** – a Metropolis walk with a
   slowly cooling temperature to escape tiny local plateaux.

All stages respect the global time‑budget (default 1000 s).  The returned
sequence is a plain Python list of floats and is guaranteed to satisfy
the non‑negativity and normalisation checks performed by
``evaluate_sequence``.
"""

import time
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, same as the original code)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (used for the objective)."""
    return _linear_convolution(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective    2·n·max(conv(a,a))  (sum(a)=1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum=1)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centered in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_linear_decreasing(n: int, slope: float) -> np.ndarray:
    """a_i = max(0, 1 - slope·i)  (monotonically decreasing)."""
    a = np.maximum(0.0, 1.0 - slope * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_truncated_exponential(n: int, lam: float, cutoff_frac: float) -> np.ndarray:
    """Exponential decay cut off after a fraction of the support."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    cut = int(cutoff_frac * n)
    a[cut:] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Baseline exploration – keep the best `max_keep` vectors
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 12):
    """Explore a large set of parametric families and random Dirichlet draws."""
    best = []                     # min‑heap of (score, vector) with size ≤ max_keep

    def maybe_insert(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            heapq.heappush(best, (-sc, vec))
        else:
            if sc < -best[0][0]:
                heapq.heapreplace(best, (-sc, vec))

    # 1) power‑law (extend p up to 5)
    for p in np.linspace(0.1, 5.0, 250):
        if time.time() >= deadline:
            break
        maybe_insert(generate_powerlaw(n, p))

    # 2) exponential (lam up to 0.2)
    for lam in np.linspace(0.0001, 0.2, 250):
        if time.time() >= deadline:
            break
        maybe_insert(generate_exponential(n, lam))

    # 3) beta (wide range, including very skewed shapes)
    al_vals = np.concatenate((np.linspace(0.05, 0.5, 12),
                              np.linspace(0.6, 5.0, 12)))
    be_vals = np.concatenate((np.linspace(0.5, 5.0, 12),
                              np.linspace(5.5, 12.0, 12)))
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            maybe_insert(generate_beta(n, alpha, beta))

    # 4) Gaussian (various widths)
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        maybe_insert(generate_gaussian(n, sigma))

    # 5) Linear decreasing (different slopes)
    for slope in np.linspace(0.0, 2.0 / (n - 1), 30):
        if time.time() >= deadline:
            break
        maybe_insert(generate_linear_decreasing(n, slope))

    # 6) Truncated exponential (different cut‑offs)
    for lam in np.linspace(0.001, 0.05, 20):
        for cut in np.linspace(0.2, 0.9, 8):
            if time.time() >= deadline:
                break
            maybe_insert(generate_truncated_exponential(n, lam, cut))

    # 7) Dirichlet draws (sorted descending to obtain monotone shapes)
    concs = [0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 800
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(np.full(n, conc))
            maybe_insert(a)
            # also try monotone version
            a_desc = np.sort(a)[::-1]
            maybe_insert(a_desc)

    # Return the vectors (sorted by score ascending)
    best_vectors = [vec for (_sc, vec) in sorted(best, key=lambda x: -x[0])]
    return best_vectors


# ----------------------------------------------------------------------
# Smooth optimisation (Adam on soft‑max parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.06,
                    init_tau_factor: float = 8.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9993) -> np.ndarray:
    """Adam on a soft‑max parametrisation with a gradually shrinking temperature."""
    a = a0.copy()
    n = a.shape[0]

    # initialise log‑variables (add tiny epsilon for numerical safety)
    eps = 1e-30
    x = np.log(a + eps)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature – proportional to the current max‑convolution
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution & smooth max weights -----
        conv = full_convolution(a)
        max_c = conv.max()

        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient wrt a (g_i = 2·∑ w_k·a_{k-i}) -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # ----- chain rule (soft‑max) -----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature annealing -----
        tau = max(min_tau, tau * tau_decay)

        # ----- occasional exact evaluation -----
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true (non‑smooth) objective
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, deadline: float,
                        init_step: float = 0.3,
                        step_decay: float = 0.9995) -> np.ndarray:
    """Minimise max(conv(a,a)) using sub‑gradient steps and simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    step = init_step
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        k = int(conv.argmax())                     # index of an active constraint

        # gradient of the active convolution entry:
        grad = np.zeros_like(a)
        if k < n:
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[:n - start][::-1]

        # step & projection
        a = a - step * grad
        a = project_simplex(a)

        # periodic evaluation
        if it % 150 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[subgrad] iter {it:6d}  score {val:.6f}")
                # if we made good progress, optionally increase the step a bit
                step = min(step * 1.05, init_step)

        step = max(step * step_decay, 1e-9)

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9992,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 5e-7) -> np.ndarray:
    """Metropolis walk with a slowly cooling temperature."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            # occasional jitter to shake the system
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = project_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted wall‑clock budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – 2048 balances expressivity and speed.
    # ------------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously saved solution
    # ------------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) Global baseline exploration (≈ 150 s)
    # ------------------------------------------------------------------
    baseline_deadline = min(deadline - 600.0, start_time + 150.0)
    if time.time() < baseline_deadline:
        print("[stage] baseline exploration")
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth optimisation with several restarts (≈ 600 s)
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    if best_seq is not None and time.time() < smooth_deadline:
        print("[stage] smooth optimisation")
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, smooth_deadline)
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # additional restarts
        while time.time() < smooth_deadline:
            remaining = smooth_deadline - time.time()
            if remaining < 30.0:
                break
            sub_deadline = time.time() + min(30.0, remaining)
            # random start – either Dirichlet or a perturbed best vector
            if np.random.rand() < 0.5:
                conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
                a_start = np.random.dirichlet(np.full(n, conc))
            else:
                # small perturbation of current best
                a_start = np.array(best_seq, dtype=float)
                jitter = np.random.randn(n) * 0.02
                a_start = a_start * np.exp(jitter)
                a_start = project_simplex(a_start)

            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected sub‑gradient descent (≈ 180 s)
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 110.0
    if best_seq is not None and time.time() < subgrad_deadline:
        print("[stage] sub‑gradient descent")
        a_start = np.array(best_seq, dtype=float)
        refined = subgradient_descent(a_start, subgrad_deadline)
        val = compute_score(refined)
        print(f"[sub‑grad] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # ------------------------------------------------------------------
    # 4) Final simulated‑annealing polish (short run)
    # ------------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        print("[stage] simulated annealing")
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # ------------------------------------------------------------------
    # 5) Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6) Official evaluation report
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
