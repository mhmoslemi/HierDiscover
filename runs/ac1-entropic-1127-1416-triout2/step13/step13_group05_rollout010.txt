def anneal(a, deadline, init_temp=0.01, final_temp=1e-6):
    best = a.copy()
    best_score = compute_score(best)
    cur = a.copy()
    cur_score = best_score
    t = 0
    while time.time() < deadline:
        # compute temperature schedule: linear in progress
        progress = (time.time() - start) / (deadline - start)
        temp = init_temp * (1 - progress) + final_temp * progress

        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # choose delta as random fraction of cur[i]
        delta = cur[i] * np.random.uniform(0, 0.1)  # maybe up to 10% of i
        # ensure delta <= cur[i]
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # no need to project if stays nonnegative
        # but we might have rounding errors leading to negativity due to delta +- ; Ensure clip
        cand = np.clip(cand, 0, None)
        cand = project_to_simplex(cand)  # ensures sum=1

        cand_score = compute_score(cand)
        delta_score = cand_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / temp):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        t += 1
        # occasionally jitter
        if t % 50000 == 0:
            jitter = np.random.randn(n) * 1e-4
            cur = np.clip(cur + jitter, 0, None)
            cur = project_to_simplex(cur)
            cur_score = compute_score(cur)
```

# ----------------------------------------------------------------------
# Helper utilities – all are pure functions, no closures.
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The objective is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors.
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^{p},   p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)  # n, n-1, …, 1
    a = idx ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    return a / a_sum


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}, 0<q<1."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Front plateau (length L) of constant height followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    return a / a_sum


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def stretched_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑λ·i),  λ>0."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s

# Random Dirichlet sampling.
# ----------------------------------------------------------------------
def random_dirichlet_vector(n: int, conc: float) -> np.ndarray:
    """Draw a Dirichlet vector with common concentration parameter `conc`."""
    if conc <= 0:
        raise ValueError("concentration must be positive")
    # Gamma(α,1) for each coordinate, then normalise.
    g = np.random.gamma(conc, 1.0, size=n)
    g_sum = g.sum()
    if g_sum == 0.0:
        return None
    return g / g_sum


# ----------------------------------------------------------------------
# Sub‑gradient polishing – additive version (classical).
# ----------------------------------------------------------------------
def subgradient_polish_additive(a0: np.ndarray, deadline: float,
                               step0: float = 0.20) -> np.ndarray:
    """Projected additive sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # indices where the max is (approximately) attained
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient: sum of shifted copies of a for each active index
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        # occasional tiny jitter to escape plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing – multiplicative (exponentiated‑gradient) variant.
# ----------------------------------------------------------------------
def subgradient_polish_multiplicative(a0: np.ndarray, deadline: float,
                                      step0: float = 0.15) -> np.ndarray:
    """Projected multiplicative (exponentiated) sub‑gradient descent."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # build the sub‑gradient (same as additive, but we will use it
        # inside an exponentiated update)
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += a[j[mask]]  # note: no factor 2 because the
                                    # exponentiated step uses symmetric update

        # step size decays like 1/√t
        step = step0 / math.sqrt(t)
        # multiplicative update; ensure positivity
        a = a * np.exp(-step * g)
        a = a / a.sum()   # projection onto simplex for the simplex geometry

        # jitter occasionally
        if t % 4000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a = np.clip(a + jitter, 0.0, None)
            a = a / a.sum()

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Annealed stochastic hill‑climb (fine‑grained local search).
# ----------------------------------------------------------------------
def annealed_hill_climb(a_start: np.ndarray, deadline: float,
                        init_temp: float = 0.02,
                        final_temp: float = 1e-6,
                        max_iter: int = 2_000_000) -> np.ndarray:
    """
    Greedy stochastic moves with a temperature schedule.
    Moves transfer a random fraction of mass from i to j.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    t = 0
    start_time = time.time()
    total_time = deadline - start_time

    while time.time() < deadline and t < max_iter:
        # linear temperature schedule
        progress = (time.time() - start_time) / total_time
        temp = init_temp * (1.0 - progress) + final_temp * progress

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # transfer a random fraction (up to 20 %) of the source mass
        frac = np.random.rand() * 0.20
        delta = frac * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure non‑negativity (tiny rounding errors may appear)
        cand = np.clip(cand, 0.0, None)
        cand = cand / cand.sum()          # projection onto simplex

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
        else:
            # Metropolis acceptance
            if np.random.rand() < math.exp(-(M_cand - best_M) / max(temp, 1e-12)):
                a = cand

        t += 1

        # occasional jitter to re‑seed the walk if we stagnate
        if t % 250_000 == 0:
            jitter = np.random.randn(n) * 1e-4
            a = np.clip(best_a + jitter, 0.0, None)
            a = a / a.sum()
            best_M = full_convolution(a).max()
            best_a = a.copy()

    return best_a

    # ------------------------------------------------------------------

        val = evaluate_sequence(fallback)
        print(f"[fallback] official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1) Exploratory parametric sweep + random draws.
    # ------------------------------------------------------------------
    n = 2048

            val_internal = compute_score(vec)
            val_official = evaluate_sequence(vec.tolist())

            print(f"[exploratory] official score = {val_official:.6f}")
            if val_official < best_val:

                # keep the numpy version around for polishing
                best_vec = vec
            else:
                best_vec = np.array(best_seq, dtype=float) if best_seq is not None else None
        else:
            best_vec = np.array(best_seq, dtype=float) if best_seq is not None else None
    else:
        best_vec = np.array(best_seq, dtype=float) if best_seq is not None else None

    # ------------------------------------------------------------------
    # 2) Two complementary polishing phases.
    # ------------------------------------------------------------------
    polish_deadline = deadline - 120.0          # keep ~2 min for final anneal
    if best_vec is not None and time.time() < polish_deadline:
        # a) additive sub‑gradient descent
        print("[polish] additive sub‑gradient")
        add_dead = min(polish_deadline, time.time() + 80.0)
        a_add = subgradient_polish_additive(best_vec, add_dead, step0=0.20)
        val_add = evaluate_sequence(a_add.tolist())
        print(f"[polish] additive result = {val_add:.6f}")
        if val_add < best_val:
            best_val, best_seq = val_add, a_add.tolist()
            best_vec = a_add

        # b) multiplicative (exponentiated) sub‑gradient descent
        if time.time() < polish_deadline:
            print("[polish] multiplicative sub‑gradient")
            mul_dead = deadline - 60.0
            a_mul = subgradient_polish_multiplicative(best_vec, mul_dead, step0=0.15)
            val_mul = evaluate_sequence(a_mul.tolist())
            print(f"[polish] multiplicative result = {val_mul:.6f}")
            if val_mul < best_val:
                best_val, best_seq = val_mul, a_mul.tolist()
                best_vec = a_mul

    # ------------------------------------------------------------------
    # 3) Final annealed hill‑climb (fine‑grained local search).
    # ------------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_vec is not None and time.time() < anneal_deadline:
        print("[anneal] starting annealed hill‑climb")
        a_final = annealed_hill_climb(best_vec, anneal_deadline,
                                      init_temp=0.02, final_temp=1e-7,
                                      max_iter=3_000_000)
        final_score = evaluate_sequence(a_final.tolist())
        print(f"[anneal] final official score = {final_score:.6f}")
        if final_score < best_val:
            best_val, best_seq = final_score, a_final.tolist()

    # ------------------------------------------------------------------
    # 4) Safeguard – if everything failed, fall back to a decent reverse‑power‑law.
    # ------------------------------------------------------------------
    if best_seq is None:
        print("[fallback] constructing a generic reverse‑power‑law")
        fallback_vec = reverse_power_law(n, 1.9)   # an empirically good exponent
        best_seq = fallback_vec.tolist()
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5) Final report.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Best reported score = {final_score:.6f}")
    return best_seq
