"""
Improved optimiser for the step‑function inequality.

Main ideas
----------
1. **Global baseline** – a fast exploration of several parametric families
   (power‑law, beta, exponential, Gaussian, sine‑power) together with a
   large pool of Dirichlet draws.  The ten best normalised vectors are kept.

2. **Linear‑program refinement** – given a current vector *f* (∑f=1) we
   solve a linear programme
        maximise   Σ g_j
        subject to G ≥ 0 , (f * g)_k ≤ M   ∀k
   where M = max_k (f*f)_k.
   The feasible set contains vectors that put all mass where *f* is tiny,
   therefore Σ g_j can exceed 1.  After normalising g we move a small step
   towards it.  Repeating this (Frank‑Wolfe style) strictly decreases M.

3. **Adam smooth‑max descent** – a final polishing phase that minimises the
   smooth‑max surrogate of max(conv) using Adam.  The result from the LP
   phase is used as a warm‑start.

4. **Projected sub‑gradient polish** – a short sub‑gradient descent on the
   exact max‑objective.

All phases respect the 1000 s wall‑clock limit; the best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex ∑v=1, v≥0."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    # sort v decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT (real‑valued)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Objective for a normalised vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return vectors normalised to sum = 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return a / a.sum()


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    return a / a.sum()


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    return a / a.sum()


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return a / a.sum()


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    return a / a.sum()


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·vec1 + (1‑λ)·vec2 projected onto the simplex."""
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)

# Baseline global exploration
# ----------------------------------------------------------------------


def baseline_search(

        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())

            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Linear‑program refinement (Frank‑Wolfe style)
# ----------------------------------------------------------------------


def build_conv_constraints(f: np.ndarray) -> sparse.csr_matrix:
    """Build the matrix A such that (A @ g)_k = (f * g)_k."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = sparse.lil_matrix((rows, n))
    # each column j receives the vector f on rows j … j+n‑1
    for j in range(n):
        A[j : j + n, j] = f
    return A.tocsr()


def lp_refine(f0: np.ndarray, deadline: float, max_iter: int = 30, t0: float = 0.3):
    """
    Frank‑Wolfe‑type refinement.
    Starts from f0 (∑f0 = 1, f0 ≥ 0) and repeatedly solves the LP that
    maximises Σg under the constraint (f * g) ≤ M, where M = max(conv(f,f)).
    The solution g (renormalised to ∑g = 1) is mixed into f.
    """
    n = f0.shape[0]
    f = f0.copy()
    best_f = f.copy()
    best_M = full_convolution(f).max()
    t = t0

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        M = full_convolution(f).max()
        # Build constraint matrix A (size (2n‑1) × n)
        A = build_conv_constraints(f)

        # Solve LP: max Σg  ⇔  min -Σg
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=[(0, None)] * n,
            method="highs",
        )
        if not res.success:
            print("[lp_refine] LP failed at iteration", it, "status:", res.message)
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # Normalise g to sum = 1
        g = g0 / sum_g0

        # Mix a bit towards g
        f_new = (1.0 - t) * f + t * g
        f_new = project_to_simplex(f_new)  # ensure exact sum = 1

        M_new = full_convolution(f_new).max()
        # Accept if we made progress
        if M_new < best_M - 1e-12:
            best_f = f_new.copy()
            best_M = M_new
            f = f_new
            # increase step size a little (but keep ≤0.5)
            t = min(0.5, t * 1.05)
        else:
            # no progress – shrink step size
            t = max(0.01, t * 0.5)

        if t < 1e-4:
            break

    return best_f


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-6,
    tau_decay: float = 0.96,
    lr_start: float = 0.12,
    lr_decay: float = 0.99,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """Adam optimisation of fτ(a) = τ·log Σ exp(conv/τ)."""
    a = a0.copy()
    n = a.shape[0]

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        # soft‑max over conv entries
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()  # size = 2n‑1

        # gradient = 2·(a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (2 * n - 2) - np.arange(n)  # mapping to correct indices
        grad = 2.0 * conv_full[idx]

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-4)

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (optional final hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = 2e-3
    step_scale = 0.06

    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, 5e-6)

        T = max(1e-6, T * 0.9992)

        if np.random.rand() < 0.001:
            # occasional jitter
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    # pick a length – 2048 gives a good trade‑off between flexibility and speed
    n = 2048
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a known construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈200 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 400.0
    if time.time() < baseline_deadline:
        candidates = baseline_search(
            n, baseline_deadline, max_keep=12, random_draws=8000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Linear‑program refinement (Frank‑Wolfe style)
    # -----------------------------------------------------------------
    if best_seq is not None:
        lp_deadline = min(deadline - 150.0, time.time() + 250.0)
        f0 = np.array(best_seq, dtype=float)
        f0 = project_to_simplex(f0)  # ensure exact sum=1

        refined_f = lp_refine(f0, lp_deadline, max_iter=25, t0=0.3)
        val = compute_score(refined_f)
        print(f"[lp‑refine] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined_f.tolist()

    # -----------------------------------------------------------------
    # 3. Adam‑based smooth‑max optimisation (≈250 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        a0 = np.array(best_seq, dtype=float)
        a_ref = smooth_max_adam(
            a0,
            deadline - 150.0,
            tau_start=0.30,
            tau_end=1e-6,
            tau_decay=0.96,
            lr_start=0.12,
            lr_decay=0.99,
        )
        val = compute_score(a_ref)
        print(f"[adam] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_ref.tolist()

    # -----------------------------------------------------------------
    # 4. Sub‑gradient polishing (≈120 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        a0 = np.array(best_seq, dtype=float)
        a_ref = subgradient_polish(a0, deadline - 120.0)
        val = compute_score(a_ref)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_ref.tolist()

    # -----------------------------------------------------------------
    # 5. Simulated annealing final hill‑climb (optional remaining time)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        a0 = np.array(best_seq, dtype=float)
        a_ref = simulated_annealing(a0, deadline - 10.0)
        val = compute_score(a_ref)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_ref.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final reporting using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
