"""
Non‑negative step‑function optimiser.

Algorithm
---------
1. Choose a modest dimension (n = 256) – short vectors keep the
   2·n·max(conv) objective low and allow many cheap optimisation steps.
2. Initialise a pool of simple candidates (uniform, power‑law, beta,
   Dirichlet draws).  Keep the best one as the starting point.
3. Iteratively improve the current vector *f* by solving a linear programme:
      maximise    Σ g_j
      subject to  (f * g)[k] ≤ M   for all k,
                  g_j ≥ 0,
   where M = max(f * f).  The LP is linear because *f* is fixed.
   The solution *g* is normalised to Σ g_j = 1 and blended with *f*
   (f ← (1‑t)·f + t·g) with a modest step size *t*.
   Each LP solve is cheap (≈ 0.5 ms for n = 256) so thousands of
   iterations fit easily into the time budget.
4. Periodically evaluate the objective 2·n·max(conv) (the “score”)
   and keep the best vector seen.
5. After the LP phase, run a short Adam‑based smooth‑max descent and a
   projected sub‑gradient polish to fine‑tune the solution.
6. Return the best sequence found as a plain list of floats.

The code respects the 1000 s wall‑clock limit, prints progress
information and works with the supplied ``evaluate_sequence`` routine.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    return np.convolve(a, a)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for the initial pool)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·vec1 + (1‑λ)·vec2, projected onto the simplex."""
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Linear‑programming step (core of the optimisation)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that (f * g)[k] = (A g)[k].
    Entry A[k, i] = f[k‑i] if 0 ≤ k‑i < n else 0.
    """
    n = f.shape[0]
    k = np.arange(2 * n - 1)[:, None]          # (2n‑1, 1)
    i = np.arange(n)[None, :]                  # (1, n)
    idx = k - i                                 # (2n‑1, n)
    valid = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n), dtype=float)
    A[valid] = f[idx[valid]]
    return A


def lp_improve(f: np.ndarray, t: float) -> np.ndarray:
    """
    One LP‑based improvement step.
    *f* is the current vector (sum = 1).  Returns the blended vector.
    """
    n = f.shape[0]
    conv_f = full_convolution(f)
    M = conv_f.max()
    A = build_conv_matrix(f)

    # maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=float)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # LP failed – fall back to a tiny random perturbation
        g = np.random.rand(n)
    else:
        g = res.x

    s = g.sum()
    if s <= 0:
        # degenerate solution – replace by a small random vector
        g = np.random.rand(n)
        s = g.sum()
    g = g / s                     # normalise to sum = 1

    # convex blend
    new_f = (1.0 - t) * f + t * g
    return new_f

# Smooth‑max Adam optimiser (fine‑grained descent)
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.20,
    tau_end: float = 1e-6,
    tau_decay: float = 0.94,
    lr_start: float = 0.10,
    lr_decay: float = 0.98,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate:
        fτ(a) = τ·log Σ_k exp(conv_k(a)/τ)
    Returns the best vector (projected on the simplex) seen before the
    deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t_iter = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # temperature schedule
        if t_iter % 150 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # smooth‑max weights
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()          # size m

        # gradient of smooth‑max: 2·(a ⋆ w)
        rev_w = w[::-1]
        grad_full = np.convolve(a, rev_w)      # length n+m‑1
        idx = (m - 1) - np.arange(n)           # map to needed slice
        grad = 2.0 * grad_full[idx]

        # Adam update
        t_iter += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t_iter)
        v_hat = v_t / (1.0 - beta2 ** t_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # learning‑rate anneal
        if t_iter % 400 == 0:
            lr = max(lr * lr_decay, 1e-4)

        # occasional jitter
        if t_iter % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t_iter))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (exact max objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.30) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t_iter = 0
    while time.time() < deadline:
        t_iter += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient (sum of contributions from all active shifts)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t_iter)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t_iter % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t_iter))
            a += jitter
            a = project_to_simplex(a)

    return best_a

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluation call
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # 0. Try the previously discovered construction (if available)
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Build a small initial pool of candidates (n = 256)
    # -----------------------------------------------------------------
    n = 256
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # a few hand‑crafted families
    init_candidates.append(generate_powerlaw(n, 0.5))
    init_candidates.append(generate_powerlaw(n, 2.0))
    init_candidates.append(generate_beta(n, 0.4, 0.8))
    init_candidates.append(generate_beta(n, 1.0, 1.0))
    init_candidates.append(generate_gaussian(n, sigma=0.15 * n))
    init_candidates.append(generate_sine_power(n, 1.5))

    # random Dirichlet draws (various concentrations)
    for conc in [0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(4):
            init_candidates.append(np.random.dirichlet([conc] * n))

    # evaluate the pool
    def update_best(vec: np.ndarray):
        nonlocal best_seq, best_val
        sc = compute_score(vec)
        if sc < best_val:
            best_val = sc
            best_seq = vec.tolist()
            print(f"[new best] score = {sc:.6f}")

    for vec in init_candidates:
        vec = project_to_simplex(vec)
        update_best(vec)

    # -----------------------------------------------------------------
    # 2. LP‑based iterative improvement (major part of the time budget)
    # -----------------------------------------------------------------
    # parameters for the LP loop
    blend_t = 0.30          # initial blend weight
    min_blend = 0.05
    outer_start = time.time()
    while time.time() < deadline - 30.0:
        # Choose a starting point – mostly the current best, occasionally random
        if best_seq is not None and np.random.rand() < 0.7:
            f = np.array(best_seq, dtype=float)
        else:
            f = np.random.dirichlet(np.ones(n))
        f = project_to_simplex(f)

        # Perform a short burst of LP steps
        inner_iters = 0
        t_local = blend_t
        while time.time() < deadline - 30.0 and inner_iters < 40:
            f = lp_improve(f, t_local)
            f = project_to_simplex(f)
            update_best(f)
            # slowly reduce the blend factor to avoid too aggressive jumps
            t_local = max(min_blend, t_local * 0.97)
            inner_iters += 1
        # reset blend weight for the next outer round
        blend_t = 0.30

        # occasional safety break if time is tight
        if time.time() - outer_start > budget_s * 0.85:
            break

    # -----------------------------------------------------------------
    # 3. Fine‑tuning with Adam (smooth‑max) and sub‑gradient polish
    # -----------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 40.0:
            # Adam phase (about half of the remaining time)
            adam_deadline = time.time() + remaining * 0.55
            a0 = np.array(best_seq, dtype=float)
            refined = smooth_max_adam(a0, adam_deadline)
            update_best(refined)

        remaining = deadline - time.time()
        if remaining > 10.0:
            # Sub‑gradient polishing (rest of the time)
            polish_deadline = time.time() + remaining * 0.95
            a0 = np.array(best_seq, dtype=float)
            refined = subgradient_polish(a0, polish_deadline)
            update_best(refined)

    # -----------------------------------------------------------------
    # 4. Fallback to a uniform vector if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final reported score (using the official evaluator)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
