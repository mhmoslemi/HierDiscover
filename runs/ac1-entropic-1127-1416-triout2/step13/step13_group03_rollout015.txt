"""
Non‑negative step‑function optimiser (≈ 1000 s budget).

Approach
--------
1. **Explore a very diverse set of candidates**  
   – classical families (power‑law, exponential, beta, Gaussian)  
   – classic window functions (Hann, Hamming, Blackman, Kaiser)

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    total = a.sum()
    return a / total


def generate_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    total = a.sum()
    return a / total


def generate_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    total = a.sum()
    return a / total


def generate_kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window (beta >= 0)."""
    i = np.arange(n, dtype=float)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = i0(beta * np.sqrt(1.0 - t ** 2)) / i0(beta)
    total = a.sum()
    return a / total


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·vec1 + (1‑λ)·vec2, projected onto the simplex."""
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Global exploration
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 6,
    random_draws: int = 4000,
) -> list[np.ndarray]:
    """Generate a wide variety of candidates and keep the best ``max_keep``."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1. Classical families -------------------------------------------------
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    for lam in np.linspace(0.0001, 0.5, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 2. Window functions ---------------------------------------------------
    for win in (generate_hann, generate_hamming, generate_blackman):
        if time.time() >= deadline:
            break
        try_candidate(win(n))

    # Kaiser with a broad β range
    for beta in np.linspace(0.0, 15.0, 31):
        if time.time() >= deadline:
            break
        try_candidate(generate_kaiser(n, beta))

    # 3. Beta family (grid + random) ---------------------------------------
    al_vals = np.linspace(0.02, 2.0, 31)
    be_vals = np.linspace(0.02, 2.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    for _ in range(300):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)   # 0.01 … 2
        beta = 10 ** np.random.uniform(-2, 0.3)
        try_candidate(generate_beta(n, alpha, beta))

    # 4. Gaussian sweep -----------------------------------------------------
    for sigma in np.linspace(0.1 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # 5. Sine‑power ---------------------------------------------------------
    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # 6. Random Dirichlet draws (wide range of concentrations) -------------
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    # 7. Random mixtures of two families ------------------------------------
    for _ in range(250):
        if time.time() >= deadline:
            break
        # pick two random generators
        gen1 = np.random.choice(
            [
                generate_powerlaw,
                generate_exponential,
                generate_hann,
                generate_hamming,
                generate_blackman,
                lambda n, _: np.full(n, 1.0 / n),
            ]
        )
        gen2 = np.random.choice(
            [
                generate_powerlaw,
                generate_exponential,
                generate_hann,
                generate_hamming,
                generate_blackman,
                lambda n, _: np.full(n, 1.0 / n),
            ]
        )
        # random parameters for each
        if gen1 in (generate_powerlaw, generate_exponential):
            p1 = np.random.uniform(0.1, 5.0)
            v1 = gen1(n, p1)
        elif gen1 == generate_hann or gen1 == generate_hamming or gen1 == generate_blackman:
            v1 = gen1(n)
        else:
            v1 = np.full(n, 1.0 / n)

        if gen2 in (generate_powerlaw, generate_exponential):
            p2 = np.random.uniform(0.1, 5.0)
            v2 = gen2(n, p2)
        elif gen2 == generate_hann or gen2 == generate_hamming or gen2 == generate_blackman:
            v2 = gen2(n)
        else:
            v2 = np.full(n, 1.0 / n)

        lam = np.random.rand()
        try_candidate(mixture(v1, v2, lam))

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Local optimiser using the *exact* max‑convolution sub‑gradient
# ----------------------------------------------------------------------


def max_subgrad_adam(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser that uses the sub‑gradient of `max(conv(a,a))`.
    Returns the best vector (projected onto the simplex) seen before the
    deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam moment buffers
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    beta1, beta2 = 0.90, 0.999
    eps = 1e-8
    lr0 = 0.15
    lr_decay = 0.9995

    while time.time() < deadline:
        # convolution and active shift(s)
        conv = full_convolution(a)
        max_val = conv.max()
        # indices whose value is within a tiny tolerance of the max
        active = np.where(conv >= max_val - 1e-12)[0]
        k = np.random.choice(active)

        # sub‑gradient of the max‑conv at shift k:
        #   ∇_i = 2·a_{k‑i}  (if 0 ≤ k‑i < n, else 0)
        idx = np.arange(n)
        mask = (k - idx >= 0) & (k - idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[k - idx[mask]]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)

        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        lr = max(lr0 * (lr_decay ** t), 1e-5)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # bookkeeping
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape flat plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (lr0 * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (as in the previous implementation)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = 2e-3
    final_T = 1e-6
    cooling = 0.9994
    step_scale = 0.06
    min_step = 5e-6

    it = 0
    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, min_step)

        T = max(final_T, T * cooling)

        if it % 7000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # -----------------------------------------------------------------
    # 0. Global parameters
    # -----------------------------------------------------------------
    n = 2048                     # a modest size (fast FFT, still good lower bound)
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0a. Fallback – previously discovered construction (if available)
    # -----------------------------------------------------------------
    try:
        # `height_sequence_1` may be defined in the global namespace by the harness
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 300 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 400.0
    if time.time() < baseline_deadline:
        print("[baseline] start global exploration")
        candidates = baseline_search(
            n, baseline_deadline, max_keep=8, random_draws=8000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Local Adam‑type refinement using exact max‑subgradient
    # -----------------------------------------------------------------
    if best_seq is not None:
        print("[local] start max‑subgradient Adam")
        # allocate a chunk of time for the local phase
        local_deadline = min(deadline - 150.0,
                             start_time + 250.0 + (deadline - start_time) * 0.35)

        # initialise a small pool of start vectors
        start_vectors = []
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        start_vectors.append(best_vec)

        # uniform start (always a useful baseline)
        start_vectors.append(np.full(n, 1.0 / n))

        # a few random Dirichlet starts
        for _ in range(4):
            start_vectors.append(project_to_simplex(np.random.rand(n)))

        refined_best = best_vec.copy()
        refined_val = best_val

        for idx, start_vec in enumerate(start_vectors):
            if time.time() >= local_deadline:
                break
            phase_deadline = local_deadline - (time.time() - start_time) - 0.5
            if phase_deadline <= time.time():
                break
            a_ref = max_subgrad_adam(start_vec, phase_deadline)
            val = compute_score(a_ref)
            print(f"[local‑adam {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_ref.copy()

        best_val, best_seq = refined_val, refined_best.tolist()
        print(f"[local] best after max‑subgrad Adam = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Simulated‑annealing polishing (≈ 120 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = simulated_annealing(a0, polish_deadline)
        val = compute_score(a_refined)
        print(f"[anneal‑polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – fall back to uniform if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")

    return best_seq
