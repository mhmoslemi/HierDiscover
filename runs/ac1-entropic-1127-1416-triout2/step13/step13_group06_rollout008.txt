3. **Random masks** – generate many random “sparse‑equal‑mass’’ masks
   (choose a random subset of indices and place equal weight on them);
   this creates irregular autocorrelations that sometimes beat the
   smooth families.
4. **Soft‑max mirror descent** – minimise a smooth approximation of the
   max‑convolution:
      L(a)=ε·log∑_k exp(conv(a,a)_k/ε)
   The gradient is obtained by a correlation of the soft‑max weights
   with ``a`` (FFT‑based, O(n log n)).  We perform a multiplicative
   (exponentiated‑gradient) update, renormalising to the simplex after
   each step.  The temperature ε is decreased geometrically, which
   gradually focuses the optimisation on the true maximum.
5. **Final simulated‑annealing hill‑climb** – a short SA phase moves
   small amounts of mass between entries to escape any remaining
   local minima.
6. Return the best sequence (plain ``list[float]``) after a final
   call to ``evaluate_sequence`` (which performs the required security
   checks).

All steps respect the supplied wall‑clock budget (default 1000 s) and
use only NumPy and the provided ``evaluate_sequence`` routine.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, scoring, simplex projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (n‑½‑i)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred on the middle of the interval."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """Symmetric triangular bump."""
    half_width = int(max(1, width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return w / w.sum()


def exp_decay_family(n: int, gamma: float, left: bool = True) -> np.ndarray:
    """Exponential decay from one side."""
    i = np.arange(n, dtype=np.float64)
    if left:
        w = np.exp(-gamma * i)
    else:
        w = np.exp(-gamma * (n - 1 - i))
    return w / w.sum()


def mixed_uniform_gaussian_family(n: int, lam: float, sigma: float) -> np.ndarray:
    """Convex mixture of uniform and centred Gaussian."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    gaussian = gaussian_center_family(n, sigma)
    w = lam * uniform + (1.0 - lam) * gaussian
    return w / w.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / s
    return w


def random_mask_family(n: int, m: int) -> np.ndarray:
    """Equal mass on a random subset of size m."""
    indices = np.random.choice(n, size=m, replace=False)
    w = np.zeros(n, dtype=np.float64)
    w[indices] = 1.0 / m
    return w

# Optimisers
# ----------------------------------------------------------------------


def smoothed_mirror_descent(start_a: np.ndarray,
                            deadline: float,
                            init_eps: float = 0.5,
                            eps_decay: float = 0.7,
                            min_eps: float = 1e-5,
                            eta0: float = 0.3,
                            eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Minimise a smoothed max‑convolution using exponentiated‑gradient
    updates.  ``eps`` is the temperature of the soft‑max; it is reduced
    geometrically.  The update is a multiplicative step followed by a
    simplex renormalisation.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    eps = init_eps
    n = a.size

    while time.time() < deadline and eps > min_eps:
        # inner loop for the current temperature
        inner = 0
        while time.time() < deadline and inner < 2000:
            # ---- compute soft‑max weights ---------------------------------
            conv = convolution_fft(a, a)
            maxc = conv.max()
            # to avoid overflow and underflow
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

            # ---- gradient = 2 * (w ⋆ a) ---------------------------------
            L = next_pow2(len(w) + n - 1)
            W = np.fft.rfft(w, L)
            A = np.fft.rfft(a, L)
            C = np.fft.irfft(W * np.conj(A), L)
            g = 2.0 * C[n - 1: n - 1 + n]          # length‑n gradient

            # ---- exponentiated‑gradient step -------------------------------
            t += 1
            eta = eta0 / math.sqrt(t)
            # Prevent overflow in exp: clip gradient magnitude
            g_clip = np.clip(g, -50.0, 50.0)
            a = a * np.exp(-eta * g_clip)
            # renormalise (KL‑projection is simple scaling)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum

            inner += 1
            if t % eval_interval == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score, best_a = sc, a.copy()
                    print(f"[{time.time():.1f}] smoothed‑MD improved to {best_score:.6f}")

        # reduce temperature
        eps = max(eps * eps_decay, min_eps)

    # final evaluation
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score, best_a = final_sc, a.copy()
        print(f"[{time.time():.1f}] smoothed‑MD final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 8_000_000,
                  temp_factor: float = 1.1e-3,
                  step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb on the simplex.
    Moves are either swaps or a fractional transfer of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature proportional to the current score
    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                         # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` under the given time budget.  Returns the best
    sequence found as a plain list of floats."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Fallback to previous best (if available)
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Working discretisation
    # -----------------------------------------------------------
    n = 4096   # a compromise between resolution and speed

    # -----------------------------------------------------------
    # 2. Coarse scan of analytic families
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] scanning analytic families (n={n})")
    families = [
        ("center_power", lambda a: symmetric_center_power_family(a, 0.0)),
        ("gaussian", lambda a: gaussian_center_family(a, 0.0)),
        ("triangular", lambda a: triangular_center_family(a, 0.5)),
        ("left_power", lambda a: left_heavy_power_family(a, 0.0)),
        ("right_power", lambda a: right_heavy_power_family(a, 0.0)),
        ("product", lambda a: symmetric_product_family(a, 0.0)),
        ("beta", lambda a: beta_family(a, 1.0, 1.0)),
    ]

    # Helper to evaluate and possibly update the current best
    def try_candidate(vec: np.ndarray, descr: str):
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] new best ({descr}) = {best_score:.6f}")

    # centre‑heavy power
    for alpha in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_center_power_family(n, alpha)
        try_candidate(vec, f"center‑α={alpha:.2f}")

    # Gaussian
    for sigma in np.linspace(0.1 * n, 0.4 * n, 30):
        if time.time() >= deadline - 30.0:
            break
        vec = gaussian_center_family(n, sigma)
        try_candidate(vec, f"gauss‑σ={sigma:.0f}")

    # Triangular
    for width in np.linspace(0.1, 0.9, 33):
        if time.time() >= deadline - 30.0:
            break
        vec = triangular_center_family(n, width)
        try_candidate(vec, f"tri‑w={width:.2f}")

    # left‑heavy power
    for beta in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        try_candidate(vec, f"left‑β={beta:.2f}")
        vec_r = right_heavy_power_family(n, beta)
        try_candidate(vec_r, f"right‑β={beta:.2f}")

    # product (U‑shaped)
    for alpha in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        try_candidate(vec, f"U‑α={alpha:.2f}")

    # beta family (randomised 2‑parameter draws)
    for _ in range(80):
        if time.time() >= deadline - 30.0:
            break
        p = np.random.uniform(0.2, 3.0)
        q = np.random.uniform(0.2, 3.0)
        vec = beta_family(n, p, q)
        try_candidate(vec, f"beta p={p:.2f},q={q:.2f}")

    # -----------------------------------------------------------
    # 3. Random dense search (including random masks)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] random dense search")
    trials = 0
    max_trials = 12000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(9)
        if choice == 0:                                   # centre‑heavy
            alpha = np.random.uniform(0.0, 15.0)
            vec = symmetric_center_power_family(n, alpha)
        elif choice == 1:                                 # Gaussian
            sigma = np.random.uniform(0.1 * n, 0.4 * n)
            vec = gaussian_center_family(n, sigma)
        elif choice == 2:                                 # triangular
            width = np.random.uniform(0.1, 0.9)
            vec = triangular_center_family(n, width)
        elif choice == 3:                                 # left‑heavy
            beta = np.random.uniform(0.0, 15.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 4:                                 # right‑heavy
            beta = np.random.uniform(0.0, 15.0)
            vec = right_heavy_power_family(n, beta)
        elif choice == 5:                                 # product
            alpha = np.random.uniform(0.0, 15.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 6:                                 # exponential decay left
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_decay_family(n, gamma, left=True)
        elif choice == 7:                                 # exponential decay right
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_decay_family(n, gamma, left=False)
        else:                                             # random mask
            m = np.random.randint(int(0.2 * n), n)        # at least 20 % of entries
            vec = random_mask_family(n, m)

        # occasional random shift (zero‑padding) to break symmetry
        if np.random.rand() < 0.15:
            d = np.random.randint(0, n // 40 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Soft‑max mirror descent refinement
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] soft‑max mirror descent start")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = smoothed_mirror_descent(
            start_a=a_start,
            deadline=deadline - 70.0,
            init_eps=0.6,
            eps_decay=0.65,
            min_eps=5e-6,
            eta0=0.4,
            eval_interval=150,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after MD best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Final simulated‑annealing hill‑climb (light)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=6_000_000,
            temp_factor=1.0e-3,
            step_scale_init=0.22,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Fallback to a safe uniform vector if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
