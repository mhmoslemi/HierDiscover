def smooth_max_optim(a0, deadline, gamma_init=5.0, gamma_max=80.0, gamma_factor=1.001, lr=0.2, max_iters=200000):
    a = a0.copy()
    n = a.size
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    gamma = gamma_init
    best_score = compute_score(a)
    best_a = a.copy()
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        S = a.sum()
        if S <= 0:
            a = np.full_like(a, 1.0/n)
            S = 1.0
        conv = convolution_fft(a, a)
        # log-sum-exp for stable softmax
        scaled = gamma * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()
        A = np.dot(w, conv)  # softmax weighted conv
        # cross correlation
        # compute via np.correlate as valid (length n)
        cw = np.correlate(w, a, mode='valid')
        # gradient
        grad = (4.0 * n / (S**2)) * cw - (4.0 * n * A / (S**3))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project to simplex
        a = project_to_simplex(a)
        # optionally evaluate periodic exact score
        if t % 2000 == 0 or time.time() >= deadline - 5:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] iter {t} new best smooth {sc:.6f}")
        # increase gamma gradually
        gamma = min(gamma * gamma_factor, gamma_max)
    # final evaluation
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score, best_a = final_sc, a.copy()
    return best_a, best_score

def smooth_optimize(start_a, deadline):
    n = start_a.size
    a = start_a.copy()
    # ensure non-neg & normalized sum to 1
    a = np.maximum(a, 0)
    a = a / a.sum()
    # initial parameters
    gamma = 5.0
    gamma_max = 80.0
    gamma_mul = 1.001
    lr = 0.2
    beta1=0.9; beta2=0.999; eps=1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t=0
    while time.time() < deadline and t < 300_000:
        t+=1
        S = a.sum()
        conv = convolution_fft(a, a)
        scaled = gamma * conv
        # stable softmax
        max_s = scaled.max()
        exp_scaled = np.exp(scaled - max_s)
        w = exp_scaled / exp_scaled.sum()
        # approximate max (soft)
        A = np.dot(w, conv)  # Or compute as (max_s + np.log(np.mean(exp_scaled))) / gamma? This is same.
        # cross-correlation cw = np.correlate(w, a, mode='valid')
        # Use FFT:
        a_rev = a[::-1]
        cw_full = convolution_fft(w, a_rev)
        start = n - 1
        cw = cw_full[start:start + n]
        # gradient
        grad = (4.0 * n / (S**2)) * cw - (4.0 * n * A / (S**3))
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection
        a = project_to_simplex(a)
        # update gamma
        gamma = min(gamma * gamma_mul, gamma_max)
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] smooth iter {t} new best {sc:.6f}")
    # final evaluate
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score, best_a = final_sc, a.copy()
    return best_a, best_score
```

"""
Propose a step‑function with a lower evaluation score.

Algorithm
---------
* Work with a relatively fine discretisation (n = 8192).
* Generate a handful of diverse starting shapes
  (uniform, previous best, a few analytic families and random Dirichlet draws).
* For each start run a *smooth‑max* optimiser:
    - Replace the hard max in the objective by a log‑sum‑exp
      (soft‑max) with a gradually increasing temperature γ.
    - Use the exact gradient of the smooth surrogate and an Adam‑style
      update, projected onto the probability simplex after every step.
    - The gradient needs the cross‑correlation   w ⋆ a,
      which we compute efficiently by FFT.
* Keep the best shape found by the smooth optimiser.
* Finish with a short simulated‑annealing hill‑climb to polish the result.
* Return the best sequence as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple, Optional

# --------------------------------------------------------------
# Helper utilities (FFT‑based convolutions, projections, etc.)
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# --------------------------------------------------------------
# Analytic families (used for initial seeds)
# --------------------------------------------------------------


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} (mass decays from the left)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Mirror of left‑heavy."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred at the middle."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """Symmetric triangular bump."""
    half_width = max(1, int(width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return w / w.sum()


def random_dirichlet(n: int, conc: float) -> np.ndarray:
    """Draw a random point on the simplex (Gamma/Dirichlet)."""
    x = np.random.gamma(conc, 1.0, size=n)
    return normalize(x)


# --------------------------------------------------------------
# Smoothed‑max optimiser (Adam + simplex projection)
# --------------------------------------------------------------


def smooth_max_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 200_000,
                        gamma_init: float = 5.0,
                        gamma_max: float = 80.0,
                        gamma_factor: float = 1.001,
                        lr: float = 0.2) -> Tuple[np.ndarray, float]:
    """
    Minimise the original objective using a smooth surrogate.
    The surrogate replaces max(conv) by (1/γ)·log‑sum‑exp(γ·conv).
    Adam updates are performed and each iterate is projected onto the simplex.
    """
    n = start_a.size
    a = normalize(start_a.copy())
    best_a = a.copy()
    best_score = compute_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    gamma = gamma_init
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        S = a.sum()
        if S <= 0.0:
            a = np.full_like(a, 1.0 / n)
            S = 1.0

        # 1) a * a convolution
        conv = convolution_fft(a, a)                     # length 2n‑1
        # 2) soft‑max weights w_k ∝ exp(γ·conv_k)
        scaled = gamma * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)         # stabilise
        w = exp_scaled / exp_scaled.sum()
        # soft approximation of the max
        A = np.dot(w, conv)                               # = (1/γ)·log‑sum‑exp up to a constant

        # 3) cross‑correlation w ⋆ a (valid part, length n)
        a_rev = a[::-1]
        cw_full = convolution_fft(w, a_rev)              # length 3n‑2
        start = n - 1
        cw = cw_full[start:start + n]                     # (w ⋆ a)[i]

        # 4) gradient of the smooth surrogate
        grad = (4.0 * n / (S * S)) * cw - (4.0 * n * A / (S ** 3))

        # 5) Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # 6) project back onto the simplex
        a = project_to_simplex(a)

        # 7) increase the sharpness γ (approach the true max)
        gamma = min(gamma * gamma_factor, gamma_max)

        # occasional exact evaluation
        if t % 2000 == 0 or time.time() >= deadline - 5.0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] smooth‑opt iter {t} new best {best_score:.6f}")

    # final check
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score, best_a = final_sc, a.copy()
        print(f"[{time.time():.1f}] smooth‑opt final improvement {best_score:.6f}")

    return best_a, best_score


# --------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polishing)
# --------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000,
                  temp_factor: float = 1.2e-3,
                  step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """Projected SA on the exact objective."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
