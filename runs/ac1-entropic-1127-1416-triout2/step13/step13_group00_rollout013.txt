import time
import numpy as np
from scipy.optimize import linprog

def full_convolution(a):
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def build_conv_matrix(f):
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A

def lp_step(f, M):
    """Solve LP to maximize sum(g) subject to 0 <= g and f * g <= M."""
    n = f.shape[0]
    A = build_conv_matrix(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # linprog minimizes, so use -1 to maximize sum
    bounds = [(0, None)] * n
    # Use HiGHS for speed
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

def blend(f, g, t, sum_target):
    """Blend f with g and renormalize to sum_target."""
    new = (1 - t) * f + t * g
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s <= 0:
        return new
    new = new * (sum_target / s)
    return new

            def subgradient_polish(a0, local_deadline, step0=0.30):

                    if s <= 0:

"""
search for a non‑negative step‑height sequence that minimises
evaluate_sequence (i.e. minimises 2·n·max(conv(a,a))/(sum a)^2).

The core of the method is an alternating scheme proposed in the
reference description:
    * start from a normalised vector  sum(a)=sqrt(2·n)
    * compute M = max( a * a )
    * solve a linear programme   maximise 1ᵀg   s.t.   a*g ≤ M , g≥0
    * rescale the solution to the same L¹‑norm and blend a with it
      (convex combination).  The bound M strictly decreases.
The linear programme is solved with SciPy’s HiGHS simplex driver,
which is fast enough for n≈2000.

After a few LP steps a short projected‑Adam refinement and a
sub‑gradient polishing phase are applied, both of which respect the
normalisation constraint.  The best vector found is returned as a
plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT based convolution)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(full_convolution(a).max())


# ----------------------------------------------------------------------
# Build the matrix describing the (a * g) convolution constraints
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the dense matrix A (shape (2·n‑1, n)) such that
    (f * g)_k = sum_i A[k,i]·g_i .
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
# Linear‑programming step (maximise L¹‑norm of g under a convolution bound)
# ----------------------------------------------------------------------


def lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max 1ᵀg   s.t.   f * g ≤ M , g ≥ 0 .
    Returns the optimal vector g or None if the solver fails.
    """
    A = build_convolution_matrix(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)          # linprog minimises ⇒ -sum(g)
    bounds = [(0.0, None)] * f.shape[0]

    # HiGHS simplex driver is fast for dense problems of this size.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Simple convex combination update (with renormalisation)
# ----------------------------------------------------------------------


def blend(f: np.ndarray, g: np.ndarray, t: float, sum_target: float) -> np.ndarray:
    """
    Return (1‑t)·f + t·g, clip to non‑negative and renormalise
    so that the L¹‑norm equals ``sum_target``.
    """
    new = (1.0 - t) * f + t * g
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s > 0:
        new *= (sum_target / s)
    return new


# ----------------------------------------------------------------------
# Helper for the Adam optimiser (smooth approximation of max)
# ----------------------------------------------------------------------


def smooth_max(conv: np.ndarray, tau: float) -> tuple[float, np.ndarray]:
    """τ·log‑sum‑exp approximation of max(conv); also returns the soft‑max weights."""
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a*a)_k   w.r.t. a.
    Implements   g_i = 2·Σ_k w_k·a_{k‑i}.
    """
    n = a.shape[0]
    m = len(w)
    rev_w = w[::-1]
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]


# ----------------------------------------------------------------------
# Projection onto the L¹‑simplex with an arbitrary target sum
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray, sum_target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x≥0, 1ᵀx = sum_target}.
    """
    if np.all(v <= 0):
        # uniform distribution in the feasible set
        return np.full_like(v, sum_target / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - sum_target
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, sum_target / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray,
                  sum_target: float,
                  deadline: float,
                  tau_start: float = 0.5,
                  tau_end: float = 1e-7,
                  tau_decay: float = 0.90,
                  lr: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """
    Minimise the smooth‑max surrogate of max(conv) with projected Adam.
    The vector is kept on the simplex sum_target.
    """
    a = a0.copy()
    n = a.shape[0]
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = max_convolution(a)

    while time.time() < deadline:
        # anneal temperature
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a, sum_target)

        cur_val = max_convolution(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (true objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray,
                       sum_target: float,
                       deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact objective
    f(a)=max(conv(a,a)).  The iterate is kept on the simplex
    with total mass ``sum_target``.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = max_convolution(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (sum_target / s)

        cur_val = max_convolution(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  The algorithm consists of

    1.  An LP‑based descent (the scheme described in the reference).
    2.  A short projected Adam refinement on a smooth approximation.
    3.  Final sub‑gradient polishing.

    The best sequence found within the time budget is returned.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # small safety margin

    # ---- problem size -------------------------------------------------
    n = 2048
    sum_target = np.sqrt(2.0 * n)      # normalisation makes the objective = max(conv)

    # ---- initialise with a uniform vector -----------------------------
    f = np.full(n, sum_target / n, dtype=np.float64)

    M = max_convolution(f)
    best_f = f.copy()
    best_M = M
    print(f"[init] n={n}, sum_target={sum_target:.4f}, max conv = {M:.6f}")

    # ---- main LP improvement loop ------------------------------------
    it = 0
    while time.time() < deadline - 60.0:      # keep ~1 min for polishing
        it += 1
        g0 = lp_step(f, M)
        if g0 is None:
            print("[lp] linear programme failed – aborting LP loop")
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print("[lp] LP returned the zero vector – aborting")
            break

        # rescale to the required L¹‑norm
        g = g0 * (sum_target / sum_g0)

        # blending factor decays slowly
        if it < 10:
            t = 0.30
        elif it < 30:
            t = 0.15
        elif it < 80:
            t = 0.08
        else:
            t = 0.04

        f = blend(f, g, t, sum_target)

        M = max_convolution(f)
        if M < best_M:
            best_M, best_f = M, f.copy()
            print(f"[lp iter {it:03d}] improved max conv = {M:.6f}")
        else:
            if it % 10 == 0:
                print(f"[lp iter {it:03d}] current max conv = {M:.6f}")

        if M <= 1.503:                     # early stop if target bound reached
            print("[lp] target bound reached")
            break

    # ---- optional Adam refinement ------------------------------------
    if time.time() < deadline - 30.0:
        try:
            for restart in range(3):
                if time.time() >= deadline - 5.0:
                    break
                print(f"[adam] restart {restart}")
                a_opt = adam_optimize(best_f, sum_target, deadline)
                val = max_convolution(a_opt)
                print(f"[adam] candidate max conv = {val:.6f}")
                if val < best_M:
                    best_M, best_f = val, a_opt.copy()
        except Exception as e:
            print("[adam] optimisation failed:", e)

    # ---- final sub‑gradient polishing --------------------------------
    if time.time() < deadline - 5.0:
        try:
            a_polished = subgradient_polish(best_f, sum_target, deadline)
            val = max_convolution(a_polished)
            print(f"[polish] final max conv = {val:.6f}")
            if val < best_M:
                best_M, best_f = val, a_polished.copy()
        except Exception as e:
            print("[polish] failed:", e)

    # ---- report and return -------------------------------------------
    final_seq = best_f.tolist()
    final_score = evaluate_sequence(final_seq)   # official evaluator
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return final_seq
