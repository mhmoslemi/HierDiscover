"""
Non‑negative step‑function optimiser (≈1000 s).

The optimiser works in three stages:

1. **Broad random exploration** – a sweep of three parametric families
   (power‑law, exponential, beta) plus a massive pool of Dirichlet draws
   (many concentration values).  The ten best vectors (by the exact
   objective `2·n·max(conv)`) are kept.

2. **Smooth‑Adam polishing** – each retained candidate is refined by a
   mirror‑descent (soft‑max) optimiser that minimises the *smooth*
   approximation of the max‑convolution:
   ```
   F(a) = logsumexp(α·conv(a,a)) / α,
   ```
   where `α` is gradually increased from a small value (≈5) to a large
   one (≈200).  The gradient of `F` is obtained analytically via a
   cross‑correlation, so each iteration uses two cheap FFT‑based
   convolutions.  Adam with bias‑corrected moments is applied; the best
   vector seen before the time limit is retained.

3. **Simulated‑annealing final polish** – a short Metropolis walk
   (transfer a small amount of mass between two coordinates) with
   exponential cooling further escapes possible plateaus.

All stages respect the 1000 s wall‑clock budget; the routine returns the
best sequence found as a plain list of floats ready for the official
`evaluate_sequence` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape:
       a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration (baseline)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 10):
    """Extensive random and parametric search, keeping the ``max_keep``
    best vectors (as numpy arrays)."""
    best = []   # [(score, vector), ...]

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s                     # project onto the simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Power‑law sweep
    for p in np.linspace(0.05, 6.0, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) Exponential sweep
    for lam in np.linspace(1e-5, 0.25, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta sweep – wide grid
    al_vals = np.concatenate([np.geomspace(0.01, 0.3, 14),
                              np.linspace(0.3, 4.0, 30)])
    be_vals = np.concatenate([np.geomspace(0.01, 0.3, 14),
                              np.linspace(0.3, 4.0, 30)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Massive random Dirichlet draws
    #   – 12 concentration values, ~30 000 draws overall (fast on modern CPUs)
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
    draws_per_conc = 2500
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth‑Adam polishing (mirror‑descent on a smooth max)
# ----------------------------------------------------------------------


def smooth_adam_polish(a0: np.ndarray,
                       deadline: float,
                       lr: float = 0.05,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8,
                       alpha_start: float = 5.0,
                       alpha_end: float = 200.0,
                       jitter_every: int = 2500) -> np.ndarray:
    """
    Adam optimiser that minimises a smooth approximation of ``max(conv)``.
    The variables are stored as ``u`` with ``a = softmax(u)``.
    Returns the best vector observed before the deadline.
    """
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)               # log‑space start (avoids zeros)
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    # pre‑compute index array for the diagonal term w[2*j]
    diag_idx = 2 * np.arange(n)

    while time.time() < deadline:
        t += 1

        # ----- soft‑max projection (numerically stable) -----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ----- convolution and smooth weighting -----
        conv = full_convolution(a)                 # length 2·n‑1
        max_val = conv.max()

        # smooth weights approximating the max (log‑sum‑exp trick)
        # w_k ∝ exp(α·(conv_k - max_val))
        alpha = alpha_start + (alpha_end - alpha_start) * (t / (t + 1e4))
        # the denominator (max_val) keeps the exponentials well‑scaled
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0:
            # avoid numerical disaster – fallback to active‑set
            active = np.where(conv >= max_val - max_val * 1e-12)[0]
            w = np.zeros_like(conv)
            w[active] = 1.0 / len(active)
        else:
            w /= w_sum

        # ----- gradient of the smooth max w.r.t. a -----
        # g = Σ_k w_k ∂conv_k/∂a   with  ∂conv_k/∂a_j = 2·a_{k‑j}  (except k = 2j)
        a_rev = a[::-1]
        # cross‑correlation S1_j = Σ_i a_i w_{i+j} = (a_rev * w)[j + n‑1]
        corr = np.convolve(a_rev, w)                         # length 3·n‑2
        S1 = corr[n - 1:n - 1 + n]                           # slice length n
        g = 2.0 * S1 - w[diag_idx] * a                       # diagonal correction

        # ----- mirror‑descent step (soft‑max Jacobian) -----
        g_u = a * (g - np.dot(g, a))

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # gentle decay of step‑size (helps final convergence)
        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of flat plateaux
        if t % jitter_every == 0:
            u += np.random.randn(n) * (lr * 0.001)

        # ----- keep the best solution seen so far -----
        cur_val = float(2.0 * n * max_val)    # identical to compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # periodic re‑evaluation to avoid drift (cheap, uses same conv)
        if t % 6000 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()

    return best_a if best_a is not None else a0


# ----------------------------------------------------------------------
# Light simulated‑annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin of a few seconds for the harness
    deadline = start_time + budget_s - 5.0

    # problem size – 2048 gives a good trade‑off between quality and speed
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previous run
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈250 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline - 650.0, start_time + 250.0)  # reserve most time for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=10)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smooth‑Adam polishing (primary + random restarts)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0          # leave ~115 s for annealing
    if best_seq is not None and time.time() < polish_deadline:
        # primary polish of the current best
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_adam_polish(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth‑adam‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # additional random restarts while time permits
        while time.time() < polish_deadline:
            conc = np.random.choice([0.01, 0.05, 0.1, 0.3, 0.7, 1.5])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_adam_polish(a_start, polish_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑adam‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (≈115 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback to uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
