def targeted_hill_climb(a, deadline):
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)
    # parameters
    tau_stop = something
    step_scale = 0.01  # initial fraction of a[i] to move
    T_start = 0.01
    T_end = 1e-5
    T = T_start
    while time < deadline:
        # compute conv
        conv = full_convolution(a)
        max_val = conv.max()
        # select a shift near max
        # get indices where conv close to max within tolerance epsilon relative
        eps = max_val * 1e-8
        active = np.where(conv >= max_val - eps)[0]
        k = int(np.random.choice(active))
        # compute v_i = a[k - i] if k-i in [0,n)
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        v = np.zeros_like(a)
        v[mask] = a[partner[mask]]
        # choose source i where a[i] > 0 and v[i] is large (so contributes strongly to max)
        # choose target j where v[j] is small (so moving mass reduces max)
        # We can define weight = a[i] * v[i] (contribution)
        contributions = a * v
        # source candidate: pick i according to contributions (probability proportional)
        # Use optimistic: choose max contribution
        src = int(np.argmax(contributions))
        # target candidate: choose j where v[j] minimal (prefer zero)
        # but we want a[j] > 0? Actually target may be any.
        # Let's pick j with minimal v, maybe zero.
        # Among indices with a[j] > 0? Could be zero, but we can add mass there.
        # Choose j randomly among those with v below some quantile. 
        low_mask = v <= np.percentile(v, 20)  # bottom 20%
        # Ensure j != src
        candidates = np.where(low_mask)[0]
        if len(candidates) == 0:
            continue
        # pick random j from candidates not src
        j = int(np.random.choice(candidates))
        if j == src:
            continue
        # Determine amount to move
        delta = step_scale * a[src]
        if delta <= 0:
            continue
        # Apply move
        cand = a.copy()
        cand[src] -= delta
        cand[j] += delta
        # Project onto simplex? Since sum unchanged and nonnegatives ok.
        # Ensure nonnegatives: cand[src] may become negative if delta > a[src]; we set delta <= a[src]
        cand = np.maximum(cand, 0.0) # already ensured
        # No need to project as sum unchanged; but might not be exactly 1 if we clamp.
        # We'll renormalize via project_to_simplex.
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        # Accept if better or random with probability exp(-Δ/T)
        Δ = val - best_score
        if val < best_score or np.random.rand() < np.exp(-Δ / max(T, 1e-12)):
            a = cand
            cur_score = val
            if val < best_score:
                best_score = val
                best = cand.copy()
        # Anneal temperature
        T = max(T_end, T * 0.9999)
        # Decrease step scale slowly
        step_scale = max(step_scale * 0.9999, 1e-5)

"""
Optimise a non‑negative step‑function (probability vector) for
``evaluate_sequence``.

The objective is  ``2·n·max(conv(a,a))/(sum(a)^2)``.
Since the ratio is scale‑invariant we normalise the vector to sum = 1 and
minimise  ``2·n·max(conv)``.

This implementation consists of three stages:

1. **Broad global search** – a large collection of parametric families
   (power‑law, exponential, beta, Gaussian, sine‑power, mixtures) and many
   Dirichlet draws.  The best few candidates are retained.

2. **Targeted stochastic hill‑climb** – starting from the best candidates
   we repeatedly transfer a small amount of mass from a “source’’ index that
   strongly contributes to the current maximal convolution entry to a
   “target’’ index that contributes little.  The move is accepted with a
   Metropolis rule (temperature anneals) so that occasional worsening steps
   allow us to escape local minima.  The move choice is guided by the
   active shift (the shift attaining the current maximum).

3. **Polishing** – a short Adam optimiser on a smooth‑max surrogate, followed
   by a final call to the official ``evaluate_sequence`` for a reliable score.

The whole search respects the 1000 s wall‑clock budget and returns the best
sequence found (as a plain Python list of floats).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        n = v.shape[0]
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT (real input)."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power‑of‑two at least m
    n_fft = 1 << ((m - 1).bit_length())
    A = rfft(a, n=n_fft)
    C = A * A
    conv = irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used during the global baseline phase)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·vec1 + (1‑λ)·vec2, projected onto the simplex."""
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Global baseline exploration (fast, low‑dimensional)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 8000,
) -> list[np.ndarray]:
    """Explore many parametric families and Dirichlet draws, keep the best."""
    best = []  # list of (score, vector)

    def consider(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law sweep
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(0.0001, 0.5, 120):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # dense beta grid
    al_vals = np.linspace(0.02, 2.0, 31)
    be_vals = np.linspace(0.02, 2.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, alpha, beta))

    # random beta draws (explore off‑grid points)
    for _ in range(200):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)   # 0.01 … 2
        beta = 10 ** np.random.uniform(-2, 0.3)
        consider(generate_beta(n, alpha, beta))

    # Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # sine‑power sweep
    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # mixtures of two families
    for _ in range(150):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3)
        )
        lam = np.random.rand()
        consider(mixture(a, b, lam))

    # Dirichlet draws with a wide range of concentrations
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            consider(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Guided stochastic hill‑climb (targeted moves)
# ----------------------------------------------------------------------


def hill_climb_targeted(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Stochastic hill‑climb that explicitly reduces the contribution to the
    current maximal convolution entry.  The algorithm works on the
    probability simplex, uses a Metropolis acceptance rule with an annealing
    schedule and constantly shrinks the transfer step size.
    """
    a = a0.copy()
    n = a.shape[0]

    # keep the best ever seen
    best_a = a.copy()
    best_score = compute_score(a)

    # annealing parameters
    T = 0.02               # initial temperature
    T_min = 1e-7
    cool_factor = 0.9995

    # step‑size (fraction of source mass that will be moved)
    step_scale = 0.04
    step_min = 5e-6

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and its maximum
        conv = full_convolution(a)
        max_val = conv.max()

        # choose a random active shift (one attaining the max)
        eps = max_val * 1e-12
        active_shifts = np.where(conv >= max_val - eps)[0]
        k = int(np.random.choice(active_shifts))

        # partner array v_i = a_{k-i} (zero outside)
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        v = np.zeros_like(a)
        v[mask] = a[partner[mask]]

        # contribution of each index to the active shift
        contrib = a * v

        # source: index with largest contribution (most harmful)
        src = int(np.argmax(contrib))

        # target: any index where partner value is relatively small
        # we bias towards indices where v is near its minimum
        low_threshold = np.percentile(v, 15)
        cand_targets = np.where(v <= low_threshold)[0]
        if cand_targets.size == 0:
            # fallback – pick a random different index
            cand_targets = np.where(np.arange(n) != src)[0]
        tgt = int(np.random.choice(cand_targets))

        if src == tgt:
            continue

        # amount to transfer – bounded by the source mass
        delta = step_scale * a[src]
        if delta <= 0.0:
            continue

        # propose new vector
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # maintain feasibility (non‑negative, sum=1) via projection
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        delta_score = cand_score - best_score

        # Metropolis acceptance
        if cand_score < best_score or np.random.rand() < np.exp(-delta_score / max(T, 1e-12)):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        # temperature and step‑size annealing
        T = max(T_min, T * cool_factor)
        step_scale = max(step_min, step_scale * 0.9999)

        # occasional global jitter to escape plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.5)
            a = a + jitter
            a = project_to_simplex(a)

    return best_a

# Smooth‑max Adam optimiser (fine‑grained polishing)
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-6,
    tau_decay: float = 0.96,
    lr_start: float = 0.08,
    lr_decay: float = 0.995,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on a smooth‑max surrogate of the max‑convolution.
    Returns the best vector observed before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # temperature schedule
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # smooth‑max weights
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()               # soft‑max over shifts

        # gradient of smooth‑max w.r.t. a: 2 * (a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = rfft(a, n=size)
        B = rfft(rev_w, n=size)
        conv_full = irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # bookkeeping
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # learning‑rate annealing
        if t % 400 == 0:
            lr = max(lr * lr_decay, 5e-4)

        # occasional jitter
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a

        hill_deadline = min(deadline - 120.0,
                            start_time + 300.0 + (deadline - start_time) * 0.35)
        # start from the current best
        start_vectors = [np.array(best_seq, dtype=float)]
        # add a few random starts to diversify
        for _ in range(3):
            start_vectors.append(project_to_simplex(np.random.rand(n)))

        refined_best = None
        refined_val = float("inf")

        for idx, start_vec in enumerate(start_vectors):
            if time.time() >= hill_deadline:
                break
            local_deadline = hill_deadline - (time.time() - start_time) - 0.5
            if local_deadline <= time.time():
                break
            a_ref = hill_climb_targeted(start_vec, local_deadline)
            cur_val = compute_score(a_ref)
            print(f"[hill‑climb {idx}] score = {cur_val:.6f}")
            if cur_val < refined_val:
                refined_val, refined_best = cur_val, a_ref.copy()

        if refined_best is not None:
            best_val, best_seq = refined_val, refined_best.tolist()
            print(f"[hill‑climb] best after stochastic phase = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Adam‑based smooth‑max polishing (≈ 150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 30.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = smooth_max_adam(a0, polish_deadline)
        val = compute_score(a_refined)
        print(f"[adam‑polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Final validation with the official evaluation function
    # -----------------------------------------------------------------
    if best_seq is None:
        # safety fallback – uniform vector
        best_seq = [1.0 / n] * n

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
