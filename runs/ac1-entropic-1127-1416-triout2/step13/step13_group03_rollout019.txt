"""
Step‑function optimiser targeting a lower objective for ``evaluate_sequence``.

Core ideas
-----------
1. Larger search space (n = 4096) – the bound (2·n)/(2·n‑1) ≈ 1.0001 is
   reachable only with a flat self‑convolution, so a larger n gives the optimiser
   more flexibility.

2. Enriched global search – besides the families used before we add:
   * uniform weight on a random subset of indices,
   * spectral‑factorisation draws: start from a (near‑uniform) convolution and
     take a random square‑root in the Fourier domain, then project back to the
     probability simplex.

3. Faster Adam optimiser – an Adam optimiser on the smooth‑max surrogate with
   bias‑correction and an aggressive learning‑rate/temperature schedule.

4. Longer polishing and annealing – after Adam we run many more sub‑gradient
   steps and a slower simulated‑annealing phase.  All phases respect the 1000 s
   wall‑clock budget.

The function ``propose_candidate`` returns the best list of floats found within
the time limit.
"""

import time
import numpy as np
from scipy.optimize import linprog  # kept for possible future extensions

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v>=0)."""
    if np.all(v <= 0):
        # all non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of suitable length."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for an initial diverse pool)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return project_to_simplex(a)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    return project_to_simplex(a)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    return project_to_simplex(a)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return project_to_simplex(a)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    return project_to_simplex(a)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·vec1 + (1‑λ)·vec2, projected onto the simplex."""
    return project_to_simplex(lam * vec1 + (1.0 - lam) * vec2)


def generate_uniform_subset(n: int, size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform weight on a random subset of `size` indices."""
    idx = rng.choice(n, size=size, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / size
    return a  # already sums to 1


def generate_spectral_draw(
    n: int,
    L: int,
    mag_sqrtU: np.ndarray,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Spectral draw: start from the (approximate) uniform convolution and take a
    random square‑root in the Fourier domain, then invert and project.
    """
    phase = np.exp(1j * rng.uniform(0.0, 2.0 * np.pi, size=mag_sqrtU.shape))
    V = mag_sqrtU * phase
    a_full = np.fft.irfft(V, n=L)
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    return project_to_simplex(a)

def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 20,
    random_draws: int = 8000,
) -> list[np.ndarray]:
    """
    Explore a large variety of families and a pool of random draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays).
    """
    best = []  # list of (score, vector)

    def consider(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    rng = np.random.default_rng()

    # ----- classic families -------------------------------------------------
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    for lam in np.linspace(0.0001, 0.5, 120):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    al_vals = np.linspace(0.02, 2.0, 31)
    be_vals = np.linspace(0.02, 2.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, alpha, beta))

    for _ in range(250):
        if time.time() >= deadline:
            break
        alpha = 10 ** rng.uniform(-2, 0.3)
        beta = 10 ** rng.uniform(-2, 0.3)
        consider(generate_beta(n, alpha, beta))

    for sigma in np.linspace(0.1 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # ----- mixtures ---------------------------------------------------------
    for _ in range(200):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, rng.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** rng.uniform(-2, 0.3),
            10 ** rng.uniform(-2, 0.3),
        )
        lam = rng.random()
        consider(mixture(a, b, lam))

    # ----- random subsets (uniform on subset) -------------------------------
    sqrt_n = int(np.sqrt(n))
    for sz in [sqrt_n, int(1.5 * sqrt_n), int(2 * sqrt_n), int(3 * sqrt_n), n // 2]:
        draws = max(1, random_draws // (len(al_vals) * len(be_vals)))
        for _ in range(draws):
            if time.time() >= deadline:
                break
            consider(generate_uniform_subset(n, sz, rng))

    # ----- Dirichlet draws (wide range of concentrations) -------------------
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = rng.dirichlet([conc] * n)
            consider(project_to_simplex(a))

    # ----- spectral factorisation draws ------------------------------------
    # target: uniform convolution of length 2n‑1
    L = 1 << ((2 * n - 2).bit_length())  # next power of two
    u = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=float)
    u_padded = np.zeros(L, dtype=float)
    u_padded[: 2 * n - 1] = u
    U = np.fft.rfft(u_padded)
    mag_sqrtU = np.sqrt(np.abs(U))

    # deterministic (zero‑phase) draw – may already be useful
    consider(generate_spectral_draw(n, L, mag_sqrtU, rng))

    # many random‑phase draws
    for _ in range(300):
        if time.time() >= deadline:
            break
        consider(generate_spectral_draw(n, L, mag_sqrtU, rng))

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

# Adam optimiser on a smooth‑max proxy of max(conv)
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-7,
    tau_decay: float = 0.92,
    lr_start: float = 0.14,
    lr_decay: float = 0.98,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate:
        fτ(a) = τ·log Σ_k exp(conv_k(a)/τ)

    Returns the best vector (projected onto the simplex) seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # ----- temperature schedule -----
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # ----- smooth‑max weights (soft‑max over the convolution) -----
        conv = full_convolution(a)            # length 2n‑1
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()         # size 2n‑1

        # ----- gradient of the smooth‑max w.r.t a -----
        # gradient = 2 * (a ⋆ w)   (cross‑correlation)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]

        # indices that map the full cross‑correlation back to length n
        idx = np.arange(n) + (rev_w.shape[0] - 1)
        grad = 2.0 * conv_full[idx]

        # ----- Adam update -----
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # ----- bookkeeping -----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # ----- learning‑rate annealing -----
        if t % 400 == 0:
            lr = max(lr * lr_decay, 5e-5)

        # occasional jitter to escape plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.004 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (more iterations than before)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient (sum of contributions from all active shifts)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 4000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (stochastic hill‑climb with a cooler schedule)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = 2e-3
    step_scale = 0.06
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        # Metropolis acceptance
        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, 5e-6)

        # exponential cooling
        T = max(1e-6, T * 0.9992)

        if it % 8000 == 0:
            # occasional jitter to diversify
            jitter = np.random.randn(n) * (step_scale * 0.015)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    n = 4096                                        # a larger vector gives a tighter bound
    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0. optional fallback from a previously discovered construction
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Global baseline exploration (≈ 250 s)
    # --------------------------------------------------------------
    baseline_deadline = deadline - 350.0
    if time.time() < baseline_deadline:
        candidates = baseline_search(
            n, baseline_deadline, max_keep=20, random_draws=10000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Adam‑based smooth‑max optimisation from the best candidates
    # --------------------------------------------------------------
    if best_seq is not None:
        grad_deadline = min(deadline - 180.0,
                            start_time + 250.0 + (deadline - start_time) * 0.33)

        # initialise a small pool of start vectors
        start_vectors = []
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        start_vectors.append(best_vec)

        # uniform start (often a good reference)
        start_vectors.append(np.full(n, 1.0 / n))

        # few random starts
        for _ in range(5):
            start_vectors.append(project_to_simplex(np.random.rand(n)))

        # pure spectral draw (no randomness) – often very flat
        rng = np.random.default_rng(seed + 999)
        L = 1 << ((2 * n - 2).bit_length())
        u = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=float)
        u_padded = np.zeros(L, dtype=float)
        u_padded[: 2 * n - 1] = u
        U = np.fft.rfft(u_padded)
        mag_sqrtU = np.sqrt(np.abs(U))
        start_vectors.append(project_to_simplex(
            generate_spectral_draw(n, L, mag_sqrtU, rng))
        )

        refined_best = best_vec.copy()
        refined_val = best_val

        for idx, sv in enumerate(start_vectors):
            if time.time() >= grad_deadline:
                break
            local_deadline = grad_deadline - (time.time() - start_time) - 0.5
            if local_deadline <= time.time():
                break
            a_ref = smooth_max_adam(sv, local_deadline)
            val = compute_score(a_ref)
            print(f"[adam‑opt {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_ref.copy()

        best_val, best_seq = refined_val, refined_best.tolist()
        print(f"[adam‑opt] best after smooth optimisation = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Sub‑gradient polishing (more iterations than before)
    # --------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_ref = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_ref)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_ref.tolist()

    # --------------------------------------------------------------
    # 4. Simulated annealing / stochastic hill‑climb
    # --------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # --------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing found
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final report using the official evaluation function
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
