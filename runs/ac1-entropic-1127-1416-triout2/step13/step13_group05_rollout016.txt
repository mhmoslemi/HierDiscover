*   After the sweep run a longer projected sub‑gradient polishing phase

*   All helpers are top‑level (no closures) and the entry point is

# Linear algebra utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n−i)^p,  p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)      # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p).  p ≥ 0."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^i with 0 < q < 1."""
    a = q ** np.arange(n, dtype=np.float64)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """front plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^α (1‑x_i)^β,  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def multi_power(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^p · (i+1)^q.
    Both p,q ≥ 0 give a front‑loaded shape that can be tuned by the
    exponent on the left‑hand side (p) and a gentle lift on the tail
    (q).
    """
    left = np.arange(n, 0, -1, dtype=np.float64) ** p      # n‑i
    right = np.arange(1, n + 1, dtype=np.float64) ** q    # i+1
    a = left * right
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def sigmoid_shape(n: int, centre_frac: float, sigma_frac: float) -> np.ndarray:
    """
    a_i ∝ 1/(1 + exp((i - c) / σ)).
    ``centre_frac`` is the location of the mid‑point as a fraction of n,
    ``sigma_frac`` controls the steepness (σ = sigma_frac * n).
    """
    c = centre_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)          # avoid division by zero
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - c) / sigma))
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a

# Projected sub‑gradient polishing (faster step‑size schedule)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max(conv) w.r.t. a:
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to avoid stagnation
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (more iterations and adaptive step‑size)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 1_200_000) -> np.ndarray:
    """Greedy hill‑climb with adaptive step‑size and restart on stagnation."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]        # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 200_000:
            # jump start from best with a tiny random perturbation
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # safety margin

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # (0) Optional fallback from an existing global variable
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # (1) Parametric sweep – bulk of the time (~300 s)
    # -------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    n = 2048
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            # quick numeric score (exact under Σa=1)
            val = compute_score(a_initial)
            print(f"[sweep] best proxy score = {val:.6f}")
            # official evaluation for safety
            val_real = evaluate_sequence(a_initial.tolist())
            print(f"[sweep] official score = {val_real:.6f}")
            if val_real < best_val:
                best_val, best_seq = val_real, a_initial.tolist()

    # -------------------------------------------------------------
    # (2) Sub‑gradient polishing – terminates ~80 s before deadline
    # -------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.12)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -------------------------------------------------------------
    # (3) Final stochastic hill‑climb
    # -------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # (4) Safeguard – fall back to a reasonable analytic candidate
    # -------------------------------------------------------------
    if best_seq is None:
        # Analytic linear decreasing shape (p = 1) is a good default.
        a_fallback = reverse_power_law(n, 1.0)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback] using linear shape, score = {best_val:.6f}")

    # -------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported final score = {final_score:.6f}")
    return best_seq
