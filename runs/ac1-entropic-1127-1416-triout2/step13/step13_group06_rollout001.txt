"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

using a two‑stage optimisation:

0.  Try the previously best construction (fallback).
1.  Optimise a *coarse* representation (256 blocks) with a fast
    simulated‑annealing on the cheap approximate objective.
    The coarse objective is exactly proportional to the full one,
    so ranking candidates is correct while evaluation is cheap.
2.  Upsample the best coarse vector to the full length (8192) and
    polish it with exponentiated‑gradient mirror descent and a final
    high‑temperature simulated‑annealing on the exact objective.

All steps respect the supplied wall‑clock budget (default 1000 s) and
return the best list of non‑negative heights found.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic FFT convolution utilities (full aperiodic convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)^2).
    a is a 1‑D non‑negative array (no further checks).
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used for SA moves
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror) descent on the full objective
# ----------------------------------------------------------------------


def exp_gradient_descent(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         eta0: float = 0.6) -> Tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using exponentiated gradients.
    Sub‑gradient is taken at the index where the auto‑convolution attains
    its maximum.  Diminishing stepsize ηₜ = η₀ / √t.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                     # index of max
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]                    # sub‑gradient

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)                     # EG step
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s                                 # renormalise (KL‑proj.)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] exp‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] exp‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb on the full objective
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 12_000_000,
                  temp_factor: float = 1.2e-3,
                  step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    Moves consist of swapping two entries or transferring a fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)                     # ensure feasibility

        # evaluate with the cheap coarse objective
        cand_small_score = compute_score(cand)
        cand_score = block * cand_small_score

        # linear cooling schedule
        T = T0 * (1.0 - step / max_steps)
        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur_coarse = cand
            cur_score = cand_score
            # slowly increase step size on success
            step_scale = min(step_scale * 1.02, 0.4)

            if cur_score < best_coarse_score:
                best_coarse = cur_coarse.copy()
                best_coarse_score = cur_score
                print(f"[{time.time():.1f}] new coarse best ≈ {best_coarse_score:.6f}")

        else:
            # shrink step size on rejection
            step_scale = max(step_scale * 0.99, 0.02)

        # occasional full‑precision check for monitoring
        if step % 250_000 == 0:
            full_vec = np.repeat(best_coarse, block)
            full_prec = evaluate_sequence(full_vec.tolist())
            print(f"[{time.time():.1f}] step {step:,} – coarse best ≈{best_coarse_score:.6f}, full ≈{full_prec:.6f}")

    # ------------------------------------------------------------------
    # 2. Upsample the best coarse vector and polish it
    # ------------------------------------------------------------------
    best_full = np.repeat(best_coarse, block)
    best_full = np.maximum(best_full, 0.0)
    s = best_full.sum()
    if s == 0:
        best_full = np.full_like(best_full, 1.0 / best_full.size)
    else:
        best_full = best_full / s

    exact_score = evaluate_sequence(best_full.tolist())
    if exact_score < best_score:
        best_score = exact_score
        best_seq = best_full.tolist()
        print(f"[{time.time():.1f}] after coarse SA exact = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Fine‑grained exponentiated‑gradient descent (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0 and best_seq:
        print(f"[{time.time():.1f}] start exponentiated‑gradient refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = exp_gradient_descent(
            start_a=a_start,
            deadline=deadline - 120.0,
            max_iters=300_000,
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final high‑temperature SA hill‑climb (full objective)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0 and best_seq:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=10_000_000,
            temp_factor=1.2e-3,
            step_scale_init=0.25,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] final SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Safety fallback (uniform) in the extremely unlikely case of failure
    # ------------------------------------------------------------------
    if not best_seq:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
