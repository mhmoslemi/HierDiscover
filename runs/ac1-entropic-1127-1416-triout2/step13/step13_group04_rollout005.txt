def triangular_tent(n, mu_frac, width_frac):
    mu = mu_frac * (n - 1)
    w = max(width_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - np.abs(i - mu) / w)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a

Note: width_frac can be >0.5, w > n/2, shape becomes wide.

def gaussian_shape(n, mu_frac, sigma_frac):
    mu = mu_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    if a.sum() == 0.0:

    a /= a.sum()
    return a

def exp_mix(n, q1, q2, w1):
    q1 = float(q1); q2 = float(q2); w1 = float(w1)
    if not (0 < q1 < 1 and 0 < q2 < 1 and 0 <= w1 <= 1):
        return None
    i = np.arange(n, dtype=np.float64)
    a = w1 * (q1 ** i) + (1 - w1) * (q2 ** i)
    if a.sum() == 0:
        return None
    a /= a.sum()
    return a

"""
search for a non‑negative step‑function sequence that minimises the
official ``evaluate_sequence`` function.

The algorithm consists of four stages:

1️⃣ Broad parametric sweep over many low‑dimensional “shape families”.
   New families (triangular, Gaussian, mixed exponentials, truncated
   power‑law) are added to the previous set.  All candidates are
   normalised to Σa = 1, because the objective is scale‑invariant.

2️⃣ Momentum‑accelerated smoothed‑gradient polishing.
   The non‑smooth max‑convolution is approximated by a soft‑max
   (log‑sum‑exp) with an increasing temperature parameter β.
   The resulting gradient is computed by a single FFT‑based convolution.
   A Euclidean projection onto the probability simplex keeps the
   iterates feasible.

3️⃣ Simulated‑annealing style stochastic mass‑transfer optimisation,
   identical to the reference implementation but with slightly tuned
   hyper‑parameters.

4️⃣ Deterministic hill‑climbing for a final fine‑polish.

All stages respect the overall 1000 s budget (a 5 s safety margin is

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def linear_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution of two real vectors via FFT."""
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:n]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a (full convolution)."""
    return linear_convolution(a, a)


def compute_score(a: np.ndarray) -> float:
    """Scale‑invariant objective for a normalised vector (Σa = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def power_law(n: int, p: float) -> np.ndarray | None:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    return a / s if s > 0 else None


def geometric(n: int, q: float) -> np.ndarray | None:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else None


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray | None:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    return a / s if s > 0 else None


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    return a / s if s > 0 else None


def logistic_shape(n: int, center_frac: float, scale_frac: float) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    scale = max(scale_frac * n, 1e-12)
    a = 1.0 / (1.0 + np.exp((i - center) / scale))
    s = a.sum()
    return a / s if s > 0 else None


def random_dirichlet(n: int, concentration: float) -> np.ndarray | None:
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    s = y.sum()
    return y / s if s > 0 else None


def triangular_tent(n: int, mu_frac: float, width_frac: float) -> np.ndarray | None:
    mu = mu_frac * (n - 1)
    w = max(width_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - np.abs(i - mu) / w)
    s = a.sum()
    return a / s if s > 0 else None


def gaussian_shape(n: int, mu_frac: float, sigma_frac: float) -> np.ndarray | None:
    mu = mu_frac * (n - 1)
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else None


def exp_mix(n: int, q1: float, q2: float, w1: float) -> np.ndarray | None:
    if not (0 < q1 < 1 and 0 < q2 < 1 and 0 <= w1 <= 1):
        return None
    i = np.arange(n, dtype=np.float64)
    a = w1 * (q1 ** i) + (1.0 - w1) * (q2 ** i)
    s = a.sum()
    return a / s if s > 0 else None


def truncated_power_law(n: int, p: float, cut_frac: float) -> np.ndarray | None:
    cut = max(1, int(cut_frac * n))
    a = np.empty(n, dtype=np.float64)
    a[:cut] = (np.arange(1, cut + 1, dtype=np.float64) ** (-p))
    a[cut:] = 0.0
    s = a.sum()
    return a / s if s > 0 else None

# Broad parametric sweep (low‑dimensional families)
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray | None:
    best_vec = None
    best_val = float('inf')

    # helpers to keep code succinct
    def consider(vec: np.ndarray):
        nonlocal best_vec, best_val
        if vec is None:
            return
        val = compute_score(vec)
        if val < best_val:
            best_val, best_vec = val, vec

    # 1. Power‑law
    for p in np.linspace(0.10, 2.00, 191):
        if time.time() >= deadline:
            return best_vec
        consider(power_law(n, p))

    # 2. Geometric
    geo_qs = np.concatenate((np.linspace(0.90, 0.99, 91),
                             np.linspace(0.990, 0.9999, 110)))
    for q in geo_qs:
        if time.time() >= deadline:
            return best_vec
        consider(geometric(n, q))

    # 3. Mixed plateau + power
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= deadline:
                return best_vec
            consider(mixed_plateau_power(n, L_frac, p))

    # 4. Beta shape
    alphas = np.linspace(-0.8, 3.0, 20)
    betas = np.linspace(-0.8, 3.0, 20)
    for a in alphas:
        for b in betas:
            if time.time() >= deadline:
                return best_vec
            consider(beta_shape(n, a, b))

    # 5. Logistic
    for center_frac in np.linspace(0.2, 0.8, 13):
        for scale_frac in np.linspace(0.01, 0.20, 20):
            if time.time() >= deadline:
                return best_vec
            consider(logistic_shape(n, center_frac, scale_frac))

    # 6. Triangular tent
    for mu_frac in np.linspace(0.0, 1.0, 21):
        for width_frac in np.linspace(0.05, 1.0, 20):
            if time.time() >= deadline:
                return best_vec
            consider(triangular_tent(n, mu_frac, width_frac))

    # 7. Gaussian
    for mu_frac in np.linspace(0.0, 1.0, 11):
        for sigma_frac in np.linspace(0.05, 0.50, 10):
            if time.time() >= deadline:
                return best_vec
            consider(gaussian_shape(n, mu_frac, sigma_frac))

    # 8. Exponential mixture
    q_vals = np.logspace(np.log10(0.80), np.log10(0.9999), 7)
    for q1 in q_vals:
        for q2 in q_vals:
            for w1 in np.linspace(0.0, 1.0, 11):
                if time.time() >= deadline:
                    return best_vec
                consider(exp_mix(n, q1, q2, w1))

    # 9. Truncated power‑law
    for p in np.linspace(0.10, 3.00, 30):
        for cut_frac in np.linspace(0.10, 0.90, 17):
            if time.time() >= deadline:
                return best_vec
            consider(truncated_power_law(n, p, cut_frac))

    # 10. Random Dirichlet (a few hundred samples)
    for _ in range(800):
        if time.time() >= deadline:
            break
        conc = 10 ** np.random.uniform(-2, 1)   # 0.01 ... 10
        consider(random_dirichlet(n, conc))

    return best_vec


# ----------------------------------------------------------------------
# Smoothed‑gradient polishing with momentum (soft‑max approximation)
# ----------------------------------------------------------------------
def smooth_gradient_polish(a0: np.ndarray, deadline: float,
                          step0: float = 0.12,
                          beta0: float = 5.0,
                          beta_max: float = 400.0,
                          beta_factor: float = 1.003,
                          momentum: float = 0.9) -> np.ndarray:
    """
    Gradient descent on a smooth approximation of max(conv(a,a)).
    Uses a soft‑max (log‑sum‑exp) with a gradually increasing beta.
    """
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    v = np.zeros_like(a)
    t = 0
    n = a.shape[0]
    eps = 1e-12

    while time.time() < deadline:
        t += 1
        # convolution and soft‑max weights
        conv = full_convolution(a)
        beta = min(beta_max, beta0 * (beta_factor ** t))
        scaled = beta * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)          # stability shift
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient of the smoothed max: 2 * (w * a_rev) cross‑correlation
        a_rev = a[::-1]
        grad_full = linear_convolution(w, a_rev)     # length 3n‑2
        g = 2.0 * grad_full[n - 1:n - 1 + n]

        # momentum update and projection
        v = momentum * v + (1.0 - momentum) * g
        step = step0 / math.sqrt(t)
        a = a - step * v
        a = project_to_simplex(a)

        # keep best observed point
        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer) – same as reference but with tuned params
# ----------------------------------------------------------------------
def anneal_search(a_start: np.ndarray, deadline: float) -> np.ndarray:
    n = a_start.shape[0]
    cur = a_start.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    temp = max(1e-4 * best_score, 1e-8)
    temp_decay = 0.9997
    step_scale = 0.05
    step_decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * cur[i] * np.random.rand()
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            accept = (np.random.rand() < prob)

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"[anneal] new best {best_score:.6f}  it={it}  time={time.time():.1f}s")
                step_scale = min(step_scale * 1.02, 0.6)

        temp = max(temp * temp_decay, 1e-12)
        step_scale = max(step_scale * step_decay, 1e-7)

    return best


# ----------------------------------------------------------------------
# Deterministic hill‑climb (greedy pairwise mass moves)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence minimising
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback (previous construction, if available)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Parametric sweep (≈ 250 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    n = 4096
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Smoothed gradient polishing with momentum (≈ 150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 250.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = smooth_gradient_polish(a0, polish_deadline)
        val = compute_score(a_polished)
        print(f"[smooth‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 3. Simulated annealing (≈ 250 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 120.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_annealed = anneal_search(a_start, anneal_deadline)
        val = compute_score(a_annealed)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_annealed.tolist()

    # -----------------------------------------------------------------
    # 4. Final hill‑climb (remaining time)
    # -----------------------------------------------------------------
    hill_deadline = deadline
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report (official evaluation)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
