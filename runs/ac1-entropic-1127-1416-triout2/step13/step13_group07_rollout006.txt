def mirror_descent_optimize(start_a, deadline, max_iters=200000, step0=0.2):
    a = start_a.copy()
    log_a = np.log(a + 1e-12)
    best_a = a.copy()
    best_score = compute_score(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        a = np.exp(log_a)
        a = a / a.sum()
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # define active set: values within tol of max
        # use absolute tolerance: 1e-12 maybe
        active = conv >= maxc - 1e-12
        # weight as uniform mass over active indices
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # gradient of weighted max (smooth)
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)[len(a)-1:len(a)-1+len(a)]
        # mirror descent step
        eta = step0 / np.sqrt(it)
        log_a -= eta * grad_a
        # optional clipping
        if it % 5000 == 0:
            # evaluate

            a_tmp = a_tmp / a_tmp.sum()

"""
propose_candidate
-----------------
Search for a non‑negative step‑function (a probability vector) that
minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm builds on the successful “product‑family” shape but adds
a few new ingredients that were not used in the previous baseline:

1. **Two‑stage resolution** – a quick coarse scan on a modest size
   (n≈2 k) to locate a promising parameter (α) of the symmetric product
   family, followed by an up‑scaling to a larger size (n≈32 k) so that
   the discrete approximation is closer to the continuous optimum.

2. **Golden‑section refinement** of the single scalar α on the large
   problem.  This gives a very accurate starting shape with virtually
   no extra cost.

3. **Exponentiated‑gradient (mirror‑descent) optimisation** on the
   exact max‑objective.  At each iteration the active set of convolution
   entries attaining the current maximum is used as the sub‑gradient.
   The update is performed in log‑space (a_i ← a_i·exp(−η·g_i)) which
   automatically respects non‑negativity; a projection onto the simplex
   (renormalisation) keeps the L¹‑norm fixed.

4. **Final greedy mass‑transfer hill‑climb** (as in the baseline) to
   polish any remaining tiny improvement.

All stages respect a single wall‑clock budget (≈1000 s).  The best
sequence found is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    a is assumed non‑negative (the optimiser keeps it on the simplex).
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised.
    Symmetric, U‑shaped for α>0.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against overflow at the ends
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


# ----------------------------------------------------------------------
# One‑dimensional optimisation helpers
# ----------------------------------------------------------------------


def golden_section(
    func, lo: float, hi: float, max_iter: int = 30
) -> Tuple[float, float]:
    """
    Unimodal minimisation on [lo, hi] using golden‑section search.
    Returns (x_opt, f_opt).
    """
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Mirror‑descent optimiser on the true max‑objective
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.25,
    min_step: float = 1e-5,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the max‑convolution
    objective.  The sub‑gradient is taken w.r.t. the active set
    (indices attaining the current maximum).  Updates are performed in
    log‑space and the vector is re‑normalised each iteration, which
    corresponds to a KL‑projection onto the simplex.
    """
    eps = 1e-12
    n = start_a.shape[0]
    log_a = np.log(start_a + eps)
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    it = 0
    while True:
        it += 1
        if time.time() >= deadline:
            break

        # recover probability vector
        a = np.exp(log_a)
        a = a / a.sum()

        # full convolution
        conv = convolution_fft(a, a)
        M = conv.max()

        # active set (within a tiny absolute tolerance)
        # the tolerance is chosen relative to M to avoid numerical noise
        tol = max(1e-12, 1e-10 * M)
        active = conv >= M - tol
        if not np.any(active):
            # degenerate situation – fall back to smooth max
            w = np.exp(1e3 * (conv - M))
            w /= w.sum()
        else:
            # uniform weight over the active indices, normalised to 1
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        # gradient of the weighted‐max (same formula as in the Adam part)
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # step size schedule (inverse square‑root)
        eta = max(step0 / math.sqrt(it), min_step)

        # mirror‑descent (exponentiated) update in log‑space
        log_a -= eta * grad

        # occasional evaluation & bookkeeping
        if it % 4000 == 0:
            a_eval = np.exp(log_a)
            a_eval = a_eval / a_eval.sum()
            sc = compute_score(a_eval)
            if sc < best_score:
                best_score = sc
                best_a = a_eval.copy()
                print(f"[{time.time():.1f}] mirror‑descent improved to {best_score:.6f}")

    # final projection & evaluation
    a_final = np.exp(log_a)
    a_final = a_final / a_final.sum()
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
        print(f"[{time.time():.1f}] mirror‑descent final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass‑transfer) – final polishing
# ----------------------------------------------------------------------


def hill_climb_greedy(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 5_000_000,
    init_scale: float = 0.15,
) -> Tuple[np.ndarray, float]:
    """
    Very simple hill‑climb that repeatedly moves a fraction of mass
    from the current largest entry to a random other entry whenever the
    move reduces the objective.  The transfer fraction is adapted
    multiplicatively.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score
    scale = init_scale

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        i = np.argmax(cur_a)            # donor (largest)
        j = np.random.randint(n)        # receiver
        if i == j:
            continue

        delta = scale * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            scale = min(scale * 1.03, 0.5)   # enlarge step on success
        else:
            scale = max(scale * 0.97, 1e-4)   # shrink on failure

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] greedy step {step}, best {best_score:.6f}")

    return best_a, best_score
