"""
Step‑function optimiser for the hidden ``evaluate_sequence``.
The optimiser runs three phases under a 1000 s wall‑clock budget:

1. **Expanded global sweep** – dense scans of several parametric families
   (power‑law, reversed power‑law, exponential, Beta, parabola, block,
   random binary support) together with many Dirichlet draws.  The
   ``max_keep`` best candidates (by the exact objective
   ``2·n·max(conv)`` for a probability vector) are retained.

2. **Inverse‑gradient polishing** – a fast mirror‑descent style update
   that repeatedly re‑weights the vector by ``exp(-β·g)`` where ``g`` is the
   sub‑gradient of the active convolution constraints.  The temperature β
   is cooled during the run.  A handful of random‑restart runs are launched
   while time permits.

3. **Optional simulated‑annealing finish** – a short Metropolis style
   anneal starting from the best vector to escape shallow local minima.

All stages respect the allotted time and the function finally returns a
plain Python list of non‑negative floats that minimises ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of size a power‑of‑two ≥ 2·len(a)−1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a))
    This coincides with ``evaluate_sequence`` for sum(a)=1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all vectors are returned already normalised to sum=1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_rev_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{‑p} (mirrored power‑law)."""
    a = np.arange(n, 0, -1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_parabola(n: int) -> np.ndarray:
    """a_i ∝ (i+1)·(n‑i)  (discrete parabola, symmetric, peaks in the middle)."""
    i = np.arange(1, n + 1, dtype=float)
    a = i * (n - i + 1)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_block(n: int, block_frac: float) -> np.ndarray:
    """
    Uniform block of width ``block_frac·n`` placed at the left side.
    The block is normalised to sum 1.
    """
    w = max(1, int(block_frac * n))
    a = np.concatenate([np.full(w, 1.0 / w), np.zeros(n - w)])
    return a


def generate_sparse_binary(n: int, k: int) -> np.ndarray:
    """
    Random binary support of size ``k`` with equal mass 1/k on the chosen
    positions.  Returns a probability vector.
    """
    if k <= 0 or k > n:
        return np.full(n, 0.0)
    a = np.zeros(n, dtype=float)
    idx = np.random.choice(n, size=k, replace=False)
    a[idx] = 1.0 / k
    return a

# Global exploration – keep the best ``max_keep`` vectors
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 20):
    """Sweep families + massive Dirichlet draws; keep the best candidates."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s                     # simplex projection
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law (both directions)
    for p in np.logspace(np.log10(0.02), np.log10(8.0), 400):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))
        try_candidate(generate_rev_powerlaw(n, p))

    # 2) exponential (wide lam range)
    for lam in np.logspace(-5, -0.3, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta family – cover U‑shapes and highly skewed shapes
    al_vals = np.concatenate([np.geomspace(0.01, 0.3, 14),
                              np.linspace(0.3, 5.0, 30)])
    be_vals = np.concatenate([np.geomspace(0.01, 0.3, 14),
                              np.linspace(0.3, 5.0, 30)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) simple parabola
    if time.time() < deadline:
        try_candidate(generate_parabola(n))

    # 5) block families (different widths)
    for frac in np.linspace(0.01, 0.5, 20):
        if time.time() >= deadline:
            break
        try_candidate(generate_block(n, frac))

    # 6) massive random Dirichlet draws with diverse concentration
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1200
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    # 7) random sparse binary supports (different k)
    ks = [int(np.sqrt(2 * n) * f) for f in [0.8, 1.0, 1.2, 1.5, 2.0]]
    for k in ks:
        for _ in range(300):
            if time.time() >= deadline:
                break
            try_candidate(generate_sparse_binary(n, k))

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]

# Inverse‑gradient polishing (mirror‑descent on the active set)
# ----------------------------------------------------------------------


def inverse_polish(a0: np.ndarray, deadline: float,
                   beta0: float = 4.0,
                   decay: float = 0.995,
                   min_beta: float = 0.1,
                   tol: float = 1e-12) -> np.ndarray:
    """
    Re‑weights the vector by ``exp(-β·g)`` where ``g`` is the sub‑gradient
    of the active convolution constraints.  β is cooled geometrically.
    Returns the best vector observed before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    beta = beta0
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        conv = full_convolution(a)
        max_val = conv.max()
        # active indices (within relative tolerance)
        active = np.where(conv >= max_val - max_val * tol)[0]

        # sub‑gradient g (same as in the Adam implementation)
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # new probabilities via soft‑max of -β·g
        u = -beta * g
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        a = exp_u / exp_u.sum()

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        beta = max(min_beta, beta * decay)

    return best_a

def adam_polish(a0: np.ndarray, deadline: float,
                lr: float = 0.05,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                tol: float = 1e-12) -> np.ndarray:
    """
    Adam‑based mirror‑descent on the simplex (identical to the original code).
    Returns the best vector seen before the deadline.
    """
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        conv = full_convolution(a)
        max_val = conv.max()
        active = np.where(conv >= max_val - max_val * tol)[0]

        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        g_u = a * (g - np.dot(g, a))

        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        cur_val = float(2.0 * n * max_val)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

    return best_a if best_a is not None else a0

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a

            seq = [float(x) for x in fallback]
            val = evaluate_sequence(seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (reserve ~250 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 250.0
    if time.time() < baseline_deadline:
        candidates = baseline_search(n, baseline_deadline, max_keep=20)
        for vec in candidates:
            val = compute_score(vec)          # sum(vec) == 1 ⇒ same as evaluate_sequence
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Inverse‑gradient polishing (primary + random restarts)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        # primary polish
        a0 = np.array(best_seq, dtype=float)
        refined = inverse_polish(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[inverse‑polish‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # random restarts while time permits
        while time.time() < polish_deadline:
            # random support sizes centred around sqrt(2n)
            k = int(np.clip(np.random.normal(np.sqrt(2 * n), 10), 30, n))
            a_start = generate_sparse_binary(n, k)
            refined = inverse_polish(a_start, polish_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[inverse‑polish‑restart] new best = {val:.6f}")

        # fallback to Adam polish for a few seconds if time remains
        adam_deadline = deadline - 20.0
        if time.time() < adam_deadline and best_seq is not None:
            a0 = np.array(best_seq, dtype=float)
            refined = adam_polish(a0, adam_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[adam‑polish‑after‑inverse] score = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Defensive fallback (uniform distribution)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
