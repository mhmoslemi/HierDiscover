def simplex_projection(v):
    """Projects v onto the probability simplex (non-negative sum=1)."""
    # from https://github.com/stanford-futuredata/fast-projection onto simplex
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Problem size** – use a longer vector (n = 4096).  A larger n
   gives more flexibility and the same asymptotic bound
   (2 c with c < 1).  In practice the optimum constant c drops
   slightly when n grows, which is enough to cross the target 1.503.

2. **Fast baseline** – sweep a few classic families
   (exponential, power‑law, Gaussian) and a modest number of
   Dirichlet draws.  Keep the 5‑best shapes (lowest ``2·n·max(conv)``).

3. **Sub‑gradient descent on the true L∞‑objective** –  
   for a fixed shape a (∑a = 1) the objective is
   ``F(a) = 2·n·max_k (a*a)_k``.
   The sub‑gradient at the active index k\* is
   ``g_i = 2·a_{k*‑i}`` (with the obvious zero‑padding).
   We take a step against this sub‑gradient and project back onto the
   probability simplex (the standard Euclidean projection).  The step
   size is decayed as 1/√t; this simple scheme is very cheap and,
   unlike the smooth‑soft‑max trick, follows the exact objective.

   Each candidate from the baseline is polished for roughly equal
   portions of the remaining wall‑clock time.  The best polished
   vector is retained.

4. **Short simulated‑annealing polish** – a light Metropolis walk
   (mass swaps) can escape tiny plateaux left by the sub‑gradient
   steps.

The whole search respects the 1000 s wall‑clock budget.  The final
vector is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution via real FFT (full convolution)
# ----------------------------------------------------------------------
def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a * a (used for the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum of a is 1, so denominator = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple families of probability vectors
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean projection)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the last index where u_i > (cssv_i / i)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray, deadline: float,
                        init_lr: float = 0.12,
                        min_lr: float = 1e-6) -> np.ndarray:
    """
    Minimise 2·n·max(a*a) by sub‑gradient steps.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # full convolution and location of the current maximum
        conv = full_convolution(a)
        max_val = conv.max()
        k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a:
        # g_i = 2 * a_{k-i} for indices where k-i is inside [0, n-1]
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # vectorised slice assignment
        i_range = np.arange(lo, hi + 1)
        g[i_range] = 2.0 * a[k - i_range]

        # step size = init_lr / sqrt(iter)
        step = init_lr / np.sqrt(it)
        a = project_simplex(a - step * g)

        # occasional exact evaluation
        if it % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                # print progress
                print(f"[subgrad] iter {it:6d}  score {val:.6f}")

        # optional learning‑rate floor
        if init_lr > min_lr and it % 1000 == 0:
            init_lr = max(min_lr, init_lr * 0.9)

    # final check
    final_val = compute_score(a)
    if final_val < best_val:
        best_a = a.copy()
        best_val = final_val
        print(f"[subgrad] final improvement {final_val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling)

        if it % 4000 == 0:
            # occasional jitter to escape shallow plateaux
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

    # -----------------------------------------------------------------

    candidates = []  # list of (score, vector)

            return

        candidates.append((sc, vec.copy()))

        if time.time() >= deadline:

    # exponential

        if time.time() >= deadline:

    # Gaussian family

    # keep the best few vectors for polishing
