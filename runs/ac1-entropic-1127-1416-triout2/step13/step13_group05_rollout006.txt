def anneal(a_start, deadline):
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = len(a)
    temp0 = 0.1
    it=0
    while time.time()<deadline:
        # temperature schedule
        t_frac = it / 1e6  # or compute based on time left
        T = max(1e-5, temp0 * (1 - t_frac))
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        # propose move: shift a fraction of mass from i to j controlled by T
        delta = T * a[i] * np.random.rand()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # enforce non-negativity
        cand = project_to_simplex(cand)
        M_cand = full_convolution(cand).max()
        if M_cand < best_M or np.random.rand() < np.exp(-(M_cand - best_M)/T):
            a = cand
            if M_cand < best_M:
                best_M = M_cand
                best_a = cand.copy()
        it+=1
    return best_a

def adam_optimize(a0, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    best_a=a.copy()
    best_M=full_convolution(a).max()
    while time.time()<deadline:
        t+=1
        conv=full_convolution(a)
        M=conv.max()
        active = np.where(conv >= M - 1e-12)[0]
        g = np.zeros_like(a)
        n=len(a)
        for k in active:
            # similar gradient compute:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2*a[j[mask]]
        # Adam updates:
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = 0.1 / (np.sqrt(v_hat) + eps)   # adjusting per-coordinate
        a = a - step * m_hat
        a = project_to_simplex(a)
        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    return best_a

"""
search_step_function_v3.py

A self‑contained optimiser for the ``evaluate_sequence`` objective.
Improvements over the previous version:

*   Adds several new parametric families (Dirichlet, extreme beta,
    combined reverse‑power‑law/geometric mixtures) and a finer sweep.
*   Uses an Adam‑style projected sub‑gradient optimiser (faster
    convergence than simple sub‑gradient descent).
*   Finishes with a simulated‑annealing hill‑climb that can accept
    occasional worsening moves – this helps escape local minima.
*   All phases respect the global time budget (default 1000 s) and
    print intermediate scores for monitoring.
"""

import time

# ----------------------------------------------------------------------
# Basic utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # if all entries are <=0 return uniform distribution
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a via FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for efficient FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The scaling is irrelevant because the official routine also
    normalises by the sum.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)ᵖ, p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p)."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ qⁱ, 0 < q < 1."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Front plateau of length L followed by a decreasing power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ xᵅ (1‑x)ᵝ,   x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def dirichlet_shape(n: int, concentration: float) -> np.ndarray:
    """Sample a Dirichlet distribution with equal concentration."""
    # concentration > 0; very small values give spiky vectors,
    # large values approach the uniform distribution.
    a = np.random.dirichlet(np.full(n, concentration))
    return a

# Adam‑style projected sub‑gradient optimiser
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray, time_deadline: float,
                  base_lr: float = 0.12,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """
    Projected Adam optimiser on the nondifferentiable objective
    f(a) = max(conv(a,a)).  The algorithm works on the sub‑gradient
    (sum of shifted copies of a for all active convolution indices).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # active indices (within a tiny tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient: g_i = 2 * sum_{k in active} a_{k-i}
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # Adam moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)

        # bias‑corrected moments
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # per‑coordinate step
        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = project_to_simplex(a)

        # optional jitter every few thousand steps (helps escape plateaus)
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (base_lr * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# Simulated‑annealing hill‑climb (accepts occasional worsening moves)
# ----------------------------------------------------------------------
def annealing_hill_climb(a_start: np.ndarray, time_deadline: float,
                         init_temp: float = 0.04,
                         decay: float = 0.99996,
                         min_temp: float = 1e-5) -> np.ndarray:
    """
    Greedy stochastic moves with a simple temperature schedule.
    Moves transfer a fraction of probability mass from one index to another.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    T = init_temp
    it = 0

    while time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose moving a small proportion of mass from i to j
        # the magnitude scales with current temperature
        delta = T * a[i] * np.random.rand()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()

        # always accept if it improves, otherwise accept with probability exp(-Δ/T)
        if M_cand < best_M or np.random.rand() < math.exp(-(M_cand - best_M) / max(T, 1e-12)):
            a = cand
            if M_cand < best_M:
                best_M = M_cand
                best_a = cand.copy()

        # temperature decay
        T = max(min_temp, T * decay)
        it += 1

        # occasional restart if we have been stagnant for a long time
        if it % 120000 == 0:
            # re‑inject a tiny random perturbation around the best so far
            a = best_a + np.random.randn(n) * 0.001
            a = project_to_simplex(a)

    return best_a
