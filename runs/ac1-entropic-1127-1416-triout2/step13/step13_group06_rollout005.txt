Algorithm

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# ----------------------------------------------------------------------
# Analytic families (used for cheap initial scan)
# ----------------------------------------------------------------------


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect from infinities
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()

def smooth_max_gradient_descent(start_a: np.ndarray,
                                deadline: float,
                                max_iters: int = 20000,
                                eta0: float = 0.6,
                                tau0: float = 0.1,
                                tau_max: float = 8.0) -> Tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using a temperature‑annealed
    log‑sum‑exp surrogate for the max‑convolution.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    # pre‑compute index array for the cross‑correlation extraction
    idxs = np.arange(n)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear annealing of temperature
        tau = tau0 + (tau_max - tau0) * (t / max_iters)
        # full self‑convolution
        conv = convolution_fft(a, a)                # length 2n‑1
        max_conv = conv.max()
        # soft‑max weights (numerically stable)
        exp_shift = np.exp(tau * (conv - max_conv))
        sum_exp = exp_shift.sum()
        p = exp_shift / sum_exp                     # length 2n‑1
        # log‑sum‑exp approximation of the max
        LSE = max_conv + np.log(sum_exp) / tau

        # cross‑correlation (p ⋆ a)
        conv_rev = convolution_fft(a, p[::-1])     # length 3n‑2
        cross_corr = conv_rev[2 * n - 2 - idxs]    # vector of length n

        # gradient of the smooth objective (ignoring constant factor)
        grad = cross_corr - LSE      # since sum(a)=1 after each renorm

        # exponentiated‑gradient step with diminishing stepsize
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex (KL‑projection is simple scaling)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional true‑objective evaluation
        if t % 200 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] smooth‑grad iter {t} improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] smooth‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000,
                  temp_factor: float = 1.5e-3,
                  step_scale_init: float = 0.2) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex (swap / transfer moves).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature proportional to current score
    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        # move type: swap two entries or transfer a fraction
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 5.0

    best_seq: List[float] = []
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback to global best (if present)
    # ------------------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a working discretisation (n = 4096 gives a good trade‑off)
    # ------------------------------------------------------------------
    n = 4096
    print(f"[{time.time():.1f}] working size n = {n}")

    # ------------------------------------------------------------------
    # 2. Very cheap scan of analytic families (centre‑heavy, left‑heavy)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] starting cheap family scan")
    alphas = np.linspace(0.0, 12.0, 61)
    for alpha in alphas:
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_center_power_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] new best centre‑α={alpha:.3f} → {best_score:.6f}")

    betas = np.linspace(0.0, 12.0, 61)
    for beta in betas:
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] new best left‑β={beta:.3f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Random Dirichlet seeds (adds asymmetry)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] random Dirichlet exploration")
    for _ in range(200):
        if time.time() >= deadline - 30.0:
            break
        vec = np.random.exponential(1.0, size=n)
        vec /= vec.sum()
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] random Dirichlet improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max mirror descent refinement
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] smooth‑max gradient descent (phase 1)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = smooth_max_gradient_descent(
            start_a=a_start,
            deadline=deadline - 15.0,
            max_iters=20000,
            eta0=0.7,
            tau0=0.05,
            tau_max=10.0,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after smooth‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional second‑phase: classic exponentiated‑gradient on true max
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 10.0:
        # reuse the earlier fast exp‑grad for a few thousand iterations
        from math import sqrt

        def exp_grad(start_a: np.ndarray,
                     deadline2: float,
                     max_iters2: int = 80000,
                     eta0_2: float = 0.5) -> Tuple[np.ndarray, float]:
            n2 = start_a.size
            a2 = start_a.copy()
            best_a2 = a2.copy()
            best_sc2 = compute_score(a2)
            for t in range(1, max_iters2 + 1):
                if time.time() >= deadline2:
                    break
                conv2 = convolution_fft(a2, a2)
                k = int(np.argmax(conv2))
                idx = np.arange(n2)
                j = k - idx
                mask = (j >= 0) & (j < n2)
                g = np.zeros_like(a2)
                g[mask] = 2.0 * a2[j[mask]]
                eta2 = eta0_2 / sqrt(t)
                a2 = a2 * np.exp(-eta2 * g)
                s2 = a2.sum()
                if s2 == 0:
                    a2 = np.full_like(a2, 1.0 / n2)
                else:
                    a2 = a2 / s2
                if t % 2000 == 0:
                    sc2 = compute_score(a2)
                    if sc2 < best_sc2:
                        best_sc2, best_a2 = sc2, a2.copy()
                        print(f"[{time.time():.1f}] exp‑grad phase2 improved to {best_sc2:.6f}")
            final_sc = compute_score(a2)
            if final_sc < best_sc2:
                best_sc2, best_a2 = final_sc, a2.copy()
                print(f"[{time.time():.1f}] exp‑grad phase2 final improvement to {best_sc2:.6f}")
            return best_a2, best_sc2

        a_start2 = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = exp_grad(
            start_a=a_start2,
            deadline2=deadline - 6.0,
            max_iters2=80000,
            eta0_2=0.5,
        )
        if sc_opt2 < best_score:
            best_score, best_seq = sc_opt2, a_opt2.tolist()
            print(f"[{time.time():.1f}] after second exp‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=5_000_000,
            temp_factor=1.5e-3,
            step_scale_init=0.2,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Fallback to uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if not best_seq:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
