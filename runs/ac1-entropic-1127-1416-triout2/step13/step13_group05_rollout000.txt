"""
Step‑function optimiser for ``evaluate_sequence``.

The objective (after fixing Σa_i = 1) is
        f(a) = 2·n·max(conv(a,a))

where conv(a,a) is the full linear convolution.
The code proceeds in three phases:

1. **Fast parametric pre‑search** – a fine sweep over the
   reverse power‑law family a_i ∝ (n‑i)^p, which is known to give
   good starting points.

2. **Projected gradient descent with a smooth approximation of the
   max** – we replace max by a soft‑max
        g_λ(c) = (1/λ)·log Σ exp(λ·c_i)
   (λ grows during the run).  The gradient of the weighted sum
   Σ w_i·c_i (with w_i = exp(λ·c_i)/Σexp) can be evaluated
   efficiently by a single FFT:
        ∇ = 2·ifft( conj(FFT(w)) * FFT(a) )
   After each step the vector is projected onto the probability
   simplex (non‑negative, Σ=1).

3. **Final stochastic hill‑climb** – a cheap greedy local search
   that swaps a small amount of mass between two coordinates.
   This often pushes the solution a little further.

All helper routines are top‑level, no closures are used.
The routine respects the 1000 s time budget and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}, front‑loaded decreasing power‑law (p ≥ 0)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    a /= total
    return a

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT.
    Length of result = 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def gradient_weighted_sum(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i·(a*a)_i w.r.t a.
    Uses the identity ∇ = 2·ifft( conj(FFT(w)) * FFT(a) )
    (linear convolution, zero‑padded to avoid circular effects).
    Returns the first len(a) components.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2*n-1
    N = n + m - 1                       # length needed for linear convolution
    N_fft = 1 << ((N - 1).bit_length())  # next power of two ≥ N
    A = np.fft.fft(a, N_fft)
    W = np.fft.fft(w, N_fft)
    grad_full = 2 * np.fft.ifft(np.conj(W) * A).real
    return grad_full[:n]

# ----------------------------------------------------------------------
# Objective (sum fixed to 1, so denominator = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Fast reverse‑power‑law sweep – gives a good deterministic start
# ----------------------------------------------------------------------
def best_reverse_power_law(n: int, deadline: float) -> np.ndarray:
    """Return the reverse‑power‑law vector with smallest compute_score
       found before *deadline* (coarse + fine sweep)."""
    best_vec = None
    best_val = float('inf')

    # coarse sweep
    for p in np.linspace(0.5, 4.0, 71):      # step ≈0.05
        if time.time() >= deadline:
            break
        a = reverse_power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # fine sweep around the best coarse p (if any)
    if best_vec is not None:
        # retrieve the approximate p of the best vector
        # (we don't store it, just sweep a tight window)
        for p in np.linspace(0.8, 2.5, 301):   # step ≈0.0057
            if time.time() >= deadline:
                break
            a = reverse_power_law(n, p)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a
    return best_vec

# ----------------------------------------------------------------------
# Stochastic hill‑climb (same as reference but a bit tighter)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 600_000) -> np.ndarray:
    """Greedy stochastic redistribution of mass with projection onto simplex."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 120_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0
    return best_a
