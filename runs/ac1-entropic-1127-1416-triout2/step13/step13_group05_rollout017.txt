def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT (real inputs)."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Front plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a

# Smoothed loss (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def smoothed_loss_and_grad(a: np.ndarray, lam: float):
    """
    Returns (loss, grad) where
        loss = (1/lam) * log sum_i exp(lam * conv_i)
    and grad = ∂loss/∂a .
    The gradient is computed via FFT for efficiency.
    """
    # convolution a*a (full)
    conv = full_convolution(a)                     # length 2n‑1
    max_c = conv.max()
    # softmax over the convolution entries (stable)
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = exp_term.sum()
    s = exp_term / sum_exp                          # length 2n‑1, sums to 1

    # gradient = 2 * (a_rev * s) evaluated on the n positions
    a_rev = a[::-1]
    full = conv_fft(a_rev, s)                     # length 3n‑2
    grad = 2.0 * full[len(a) - 1: len(a) - 1 + len(a)]

    # loss value (the smooth surrogate of max)
    loss = max_c + np.log(sum_exp) / lam
    return loss, grad

    """
    a = a0.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    for lam in lambda_schedule:
        for _ in range(iters_per_stage):
            if time.time() >= time_deadline:
                return a
            t += 1
            loss, grad = smoothed_loss_and_grad(a, lam)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a -= step0 * m_hat / (np.sqrt(v_hat) + eps)

            # keep on the simplex
            a = project_to_simplex(a)

    return a


# ----------------------------------------------------------------------
# Simple hill‑climb (aggressive final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               time_deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 120_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # keep a safety margin

    best_seq = None
    best_val = float('inf')
    n = 2048

    # -----------------------------------------------------------------
    # 0. optional fallback from previous runs (if present)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. parametric sweep (≈ 250 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 250.0
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = evaluate_sequence(a_initial.tolist())
            print(f"[parametric sweep] official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Adam smoothing optimisation (the new core)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    if best_seq is not None and time.time() < smooth_deadline:
        a0 = np.array(best_seq, dtype=float)
        # λ schedule – start soft, end close to true max
        lambda_sched = [2.0, 8.0, 30.0, 120.0, 500.0, 2000.0]
        a_opt = adam_optimize_on_simplex(
            a0,
            smooth_deadline,
            lambda_schedule=lambda_sched,
            iters_per_stage=500,
            step0=0.12,
        )
        val = evaluate_sequence(a_opt.tolist())
        print(f"[adam] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()

    # -----------------------------------------------------------------
    # 3. Random restarts + brief Adam refinements
    # -----------------------------------------------------------------
    restart_deadline = deadline - 120.0
    restarts = 5
    while restarts > 0 and time.time() < restart_deadline:
        restarts -= 1
        # Dirichlet (alpha=0.7 gives a bit more front‑loading)
        a_rand = np.random.dirichlet(np.full(n, 0.7))
        a_refined = adam_optimize_on_simplex(
            a_rand,
            restart_deadline,
            lambda_schedule=[5.0, 50.0, 300.0, 1500.0],
            iters_per_stage=300,
            step0=0.14,
        )
        val = evaluate_sequence(a_refined.tolist())
        print(f"[restart] candidate score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Final hill‑climb polish (bulk of remaining time)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 5. Safeguard fallback (very small reverse‑power‑law)
    # -----------------------------------------------------------------
    if best_seq is None:
        a_fallback = reverse_power_law(n, 1.8)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
