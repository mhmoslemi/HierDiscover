"""
Search for a non‑negative step‑function (vector) that minimises
``evaluate_sequence``.
The objective (after normalising the vector to sum = 1) is
    2·n·max(conv(a,a)),
where ``conv`` is the full linear convolution.
The programme follows three phases:

1. **Broad candidate generation** – several parametric families
   (beta, power‑law, inverse‑product, edge‑heavy) and random Dirichlet
   draws.  The ten best vectors (by the exact score) are kept.

2. **Linear‑programming refinement** – for a fixed vector *f* we
   maximise the ℓ₁‑mass of a new vector *g* under the linear
   constraints ``conv(f,g) ≤ M`` where ``M = max(conv(f,f))``.
   The optimal *g* is normalised to sum = 1 and mixed with *f*:
   ``f ← (1‑t)·f + t·g`` (projected onto the simplex).  Repeating this
   step (≈ 20 iterations) monotonically reduces the maximum
   autocorrelation.  The LP uses a sparse “convolution matrix’’ that
   is built once and re‑used.

3. **Sub‑gradient polishing** – a classic projected sub‑gradient descent
   on the true non‑smooth objective finishes the optimisation.

All steps respect the 1000 s wall‑clock budget.  The best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def generate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_edge_powerlaw(n: int, p: float) -> np.ndarray:
    """Symmetric edge‑heavy shape: a_i ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p) + (n - i + 1) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_inverse_product(n: int, p: float) -> np.ndarray:
    """Inverse‑product shape: a_i ∝ ((i+1)*(n‑i))^{‑p}."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i * (n - i + 1)) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_random_dirichlet(n: int, conc: float) -> np.ndarray:
    return np.random.dirichlet([conc] * n)


# ----------------------------------------------------------------------
# Convolution matrix (row/col pattern) – reused for every LP step
# ----------------------------------------------------------------------


def build_conv_pattern(n: int):
    """Return row indices, column indices for the n×n convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


# ----------------------------------------------------------------------
# Linear‑programming refinement of a single vector
# ----------------------------------------------------------------------


def lp_refine(f: np.ndarray, pattern, max_iters: int, deadline: float):
    """Iteratively improve f using the LP step described in the paper."""
    rows, cols = pattern
    n = f.shape[0]
    m = 2 * n - 1

    best_f = f.copy()
    best_score = compute_score(best_f)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # current bound
        conv = full_convolution(best_f)
        M = conv.max()

        # build sparse matrix A (weights = f_i for each occurrence)
        data = best_f[cols]
        A = sparse.csr_matrix((data, (rows, cols)), shape=(m, n))

        # LP: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(m, M, dtype=np.float64)
        bounds = [(0, None)] * n

        res = linprog(
            c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options={"presolve": True}
        )
        if not res.success or res.x is None:
            # LP failed – abort refinement
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            break

        # normalise to sum = 1 (simplex)
        g = g0 / sum_g0

        # mixing coefficient – start aggressive, then decay
        tau = max(0.1, 0.5 / np.sqrt(t))

        new_f = (1.0 - tau) * best_f + tau * g
        new_f = project_to_simplex(new_f)

        new_score = compute_score(new_f)
        if new_score < best_score:
            best_score, best_f = new_score, new_f

        # early exit if improvement negligible
        if best_score / new_score < 1.000001:
            # not improving much
            if time.time() > deadline - 5.0:
                break

    return best_f, best_score


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as used previously)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a
