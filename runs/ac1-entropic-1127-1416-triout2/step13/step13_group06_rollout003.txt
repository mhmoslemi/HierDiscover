"""
Propose a step‑function sequence with a lower upper bound.

Algorithm overview
------------------
1. **Coarse analytic search (n_small = 8192)** – Scan several one‑parameter
   families (center‑heavy power, Gaussian, triangular, product, beta) and a
   dense random search.  Keep the best candidate.
2. **Local refinement (exponentiated‑gradient)** – Mirror‑descent on the
   simplex (KL‑projection) for a few 10⁵ iterations.
3. **Upsample** the current best vector to a larger discretisation
   (n_big = 16384) by linear interpolation and renormalise.
4. **Smooth‑max Adam descent** – Minimise a smooth approximation of the
   ``max`` of the self‑convolution using an Adam optimiser and a
   temperature‑like parameter β that is increased geometrically.
5. **Optional random Dirichlet trials** – Sample a few random vectors on the
   large grid and keep any improvement.
6. **Final simulated‑annealing hill‑climb** – A short SA run on the large
   grid to escape possible local minima.
7. Return the best sequence after a final call to ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²).  Assumes sum(a)>0."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s

def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred on the middle of the interval."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """Symmetric triangular bump."""
    half_width = int(max(1, width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return w / w.sum()


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / s
    return w

def exp_gradient_descent(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         eta0: float = 0.6) -> Tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using exponentiated gradients.
    Uses sub‑gradient at the index where the convolution attains its maximum.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] exp‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] exp‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def smooth_max_adam_descent(start_a: np.ndarray,
                           deadline: float,
                           max_iters: int = 4000,
                           alpha: float = 0.05,
                           beta_start: float = 5.0,
                           beta_end: float = 2000.0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of the max‑convolution.
    The smooth function is  S(a)=log∑_k exp(β·conv_k)  (β increases over time).
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = beta_start
    if max_iters > 0:
        beta_factor = (beta_end / beta_start) ** (1.0 / max_iters)
    else:
        beta_factor = 1.0

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_conv = conv.max()
        shifted = conv - max_conv                # ≤ 0
        # weights for the smooth max (softmax over conv values)
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth max: g_i = 2·∑_j a_j·w_{i+j}
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        mlen = w.size
        idx = mlen - 1 - np.arange(n)
        g = 2.0 * conv_rev[idx]

        # Adam update
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        step = alpha / (np.sqrt(v_hat) + 1e-8)
        a = a - step * m_hat

        # keep a on the simplex
        a = project_to_simplex(a)

        # increase β slowly
        beta *= beta_factor

        if t % 200 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] smooth‑Adam iter {t} best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] smooth‑Adam final improvement = {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000,
                  temp_factor: float = 1.2e-3,
                  step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex (same code as before,
    but with a reduced step budget for the large grid).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def upsample_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation up‑sampling to a larger length."""
    old_n = v.size
    if new_n == old_n:
        return v.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    v_big = np.interp(new_x, old_x, v)
    return normalize(v_big)


def random_dirichlet_like(n: int, concentration: float) -> np.ndarray:
    """Generate a random point on the simplex (Gamma/Dirichlet sampler)."""
    # For concentration <= 0 the distribution becomes very sparse,
    # for large concentration it approaches uniform.
    if concentration <= 0.0:
        concentration = 1e-3
    g = np.random.gamma(concentration, 1.0, size=n)
    return normalize(g)

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin for final eval

    best_seq: List[float] | None = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Optional fallback to the previous best construction
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Coarse analytic search on a modest grid (n_small = 8192)
    # -----------------------------------------------------------
    n_small = 8192
    print(f"[{time.time():.1f}] starting coarse family scan (n={n_small})")

    # Helper to update the global best
    def consider(vec: np.ndarray, src: str):
        nonlocal best_score, best_seq
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best via {src} = {best_score:.6f}")

    # center‑heavy power
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        consider(symmetric_center_power_family(n_small, alpha), f"center_power α={alpha:.2f}")

    # Gaussian centre
    for sigma in np.linspace(0.2 * n_small, 0.5 * n_small, 30):
        if time.time() >= deadline - 30.0:
            break
        consider(gaussian_center_family(n_small, sigma), f"gaussian σ={sigma:.0f}")

    # Triangular centre
    for width in np.linspace(0.1, 0.9, 33):
        if time.time() >= deadline - 30.0:
            break
        consider(triangular_center_family(n_small, width), f"triangular w={width:.2f}")

    # Product (U‑shaped) family
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        consider(symmetric_product_family(n_small, alpha), f"product α={alpha:.2f}")

    # Beta family (random two‑parameter draws)
    for _ in range(80):
        if time.time() >= deadline - 30.0:
            break
        p = np.random.uniform(0.2, 3.0)
        q = np.random.uniform(0.2, 3.0)
        consider(beta_family(n_small, p, q), f"beta p={p:.2f},q={q:.2f}")

    # -----------------------------------------------------------
    # 2. Random dense search over families (≈8000 evaluations)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] random dense search (n={n_small})")
    trials = 0
    max_trials = 8000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(7)
        if choice == 0:
            vec = symmetric_center_power_family(n_small, np.random.uniform(0.0, 12.0))
            src = "rnd_center_power"
        elif choice == 1:
            vec = gaussian_center_family(n_small, np.random.uniform(0.2 * n_small, 0.5 * n_small))
            src = "rnd_gaussian"
        elif choice == 2:
            vec = triangular_center_family(n_small, np.random.uniform(0.1, 0.9))
            src = "rnd_triangular"
        elif choice == 3:
            vec = symmetric_product_family(n_small, np.random.uniform(0.0, 12.0))
            src = "rnd_product"
        elif choice == 4:
            p = np.random.uniform(0.2, 3.0)
            q = np.random.uniform(0.2, 3.0)
            vec = beta_family(n_small, p, q)
            src = "rnd_beta"
        elif choice == 5:
            vec = np.random.exponential(1.0, size=n_small)
            vec = normalize(vec)
            src = "rnd_exponential"
        else:
            # random uniform + Gaussian mixture
            lam = np.random.rand()
            sigma = np.random.uniform(0.2 * n_small, 0.5 * n_small)
            vec = lam * np.full(n_small, 1.0 / n_small) + (1.0 - lam) * gaussian_center_family(n_small, sigma)
            vec = normalize(vec)
            src = "rnd_mixture"

        # occasional zero‑padding shift
        if np.random.rand() < 0.2:
            d = np.random.randint(0, n_small // 30 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        consider(vec, src)

    # -----------------------------------------------------------
    # 3. Mirror‑descent (exponentiated‑gradient) refinement
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 260.0:
        print(f"[{time.time():.1f}] exponentiated‑gradient refinement (n={n_small})")
        start_a = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = exp_gradient_descent(
            start_a=start_a,
            deadline=deadline - 150.0,
            max_iters=250_000,
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Upsample to a larger grid for finer optimisation (n_big = 16384)
    # -----------------------------------------------------------
    n_big = 16384
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] up‑sampling from {n_small} to {n_big}")
        a_big = upsample_vector(np.array(best_seq, dtype=np.float64), n_big)
    else:
        # fallback uniform if we have nothing yet
        a_big = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    # -----------------------------------------------------------
    # 5. Smooth‑max Adam descent on the large grid
    # -----------------------------------------------------------
    if time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] smooth‑max Adam descent (n={n_big})")
        a_smooth, sc_smooth = smooth_max_adam_descent(
            start_a=a_big,
            deadline=deadline - 30.0,
            max_iters=4000,
            alpha=0.05,
            beta_start=5.0,
            beta_end=2000.0,
        )
        if sc_smooth < best_score:
            best_score, best_seq = sc_smooth, a_smooth.tolist()
            print(f"[{time.time():.1f}] smooth‑Adam gave new best = {best_score:.6f}")
        else:
            a_big = a_smooth  # keep for possible SA stage

    # -----------------------------------------------------------
    # 6. Optional random Dirichlet trials on the large grid
    # -----------------------------------------------------------
    if time.time() < deadline - 40.0:
        print(f"[{time.time():.1f}] random Dirichlet trials (n={n_big})")
        trials_big = 0
        while time.time() < deadline - 30.0 and trials_big < 300:
            trials_big += 1
            conc = np.random.uniform(0.05, 2.0)   # vary sparsity
            vec = random_dirichlet_like(n_big, conc)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] Dirichlet trial #{trials_big} improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Final simulated‑annealing hill‑climb (high temperature)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb (n={n_big})")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=5_000_000,
            temp_factor=1.2e-3,
            step_scale_init=0.25,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 8. Fallback to a safe uniform vector if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
