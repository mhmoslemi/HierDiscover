"""
Search for a non‑negative step height vector minimizing the objective
``evaluate_sequence`` (equivalently `2 * n * max(conv(a,a))` after normalising to a
probability simplex).  The approach consists of

* a large, cheap random / parametric exploration phase that keeps the
  best few candidates,
* a projected Adam optimisation on a smooth log‑sum‑exp surrogate of the
  max‑convolution,
* a short sub‑gradient polishing phase,
* a final simulated‑annealing hill‑climb.

All phases respect the 1000 s wall‑clock budget.
"""
import time
import numpy as np

# ----- basic utilities -------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x >=0, sum(x)=1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear full convolution a*a using a real FFT.
    Length of result = 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def smooth_max(conv: np.ndarray, tau: float):
    """τ·log‑sum‑exp approximation of max(conv).  Returns (S, w)."""
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w

def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k(a,a) w.r.t. a.
    Returns g_i = 2·Σ_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    m = w.shape[0]          # = 2*n-1
    rev_w = w[::-1]
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]

def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α−1}·(n−i)^{β−1}."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(−(i−mid)²/(2σ²))."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin^p (π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_ushape(n: int, p: float) -> np.ndarray:
    """U‑shape a_i ∝ ((i+1)*(n−i))^{p}."""
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1 / sqrt( (i+1)*(n-i) + eps )."""
    eps = 1e-6
    i = np.arange(n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(i * (n - i + 0.5) + eps)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_cosine_u(n: int, p: float) -> np.ndarray:
    """a_i ∝ (1+cos(π·i/(n-1)))^{p} (U‑shape, p>=0)."""
    i = np.arange(n, dtype=float)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

    """Sparse periodic pattern: weight 1 on indices that are multiples of step."""
    a = np.zeros(n, dtype=float)
    a[::step] = 1.0

def explore_candidates(n: int, deadline: float, keep: int = 20) -> list[np.ndarray]:
    """Generate a large pool of vectors and keep the ``keep`` best ones."""
    best = []   # list of (score, vector)

    def consider(v: np.ndarray):
        if not np.isfinite(v).all():
            return
        v = np.maximum(v, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        if len(best) < keep:
            best.append((sc, v.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, v.copy())
            best.sort(key=lambda x: x[0])

    # 1) Uniform baseline
    consider(np.full(n, 1.0 / n))

    # 2) Beta shapes – enlarge range, especially near zero
    alphas = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    betas = alphas
    for a in alphas:
        for b in betas:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    # 3) U‑shape with stronger exponents
    for p in np.linspace(0.1, 12.0, 24):
        if time.time() >= deadline:
            break
        consider(generate_ushape(n, p))

    # 4) Power‑law (very steep)
    for p in np.linspace(0.0, 12.0, 25):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))
        consider(generate_powerlaw(n, -p))

    # 5) Exponential sweep – larger magnitude
    for lam in np.linspace(-1.0, 1.0, 41):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # 6) Sine‑power (high frequency)
    for p in np.linspace(0.5, 30.0, 20):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # 7) Inverse‑sqrt edge‑heavy shape
    if time.time() < deadline:
        consider(generate_inverse_sqrt(n))

    # 8) Cosine‑U family
    for p in np.linspace(0.5, 10.0, 15):
        if time.time() >= deadline:
            break
        consider(generate_cosine_u(n, p))

    # 9) Periodic sparse patterns
    for step in range(2, min(200, n // 5), 10):
        if time.time() >= deadline:
            break
        consider(generate_periodic(n, step))

    # 10) Random Dirichlet draws with a wide range of concentrations
    #   concentration parameter drawn log‑uniformly from 1e‑3 … 10
    while time.time() < deadline:
        conc = 10 ** np.random.uniform(-3, 1)   # 0.001 … 10
        # use symmetric Dirichlet (same α for all components)
        g = np.random.gamma(conc, 1.0, size=n)
        consider(g)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]

def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 0.3,
                  tau_end: float = 1e-8,
                  tau_decay: float = 0.88,
                  lr: float = 0.07,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """
    Projected Adam minimising a smooth log‑sum‑exp surrogate of max(conv).
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        # anneal temperature
        if t % 150 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape flat regions
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (lr * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a

def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient corresponding to all active indices
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        cooling: float = 0.9998,
                        step_scale0: float = 0.06,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Simple Metropolis hill‑climb on the true objective."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale0
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling)

        # occasional global jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.008)
            a = a + jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

``evaluate_sequence`` (equivalently 2·n·max(conv(a,a)) after normalising to a
probability simplex).  The algorithm consists of

1. a large, cheap random / parametric exploration phase that keeps the
   best few candidates,
2. projected Adam optimisation on a smooth log‑sum‑exp surrogate of the
   max‑convolution,
3. a short sub‑gradient polishing phase,
4. a final simulated‑annealing hill‑climb.

All phases respect the 1000 s wall‑clock budget.

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution ``a*a`` using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def smooth_max(conv: np.ndarray, tau: float):
    """τ·log‑sum‑exp approximation of ``max(conv)``; returns (S, w)."""
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k(a,a) w.r.t. ``a``.
    Returns g_i = 2·Σ_k w_k·a_{k−i}.
    """
    n = a.shape[0]
    m = w.shape[0]                         # = 2·n−1
    rev_w = w[::-1]
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]

        _, w = smooth_max(conv, tau)
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (lr * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (true objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on the exact L∞‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final refinement)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        cooling: float = 0.9998,
                        step_scale0: float = 0.06,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale0
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.008)
            a = a + jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # a slightly smaller length gives faster iterations while preserving the
    # theoretical lower bound (2·n/(2·n−1) ≈ 1); 1024 works well.
    n = 1024

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. quick fallback if a known construction exists
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. broad exploration (≈ 250 s)
    # --------------------------------------------------------------
    explore_deadline = deadline - 550.0
    if time.time() < explore_deadline:
        print("[explore] generating candidates …")
        cand_vecs = explore_candidates(n, explore_deadline, keep=30)
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_score:
                best_score, best_seq = val, vec.tolist()
        print(f"[explore] best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Adam optimisation from elite points (≈ 300 s)
    # --------------------------------------------------------------
    if best_seq is not None:
        adam_deadline = min(deadline - 200.0,
                            start + (deadline - start) * 0.45)
        elite = [np.array(best_seq, dtype=float)]
        elite.append(np.full(n, 1.0 / n))
        for _ in range(4):
            elite.append(project_to_simplex(np.random.rand(n)))
        for vec in cand_vecs[:5]:
            elite.append(vec)

        refined_best = elite[0].copy()
        refined_val = best_score

        for idx, a0 in enumerate(elite):
            if time.time() >= adam_deadline:
                break
            local_deadline = adam_deadline - (time.time() - start) - 0.5
            if local_deadline <= time.time():
                break
            print(f"[adam {idx}] start")
            a_opt = adam_optimize(a0, local_deadline)
            val = compute_score(a_opt)
            print(f"[adam {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_opt.copy()

        best_score, best_seq = refined_val, refined_best.tolist()
        print(f"[adam] best after Adam = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 150 s)
    # --------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[polish] start")
        a_polished = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # --------------------------------------------------------------
    # 4. Simulated annealing (final refinement, ≈ 100 s)
    # --------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[anneal] start")
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_final.tolist()

    # --------------------------------------------------------------
    # safeguard – uniform vector if nothing has been found
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # final official evaluation
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
