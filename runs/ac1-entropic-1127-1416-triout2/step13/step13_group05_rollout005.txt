def simulated_annealing_optimize(a_init, deadline):
    a = a_init.copy()
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)
    T0 = 0.01  # initial temperature (maybe relative to objective)
    t0 = time.time()
    # schedule: T = T0 * (1 - (elapsed / total))  # linear decay
    while time.time() < deadline:
        elapsed = time.time() - t0
        total = deadline - t0
        T = T0 * max(0.001, 1 - elapsed/total)  # temperature decays
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # choose delta between -a[i] and a[i]
        delta = np.random.uniform(0, min(a[i], 0.5*a[i]))  # move fraction
        # transfer from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        # compute score
        score = compute_score(cand)
        diff = score - best_score
        if diff < 0 or np.random.rand() < np.exp(-diff / T):
            a = cand
            if score < best_score:
                best_score = score
                best = cand.copy()
        # occasionally reheat? maybe after some iterations assign a=best
        if np.random.rand() < 0.001:
            a = best.copy()
    return best

    return a

"""
search_step_function_v3.py

A fresh optimiser for the ``evaluate_sequence`` objective.

Strategy
--------
1.  **Coarse / fine parametric sweep** over the *reverse‑power‑law*
    family a_i ∝ (n−i)^p (the single most promising shape discovered
    so far) together with a few random convex combinations of basic
    shapes (uniform, geometric, reverse‑power‑law). The best vector of
    this stage seeds the continuous optimiser.

2.  **Smooth‑max Adam optimisation** – minimise a differentiable
    approximation of the max‑convolution (soft‑max) using Adam.  The
    temperature‑parameter β starts small (smooth) and is increased
    linearly to a large value so that the optimiser finally focuses on
    the true peak of the convolution.

3.  **Projected sub‑gradient polishing** – a classic sub‑gradient
    descent on the exact max‑convolution with a decaying step‑size.

4.  **Stochastic hill‑climb** – greedy mass‑transfer moves with an
    adaptive step‑scale to fine‑tune the last few digits.

All stages respect the simplex constraint Σ a_i = 1, a_i ≥ 0, and the
total running time is bounded by the supplied budget (≈ 1000 s).  The
final sequence (a plain Python list) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # tie‑break: uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    # Numerical noise may create tiny negatives – clamp safely
    conv = np.where(conv < 0, 0.0, conv)
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (n-i)^p   (front‑loaded decreasing power‑law).
    p ≥ 0.
    """
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a

def parametric_sweep(n: int, time_deadline: float) -> np.ndarray:
    """Broad sweep over reverse‑power‑law and a few random mixes."""
    best_vec = None
    best_val = float('inf')
    best_p = 1.0                          # placeholder for fine grid

    # ---- coarse reverse‑power‑law -------------------------------------------------
    for p in np.linspace(0.5, 4.0, 71):   # step ≈0.05
        if time.time() >= time_deadline:
            return best_vec
        a = reverse_power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec, best_p = val, a, p

    # ---- fine grid around the best coarse p ---------------------------------------
    if best_vec is not None:
        lo = max(0.3, best_p - 0.6)
        hi = min(5.0, best_p + 0.6)
        for p in np.linspace(lo, hi, 141):   # step ≈0.01
            if time.time() >= time_deadline:
                break
            a = reverse_power_law(n, p)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec, best_p = val, a, p

    # ---- random convex combinations of a few basic shapes -------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    pw2 = reverse_power_law(n, 2.0)
    geom = geometric(n, 0.98)

    for _ in range(2000):

        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec

# 2. Smooth‑max Adam optimisation
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray, time_deadline: float,
                  beta_start: float = 5.0,
                  beta_end: float = 5000.0,
                  lr0: float = 0.02) -> np.ndarray:
    """
    Adam optimiser on the smooth approximation
        f_β(a) = (1/β) * log Σ exp(β·conv(a,a)) .
    β grows linearly from ``beta_start`` to ``beta_end``.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    start = time.time()
    total = max(time_deadline - start, 1e-6)

    while time.time() < time_deadline:
        t += 1
        # current temperature for the smooth max
        elapsed = time.time() - start
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        conv = full_convolution(a)                     # length 2n‑1
        # soft‑max weights (stable computation)
        exp_arg = beta * conv - np.max(beta * conv)
        w = np.exp(exp_arg)
        w /= w.sum()                                   # shape (2n‑1,)

        # gradient of smooth‑max w.r.t. a  (2·correlation of w and a)
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # Adam update (bias‑corrected)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        step = lr0 / (np.sqrt(v_hat) + 1e-8)
        a = a - step * m_hat
        a = project_to_simplex(a)

        # occasional logging
        if t % 2000 == 0:
            cur = compute_score(a)
            print(f"[adam] iter {t:6d} – approx β={beta:9.1f} – cur score {cur:.6f}")

    return a

# 3. Projected sub‑gradient polishing
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # active set – indices whose value is within 1e‑12 of the max
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient: 2·a shifted for each active index
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # small jitter every few thousand iterations
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# 4. Simple stochastic hill‑climb (final fine‑tuning)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy stochastic mass‑transfer moves with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]           # fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            # escape stagnation
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a
