"""
Improved search for a low‑max‑convolution step‑function.

Algorithm
---------
1.  Choose a moderately large size (n≈16384) – larger n gives a
    smaller theoretical lower bound (≈1) while still being cheap
    for FFT‑based convolutions.
2.  Initialise the vector with the best known construction (if
    available) or with a simple symmetric centre‑heavy family.
3.  Run a *smoothed‑max* gradient descent:
        –  f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)  (smooth approximation to
          max_k (a*a)_k).
        –  Its gradient can be written as a correlation of the weight
          vector w_k ∝ exp(λ·(a*a)_k) with a, computed efficiently via
          FFTs.
        –  After each step the vector is projected onto the simplex
          (non‑negative, sum = 1).  The stepsize η_t = η₀/√t and λ is
          slowly increased so that the optimisation focuses more on the
          true max as it converges.
4.  Refine the result with a long simulated‑annealing hill‑climb
    (mass‑transfer moves) on the exact objective.
5.  Return the best sequence discovered (as a plain list of floats)
    after a final call to ``evaluate_sequence`` (which performs the
    security checks).

The whole routine respects the 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s

# Analytic families (used for quick initialisation)
# ----------------------------------------------------------------------


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def uniform_family(n: int) -> np.ndarray:
    """Flat distribution (used as a sanity check)."""
    return np.full(n, 1.0 / n, dtype=np.float64)

def smooth_max_gradient_descent(start_a: np.ndarray,
                                deadline: float,
                                max_iters: int = 100_000,
                                eta0: float = 0.5,
                                lam_start: float = 1.0,
                                lam_inc: float = 1e-4) -> Tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of the max‑convolution.
    Uses the soft‑max (log‑sum‑exp) surrogate and projects onto the simplex.
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(a)

    lam = lam_start
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # 1) current a*a
        conv = convolution_fft(a, a)          # length 2n‑1
        max_conv = conv.max()

        # 2) soft‑max weights (avoid overflow)
        w = np.exp(lam * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # 3) gradient of the surrogate:
        #    ∂f/∂a_i = 2 * Σ_k w_k a_{k-i}  (indices out of range ignored)
        #    This is a correlation – compute via FFT.
        grad_full = convolution_fft(w, a[::-1])          # length 3n‑2
        #   grad_i = 2 * grad_full[0:n][::-1]
        grad = 2.0 * grad_full[:n][::-1]

        # 4) step (inverse‑time schedule)
        step = eta0 / math.sqrt(t)
        a = a - step * grad

        # 5) project back onto the simplex
        a = project_to_simplex(a)

        # 6) slowly increase λ so the surrogate gets sharper
        lam = lam_start + t * lam_inc

        # occasional evaluation of the true objective
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] smooth‑grad improved to {best_score:.6f}  (iter {t})")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] smooth‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 6_000_000,
                  temp_factor: float = 1.2e-3,
                  step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex (mass‑transfer moves).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule – based on current score
    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        # move type: swap two entries or transfer a fraction of mass
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

        # allocate most of the remaining budget to the gradient phase
        grad_deadline = deadline - 120.0
        a_opt, sc_opt = smooth_max_gradient_descent(
            start_a=a_start,
            deadline=grad_deadline,
            max_iters=120_000,
            eta0=0.8,
            lam_start=1.0,
            lam_inc=5e-4,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after smooth‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing refinement (high‑temperature)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=8_000_000,
            temp_factor=1.2e-3,
            step_scale_init=0.25,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback to a simple uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6. Final official evaluation (security checks are performed inside)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
