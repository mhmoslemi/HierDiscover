```
def max_conv_gradient(a):
    conv = full_convolution(a)
    max_val = conv.max()
    # w a vector of same length as conv, with 1 where conv == max_val (within tolerance)
    eps = max_val * 1e-12
    w = (conv >= max_val - eps).astype(float)
    # compute gradient = 2 * cross-correlation between a and w
    rev_w = w[::-1]
    size = 1 << ((a.shape[0] + rev_w.shape[0] - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    corr = np.fft.irfft(A * B, n=size)[: a.shape[0] + rev_w.shape[0] - 1]
    # indices mapping: conv length m = 2n-1, rev_w length m => correlation length n+m-1? Wait...
    # In earlier code, they did idx = (m - 1) - np.arange(n). Which picks appropriate slice.
    m = conv.shape[0]
    idx = (m - 1) - np.arange(a.shape[0])
    grad = 2.0 * corr[idx]
    return grad
```

```
def mirror_descent_optimize(a0, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    start = time.time()
    eta_start = 0.5
    eta_end = 0.001
    total = deadline - start
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        # compute step size linearly decreasing in log scale:
        if total > 0:
            eta = eta_start * ((eta_end/eta_start) ** (elapsed / total))
        else:
            eta = eta_end
        grad = max_conv_gradient(a)
        # mirror descent step:
        exp_term = np.exp(-eta * grad)
        a = a * exp_term
        # renormalize
        s = a.sum()
        if s == 0:
            # reinitialize to uniform
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / s
        # evaluate
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # optional print
        # jitter every 5000 steps
        it += 1
        if it % 5000 == 0:
            # small random perturbation
            noise = np.random.randn(a.shape[0]) * (eta * 0.01)
            a = a * np.exp(noise)  # multiplicative noise
            a = a / a.sum()
    return best_a
```

```
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
```

The program searches for a non‑negative step‑function (a probability vector) that minimises  

```
score = 2·n·max(conv(a,a))          (∑a = 1)
```

which is exactly the value returned by `evaluate_sequence`.  
The lower bound `2·n/(2·n‑1) ≃ 1` is attained only for a *perfectly flat* auto‑convolution.  
Our optimiser works in three stages:

1. **Initialisation** – start from the known construction (`height_sequence_1` if it exists),
   from a uniform vector and from a few random Dirichlet draws.  Keep the best of them.
2. **Smooth‑max Adam optimisation** – minimise a smooth approximation of the
   `max` using Adam with a slowly decreasing temperature.  The gradient of the
   smoothed objective is `2·(a ⋆ w)` where `w` is the soft‑max over the convolution.
   This phase consumes the bulk of the time budget.
3. **Permutation & mass‑transfer hill‑climb** – the ordering of the coefficients
   influences the convolution.  We perform two very cheap local searches:
   * random swaps of two entries (permutation hill‑climb),
   * random transfer of a small amount of mass from one index to another
     (mass‑transfer hill‑climb).  
   Both keep the vector on the probability simplex.

All phases respect the 1000 s wall‑clock limit, keep the best vector found,
and finally return it as a plain list of Python floats.

```python
"""
Non‑negative step‑function optimiser.

The optimiser combines:
  * a quick global initialisation (fallback / uniform / Dirichlet),
  * a long Adam‑based smooth‑max descent on the objective
    `τ·log Σ exp(conv/τ)`,
  * a permutation‑swap hill‑climb,
  * a small mass‑transfer hill‑climb.

All vectors are kept on the probability simplex (∑a = 1, a_i ≥ 0).
The algorithm is deliberately time‑aware: each phase stops before the
deadline and the current best solution is always updated.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 2.0,
    tau_end: float = 1e-6,
    tau_decay: float = 0.995,
    lr_start: float = 0.05,
    lr_decay: float = 0.999,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate
        f_τ(a) = τ·log Σ_k exp(conv_k(a)/τ)

    The function returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start
    start = time.time()

    while time.time() < deadline:
        # temperature schedule (geometric)
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # full convolution
        conv = full_convolution(a)               # length m
        shift = conv.max()
        # soft‑max weights (stable formulation)
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()            # shape (m,)

        # gradient of smooth‑max: 2·(a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # bookkeeping
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter={t}, τ={tau:.1e}, lr={lr:.3e}, score={cur_val:.6f}")

        # learning‑rate decay
        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-6)

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simple hill‑climbs on the exact objective
# ----------------------------------------------------------------------


def permute_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Swap two entries at a time; keep the best permutation."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]          # perform swap
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional log
            if it % 5000 == 0:
                print(f"[permute] iter={it}, score={val:.6f}")
        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]
        it += 1
    return best_a


def mass_transfer_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Move a small fraction of mass from one index to another."""
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # transfer a random fraction (up to 30 % of the source mass)
        frac = np.random.rand() * 0.30
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # already stays on the simplex
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            if it % 5000 == 0:
                print(f"[mass] iter={it}, score={val:.6f}")
        it += 1
    return best_a

    # --------------------------------------------------------------
    # 0. Load the known construction (if any) and initialise a start.

    fallback = None

        fallback = [float(x) for x in height_sequence_1]   # type: ignore

    # we keep a small pool of starts (fallback, uniform, a few Dirichlet)
```
