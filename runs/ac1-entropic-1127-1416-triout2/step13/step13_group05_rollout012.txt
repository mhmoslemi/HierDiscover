"""
search_step_function_v3.py

A time‑bounded optimiser for ``evaluate_sequence``.
Main ideas
----------

*   Systematic (golden‑section) optimisation of the *reverse power‑law*
    family a_i ∝ (n‑i)^p.  The optimal exponent p is found to several
    1e‑4 precision – the previous version only used a coarse grid.
*   A cheap *mixed* family a = (1‑w)·rev(p) + w·uniform is explored on a
    fine (p,w) grid around the best pure reverse‑power‑law.
*   A small pool of the best‑scoring vectors (size ≤ 30) is kept.
    Random convex combinations of these elite vectors are examined.
*   The usual sub‑gradient polishing and stochastic hill‑climb are kept,
    but with a longer budget and a slightly more aggressive step‑size
    schedule.
*   Two sequence lengths (1024 and 2048) are tried; the best result of
    either length is returned.

The code follows the harness contract: a top‑level ``propose_candidate``
function is provided, all helpers are defined at module level and no
closures or file I/O are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / projection utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # All non‑positive -> return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Result length = 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p},  p ≥ 0, front‑loaded decreasing shape."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def uniform_vec(n: int) -> np.ndarray:
    """Uniform distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def mix_rev_uniform(n: int, p: float, w: float) -> np.ndarray:
    """
    Convex mixture (1‑w)·reverse_power_law(p) + w·uniform.
    0 ≤ w ≤ 1.
    """
    a_rev = reverse_power_law(n, p)
    if a_rev is None:
        return None
    a_uni = uniform_vec(n)
    a = (1.0 - w) * a_rev + w * a_uni
    # projection only needed to clean up tiny numerical drift
    a = project_to_simplex(a)
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation helpers
# ----------------------------------------------------------------------
def golden_section_search(func, a: float, b: float, tol: float, deadline: float):
    """
    Minimise ``func`` on [a,b] with golden‑section, respecting the time limit.
    Returns (best_x, best_f).
    """
    phi = (1 + math.sqrt(5)) / 2
    resphi = 2 - phi
    x1 = a + resphi * (b - a)
    x2 = b - resphi * (b - a)
    f1 = func(x1)
    f2 = func(x2)

    while time.time() < deadline and (b - a) > tol:
        if f1 < f2:
            b, x2, f2 = x2, x1, f1
            x1 = a + resphi * (b - a)
            f1 = func(x1)
        else:
            a, x1, f1 = x1, x2, f2
            x2 = b - resphi * (b - a)
            f2 = func(x2)

    # Pick the best of the two interior points and the interval ends
    candidates = [(x1, f1), (x2, f2), (a, func(a)), (b, func(b))]
    best_x, best_f = min(candidates, key=lambda kv: kv[1])
    return best_x, best_f


def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.10) -> np.ndarray:
    """Projected sub‑gradient descent (same idea as the reference)."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # Identify all (almost) active indices
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient: sum of shifted copies of a
        g = np.zeros_like(a)
        for k in active:
            # contribution of a shifted by k
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape plateaus
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a = project_to_simplex(a + jitter)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_iter: int = 1_200_000) -> np.ndarray:
    """
    Greedy stochastic hill‑climb.  Moves a small fraction of mass from i→j.
    Accepts only improving moves; step size is adapted.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a *fraction* of the source mass
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)      # keep non‑negative
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.04, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        if no_improve > 120_000:
            # shake the current solution
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # 5 s safety margin

    best_seq = None
    best_val = float('inf')

    # ---- 0. optional provided fallback ---------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]    # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ---- 1. systematic search over a few lengths ----------------------
    lengths = [1024, 2048]                      # try both, keep the better
    elite_pool = []                             # (score, vector) pairs

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n--- Searching length n = {n} ---")

        # ---- a) pure reverse‑power‑law (golden‑section) -------------
        def rev_score(p):
            vec = reverse_power_law(n, p)
            return compute_score(vec)

        p_lo, p_hi = 0.2, 4.5
        tol = 1e-4
        p_opt, val_opt = golden_section_search(
            rev_score, p_lo, p_hi, tol, deadline)
        vec_opt = reverse_power_law(n, p_opt)
        print(f"[rev‑pow] p* = {p_opt:.5f}, score = {val_opt:.6f}")

        # store elite
        elite_pool.append((val_opt, vec_opt))

        # ---- b) mixture with uniform --------------------------------
        best_mix_val = float('inf')
        best_mix_vec = None
        # explore w ∈ [0,0.30] and p in a neighbourhood of p_opt
        w_grid = np.linspace(0.0, 0.30, 31)
        p_grid = np.linspace(max(0.0, p_opt - 0.5), p_opt + 0.5, 101)
        for w in w_grid:
            if time.time() >= deadline:
                break
            for p in p_grid:
                vec = mix_rev_uniform(n, p, w)
                if vec is None:
                    continue
                val = compute_score(vec)
                if val < best_mix_val:
                    best_mix_val, best_mix_vec = val, vec
            # keep the best so far in the elite pool
        if best_mix_vec is not None:
            print(f"[mix‑uni] best (p,w)=({p_opt:.3f},?) score = {best_mix_val:.6f}")
            elite_pool.append((best_mix_val, best_mix_vec))

        # ---- c) beta‑shape random sampling ----------------------------
        # 400 random (α,β) samples, bounded to a broad region
        for _ in range(400):
            if time.time() >= deadline:
                break
            alpha = np.random.uniform(-1.5, 6.0)
            beta = np.random.uniform(-1.5, 6.0)
            vec = beta_shape(n, alpha, beta)
            if vec is None:
                continue
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
            elite_pool.append((val, vec))

        # ---- d) random Dirichlet vectors --------------------------------
        # 800 draws from Dirichlet(0.3,…,0.3) – very sparse shapes
        alpha_dir = np.full(n, 0.3)
        for _ in range(800):
            if time.time() >= deadline:
                break
            vec = np.random.gamma(alpha_dir, 1.0)
            vec /= vec.sum()
            val = compute_score(vec)
            elite_pool.append((val, vec))

        # keep only the top‑30 candidates to limit memory & later work
        elite_pool.sort(key=lambda kv: kv[0])
        elite_pool = elite_pool[:30]

    # ------------------------------------------------------------------
    # 2. random convex combinations of elite vectors
    # ------------------------------------------------------------------
    if elite_pool and time.time() < deadline:
        print("\n[elite‑mix] exploring convex combinations")
        elite_vecs = [vec for _, vec in elite_pool]
        for _ in range(2500):
            if time.time() >= deadline:
                break
            # pick two distinct elites at random
            i, j = np.random.choice(len(elite_vecs), size=2, replace=False)
            α = np.random.rand()
            comb = α * elite_vecs[i] + (1 - α) * elite_vecs[j]
            comb = project_to_simplex(comb)
            val = compute_score(comb)
            if val < best_val:
                best_val, best_seq = val, comb.tolist()
            # keep track of really good combos
            if val < elite_pool[-1][0]:
                elite_pool.append((val, comb))
                elite_pool.sort(key=lambda kv: kv[0])
                elite_pool = elite_pool[:30]

    # ------------------------------------------------------------------
    # 3. sub‑gradient polishing of the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a0 = np.array(best_seq, dtype=float)
        polish_deadline = deadline - 80.0
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.10)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # ------------------------------------------------------------------
    # 4. final hill‑climb (longest remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_start = np.array(best_seq, dtype=float)
        hill_deadline = deadline - 5.0
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # safeguard – if nothing succeeded, fall back to a well‑known shape
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2048
        fallback_vec = reverse_power_law(n, 1.85)    # near‑optimal exponent
        best_seq = fallback_vec.tolist()
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # final reporting
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Best reported score = {final_score:.6f}")
    return best_seq
