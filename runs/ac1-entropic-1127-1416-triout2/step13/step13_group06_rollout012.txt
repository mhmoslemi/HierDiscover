"""
Propose a step‑function with a lower bound on the evaluation metric.

The algorithm works in three phases:
0. (fallback) try the previously known construction ``height_sequence_1``.
1. Scan a modest collection of analytic families (including new
   arcsine/raised‑cosine shapes) on a moderate grid and keep the best
   candidate.
2. Refine the best candidate with two complementary mirror‑descent
   procedures:
   • “hard” exponentiated‑gradient using the sub‑gradient at the
     location of the maximal convolution entry.
   • “soft” exponentiated‑gradient that minimises a smooth
     log‑sum‑exp approximation of the max‑convolution (beta is
     increased gradually).
3. Finish with a longer simulated‑annealing hill‑climb that moves mass
   between entries.
All steps respect the global wall‑clock budget (default 1000 s).  The
search is performed on a fixed discretisation (n = 8192), but the
objective is homogeneous, so the exact length does not affect the
final bound.
"""

import time

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max()) / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v

def uniform_family(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{α} (centre‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """Symmetric triangular bump; width_frac∈(0,1]."""
    half_width = max(1, int(width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return w / w.sum()


def exp_decay_family(n: int, gamma: float, left: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    if left:
        w = np.exp(-gamma * i)
    else:
        w = np.exp(-gamma * (n - 1 - i))
    return w / w.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Discrete Beta(p,q) density on 0…n‑1."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0:
        return uniform_family(n)
    return w / s


def arcsine_family(n: int) -> np.ndarray:
    """a_i ∝ 1 / sqrt((i+½)(n‑½‑i)).  (Beta(½,½) shape)"""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0:
        return uniform_family(n)
    return w / s


def raised_cosine_family(n: int) -> np.ndarray:
    """a_i ∝ (cos(π * (i‑(n‑1)/2) / n) + 1) / 2  (non‑negative)."""
    i = np.arange(n, dtype=np.float64)
    w = (np.cos(np.pi * (i - (n - 1) / 2.0) / n) + 1.0) / 2.0
    s = w.sum()
    return w / s


def sine_half_family(n: int) -> np.ndarray:
    """a_i ∝ sin(π * (i+½) / (n+1)).  Positive on (0,π)."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / (n + 1))
    s = w.sum()
    return w / s

def exp_gradient_subgrad(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         eta0: float = 0.6) -> Tuple[np.ndarray, float]:
    """
    Mirror descent (exponentiated gradient) using the exact
    sub‑gradient at the index where the convolution attains its
    maximum.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        a = a / a.sum()          # KL‑projection is just normalisation

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] subgrad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] subgrad final improvement to {best_score:.6f}")

    return best_a, best_score


def exp_gradient_softmax(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 150_000,
                        eta0: float = 0.5,
                        beta_start: float = 5.0,
                        beta_end: float = 30.0,
                        beta_step: float = 0.0002) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smooth log‑sum‑exp
    approximation of the max‑convolution.
    The inverse temperature ``beta`` is increased slowly from
    ``beta_start`` to ``beta_end``.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    beta = beta_start
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_c = float(conv.max())
        # softmax weights (normalised)
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient = sum_l w[l] * ∂conv[l]/∂a_i  = 2 * (a * w_rev)[anti‑diag]
        w_rev = w[::-1]
        grad_full = convolution_fft(a, w_rev)          # length = 3n‑2
        seg = grad_full[n - 1:2 * n - 1]                # length n, corresponds to indices n‑1 … 2n‑2
        g = 2.0 * seg[::-1]                            # reverse to obtain g_i

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        a = a / a.sum()

        # increase beta gradually
        if beta < beta_end:
            beta = min(beta_end, beta + beta_step)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] soft‑max improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] soft‑max final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 9_000_000,
                  temp_factor: float = 1.3e-3,
                  step_scale_init: float = 0.3) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.6)
        else:
            step_scale *= 0.96

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 12) -> Tuple[float, float]:
    """Golden‑section search for the minimum of func on [lo,hi]."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    return (c, fc) if fc < fd else (d, fd)

def propose_candidate(seed: int = 12345,

                    print(f"[{time.time():.1f}] new best ({name}={p:.3f}) → {best_score:.6f}")

        elif name == "gaussian":

                if time.time() >= deadline - 30.0:
                    break
                vec = maker(sigma)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best ({name}=σ={sigma:.0f}) → {best_score:.6f}")

        elif name == "triangular":
            for w in np.linspace(0.15, 0.85, 25):
                if time.time() >= deadline - 30.0:
                    break
                vec = maker(w)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best ({name}=w={w:.3f}) → {best_score:.6f}")

        elif name in ("exp_left", "exp_right"):
            for g in np.linspace(0.0, 6.0, 25):
                if time.time() >= deadline - 30.0:
                    break
                vec = maker(g)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best ({name}=γ={g:.3f}) → {best_score:.6f}")

        else:   # families without a parameter
            if time.time() >= deadline - 30.0:
                break
            vec = maker(None)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best ({name}) → {best_score:.6f}")

    # -----------------------------------------------------------
    # 2. Random dense search over families (≈8000 trials)

    print(f"[{time.time():.1f}] random dense search")
    max_trials = 8000
    trials = 0
    while time.time() < deadline - 30.0 and trials < max_trials:

        elif choice == 1:    # product (U‑shaped)
            alpha = np.random.uniform(0.0, 12.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 2:    # gaussian
            sigma = np.random.uniform(0.25 * n, 0.45 * n)
            vec = gaussian_center_family(n, sigma)
        elif choice == 3:    # triangular
            width = np.random.uniform(0.15, 0.85)
            vec = triangular_center_family(n, width)
        elif choice == 4:    # left‑heavy power
            beta = np.random.uniform(0.0, 12.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 5:    # right‑heavy power
            beta = np.random.uniform(0.0, 12.0)
            vec = right_heavy_power_family(n, beta)
        elif choice == 6:    # exponential left
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_decay_family(n, gamma, left=True)
        elif choice == 7:    # exponential right
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_decay_family(n, gamma, left=False)
        elif choice == 8:    # arcsine
            vec = arcsine_family(n)
        elif choice == 9:    # raised cosine
            vec = raised_cosine_family(n)
        else:                # sine half
            vec = sine_half_family(n)

        # Occasionally apply a random cyclic shift (breaking symmetry)
        if np.random.rand() < 0.15:
            shift = np.random.randint(0, n // 20 + 1)
            vec = np.concatenate([vec[shift:], np.zeros(shift)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved → {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Local golden‑section refinement for the best family (if any)
    # -----------------------------------------------------------
    # (Skip; the random search already covers many families; the heavy
    #  refinements are performed by the gradient‑based steps below.)

    # -----------------------------------------------------------
    # 4. Gradient‑based refinements
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 300.0:
        a0 = np.array(best_seq, dtype=np.float64)

        # --- hard sub‑gradient exponentiated descent ---
        print(f"[{time.time():.1f}] hard exponentiated‑gradient refinement")
        a_hard, score_hard = exp_gradient_subgrad(
            start_a=a0,
            deadline=deadline - 250.0,
            max_iters=250_000,
            eta0=0.6,
        )
        if score_hard < best_score:
            best_score, best_seq = score_hard, a_hard.tolist()
            print(f"[{time.time():.1f}] after hard EG best = {best_score:.6f}")

        # --- soft (log‑sum‑exp) exponentiated descent ---
        if time.time() < deadline - 150.0:
            print(f"[{time.time():.1f}] soft exponentiated‑gradient refinement")
            a_soft, score_soft = exp_gradient_softmax(
                start_a=np.array(best_seq, dtype=np.float64),
                deadline=deadline - 100.0,
                max_iters=120_000,
                eta0=0.5,
                beta_start=5.0,
                beta_end=35.0,
                beta_step=0.0005,
            )
            if score_soft < best_score:
                best_score, best_seq = score_soft, a_soft.tolist()
                print(f"[{time.time():.1f}] after soft EG best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Final simulated‑annealing hill‑climb
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=10_000_000,
            temp_factor=1.3e-3,
            step_scale_init=0.3,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Fallback – uniform vector (should never happen)
    # -----------------------------------------------------------
    if best_seq is None:
        print(f"[{time.time():.1f}] fallback to uniform")
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
