def compute_optimal_g(f, M):
    n = len(f)
    b = cp.Variable(n)
    # constraints: b >= 0
    constraints = [b >= 0]
    # compute convolution linear matrix: conv = convmtx(f) * b <= M
    # We need to generate Toeplitz matrix of f appropriate for full convolution.
    # We can use scipy.signal.convolve? Not linear. Build matrix manually: For k from 0 to 2n-2:
    # sum_{i} f[i] * b[k-i] where k-i in [0,n-1].
    # Equivalent to linear constraints: sum_{i=max(0, k-(n-1))}^{min(k, n-1)} f[i] * b[k-i] <= M.
    # We'll build for each k.
    for k in range(2*n-1):

    # maximize sum b
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    b_opt = b.value

"""
Propose a candidate step‑function minimizing the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The approach builds on the previous search but adds a *linear‑programming
refinement* inspired by the original iterative scheme described in the
statement.

Key ideas
---------
1. Choose a moderate length (n = 2048) – large enough for a good bound but
   small enough that the O(n²) LP can be solved many times within the 1000 s
   budget.
2. Start from the best known construction (the global ``height_sequence_1``
   if present, otherwise a simple product‑family shape) and keep the vector
   on the probability simplex (non‑negative, sum = 1).
3. In each refinement iteration
   * compute the current maximal self‑convolution M,
   * solve the LP
         maximise   1ᵀb
         subject to  b ≥ 0,
                     (f * b) ≤ M   (component‑wise),
   * normalise the LP solution to a probability vector g,
   * mix the current distribution ``f`` with ``g`` (mixing weight t ≈ 0.3)
     and re‑normalise.
   The LP pushes mass into positions where the convolution with the current
   distribution is still small, thereby flattening the whole convolution
   profile and reducing the max.
4. After a few LP steps run a short Adam smooth‑max optimisation as a
   final polish.
5. Return the best sequence found (as a plain ``list[float]``).

All helper routines are top‑level, no closures are used, and the code works
with only ``numpy`` and ``scipy`` (installed in the competition environment).
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum²).  a must be non‑negative."""
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# ----------------------------------------------------------------------
# Analytic families (used only for an initial guess)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix T such that  T @ b  =  f * b  (full convolution).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=np.float64)
    # column j contributes f[0], f[1], …, f[n‑1] to rows j … j+n‑1
    for j in range(n):
        T[j:j + n, j] = f
    return T


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise 1ᵀb   s.t.  b ≥ 0  and  (f * b) ≤ M
    using scipy.optimize.linprog (highs method).  Returns the optimal b.
    """
    try:
        from scipy.optimize import linprog
    except Exception as e:  # pragma: no cover
        raise RuntimeError("scipy.optimize.linprog is required for LP refinement") from e

    n = f.shape[0]
    T = build_conv_matrix(f)          # shape (2n‑1, n)
    # Linear constraints: T @ b <= M (vector of length 2n‑1)
    A_ub = T
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Objective: maximise sum(b)  →  minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n
    # Using the high‑performance HiGHS backend
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        # fall back to a uniform vector – it is always feasible
        return np.full(n, 1.0 / n, dtype=np.float64)
    b_opt = np.maximum(res.x, 0.0)   # numerical safety
    return b_opt


# ----------------------------------------------------------------------
# Adam smooth‑max optimiser (lightweight version)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 10_000.0,
                         lr: float = 0.015) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of the max‑convolution.
    This is used only as a final polishing step after the LP refinement.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)        # log‑softmax variables

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = normalize(np.exp(z))
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = normalize(a)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t a
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad)
        grad_z = a * grad - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 4000 == 0:
            a_tmp = normalize(np.exp(z))
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = normalize(np.exp(z))
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The method consists of:

    1. Initialise from a known good construction (``height_sequence_1``) or a
       product‑family guess.
    2. Repeatedly solve a linear programme that spreads mass into
       low‑convolution regions (LP refinement).
    3. Finish with a short Adam optimisation on the smooth‑max surrogate.
    4. Return the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for final checks
    deadline = start_time + budget_s - 5.0

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. initialise from fallback if available
    # ------------------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = np.array([float(x) for x in height_sequence_1], dtype=np.float64)  # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose working length (2048 gives a good trade‑off)
    # ------------------------------------------------------------------
    n = 2048

    # ------------------------------------------------------------------
    # 2. If we have no starting point, create a product‑family guess
    # ------------------------------------------------------------------
    if best_seq is None:
        # preliminary alpha found to be decent in earlier runs
        alpha0 = 3.0
        best_seq = symmetric_product_family(n, alpha0)
        best_score = compute_score(best_seq)
        print(f"[{time.time():.1f}] initial product‑family α={alpha0:.2f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. LP refinement loop
    # ------------------------------------------------------------------
    f = normalize(best_seq)          # ensure sum = 1
    iter_cnt = 0
    while time.time() < deadline - 30.0:
        iter_cnt += 1
        conv_f = convolution_fft(f, f)
        M = conv_f.max()
        # solve LP: maximise sum(b)  s.t.  (f * b) ≤ M,  b ≥ 0
        b = solve_lp_max_sum(f, M)
        sum_b = b.sum()
        if sum_b < 1e-12:
            print(f"[{time.time():.1f}] LP produced near‑zero vector, stopping.")
            break
        g = b / sum_b                # normalise to a probability vector
        # mixing weight – start relatively aggressive, then decay
        t = 0.30
        f_new = normalize((1.0 - t) * f + t * g)
        new_score = compute_score(f_new)
        if new_score < best_score:
            best_score = new_score
            best_seq = f_new.copy()
            print(f"[{time.time():.1f}] LP iteration {iter_cnt} improved to {best_score:.6f}")
        # very small improvement → stop early
        if abs(new_score - best_score) < 1e-7:
            break
        f = f_new

    # ------------------------------------------------------------------
    # 4. Final Adam smooth‑max polishing (optional, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final Adam polishing")
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=best_seq,
            deadline=deadline,
            max_iters=200_000,
            beta_start=5.0,
            beta_end=8000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq.tolist()
