def propose_candidate(...):

    deadline = start_time + budget_s - 5

    # 1. Scan families

    for each family and coarse param ranges -> compute vectors and evaluate (using compute_score which we have replicates of evaluate_sequence but more efficient). Keep best.

    # 2. Subgradient descent.

    # Choose initial a as best_seq if found else uniform.

    a = np.array(best_seq) if best_seq else np.full(n, 1.0/n)

    a = normalize(a) # ensures sum=1.

    best_a = a.copy()
    best_score = compute_score(a)

    # subgradient loop
    iteration = 0
    last_improve_iter = 0
    while time.time() < deadline - 30:
        iteration += 1
        # compute conv
        conv = convolution_fft(a, a)
        k = np.argmax(conv)
        max_conv = conv[k]

        # gradient
        idx = np.arange(n)
        j = k - idx
        mask = (j >=0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]  # subgradient

        # step size schedule
        eta = 0.3 / np.sqrt(iteration) # tune

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        sum_a = a_new.sum()
        if sum_a == 0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new = a_new / sum_a

        a = a_new

        if iteration % 500 == 0:
            sc = compute_score(a)
            if sc < best_score - 1e-6:
                best_score = sc
                best_a = a.copy()
                last_improve_iter = iteration
                print(f"[{time.time():.1f}] subgrad iter {iteration} improved to {best_score:.6f}")

        # if stagnation
        if iteration - last_improve_iter > 5000:
            # random perturbation
            perturb = np.random.rand(n)
            perturb = perturb - perturb.mean()
            perturb = perturb / np.linalg.norm(perturb) * 0.01  # small L2 norm
            a = a + perturb
            a = np.maximum(a, 0)
            a = a / a.sum()
            print(f"[{time.time():.1f}] stagnation, perturbation applied at iter {iteration}")
            last_improve_iter = iteration

    # After loop, best_a is candidate.

    # Optional: final short SA to polish
    cur_a = best_a.copy()
    cur_score = compute_score(cur_a)
    best_local_a = cur_a.copy()
    best_local_score = cur_score

    T = cur_score * 1e-3
    steps = 0
    max_steps = 2000000
    while time.time() < deadline - 5 and steps < max_steps:
        steps += 1
        i, j = np.random.randint(n, size=2)
        # transfer δ from i to j
        if cur_a[i] > 1e-12:
            δ = np.random.rand() * cur_a[i] * 0.1

            cand[j] += δ

                    best_local_score = cur_score
                    best_local_a = cur_a.copy()
        # cooling
        T *= 0.999999

    # choose final best
    final_a = best_local_a if best_local_score < best_score else best_a
    final_seq = final_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] final score = {final_score:.6f}")

"""
Hybrid optimizer for the step‑function attack:

1.  Scan a few classic analytic families (U‑shaped product, centred Gaussian,
    triangular, left/right power, exponential decay, Beta, and an inverted
    Hann window) on a modest grid – this gives a good warm‑start.
2.  Refine the best candidate with a *scaled additive sub‑gradient* method.
    The sub‑gradient of the max‑convolution is computed analytically,
    an AdaGrad‑like per‑coordinate step‑size is used, and after each update
    the vector is projected back onto the probability simplex (sum = 1,
    non‑negative).  The step‑size decays as 1/√t.
3.  If the descent stagnates for a while a tiny random perturbation is added
    (to escape shallow local minima).
4.  A short simulated‑annealing hill‑climb is executed for a few seconds to
    polish the result.
5.  The final sequence (as a plain list) is returned after a call to the
    provided ``evaluate_sequence`` (which also applies the required clipping).

The optimiser works on the full resolution (default n = 8192) but runs the
expensive gradient phase for many thousands of iterations – the bulk of the
1000 s budget is spent here, which is crucial for pushing the score below
the previous 1.51077 bound.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, scoring, simplex projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_self_fft(a: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution a * a via real FFT (fast O(N log N))."""
    n = a.size
    m = 2 * n - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    C = A * A                      # element‑wise square = FFT of a*a
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_self_fft(a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    # based on the algorithm of Wang & Carreira‑Perpiñán (2013)
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, v.shape[0] + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip negative values to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Mirror of left‑heavy."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return normalize(w)


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return normalize(w)


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """Symmetric triangular bump (width_frac ∈ (0,1])."""
    half_width = max(1, int(width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return normalize(w)


def exp_decay_family(n: int, gamma: float, left: bool = True) -> np.ndarray:
    """Exponential decay from the chosen side."""
    i = np.arange(n, dtype=np.float64)
    if left:
        w = np.exp(-gamma * i)
    else:
        w = np.exp(-gamma * (n - 1 - i))
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Discrete Beta distribution on the grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)


def inverted_hann_family(n: int) -> np.ndarray:
    """U‑shaped Hann window: a_i ∝ 1‑cos(2π i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    return normalize(w)

# Optimisers

def additive_subgradient_descent(start_a: np.ndarray,
                                 deadline: float,
                                 max_iters: int = 800_000,
                                 eta0: float = 0.6) -> Tuple[np.ndarray, float]:
    """
    Projected additive sub‑gradient descent with AdaGrad‑style per‑coordinate
    scaling.  The objective is the max‑convolution (the denominator is constant
    because we keep sum(a)=1).  Returns the best vector seen and its score.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    # AdaGrad accumulator
    G = np.zeros_like(a)
    eps = 1e-12

    last_improve = 0
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_self_fft(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max entry (index k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # AdaGrad update
        G += g * g
        step = eta0 / math.sqrt(t)
        a_new = a - step * g / np.sqrt(G + eps)

        # project back to simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= s

        a = a_new

        # periodic evaluation
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score - 1e-8:
                best_score, best_a = sc, a.copy()
                last_improve = t
                print(f"[{time.time():.1f}] sub‑grad iter {t} improved to {best_score:.6f}")

        # stagnation handling: tiny random perturbation
        if t - last_improve > 100_000:
            # add a small zero‑mean noise and re‑normalise
            delta = np.random.randn(n) * 1e-4
            a = a + delta
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve = t
            print(f"[{time.time():.1f}] sub‑grad iter {t} stagnation -> perturbation")

    # final check
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score, best_a = final_sc, a.copy()
        print(f"[{time.time():.1f}] sub‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000,
                  temp_factor: float = 1.0e-3,
                  step_scale_init: float = 0.3) -> Tuple[np.ndarray, float]:
    """
    Short simulated‑annealing hill‑climb (move small mass between two
    coordinates).  Used only for polishing.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # transfer a fraction of mass from i to j
        delta = step_scale * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        diff = cand_score - cur_score
        if diff < 0 or np.random.rand() < np.exp(-diff / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        # simple linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

            if fam2[0] != "beta":
                if fam2[0] == "product":
                    p2 = np.random.uniform(0.0, 12.0)
                    vec2 = symmetric_product_family(n, p2)
                elif fam2[0] == "center_power":
                    p2 = np.random.uniform(0.0, 12.0)
                    vec2 = symmetric_center_power_family(n, p2)
                elif fam2[0] == "gaussian":
                    s2 = np.random.uniform(0.2 * n, 0.5 * n)
                    vec2 = gaussian_center_family(n, s2)
                elif fam2[0] == "triangular":
                    w2 = np.random.uniform(0.1, 0.9)
                    vec2 = triangular_center_family(n, w2)
                elif fam2[0] == "left_power":
                    b2 = np.random.uniform(0.0, 12.0)
                    vec2 = left_heavy_power_family(n, b2)
                elif fam2[0] == "right_power":
                    b2 = np.random.uniform(0.0, 12.0)
                    vec2 = right_heavy_power_family(n, b2)
                elif fam2[0] == "exp_decay_left":
                    g2 = np.random.uniform(0.0, 6.0)
                    vec2 = exp_decay_family(n, g2, left=True)
                elif fam2[0] == "exp_decay_right":
                    g2 = np.random.uniform(0.0, 6.0)
                    vec2 = exp_decay_family(n, g2, left=False)
                else:
                    vec2 = inverted_hann_family(n)
                vec = lam * vec + (1 - lam) * vec2
                vec = normalize(vec)

        # occasional zero‑padding (break symmetry)
        if np.random.rand() < 0.2:
            d = np.random.randint(0, n // 30 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Aggressive additive sub‑gradient descent (AdaGrad)
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely, but fallback to uniform
        best_seq = [1.0 / n] * n

    a_start = np.array(best_seq, dtype=np.float64)
    a_start = normalize(a_start)

    if time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting additive sub‑gradient refinement")
        a_opt, sc_opt = additive_subgradient_descent(
            start_a=a_start,
            deadline=deadline - 150.0,
            max_iters=600_000,
            eta0=0.8,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final short SA polish
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final simulated‑annealing polish")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=2_000_000,
            temp_factor=1.0e-3,
            step_scale_init=0.3,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Official evaluation (applies security clipping etc.)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
