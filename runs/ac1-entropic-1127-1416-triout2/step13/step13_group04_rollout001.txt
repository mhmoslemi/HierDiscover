def smooth_max_gradient(a, alpha):
    conv = full_convolution(a)
    # shift conv for stable exp
    maxc = conv.max()
    exp_terms = np.exp(alpha * (conv - maxc))
    w = exp_terms / exp_terms.sum()
    # compute gradient via convolution of a_rev with w
    a_rev = a[::-1]
    # w length = 2n-1
    conv_rev = np.convolve(a_rev, w)
    # gradient = 2 * conv_rev[n-1:n-1+n]
    g = 2.0 * conv_rev[n-1:n-1+n]
    return g, conv

"""
Propose a high‑quality step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.  The method differs from the reference by
* using a smooth approximation of the max‑convolution (soft‑max) and
  Adam‑style adaptive gradient descent on the simplex,
* a geometric schedule for the soft‑max sharpness (α) that gradually
  approaches the true max,
* multiple random restarts drawn from a variety of parametric families,
* a final short sub‑gradient polishing pass.

The objective is scale‑invariant, therefore all vectors are kept normalised
to sum = 1 (a probability simplex).  The routine respects the 1000 s
budget, projects after each update and keeps the best vector according to
the official ``evaluate_sequence`` call.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x >=0, sum=1}."""
    if np.all(v <= 0):
        # uniform vector is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (length = len(x)+len(y)-1)."""
    m = 1 << ((len(x) + len(y) - 2).bit_length())   # next power of two
    X = np.fft.rfft(x, n=m)
    Y = np.fft.rfft(y, n=m)
    conv = np.fft.irfft(X * Y, n=m)
    return conv[:len(x) + len(y) - 1]


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) self‑convolution a * a using FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The objective used by ``evaluate_sequence`` is scale‑invariant,
    therefore we always enforce Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for random restarts)
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}   (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Sample from Dirichlet(conc,…,conc)."""
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    s = y.sum()
    if s == 0.0:
        return None
    return y / s


# ----------------------------------------------------------------------
# Smooth‑max + Adam optimisation on the simplex
# ----------------------------------------------------------------------
def smooth_max_optimize(a0: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    Minimise the true max‑convolution by optimising a smooth surrogate:
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k).

    The gradient of f_α is a weighted sum of the convolution terms
    (weights = soft‑max of the entries).  Adam with an adaptive step
    size and projection onto the simplex is used.
    The sharpness α is increased geometrically during the run,
    ending with a value large enough (≈2000) to approximate the true max.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.02                # base learning rate

    # soft‑max sharpness schedule
    alpha = 10.0
    alpha_max = 2000.0
    alpha_factor = 1.02

    # bookkeeping
    best_a = a.copy()
    best_score = compute_score(a)
    step = 0

    while time.time() < time_deadline:
        step += 1

        # ----- compute smooth gradient -----
        conv = full_convolution(a)                     # length 2n‑1
        maxc = conv.max()
        # soft‑max weights (stable)
        scaled = alpha * (conv - maxc)                # max term = 0
        # prevent overflow/underflow
        scaled = np.clip(scaled, -700, 0)              # exp(≤0) ≤ 1
        exp_terms = np.exp(scaled)
        w = exp_terms / exp_terms.sum()                # sum_{k} w_k = 1

        # gradient = 2 * (a_rev * w)[n-1 : n-1+n]
        a_rev = a[::-1]
        conv_rev = convolve_fft(a_rev, w)             # length n + (2n‑1) - 1 = 3n‑2
        g = 2.0 * conv_rev[n - 1:n - 1 + n]            # shape (n,)

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step_vec
        a = project_to_simplex(a)

        # ----- keep best according to true objective -----
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- increase α gradually -----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to escape shallow plateaus
        if step % 5000 == 0:
            jitter = np.random.randn(n) * 1e-4
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as in the reference implementation)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of each active entry: 2 * a shifted
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

    if best_seq is not None and time.time() < deadline:
        print("[polish] running sub‑gradient polishing")
        a_best = np.array(best_seq, dtype=np.float64)
        a_polished = subgradient_polish(a_best, deadline, step0=0.12)
        update_best(a_polished)

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
