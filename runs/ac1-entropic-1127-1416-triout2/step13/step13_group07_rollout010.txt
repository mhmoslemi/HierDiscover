"""
propose_candidate – a multi‑stage optimizer for the max‑convolution objective.

The search works as follows (all stages respect the global 1000 s wall‑clock
budget):

1. **Coarse deterministic scan** of a handful of analytic families
   (symmetric product, left‑heavy power, exponential‑U, a 2‑parameter
   Beta family and a new *asymmetric product* family).

2. **Fine one‑dimensional refinement** of the best symmetric product
   exponent using a golden‑section search.

3. **Systematic right‑shifts** of the refined product shape (up to
   a few hundred offsets) – a cheap way of breaking the symmetry and
   reducing the most overloaded convolution bin.

4. **Mixture optimisation** – a convex blend of the refined product shape
   with the uniform vector.  The mixing weight is also tuned by a
   golden‑section search.

5. **Random dense exploration** of all families (≈3000 samples) and of
   random mixtures of two families.

6. **Adam optimisation** of the exact smooth‑max surrogate of the max‑convolution.
   A long continuation schedule (β from 5 to 2·10⁴) and a moderate learning
   rate are used; the number of iterations is limited only by remaining
   wall‑clock time.

7. **Projected sub‑gradient descent** on the true ‑max‑objective.

8. **Greedy hill‑climb** that transfers a tiny amount of mass between
   two randomly chosen entries whenever this improves the score.

The best sequence found is returned as a plain Python list of floats.
All helper functions are top‑level; the code uses only NumPy and the
provided ``evaluate_sequence`` for the final verification.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    a is assumed non‑negative (the optimiser keeps it on the simplex).
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


def shift_and_normalize(v: np.ndarray, d: int) -> np.ndarray:
    """Right‑shift by d (drop first d entries, pad zeros) and renormalise."""
    if d <= 0:
        return v.copy()
    n = v.shape[0]
    shifted = np.concatenate([v[d:], np.zeros(d, dtype=v.dtype)])
    return normalize(shifted)

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    # protect against log(0)
    eps = 1e-12
    logprod = np.log(i + 0.5 + eps) + np.log(n - 0.5 - i + eps)
    w = np.exp(-alpha * logprod)
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Discrete beta‑type density on 0…n‑1.
    p,q>0.  When p,q<1 the shape is U‑shaped; p,q>1 gives a bell.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)


def asymmetric_product_family(n: int, alpha_l: float, alpha_r: float) -> np.ndarray:
    """
    Two‑sided product family:
        a_i ∝ (i+½)^{‑αₗ}·(n‑½‑i)^{‑αᵣ}
    Allows the distribution to be skewed.
    """
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    left = np.log(i + 0.5 + eps)
    right = np.log(n - 0.5 - i + eps)
    w = np.exp(-alpha_l * left - alpha_r * right)
    return normalize(w)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 1_000_000,
                         beta_start: float = 5.0,
                         beta_end: float = 20_000.0,
                         lr: float = 0.01,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    ``beta`` grows geometrically from ``beta_start`` to ``beta_end``.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # Initialise via a log‑softmax to guarantee positivity
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 4000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.040) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active convolution bin
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        # each ordered pair contributes a[j] to gradient
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_greedy(start_a: np.ndarray,
                      deadline: float,
                      max_steps: int = 8_000_000,
                      init_scale: float = 0.13) -> Tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a small fraction of mass from a random
    donor to a random receiver whenever this reduces the objective.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    scale = init_scale

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # pick donor (largest entry) and a random receiver
        i = np.argmax(cur_a)
        j = np.random.randint(n)
        if i == j:
            continue

        delta = scale * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            # success → increase step size slowly
            scale = min(scale * 1.035, 0.5)
        else:
            # failure → shrink step size
            scale = max(scale * 0.965, 1e-4)

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] greedy step {step}, best {best_score:.6f}")

    return best_a, best_score


def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 16) -> Tuple[float, float]:
    """Minimise ``func`` on [lo, hi] using golden‑section."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Adam refinement (smooth‑max, unrestricted)")
        a_start = best_seq.astype(np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=a_start,
            deadline=deadline - 150.0,
            max_iters=800_000,
            beta_start=5.0,
            beta_end=30_000.0,
            lr=0.009,
            sym_interval=0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Projected sub‑gradient descent on the true max‑objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            start_a=best_seq.astype(np.float64),
            deadline=deadline - 60.0,
            max_iters=1_500_000,
            step0=0.040,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub
            print(f"[{time.time():.1f}] after sub‑gradient best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Final greedy hill‑climb (mass‑transfer moves)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final greedy hill‑climb")
        a_final, sc_final = hill_climb_greedy(
            start_a=best_seq.astype(np.float64),
            deadline=deadline,
            max_steps=8_000_000,
            init_scale=0.13,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final
            print(f"[{time.time():.1f}] greedy improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Fallback (very unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] fallback uniform; score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (includes all security checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq.tolist()
