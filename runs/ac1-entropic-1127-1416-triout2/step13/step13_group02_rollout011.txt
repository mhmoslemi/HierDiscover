    best_a = None

    # baseline

    cand_vecs = baseline_search(n, baseline_deadline, max_keep=top_k)

        if val < best_val:
            best_val, best_a = val, v.copy()
    best_vecs.extend(cand_vecs)

    # Main polishing loop
    while time.time() < deadline - 5:
        # choose start vector: sometimes best, sometimes random
        if np.random.rand() < 0.3 and best_a is not None:
            start_vec = best_a.copy()
        else:
            # random Dirichlet with random concentration
            conc = np.random.choice([0.05,0.1,0.2,0.5,1,2,4])
            start_vec = np.random.dirichlet(np.full(n, conc))
        # allocate slice of time
        slice_dead = time.time() + min(8.0, deadline - time.time() - 2.0)
        # run Adam smooth for that slice
        refined = adam_smooth(start_vec, slice_dead)
        val = compute_score(refined)
        if val < best_val:
            best_val, best_a = val, refined.copy()
            print(f"[polish] New best {best_val:.6f}")
        # sometimes add to pool for further polishing
        if len(best_vecs) < top_k:
            best_vecs.append(refined.copy())
        else:
            # replace worst in pool if improved
            worst_val = max(compute_score(v) for v in best_vecs)
            if val < worst_val:
                # replace worst
                idx = np.argmax([compute_score(v) for v in best_vecs])
                best_vecs[idx] = refined.copy()
        # optional jitter: small random tweak of best vector
        if np.random.rand() < 0.1 and best_a is not None:
            # jitter via random Dirichlet mixed with best
            jitter = np.random.dirichlet(np.full(n, 0.5))*0.01
            cand = best_a * (1-0.01) + jitter
            cand = cand / cand.sum()
            best_a = cand

    # final local refinement (simple exchange)
    if best_a is not None and time.time() < deadline - 1:

def convolve_fft(a, b):
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n+m-2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[:n+m-1]

"""
Enhanced optimizer for the step‑function benchmark.

The search proceeds in three stages:

1. **Broad global sweep** – exhaustive scans of simple parametric families
   (power‑law, exponential, beta) plus a large pool of random Dirichlet draws.
   The best 30 candidates (in terms of the exact objective
   ``2·n·max(conv(a,a))`` – i.e. the value of ``evaluate_sequence`` assuming
   ``∑a=1``) are kept.

2. **Smooth‑gradient polishing** – each retained candidate is refined with an
   Adam optimiser that minimizes a *soft‑max* approximation of the convolution
   peak.  The loss is

       L(a) = (1/α)·log∑_k exp(α·(a*a)_k)

   whose gradient is a weighted sum of all convolution entries, giving a much
   richer signal than the pure sub‑gradient of the max.  The variables are kept
   on the simplex via a soft‑max parametrisation (mirror‑descent).  The primary
   best vector is polished for the majority of the remaining budget, while
   many short restarts explore the landscape.

3. **Local exchange fine‑tuning** – a cheap hill‑climbing routine that moves a
   small amount of mass between two random positions and keeps any improvement.
   This often extracts a few extra digits of precision.

All stages respect the global time limit (default 1000 s).  The routine finally
returns a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT. Returns the full result."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 2).bit_length())      # next power‑of‑two ≥ n+m‑1
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[: n + m - 1]


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return conv_fft(a, a)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_mixed_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-lam·i)."""
    a = (np.arange(1, n + 1, dtype=float) ** (-p)) * np.exp(-lam * np.arange(n))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration – keep the best few vectors
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 30,
) -> list[np.ndarray]:
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep
    for p in np.linspace(0.01, 6.0, 500):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(1e-6, 0.4, 500):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep (covers U‑shapes and asymmetric shapes)
    al_vals = np.concatenate(
        [
            np.geomspace(0.01, 0.3, 15),
            np.linspace(0.3, 5.0, 25),
        ]
    )
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) mixed power‑law / exponential
    for p in np.linspace(0.01, 3.0, 150):
        for lam in np.linspace(1e-5, 0.2, 150):
            if time.time() >= deadline:
                break
            try_candidate(generate_mixed_powerexp(n, p, lam))

    # 5) massive Dirichlet pool
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 2000
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth‑gradient Adam optimiser (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def adam_smooth(
    a0: np.ndarray,
    deadline: float,
    lr: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha: float = 15.0,
) -> np.ndarray:
    """
    Adam optimiser that minimises a smooth approximation of the peak
    of the convolution.  The variables are kept on the simplex via a
    soft‑max (mirror‑descent) parametrisation.
    """
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)               # log‑space variables
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # ---- soft‑max projection ----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ---- convolution and smooth weights ----
        conv = full_convolution(a)
        max_c = conv.max()
        # soft‑max over the whole convolution vector
        w_raw = np.exp(alpha * (conv - max_c))
        w = w_raw / w_raw.sum()                      # shape (2n‑1,)

        # ---- gradient of the smoothed loss ----
        # g_i = 2·∑_k w_k·a_{k-i}  (cross‑correlation)
        w_rev = w[::-1]                               # reversed weight vector
        grad_full = conv_fft(a, w_rev)                # length = n + (2n‑1) - 1 = 3n‑2
        offset = len(w_rev) - 1                       # = 2n‑2
        g = 2.0 * grad_full[offset - np.arange(n)]

        # ---- mirror‑descent gradient (soft‑max Jacobian) ----
        g_u = a * (g - np.dot(g, a))

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ---- keep the best observed point ----
        cur_val = float(2.0 * n * max_c)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

    return best_a if best_a is not None else a0

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    n = 2048  # problem size – balances expressiveness and speed

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previous run
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback]   score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline sweep (≈ 200 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline - 550.0, start_time + 200.0)
    if time.time() < baseline_deadline:
        candidates = baseline_search(n, baseline_deadline, max_keep=30)
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline]  best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Main polishing with smooth‑gradient Adam
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and best_seq is not None:
        # a) long polish of the current best
        polish_end = time.time() + 0.55 * remaining
        a0 = np.array(best_seq, dtype=float)
        refined = adam_smooth(a0, polish_end, lr=0.15, alpha=20.0)
        val = compute_score(refined)
        print(f"[polish‑main] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # b) many short restarts while we still have time
        while time.time() < deadline - 5.0:
            # random Dirichlet start
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            slice_end = time.time() + min(6.0, deadline - time.time() - 2.0)
            refined = adam_smooth(a_start, slice_end, lr=0.12, alpha=15.0)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[polish‑restart] new best = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final local exchange fine‑tuning (≈ last 5 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        a_start = np.array(best_seq, dtype=float)
        final = local_exchange(a_start, deadline)
        val = compute_score(final)
        print(f"[exchange] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback to uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    reported = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {reported:.6f}")
    return best_seq
