def shape_from_knots(v, n):
    K = len(v)
    # ensure positivity
    v = np.maximum(v, 0.0)
    # create positions for knots evenly spaced
    xs = np.linspace(0, n-1, K)
    xi = np.arange(n)
    a = np.interp(xi, xs, v)
    # Clip to >=0
    a = np.maximum(a, 0.0)
    sum_a = a.sum()
    if sum_a == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / sum_a
    return a
```

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    n = 2048
    best_seq = None
    best_score = float('inf')

    # fallback
    try:
        fallback = [float(x) for x in height_sequence_1]
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_seq = fallback
            best_score = val
    except Exception:
        pass

    # basic uniform baseline
    uniform = np.full(n, 1.0 / n)
    uniform_score = compute_score(uniform)
    if uniform_score < best_score:
        best_seq = uniform.tolist()
        best_score = uniform_score

    # DE search
    from scipy.optimize import differential_evolution

    K = 12  # number of knots
    # initial bounds for knot values, we can use [0,1] (relative amplitude)
    bounds = [(0.0, 1.0) for _ in range(K)]

    def shape_score(knots):
        a = shape_from_knots(knots, n)
        return compute_score(a)

    # callback for time limit
    def de_callback(xk, convergence):
        if time.time() >= deadline:
            return True
        a = shape_from_knots(xk, n)
        sc = compute_score(a)
        nonlocal best_seq, best_score
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[DE] new best {best_score:.6f}")
        return False

    # compute remaining time for DE
    de_deadline = deadline - 200  # leave time for polishing
    if time.time() < de_deadline:
        maxiter = max(10, int((de_deadline - time.time()) // 0.2))  # rough estimate?

        maxiter = 150
        print("[DE] start")
        result = differential_evolution(shape_score, bounds,
                                        maxiter=maxiter, polish=False,
                                        init='latinhypercube',
                                        seed=seed,
                                        callback=de_callback,
                                        disp=False)
        # after DE, evaluate best result
        a_de = shape_from_knots(result.x, n)
        sc_de = compute_score(a_de)
        if sc_de < best_score:
            best_score = sc_de
            best_seq = a_de.tolist()
        print(f"[DE] finished best {best_score:.6f}")

    # local Adam refinement
    if best_seq is not None and time.time() < deadline - 30:
        a0 = np.array(best_seq, dtype=float)
        print("[Adam] start local refinement")
        a_opt = adam_optimize(a0, deadline - 10, tau_start=0.5, tau_end=1e-7,

        print(f"[Adam] refined score = {sc_opt:.6f}")
        if sc_opt < best_score:

            best_seq = a_opt.tolist()

    # final polish via subgradient

"""
Search for a non‑negative step‑height vector that minimizes
evaluate_sequence (i.e. 2·n·max(conv(a,a)) for a simplex‑normalised a).

Algorithm
---------
1. Quick fallback to a known construction (if available) and a uniform baseline.
2. Global optimisation of a low‑dimensional “knot” representation using
   SciPy's differential evolution (piecewise‑linear shapes, all knots are
   normalised to a probability vector).  This quickly discovers promising
   edge‑heavy or U‑shaped profiles.
3. Local refinement of the best candidate with a projected Adam optimiser that
   minimises a smooth log‑sum‑exp surrogate of the max‑convolution.
4. Short sub‑gradient polishing on the true objective.
5. Return the best vector found (as a plain Python list of floats).

All helpers are defined at module level to avoid closures; the code respects
the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities (same as the previous implementation)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT (real, power‑of‑2 length)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·conv_k(a,a)   w.r.t. a.
    Returns g_i = 2·Σ_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    m = len(w)
    rev_w = w[::-1]
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def smooth_max(conv: np.ndarray, tau: float) -> (float, np.ndarray):
    """
    Returns the smooth approximation S = τ·log(∑ exp(conv/τ))
    together with the soft‑max weights w (∑ w = 1).
    """
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


# ----------------------------------------------------------------------
# Piecewise‑linear shape utilities (used by the global DE search)
# ----------------------------------------------------------------------


GLOBAL_N = None  # will be set before calling DE


def shape_from_knots(v: np.ndarray, n: int) -> np.ndarray:
    """
    Interpolate a piecewise‑linear probability density from `len(v)` knots.
    The knots are placed uniformly on the index interval [0, n‑1],
    the result is clipped to non‑negative and normalised to sum = 1.
    """
    K = len(v)
    # ensure non‑negativity before interpolation
    v = np.maximum(v, 0.0)
    # positions of the knots
    xs = np.linspace(0, n - 1, K)
    xi = np.arange(n)
    a = np.interp(xi, xs, v)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def shape_score_from_knots(knots: np.ndarray) -> float:
    """Objective passed to DE – lower is better."""
    a = shape_from_knots(knots, GLOBAL_N)
    return compute_score(a)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate (projected onto simplex)
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 0.5,
                  tau_end: float = 1e-7,
                  tau_decay: float = 0.90,
                  lr: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """
    Projected Adam minimising τ·log‑sum‑exp(conv) with exponential τ‑schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)          # soft‑max weights for the surrogate
        grad = conv_with_weights(a, w)        # ∇_a τ·log‑sum‑exp(conv)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 1200 == 0:
            jitter = np.random.randn(n) * (lr * 0.002 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as before, but a bit longer if time permits)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a
