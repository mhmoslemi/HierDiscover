def smooth_optimize(a0, time_deadline):
    a = a0.copy()
    best = a.copy()
    best_val = compute_score(best)  # 2*n*maxconv
    t = 0
    lam = 5.0
    while time.time() < time_deadline:
        t += 1
        # optionally increase lam gradually
        if t % 500 == 0 and lam < 500:
            lam *= 1.25
        conv = full_convolution(a)
        # weight smoothing
        maxc = conv.max()
        # compute normalized weights
        w = np.exp(lam * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum
        # gradient
        a_rev = a[::-1]
        grad = 2 * np.convolve(a_rev, w, mode='full')[len(a)-1 : len(a)-1+len(a)]
        # step size schedule
        step = 0.1 / np.sqrt(t)
        a = a - step * grad
        a = project_to_simplex(a)
        # evaluate
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
    return best
```

"""
Algorithm for minimizing `evaluate_sequence`.

1.  Quick parametric sweep over the reverse‑power‑law family
    a_i ∝ (n‑i)^p to obtain a decent starting shape.
2.  Projected smooth‑max gradient descent.
    The non‑smooth objective  max(conv(a,a)) is approximated by a
    soft‑max  f_λ(a)= (1/λ)·log∑_k exp(λ·conv_k).  Its gradient can be
    computed efficiently with one FFT.  λ is increased gradually so
    the optimisation first follows a smoothed landscape and later
    concentrates on the true maximum.
3.  A short stochastic hill‑climb (mass transfer between two coordinates)
    is applied at the end for final polishing.
All steps respect the overall time budget (≈1000 s).  The returned
sequence contains non‑negative values and is already normalised
to sum = 1 (scaling does not affect the score).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        # tie‑break: uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1): 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (n-i)^{p}   (front‑loaded decreasing power‑law, p ≥ 0).
    The vector is normalised to sum = 1.
    """
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n-1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft approximation of the max)
# ----------------------------------------------------------------------
def gradient_smooth(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation
        f_λ(a) = (1/λ)·log(∑_k exp(λ·conv_k))
    with respect to a, under the constraint Σa = 1.
    The result is a vector of length n.
    """
    n = a.shape[0]

    # 1) convolution a*a
    conv = full_convolution(a)                      # length 2n‑1
    maxc = conv.max()

    # 2) soft‑max weights (probability distribution over the convolution indices)
    w = np.exp(lam * (conv - maxc))                 # ≤ 1, max = 1
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                                   # now Σ w = 1

    # 3) gradient = 2 * (a_rev * w)_{n‑1 … n‑1+n‑1}
    #   where a_rev is a reversed.
    m = 2 * n - 1
    N_fft = next_pow2(3 * n - 2)                     # length ≥ n + m – 1
    a_rev = a[::-1]

    A = np.fft.rfft(a_rev, n=N_fft)
    W = np.fft.rfft(w, n=N_fft)
    G = A * W
    conv_rev = np.fft.irfft(G, n=N_fft)[: 3 * n - 2]   # length 3n‑2

    start = n - 1
    grad = 2.0 * conv_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Smoothed gradient descent on the simplex
# ----------------------------------------------------------------------
def smooth_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the soft‑max surrogate.
    λ (smoothness) is increased gradually; step size follows 1/√t.
    Returns the best vector found before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(best_a)
    t = 0
    lam = 5.0                      # start with a mild smoothing
    step0 = 0.4                    # base step (will be scaled)

    while time.time() < deadline:
        t += 1
        # slowly sharpen the approximation
        if t % 500 == 0:
            lam = min(lam * 1.25, 200.0)

        grad = gradient_smooth(a, lam)
        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0:
            break

        # diminishing step, normalised by gradient magnitude
        step = step0 / (math.sqrt(t) * gnorm)
        a = a - step * grad
        a = project_to_simplex(a)

        # occasional jitter to avoid stagnation
        if t % 2000 == 0:
            a = a + np.random.randn(a.shape[0]) * 0.001
            a = project_to_simplex(a)

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """
    Greedy stochastic moves: transfer a fraction of mass from one index
    to another.  Each accepted move reduces the objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]                 # move a fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        score_cand = compute_score(cand)
        if score_cand < best_score:
            best_score = score_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            # escape possible plateau
            a = best_a + np.random.randn(n) * 0.001
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # 0. Optional fallback (previous construction, if available)
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Very fast parametric sweep over reverse power‑law shapes
    # -------------------------------------------------------------
    n = 2048
    sweep_deadline = deadline - 800.0   # reserve most time for optimisation
    if time.time() < sweep_deadline:
        best_local = None
        best_local_val = float('inf')
        # coarse grid (step ≈0.02)
        for p in np.linspace(0.5, 3.0, 126):
            if time.time() >= sweep_deadline:
                break
            a = reverse_power_law(n, p)
            if a is None:
                continue
            val = evaluate_sequence(a.tolist())
            if val < best_local_val:
                best_local_val, best_local = val, a.tolist()
        # fine grid around the best p found (if any)
        if best_local is not None:
            # retrieve approximate p from the coarse sweep by locating index
            # (for simplicity we just sweep a narrower interval)
            for p in np.linspace(0.8, 2.2, 141):
                if time.time() >= sweep_deadline:
                    break
                a = reverse_power_law(n, p)
                val = evaluate_sequence(a.tolist())
                if val < best_local_val:
                    best_local_val, best_local = val, a.tolist()
        # remember the best candidate from the sweep
        if best_local is not None and best_local_val < best_val:
            best_val, best_seq = best_local_val, best_local
            print(f"[parametric sweep] best score = {best_val:.6f}")

    # -------------------------------------------------------------
    # 2. Smooth‑max gradient descent (bulk of remaining time)
    # -------------------------------------------------------------
    if best_seq is not None:
        a_start = np.array(best_seq, dtype=float)
    else:
        # if everything failed, start from a reasonable reverse power‑law
        a_start = reverse_power_law(n, 1.8)

    grad_deadline = deadline - 80.0
    a_opt = smooth_descent(a_start, grad_deadline)
    val_opt = evaluate_sequence(a_opt.tolist())
    print(f"[smooth descent] score = {val_opt:.6f}")
    if val_opt < best_val:
        best_val, best_seq = val_opt, a_opt.tolist()

    # -------------------------------------------------------------
    # 3. Final stochastic hill‑climb for polishing
    # -------------------------------------------------------------
    if best_seq is not None:
        hill_deadline = deadline - 5.0
        a_hill = hill_climb(np.array(best_seq, dtype=float), hill_deadline)
        val_hill = evaluate_sequence(a_hill.tolist())
        print(f"[hill‑climb] final score = {val_hill:.6f}")
        if val_hill < best_val:
            best_val, best_seq = val_hill, a_hill.tolist()

    # -------------------------------------------------------------
    # 4. Safeguard – if nothing was produced, fallback to a simple shape
    # -------------------------------------------------------------
    if best_seq is None:
        a_fallback = reverse_power_law(n, 1.8)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final report
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
