def simulated_annealing(a_start, time_deadline):
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = len(a)
    T0 = 0.01 * best_M  # initial temperature relative to M
    T = T0
    it = 0
    while time.time() < time_deadline:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        # pick delta as random fraction of source mass
        max_delta = a[i]
        if max_delta == 0:
            continue
        delta = np.random.rand() * max_delta * 0.5  # up to 50% of source
        # produce candidate
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum unchanged, non-negative automatically
        # no need for projection if 0 <= cand[i]
        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
        # Metropolis acceptance
        if M_cand < full_convolution(a).max():
            accept = True
        else:
            dE = M_cand - full_convolution(a).max()
            accept = np.random.rand() < np.exp(-dE / (T + 1e-12))
        if accept:
            a = cand
        # temperature schedule
        it += 1
        # reduce temperature slowly
        if it % 1000 == 0:
            T *= 0.995
            if T < 1e-6:
                T = 1e-6
    return best_a

"""
search_step_function_v3.py

A multi‑phase optimiser for the ``evaluate_sequence`` objective.

Phases
------
1. **Broad parametric sweep** – explores reverse power‑law,
   β‑shapes, geometric and random mixtures on a relatively large
   support (n≈4096).  The best vector from this stage seeds the next.

2. **Smooth max‑gradient descent** – minimises a smooth approximation
   of the max‑convolution using a soft‑max (log‑sum‑exp) with a
   continuation on the temperature λ.  After each step the vector is
   projected onto the probability simplex.

3. **Simulated‑annealing hill‑climb** – starting from the polished
   candidate, performs stochastic mass‑transfer moves that are
   occasionally accepted even if they worsen the objective, with a
   slowly decreasing temperature.  This helps escape local minima.

All vectors are kept non‑negative and normalised (sum = 1), which is
invariant for the evaluation function.  Helper utilities for the
probability‑simplex projection and FFT‑based convolutions are provided
at the top level.

The function ``propose_candidate`` is the entry point required by the
harness and returns a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # tie‑break: uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (returns length len(x)+len(y)-1)."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())            # next power of two ≥ m
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a."""
    return full_convolution_fft(a, a)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law).  p ≥ 0."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    return a / total


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    return a / a.sum()


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    if not (0.0 < q < 1.0):
        return None
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    return a / a.sum()


# ----------------------------------------------------------------------
# Phase 1 – Parametric sweep
# ----------------------------------------------------------------------
def parametric_sweep(n: int, time_deadline: float) -> np.ndarray:
    """Explore a few parametric families and keep the best vector."""
    best_vec = None
    best_val = float('inf')

    # ---- reverse power‑law (core family) ----
    for p in np.linspace(0.5, 4.0, 71):           # step ≈0.05
        if time.time() >= time_deadline:
            return best_vec
        a = reverse_power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # finer search – full‑resolution sweep (cheaper than a separate pass)
    for p in np.linspace(0.5, 4.0, 351):          # step ≈0.01
        if time.time() >= time_deadline:
            break
        a = reverse_power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- β‑shapes (allow U‑shape and centre‑heavy) ----
    alphas = np.linspace(-2.0, 8.0, 27)   # step ≈0.4
    betas = np.linspace(-2.0, 8.0, 27)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= time_deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- geometric series (very close to 1) ----
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 91),
            np.linspace(0.990, 0.9999, 120))):
        if time.time() >= time_deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- random convex combinations of a few distinguished shapes ----
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    pw2 = reverse_power_law(n, 2.0)
    geom = geometric(n, 0.98)
    basics = [uniform, pw2, geom]

    for _ in range(2000):
        if time.time() >= time_deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Phase 2 – Smooth max gradient descent (log‑sum‑exp continuation)
# ----------------------------------------------------------------------
def gradient_descent_smooth(a0: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed objective
        f_λ(a) = (1/λ)·log Σ_k exp(λ·conv_k(a,a))

    λ is increased gradually (continuation) to approach the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0

    # continuation schedule – start gentle, finish sharp
    lam_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0]
    steps_per_lambda = 3000  # will be trimmed by the time budget

    for lam in lam_schedule:
        for _ in range(steps_per_lambda):
            if time.time() >= time_deadline:
                return best_a
            # convolution of a with itself
            C = full_convolution(a)                     # length 2n‑1
            # soft‑max weights
            Cmax = C.max()
            exp_vals = np.exp(lam * (C - Cmax))         # subtract max for stability
            w = exp_vals / exp_vals.sum()               # w_k ≥ 0, Σ w = 1
            # gradient: 2 * (w * a_rev)  (full convolution, then truncate)
            a_rev = a[::-1]
            grad_full = full_convolution_fft(w, a_rev)  # length 3n‑2
            grad = 2.0 * grad_full[:n]                  # keep first n components
            # step size schedule
            t += 1
            step = 0.2 / math.sqrt(t)
            a = a - step * grad
            a = project_to_simplex(a)

            M = full_convolution(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Phase 3 – Simulated‑annealing stochastic hill‑climb
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    Stochastic mass‑transfer moves with Metropolis acceptance.
    The temperature slowly cools, allowing the search to escape local minima.
    """
    a = a_start.copy()
    n = a.shape[0]
    M_current = full_convolution(a).max()
    best_a = a.copy()
    best_M = M_current

    # initial temperature – a small fraction of the current max
    T = max(1e-4, 0.05 * M_current)
    it = 0

    while time.time() < time_deadline:
        # pick a source index with some mass and a distinct target
        i = np.random.randint(n)
        if a[i] == 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of the source mass (up to 50 %)
        delta = np.random.rand() * a[i] * 0.5
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate candidate
        M_cand = full_convolution(cand).max()

        # Metropolis acceptance
        if M_cand < M_current:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(M_cand - M_current) / (T + 1e-12))

        if accept:
            a = cand
            M_current = M_cand
            if M_cand < best_M:
                best_M = M_cand
                best_a = cand.copy()

        # temperature schedule – very slow cooling
        it += 1
        if it % 2000 == 0:
            T *= 0.99
            T = max(T, 1e-8)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Searches for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns a plain Python list of floats."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a few seconds margin for final I/O

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # 0. Optional fallback from a previously found construction
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore[name-defined]
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Parametric sweep (≈ 150 s)
    # -------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    n = 4096                      # a larger support gives a finer approximation
    if time.time() < sweep_deadline:
        a0 = parametric_sweep(n, sweep_deadline)
        if a0 is not None:
            val = evaluate_sequence(a0.tolist())
            print(f"[parametric sweep] official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a0.tolist()

    # -------------------------------------------------------------
    # 2. Smooth‑max gradient descent (≈ 150 s)
    # -------------------------------------------------------------
    gd_deadline = deadline - 120.0
    if best_seq is not None and time.time() < gd_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_gd = gradient_descent_smooth(a_start, gd_deadline)
        val = evaluate_sequence(a_gd.tolist())
        print(f"[smooth GD] polished score = {val:.6f}")
        if val < best_val:

    # 3. Simulated‑annealing hill‑climb (uses the remaining time)

    if best_seq is not None and time.time() < deadline:

        a_final = simulated_annealing(a_start, deadline)
        val = evaluate_sequence(a_final.tolist())

        if val < best_val:

    # -------------------------------------------------------------
    # Final reporting
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
