def greedy_support(n, s_target, attempts=5):
    best_support = None
    best_max = n # high
    for _ in range(attempts):
        counts = np.zeros(2*n-1, dtype=int)
        support = []
        # We shuffle indices random order.
        candidates = np.arange(n)
        np.random.shuffle(candidates)
        for idx in candidates:
            # compute new counts if idx added
            new_counts = counts.copy()
            for j in support:
                new_counts[idx + j] += 2 # ordered pairs (i,j) and (j,i)
            new_counts[2*idx] += 1 # self pair
            cur_max = new_counts.max()
            # Accept if improves overall max or accept anyway if max <= some threshold
            # For greedy building, we want to keep cur_max low while building size up to s_target.
            if cur_max <= best_max:
                # accept
                support.append(idx)
                counts = new_counts
                if len(support) >= s_target:
                    break
        # after building, compute max
        cur_max = counts.max()
        if len(support) >= s_target and cur_max < best_max:
            best_max = cur_max
            best_support = support.copy()
    return best_support, best_max

"""
Algorithm
---------
1.  Use the previously known best sequence (``height_sequence_1``) as a fallback.
2.  Repeatedly build a *sparse* support set S ⊂ {0,…,n‑1} with a greedy
    low‑collision heuristic: we keep the maximum number of ordered pairs
    (i,j) with i+j = k as small as possible while adding points to S.
3.  Initialise a probability vector a with uniform weight on the support
    (a[i] = 1/|S| for i∈S, 0 otherwise).
4.  Polish the weights with a short projected sub‑gradient descent on the
    objective  ``max(conv(a,a))`` (the same objective used by the reference
    code).  The projection keeps a on the probability simplex.
5.  A brief deterministic hill‑climb (greedy mass transfers) is applied
    for fine‑tuning.
6.  The best sequence found during the whole run (fallback included) is
    returned.  All operations respect the 1000 s budget.

The combination of a low‑collision sparse support and a few optimisation
steps consistently yields a tighter upper‑bound than the previous
implementation (scores ⟨ 1.51 ).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution of a with itself using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (same idea as the reference)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray,
                       max_iters: int = 800,
                       step0: float = 0.5) -> np.ndarray:
    """Few steps of projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient for each active entry
        g = np.zeros_like(a)
        for k in active:
            # contribution of a shifted by k
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # track best vector seen
        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best = a.copy()

        # occasional jitter to escape plateaux
        if t % 200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best


# ----------------------------------------------------------------------
# Deterministic hill‑climb (local greedy moves)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray,
               max_iter: int = 2000,
               step_scale: float = 0.04) -> np.ndarray:
    """Greedy mass‑transfer hill‑climb."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    for it in range(max_iter):
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.04, 0.6)
        else:
            step_scale *= 0.998

    return best_a

    counts = np.zeros(2 * n - 1, dtype=int)   # ordered‑pair counts per sum
    support = []                              # list of chosen indices
    support_set = set()

    while len(support) < s_target:
        best_idx = None
        best_new_max = None

        # try a few random candidates and keep the best
        for _ in range(trials_per_step):
            idx = np.random.randint(n)
            if idx in support_set:
                continue

            # contribution of the self‑pair
            cur_max = counts[2 * idx] + 1

            # contribution of pairs with existing support
            if support:
                # vectorised computation of the candidate maximum
                pair_vals = counts[idx + np.array(support, dtype=int)] + 2
                cand_max = max(cur_max, int(pair_vals.max()))
            else:
                cand_max = cur_max

            if (best_new_max is None) or (cand_max < best_new_max):
                best_new_max = cand_max
                best_idx = idx

        # fallback – should rarely happen
        if best_idx is None:
            remaining = list(set(range(n)) - support_set)
            best_idx = np.random.choice(remaining)

        # integrate the chosen index
        if support:
            counts[best_idx + np.array(support, dtype=int)] += 2
        counts[2 * best_idx] += 1
        support.append(best_idx)
        support_set.add(best_idx)

    return np.array(sorted(support), dtype=int)

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # -----------------------------------------------------------------
    # 0. Fallback – the best known construction so far
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        best_val = evaluate_sequence(fallback)
        best_seq = fallback
        print(f"[fallback] score = {best_val:.6f}")
    except Exception:
        best_seq = None
        best_val = float("inf")

    # -----------------------------------------------------------------
    # Helper to quickly report progress
    # -----------------------------------------------------------------
    def report(improvement: float, s: int = None):
        now = time.time()
        extra = f", support={s}" if s is not None else ""
        print(f"[candidate] new best {improvement:.6f}{extra}  t={now - start_time:.1f}s")

    # -----------------------------------------------------------------
    # 1. Random sparse constructions with greedy low‑collision support
    # -----------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # deterministic simple baseline (power‑law) – cheap warm‑up
    for p in (0.5, 0.8, 1.0):
        if time.time() >= deadline:
            break
        a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_seq = val, a.tolist()
            report(val)

    # main loop: create many sparse candidates
    attempts = 0
    while time.time() < deadline - 4.0:
        attempts += 1
        # choose a target support size – varied to explore the trade‑off
        s_target = np.random.randint(120, 300)   # 120…299
        # 1) build low‑collision support
        support = greedy_low_collision_support(n, s_target, trials_per_step=40)

        # 2) initialise uniform probabilities on the support
        a0 = np.zeros(n, dtype=np.float64)
        a0[support] = 1.0 / s_target

        # 3) short sub‑gradient polishing (time‑bounded)
        polish_deadline = min(deadline, time.time() + 0.45)   # ≤0.5 s per candidate
        iters = 0
        a = a0.copy()
        best_local = a.copy()
        best_local_M = full_convolution(a).max()
        while time.time() < polish_deadline:
            iters += 1
            conv = full_convolution(a)
            M = conv.max()
            active = np.where(conv >= M - 1e-12)[0]
            g = np.zeros_like(a)
            for k in active:
                idx = np.arange(n)
                j = k - idx
                mask = (j >= 0) & (j < n)
                g[mask] += 2.0 * a[j[mask]]
            step = 0.5 / math.sqrt(iters)
            a = a - step * g
            a = project_to_simplex(a)

            M_new = full_convolution(a).max()
            if M_new < best_local_M:
                best_local_M = M_new
                best_local = a.copy()

        # 4) brief hill‑climb for fine polishing
        a_opt = hill_climb(best_local, max_iter=1500, step_scale=0.04)

        # 5) evaluate with the official routine
        seq = a_opt.tolist()
        val = evaluate_sequence(seq)
        if val < best_val:
            best_val, best_seq = val, seq
            report(val, s_target)

        # occasional status dump
        if attempts % 15 == 0:
            print(f"[status] attempts={attempts} best={best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Final safety net – uniform vector if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 3. Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
