"""
Propose a step‑height sequence that lowers the objective
2·n·max(conv(a,a))/(sum(a)^2).

Core ideas
---------
*   The average of the self‑convolution is fixed ( = 1/(2n‑1) after
    normalising the sum to 1 ).  If the convolution were perfectly flat,
    the objective would be 1 – the theoretical optimum.
*   We therefore minimise the **L2 distance** of the convolution to this
    constant vector.  The loss
        L(a) = ‖a*a – c‖₂² ,   c_i = 1/(2n‑1)
    is smooth; its gradient can be written in closed form and computed
    by a single FFT‑based convolution.
*   A projected Adam optimiser (with the simplex projection) drives the
    sequence toward a flatter convolution.
*   The result is then polished with the original sub‑gradient method
    and a short simulated‑annealing hill‑climb, exactly as in the
    previous state‑of‑the‑art code.
*   Several parametric families (including a new “arcsine’’ shape) and a
    pool of random Dirichlet draws are explored to obtain diverse
    starting points.

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT (real‑valued)."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1 - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[: n + m - 1]


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (self‑convolution)."""
    return fft_convolve(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_ushape(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_arcsine(n: int) -> np.ndarray:
    """Arcsine‑type density: a_i ∝ 1/√(i·(n+1−i)).  Heavily weighted at the ends."""
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(i * (n + 1 - i))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Exploration – keep a few promising vectors
# ----------------------------------------------------------------------


def explore_candidates(n: int, deadline: float, keep: int = 12) -> list[np.ndarray]:
    """Generate a wide range of vectors and keep the best ``keep`` of them."""
    best = []  # list of (score, vector)

    def consider(v: np.ndarray):
        if not np.isfinite(v).all():
            return
        v = np.maximum(v, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        if len(best) < keep:
            best.append((sc, v.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, v.copy())
            best.sort(key=lambda x: x[0])

    # 0) uniform – a useful reference
    consider(np.full(n, 1.0 / n))

    # 1) power‑law (both decreasing and increasing)
    for p in np.linspace(0.0, 4.0, 41):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))
        consider(generate_powerlaw(n, -p))

    # 2) exponential sweep
    for lam in np.linspace(-0.3, 0.3, 61):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # 3) beta family
    al_vals = np.linspace(0.1, 5.0, 21)
    be_vals = np.linspace(0.1, 5.0, 21)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    # 4) Gaussian widths
    for sigma in np.linspace(0.05 * n, 0.5 * n, 15):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # 5) sine‑power shapes
    for p in np.linspace(0.1, 8.0, 20):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # 6) U‑shape (edge‑heavy)
    for p in np.linspace(0.1, 5.0, 20):
        if time.time() >= deadline:
            break
        consider(generate_ushape(n, p))

    # 7) arcsine shape (new)
    consider(generate_arcsine(n))

    # 8) random Dirichlet draws with a variety of concentration parameters
    concentrations = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws = max(1, (keep * 250) // len(concentrations))
    for conc in concentrations:
        for _ in range(draws):
            if time.time() >= deadline:
                break
            consider(np.random.dirichlet([conc] * n))

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser on the **L2‑flattening** surrogate
# ----------------------------------------------------------------------


def adam_optimize_l2(a0: np.ndarray, deadline: float,
                     lr: float = 0.05,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8) -> np.ndarray:
    """
    Projected Adam minimising ‖a*a – c‖₂² where c is the flat target.
    Returns the best vector (in terms of the real objective) seen up to
    ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]
    const_val = 1.0 / (2 * n - 1)
    const_vec = np.full(2 * n - 1, const_val)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_obj = compute_score(a)

    while time.time() < deadline:
        # ----- compute loss and gradient -----
        conv = full_convolution(a)            # length 2n‑1
        err = conv - const_vec                # deviation from flatness
        # (objective for Adam) – L2 loss
        # loss = np.dot(err, err)   # not used directly, only for diagnostics
        # gradient of L2 loss: 4 * (err  *  a_rev) restricted to n entries
        grad = 4.0 * fft_convolve(err, a[::-1])[n - 1:n - 1 + n]

        # ----- Adam update (projected) -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # ----- keep the best according to the true objective -----
        cur_obj = compute_score(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # occasional jitter to help escape flat regions
        if t % 2000 == 0:
            a += np.random.randn(n) * (lr * 0.001 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as before)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb (final refinement)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can escape shallow local minima."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            a += np.random.randn(n) * (step_scale * 0.01)
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑height vector that minimises
    ``evaluate_sequence``.  The routine follows the three‑phase scheme
    (exploration → L2‑Adam optimisation → polishing → annealing) while
    respecting the wall‑clock budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # small safety margin

    n = 2048                                     # length of the step function
    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Fallback – use the previously known construction if it exists
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad exploration (≈ 120 s)
    # -----------------------------------------------------------------
    explore_deadline = deadline - 600.0
    if time.time() < explore_deadline:
        print("[explore] generating candidates …")
        cand_vecs = explore_candidates(n, explore_deadline, keep=12)
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_score:
                best_score, best_seq = val, vec.tolist()
        print(f"[explore] best score after exploration = {best_score:.6f}")
    else:
        cand_vecs = []

    # -----------------------------------------------------------------
    # 2. Adam optimisation on the L2‑flattening loss (≈ 400 s)
    # -----------------------------------------------------------------
    if best_seq is not None:
        grad_deadline = min(deadline - 250.0,
                            start + 150.0 + (deadline - start) * 0.35)
        start_points = []

        # a) best vector from exploration
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        start_points.append(best_vec)

        # b) uniform start
        start_points.append(np.full(n, 1.0 / n))

        # c) a few pure random starts
        for _ in range(3):
            start_points.append(project_to_simplex(np.random.rand(n)))

        # d) top candidates from exploration if any
        for vec in cand_vecs[:3]:
            start_points.append(vec)

        refined_best = best_vec.copy()
        refined_val = best_score

        for idx, a0 in enumerate(start_points):
            if time.time() >= grad_deadline:
                break
            local_deadline = grad_deadline - (time.time() - start) - 0.5
            if local_deadline <= time.time():
                break
            print(f"[adam L2 {idx}] start")
            a_opt = adam_optimize_l2(a0, local_deadline, lr=0.07)
            val = compute_score(a_opt)
            print(f"[adam L2 {idx}] objective = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_opt.copy()

        best_score, best_seq = refined_val, refined_best.tolist()
        print(f"[adam L2] best after L2‑Adam = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[polish] start")
        a_polished = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing (final tweak, ≈ 80 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[anneal] start")
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
