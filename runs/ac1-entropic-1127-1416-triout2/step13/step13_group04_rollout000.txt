def anneal_search_improved(a_start, deadline):
    n = len(a_start)
    cur = a_start.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    # initial temperature based on cur_score
    temp = max(1e-4 * cur_score, 1e-6)
    decay = 0.9996
    step_scale = 0.07
    step_decay = 0.9999
    no_improve_counter = 0
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1
        # randomly decide move type
        if np.random.rand() < 0.8:
            # simple transfer
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
            delta = step_scale * cur[i] * np.random.rand()
            if delta <= 0:
                continue
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # block transfer
            block_len = np.random.randint(1, 5)  # up to 4
            i_start = np.random.randint(0, n - block_len + 1)
            j_start = np.random.randint(0, n - block_len + 1)
            # ensure they don't overlap? not needed
            # compute total mass in source block:
            src_mass = cur[i_start:i_start+block_len].sum()
            if src_mass <= 0:
                continue
            # transfer fraction
            frac = step_scale * np.random.rand()
            delta_block = src_mass * frac
            # transfer proportionally across source indices:
            src = cur[i_start:i_start+block_len] * (delta_block / src_mass)
            cand = cur.copy()
            cand[i_start:i_start+block_len] -= src
            # distribute delta equally among target block
            cand[j_start:j_start+block_len] += delta_block / block_len
        # ensure nonneg, projection
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        delta_score = cand_score - cur_score
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-delta_score / max(temp, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                # increase step_scale slightly
                step_scale = min(step_scale * 1.02, 0.7)
                no_improve_counter = 0
                print(f"[anneal] new best {best_score:.6f} iter={iter_cnt}")
            else:
                no_improve_counter += 1
        else:
            no_improve_counter += 1

        # update temperature and step_scale
        temp = max(temp * decay, 1e-12)
        step_scale = max(step_scale * step_decay, 1e-7)

        if no_improve_counter > 200000:
            # random perturbation
            jitter = np.random.randn(n) * 0.001
            cur = best + jitter
            cur = project_to_simplex(cur)
            cur_score = compute_score(cur)
            no_improve_counter = 0

    return best
```

"""
Advanced stochastic optimisation for the self‑convolution bound.

The optimiser works in four stages:

1. **Broad parametric sweep** – a collection of analytical families
   (power‑law, geometric, mixed plateau+power, beta, logistic,
   shifted‑power, exponential‑cutoff, power‑geometric) is explored.
   The ten best normalised vectors (Σa=1) are retained.

2. **Momentum sub‑gradient polishing** – starting from the current best
   vector we perform a heavy‑ball sub‑gradient descent on the maximal
   convolution entry.  Projection onto the probability simplex keeps the
   sum fixed and the non‑negativity constraint satisfied.

3. **Multi‑restart simulated annealing** – each of the retained vectors
   seeds an independent annealing run.  Moves consist of single‑mass
   transfers as well as small block transfers; temperature decays slowly
   and an adaptive step‑size is used.  The best result of all restarts
   is kept.

4. **Deterministic hill‑climb** – a final greedy mass‑transfer pass
   polishes the current best solution.

All time budgets are respected via explicit deadline checks, and the
final list of non‑negative heights is returned as a plain Python list.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())        # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def geometric(n: int, q: float) -> np.ndarray:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def logistic_shape(n: int, center_frac: float, scale_frac: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    scale = max(scale_frac * n, 1e-6)
    a = 1.0 / (1.0 + np.exp((i - center) / scale))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def shifted_power(n: int, p: float, shift: int) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^(-p)"""
    idx = np.arange(shift, shift + n, dtype=np.float64) + 1.0
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def exp_power(n: int, p: float, tau: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) * exp(-i/tau)"""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-p) * np.exp(-i / tau)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def power_geometric(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) * q**i"""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-p) * (q ** i)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc)."""
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    s = y.sum()
    if s == 0.0:
        return None
    return y / s

        if len(best) < top_k:
            best.append((score, a.copy()))
            best.sort(key=lambda x: x[0])
        else:

                best[-1] = (score, a.copy())
                best.sort(key=lambda x: x[0])

    # ----- power‑law -------------------------------------------------
    for p in np.linspace(0.10, 2.40, 231):

            return best
        consider(power_law(n, p))

    # ----- geometric -------------------------------------------------
    for q in np.concatenate((
            np.linspace(0.85, 0.97, 81),

        if time.time() >= deadline:

        consider(geometric(n, q))

    # ----- mixed plateau + power ------------------------------------
    for L_frac in np.linspace(0.02, 0.35, 15):
        for p in np.linspace(0.30, 1.00, 36):
            if time.time() >= deadline:

            consider(mixed_plateau_power(n, L_frac, p))

    # ----- beta shape ------------------------------------------------
    alphas = np.linspace(-0.8, 3.2, 22)

    for alpha in alphas:
        for beta in betas:

    for q in np.linspace(0.90, 0.999, 55):

# Momentum sub‑gradient polishing
# ----------------------------------------------------------------------
def subgradient_polish_improved(a0: np.ndarray, deadline: float,
                                step0: float = 0.12) -> np.ndarray:
    """
    Heavy‑ball sub‑gradient descent on the maximal convolution entry.
    The iterate is always projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    t = 0
    momentum = np.zeros_like(a)
    beta = 0.85                     # momentum coefficient

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of each active entry: 2 * a shifted
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        momentum = beta * momentum + step * g
        a = a - momentum
        a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a.copy()

        # occasional jitter to escape shallow plateaux
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

# Simulated annealing – multi‑restart, block moves
# ----------------------------------------------------------------------
def anneal_search_improved(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing style stochastic optimisation.
    Supports single‑mass and small block transfers.
    """
    n = a_start.shape[0]
    cur = a_start.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    # temperature starts as a small fraction of the current score
    temp = max(1e-4 * best_score, 1e-8)
    temp_decay = 0.9996

    step_scale = 0.07      # typical fraction of source mass to move
    step_decay = 0.9999

    it = 0
    no_progress = 0

    while time.time() < deadline:
        it += 1
        # choose move type
        if np.random.rand() < 0.78:
            # --- single mass transfer ---
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = step_scale * cur[i] * np.random.rand()
            if delta <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # --- block transfer (1–4 consecutive sources) ---
            blk_len = np.random.randint(1, 5)
            i0 = np.random.randint(0, n - blk_len + 1)
            j0 = np.random.randint(0, n - blk_len + 1)
            src_mass = cur[i0:i0 + blk_len].sum()
            if src_mass <= 0.0:
                continue
            frac = step_scale * np.random.rand()
            delta_blk = src_mass * frac
            # take proportionally from the source block
            src = cur[i0:i0 + blk_len] * (delta_blk / src_mass)
            cand = cur.copy()
            cand[i0:i0 + blk_len] -= src
            # spread uniformly to the target block
            cand[j0:j0 + blk_len] += delta_blk / blk_len

        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            accept = (np.random.rand() < prob)

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                step_scale = min(step_scale * 1.02, 0.6)
                no_progress = 0
                print(f"[anneal] new best {best_score:.6f}  it={it}")
            else:
                no_progress += 1
        else:
            no_progress += 1

        # temperature and step decay
        temp = max(temp * temp_decay, 1e-12)
        step_scale = max(step_scale * step_decay, 1e-7)

        # if stuck for a while, inject a small random jitter
        if no_progress > 150_000:
            jitter = np.random.randn(n) * 0.001
            cur = best + jitter
            cur = project_to_simplex(cur)
            cur_score = compute_score(cur)
            no_progress = 0

    return best

# Deterministic hill‑climb (greedy)
# ----------------------------------------------------------------------
def hill_climb_improved(a_start: np.ndarray, deadline: float,
                        max_iter: int = 900_000) -> np.ndarray:
    """Greedy mass‑transfer hill climbing with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.025
    it = 0
    stagnation = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M - 1e-12:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.6)
            stagnation = 0
        else:
            step_scale *= 0.996
            stagnation += 1

        it += 1
        if stagnation > 180_000:
            # escape flat region
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            stagnation = 0

    return best_a

    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Parametric sweep (≈ 250–300 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 350.0
    n = 4096
    if time.time() < sweep_deadline:
        top_candidates = parametric_sweep_extended(n, sweep_deadline, top_k=10)
        if top_candidates:
            # the very best from the sweep
            best_sweep_score, best_sweep_vec = top_candidates[0]
            print(f"[parametric sweep] best score = {best_sweep_score:.6f}")
            if best_sweep_score < best_val:
                best_val = best_sweep_score
                best_seq = best_sweep_vec.tolist()
        else:
            top_candidates = []
    else:
        top_candidates = []

    # -----------------------------------------------------------------
    # 2. Momentum sub‑gradient polishing (≈ 120–150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 250.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish_improved(a0, polish_deadline, step0=0.12)
        val = compute_score(a_polished)
        print(f"[sub‑grad] polished score = {val:.6f}")

            best_val = val
            best_seq = a_polished.tolist()

    # -----------------------------------------------------------------
    # 3. Multi‑restart simulated annealing (≈ 250–300 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 120.0
    if time.time() < anneal_deadline:
        # Build a list of start vectors: best from sweep + the current best
        starts = []
        for _, vec in top_candidates[:5]:
            starts.append(np.array(vec, copy=True))
        if best_seq is not None:
            starts.append(np.array(best_seq, copy=True))

        # Run independent annealing runs, each respecting the remaining budget
        for a_start in starts:
            if time.time() >= anneal_deadline:
                break
            a_best = anneal_search_improved(a_start, anneal_deadline)
            val = compute_score(a_best)
            print(f"[anneal‑run] score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_best.tolist()

    # -----------------------------------------------------------------
    # 4. Final deterministic hill‑climb (remaining time)
    # -----------------------------------------------------------------
    hill_deadline = deadline
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb_improved(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()
    else:
        # As a last resort, fall back to a uniform distribution
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report with the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
