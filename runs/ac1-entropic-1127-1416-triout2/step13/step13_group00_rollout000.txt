"""
Iterative linear‑programming refinement for the step‑height problem.

The objective to minimise is  

    2·n·max(conv(a,a)) / (sum(a)**2)  

which is scale‑invariant (ignoring the clipping in ``evaluate_sequence``).
Hence we normalise every candidate to a probability vector (sum = 1).

Algorithm
---------
1. Start from a reasonable non‑negative vector (the fallback ``height_sequence_1`,
   otherwise the uniform distribution) and normalise it.
2. Repeatedly solve the linear programme  

        maximise   1ᵀb
        subject to b ≥ 0,
                   conv(f,b) ≤ M

   where ``f`` is the current vector and ``M = max(conv(f,f))``.
   The optimal ``b`` carries more total mass than ``f`` while respecting the same
   peak‑convolution bound.  After normalising ``b`` back to a probability vector we
   move a small step towards it:

        f ← (1‑t)·f + t·b_norm   (t ≈ 0.2)

   This strictly reduces the peak of the autocorrelation (theoretical guarantee
   from the original “alternating LP” method).  The loop stops when the time
   budget expires or improvements become negligible.
3. The best vector obtained is polished with a projected sub‑gradient descent
   on the exact objective (as in the previous baseline).
4. Finally the vector is returned as a plain Python list of floats.

If ``cvxpy`` is unavailable the programme falls back to the earlier Adam‑+‑polish
pipeline, guaranteeing a valid answer even in restricted environments.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (requires cvxpy)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVXPY_AVAILABLE = False


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀb   subject to b ≥ 0, conv(f,b) ≤ M   (full linear conv).
    Returns the optimal vector ``b`` (may have any total mass) or ``None`` on failure.
    """
    if not _CVXPY_AVAILABLE:
        return None

    n = f.shape[0]
    # constant parameters
    f_param = cp.Parameter(n, nonneg=True)
    f_param.value = f
    M_param = cp.Parameter(nonneg=True)
    M_param.value = M

    # variable
    b = cp.Variable(n, nonneg=True)

    # convolution (linear because f is a Parameter)
    conv_fb = cp.conv(f_param, b)           # length 2n‑1
    constraints = [conv_fb <= M_param]      # element‑wise inequality

    # maximise total mass of b
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:                       # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None

    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)


def _lp_refine(start_f: np.ndarray,
               deadline: float,
               t_step: float = 0.20,
               max_iter: int = 200) -> tuple[np.ndarray, float]:
    """
    Iteratively improve ``start_f`` by the alternating LP scheme.
    Returns the best vector found and its score.
    """
    f = start_f.copy()
    best_f = f.copy()
    best_val = compute_score(f)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = full_convolution(f)
        M = conv.max()

        b = _solve_lp(f, M)
        if b is None:
            # LP could not be solved – give up the refinement loop.
            break

        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # Normalise b to a probability vector (same sum as f, which is 1).
        b_norm = b / sum_b

        # Convex combination – stays on the simplex.
        f_new = (1.0 - t_step) * f + t_step * b_norm

        new_val = compute_score(f_new)
        if new_val < best_val:
            best_val = new_val
            best_f = f_new.copy()
            print(f"[LP] iter {it:3d} – new best score {best_val:.6f}")

        # Accept the new vector for the next iteration.
        f = f_new

        # If progress stalls, shrink the step size.
        if it % 20 == 0 and new_val >= best_val * 0.9999:
            t_step = max(t_step * 0.5, 0.02)

    return best_f, best_val


# ----------------------------------------------------------------------
# Sub‑gradient polishing (identical to the previous baseline)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # Compute a sub‑gradient of the max‑convolution.
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # Occasional jitter to escape shallow basins.
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    n = 2048
    best_seq: list[float] | None = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Quick fallback using the previously known construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Initialise a probability vector (sum = 1)
    # -----------------------------------------------------------------
    if best_seq is not None:
        init_vec = np.array(best_seq, dtype=float)
    else:
        init_vec = np.full(n, 1.0 / n, dtype=float)   # uniform start

    init_vec = np.maximum(init_vec, 0.0)
    init_vec = init_vec / init_vec.sum()            # ensure sum = 1

    # -----------------------------------------------------------------
    # 2. Linear‑programming alternating refinement (≈ 600 s)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 250.0                  # leave time for polishing
    if time.time() < lp_deadline:
        print("[LP] start alternating refinement")
        f_opt, f_score = _lp_refine(init_vec, lp_deadline,
                                    t_step=0.20, max_iter=120)
        seq = f_opt.tolist()
        val = evaluate_sequence(seq)
        print(f"[LP] evaluated score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, seq

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ remaining budget)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 5.0
    if best_seq is not None and time.time() < polish_deadline:
        print("[polish] start")
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[polish] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – uniform vector if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
