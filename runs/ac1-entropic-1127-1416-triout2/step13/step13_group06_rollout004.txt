def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both a and b are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by the optimiser (sum(a) is assumed ≈1)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (∑v=1, v≥0).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families (all return a normalised vector of length n)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (n‑½‑i)^{‑β} (mirror of left‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred on the middle."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def double_exponential_family(n: int, gamma: float) -> np.ndarray:
    """Sum of two exponentials decaying from both ends."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = np.exp(-gamma * (n - 1 - i))
    w = left + right
    # protect against all‑zero (gamma huge)
    if w.max() == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w / w.sum()

def adam_smoothing_descent(start_a: np.ndarray,
                           deadline: float,
                           max_iters: int = 250_000,
                           eps_start: float = 1e-3,
                           eps_end: float = 2e-4,
                           lr: float = 0.02,
                           beta1: float = 0.9,
                           beta2: float = 0.999) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective
        Fε(a)=2·n·ε·log(∑exp((a*a)_k/ε)),
    with ε decreasing from eps_start to eps_end.
    The iterate is kept on the simplex via Euclidean projection.
    """
    n = start_a.size
    a = normalize(start_a.copy())
    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # geometric schedule for ε
    if max_iters > 1:
        log_eps_ratio = math.log(eps_end / eps_start)
    else:
        log_eps_ratio = 0.0

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current smoothing parameter
        eps = eps_start * math.exp(log_eps_ratio * (t - 1) / (max_iters - 1))

        # convolution a*a
        conv = convolution_fft(a, a)                     # length 2n‑1

        # soft‑max weights
        w = np.exp(conv / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed loss (ignoring the constant term)
        # g = 2 * sum_i w_i * d conv_i/da  = 2 * 2 * conv_rev[n‑1:2n‑1]
        w_rev = w[::-1]
        conv_rev = convolution_fft(w_rev, a)            # length 3n‑2
        # slice that corresponds to the needed indices
        g_slice = conv_rev[n - 1:2 * n - 1]              # size n
        grad = 2.0 * g_slice                             # factor 2 from derivative

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(v_hat) + 1e-8)

        a = a - step * m_hat
        a = project_to_simplex(a)

        # occasional real‑objective check
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] Adam iter {t:6d} ε={eps:.2e} improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 12_000_000,
                  temp_factor: float = 1.2e-3,
                  step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()

        # move type: swap two entries or transfer a fraction
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # 5 seconds safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 5.0

    best_seq: List[float] = []
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0.  Fallback to the previously known solution (if it exists)
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1.  Choose working discretisation
    # -----------------------------------------------------------
    n = 4096                     # moderate size – enough freedom, still fast

    # -----------------------------------------------------------
    # 2.  Coarse scan of analytic families (very cheap)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] coarse analytic scan (n={n})")
    families = []

    # product (U‑shaped)
    families.append(("product", lambda a: symmetric_product_family(n, a), np.linspace(0.0, 12.0, 25)))
    # centre‑heavy power
    families.append(("center_power", lambda a: symmetric_center_power_family(n, a), np.linspace(0.0, 12.0, 25)))
    # left‑heavy power
    families.append(("left_power", lambda a: left_heavy_power_family(n, a), np.linspace(0.0, 12.0, 25)))
    # right‑heavy power
    families.append(("right_power", lambda a: right_heavy_power_family(n, a), np.linspace(0.0, 12.0, 25)))
    # Gaussian centre
    families.append(("gaussian", lambda s: gaussian_center_family(n, s),
                     np.linspace(0.2 * n, 0.5 * n, 15)))
    # new double‑exponential family
    families.append(("double_exp", lambda g: double_exponential_family(n, g),
                     np.linspace(0.0, 5.0, 20)))

    for name, generator, grid in families:
        for p in grid:
            if time.time() >= deadline - 30.0:
                break
            vec = generator(p)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best from {name} ({p:.3f}) = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3.  Random dense search (≈4000 trials)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] random dense search")
    trials = 0
    max_trials = 4000
    family_funcs = [
        lambda: symmetric_product_family(n, np.random.uniform(0.0, 12.0)),
        lambda: symmetric_center_power_family(n, np.random.uniform(0.0, 12.0)),
        lambda: left_heavy_power_family(n, np.random.uniform(0.0, 12.0)),
        lambda: right_heavy_power_family(n, np.random.uniform(0.0, 12.0)),
        lambda: gaussian_center_family(n, np.random.uniform(0.2 * n, 0.5 * n)),
        lambda: double_exponential_family(n, np.random.uniform(0.0, 5.0)),
    ]

    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        vec = family_funcs[np.random.randint(len(family_funcs))]()

        # Occasionally apply a random zero‑padding shift (break symmetry)
        if np.random.rand() < 0.15:
            d = np.random.randint(0, n // 30 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 4.  Smoothed gradient descent with Adam
    # -----------------------------------------------------------
    if best_seq and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] starting Adam smoothed‑gradient descent")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smoothing_descent(
            start_a=a_start,
            deadline=deadline - 150.0,
            max_iters=200_000,
            eps_start=1e-3,
            eps_end=2e-4,
            lr=0.02,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5.  Final simulated‑annealing hill‑climb
    # -----------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=12_000_000,
            temp_factor=1.2e-3,
            step_scale_init=0.25,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6.  Fallback to a safe uniform vector (should never happen)
    # -----------------------------------------------------------
    if not best_seq:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (security checks applied)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
