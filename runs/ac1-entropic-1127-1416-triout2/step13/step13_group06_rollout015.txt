"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

using a multi‑stage search:

1.  Scan a richer set of analytic families (including a regularised
    “offset‑product” family) on a fine grid.
2.  Random dense search over the same families plus random zero‑padding.
3.  Local golden‑section refinement of the currently best one‑parameter
    family.
4.  Smoothed‑max (log‑sum‑exp) gradient descent with exponentiated
    updates (KL‑mirror descent).  The gradient is obtained via a single
    FFT‑based cross‑correlation, giving a cheap estimate of the full
    sub‑gradient of the max‑convolution.
5.  A final simulated‑annealing hill‑climb to polish the result.
All steps respect the supplied wall‑clock budget (default 1000 s).  The
returned list is a plain Python list of floats ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, scoring, projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 0.5 - i)
    if alpha == 0.0:
        w = np.ones(n, dtype=np.float64)
    else:
        # avoid overflow/underflow by working in log‑space
        logw = -alpha * np.log(prod)
        w = np.exp(logw)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def offset_product_family(n: int, alpha: float, offset: float) -> np.ndarray:
    """Regularised U‑shape: a_i ∝ ( (i+½)(n‑½‑i) + offset )^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 0.5 - i)
    base = prod + offset
    if alpha == 0.0:
        w = np.ones(n, dtype=np.float64)
    else:
        w = np.power(base, -alpha)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 0.5 - i)
    expo = np.clip(alpha * np.log(prod), -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (n‑½‑i)^{‑β} (mirror of left‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred on the middle of the interval."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """
    Symmetric triangular bump.
    width_frac ∈ (0,1] controls half‑width relative to n.
    """
    half_width = max(1, int(width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return w / w.sum()


def exp_decay_family(n: int, gamma: float, left: bool = True) -> np.ndarray:
    """Exponential decay from one side."""
    i = np.arange(n, dtype=np.float64)
    if left:
        w = np.exp(-gamma * i)
    else:
        w = np.exp(-gamma * (n - 1 - i))
    return w / w.sum()


def mixed_uniform_gaussian_family(n: int, lam: float, sigma: float) -> np.ndarray:
    """Convex mixture of uniform and centred Gaussian."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    gaussian = gaussian_center_family(n, sigma)
    w = lam * uniform + (1.0 - lam) * gaussian
    return w / w.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / s
    return w

# Gradient utilities (soft‑max of the convolution)
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Return normalized soft‑max weights for a given λ."""
    # shift by max to avoid overflow
    shift = conv.max()
    w = np.exp(lam * (conv - shift))
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def cross_corr_gradient(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_j w_{i+j} a_j  (linear cross‑correlation).
    Returned vector has length len(a).
    """
    L = w.shape[0]
    n = a.shape[0]
    N = next_pow2(L + n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    corr = corr[: L + n - 1]
    # slice n‑1 … n‑1+n‑1
    return 2.0 * corr[n - 1 : n - 1 + n]


def softmax_exp_gradient_descent(start_a: np.ndarray,
                                 deadline: float,
                                 max_iters: int = 200_000,
                                 eta0: float = 0.5,
                                 lam_start: float = 10.0,
                                 lam_factor: float = 30.0) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a log‑sum‑exp smooth approximation
    of the max‑convolution.  The smoothing parameter λ grows linearly
    from lam_start to lam_start*lam_factor.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        # λ schedule
        lam = lam_start * (1.0 + (t - 1) / max_iters * (lam_factor - 1.0))

        # soft‑max weights
        w = softmax_weights(conv, lam)

        # gradient of the smoothed objective
        g = cross_corr_gradient(w, a)

        # exponentiated‑gradient step (mirror descent w.r.t. KL)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional true‑objective evaluation
        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] soft‑max‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] soft‑max‑grad final improvement to {best_score:.6f}")

    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 12_000_000,
                  temp_factor: float = 1.2e-3,
                  step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """Simple SA on the simplex (same as previous implementation)."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 12) -> Tuple[float, float]:
    """Golden‑section line‑search for the minimum of func on [lo, hi]."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local golden‑section refinement of the best family (if applicable)
    # ------------------------------------------------------------------
    if best_family is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining best family {best_family}")

        if best_family == "center_power":
            cur_alpha = best_params["alpha"]
            lo, hi = max(0.0, cur_alpha - 0.8), min(12.0, cur_alpha + 0.8)

            def f_alpha(a):
                return compute_score(symmetric_center_power_family(n, a))

            new_alpha, _ = golden_section_optimize(f_alpha, lo, hi, max_iter=14)
            vec = symmetric_center_power_family(n, new_alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["alpha"] = new_alpha
                print(f"[{time.time():.1f}] refined centre‑α={new_alpha:.4f} → {best_score:.6f}")

        elif best_family == "gaussian":
            cur_sigma = best_params["sigma"]
            lo, hi = max(0.2 * n, cur_sigma - 0.2 * n), min(0.5 * n, cur_sigma + 0.2 * n)

            def f_sigma(s):
                return compute_score(gaussian_center_family(n, s))

            new_sigma, _ = golden_section_optimize(f_sigma, lo, hi, max_iter=14)
            vec = gaussian_center_family(n, new_sigma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["sigma"] = new_sigma
                print(f"[{time.time():.1f}] refined σ={new_sigma:.1f} → {best_score:.6f}")

        elif best_family == "triangular":
            cur_w = best_params["width_frac"]
            lo, hi = max(0.05, cur_w - 0.2), min(0.95, cur_w + 0.2)

            def f_w(w):
                return compute_score(triangular_center_family(n, w))

            new_w, _ = golden_section_optimize(f_w, lo, hi, max_iter=14)
            vec = triangular_center_family(n, new_w)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["width_frac"] = new_w
                print(f"[{time.time():.1f}] refined tri‑w={new_w:.4f} → {best_score:.6f}")

        elif best_family == "product":
            cur_alpha = best_params["alpha"]
            lo, hi = max(0.0, cur_alpha - 0.6), min(12.0, cur_alpha + 0.6)

            def f_alpha(a):
                return compute_score(symmetric_product_family(n, a))

            new_alpha, _ = golden_section_optimize(f_alpha, lo, hi, max_iter=14)
            vec = symmetric_product_family(n, new_alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["alpha"] = new_alpha
                print(f"[{time.time():.1f}] refined product‑α={new_alpha:.4f} → {best_score:.6f}")

        elif best_family == "offset_product":
            cur_alpha = best_params["alpha"]
            cur_off = best_params["offset_frac"]
            lo_a, hi_a = max(0.0, cur_alpha - 0.5), min(12.0, cur_alpha + 0.5)
            lo_o, hi_o = max(0.0, cur_off - 0.5), min(4.0, cur_off + 0.5)

            def f_pair(x):
                a, o = x
                offset = o * base_offset
                return compute_score(offset_product_family(n, a, offset))

            # simple random local search (no golden section for two params)
            for _ in range(200):
                cand_a = np.clip(cur_alpha + np.random.randn() * 0.1, 0.0, 12.0)
                cand_o = np.clip(cur_off + np.random.randn() * 0.1, 0.0, 4.0)
                sc = f_pair((cand_a, cand_o))
                if sc < best_score:
                    best_score, best_seq = sc, offset_product_family(n, cand_a, cand_o * base_offset).tolist()
                    cur_alpha, cur_off = cand_a, cand_o
                    print(f"[{time.time():.1f}] refined offset‑α={cand_a:.3f},off={cand_o:.3f} → {best_score:.6f}")

        elif best_family == "beta":
            p0, q0 = best_params["p"], best_params["q"]
            for _ in range(200):
                if time.time() >= deadline - 5.0:
                    break
                p = np.clip(p0 + np.random.randn() * 0.2, 0.1, 5.0)
                q = np.clip(q0 + np.random.randn() * 0.2, 0.1, 5.0)
                vec = beta_family(n, p, q)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    best_params["p"], best_params["q"] = p, q
                    print(f"[{time.time():.1f}] refined beta p={p:.3f},q={q:.3f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smoothed‑max exponentiated‑gradient descent refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] soft‑max gradient descent refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = softmax_exp_gradient_descent(
            start_a=a_start,
            deadline=deadline - 150.0,
            max_iters=250_000,
            eta0=0.6,
            lam_start=10.0,
            lam_factor=30.0,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after soft‑max grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=12_000_000,
            temp_factor=1.2e-3,
            step_scale_init=0.25,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Fallback to a safe uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 7. Final official evaluation (includes security checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
