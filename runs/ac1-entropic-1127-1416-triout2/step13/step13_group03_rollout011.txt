"""
Non‑negative step‑function optimiser – genetic+gradient hybrid.

The optimiser works with a configurable length `n` (default 4096) and
maintains a small population of candidate sequences.  Candidates are
generated from a variety of parametric families, random Dirichlet draws,
and convex mixtures thereof.  Evolution proceeds by convex recombination,
Gaussian mutation and projection onto the probability simplex.
Periodically the current best is polished with a smooth‑max Adam step
and a projected sub‑gradient descent.  The final sequence is returned
as a plain list of floats and is evaluated with the provided
``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        # all non‑positive: return uniform vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT of size power‑of‑2."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for seeding the population)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^{-p}, i = 1..n, then normalised."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), i = 0..n‑1, then normalised."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑type shape a_i ∝ i^{α‑1}·(n‑i+1)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the interval."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/n)^{p}."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·vec1 + (1‑λ)·vec2, projected onto the simplex."""
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.20,
    tau_end: float = 1e-6,
    tau_decay: float = 0.92,
    lr_start: float = 0.20,
    lr_decay: float = 0.96,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        f_τ(a) = τ·log Σ_k exp(conv_k(a)/τ)

    Returns the best vector (projected onto the simplex) observed before
    ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # temperature schedule
        if t % 150 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # smooth‑max weights
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient = 2 * (a ⋆ w)   (cross‑correlation)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # learning‑rate anneal
        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-4)

        # occasional jitter
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.004 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.30) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient (sum of contributions from all active shifts)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # -----------------------------------------------------------------
    # Problem size (larger n gives more optimisation freedom)
    # -----------------------------------------------------------------
    n = 4096

    # -----------------------------------------------------------------
    # Helper to evaluate a candidate quickly (exact objective)
    # -----------------------------------------------------------------
    def score_of(vec: np.ndarray) -> float:
        return compute_score(vec)

    # -----------------------------------------------------------------
    # Initial population (size ≤ 20)
    # -----------------------------------------------------------------
    POP_SIZE = 20
    population = []          # list of (score, vector)

    def add_to_pop(vec: np.ndarray):
        """Project, score and maintain a bounded sorted population."""
        vec = project_to_simplex(np.maximum(vec, 0.0))
        sc = score_of(vec)
        if len(population) < POP_SIZE:
            population.append((sc, vec))
            population.sort(key=lambda x: x[0])
        elif sc < population[-1][0]:
            population[-1] = (sc, vec)
            population.sort(key=lambda x: x[0])

    # 0. fallback (if available)
    try:
        fallback = np.array(height_sequence_1, dtype=float)
        if fallback.shape[0] != n:
            # simple up‑sampling by linear interpolation to match n
            x_old = np.linspace(0, 1, fallback.shape[0])
            x_new = np.linspace(0, 1, n)
            fallback = np.interp(x_new, x_old, fallback)
        add_to_pop(fallback)
        print(f"[init] fallback added, score = {population[0][0]:.6f}")
    except Exception:
        pass

    # 1. uniform vector
    add_to_pop(np.full(n, 1.0 / n))

    # 2. random Dirichlet draws (moderate concentrations)
    for _ in range(30):
        conc = 10 ** np.random.uniform(-1.0, 1.0)   # 0.1 … 10
        vec = np.random.dirichlet([conc] * n)
        add_to_pop(vec)

    # 3. parametric families with random parameters
    for _ in range(12):
        add_to_pop(generate_powerlaw(n, np.random.uniform(0.1, 5.0)))
    for _ in range(12):
        add_to_pop(generate_exponential(n, np.random.uniform(0.0001, 0.6)))
    for _ in range(12):
        a = 10 ** np.random.uniform(-2, 0.3)
        b = 10 ** np.random.uniform(-2, 0.3)
        add_to_pop(generate_beta(n, a, b))
    for _ in range(8):
        sigma = np.random.uniform(0.1 * n, 0.45 * n)
        add_to_pop(generate_gaussian(n, sigma))
    for _ in range(8):
        add_to_pop(generate_sine_power(n, np.random.uniform(0.1, 8.0)))

    # 4. mixtures of two random members of the current pool
    for _ in range(15):
        if len(population) < 2:
            break
        i, j = np.random.choice(len(population), size=2, replace=False)
        lam = np.random.rand()
        mix = mixture(population[i][1], population[j][1], lam)
        add_to_pop(mix)

    # keep best as global incumbent
    best_score, best_vec = population[0]
    print(f"[init] best initial score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # Genetic / evolutionary loop
    # -----------------------------------------------------------------
    ITER_BATCH = 500          # after this many children we run a quick Adam refinement
    it = 0
    # Reserve about 150 s for final polishing phases
    final_phase_time = 150.0
    while time.time() < deadline - final_phase_time:
        it += 1
        # ---- selection (tournament among top half) ----
        top = POP_SIZE // 2
        p1 = population[np.random.randint(top)][1]
        p2 = population[np.random.randint(top)][1]

        # ---- recombination (convex blend) ----
        lam = np.random.rand()
        child = lam * p1 + (1.0 - lam) * p2

        # ---- mutation (relative Gaussian perturbation) ----
        sigma_mut = 0.02
        child = child + sigma_mut * np.random.randn(n) * child
        child = np.maximum(child, 0.0)
        child = project_to_simplex(child)

        # ---- evaluate and possibly insert ----
        child_score = score_of(child)
        if child_score < population[-1][0]:
            population[-1] = (child_score, child)
            population.sort(key=lambda x: x[0])
            if child_score < best_score:
                best_score, best_vec = child_score, child.copy()
                print(f"[GA iter {it}] new best = {best_score:.6f}")

        # ---- occasional local Adam refinement ----
        if it % ITER_BATCH == 0:
            # 5‑second Adam refinement on the current best
            sub_deadline = min(deadline, time.time() + 5.0)
            refined = smooth_max_adam(best_vec, sub_deadline,
                                     tau_start=0.16, tau_end=1e-6,
                                     lr_start=0.18)
            refined_score = score_of(refined)
            if refined_score < best_score:
                best_score, best_vec = refined_score, refined.copy()
                print(f"[refine] Adam after {it} iters -> {best_score:.6f}")

    # -----------------------------------------------------------------
    # Final polishing – Adam → sub‑gradient → optional simulated anneal
    # -----------------------------------------------------------------
    # 1. Adam smooth‑max (longer run)
    if time.time() < deadline - 40.0:
        adam_deadline = deadline - 40.0
        best_vec = smooth_max_adam(best_vec, adam_deadline,
                                   tau_start=0.12, tau_end=1e-6,
                                   lr_start=0.15)

    # 2. Sub‑gradient polishing
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 5.0
        best_vec = subgradient_polish(best_vec, sub_deadline, step0=0.28)

    # -----------------------------------------------------------------
    # Return the final candidate
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_vec.tolist()
