def adam_optimize(a0, time_deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1  # step size maybe adjust
    t=0
    best_a = a.copy()
    best_score = compute_score(a)
    beta = 10.0  # initial sharpness for softmax
    while time.time() < time_deadline:
        t+=1
        # compute conv
        conv = full_convolution(a)
        # optionally increase beta as we progress to get closer to true max
        if t % 200 == 0 and beta < 2000:
            beta *= 1.2
        # compute weights
        max_conv = conv.max()
        # compute w as softmax
        # use shifted values to avoid overflow
        shift = max_conv
        exp_vals = np.exp(beta * (conv - shift))
        Z = np.sum(exp_vals)
        w = exp_vals / Z
        # compute gradient = 2 * correlation of a with w
        N = 1 << (( (len(w)+len(a)-1).bit_length()))
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        # correlation = irfft( conj(A) * W )
        corr = np.fft.irfft(np.conj(A) * W, N)[:a.shape[0]]
        grad = 2.0 * corr.real
        # add projection gradient to handle sum constraint? We'll project after step.
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a = project_to_simplex(a)
        # occasional restart: if score worse than best for many iterations, maybe jitter
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # maybe increase step size
            lr = min(lr * 1.05, 0.5)
        else:
            # optionally adapt lr downwards
            lr = max(lr * 0.999, 1e-4)
        # optionally print progress every few seconds
        if t % 500 == 0:
            print(f"[adam] iter {t}, beta={beta:.2f}, best={best_score:.6f}, time={time.time():.1f}s")
    return best_a

def adam_optimize(a0, time_deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    t = 0
    best_a = a.copy()
    best_score = compute_score(a)
    beta = 2.0   # softmax sharpness
    while time.time() < time_deadline:
        t += 1
        # compute conv
        conv = full_convolution(a)
        # softmax weight
        mx = conv.max()
        # increase beta slowly
        if t % 1000 == 0:
            beta = min(beta * 1.1, 2000.0)
        # compute softmax probabilities
        # use stable exp
        exp_vals = np.exp(beta * (conv - mx))
        w = exp_vals / exp_vals.sum()
        # compute gradient via cross-correlation
        N = 1 << ((len(w) + len(a) - 1).bit_length())
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        corr = np.fft.irfft(np.conj(A) * W, N)[:len(a)]
        grad = 2.0 * corr.real
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = project_to_simplex(a)
        # evaluate score occasionally
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[adam] iter={t}, best_score={best_score:.6f}, time={time.time():.1f}s")
            # if stuck, maybe add jitter
    return best_a
```

"""
Propose a coefficient vector that minimises the self‑convolution bound.

The algorithm works in four stages:

1. **Broad parametric sweep** – explore a few analytic families
   (power‑law, geometric, plateau‑power, beta, logistic) and keep the
   best vector found.
2. **Adam optimiser with a smooth “soft‑max” approximation** – minimise
   `max(conv(a,a))` by replacing the non‑smooth max with a
   log‑sum‑exp; the gradient is obtained analytically via a
   cross‑correlation (FFT) and the iterate is projected onto the
   probability simplex after each step.  The temperature‑like sharpness
   parameter β is increased gradually.
3. **Projected sub‑gradient polishing** – a short descent phase that
   uses the exact active‑set sub‑gradient of the max‑convolution.
4. **Deterministic hill‑climb** – greedy mass‑transfer moves for final
   fine‑tuning.

The whole search respects the 1000 s budget.  The returned list is
ready to be passed to ``evaluate_sequence``.  The code contains no
closures, no file I/O and only uses the standard scientific Python stack.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Projection onto the probability simplex  (Euclidean projection)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x >= 0, sum x = 1}."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, v.shape[0] + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    N = 1 << ((m - 1).bit_length())                # next power of two ≥ m
    A = np.fft.rfft(a, N)
    C = A * A
    conv = np.fft.irfft(C, N)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a vector with Σa = 1:
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
#  Parametric families (lightweight sweep)
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray | None:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def geometric(n: int, q: float) -> np.ndarray | None:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray | None:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def logistic_shape(n: int, center_frac: float, scale_frac: float) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    scale = max(scale_frac * n, 1e-9)
    a = 1.0 / (1.0 + np.exp((i - center) / scale))
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def random_dirichlet(n: int, concentration: float) -> np.ndarray | None:
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    s = y.sum()
    if s == 0:
        return None
    return y / s


# ----------------------------------------------------------------------
#  Broad parametric sweep (time‑controlled)
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray | None:
    best_vec = None
    best_val = float('inf')

    # power law
    for p in np.linspace(0.2, 2.0, 19):
        if time.time() >= deadline:
            return best_vec
        a = power_law(n, p)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # geometric
    for q in np.linspace(0.90, 0.9999, 60):
        if time.time() >= deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # mixed plateau + power
    for L_frac in np.linspace(0.05, 0.30, 6):
        for p in np.linspace(0.3, 1.0, 8):
            if time.time() >= deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # beta shape
    alphas = np.linspace(-0.5, 2.5, 7)
    betas = np.linspace(-0.5, 2.5, 7)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # logistic shape
    for center_frac in np.linspace(0.2, 0.8, 7):
        for scale_frac in np.linspace(0.02, 0.15, 7):
            if time.time() >= deadline:
                return best_vec
            a = logistic_shape(n, center_frac, scale_frac)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # random Dirichlet samples (cheap)
    for _ in range(300):
        if time.time() >= deadline:
            break
        conc = 10 ** np.random.uniform(-2, 0)   # 0.01 – 1
        a = random_dirichlet(n, conc)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
#  Projected sub‑gradient polishing (as in the reference)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.15) -> np.ndarray:
    a = a0.copy()
    best = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient: 2 * a shifted for each active index
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best = a.copy()

    return best


# ----------------------------------------------------------------------
#  Adam optimiser on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]

    # FFT size for correlation (a length n, w length 2n‑1)
    conv_len = 2 * n - 1
    N = 1 << ((n + conv_len - 1).bit_length())   # power‑of‑two ≥ n+2n‑2

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.07

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    best_a = a.copy()
    best_val = compute_score(a)

    beta = 2.0                # initial soft‑max sharpness
    beta_max = 2000.0

    while time.time() < deadline:
        t += 1

        # --- current convolution and soft‑max weights ---
        conv = full_convolution(a)                 # length 2n‑1
        mx = conv.max()
        # increase β gradually (log‑scale)
        if t % 500 == 0 and beta < beta_max:
            beta = min(beta * 1.12, beta_max)

        # stable soft‑max
        exp_vals = np.exp(beta * (conv - mx))
        w = exp_vals / exp_vals.sum()              # weights, sum to 1

        # --- gradient = 2 * cross‑correlation(a, w) ---
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        corr = np.fft.irfft(np.conj(A) * W, N)[:n]  # cross‑correlation (real)
        grad = 2.0 * corr.real

        # --- Adam update + projection ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)

        # occasionally record improvement
        if t % 400 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[adam] iter={t}  best={best_val:.6f}  time={time.time():.1f}s")
                # be a little more aggressive after a success
                lr = min(lr * 1.04, 0.5)

    return best_a


# ----------------------------------------------------------------------
#  Simple greedy hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    best = a.copy()
    best_M = full_convolution(a).max()
    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best


# ----------------------------------------------------------------------
#  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # 0. Optional fallback (previously discovered vector)
    # ------------------------------------------------------------------
    try:
        fallback = np.array(height_sequence_1, dtype=float)
        # adapt length if needed – pad with zeros (zero entries are harmless)
        target_len = 4096   # we will work primarily with this length
        if fallback.size != target_len:
            if fallback.size < target_len:
                fallback = np.concatenate([fallback,
                                           np.zeros(target_len - fallback.size)])
            else:
                fallback = fallback[:target_len]
        fallback = project_to_simplex(fallback)  # enforce Σ=1
        val = compute_score(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback.tolist()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Parametric sweep (≈ 120 s)
    # ------------------------------------------------------------------
    sweep_deadline = deadline - 660.0   # leave ~660 s for refinement
    n = 4096
    if time.time() < sweep_deadline:
        a0 = parametric_sweep(n, sweep_deadline)
        if a0 is not None:
            val = compute_score(a0)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a0.tolist()

    # ------------------------------------------------------------------
    # 2. Adam optimisation (≈ 350 s)
    # ------------------------------------------------------------------
    adam_deadline = deadline - 280.0
    if best_seq is not None and time.time() < adam_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_opt = adam_optimize(a_start, adam_deadline)
        val = compute_score(a_opt)
        print(f"[adam] after optimisation score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()

    # ------------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 150 s)
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 120.0
    if best_seq is not None and time.time() < subgrad_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, subgrad_deadline, step0=0.15)
        val = compute_score(a_polished)
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # ------------------------------------------------------------------
    # 4. Final hill‑climb (remaining time)
    # ------------------------------------------------------------------
    hill_deadline = deadline
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq, dtype=float))

    # ------------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
