"""
Optimiser for the self‑convolution upper bound.

Approach
--------
* Start from the known construction (global variable ``height_sequence_1``) if it
  exists, otherwise from a uniform distribution.
* Refine the vector with a smooth approximation of the max‑convolution:
  - Use a soft‑max (log‑sum‑exp) over the full auto‑convolution to obtain a
    differentiable surrogate of the objective.
  - Compute the gradient of the surrogate analytically by a second FFT‑based
    convolution.
  - Apply Adam (adaptive moment) updates on the simplex (projected after each
    step).
* Increase the soft‑max temperature (parameter `beta`) gradually so that the
  surrogate converges to the true max.
* After the Adam phase, perform a short projected sub‑gradient polishing
  (as in the reference implementation) and a final greedy hill‑climb for
  fine‑grained polishing.
* The best vector encountered (according to the exact objective) is returned
  as a plain Python list.

The optimiser runs for the whole time budget (default 1000 s) and prints
intermediate progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using an FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as in the reference)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    t = 0

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of each active entry: 2 * a shifted
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape shallow plateaux
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Deterministic hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """Greedy mass‑transfer hill‑climb."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.04, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 120_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for harness bookkeeping
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # 0. Baseline (fallback) – previous construction if available
    # -----------------------------------------------------------------
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=np.float64)
        fallback = np.maximum(fallback, 0.0)
        s = fallback.sum()
        if s < 0.01:
            fallback = np.full_like(fallback, 1.0 / fallback.shape[0])
        else:
            fallback = fallback / s
        best_seq = fallback.tolist()
        best_val = compute_score(fallback)
        print(f"[fallback] score = {best_val:.6f}")
    except Exception:
        fallback = None
        best_seq = None
        best_val = float('inf')

    # -----------------------------------------------------------------
    # 1. Initialise – use fallback if present, otherwise uniform
    # -----------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, dtype=np.float64)
    else:
        n = 2048
        a = np.full(n, 1.0 / n, dtype=np.float64)
        best_seq = a.tolist()
        best_val = compute_score(a)
        print(f"[init] uniform start score = {best_val:.6f}")

    n = a.shape[0]
    m = 2 * n - 1               # length of the auto‑convolution
    # FFT sizes (powers of two)
    fft_len_conv = 1 << ((m - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2 - 1).bit_length())  # for a * w_rev

    # -----------------------------------------------------------------
    # 2. Adam optimisation on a smooth max‑convolution surrogate
    # -----------------------------------------------------------------
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.06                     # base learning rate
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)

    beta_smooth = 8.0              # start soft‑max temperature
    beta_target = 2000.0
    beta_inc_iter = 12_000
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- full auto‑convolution (FFT) ----
        A = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(A * A, n=fft_len_conv)[:m]
        max_c = conv_full.max()

        # ---- soft‑max weights ----
        # exponents are non‑positive after subtraction of max_c
        exps = np.exp(beta_smooth * (conv_full - max_c))
        w_sum = exps.sum()
        if w_sum == 0.0:                     # numerical safeguard
            w = np.full_like(exps, 1.0 / exps.shape[0])
        else:
            w = exps / w_sum

        # ---- gradient of the smooth max (FFT) ----
        w_rev = w[::-1]
        A2 = np.fft.rfft(a, n=fft_len_grad)
        B = np.fft.rfft(w_rev, n=fft_len_grad)
        grad_full = np.fft.irfft(A2 * B, n=fft_len_grad)[:(3 * n - 2)]

        # index conversion: grad_i = 2 * grad_full[(m-1) - i]
        offset = m - 1           # equals 2*n-2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # ---- Adam update on the simplex ----
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad ** 2)
        m_hat = m_adam / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # ---- occasional reporting & beta schedule ----
        if it % 2000 == 0 or time.time() > deadline - 10.0:
            cur_score = compute_score(a)
            if cur_score < best_val:
                best_val = cur_score
                best_seq = a.tolist()
                print(f"[iter {it}] new best {best_val:.6f}  time={time.time()-start_time:.1f}s")
            # increase the soft‑max temperature slowly
            if it % beta_inc_iter == 0 and beta_smooth < beta_target:
                beta_smooth = min(beta_target, beta_smooth * 1.12)

        # safety break
        if time.time() >= deadline:
            break

    # -----------------------------------------------------------------
    # 3. Short projected sub‑gradient polishing
    # -----------------------------------------------------------------
    polish_deadline = deadline - 15.0
    if time.time() < polish_deadline:
        a_best = np.array(best_seq, dtype=np.float64)
        a_polished = subgradient_polish(a_best, polish_deadline, step0=0.10)
        val = compute_score(a_polished)
        print(f"[sub‑grad] polish score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_polished.tolist()
            a_best = a_polished
        else:
            a_best = a_best
    else:
        a_best = np.array(best_seq, dtype=np.float64)

    # -----------------------------------------------------------------
    # 4. Final deterministic hill‑climb (very short)
    # -----------------------------------------------------------------
    hill_deadline = deadline
    if time.time() < hill_deadline:
        a_final = hill_climb(a_best, hill_deadline, max_iter=300_000)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()

    # -----------------------------------------------------------------
    # 5. Safeguard – fall back to uniform if nothing decent was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 2048
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final evaluation with the official function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
