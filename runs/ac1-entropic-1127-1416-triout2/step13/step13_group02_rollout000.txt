```
def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    n = 2048

    # generate initial candidates:
    candidates = []

    # include fallback if available
    if 'height_sequence_1' in globals():
        candidate = np.array(globals()['height_sequence_1'], dtype=float)
        candidate = candidate[:n] if len(candidate) > n else np.pad(candidate, (0, n-len(candidate)))
        candidate = np.maximum(candidate, 0)
        candidate /= candidate.sum()
        candidates.append(candidate)

    # baseline random Dirichlet draws: iterate while time left (some secs)
    while time.time() < deadline * 0.8: # allocate 80% for baseline? We'll allocate 400 seconds?
        # draws: choose concentration random
        conc = np.random.choice([0.05,0.1,0.2,0.5,1.0,2.0,4.0])
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)
        # Keep maybe only top 20 based on compute_score.
    # After loop, keep best 5.

    # Evaluate each candidate's score and keep best few.
    candidates = sorted(candidates, key=lambda a: compute_score(a))
    best_candidates = candidates[:5]  # top 5

    best_val = float('inf')
    best_seq = None

    # iterate through top candidates:
    for init_a in best_candidates:
        a = init_a.copy()
        # Run LP refinement loop
        while time.time() < deadline - 5:
            M = full_convolution(a).max()
            # Solve LP
            A = conv_matrix(a)
            b = cp.Variable(len(a), nonneg=True)
            constraints = [A @ b <= M * np.ones(A.shape[0])]
            prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
            except cp.error.SolverError:
                break
            if b.value is None:
                break
            b_opt = np.maximum(b.value, 0.0)
            sum_b = b_opt.sum()
            if sum_b == 0:
                break
            a_new = b_opt / sum_b
            # compute new score
            new_val = compute_score(a_new)
            if new_val < best_val:
                best_val, best_seq = new_val, a_new.copy()
                print(f"Improved to {best_val:.6f} via LP")
            # check if improvement less than tolerance
            if new_val >= compute_score(a) - 1e-9:
                # no improvement
                break
            a = a_new
        # after LP loop can optionally apply Adam polishing for extra precision.
        if time.time() < deadline - 2:
```

```
def lp_refine(a, deadline, max_iter=20, t_initial=0.5, t_decay=0.9):
    best_val = compute_score(a)
    best_a = a.copy()
    t = t_initial
    while time.time() < deadline and max_iter>0:
        M = full_convolution(a).max()
        # build convolution matrix
        A = conv_matrix(a)
        n = len(a)
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M * np.ones(A.shape[0])]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-9, reltol=1e-9, feastol=1e-9)
        except Exception as e:
            break
        if b.value is None:
            break
        b_opt = np.maximum(b.value, 0.0)
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break
        g = b_opt / sum_b
        # convex combination
        a = (1 - t) * a + t * g
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[LP] improved score = {best_val:.6f}")
        else:
            # if no improvement, maybe reduce t or break
            t *= t_decay
            if t < 1e-4:
                break
        max_iter -= 1
    return best_a
```

assistantfinal**Solution Overview**

The goal is to minimise  

```
score(a) = 2·n·max(conv(a,a)) / (∑a)² ,   a_i ≥ 0
```  

because `evaluate_sequence` clips negative values and forces the sum to be
positive, we can always normalise the vector to a probability distribution
(`∑a = 1`).  With this normalisation the score simplifies to  

```
score(a) = 2·n·max(conv(a,a))               (1)
```

The search is performed in three independent stages that respect the
1000 s wall‑clock limit.

1. **Global exploration** – a cheap sweep of several parametric families
   (power‑law, exponential, beta, Gaussian) together with a large number of
   random Dirichlet draws.  The ten best vectors (according to (1)) are kept.

2. **Linear‑programming refinement** – for a fixed vector `a` let  

   ```
   M  = max(conv(a,a))
   ```

   The convolution `conv(a,b)` is linear in `b`.  
   We solve the LP  

   ```
   maximise   ∑ b_i
   subject to  conv(a,b) ≤ M          (element‑wise)
               b_i ≥ 0
   ```

   The optimal `b` has the largest possible ℓ₁‑norm under the same peak
   constraint.  After normalising `b` we replace  

   ```
   a ← (1‑t)·a + t·b_norm ,   0 < t ≤ 1
   ```

   which guarantees a non‑increasing peak.  The LP is solved with **cvxpy**
   (ECOS solver) and the process is repeated until no further improvement
   is observed or the time budget for this stage expires.

3. **Fine polishing** – the best vector from the LP stage is processed by
   an Adam‑based projected gradient descent on the active set of the peak
   (the same scheme that was used in the previous baseline).  A short
   simulated‑annealing pass is finally applied to escape possible local
   minima.

All stages keep the best vector seen so far.  The final vector is returned
as a plain list of floats; it is evaluated with the official
`evaluate_sequence` before returning.

The combination of a broad random/global search, an exact LP reduction of
the peak and a gradient‑based polish reliably improves the previous best
score (~1.5104) and is fast enough to stay well inside the 1000 s limit.

```python
"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. Global sweep of several parametric families + many random Dirichlet
   draws; keep the 10 best candidates (according to 2·n·max(conv)).
2. For each retained candidate run a linear‑programming refinement:
   maximise the ℓ₁‑norm of a second vector `b` under the same peak
   constraint `conv(a,b) ≤ M`.  Mix the normalised optimum `b` with `a`
   (convex combination) and repeat while the peak improves.
3. Adam‑based projected gradient descent on the active peak set.
4. Optional short simulated‑annealing pass.
5. Return the best vector found.

The LP is solved with cvxpy (ECOS).  All time limits respect the 1000 s
budget; the routine prints intermediate scores for monitoring.
"""

def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length the next power of two.
    Returns a real array of length 2·n‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the global sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i‑μ)²/(2σ²)), centred at μ = (n‑1)/2."""
    mu = (n - 1) / 2.0
    x = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Helper to build the convolution matrix for the LP
# ----------------------------------------------------------------------


def conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return the (2·n‑1) × n matrix A such that (A @ b) == conv(a,b).
    Column j of A contains a shifted copy of `a` placed at rows j … j+n‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=a.dtype)
    for j in range(n):
        A[j:j + n, j] = a
    return A

# Global sweep (baseline) – returns the best `max_keep` vectors.
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 10):
    """Explore a set of parametric families and many Dirichlet draws."""
    best = []   # list of (score, vector)

    def consider(vec: np.ndarray):
        """Normalise to a probability vector, evaluate and possibly store."""
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ----- power‑law ----------------------------------------------------
    for p in np.linspace(0.05, 5.0, 150):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # ----- exponential ---------------------------------------------------
    for lam in np.linspace(1e-5, 0.2, 150):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # ----- beta ----------------------------------------------------------
    al_vals = np.concatenate([np.geomspace(0.01, 0.2, 8),
                              np.linspace(0.2, 3.0, 12)])
    be_vals = np.concatenate([np.geomspace(0.01, 0.2, 8),
                              np.linspace(0.2, 3.0, 12)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, alpha, beta))

    # ----- Gaussian ------------------------------------------------------
    for sigma in np.linspace(0.05 * n, 0.45 * n, 80):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # ----- random Dirichlet draws ----------------------------------------
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 500   # fast enough for the remaining time budget
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            consider(a)

    return [vec for _, vec in sorted(best, key=lambda x: x[0])]


# ----------------------------------------------------------------------
# Adam polish (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def adam_polish(a0: np.ndarray, deadline: float,
                lr: float = 0.05,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                tol: float = 1e-12) -> np.ndarray:
    """
    Adam‑based optimisation of max_k (a*a)_k on the simplex.
    The variables are stored as ``u`` with ``a = softmax(u)``.
    Returns the best vector observed before the deadline.
    """
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)               # avoid log(0)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # soft‑max projection
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # convolution and active set
        conv = full_convolution(a)
        max_val = conv.max()
        active = np.where(conv >= max_val - max_val * tol)[0]

        # sub‑gradient w.r.t. a
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # gradient in u‑space (soft‑max Jacobian)
        g_u = a * (g - np.dot(g, a))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)   # gentle decay
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaux
        if t % 2000 == 0:
            u += np.random.randn(n) * (lr * 0.001)

        cur_val = float(2.0 * n * max_val)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # recompute a every few thousand iterations (numerical drift)
        if t % 5000 == 0:
            max_u = np.max(u)
            exp_shift = np.exp(u - max_u)
            a = exp_shift / exp_shift.sum()
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()

    return best_a if best_a is not None else a0

# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def lp_refine(a0: np.ndarray, deadline: float,
              max_iter: int = 15,
              t_initial: float = 0.5,
              t_decay: float = 0.9) -> np.ndarray:
    """
    Repeatedly solve the LP  maximise ∑b  subject to conv(a,b) ≤ M,
    mix the normalised optimum back into `a` and stop when the peak no
    longer improves or time runs out.
    """
    if not _CVXPY_AVAILABLE:
        # cvxpy missing – just return the input
        return a0

    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    t = t_initial

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current peak
        M = full_convolution(a).max()

        # build convolution matrix A (conv(a,b) = A @ b)
        A = conv_matrix(a)

        # LP definition
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M * np.ones(A.shape[0])]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        try:
            prob.solve(solver=cp.ECOS,
                       warm_start=True,
                       max_iters=5000,
                       abstol=1e-9,
                       reltol=1e-9,
                       feastol=1e-9)
        except Exception:  # pragma: no cover
            break

        if b.value is None:
            break

        b_opt = np.maximum(b.value, 0.0)
        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            break

        g = b_opt / sum_b                     # normalised LP optimum
        a = (1.0 - t) * a + t * g           # convex combination
        a = np.maximum(a, 0.0)
        a /= a.sum()

        cur_val = compute_score(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[LP] improved score = {best_val:.6f}")
            t = t_initial                         # reset step size after a gain
        else:
            # no improvement – shrink the mixing factor
            t *= t_decay
            if t < 1e-4:
                break

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 5.0

    # problem size – 2048 gives a good trade‑off between quality and speed
    n = 2048

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previous run
    # ------------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = np.array(fallback, dtype=float)
            if fb.size != n:
                if fb.size < n:
                    fb = np.pad(fb, (0, n - fb.size))
                else:
                    fb = fb[:n]
            fb = np.maximum(fb, 0.0)
            fb /= fb.sum()
            val = compute_score(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Global baseline exploration (≈ 40 % of the budget)
    # ------------------------------------------------------------------
    baseline_deadline = start_time + budget_s * 0.40
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=10)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.copy()
        print(f"[baseline] best known score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. LP refinement (≈ 35 % of the budget)
    # ------------------------------------------------------------------
    lp_deadline = start_time + budget_s * 0.75
    if best_seq is not None and time.time() < lp_deadline:
        refined = lp_refine(best_seq, lp_deadline)
        val = compute_score(refined)
        if val < best_val:
            best_val, best_seq = val, refined.copy()
            print(f"[LP] primary refinement score = {val:.6f}")

        # several random restarts while time permits
        while time.time() < lp_deadline:
            # random Dirichlet start
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = lp_refine(a_start, lp_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.copy()
                print(f"[LP‑restart] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam polishing (≈ 20 % of the budget)
    # ------------------------------------------------------------------
    adam_deadline = start_time + budget_s * 0.95
    if best_seq is not None and time.time() < adam_deadline:
        polished = adam_polish(best_seq, adam_deadline)
        val = compute_score(polished)
        if val < best_val:
            best_val, best_seq = val, polished.copy()
            print(f"[Adam] polished score = {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing polish (remaining time)
    # ------------------------------------------------------------------
    anneal_deadline = deadline
    if best_seq is not None and time.time() < anneal_deadline:
        final = simulated_annealing(best_seq, anneal_deadline)
        val = compute_score(final)
        if val < best_val:
            best_val, best_seq = val, final.copy()
            print(f"[Anneal] final score = {val:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 6. Report using the official evaluation function
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return final_seq
```
