def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective: 2·n·max(conv(a,a)) / (sum(a)**2).
    The sequence a is assumed non‑negative.  No clipping is performed here;
    the outer ``evaluate_sequence`` will enforce the limits.
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Mirror of left‑heavy."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred on the middle of the interval."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """
    Symmetric triangular bump.
    width_frac ∈ (0,1] controls half‑width relative to n.
    """
    half_width = max(1, int(width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return w / w.sum()


def exp_decay_family(n: int, gamma: float, left: bool = True) -> np.ndarray:
    """Exponential decay from one side."""
    i = np.arange(n, dtype=np.float64)
    if left:
        w = np.exp(-gamma * i)
    else:
        w = np.exp(-gamma * (n - 1 - i))
    return w / w.sum()


def mixed_uniform_gaussian_family(n: int, lam: float, sigma: float) -> np.ndarray:
    """Convex mixture of uniform and centred Gaussian."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    gaussian = gaussian_center_family(n, sigma)
    w = lam * uniform + (1.0 - lam) * gaussian
    return w / w.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / s
    return w

def exp_gradient_descent(start_a: np.ndarray,
                         deadline: float,

    Mirror descent on the simplex using exponentiated gradients.
    Sub‑gradient is taken at the index where the convolution attains its maximum.
    """

    a = start_a.copy()

        if time.time() >= deadline:

        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        # Renormalise (KL‑projection is scaling)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] exp‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] exp‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def smoothed_gradient_descent(start_a: np.ndarray,
                              deadline: float,
                              tau0: float = 5e-4,
                              tau_decay: float = 0.96,
                              eta0: float = 1.0,
                              max_iters: int = 100_000) -> Tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of the
    max‑convolution (log‑sum‑exp).  Temperature τ is geometrically
    decreased.  Each iteration performs:
        w_i ∝ exp(conv_i / τ)   (soft‑max weights)
        g = 2 * (a * w_rev) reversed
        a ← a – η g   (Euclidean step)  → project onto simplex
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    tau = tau0
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)          # length 2n‑1
        # soft‑max weights (stable version)
        mx = conv.max()
        exp_terms = np.exp((conv - mx) / tau)
        w = exp_terms / exp_terms.sum()

        # compute reversed weights (needed for gradient)
        w_rev = w[::-1]

        # gradient = 2 * (a * w_rev) evaluated at reversed positions
        grad_full = convolution_fft(a, w_rev)    # length n + (2n‑1) -1 = 3n‑2
        # we need values corresponding to k = 0…n‑1:
        # from earlier derivation: g_k = 2 * (a * w_rev)[2n‑2‑k]
        g = 2.0 * grad_full[2 * n - 2 - np.arange(n)]

        # step size schedule
        eta = eta0 / math.sqrt(t)

        a = a - eta * g
        a = normalize(a)      # Euclidean projection onto simplex

        # decay temperature occasionally
        if t % 2000 == 0:
            tau = max(tau * tau_decay, 1e-8)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] smoothed‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] smoothed‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 12_000_000,
                  temp_factor: float = 1.5e-3,
                  step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    Moves are either a swap of two entries or a fractional transfer.
    Linear cooling of temperature.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                 # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.08, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 14) -> Tuple[float, float]:
    """Golden‑section search for the minimum of func on [lo,hi]."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback (previous best) if available
    # ------------------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Working discretisation
    # ------------------------------------------------------------------
    n = 8192
    print(f"[{time.time():.1f}] start search with n={n}")

    # ------------------------------------------------------------------
    # 2. Coarse scan of analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse family scan")
    best_family = None
    best_params = None

    # centre‑heavy power (alpha >= 0)
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_center_power_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "center_power"
            best_params = {"alpha": alpha}
            print(f"[{time.time():.1f}] new best centre‑α={alpha:.3f} → {best_score:.6f}")

    # Gaussian family
    for sigma in np.linspace(0.2 * n, 0.5 * n, 30):
        if time.time() >= deadline - 30.0:
            break
        vec = gaussian_center_family(n, sigma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "gaussian"
            best_params = {"sigma": sigma}
            print(f"[{time.time():.1f}] new best gauss‑σ={sigma:.0f} → {best_score:.6f}")

    # Triangular family
    for width in np.linspace(0.1, 0.9, 33):
        if time.time() >= deadline - 30.0:
            break
        vec = triangular_center_family(n, width)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "triangular"
            best_params = {"width_frac": width}
            print(f"[{time.time():.1f}] new best tri‑w={width:.3f} → {best_score:.6f}")

    # Left‑heavy power (β)
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "left_power"
            best_params = {"beta": beta}
            print(f"[{time.time():.1f}] new best left‑β={beta:.3f} → {best_score:.6f}")

        vec_r = right_heavy_power_family(n, beta)
        sc_r = compute_score(vec_r)
        if sc_r < best_score:
            best_score, best_seq = sc_r, vec_r.tolist()
            best_family = "right_power"
            best_params = {"beta": beta}
            print(f"[{time.time():.1f}] new best right‑β={beta:.3f} → {best_score:.6f}")

    # Product (U‑shaped) family
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "product"
            best_params = {"alpha": alpha}
            print(f"[{time.time():.1f}] new best product‑α={alpha:.3f} → {best_score:.6f}")

    # Beta family (two parameters)
    for _ in range(120):
        if time.time() >= deadline - 30.0:
            break
        p = np.random.uniform(0.2, 3.0)
        q = np.random.uniform(0.2, 3.0)
        vec = beta_family(n, p, q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            best_family = "beta"
            best_params = {"p": p, "q": q}
            print(f"[{time.time():.1f}] new best beta p={p:.3f},q={q:.3f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Random dense search (≈15000 trials)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] random dense search")
    trials = 0
    max_trials = 15000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        vec = random_mixture(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local golden‑section refinement of the best family (if any)
    # ------------------------------------------------------------------
    if best_family is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining best family {best_family}")

        if best_family == "center_power":
            cur_alpha = best_params["alpha"]
            lo, hi = max(0.0, cur_alpha - 0.8), min(12.0, cur_alpha + 0.8)

            def f_alpha(a):
                return compute_score(symmetric_center_power_family(n, a))

            new_alpha, _ = golden_section_optimize(f_alpha, lo, hi)
            vec = symmetric_center_power_family(n, new_alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["alpha"] = new_alpha
                print(f"[{time.time():.1f}] refined centre‑α={new_alpha:.4f} → {best_score:.6f}")

        elif best_family == "gaussian":
            cur_sigma = best_params["sigma"]
            lo, hi = max(0.2 * n, cur_sigma - 0.2 * n), min(0.5 * n, cur_sigma + 0.2 * n)

            def f_sigma(s):
                return compute_score(gaussian_center_family(n, s))

            new_sigma, _ = golden_section_optimize(f_sigma, lo, hi)
            vec = gaussian_center_family(n, new_sigma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["sigma"] = new_sigma
                print(f"[{time.time():.1f}] refined gauss‑σ={new_sigma:.1f} → {best_score:.6f}")

        elif best_family == "triangular":
            cur_w = best_params["width_frac"]
            lo, hi = max(0.05, cur_w - 0.2), min(0.95, cur_w + 0.2)

            def f_w(w):
                return compute_score(triangular_center_family(n, w))

            new_w, _ = golden_section_optimize(f_w, lo, hi)
            vec = triangular_center_family(n, new_w)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["width_frac"] = new_w
                print(f"[{time.time():.1f}] refined tri‑w={new_w:.4f} → {best_score:.6f}")

        elif best_family == "product":
            cur_alpha = best_params["alpha"]
            lo, hi = max(0.0, cur_alpha - 0.6), min(12.0, cur_alpha + 0.6)

            def f_alpha(a):
                return compute_score(symmetric_product_family(n, a))

            new_alpha, _ = golden_section_optimize(f_alpha, lo, hi)
            vec = symmetric_product_family(n, new_alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                best_params["alpha"] = new_alpha
                print(f"[{time.time():.1f}] refined product‑α={new_alpha:.4f} → {best_score:.6f}")

        elif best_family == "beta":
            # Local random jitter around best (p,q)
            p0, q0 = best_params["p"], best_params["q"]
            for _ in range(250):
                if time.time() >= deadline - 5.0:
                    break
                p = np.clip(p0 + np.random.randn() * 0.15, 0.1, 5.0)
                q = np.clip(q0 + np.random.randn() * 0.15, 0.1, 5.0)
                vec = beta_family(n, p, q)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    best_params["p"], best_params["q"] = p, q
                    print(f"[{time.time():.1f}] refined beta p={p:.3f},q={q:.3f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Multi‑restart local optimisation (exp‑grad + SA)
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 20.0:
        restart += 1
        print(f"[{time.time():.1f}] restart #{restart}")

        # start from the current best, but add a tiny random perturbation to break symmetry
        a_start = np.array(best_seq, dtype=np.float64)
        eps = 1e-4
        perturb = np.exp(eps * np.random.randn(n))
        a_start = normalize(a_start * perturb)

        # ------ exponentiated‑gradient phase (shortened) ------
        eg_deadline = min(deadline - 8.0, time.time() + 30.0)   # ~30 s
        a_eg, sc_eg = exp_gradient_descent(
            start_a=a_start,
            deadline=eg_deadline,
            max_iters=150_000,
            eta0=0.7,
        )
        if sc_eg < best_score:
            best_score, best_seq = sc_eg, a_eg.tolist()
            print(f"[{time.time():.1f}] after exp‑grad restart {restart} best = {best_score:.6f}")

        # ------ smoothed‑gradient phase (optional, useful when time permits) ------
        if time.time() < deadline - 10.0:
            sg_deadline = min(deadline - 6.0, time.time() + 20.0)
            a_sg, sc_sg = smoothed_gradient_descent(
                start_a=np.array(best_seq, dtype=np.float64),
                deadline=sg_deadline,
                tau0=5e-4,
                tau_decay=0.94,
                eta0=0.8,
                max_iters=80_000,
            )
            if sc_sg < best_score:
                best_score, best_seq = sc_sg, a_sg.tolist()
                print(f"[{time.time():.1f}] after smoothed‑grad restart {restart} best = {best_score:.6f}")

        # ------ simulated‑annealing phase (longer) ------
        if time.time() < deadline - 5.0:
            sa_deadline = deadline - 2.0
            a_sa, sc_sa = hill_climb_sa(
                start_a=np.array(best_seq, dtype=np.float64),
                deadline=sa_deadline,
                max_steps=8_000_000,
                temp_factor=1.2e-3,
                step_scale_init=0.28,
            )
            if sc_sa < best_score:
                best_score, best_seq = sc_sa, a_sa.tolist()
                print(f"[{time.time():.1f}] SA restart {restart} improved best = {best_score:.6f}")

        # safety break if we are very close to the time limit
        if time.time() >= deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 6. Final safety fallback (uniform) if nothing found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 7. Final official evaluation (includes all security checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
