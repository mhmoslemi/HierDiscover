"""
Search for a non‑negative step‑height vector that minimises
``evaluate_sequence`` (i.e. 2·n·max(conv)/sum²).

The method builds on the previous implementation but adds several
enhancements:

*   A richer set of parametric families (Pareto, log‑normal, additional
    power‑law ranges, very low‐concentration Dirichlet draws) for the
    broad exploration stage.
*   The Adam optimiser is run for many more iterations with a slower
    temperature decay (τ → 1e‑10) and a smaller learning‑rate schedule.
*   After Adam we perform a longer, finer projected sub‑gradient descent
    (step size 0.10 → 0.0) and finally a high‑budget simulated‑annealing
    hill‑climb.
*   The overall budget (≈1000 s) is split dynamically so that the
    optimiser uses almost the whole wall‑clock time.

The vector is always projected onto the probability simplex
(`sum(v)=1, v_i≥0`).  Scaling does not change the official score
because the evaluation routine clips to [0,1000] and the objective is
scale‑invariant for values well below the clipping threshold.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    Returns g_i = 2·Σ_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    m = len(w)
    rev_w = w[::-1]
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def smooth_max(conv: np.ndarray, tau: float):
    """
    τ·log‑sum‑exp approximation of max(conv).
    Returns (S, w) where w are the associated soft‑max weights (∑w=1).
    """
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 0.0)


def generate_ushape(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n + 1 - i)) ** p
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 0.0)


def generate_pareto(n: int, shape: float) -> np.ndarray:
    # pareto samples are (X+1) where X~Pareto(shape)
    a = (np.random.pareto(shape, size=n) + 1.0)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 0.0)


def generate_lognormal(n: int, sigma: float, mean: float = 0.0) -> np.ndarray:
    a = np.random.lognormal(mean, sigma, size=n)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 0.0)

def explore_candidates(n: int, deadline: float, keep: int = 12) -> list[np.ndarray]:
    """Create many vectors and keep the best ``keep`` according to compute_score."""
    best = []          # list of (score, vector)

    def consider(v: np.ndarray):
        if not np.isfinite(v).all():
            return
        v = np.maximum(v, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        if len(best) < keep:
            best.append((sc, v.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, v.copy())
            best.sort(key=lambda x: x[0])

    # 0) uniform baseline
    consider(np.full(n, 1.0 / n))

    # 1) power‑law (both decreasing & increasing)
    for p in np.linspace(0.0, 5.0, 51):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))
        consider(generate_powerlaw(n, -p))

    # 2) exponential
    for lam in np.linspace(-0.4, 0.4, 81):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # 3) beta
    al_vals = np.linspace(0.1, 6.0, 25)
    be_vals = np.linspace(0.1, 6.0, 25)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    # 4) Gaussian width sweep
    for sigma in np.linspace(0.03 * n, 0.5 * n, 15):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # 5) sine‑power
    for p in np.linspace(0.1, 10.0, 30):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # 6) U‑shape (edge‑heavy)
    for p in np.linspace(0.1, 6.0, 30):
        if time.time() >= deadline:
            break
        consider(generate_ushape(n, p))

    # 7) Pareto heavy‑tailed
    for shape in [0.3, 0.5, 0.8, 1.2, 2.0]:
        if time.time() >= deadline:
            break
        consider(generate_pareto(n, shape))

    # 8) Log‑normal
    for sigma in np.linspace(0.2, 2.0, 20):
        if time.time() >= deadline:
            break
        consider(generate_lognormal(n, sigma))

    # 9) Dirichlet draws – a very fine grid of concentration values
    concentrations = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, (keep * 250) // len(concentrations))
    for conc in concentrations:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            consider(np.random.dirichlet([conc] * n))

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 0.5,
                  tau_end: float = 1e-10,
                  tau_decay: float = 0.93,
                  lr: float = 0.03,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-9) -> np.ndarray:
    """Projected Adam minimising τ·log‑sum‑exp(conv) with a slow τ‑schedule."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        # decay τ every few hundred steps
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)          # soft‑max weights
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape shallow wells
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (lr * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (more iterations, smaller start step)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑conv objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # gradient of the active max‑indices
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-3,
                        final_temp: float = 5e-8,
                        cooling_factor: float = 0.9998,
                        step_scale_init: float = 0.06,
                        step_scale_min: float = 5e-7) -> np.ndarray:
    """Metropolis hill‑climb with a very slow temperature schedule."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            # if we accepted, slowly increase step size (allows larger moves later)
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional global jitter to diversify
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.02)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # a small safety margin

    n = 2048
    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0.  Simple fallback if a known construction exists
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1.  Broad generation of candidates (≈ 180 s)
    # -----------------------------------------------------------------
    explore_deadline = deadline - 620.0
    if time.time() < explore_deadline:
        print("[explore] generating candidates …")
        cand_vecs = explore_candidates(n, explore_deadline, keep=16)
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_score:
                best_score, best_seq = val, vec.tolist()
        print(f"[explore] best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2.  Long Adam optimisation from the best few candidates (≈ 400 s)
    # -----------------------------------------------------------------
    if best_seq is not None:
        # leave roughly 260 s for polishing + annealing
        adam_deadline = min(deadline - 260.0,
                            start + 150.0 + (deadline - start) * 0.40)
        print("[adam] stage start")
        # prepare a small pool of start points
        start_points = []

        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        start_points.append(best_vec)

        start_points.append(np.full(n, 1.0 / n))
        for _ in range(5):
            start_points.append(project_to_simplex(np.random.rand(n)))

        # top exploration vectors (if any)
        for vec in cand_vecs[:4]:
            start_points.append(vec)

        refined_best = best_vec.copy()
        refined_val = best_score

        for idx, a0 in enumerate(start_points):
            if time.time() >= adam_deadline:
                break
            local_deadline = adam_deadline - (time.time() - start) - 0.5
            if local_deadline <= time.time():
                break
            print(f"[adam {idx}] start")
            a_opt = adam_optimize(a0, local_deadline,
                                  tau_start=0.5,
                                  tau_end=1e-10,
                                  tau_decay=0.93,
                                  lr=0.03)
            val = compute_score(a_opt)
            print(f"[adam {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_opt.copy()

        best_score, best_seq = refined_val, refined_best.tolist()
        print(f"[adam] best after long Adam = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3.  Sub‑gradient polishing (≈ 180 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[polish] start")
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.15)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4.  Simulated annealing – final fine tuning (≈ 60 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[anneal] start")
        a_final = simulated_annealing(a_start, anneal_deadline,
                                      init_temp=5e-3,
                                      final_temp=5e-8,
                                      cooling_factor=0.9998,
                                      step_scale_init=0.06)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – return a uniform vector if nothing was produced
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
