"""
A compact optimiser for the step‑function lower‑bound problem.

Main ideas
----------
1. **Fast global exploration** – many random Dirichlet draws together with a
   fine grid of discrete Beta‑distributions (a_i ∝ i^{α‑1}(n‑i+1)^{β‑1}).
   All candidates are projected onto the probability simplex and scored with
   the exact objective 2·n·max(conv) (the sum of the coefficients is 1).

2. **LP‑based refinement** – given a current vector `f` we solve the linear
   program  

        maximise   Σ g_j
        subject to 0 ≤ g               (non‑negative)
                   (f * g)_k ≤ M  for all k

   where `M = max(f*f)`.  The LP finds a vector `g` that can be added to `f`
   without increasing the worst convolution entry.  By mixing `f` and the
   normalised `g` we obtain a strictly improved step‑function.  Repeating
   this step quickly drives the objective down.

3. **Projected sub‑gradient polishing** – once the LP phase stalls we run a
   few thousand iterations of exact sub‑gradient descent on the true max‑
   objective, always projecting back onto the simplex.

All phases respect the 1000 s wall‑clock budget and the best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Geometry utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for the global search)
# ----------------------------------------------------------------------


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta distribution a_i ∝ i^{α‑1}(n‑i+1)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_dirichlet(n: int, conc: float) -> np.ndarray:
    """Draw a random point on the simplex from a symmetric Dirichlet."""
    a = np.random.dirichlet([conc] * n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def lp_refine(a0: np.ndarray, deadline: float, mix: float = 0.5) -> np.ndarray:
    """
    Iteratively improve a vector using the LP described in the statement.
    `mix` is the convex‑combination weight (0 < mix ≤ 1).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    # pre‑allocate the convolution matrix (will be overwritten each loop)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)

    while time.time() < deadline:
        # current worst convolution value
        M = full_convolution(a).max()

        # build the Toeplitz matrix for (a * g) ≤ M
        # each column i contributes a shifted copy of a
        for i in range(n):
            A[i:i + n, i] = a

        # LP: maximise Σ g_j  <=> minimise -Σ g_j
        c = -np.ones(n, dtype=float)
        b = np.full(m, M, dtype=float)
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # LP failed – abort the refinement phase
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # scale to the same ℓ₁‑norm as a (which is 1)
        g = g0 / sum_g0

        # convex combination (preserves the sum = 1)
        a_new = (1.0 - mix) * a + mix * g
        a_new = project_to_simplex(a_new)

        cur_val = compute_score(a_new)

        # keep the best vector seen so far
        if cur_val < best_val:
            best_val, best_a = cur_val, a_new.copy()

        # accept the new point for the next iteration if it does not worsen M
        if cur_val <= best_val + 1e-12:
            a = a_new
        else:
            # no improvement – try a smaller mixing factor
            mix *= 0.5
            if mix < 1e-3:
                break

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as in the reference implementation)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient (sum of contributions from all active shifts)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # -----------------------------------------------------------------
    # Parameters
    # -----------------------------------------------------------------
    n = 1024                         # moderate size – LP is cheap enough
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global exploration (random Dirichlet + fine Beta grid)
    # -----------------------------------------------------------------
    # keep the top few candidates (sorted by score)
    top_k = 8
    best_candidates = []   # list of (score, vector)

    def consider(vec: np.ndarray):
        nonlocal best_candidates, best_val, best_seq
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best_candidates) < top_k:
            best_candidates.append((sc, vec.copy()))
            best_candidates.sort(key=lambda x: x[0])
        elif sc < best_candidates[-1][0]:
            best_candidates[-1] = (sc, vec.copy())
            best_candidates.sort(key=lambda x: x[0])

        if sc < best_val:
            best_val, best_seq = sc, vec.tolist()
            print(f"[global] new best score = {best_val:.6f}")

    # ---- random Dirichlet draws (wide concentration range) --------------
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = 1500 // len(concs)
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            consider(generate_dirichlet(n, conc))

    # ---- systematic Beta grid (α,β ∈ [0.02, 2] on log‑scale) ------------
    al_vals = np.logspace(np.log10(0.02), np.log10(2.0), 30)
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, alpha, beta))

    # ---- random Beta draws (more fine‑grained) ---------------------------
    for _ in range(2000):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)   # roughly 0.01 … 2
        beta = 10 ** np.random.uniform(-2, 0.3)
        consider(generate_beta(n, alpha, beta))

    # keep the best candidate from the global stage as a start vector
    if best_seq is None:
        # fallback to uniform if everything failed
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    print(f"[global] best score after exploration = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. LP‑based refinement (main driver of improvement)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 200.0   # reserve time for polishing
    if time.time() < lp_deadline:
        start_vec = np.array(best_seq, dtype=float)
        start_vec = project_to_simplex(start_vec)
        refined = lp_refine(start_vec, lp_deadline, mix=0.5)
        refined_val = compute_score(refined)
        print(f"[lp] refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val, best_seq = refined_val, refined.tolist()

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (finer local optimisation)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 30.0
    if time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.45)
        polished_val = compute_score(a_polished)
        print(f"[polish] polished score = {polished_val:.6f}")
        if polished_val < best_val:
            best_val, best_seq = polished_val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Final safety net – ensure we return something sensible
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
