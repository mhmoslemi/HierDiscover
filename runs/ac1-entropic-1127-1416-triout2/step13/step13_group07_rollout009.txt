"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The implementation follows a multi‑stage strategy:

1. **Coarse exploration** of several analytic families (product‑V,
   left‑heavy power, exponential‑U, mixed uniform+U, Beta).  The best
   parameters are kept.

2. **Random dense sampling** of those families.

3. **Local refinement** of the best product‑α and Beta parameters by a
   simple 1‑D hill‑climb.

4. **Adam optimisation** of a smooth‑max surrogate of the true objective.
   The surrogate temperature β is increased gradually (continuation) and
   the vector is kept symmetric after every few steps.

5. **Projected sub‑gradient descent** on the exact ‑max‑objective.

6. **Greedy hill‑climb** that transfers a small amount of mass from a
   high‑gradient index to a low‑gradient one.  The move is accepted only
   if it improves the true score.

All stages respect a common wall‑clock budget (default 1000 s).  The
best vector found is returned as a plain list of floats.  The code uses
only NumPy and the provided ``evaluate_sequence`` for the final official
score.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# Core utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using real FFT (O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective:
        2·n·max(conv(a,a)) / (sum(a)**2)
    a is assumed non‑negative.
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v

# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """V‑shaped family a_i ∝ [(i+½)(n‑½‑i)]^{‑α} , then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} , left‑heavy."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Discrete Beta‑type density on the grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)

def adam_optimize(start_a: np.ndarray,
                  start_time: float,
                  deadline: float,
                  max_iters: int = 300_000,
                  beta_start: float = 5.0,
                  beta_end: float = 20_000.0,
                  lr: float = 0.014,
                  sym_interval: int = 25) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    ``beta`` grows geometrically from ``beta_start`` to ``beta_end``.
    Symmetry (a_i = a_{n-1-i}) is enforced every ``sym_interval`` steps.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise via a log‑softmax (guarantees positivity)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    a_cur = np.exp(z)
    a_cur = a_cur / a_cur.sum()
    best_a = a_cur.copy()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)
            # recompute a after symmetrisation
            a = np.exp(z)
            a = a / a.sum()

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the log‑softmax layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # periodic true‑objective check
        if it % 4000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     start_time: float,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_greedy(start_a: np.ndarray,
                      start_time: float,
                      deadline: float,
                      max_steps: int = 8_000_000,
                      init_scale: float = 0.15) -> Tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a small fraction of mass from a high‑gradient
    entry to a random other entry whenever this reduces the objective.
    The step size ``scale`` is adapted based on success.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    scale = init_scale

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # donor = index with largest contribution to the current max‑conv
        # (a simple proxy: the currently largest entry)
        i = int(np.argmax(cur_a))
        j = np.random.randint(n)
        if i == j:
            continue

        delta = scale * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            scale = min(scale * 1.03, 0.5)
        else:
            scale = max(scale * 0.97, 1e-4)

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] greedy step {step}, best {best_score:.6f}")

    return best_a, best_score


def hill_climb_alpha(alpha0: float, n: int,
                     start_time: float,
                     deadline: float) -> float:
    """1‑D hill‑climb for the product‑family exponent α."""
    best = alpha0
    best_score = compute_score(symmetric_product_family(n, best))
    step = 0.8
    while time.time() < deadline and step > 1e-4:
        improved = False
        for sign in (-1, 1):
            cand = max(0.0, best + sign * step)
            sc = compute_score(symmetric_product_family(n, cand))
            if sc < best_score:
                best_score = sc
                best = cand
                improved = True
                print(f"[{time.time():.1f}] α refined to {best:.6f} score {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best


def hill_climb_beta(p0: float, q0: float, n: int,
                    start_time: float,
                    deadline: float) -> Tuple[float, float]:
    """2‑D hill‑climb for the Beta‑family parameters (p,q)."""
    best_p, best_q = p0, q0
    best_score = compute_score(beta_family(n, best_p, best_q))
    step = 0.4
    while time.time() < deadline and step > 1e-3:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(0.01, best_p + dp)
                cand_q = max(0.01, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_score = sc
                    best_p, best_q = cand_p, cand_q
                    improved = True
                    print(f"[{time.time():.1f}] β refined to p={best_p:.6f}, q={best_q:.6f} score {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
    return best_p, best_q

        sc = compute_score(vec)

            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] refined Beta p={refined_p:.6f}, q={refined_q:.6f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 5.  Adam refinement (smooth‑max, enforced symmetry)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        remaining = deadline - time.time() - 150.0
        max_iters_adam = int(min(800_000, remaining * 1500))   # heuristic
        print(f"[{time.time():.1f}] Adam refinement (max_iters={max_iters_adam})")
        a_opt, sc_opt = adam_optimize(
            start_a=best_seq,
            start_time=start_time,
            deadline=deadline - 150.0,
            max_iters=max_iters_adam,
            beta_start=5.0,
            beta_end=30_000.0,
            lr=0.012,
            sym_interval=25,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6.  Projected sub‑gradient descent (exact max)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        remaining = deadline - time.time() - 60.0
        max_iters_sub = int(min(2_500_000, remaining * 2500))
        print(f"[{time.time():.1f}] sub‑gradient descent (max_iters={max_iters_sub})")
        a_sub, sc_sub = subgrad_optimize(
            start_a=best_seq,
            start_time=start_time,
            deadline=deadline - 60.0,
            max_iters=max_iters_sub,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub
            print(f"[{time.time():.1f}] after sub‑gradient best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7.  Final greedy hill‑climb (mass‑transfer moves)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_steps_greedy = int(min(8_000_000, remaining * 2000))
        print(f"[{time.time():.1f}] final greedy hill‑climb (steps={max_steps_greedy})")
        a_final, sc_final = hill_climb_greedy(
            start_a=best_seq,
            start_time=start_time,
            deadline=deadline,
            max_steps=max_steps_greedy,
            init_scale=0.15,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final
            print(f"[{time.time():.1f}] greedy improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 8.  Fallback to a safe uniform vector if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] fallback uniform score = {best_score:.6f}")

    # Final official evaluation (security checks are enforced inside)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the competition harness
    return best_seq.tolist()
