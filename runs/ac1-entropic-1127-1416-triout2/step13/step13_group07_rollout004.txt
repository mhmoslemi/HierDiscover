def piecewise_log_to_seq(params: np.ndarray, n: int) -> np.ndarray:
    """
    Convert a vector of log‑heights defined at equally spaced knots into a full
    length‑n non‑negative sequence that is normalised to unit sum.
    """
    m = params.shape[0]
    # exponentiate to get positive heights at knots
    knots = np.exp(params)
    # interpolation grid
    x_full = np.linspace(0.0, 1.0, n, endpoint=False)
    x_knots = np.linspace(0.0, 1.0, m)
    seq = np.interp(x_full, x_knots, knots)
    # ensure non‑negativity and normalise
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= total
    return seq

def de_optimize_piecewise(n: int,
                          num_knots: int,
                          deadline: float,

# Up‑sampling helper
def upscale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Up‑sample a distribution to a larger support by linear interpolation.
    The result is re‑normalised to unit sum.
    """
    n_old = seq.shape[0]
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = np.maximum(seq_new, 0.0)
    total = seq_new.sum()
    if total == 0.0:
        seq_new = np.full(new_n, 1.0 / new_n)
    else:
        seq_new /= total
    return seq_new

"""
Hybrid optimisation for the step‑function attack.

The routine mixes three complementary ideas:

1. **Low‑dimensional differential evolution** on a piecewise‑linear log‑height
   model (few knots, many degrees of freedom are handled by the interpolation).
   This quickly finds a promising continuous shape.

2. **Hierarchical up‑sampling** – the best shape is linearly interpolated to
   larger supports (16384 → 32768 → 65536).  Each size is refined with
   *Adam* on a smooth‑max surrogate, followed by a projected sub‑gradient
   descent on the exact max‑objective and a final greedy mass‑transfer hill‑climb.

3. **Time‑aware scheduling** – every stage checks the global wall‑clock deadline
   (the budget is 1000 s, a few seconds are kept as safety margin).  The best
   sequence found is finally evaluated with the supplied ``evaluate_sequence``.

All helper functions are defined at module level (no closures) and only NumPy
(and SciPy if available) are used.
"""

import time
import math
import random
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for differential evolution
# ----------------------------------------------------------------------
try:
    from scipy.optimize import differential_evolution
except Exception:  # pragma: no cover
    differential_evolution = None  # type: ignore

# ----------------------------------------------------------------------
# Global state used by the DE callbacks (required to avoid closures)
# ----------------------------------------------------------------------
_DE_STATE = {
    "n": None,          # problem size for the current DE run
    "deadline": None,   # time at which we must stop
    "best_seq": None,   # best full‑length sequence seen so far
    "best_score": float("inf"),
}


# ----------------------------------------------------------------------
# Core utilities (FFT convolution, scoring, projections)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)**2).
    The vector ``a`` is assumed non‑negative.
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    (sum = 1, v ≥ 0).  Implementation follows
    Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Optimisers used later (Adam, sub‑gradient, greedy hill‑climb)
# ----------------------------------------------------------------------
def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 5.0,
                         beta_end: float = 20_000.0,
                         lr: float = 0.012,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of the max‑convolution."""
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)               # log‑softmax variables
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a (see derivation in the original script)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 3000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_greedy(start_a: np.ndarray,
                      deadline: float,
                      max_steps: int = 10_000_000,
                      init_scale: float = 0.15) -> Tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb that moves a fraction of mass from the largest
    entry to a random other entry whenever this reduces the objective.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score
    scale = init_scale

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        i = np.argmax(cur_a)                # donor
        j = np.random.randint(n)            # receiver
        if i == j:
            continue

        delta = scale * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            scale = min(scale * 1.03, 0.5)
        else:
            scale = max(scale * 0.97, 1e-4)

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] greedy step {step}, best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Piecewise‑linear log‑height model (used by differential evolution)
# ----------------------------------------------------------------------
def piecewise_log_to_seq(params: np.ndarray, n: int) -> np.ndarray:
    """
    Convert a vector of log‑heights defined at equally spaced knots into a
    full length‑n non‑negative probability vector (unit sum).
    """
    m = params.shape[0]
    knots = np.exp(params)                     # positive heights at knots
    x_full = np.linspace(0.0, 1.0, n, endpoint=False)
    x_knots = np.linspace(0.0, 1.0, m)
    seq = np.interp(x_full, x_knots, knots)
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq = seq / total
    return seq


def de_objective(p):
    """Objective wrapper for scipy's differential evolution."""
    n = _DE_STATE["n"]
    deadline = _DE_STATE["deadline"]
    if time.time() >= deadline:
        # large penalty – this point will be ignored by the optimiser
        return 1e12
    seq = piecewise_log_to_seq(np.array(p, dtype=np.float64), n)
    return compute_score(seq)


def de_callback(xk, convergence):
    """Called by DE after each generation; updates the global best."""
    if time.time() >= _DE_STATE["deadline"]:
        raise RuntimeError("deadline reached")
    seq = piecewise_log_to_seq(np.array(xk, dtype=np.float64), _DE_STATE["n"])
    sc = compute_score(seq)
    if sc < _DE_STATE["best_score"]:
        _DE_STATE["best_score"] = sc
        _DE_STATE["best_seq"] = seq.copy()
    return False


def de_optimize_piecewise(n: int,
                          num_knots: int,
                          deadline: float,
                          maxiter: int = 30,
                          popsize: int = 15) -> Tuple[np.ndarray, float]:
    """
    Run SciPy's differential evolution on the low‑dimensional log‑height
    parameters of a piecewise‑linear distribution.
    Returns the best full‑length sequence and its score.
    """
    if differential_evolution is None:
        # SciPy unavailable – fall back to uniform distribution
        uni = np.full(n, 1.0 / n, dtype=np.float64)
        return uni, compute_score(uni)

    # initialise global state for the callbacks
    _DE_STATE["n"] = n
    _DE_STATE["deadline"] = deadline
    _DE_STATE["best_seq"] = None
    _DE_STATE["best_score"] = float("inf")

    bounds = [(-10.0, 10.0)] * num_knots

    try:
        differential_evolution(
            de_objective,
            bounds,
            maxiter=maxiter,
            popsize=popsize,
            polish=False,
            updating='deferred',
            seed=int(time.time()) % (2**32 - 1),
            callback=de_callback,
            disp=False,
        )
    except RuntimeError:
        # deadline hit inside the callback – we simply continue
        pass

    if _DE_STATE["best_seq"] is not None:
        return _DE_STATE["best_seq"], _DE_STATE["best_score"]
    # fallback (should rarely happen)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    return uni, compute_score(uni)

# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------
def upscale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Upsample ``seq`` to length ``new_n`` by linear interpolation and renormalise.
    """
    old_n = seq.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = np.maximum(seq_new, 0.0)
    total = seq_new.sum()
    if total == 0.0:
        seq_new = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        seq_new = seq_new / total
    return seq_new
