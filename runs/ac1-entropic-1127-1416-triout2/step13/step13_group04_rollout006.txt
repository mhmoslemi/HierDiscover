"""
Improved optimiser for the self‑convolution bound.

Main ideas
----------
* A fast parametric sweep (same families as the reference) to obtain a
  decent starter vector.
* A new “smooth‑max” gradient descent: we replace the nonsmooth
  max (conv) by a softmax with an increasing temperature (β) and perform
  projected gradient steps on the simplex.  The gradient of the
  soft‑max objective can be obtained by a single FFT‑based convolution,
  making each iteration cheap.
* Several short runs of the gradient method from different random
  initial points, keeping the best result.
* A final deterministic hill‑climb (greedy mass transfers) for fine
  polishing.

All vectors are kept normalised to Σa = 1, which makes the objective
scale‑invariant.  The final sequence is returned as a plain Python list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a (full linear convolution)."""
    return convolve_fft(a, a)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a)) .
    The objective is invariant under scaling, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the broad sweep)
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def geometric(n: int, q: float) -> np.ndarray:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def logistic_shape(n: int, center_frac: float, scale_frac: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    scale = max(scale_frac * n, 1e-6)
    a = 1.0 / (1.0 + np.exp((i - center) / scale))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def random_dirichlet(n: int, concentration: float) -> np.ndarray:
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    s = y.sum()
    if s == 0.0:
        return None
    return y / s


# ----------------------------------------------------------------------
# Broad parametric sweep
# ----------------------------------------------------------------------
def parametric_sweep(n: int, time_deadline: float) -> np.ndarray:
    """Explore a number of families and keep the best vector found."""
    best_vec = None
    best_val = float('inf')

    # power‑law
    for p in np.linspace(0.10, 2.00, 191):
        if time.time() >= time_deadline:
            return best_vec
        a = power_law(n, p)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # geometric
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 91),
            np.linspace(0.990, 0.9999, 110))):
        if time.time() >= time_deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # mixed plateau + power
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= time_deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # beta shape
    alphas = np.linspace(-0.8, 3.0, 20)
    betas = np.linspace(-0.8, 3.0, 20)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= time_deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # logistic
    for center_frac in np.linspace(0.2, 0.8, 13):
        for scale_frac in np.linspace(0.01, 0.20, 20):
            if time.time() >= time_deadline:
                return best_vec
            a = logistic_shape(n, center_frac, scale_frac)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # random Dirichlet
    for _ in range(1500):
        if time.time() >= time_deadline:
            break
        conc = 10 ** np.random.uniform(-2, 1)   # 0.01 … 10
        a = random_dirichlet(n, conc)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (new component)
# ----------------------------------------------------------------------
def smooth_max_descent(
    a0: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    beta0: float = 0.5,
    beta_mult: float = 1.015,
    beta_max: float = 180.0,
) -> np.ndarray:
    """
    Projected gradient descent on the smooth approximation of max(conv).
    The smoothness parameter β is increased slowly so that the algorithm
    first moves on a gentle landscape and then focuses on the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    offset = n - 1                     # position of a_i in the convolution with w
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    step = step0
    beta = beta0

    while time.time() < time_deadline:
        t += 1
        # current convolution and smooth weights
        conv = full_convolution(a)
        M = conv.max()
        # soft‑max weights (numerically stable)
        w = np.exp(beta * (conv - M))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient = 2 * (w * a_rev) shifted by offset
        a_rev = a[::-1]
        grad_full = convolve_fft(w, a_rev)          # length = 3n‑2
        g = 2.0 * grad_full[offset: offset + n]

        # step‑size schedule (√t decay)
        step = step0 / math.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        # evaluate true objective for the new point
        cur_score = compute_score(a_new)

        # accept if improvement (greedy descent)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # after a good move be slightly bolder
            step0 = min(step0 * 1.02, 1.0)

        a = a_new

        # increase smoothness
        if beta < beta_max:
            beta = min(beta * beta_mult, beta_max)

        # occasional jitter to escape tiny plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a


# ----------------------------------------------------------------------
# Hill‑climb (deterministic greedy mass transfers) – same as reference
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback (previously discovered construction)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore[name‑defined]
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad parametric sweep (≈ 150 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 350.0
    n = 4096
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Smooth‑max gradient descent (several short runs)
    # -----------------------------------------------------------------
    grad_deadline = deadline - 200.0
    if best_seq is not None and time.time() < grad_deadline:
        a0 = np.array(best_seq, dtype=float)
        # we will perform a few independent runs (different seeds)
        remaining = grad_deadline - time.time()
        runs = max(1, int(remaining // 60))   # aim for roughly a minute per run
        for r in range(runs):
            run_deadline = min(time.time() + 55.0, grad_deadline)
            # occasionally jitter the start point to diversify
            if r > 0:
                jitter = np.random.randn(a0.size) * 0.001
                start_vec = project_to_simplex(a0 + jitter)
            else:
                start_vec = a0
            a_opt = smooth_max_descent(start_vec, run_deadline)
            val = compute_score(a_opt)
            print(f"[smooth‑max] run {r+1}/{runs} score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                a0 = a_opt   # seed next runs from the improved point

    # -----------------------------------------------------------------
    # 3. Additional stochastic exploration (short random restarts)
    # -----------------------------------------------------------------
    random_deadline = deadline - 80.0
    if best_seq is not None and time.time() < random_deadline:
        while time.time() < random_deadline:
            # draw a fresh random Dirichlet point and give it a quick polish
            conc = 10 ** np.random.uniform(-2, 1)
            a_rand = random_dirichlet(n, conc)
            if a_rand is None:
                continue
            # a brief gradient run (≈ 10 s)
            run_deadline = min(time.time() + 12.0, random_deadline)
            a_opt = smooth_max_descent(a_rand, run_deadline, step0=0.15,
                                       beta0=0.4, beta_mult=1.02, beta_max=150.0)
            val = compute_score(a_opt)
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                print(f"[random‑restart] improved to {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Final hill‑climb (remaining time)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluate_sequence
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
