def subgradient_adagrad(a0, deadline, step0=0.5, eps=1e-8):
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_val = compute_score(a)
    g_sum_sq = np.zeros(n)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # active indices: those achieving max within tolerance
        eps_active = max_val * 1e-12
        active = np.where(conv >= max_val - eps_active)[0]
        # compute subgradient
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # Adagrad step
        # accumulate squared gradients
        g_sum_sq += g**2
        # adapt step size
        adj_lr = step0 / (np.sqrt(g_sum_sq) + eps)
        a = a - adj_lr * g
        # project to simplex
        a = project_to_simplex(a)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
        # optionally jitter
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)
    return best
```

def subgradient(a):
    conv = full_convolution(a)
    max_val = conv.max()
    # tolerance for active; could use relative tolerance
    eps = max_val * 1e-12
    active = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    idx = np.arange(a.shape[0])
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < a.shape[0])
        g[mask] += 2.0 * a[j[mask]]
    return g, max_val
```

def generate_piecewise_linear(n, num_segments=3):
    # Choose breakpoints at integer positions, ensure start at 0 and end at n-1
    # Random positions sorted inside
    breaks = np.sort(np.random.choice(np.arange(1, n-1), size=num_segments-1, replace=False))
    # Add endpoints
    points = np.concatenate(([0], breaks, [n-1]))
    # Random heights for each breakpoint (positive)
    heights = np.random.rand(len(points)) + 1e-6
    # Interpolate linearly
    x = np.arange(n)
    a = np.interp(x, points, heights)
    a = np.maximum(a, 0.0)
    a = project_to_simplex(a)
    return a
```

"""
Improved search for a step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Broad candidate pool** – generate a diverse set of simplex‑normalised
   vectors from many parametric families (power‑law, exponential, beta,
   Gaussian, sine‑power, U‑shape) together with a large number of random
   Dirichlet draws (including extremely sparse draws) and random
   piece‑wise‑linear shapes.

2. **Smooth‑max Adam** – from the best few candidates run a projected Adam
   optimiser on the smooth approximation
      τ·log‑sum‑exp(conv(a,a)/τ)
   with an aggressive annealing schedule (τ down to 1e‑12).  This quickly
   reduces the true objective while keeping the iterates on the simplex.

3. **Adaptive sub‑gradient (AdaGrad)** – starting from the best Adam result
   perform a projected AdaGrad descent on the *exact* max‑convolution
   (sub‑gradient obtained from the active lag indices).  The per‑coordinate
   learning rates adapt automatically, allowing a very fine‑grained
   polishing phase.

4. **Final simulated‑annealing hill‑climb** – a Metropolis walk with
   temperature cooling makes a last escape from shallow local minima.

All phases respect the 1000 s wall‑clock budget; the best sequence found
is returned as a plain Python list of ``float``.  The implementation uses
only ``numpy`` and the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    Returns g_i = 2·Σ_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    m = len(w)
    rev_w = w[::-1]
    size = 1 << ((n + m - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def smooth_max(conv: np.ndarray, tau: float) -> (float, np.ndarray):
    """
    Smooth approximation S = τ·log(∑ exp(conv/τ)).
    Returns also the soft‑max weights (∑ w = 1).
    """
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_ushape(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_piecewise_linear(n: int, num_segments: int = 4) -> np.ndarray:
    """Random piece‑wise linear shape with ``num_segments`` linear pieces."""
    if num_segments < 2:
        num_segments = 2
    # random interior break points
    interior = np.sort(
        np.random.choice(np.arange(1, n - 1), size=num_segments - 1, replace=False)
    )
    points = np.concatenate(([0], interior, [n - 1]))
    heights = np.random.rand(len(points)) + 1e-6
    x = np.arange(n)
    a = np.interp(x, points, heights)
    a = np.maximum(a, 0.0)
    a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# Exploration – generate a pool of diverse candidates
# ----------------------------------------------------------------------
def explore_candidates(
    n: int, deadline: float, keep: int = 16
) -> list[np.ndarray]:
    """Generate many vectors, keep the ``keep`` best (according to compute_score)."""
    best = []  # list of (score, vector)

    def consider(v: np.ndarray):
        if not np.isfinite(v).all():
            return
        v = np.maximum(v, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        if len(best) < keep:
            best.append((sc, v.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, v.copy())
            best.sort(key=lambda x: x[0])

    # 0) uniform
    consider(np.full(n, 1.0 / n))

    # 1) power‑law sweeps
    for p in np.linspace(0.0, 4.0, 41):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))
        consider(generate_powerlaw(n, -p))

    # 2) exponential sweeps
    for lam in np.linspace(-0.5, 0.5, 81):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # 3) beta family
    al_vals = np.linspace(0.1, 5.0, 21)
    be_vals = np.linspace(0.1, 5.0, 21)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    # 4) Gaussian widths
    for sigma in np.linspace(0.02 * n, 0.6 * n, 15):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # 5) sine‑power
    for p in np.linspace(0.1, 8.0, 20):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # 6) U‑shape
    for p in np.linspace(0.1, 5.0, 20):
        if time.time() >= deadline:
            break
        consider(generate_ushape(n, p))

    # 7) random Dirichlet draws (including very sparse)
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, (keep * 300) // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            consider(np.random.dirichlet([conc] * n))

    # 8) random piece‑wise‑linear shapes
    while time.time() < deadline and len(best) < keep * 2:
        consider(generate_piecewise_linear(n, num_segments=np.random.randint(2, 6)))

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.5,
    tau_end: float = 1e-12,
    tau_decay: float = 0.85,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """Projected Adam minimising τ·log‑sum‑exp(conv) with exponential τ‑schedule."""
    a = a0.copy()
    n = a.shape[0]
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        # anneal τ every few hundred steps
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)          # soft‑max weights
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (lr * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Adaptive sub‑gradient descent (AdaGrad) on the exact max‑conv objective
# ----------------------------------------------------------------------
def subgradient_adagrad(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.25,
) -> np.ndarray:
    """AdaGrad on the true objective 2·n·max(conv) with per‑coordinate steps."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    g_acc = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # active set – lags within tolerance of the max
        eps = max_val * 1e-12 + 1e-15
        active = np.flatnonzero(conv >= max_val - eps)

        # build weight vector for the active lags (average of sub‑gradients)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        g = conv_with_weights(a, w)   # exact sub‑gradient of max (averaged)

        # AdaGrad accumulation and adaptive step
        g_acc += g ** 2
        adapt_lr = step0 / (np.sqrt(g_acc) + 1e-12)
        a = a - adapt_lr * g

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional random jitter
        if it % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.008 / np.sqrt(it))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------
def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 5e-4,
    final_temp: float = 1e-9,
    cooling_factor: float = 0.9992,
    step_scale_init: float = 0.08,
    step_scale_min: float = 1e-6,
) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.04, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.96, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    n = 2048
    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Known construction (if available)
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fall = [float(x) for x in fallback]
            val = evaluate_sequence(fall)
            print(f"[fallback] score = {val:.6f}")
            if val < best_score:
                best_score, best_seq = val, fall
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Broad exploration (≈ 120 s)
    # -----------------------------------------------------------------
    explore_deadline = deadline - 700.0
    if time.time() < explore_deadline:
        print("[explore] generating candidates …")
        cand_vecs = explore_candidates(n, explore_deadline, keep=20)
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_score:
                best_score, best_seq = val, vec.tolist()
        print(f"[explore] best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam optimisation from the best few candidates (≈ 300 s)
    # -----------------------------------------------------------------
    if best_seq is not None:
        # allocate roughly 30 % of remaining time to Adam
        adam_deadline = min(deadline - 250.0,
                            time.time() + (deadline - time.time()) * 0.30)
        start_points = []

        # ensure we have a simplex‑normalised version
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        start_points.append(best_vec)

        # uniform start
        start_points.append(np.full(n, 1.0 / n))

        # a few random starts
        for _ in range(3):
            start_points.append(project_to_simplex(np.random.rand(n)))

        # top candidates from exploration
        for vec in cand_vecs[:4]:
            start_points.append(vec)

        refined_best = best_vec.copy()
        refined_val = best_score

        for idx, a0 in enumerate(start_points):
            if time.time() >= adam_deadline:
                break
            local_deadline = adam_deadline - (time.time() - start) - 0.5
            if local_deadline <= time.time():
                break
            print(f"[adam {idx}] start")
            a_opt = adam_optimize(a0, local_deadline)
            val = compute_score(a_opt)
            print(f"[adam {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_opt.copy()

        best_score, best_seq = refined_val, refined_best.tolist()
        print(f"[adam] best after Adam = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Adaptive sub‑gradient polishing (≈ 300 s)
    # -----------------------------------------------------------------
    if best_seq is not None:
        a0 = np.array(best_seq, dtype=float)
        adagrad_deadline = deadline - 150.0
        if time.time() < adagrad_deadline:
            print("[adagrad] start")
            a_ref = subgradient_adagrad(a0, adagrad_deadline)
            val = compute_score(a_ref)
            print(f"[adagrad] polished score = {val:.6f}")
            if val < best_score:
                best_score, best_seq = val, a_ref.tolist()

    # -----------------------------------------------------------------
    # 4. Final simulated annealing (≈ 200 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_start = np.array(best_seq, dtype=float)
        print("[anneal] start")
        a_final = simulated_annealing(a_start, deadline - 5.0)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq, dtype=float))

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
