def smooth_gradient_descent(a0, time_deadline, init_lr=0.1, gamma=0.15, min_lr=1e-5):
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    n = a.shape[0]
    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # smoothing parameter eta
        eta = max(gamma * M, 1e-6)
        # stable softmax
        exp_vals = np.exp((conv - M) / eta)
        w = exp_vals / exp_vals.sum()
        # compute gradient via convolution with reversed weights
        w_rev = w[::-1]
        grad_full = np.convolve(a, w_rev, mode='full')
        # index offset:
        idx_offset = 2 * n - 2
        g = 2.0 * grad_full[idx_offset - np.arange(n)]
        # step size schedule
        lr = init_lr / math.sqrt(t)
        # update
        a = a - lr * g
        # projection onto simplex
        a = project_to_simplex(a)
        # optionally, jitter occasionally to escape
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (init_lr * 0.01 / math.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

        # evaluate
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # boost step size perhaps
            init_lr = min(init_lr * 1.02, 0.5)
    return best_a

"""
Propose a low‑score coefficient sequence for `evaluate_sequence`.

Algorithm
---------
1. **Fallback** – try the historic best (`height_sequence_1`) if present.
2. **Broad parametric sweep** – scan several analytic families
   (power‑law, geometric, mixed plateau‑power, beta, logistic,
   sine, quadratic) and many random Dirichlet draws.  The best
   vector (normalized to sum = 1) is kept.
3. **Smooth‑gradient polishing** – minimise a smooth approximation of
   the max‑convolution using an entropic soft‑max:
       Φ_η(a) = 2·n·η·log∑_t exp((a*a)[t]/η) .
   The gradient is obtained analytically via two FFT‑based convolutions.
   A projected (onto the simplex) gradient step with a decreasing
   learning‑rate is performed until the allotted time expires.
4. **Stochastic annealing** – mass‑transfer moves (move a random
   fraction of mass from one coordinate to another) are accepted
   according to a temperature schedule.  This helps escape shallow
   local minima left by the gradient phase.
5. **Deterministic hill‑climb** – final fine‑grained greedy moves
   (transfer a small fraction of mass) that strictly improve the
   objective.

All vectors are kept on the probability simplex (Σa_i = 1) so that
the objective is scale‑invariant.  The code monitors the remaining
budget and stops gracefully before the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------
def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution (full) of a and b using a real FFT.
    Length of result = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenience wrapper: a * a (full) via FFT."""
    return fft_convolve(a, a)


# ----------------------------------------------------------------------
# Objective (sum = 1 → 2·n·max(conv))
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray | None:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def geometric(n: int, q: float) -> np.ndarray | None:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray | None:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def logistic_shape(n: int, center_frac: float, scale_frac: float) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    scale = max(scale_frac * n, 1e-6)
    a = 1.0 / (1.0 + np.exp((i - center) / scale))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def sine_shape(n: int) -> np.ndarray:
    """A single half‑sine bump."""
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    return a / a.sum()


def quadratic_shape(n: int) -> np.ndarray:
    """Tent‑like: a_i ∝ (i+1)*(n-i). Symmetric with peak in the centre."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    return a / a.sum()


def random_dirichlet(n: int, concentration: float) -> np.ndarray | None:
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    s = y.sum()
    if s == 0.0:
        return None
    return y / s

def smooth_gradient_descent(a0: np.ndarray, time_deadline: float,

    best_a = a.copy()
    best_score = compute_score(a)

    n = a.shape[0]
    it = 0
    while time.time() < time_deadline:
        it += 1
        # current convolution
        conv = full_convolution(a)
        M = conv.max()
        # smoothing parameter (never zero)
        eta = max(gamma * M, 1e-9)

        # stable soft‑max weights
        exp_shift = np.exp((conv - M) / eta)
        w = exp_shift / exp_shift.sum()

        # gradient = 2 * (a * w reversed) evaluated at specific offsets
        w_rev = w[::-1]
        grad_full = fft_convolve(a, w_rev)          # length = 3n-2
        idx = 2 * n - 2 - np.arange(n)              # map to required positions
        g = 2.0 * grad_full[idx]

        # learning‑rate schedule
        lr = max(init_lr / math.sqrt(it), min_lr)

        # gradient step + projection
        a = a - lr * g
        a = project_to_simplex(a)

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = np.random.randn(n) * (init_lr * 0.02 / math.sqrt(it))
            a = a + jitter
            a = project_to_simplex(a)

        # evaluation & bookkeeping
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                # slightly enlarge step size after a good move
                init_lr = min(init_lr * 1.05, 0.5)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer)
# ----------------------------------------------------------------------
def anneal_search(a_start: np.ndarray, time_deadline: float) -> np.ndarray:
    """Mass‑transfer simulated annealing."""
    n = a_start.shape[0]
    cur = a_start.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    temp = max(1e-4 * best_score, 1e-9)
    temp_decay = 0.9995
    step_scale = 0.07
    step_decay = 0.9998

    it = 0
    while time.time() < time_deadline:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * cur[i] * np.random.rand()
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best, best_score = cand, cand_score
                # become bolder after improvement
                step_scale = min(step_scale * 1.03, 0.6)

        temp = max(temp * temp_decay, 1e-12)
        step_scale = max(step_scale * step_decay, 1e-7)

    return best


# ----------------------------------------------------------------------
# Deterministic hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_iter: int = 600_000) -> np.ndarray:
    """Greedy coordinate transfers that strictly improve the objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the outer harness
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional historical fallback
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Parametric sweep (quick broad exploration)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 600.0          # keep ~600 s for refinement
    n_sweep = 4096
    if time.time() < sweep_deadline:
        vec = parametric_sweep(n_sweep, sweep_deadline)
        if vec is not None:
            val = compute_score(vec)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, vec.tolist()

    # -----------------------------------------------------------------
    # 2. Smooth‑gradient polishing (entropic smoothing of max)
    # -----------------------------------------------------------------
    grad_deadline = deadline - 300.0
    if best_seq is not None and time.time() < grad_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = smooth_gradient_descent(a0, grad_deadline)
        val = compute_score(a_polished)
        print(f"[smooth‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 3. Simulated annealing (mass‑transfer)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 100.0
    if best_seq is not None and time.time() < anneal_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_ann = anneal_search(a0, anneal_deadline)
        val = compute_score(a_ann)
        print(f"[anneal] after anneal score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_ann.tolist()

    # -----------------------------------------------------------------
    # 4. Final deterministic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline
    if best_seq is not None and time.time() < hill_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_final = hill_climb(a0, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 4096
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluator
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
