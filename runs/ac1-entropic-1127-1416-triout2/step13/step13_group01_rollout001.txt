"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Global exploration** – generate candidates from several
   parametric families (power‑law, exponential, beta, Gaussian,
   arcsine) and from random Dirichlet draws as well as a few
   “spike” vectors.  Keep the best few (default 12) after each
   generation.

2. **Smoothed minimisation** – minimise the max‑convolution
   (equivalently the objective ``2·n·max(conv)``) using Adam on the
   log‑softmax parametrisation.  The temperature τ controls a
   soft‑max approximation of the true max; it is started relatively
   high (10·max) and annealed multiplicatively (τ←τ·0.9998) until a
   hard max is used.  The best vector seen is recorded.

3. **Multiple restarts** – while time permits run the optimiser from
   the current best and from a few random Dirichlet starts.

4. **Final polishing** – a short Metropolis‑style simulated‑annealing
   pass that moves a small amount of mass between two indices and
   accepts worsening moves with a decreasing temperature.

The whole search respects the given wall‑clock budget (default 1000 s)
and finally returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of a and b using real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())         # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (used for the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant – sum of a is 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalized to sum = 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a normalized to sum = 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_arcsine(n: int) -> np.ndarray:
    """Beta(0.5,0.5) – the discrete arcsine law."""
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_spike_vector(n: int, k: int) -> np.ndarray:
    """Place equal mass on k random positions (k ≥ 2)."""
    if k >= n:
        k = n
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Helper – keep the best ‘max_keep’ candidates seen so far
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best: list, max_keep: int):
    """Project vec onto the simplex, evaluate and possibly store it."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return
    vec = vec / s                     # simplex projection
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Global baseline exploration
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 12):
    """Sweep several families and many Dirichlet samples."""
    best = []   # (score, vector)

    # ---- 1) Power‑law -------------------------------------------------
    for p in np.linspace(0.05, 2.5, 220):
        if time.time() >= deadline:
            break
        _try_candidate(generate_powerlaw(n, p), best, max_keep)

    # ---- 2) Exponential ------------------------------------------------
    for lam in np.linspace(1e-5, 0.08, 350):
        if time.time() >= deadline:
            break
        _try_candidate(generate_exponential(n, lam), best, max_keep)

    # ---- 3) Beta (including arcsine region) ---------------------------
    al_vals = np.linspace(0.2, 5.0, 30)
    be_vals = np.linspace(0.2, 5.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # arcsine (beta=0.5,0.5) as a dedicated candidate
    if time.time() < deadline:
        _try_candidate(generate_arcsine(n), best, max_keep)

    # ---- 4) Gaussian ----------------------------------------------------
    sigma_vals = np.linspace(n / 24.0, n / 2.0, 35)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        _try_candidate(generate_gaussian(n, sigma), best, max_keep)

    # ---- 5) Spike vectors (few‑point support) -------------------------
    for k in [2, 3, 5, 8, 13]:
        for _ in range(30):
            if time.time() >= deadline:
                break
            _try_candidate(generate_spike_vector(n, k), best, max_keep)

    # ---- 6) Dirichlet draws --------------------------------------------
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 800
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on log‑softmax) – core routine
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.15,
                    init_tau_factor: float = 8.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9998) -> np.ndarray:
    """
    Minimise max(conv(a,a)) using a smooth max (soft‑max) and Adam.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise logits (log‑softmax)
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature (smoothness)
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max → a (simplex‑preserving) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution & max -----
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- smooth‑max weights -----
        if tau < 1e-4:
            # use hard argmax for efficiency when τ is tiny
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # numerically stable soft‑max over convolution values
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient wrt a :  g = 2 * (a_rev * w)  (slice central part) -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)    # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for log‑softmax (x‑space) -----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature -----
        tau = max(min_tau, tau * tau_decay)

        # ----- periodic exact evaluation -----
        if it % 500 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 5e-8,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb with occasional random jitter."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)

        # propose moving a small fraction from i to j
        delta = step_scale * a[i]
        if delta == 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑project onto simplex (the move conserves the sum, only non‑negativity may be violated)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ---------------------------------------------------------------
    # Problem size – 4096 gives a finer discretisation while still fast
    # ---------------------------------------------------------------
    n = 4096

    best_seq = None
    best_val = float("inf")

    # ---------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # ---------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # ---------------------------------------------------------------
    # 1. Global baseline exploration (≈ 220 s)
    # ---------------------------------------------------------------
    baseline_deadline = deadline - 550.0   # reserve the rest for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # ---------------------------------------------------------------
    # 2. Smoothed Adam optimisation with several restarts (≈ 700 s)
    # ---------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        # single long run from the current best
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # additional short restarts while time permits
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(35.0, remaining)
            # start from a random Dirichlet (vary concentration)
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # ---------------------------------------------------------------
    # 3. Final simulated‑annealing polish (short run)
    # ---------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # ---------------------------------------------------------------
    # 4. Fallback – uniform distribution (should never be needed)
    # ---------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ---------------------------------------------------------------
    # 5. Report the official evaluation (the harness uses this)
    # ---------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
