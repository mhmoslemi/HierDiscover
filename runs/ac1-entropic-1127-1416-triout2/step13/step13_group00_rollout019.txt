    rho = np.where(mask)[0][-1]

"""
Improved search for a low‐value step‐function.

Algorithm
---------
1.  Load the historic best construction (height_sequence_1) if it exists.
2.  Generate a large pool of random/simple candidate vectors (uniform,
    power‑law, exponential, beta, sine‑power, Dirichlet draws) and keep the
    few with the smallest exact score 2·n·max(conv).
    All candidates are projected onto the probability simplex (sum = 1).
3.  For each kept candidate run an aggressive simulated‑annealing
    hill‑climber.  The move transfers a fraction of mass from a source
    index (chosen proportionally to its current value) to a random
    destination index.  Temperature and step size decay linearly with the
    elapsed fraction of the dedicated time budget.
    The objective is the exact score ``compute_score`` (which equals the
    official ``evaluate_sequence`` when the sum is 1).  The best sequence
    seen during all annealing runs is retained.
4.  Return the best sequence found (as a plain list of floats).  The whole
    routine respects the 1000‑second wall‑clock limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (no padding artefacts)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)).  Here a is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used only for seeding the search)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (p may be negative)."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i).  lam may be negative."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α−1}·(n−i)^{β−1}."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n).  p>0 concentrates mass in the centre."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Simulated annealing (core optimiser)
# ----------------------------------------------------------------------


def simulated_annealing_fast(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Aggressive Metropolis hill‑climb with an annealed temperature.
    The move transfers a fraction of the source mass to a random target.
    The sum of the vector stays 1, therefore the objective equals the
    official evaluation metric.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    init_temp = 5e-1         # start temperature
    final_temp = 1e-8        # final temperature
    init_step = 0.20         # start fraction of source mass moved
    min_step = 1e-3

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total if total > 0 else 0.0

        # exponential cooling for the temperature
        T = init_temp * ((final_temp / init_temp) ** frac)

        # step fraction decays linearly
        step_frac = init_step * (1.0 - frac) + min_step

        # pick a source index, biasing towards larger entries
        # (np.random.choice with probabilities is fast enough for n≈2000)
        src = np.random.choice(n, p=a / a.sum())
        if a[src] <= 0.0:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        delta = step_frac * a[src]          # amount moved
        if delta <= 1e-15:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # numerical drift may break the simplex – re‑project occasionally
        if np.abs(cand.sum() - 1.0) > 1e-9:
            cand = project_to_simplex(cand)

        val = compute_score(cand)

        # Metropolis acceptance
        if val < cur_val or np.random.rand() < np.exp((cur_val - val) / max(T, 1e-15)):
            a = cand
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = a.copy()

        # occasional small jitter to escape flat regions
        if np.random.rand() < 0.001:
            jitter = np.random.randn(n) * (step_frac * 0.01)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = project_to_simplex(a)
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()

    return best_a

        val = evaluate_sequence(fallback_seq)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback_seq.copy()

    # --------------------------------------------------------------
    # 1.  Broad generation of candidates (≈ 150 s)
    # --------------------------------------------------------------
    cand_pool = []                     # list of (score, vector)

    def consider(v: np.ndarray):
        """Project, score and possibly keep the vector."""
        if not np.isfinite(v).all():
            return
        v = np.maximum(v, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        cand_pool.append((sc, v))

    # simple families
    consider(np.full(n, 1.0 / n))                     # uniform

    # power‑law sweep
    for p in np.linspace(0.0, 4.0, 41):
        consider(generate_powerlaw(n, p))
        consider(generate_powerlaw(n, -p))

    # exponential sweep
    for lam in np.linspace(-0.6, 0.6, 49):
        consider(generate_exponential(n, lam))

    # beta family
    al_vals = np.linspace(0.2, 5.0, 13)
    be_vals = np.linspace(0.2, 5.0, 13)
    for a in al_vals:
        for b in be_vals:
            consider(generate_beta(n, a, b))

    # sine‑power sweep
    for p in np.linspace(0.1, 8.0, 24):
        consider(generate_sine_power(n, p))

    # random Dirichlet draws (various concentrations)
    concentrations = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    for conc in concentrations:
        for _ in range(300):
            if time.time() >= deadline:
                break
            consider(np.random.dirichlet([conc] * n))

    # keep the best few (they will be refined later)
    cand_pool.sort(key=lambda x: x[0])
    top_vectors = [vec for sc, vec in cand_pool[:6]]   # keep 6 starters

    # ensure the historic fallback (projected to simplex) is also in the list
    if fallback_seq is not None:
        fb = np.array(fallback_seq, dtype=float)
        fb = project_to_simplex(fb)
        top_vectors.append(fb)

    print(f"[explore] best preliminary score = {cand_pool[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2.  Simulated‑annealing refinement (major part of the budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0:
        # split remaining time roughly equally among the starters
        per_vec_time = remaining / max(len(top_vectors), 1)
        for idx, vec in enumerate(top_vectors):
            if time.time() >= deadline:
                break
            local_deadline = time.time() + per_vec_time
            print(f"[anneal {idx}] start (budget {per_vec_time:.1f}s)")
            refined = simulated_annealing_fast(vec, local_deadline)
            val = compute_score(refined)
            print(f"[anneal {idx}] refined score = {val:.6f}")
            if val < best_score:
                best_score, best_seq = val, refined.tolist()

    # --------------------------------------------------------------
    # 3.  Fallback – uniform vector if nothing was discovered
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 4.  Final report using the official evaluation function
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
