def adam_polish(a0, deadline, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=0
    best_a=a.copy()
    best_M=full_convolution(a).max()
    while time.time() < deadline:
        t+=1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]
        g = np.zeros_like(a)
        # subgrad
        n = len(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >=0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # Adam moments update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)
        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    return best_a

"""
search_step_function_v3.py

A new optimiser for the ``evaluate_sequence`` objective.

Main ideas
----------

1. **Extended parametric families** – In addition to the pure
   reverse power‑law we optimise a *reverse power‑law multiplied by a
   geometric factor*  a_i ∝ (n‑i)^p·q^i.  This extra exponential decay
   gives us an extra degree of freedom to push the auto‑convolution
   peak below the best value obtained previously.

2. **Adam‑style projected gradient descent** – The original sub‑gradient
   descent uses a diminishing step size and can stall.  We replace it by
   a projected Adam optimiser on the same sub‑gradient (active‑set)
   which adapts the step automatically and converges faster.

3. **Two‑stage refinement** – After Adam we run a short stochastic
   hill‑climb (pairwise mass transfer) to escape any remaining shallow
   local minima.

4. **Larger dimension** – The search is performed on ``n = 4096``
   (instead of 2048).  With more degrees of freedom the theoretical
   lower bound 2·n·max(conv) → 1 becomes attainable, giving a chance
   to beat the 1.503 target.

The programme respects the 1000 s budget, prints intermediate progress
and finally returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # Based on the algorithm by Duchi et al. (2008).
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Result length = 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σ a = 1):
        2·n·max(conv(a,a))
    The value is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (p ≥ 0, front‑loaded decreasing power‑law)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)      # n, n‑1, … , 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def rev_power_exp(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^{p}·q^{i}.
    ``p`` controls a polynomial front‑load, ``q`` (0<q≤1) adds an
    exponential decay.  When q=1 it reduces to ``reverse_power_law``.
    """
    i = np.arange(n, dtype=np.float64)          # i = 0,…,n‑1
    front = (n - i) ** p
    expo = q ** i
    a = front * expo
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α}·(1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Parametric sweep – broad search over the families
# ----------------------------------------------------------------------


def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """Explore several families, keep the best vector found."""
    best_vec = None
    best_val = float("inf")

    # ---- reverse power‑law (baseline) ---------------------------------
    for p in np.linspace(0.5, 3.0, 51):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- reverse power‑law × geometric factor (new family) ------------
    # coarse grid
    for p in np.linspace(0.5, 3.0, 31):
        for q in np.linspace(0.90, 0.999, 35):
            if time.time() >= deadline:
                return best_vec
            a = rev_power_exp(n, p, q)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # fine grid around current best (if any)
    if best_vec is not None:
        # try to tighten around the (p,q) that produced it – we simply
        # re‑sweep a narrower window.
        for p in np.linspace(0.8, 2.2, 71):
            for q in np.linspace(0.95, 0.9999, 71):
                if time.time() >= deadline:
                    return best_vec
                a = rev_power_exp(n, p, q)
                if a is None:
                    continue
                val = compute_score(a)
                if val < best_val:
                    best_val, best_vec = val, a

    # ---- geometric ----------------------------------------------------
    for q in np.concatenate((
        np.linspace(0.90, 0.99, 41),
        np.linspace(0.990, 0.9999, 60))):
        if time.time() >= deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- beta distributions (including negative exponents) -------------
    alphas = np.linspace(-0.8, 5.0, 27)
    betas = np.linspace(-0.8, 5.0, 27)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- random convex mixtures of a few primitives --------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    rev2 = reverse_power_law(n, 2.0)
    geom = geometric(n, 0.98)
    basics = [uniform, rev2, geom]

    for _ in range(2000):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient optimiser
# ----------------------------------------------------------------------


def adam_polish(a0: np.ndarray, deadline: float,
                lr: float = 0.12,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimiser on the max‑convolution sub‑gradient.
    The iterate is always projected onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # active set – indices where the max is attained (within tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max(conv): sum of gradients of active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # keep the best encountered point
        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

        # occasional jitter to shake out of plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr * 0.003 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (pairwise mass transfer)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """
    Greedy stochastic moves: pick i≠j, move a fraction of a[i] to a[j].
    Accept only improving moves; occasionally inject jitter.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.015
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.55)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 120_000:
            # shake‑up
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # 5 s safety margin

    best_seq = None
    best_val = float("inf")

    # -------------------------------------------------------------
    # 0. Optional fallback – use the previously known construction
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Parametric sweep (≈ 250 s)
    # -------------------------------------------------------------
    sweep_deadline = deadline - 500.0               # leave >500 s for polishing
    n = 4096                                        # a larger dimension
    if time.time() < sweep_deadline:
        print("[search] starting parametric sweep (n =", n, ")")
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val_norm = compute_score(a_initial)          # normalised score
            print(f"[sweep] normalised score = {val_norm:.6f}")
            # Evaluate via the official routine
            val_real = evaluate_sequence(a_initial.tolist())
            print(f"[sweep] official score = {val_real:.6f}")
            if val_real < best_val:
                best_val, best_seq = val_real, a_initial.tolist()

    # -------------------------------------------------------------
    # 2. Adam‑style polishing (bulk of remaining time)
    # -------------------------------------------------------------
    polish_deadline = deadline - 150.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[optim] starting Adam polishing")
        a_polished = adam_polish(a0, polish_deadline, lr=0.12)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[optim] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -------------------------------------------------------------
    # 3. Final stochastic hill‑climb
    # -------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[hill] starting final hill‑climb")
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # 4. Safeguard – if nothing succeeded, fall back to a good heuristic
    # -------------------------------------------------------------
    if best_seq is None:
        print("[fallback] no candidate found – using reverse power‑law p=1.8")
        a_fallback = reverse_power_law(n, 1.8)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final report
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
