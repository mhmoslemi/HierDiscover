   - best_a = a, best_score = compute_score(a)

   - For iteration i (up to max_iters or until deadline):
        if time.time() >= deadline: break
        M = full_convolution(a).max()
        # Build constraint matrix using a

        A = csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))

"""
Optimisation of a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of five stages:

1. **Global exploration** – sample a handful of parametric families
   (beta‑shapes, power‑laws, exponentials, Gaussians, sine‑powers) and a
   few Dirichlet draws.  All candidates are projected onto the simplex
   (∑a_i = 1) and the best few are kept.

2. **Linear‑programming refinement** – for a current vector `f` compute
   `M = max(conv(f,f))`.  Solve the LP

       maximise   ∑ g_i
       subject to  f * g ≤ M   (element‑wise)
                    g_i ≥ 0

   The solution `g` has a larger L1‑norm than `f`.  By scaling `g` back
   to unit sum and mixing it with `f` (convex combination) the maximum
   convolution never increases, while the distribution usually becomes
   flatter.  This step is repeated many times (≈30 – 50 iterations) with a
   decreasing mixing coefficient.

3. **Projected sub‑gradient polishing** – minimise the exact max‑convolution
   using a simple sub‑gradient descent with a diminishing step size.

4. **Stochastic hill‑climbing (simulated annealing)** – a final noisy
   optimisation that can escape tiny basins of attraction.

5. **Return the best feasible sequence** – the result is a plain list of
   floats; any scaling leaves the objective unchanged, but the implementation
   keeps the L1‑norm equal to 1 for numerical stability.

All stages respect a total wall‑clock budget of ~1000 s.  The LP phase
uses a pre‑computed sparse Toeplitz matrix for efficiency.  The code
depends only on NumPy and SciPy (already available in the judging
environment).
"""

import time

from scipy.sparse import csr_matrix

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·vec1 + (1‑λ)·vec2, projected onto the simplex."""
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Baseline global exploration
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 3000,
):
    """Explore several families and keep the best ``max_keep`` vectors."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law sweep
    for p in np.linspace(0.1, 5.0, 61):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(0.0001, 0.5, 80):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta grid (coarse)
    al_vals = np.linspace(0.02, 2.0, 25)
    be_vals = np.linspace(0.02, 2.0, 25)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # random Beta draws
    for _ in range(200):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)   # 0.01 … 2
        beta = 10 ** np.random.uniform(-2, 0.3)
        try_candidate(generate_beta(n, alpha, beta))

    # Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.45 * n, 20):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # sine‑power sweep
    for p in np.linspace(0.1, 8.0, 25):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # mixtures of two families
    for _ in range(80):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # Dirichlet draws with a range of concentration parameters
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Linear‑programming based refinement (core novelty)
# ----------------------------------------------------------------------


def precompute_conv_indices(n: int):
    """Return row/col index arrays for the Toeplitz convolution matrix."""
    # Using 32‑bit integers to save memory
    i = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + i[None, :]).ravel()
    cols = np.tile(i, n)
    return rows, cols


def lp_refine(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 40,
    t_start: float = 0.5,
    t_decay: float = 0.93,
):
    """
    Perform the classic “LP + mixing” iteration.
    Returns the best vector found (∑a = 1, a ≥ 0).
    """
    n = a0.shape[0]
    rows, cols = precompute_conv_indices(n)

    # ensure all entries are strictly positive to keep the LP bounded
    a = np.maximum(a0, 1e-12)
    a = project_to_simplex(a)

    best_a = a.copy()
    best_score = compute_score(a)

    t = t_start
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # current upper bound on convolution
        conv = full_convolution(a)
        M = conv.max()

        # Build sparse constraint matrix A (shape (2n‑1, n))
        data = np.repeat(a, n)            # each a_i appears n times
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # Linear program: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n

        # Solve
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            print(f"[lp] iteration {it}: LP failed – {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print(f"[lp] iteration {it}: got zero sum from LP")
            break

        # Scale g back to unit L1‑norm (guaranteed not to violate the bound)
        g_scaled = g / sum_g

        # Convex mix with current a
        a_new = (1.0 - t) * a + t * g_scaled
        a_new = project_to_simplex(a_new)

        # Evaluate
        score_new = compute_score(a_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            print(f"[lp] iter {it}: improved score = {best_score:.6f}")

        # Reduce mixing coefficient slowly
        t = max(t * t_decay, 0.02)

        a = a_new

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35):
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient (sum of contributions from all active shifts)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[polish] iter {t}: improved score = {best_val:.6f}")

        if t % 4000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9993,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 5e-6):
    """Metropolis hill‑climb that accepts occasional worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[anneal] iter {it}: new best = {best_val:.6f}")
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises the
    given ``evaluate_sequence`` function.  Returns a plain Python list of
    floats (the best vector found before the time budget expires).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0       # keep a safety margin

    # Problem dimension – 2048 gives a tight lower bound while staying fast
    n = 2048
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Fallback (if a previously discovered construction is available)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈120 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 550.0
    if time.time() < baseline_deadline:
        candidates = baseline_search(
            n, baseline_deadline, max_keep=10, random_draws=4000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. LP‑based refinement (≈300 s)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 300.0
    if best_seq is not None and time.time() < lp_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = lp_refine(a0, lp_deadline, max_iters=45, t_start=0.7)
        val = compute_score(refined)
        print(f"[lp‑phase] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 150.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_refined)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing (remaining time)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 5. Safeguard – uniform vector if nothing was ever found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final reporting via the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
