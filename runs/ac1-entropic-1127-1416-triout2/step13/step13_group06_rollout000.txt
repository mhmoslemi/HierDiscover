"""
Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence().  The algorithm is deliberately different from the
previous implementation:

1.  Random search on the probability simplex using Dirichlet draws.
    Dirichlet(alpha) samples uniformly for α=1 and creates moderate
    variance for other α; the resulting autocorrelation is typically
    close to its average value, giving a low max‑convolution peak.

2.  After a large number of cheap random samples the best vector is
    refined by exponentiated‑gradient (KL‑mirror) descent.  The
    gradient is assembled from *all* convolution entries whose value
    lies within 1 % of the current maximum, which yields a smoother
    descent than using a single index.

3.  A final simulated‑annealing hill‑climb (same routine as before)
    polishes the solution.

The whole pipeline stops before the 1000 s wall‑clock budget (a 5 s
margin is kept for the final official evaluate_sequence call).  The
returned list contains the raw floats (no clipping – evaluate_sequence
will do its own security checks).
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities (independent of evaluate_sequence)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast O(N log N)).
    """
    m = a.size + b.size - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)^2).
    The vector a is assumed to be non‑negative; the function does *not*
    apply the security checks performed by evaluate_sequence().
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * float(conv.max()) / (s * s)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _exp_gradient_descent(start_a: np.ndarray,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.5,
                          tol_frac: float = 0.01) -> Tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) descent.
    The gradient is summed over all convolution indices whose value
    is within `tol_frac` of the current maximum.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = _score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        # gather indices that are “almost maximal”
        near = np.where(conv >= max_val * (1.0 - tol_frac))[0]

        # build sub‑gradient (sum over near‑max indices)
        grad = np.zeros_like(a)
        for k in near:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            # each pair (i, j) contributes a_j to gradient of a_i
            grad[mask] += 2.0 * a[j[mask]]

        # average to keep scale stable
        grad /= max(1, len(near))

        # exponentiated‑gradient step
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex (KL‑projection = scaling)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional full evaluation
        if t % 2000 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] exp‑grad improved to {best_score:.6f}")

    # final check
    sc = _score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] exp‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def _hill_climb_sa(start_a: np.ndarray,
                   deadline: float,
                   max_steps: int = 6_000_000,
                   temp_factor: float = 1.2e-3,
                   step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    Very similar to the routine used in the previous run, but with a
    slightly reduced step budget to respect the overall time limit.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = _score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        # two possible moves: swap two entries or transfer a fraction of mass
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = _project_to_simplex(cand)
        cand_score = _score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    evaluate_sequence().  The strategy is:

    * Random Dirichlet draws on a moderately large simplex (n≈4096) to
      locate a region where the max‑convolution peak is close to its
      theoretical lower bound (≈mean value).
    * KL‑mirror descent with a “soft‑max” sub‑gradient (all near‑max
      indices) to fine‑tune the candidate.
    * A short simulated‑annealing hill‑climb for final polishing.

    The function respects the supplied time budget and returns the best
    sequence found as a plain list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 5‑second safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Optional fallback to a previously known construction
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Choose problem size
    # -----------------------------------------------------------
    n = 4096                     # a compromise between quality and speed

    # -----------------------------------------------------------
    # 2. Random Dirichlet search (≈ half of the time budget)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] starting random Dirichlet search (n={n})")
    # goal: use roughly 30 s for set‑up then consume up to 45 % of total time
    random_deadline = start_time + 0.45 * budget_s
    samples = 0
    while time.time() < random_deadline and samples < 200_000:
        samples += 1

        # sample a concentration α (log‑uniform in [0.3, 5])
        log_alpha = np.random.uniform(np.log(0.3), np.log(5.0))
        alpha = math.exp(log_alpha)

        # Dirichlet draw on the simplex
        a = np.random.dirichlet(np.full(n, alpha))

        # Occasionally apply a power‑law reshaping to explore different variances
        if np.random.rand() < 0.25:
            p = np.random.uniform(0.5, 2.5)
            a = np.power(a, p)
            a /= a.sum()

        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] random sample #{samples} improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. KL‑mirror (exponentiated‑gradient) refinement

        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Final simulated‑annealing hill‑climb
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = _hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=6_000_000,
            temp_factor=1.2e-3,
            step_scale_init=0.25,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.copy()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Fallback to a safe uniform vector if nothing was found

    if best_seq is None:

        print(f"[{time.time():.1f}] fallback to uniform (score={best_score:.6f})")

    # -----------------------------------------------------------

    # -----------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # return a plain Python list (required by the outer harness)
    return [float(x) for x in best_seq]
