def project_to_simplex(v):
    """Projects vector v onto the probability simplex {x>=0, sum=1}."""
    # Implementation based on sorting.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    """Projection onto simplex."""
    # Based on algorithm from Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Search for a non‑negative step‑function that (approximately) minimises
``evaluate_sequence``.  The pipeline is:

1. **Exploratory families** – power laws, exponentials, beta,
   Gaussian, Hann, Tukey, sine‑power and many random Dirichlet draws.
   The best few candidates are kept.

2. **Smooth Adam optimisation** – minimise a soft‑max approximation of
   the supremum of the auto‑convolution using an Adam optimiser on the
   soft‑max (entropic) parametrisation.  The temperature τ is annealed
   gradually to zero.  Several restarts are performed.

3. **Projected sub‑gradient descent** – after the Adam phase a plain
   sub‑gradient method (gradient of the true max) with simplex projection
   is applied to polish the solution further.

4. **Fast simulated‑annealing polish** – a short Metropolis walk to
   escape any tiny local plateaux.

The whole search respects a 1000 s wall‑clock budget.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional, not used in the final pipeline


# ----------------------------------------------------------------------
# Utility: linear convolution via real FFT (fast for long vectors)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective used for internal ranking (identical to evaluate_sequence
# after normalising the vector to sum = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (entropic mirror)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a real vector onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the global exploration phase
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_tukey(n: int, alpha: float) -> np.ndarray:
    """Simple Tukey window (alpha∈[0,1]; 0 → rectangle, 1 → Hann)."""
    if alpha <= 0.0:
        return np.full(n, 1.0 / n)
    if alpha >= 1.0:
        return generate_hann(n)
    i = np.arange(n, dtype=float) / (n - 1)
    a = np.ones_like(i)
    taper_len = alpha / 2.0
    left = i < taper_len
    right = i > (1.0 - taper_len)
    a[left] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * i[left] / alpha - 1.0)))
    a[right] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * i[right] / alpha - 2.0 / alpha + 1.0)))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * i / (n - 1)) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration (keeps up to ``max_keep`` best vectors)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 12):
    """Sweep families and many Dirichlet draws, remembering the best ones."""
    best = []  # (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec /= s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # -- power law -------------------------------------------------------
    for p in np.linspace(0.1, 5.0, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # -- exponential ------------------------------------------------------
    for lam in np.linspace(0.0001, 0.10, 400):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # -- beta -------------------------------------------------------------
    al_vals = np.linspace(0.2, 10.0, 30)
    be_vals = np.linspace(0.2, 10.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # -- Gaussian ---------------------------------------------------------
    sigma_vals = np.linspace(n / 32.0, n / 1.5, 35)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # -- Hann -------------------------------------------------------------
    try_candidate(generate_hann(n))

    # -- Tukey ------------------------------------------------------------
    for alpha in np.linspace(0.0, 1.0, 25):
        if time.time() >= deadline:
            break
        try_candidate(generate_tukey(n, alpha))

    # -- Sine‑power -------------------------------------------------------
    for p in np.linspace(0.5, 8.0, 40):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # -- Dirichlet draws --------------------------------------------------
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 800
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.18,
                    init_tau_factor: float = 12.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9998) -> np.ndarray:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam
    on the entropic (soft‑max) parametrisation.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)          # log‑softmax variables

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature (roughly the current max)
    tau = max(min_tau, init_tau_factor * full_convolution(a).max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping (a = softmax(x)) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution & smooth max weights -----
        conv = full_convolution(a)
        max_c = conv.max()

        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # soft‑max over the convolution values
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient w.r.t. a (g_i = 2·∑_k w_k·a_{k-i}) -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for the log‑softmax (x‑space) -----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        it_adj = it + 1  # bias correction factor denominator
        m_hat = m / (1.0 - beta1 ** it_adj)
        v_hat = v / (1.0 - beta2 ** it_adj)
        lr = init_lr / (np.sqrt(it_adj) + 1.0)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal τ -----
        tau = max(min_tau, tau * tau_decay)

        # ----- occasional exact evaluation -----
        if it % 400 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, deadline: float,
                        init_eta: float = 0.15,
                        decay: float = 0.99995,
                        top_k: int = 3) -> np.ndarray:
    """
    Simple sub‑gradient descent on `max(conv(a,a))` with simplex projection.
    ``top_k`` gradients from the K largest convolution entries are averaged.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_c = conv.max()

        # indices of the K largest values (avoid full sort)
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]

        # accumulate gradient contributions
        grad = np.zeros_like(a)
        for k in idxs:
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            i_range = np.arange(i_min, i_max + 1)
            j_range = k - i_range
            grad[i_range] += 2.0 * a[j_range]
        grad /= float(top_k)

        # step size schedule
        eta = init_eta * (decay ** it)

        a_new = a - eta * grad
        a_new = project_to_simplex(a_new)

        cur_val = compute_score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            if it % 2000 == 0:
                print(f"[subgrad] iter {it:7d}  best {best_val:.6f}")

        a = a_new

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9993,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb with occasional jitter."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-14)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            # occasional jitter to shake out plateaux
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the given wall‑clock budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 is a good balance between expressivity and speed.
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 150 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 650.0  # reserve the rest for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smoothed Adam optimisation with several restarts (≈ 600 s)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 120.0
    if time.time() < smooth_deadline:
        # start from the current best, otherwise from uniform
        if best_seq is not None:
            a0 = np.array(best_seq, dtype=float)
        else:
            a0 = np.full(n, 1.0 / n)
        refined = smooth_optimize(a0, smooth_deadline)
        val = compute_score(refined)
        if val < best_val:
            best_val, best_seq = val, refined.tolist()
            print(f"[smooth‑primary] refined score = {val:.6f}")

        # a handful of random restarts
        while time.time() < smooth_deadline:
            remaining = smooth_deadline - time.time()
            sub_deadline = time.time() + min(35.0, remaining)
            conc = np.random.choice([0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Projected sub‑gradient polishing (≈ 200 s)
    # -----------------------------------------------------------------
    subgrad_deadline = deadline - 50.0
    if best_seq is not None and time.time() < subgrad_deadline:
        a_start = np.array(best_seq, dtype=float)
        refined = subgradient_descent(a_start, subgrad_deadline)
        val = compute_score(refined)
        if val < best_val:
            best_val, best_seq = val, refined.tolist()
            print(f"[subgrad‑polish] refined score = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        if val < best_val:
            best_val, best_seq = val, final.tolist()
            print(f"[anneal] final score = {val:.6f}")

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Official evaluation report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
