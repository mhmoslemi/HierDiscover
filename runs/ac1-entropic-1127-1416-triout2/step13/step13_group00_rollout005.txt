"""
Propose a step-height sequence that minimises
    2 * n * max(conv(a, a))   (with a_i ≥ 0, Σ a_i = 1)

The algorithm consists of four stages:

1. **Broad candidate generation** – a mixture of parametric families
   (power‑law, exponential, beta, Gaussian, sine‑power, U‑shape,
   triangle) and many random Dirichlet draws.  The best few are kept.

2. **Projected Adam optimisation** – smooth‑max surrogate
   (τ·log‑sum‑exp) is minimised with Adam.  τ is exponentially
   decayed to a very small value; after each step the vector is
   projected onto the probability simplex.

3. **Sub‑gradient polishing** – true objective (max of the exact
   convolution) is reduced by a diminishing‑step sub‑gradient method.

4. **Simulated‑annealing hill‑climb** – a Metropolis walk that mixes
   transfer moves and occasional swaps, with a slow cooling schedule.

All phases respect the 1000 s wall‑clock budget.  The best sequence
found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    Returns g_i = 2·Σ_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    m = len(w)
    rev_w = w[::-1]
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def smooth_max(conv: np.ndarray, tau: float) -> (float, np.ndarray):
    """Smooth approximation S = τ·log∑exp(conv/τ) and soft‑max weights."""
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_ushape(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n + 1.0 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_triangle(n: int) -> np.ndarray:
    """Symmetric triangle: a_i ∝ min(i+1, n-i)."""
    i = np.arange(n, dtype=float) + 1.0
    a = np.minimum(i, n + 1.0 - i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_random_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.random.rand(n) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Exploration – keep a few promising vectors
# ----------------------------------------------------------------------


def explore_candidates(n: int, deadline: float, keep: int = 24) -> list[np.ndarray]:
    """Generate many vectors, keep the ``keep`` best according to compute_score."""
    best = []   # list of (score, vector)

    def consider(v: np.ndarray):
        if not np.isfinite(v).all():
            return
        v = np.maximum(v, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        if len(best) < keep:
            best.append((sc, v.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, v.copy())
            best.sort(key=lambda x: x[0])

    # 1) Uniform baseline
    consider(np.full(n, 1.0 / n))

    # 2) Power‑law sweeps
    for p in np.linspace(0.0, 4.0, 41):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))
        consider(generate_powerlaw(n, -p))

    # 3) Exponential sweeps
    for lam in np.linspace(-0.4, 0.4, 81):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # 4) Beta family (covers U‑shape & peaked shapes)
    al_vals = np.linspace(0.1, 5.0, 21)
    be_vals = np.linspace(0.1, 5.0, 21)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    # 5) Gaussian widths
    for sigma in np.linspace(0.03 * n, 0.4 * n, 12):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # 6) Sine‑power shapes
    for p in np.linspace(0.1, 8.0, 24):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # 7) U‑shape (edge‑heavy)
    for p in np.linspace(0.1, 5.0, 24):
        if time.time() >= deadline:
            break
        consider(generate_ushape(n, p))

    # 8) Triangle shape
    consider(generate_triangle(n))

    # 9) Random power‑law transforms
    for p in np.linspace(-2.0, 2.0, 16):
        if time.time() >= deadline:
            break
        consider(generate_random_powerlaw(n, p))

    # 10) Random Dirichlet draws
    concentrations = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per = max(1, (keep * 300) // len(concentrations))
    for conc in concentrations:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            consider(np.random.dirichlet([conc] * n))

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 0.4,
                  tau_end: float = 1e-8,
                  tau_decay: float = 0.92,
                  lr: float = 0.04,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Projected Adam minimising τ·log‑sum‑exp(conv) with exponential τ‑schedule."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        # decay temperature of the smooth‑max
        if t % 150 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)          # soft‑max weights (∑ w = 1)
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape shallow plateaux
        if t % 800 == 0:
            jitter = np.random.randn(n) * (lr * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (true max‑conv objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.25) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # collect all indices within relative tolerance of the max
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # accumulate sub‑gradient from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing with transfer and swap moves
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-8,
                        cooling_factor: float = 0.9998,
                        step_scale_init: float = 0.09,
                        step_scale_min: float = 5e-7,
                        swap_prob: float = 0.12) -> np.ndarray:
    """Metropolis walk that mixes transfer and swap moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        if np.random.rand() < swap_prob:
            # simple swap of two coordinates
            i, j = np.random.choice(n, 2, replace=False)
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i = np.random.randint(n)
            if a[i] <= 0.0:
                continue
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = step_scale * a[i] * np.random.rand()
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta

        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
            # modest increase of step_scale on accepted moves
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            # shrink step_scale on rejections
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            # occasional jitter to shake out of plateaux
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # Use the length of the known fallback if it exists, else 2048
    try:
        n_fallback = len(height_sequence_1)
        n = max(2048, n_fallback)
    except Exception:
        n = 2048

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Quick deterministic fallback (triangular / U‑shape)
    # -----------------------------------------------------------------
    try:
        fallback_vec = generate_ushape(n, 1.5)
        fallback = fallback_vec.tolist()
        val = evaluate_sequence(fallback)
        print(f"[fallback] U‑shape(p=1.5) score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad exploration (≈ 130 s)
    # -----------------------------------------------------------------
    explore_deadline = deadline - 650.0
    if time.time() < explore_deadline:
        print("[explore] generating candidates …")
        cand_vecs = explore_candidates(n, explore_deadline, keep=24)
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_score:
                best_score, best_seq = val, vec.tolist()
        print(f"[explore] best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam optimisation from several promising starts (≈ 450 s)
    # -----------------------------------------------------------------
    if best_seq is not None:
        grad_deadline = min(deadline - 260.0,
                            start_time + 150.0 + (deadline - start_time) * 0.45)
        start_points = []

        # best so far
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        start_points.append(best_vec)

        # uniform
        start_points.append(np.full(n, 1.0 / n))

        # a few random Gaussian‑based starts
        for _ in range(4):
            start_points.append(project_to_simplex(np.random.rand(n)))

        # include top three candidates from the exploration
        for vec in cand_vecs[:3]:
            start_points.append(vec)

        refined_best = best_vec.copy()
        refined_val = best_score

        for idx, a0 in enumerate(start_points):
            if time.time() >= grad_deadline:
                break
            local_deadline = grad_deadline - (time.time() - start_time) - 0.5
            if local_deadline <= time.time():
                break
            print(f"[adam {idx}] start")
            a_opt = adam_optimize(a0, local_deadline)
            val = compute_score(a_opt)
            print(f"[adam {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_opt.copy()

        best_score, best_seq = refined_val, refined_best.tolist()
        print(f"[adam] best after Adam = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 170 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[polish] start")
        a_polished = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing (≈ 80 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[anneal] start")
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final evaluation with the official function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
