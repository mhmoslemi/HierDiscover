assistantfinal**Solution Overview**

The goal is to find a non‑negative sequence `a` (a step‑function) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max(\,a*a\,)}{(\sum a)^2},
\]

where `*` denotes the aperiodic convolution.  
The score is homogeneous – scaling `a` does not change the value – therefore we only have to optimise the *shape* of the vector.

We combine three ideas:

1. **Analytic families** – a dense scan of several parameterised families
   (U‑shaped product, asymmetric Beta, pure exponential decay, left‑heavy power).
   The scan is performed on a relatively large dimension `n = 16384` and
   keeps the best candidate.

2. **Local refinement** – once the most promising family and its parameters are known
   we perform a one‑dimensional golden‑section search (or simple hill‑climb)
   to obtain a highly‑accurate parameter value.

3. **Gradient‑based polishing** – starting from the refined candidate we run an
   Adam optimiser on the smooth‑max surrogate of the max‑convolution.
   The temperature (`β`) is increased from a modest value to a very large one,
   so the optimiser first sees a smooth landscape and finally converges to the
   true max‑objective.

4. **Final stochastic hill‑climb** – a cheap random‑mass‑transfer move
   (move a fraction of probability mass from a large entry to a random
   small entry) is repeated while a time budget remains.  This often
   squeezes out a few extra points.

All steps respect the global 1000 s wall‑clock limit.  The code is self‑contained,
uses only `numpy` and the provided `evaluate_sequence`, and prints intermediate
scores so that progress can be monitored.

```python
"""
Search for a non‑negative step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of:
1) a dense deterministic scan of several analytic families (product‑U,
   asymmetric Beta, exponential decay, left‑heavy power);
2) one‑dimensional refinement of the best parameters (golden‑section);
3) Adam optimisation of a smooth‑max surrogate of the max‑convolution;
4) a final greedy stochastic hill‑climb that moves a small amount of mass
   between entries whenever this reduces the true objective.

All stages stop when the remaining wall‑clock budget is exhausted.
The best sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)**2).
    a is assumed non‑negative.
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families (all return a normalised vector of length n)
# ----------------------------------------------------------------------


def product_u_family(n: int, alpha: float, logprod: np.ndarray) -> np.ndarray:
    """
    a_i ∝ [(i+½)(n‑½‑i)]^{‑α}   (U‑shaped, symmetric).
    logprod must be pre‑computed: log(i+½)+log(n‑½‑i).
    """
    w = np.exp(-alpha * logprod)
    # numeric guard – replace infinities by a huge finite number
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float, logi: np.ndarray) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    w = np.exp(-beta * logi)
    return normalize(w)


def exp_decay_family(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    return normalize(w)


def beta_family(n: int, p: float, q: float, logi: np.ndarray, logrev: np.ndarray) -> np.ndarray:
    """
    Discrete Beta density:
        a_i ∝ (i+½)^{p‑1} · (n‑½‑i)^{q‑1}
    p,q>0 .
    """
    logw = (p - 1.0) * logi + (q - 1.0) * logrev
    # subtract max to avoid overflow, then exponentiate
    logw -= np.max(logw)
    w = np.exp(logw)
    return normalize(w)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 800_000,
                         beta_start: float = 5.0,
                         beta_end: float = 30_000.0,
                         lr: float = 0.012,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    The temperature β grows geometrically from β_start to β_end.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)               # log‑softmax parametrisation

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a /= best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a /= a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 5000 == 0:
            a_tmp = np.exp(z)
            a_tmp /= a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z)
    a_fin /= a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def greedy_hill_climb(start_a: np.ndarray,
                      deadline: float,
                      max_steps: int = 5_000_000,
                      init_scale: float = 0.15) -> Tuple[np.ndarray, float]:
    """
    Simple hill‑climb that transfers a small fraction of mass from the current
    largest entry to a random other entry whenever this reduces the objective.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score
    scale = init_scale

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        i = np.argmax(cur_a)                # donor (largest)
        j = np.random.randint(n)            # receiver
        if i == j:
            continue

        delta = scale * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            scale = min(scale * 1.03, 0.5)   # expand step size on success
        else:
            scale = max(scale * 0.97, 1e-4)  # contract otherwise

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] greedy step {step}, best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# One‑dimensional refinement helpers
# ----------------------------------------------------------------------


def golden_section(func, lo: float, hi: float, max_iter: int = 20) -> Tuple[float, float]:
    """Golden‑section search for the minimum of a unimodal function on [lo, hi]."""
    phi = (1.0 + 5.0 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under a fixed wall‑clock budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin for the final call

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 16384                                      # a compromise between resolution and speed

    # Pre‑compute logs that are reused many times
    i = np.arange(n, dtype=np.float64)
    logi = np.log(i + 0.5)
    logrev = np.log(n - 0.5 - i)
    logprod = logi + logrev                         # for the product‑U family

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Coarse deterministic scan of analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] START coarse scan (n={n})")

    # ----- product‑U family ------------------------------------------------
    alphas = np.linspace(0.0, 12.0, 73)            # 0 … 12 in steps of 0.166…
    best_alpha = None
    for alpha in alphas:
        if time.time() >= deadline - 30.0:
            break
        vec = product_u_family(n, alpha, logprod)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best (product α={alpha:.4f}) → {best_score:.6f}")

    # ----- left‑heavy power family ----------------------------------------
    betas = np.linspace(0.0, 12.0, 73)
    for beta in betas:
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta, logi)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] new best (power β={beta:.4f}) → {best_score:.6f}")

    # ----- exponential decay family -----------------------------------------
    # log‑space grid for γ (more resolution for tiny decay rates)
    gammas = np.logspace(-5, -0.1, 40)  # γ ∈ [1e‑5, 0.79]
    for gamma in gammas:
        if time.time() >= deadline - 30.0:
            break
        vec = exp_decay_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] new best (exp‑decay γ={gamma:.6f}) → {best_score:.6f}")

    # ----- Beta family (asymmetric) ----------------------------------------
    # p and q are explored on a log‑scale, allowing values < 0.2
    p_vals = np.logspace(np.log10(0.05), np.log10(4.0), 20)
    q_vals = np.logspace(np.log10(0.05), np.log10(4.0), 20)
    best_pq = None
    for p in p_vals:
        for q in q_vals:
            if time.time() >= deadline - 30.0:
                break
            vec = beta_family(n, p, q, logi, logrev)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec
                best_pq = (p, q)
                print(f"[{time.time():.1f}] new best (Beta p={p:.4f}, q={q:.4f}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Fine 1‑D refinement of the best analytic family
    # ------------------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        # Refine the product‑U parameter α by golden‑section
        lo = max(0.0, best_alpha - 1.0)
        hi = best_alpha + 1.0

        def f_alpha(a):
            vec = product_u_family(n, a, logprod)
            return compute_score(vec)

        refined_alpha, sc = golden_section(f_alpha, lo, hi, max_iter=25)
        vec = product_u_family(n, refined_alpha, logprod)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha = refined_alpha
            print(f"[{time.time():.1f}] refined product α={refined_alpha:.6f} → {best_score:.6f}")

    if best_pq is not None and time.time() < deadline - 20.0:
        # Refine the Beta parameters (p,q) jointly by simple hill‑climb
        p, q = best_pq
        step = 0.2
        while time.time() < deadline - 10.0 and step > 1e-3:
            improved = False
            for dp in (-step, step):
                for dq in (-step, step):
                    p_cand = max(0.01, p + dp)
                    q_cand = max(0.01, q + dq)
                    vec = beta_family(n, p_cand, q_cand, logi, logrev)
                    sc = compute_score(vec)
                    if sc < best_score:
                        best_score = sc
                        best_seq = vec
                        p, q = p_cand, q_cand
                        improved = True
                        print(f"[{time.time():.1f}] refined Beta p={p:.6f}, q={q:.6f} → {best_score:.6f}")
                        break
                if improved:
                    break
            if not improved:
                step *= 0.5

    # ------------------------------------------------------------------
    # 3) Adam smooth‑max optimisation (no symmetry constraints)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] start Adam optimisation")
        a_start = best_seq.astype(np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=a_start,
            deadline=deadline - 150.0,
            max_iters=1_200_000,
            beta_start=5.0,
            beta_end=40_000.0,
            lr=0.010,
            sym_interval=0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final stochastic greedy hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final greedy hill‑climb")
        a_final, sc_final = greedy_hill_climb(
            start_a=best_seq.astype(np.float64),
            deadline=deadline,
            max_steps=8_000_000,
            init_scale=0.12,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final
            print(f"[{time.time():.1f}] greedy improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Fallback safety net
    # ------------------------------------------------------------------
    if best_seq is None:
        # Uniform vector is always a valid fallback
        n_small = 256
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)

    # Final official evaluation (contains the security checks)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq.tolist()
```
