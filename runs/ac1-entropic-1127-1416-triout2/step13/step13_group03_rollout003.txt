    for lam in [0.005, 0.01, 0.02]:

        start_vectors.append(project_to_simplex(a))

    for p in [0.5, 1.0, 2.0]:
        a = np.arange(1,n+1)**(-p)
        start_vectors.append(project_to_simplex(a))

    for _ in range(5):
        a = np.random.dirichlet(np.full(n, 0.5))
        b = np.random.dirichlet(np.full(n, 0.5))
        lam = np.random.rand()
        a_mix = lam*a + (1-lam)*b
        start_vectors.append(project_to_simplex(a_mix))

    # allocate time: compute per-start deadline
    remaining = deadline - time.time()

    for idx, start_vec in enumerate(start_vectors):
        now = time.time()
        if now >= deadline:

        if score < best_score:

            best_seq = a_opt.copy()

"""
Non‑negative step‑function optimiser.

The objective (after normalising the vector to sum = 1) is
    score = 2·n·max(conv(a,a)),
where conv is the full linear convolution.
The theoretical lower bound is 2·n/(2·n‑1) ≈ 1.0, so the task is to make the
self‑convolution as flat as possible.

Approach
--------
1. **Many random starts** – Dirichlet draws, simple exponentials,
   power‑laws and a few mixtures give a diverse pool of initial points.
2. **Smooth‑max Adam** – minimise a smooth approximation of the
   max‑operator:
       fτ(a) = τ·log Σ exp(conv/τ) .
   A small initial temperature (τ≈5·10⁻⁴) makes the soft‑max focus on the
   current peak of the convolution, so the gradient actually changes the
   vector (the earlier version used a huge τ and got stuck).
   Adam with bias‑correction, geometric τ‑annealing and a decaying learning
   rate performs fast descent.
3. **Projected sub‑gradient polishing** – once the Adam phase is finished
   we switch to the exact sub‑gradient of the max‑operator and run many
   iterations with a modest step‑size, again projecting onto the simplex.
4. **Final evaluation** – the best vector found is returned as a plain
   list of floats; `evaluate_sequence` will clip/normalise it automatically.

The whole search respects the 1000 s wall‑clock budget and prints a short
progress log.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (simple, no external dependencies)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam)
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 5e-4,
    tau_end: float = 1e-7,
    tau_decay: float = 0.96,
    tau_decay_steps: int = 100,
    lr_start: float = 0.15,
    lr_decay: float = 0.98,
    lr_decay_steps: int = 200,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate
        fτ(a) = τ·log Σ_k exp(conv_k(a)/τ)

    Returns the best vector (projected on the simplex) seen before the
    deadline.
    """
    a = project_to_simplex(a0.astype(float))
    n = a.shape[0]
    m = 2 * n - 1

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # ---- temperature schedule ----
        if t % tau_decay_steps == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # ---- compute convolution and soft‑max weights ----
        conv = full_convolution(a)                # length m
        shift = conv.max()
        # stable soft‑max
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()             # shape (m,)

        # ---- gradient of the smooth‑max w.r.t a ----
        # gradient = 2 * (a ⋆ w)   (cross‑correlation)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)               # pick the needed slice
        grad = 2.0 * conv_full[idx]

        # ---- Adam update ----
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # ---- bookkeeping ----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # ---- learning‑rate decay ----
        if t % lr_decay_steps == 0:
            lr = max(lr * lr_decay, 1e-4)

        # occasional jitter to escape tiny plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.20) -> np.ndarray:
    """
    Projected sub‑gradient descent on   max(conv(a,a)).
    Uses a decaying step size  step0/√t .
    """
    a = project_to_simplex(a0.astype(float))
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient = sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns the sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so that the harness can still call ``evaluate_sequence``
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 gives a good trade‑off between speed and bound
    # -----------------------------------------------------------------
    n = 2048
    print(f"[init] n = {n}, total budget = {budget_s}s")

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (if a historic construction is available)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Generate a diverse pool of start vectors
    # -----------------------------------------------------------------
    start_vectors = []

    # uniform
    start_vectors.append(np.full(n, 1.0 / n))

    # Dirichlet draws – concentration 0.5 … 2.0 gives a wide variety
    for _ in range(25):
        conc = 10 ** np.random.uniform(-0.3, 0.3)   # roughly 0.5 … 2
        a = np.random.dirichlet(np.full(n, conc))
        start_vectors.append(project_to_simplex(a))

    # exponential families
    for lam in (0.002, 0.005, 0.01):
        a = np.exp(-lam * np.arange(n, dtype=float))
        start_vectors.append(project_to_simplex(a))

    # power‑law families
    for p in (0.5, 1.0, 2.0):
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        start_vectors.append(project_to_simplex(a))

    # random mixtures of two Dirichlet draws
    for _ in range(8):
        a = np.random.dirichlet(np.full(n, 0.4))
        b = np.random.dirichlet(np.full(n, 0.4))
        lam = np.random.rand()
        mix = lam * a + (1.0 - lam) * b
        start_vectors.append(project_to_simplex(mix))

    # -----------------------------------------------------------------
    # 2. Adam‑based smooth‑max optimisation from each start vector
    # -----------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        remaining_time = 0
    per_start_budget = max(5.0, remaining_time / max(len(start_vectors), 1))

    for idx, vec in enumerate(start_vectors):
        now = time.time()
        if now >= deadline:
            break
        local_deadline = min(deadline, now + per_start_budget)
        print(f"[adam start {idx}] deadline in {local_deadline - now:.1f}s")
        refined = smooth_max_adam(
            vec,
            local_deadline,
            tau_start=5e-4,
            tau_end=1e-7,
            tau_decay=0.96,
            tau_decay_steps=100,
            lr_start=0.15,
            lr_decay=0.98,
            lr_decay_steps=200,
        )
        val = compute_score(refined)
        print(f"[adam {idx}] score = {val:.6f}")
        if val < best_score:
            best_score = val
            best_seq = refined.copy()
        # keep a little time buffer between runs
        if time.time() >= deadline:
            break

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (exact max)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 5.0
    if best_seq is not None and time.time() < polish_deadline:
        print("[polish] starting projected sub‑gradient polishing")
        polished = subgradient_polish(best_seq, polish_deadline, step0=0.20)
        val = compute_score(polished)
        print(f"[polish] polished score = {val:.6f}")
        if val < best_score:
            best_score = val
            best_seq = polished.copy()

    # -----------------------------------------------------------------
    # 4. Safeguard – if nothing was found, fall back to uniform
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_score(best_seq)

    # -----------------------------------------------------------------
    # 5. Final official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq.tolist()
