def fast_hill_climb(a, conv, time_deadline, max_iters=None):

    M = conv.max()

    step_scale = 0.1

    while time.time() < time_deadline and (max_iters is None or it < max_iters):

"""
Improved optimiser for the self‑convolution upper bound.

The optimiser works with three stages:

1.  A broader parametric sweep (power‑law, geometric, mixed
    plateau‑power, beta, logistic, Dirichlet) – the best vector is kept.

2.  A long deterministic hill‑climb that updates the full convolution
    by a *rank‑1* update when a small amount of mass δ is moved from an
    index i to j.  This update is O(n) and avoids a full FFT each move,
    allowing millions of cheap proposals inside the 1000 s budget.

3.  The final candidate is reported using the official ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
#  Parametric families (all return a normalised vector Σa = 1)
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray | None:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def geometric(n: int, q: float) -> np.ndarray | None:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray | None:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def logistic_shape(n: int, centre_frac: float, scale_frac: float) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64)
    centre = centre_frac * n
    scale = max(scale_frac * n, 1e-9)
    a = 1.0 / (1.0 + np.exp((i - centre) / scale))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def random_dirichlet(n: int, concentration: float) -> np.ndarray | None:
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    s = y.sum()
    if s == 0.0:
        return None
    return y / s

#  Broad parametric sweep (larger ranges + finer grids)
# ----------------------------------------------------------------------
def parametric_sweep_extended(n: int, deadline: float) -> np.ndarray | None:
    """Explore many families with extended ranges.  Returns the best vector."""
    best_vec = None
    best_val = float('inf')
    # ----- power‑law -------------------------------------------------
    for p in np.linspace(0.05, 5.0, 251):   # finer, wider range
        if time.time() >= deadline:
            return best_vec
        a = power_law(n, p)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a
    # ----- geometric -------------------------------------------------
    for q in np.concatenate((
            np.linspace(0.90, 0.999, 151),
            np.linspace(0.999, 0.99999, 151))):
        if time.time() >= deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a
    # ----- mixed plateau + power ------------------------------------
    for L_frac in np.linspace(0.01, 0.45, 23):
        for p in np.linspace(0.10, 2.5, 46):
            if time.time() >= deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a
    # ----- beta shape ------------------------------------------------
    alphas = np.linspace(-1.5, 4.0, 28)
    betas = np.linspace(-1.5, 4.0, 28)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a
    # ----- logistic --------------------------------------------------
    for centre_frac in np.linspace(0.1, 0.9, 17):
        for scale_frac in np.linspace(0.005, 0.15, 26):
            if time.time() >= deadline:
                return best_vec
            a = logistic_shape(n, centre_frac, scale_frac)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a
    # ----- random Dirichlet samples ----------------------------------
    for _ in range(2000):
        if time.time() >= deadline:
            break
        conc = 10 ** np.random.uniform(-2, 1)   # 0.01 – 10
        a = random_dirichlet(n, conc)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a
    return best_vec


# ----------------------------------------------------------------------
#  Fast hill‑climb using O(n) rank‑1 update of the convolution
# ----------------------------------------------------------------------
def fast_hill_climb(a: np.ndarray, time_deadline: float,
                    max_iters: int | None = None) -> np.ndarray:
    """
    Deterministic + simulated‑annealing hill‑climb.
    Moves a small fraction of mass from one coordinate to another.
    The convolution vector is updated with a cheap O(n) formula.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    M = conv.max()
    best_a = a.copy()
    best_M = M

    step_scale = 0.12               # typical fraction of source mass
    temperature = best_M * 0.02      # start temperature (relative)
    temp_decay = 0.99995
    step_decay = 0.99995
    it = 0
    no_improve = 0
    recompute_interval = 200_000    # occasional exact recomputation

    while time.time() < time_deadline and (max_iters is None or it < max_iters):
        it += 1

        # source index – weighted by current mass (focus on large entries)
        i = np.random.choice(n, p=a)

        # destination – uniform among the remaining indices
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # propose a move
        delta = step_scale * a[i] * np.random.rand()
        if delta <= 0.0 or delta > a[i]:
            continue

        # ----- fast convolution update (copy‑on‑write) -----
        conv_cand = conv.copy()
        conv_cand[i:i + n] -= 2.0 * delta * a
        conv_cand[j:j + n] += 2.0 * delta * a
        conv_cand[2 * i] += delta * delta
        conv_cand[2 * j] += delta * delta
        conv_cand[i + j] -= 2.0 * delta * delta
        M_cand = conv_cand.max()

        # acceptance rule (deterministic improvement + annealed worsening)
        accept = False
        if M_cand < M:
            accept = True
        else:
            prob = math.exp(-(M_cand - M) / max(temperature, 1e-15))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # apply the move permanently
            a[i] -= delta
            a[j] += delta
            conv = conv_cand
            M = M_cand

            # keep the best ever seen
            if M < best_M:
                best_M = M
                best_a = a.copy()
                # become a bit bolder after a good step
                step_scale = min(step_scale * 1.02, 0.6)
                no_improve = 0
            else:
                # slight back‑off after a non‑recording acceptance
                step_scale = max(step_scale * 0.998, 1e-8)

            # temperature decay
            temperature = max(temperature * temp_decay, 1e-12)
        else:
            # rejected move – shrink step and cool down a little
            step_scale = max(step_scale * step_decay, 1e-8)
            temperature = max(temperature * temp_decay, 1e-12)
            no_improve += 1

        # occasional full recomputation to control numerical drift
        if (it % recompute_interval) == 0:
            conv = full_convolution(a)
            M = conv.max()
            # renormalise to combat tiny drift
            a_sum = a.sum()
            if a_sum != 0.0:
                a /= a_sum

        # if we have been stuck for a long period, inject a small jitter
        if no_improve > 500_000:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = full_convolution(a)
            M = conv.max()
            step_scale = 0.12
            temperature = best_M * 0.02
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
#  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback (previously discovered construction)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Parametric sweep – generous but bounded time slice
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 500.0          # reserve the bulk of the budget for hill‑climb
    if time.time() < sweep_deadline:
        n = 4096
        # if we already have a fallback, keep its length
        if best_seq is not None:
            n = len(best_seq)
        a0 = parametric_sweep_extended(n, sweep_deadline)
        if a0 is not None:
            val = compute_score(a0)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a0.tolist()

    # -----------------------------------------------------------------
    # 2. Fast hill‑climb on the current best vector
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = np.array(best_seq, dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a /= s

        a_opt = fast_hill_climb(a, deadline)
        val = compute_score(a_opt)
        print(f"[fast hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()

    # -----------------------------------------------------------------
    # 3. Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 4096
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
