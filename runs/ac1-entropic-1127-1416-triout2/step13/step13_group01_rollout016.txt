def subgradient_optimize(a0, deadline: float, eta0: float = 0.5) -> np.ndarray:

   a = a0.copy()
   best_a = a.copy()
   best_val = compute_score(a)
   it=0
   while time.time() < deadline:
        it+=1
        # compute convolution
        conv = full_convolution(a)
        k = int(conv.argmax())
        # compute subgradient
        indices = k - np.arange(len(a))
        mask = (indices >= 0) & (indices < len(a))
        g = np.zeros_like(a)
        g[mask] = 2 * a[indices[mask]]
        # step size schedule: eta = eta0 / sqrt(it)
        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        # project onto simplex
        a = project_onto_simplex(a)
        # periodically evaluate
        if it % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[subgrad] iter {it} score {val:.6f}")
   return best_a

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Expanded global search** – sweep the classic families
   (power‑law, exponential, beta, Gaussian) **and a new “distance‑power”
   family** a_i ∝ |i‑(n‑1)/2|^p.  For each family a fine grid of the
   shape parameter(s) is scanned, plus many random Dirichlet draws.
   The best ``max_keep`` candidates (by the exact objective
   ``2·n·max(conv(a,a))``) are retained.

2. **Two refinement stages**
   * *Smooth max‑Adam* – minimise a soft‑max approximation of the
     max‑convolution using Adam on the log‑softmax parametrisation.
     Several restarts with different hyper‑parameters are performed.
   * *Targeted simulated annealing* – a Metropolis walk that always
     moves mass away from the indices that actively contribute to the
     currently maximal convolution entry.  This helps to flatten the
     convolved profile further.

3. **Final reporting** – the best vector found is normalised,
   clipped by the security checks in ``evaluate_sequence`` and
   returned as a plain Python list.

The whole search respects the 1000 s wall‑clock budget; the
implementation keeps a small safety margin.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = rfft(a, n=n_fft)
    B = rfft(b, n=n_fft)
    C = A * B
    conv = irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a (used for the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (no clipping – all entries already non‑negative)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by sub‑gradient descent
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should not happen)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_distance_power(n: int, p: float) -> np.ndarray:
    """U‑shaped family: a_i ∝ |i‑c|^p,  c = (n‑1)/2."""
    c = (n - 1) / 2.0
    d = np.abs(np.arange(n, dtype=float) - c) ** p
    s = d.sum()
    return d / s if s > 0 else np.full(n, 0.0)

# ----------------------------------------------------------------------

    best = []  # list of (score, vector)

        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0:
            return
        vec = vec / vec.sum()
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- 1) power‑law -------------------------------------------------
    for p in np.linspace(0.05, 3.0, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # ---- 2) exponential ------------------------------------------------
    for lam in np.linspace(5e-5, 0.10, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # ---- 3) beta -------------------------------------------------------
    al_vals = np.concatenate(
        [np.geomspace(0.02, 0.5, 12), np.geomspace(0.5, 5.0, 14)]
    )
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # ---- 4) Gaussian ----------------------------------------------------
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 35)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # ---- 5) Distance‑power (new family) -------------------------------
    for p in np.linspace(0.1, 4.0, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_distance_power(n, p))

    # ---- 6) Dirichlet draws --------------------------------------------
    concs = [0.05, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 800
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (soft‑max temperature τ annealed)
# ----------------------------------------------------------------------


def smooth_optimize(
    a0: np.ndarray,
    deadline: float,
    init_lr: float = 0.08,
    init_tau_factor: float = 25.0,
    min_tau: float = 1e-10,
    tau_decay: float = 0.9997,
) -> np.ndarray:
    """
    Adam optimiser on the log‑softmax parametrisation of ``a``.
    The soft‑max temperature τ starts relatively large (≈ init_tau_factor·max conv)
    and is geometrically annealed to ``min_tau``.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise τ from the current convolution
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max -----
        mx = np.max(x)
        exp_x = np.exp(x - mx)
        a = exp_x / exp_x.sum()

        # ----- convolution & smooth max weights -----
        conv = full_convolution(a)
        mx_conv = conv.max()

        if tau < 1e-12:  # essentially exact max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - mx_conv) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient w.r.t a -----
        # g = 2 * (a_rev * w)  (convolution of reversed a with w)
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)  # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start : start + n]

        # ----- chain rule back to x (log‑softmax) -----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature anneal -----
        tau = max(min_tau, tau * tau_decay)

        # ----- occasional exact evaluation -----
        if it % 500 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:8d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Targeted simulated annealing (focus on the active max‑convolution entry)
# ----------------------------------------------------------------------


def targeted_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 5e-4,
    final_temp: float = 1e-7,
    cooling: float = 0.9995,
    step_scale_init: float = 0.05,
) -> np.ndarray:
    """
    Metropolis walk that preferentially steals mass from indices that
    actively contribute to the current maximal convolution entry.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1

        # locate the active shift and the most “guilty” index
        conv = full_convolution(a)
        k_max = int(conv.argmax())

        # contributions to the active shift
        idx = np.arange(n, dtype=int)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]
        # pick an index with large contribution (avoid zero)
        if np.any(contrib > 0):
            i = np.argmax(contrib)
        else:
            i = np.random.randint(n)

        # pick a distinct target index (prefer one with small contribution)
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        delta = step_scale * a[i]
        if delta == 0.0:
            # nothing to move – pick a fresh random move
            cand = a.copy()
        else:
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, 1e-6)

        T = max(final_temp, T * cooling)

        if it % 4000 == 0:
            # occasional jitter to escape plateaux
            jitter = np.random.randn(n) * (step_scale * 0.02)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 offers a good balance of expressive power
    # and speed.
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (previous best known solution)
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 180 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 320.0  # leave ~320 s for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smooth‑max Adam optimisation with several restarts
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)

        # primary run with slightly more aggressive temperature schedule
        refined = smooth_optimize(
            a0,
            polish_deadline,
            init_lr=0.06,
            init_tau_factor=30.0,
            min_tau=1e-11,
            tau_decay=0.9998,
        )
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # a few short random restarts
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(
                a_start,
                sub_deadline,
                init_lr=0.07,
                init_tau_factor=20.0,
                min_tau=1e-10,
                tau_decay=0.9996,
            )
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Targeted simulated annealing polish (≈ 100 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = targeted_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
