"""
search_step_function_v3.py

A new optimiser for the ``evaluate_sequence`` objective.

Key ideas
---------
*   Introduce a *shifted* reverse‑power‑law family  
        a_i ∝ (n‑i + c)^p .  
    The offset ``c`` reduces the peak at the very front and gives
    a finer trade‑off between the large‑index edge (conv[0]) and the
    centre (conv[n‑1]) of the convolution.
*   Sweep the two parameters ``p`` and ``c`` on a coarse grid and then
    refine locally.  A secondary sweep mixes the shifted shape with a
    uniform distribution (parameter λ) – this permits interpolation
    between the “U‑shaped” optimum and the flat baseline.
*   After the parametric phase a longer projected sub‑gradient polishing
    (larger step‑size, more iterations) is performed.
*   Finally a more aggressive stochastic hill‑climb with adaptive
    step‑scale, occasional large jumps and a simple simulated‑annealing
    acceptance rule is used to escape shallow minima.
*   All helpers are top‑level, the entry point is ``propose_candidate``
    as required by the harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v>=0)."""
    if np.all(v <= 0):
        # all zero – fall back to the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two ≥ m for efficient FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


def shifted_reverse_power_law(n: int, p: float, offset: float) -> np.ndarray:
    """
    Shifted reverse power‑law:
        a_i ∝ (n‑i + offset)^{p}   (offset ≥ 0)
    """
    idx = np.arange(n, 0, -1, dtype=np.float64) + offset
    a = idx ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def uniform_vec(n: int) -> np.ndarray:
    """Uniform distribution of length n."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def mixture_rev_uniform(n: int, p: float, offset: float, lam: float) -> np.ndarray:
    """
    Convex mixture of a shifted reverse‑power‑law and the uniform vector:
        a = λ·rev(p,offset) + (1‑λ)·uniform .
    λ∈[0,1].
    """
    rev = shifted_reverse_power_law(n, p, offset)
    if rev is None:
        return None
    uni = uniform_vec(n)
    a = lam * rev + (1.0 - lam) * uni
    # re‑project onto the simplex to avoid any rounding drift
    a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# Parametric sweep – broad exploration of the new families
# ----------------------------------------------------------------------
def parametric_sweep(n: int, time_deadline: float) -> np.ndarray:
    """
    Sweep the (p, offset, λ) space and return the best discovered vector.
    """
    best_vec = None
    best_val = float('inf')

    # ----- coarse sweep over (p, offset) for the pure shifted family -----
    p_grid = np.linspace(0.5, 3.5, 61)            # step ≈0.05
    offset_grid = np.linspace(0.0, 5.0, 26)       # step 0.2
    for p in p_grid:
        for off in offset_grid:
            if time.time() >= time_deadline:
                return best_vec
            a = shifted_reverse_power_law(n, p, off)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- fine grid around the currently best (p, offset) -----
    if best_vec is not None:
        # crude extraction of the parameters from the current best vector:
        # we simply search a narrow window centred on the previous grids.
        p_fine = np.linspace(max(0.1, p - 0.3), p + 0.3, 61)
        off_fine = np.linspace(max(0.0, off - 0.5), off + 0.5, 51)
        for p in p_fine:
            for off in off_fine:
                if time.time() >= time_deadline:
                    return best_vec
                a = shifted_reverse_power_law(n, p, off)
                if a is None:
                    continue
                val = compute_score(a)
                if val < best_val:
                    best_val, best_vec = val, a

    # ----- mixture with uniform (λ) -----
    # Use the best (p,offset) found above as a good seed.
    if best_vec is not None:
        # retrieve plausible p/off from the best vector by a tiny optimisation:
        # we simply reuse the same grids but now add λ.
        for lam in np.linspace(0.0, 1.0, 21):
            a = mixture_rev_uniform(n, p, off, lam)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- random convex combinations of a handful of basic shapes -----
    uniform = uniform_vec(n)
    rev2 = shifted_reverse_power_law(n, 2.0, 0.0)
    rev3 = shifted_reverse_power_law(n, 2.0, 2.0)
    geom = geometric(n, 0.99)
    basics = [uniform, rev2, rev3, geom]

    for _ in range(2500):
        if time.time() >= time_deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Sub‑gradient polishing (more iterations, slightly larger initial step)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """
    Projected sub‑gradient descent on  max(conv(a,a)).
    Uses a diminishing step size  step0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    t = 0

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # indices where the current maximum is attained (within tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        # build sub‑gradient: for each active shift k we add 2*a shifted
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        # small random jitter every few thousand iterations
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb with a simple simulated‑annealing flavour
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_iter: int = 1_500_000) -> np.ndarray:
    """
    Greedy stochastic moves with occasional uphill acceptance.
    Moves a fraction of the mass from one index to another.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    # step scale controls the fraction of the source mass that is moved
    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # numerical safety
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        improve = M_cand < best_M
        # simulated‑annealing acceptance: allow a small probability of uphill moves
        if improve or (np.random.rand() < 0.001):
            if improve:
                best_M = M_cand
                best_a = cand.copy()
                # success – slowly increase step scale
                step_scale = min(step_scale * 1.03, 0.6)
                no_improve = 0
            else:
                # uphill – shrink step scale to stay cautious
                step_scale *= 0.95
                no_improve += 1
            a = cand
        else:
            # failed – decrease step scale aggressively
            step_scale *= 0.995
            no_improve += 1

        it += 1
        # long streak without improvement → random restart jitter
        if no_improve > 200_000:
            jitter = np.random.randn(n) * 0.0015
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 7‑second safety margin for the final evaluation/printing
    deadline = start_time + budget_s - 7.0

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # 0. Optional fallback (if a previous construction is present)
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Parametric sweep (≈ 200 s)
    # -------------------------------------------------------------
    sweep_deadline = deadline - 500.0   # leave a lot of time for polishing
    n = 2048
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val_raw = compute_score(a_initial)               # quick surrogate
            val_real = evaluate_sequence(a_initial.tolist())  # official score
            print(f"[parametric sweep] surrogate = {val_raw:.6f}, "
                  f"official = {val_real:.6f}")
            if val_real < best_val:
                best_val, best_seq = val_real, a_initial.tolist()

    # -------------------------------------------------------------
    # 2. Sub‑gradient polishing (bulk of remaining time)
    # -------------------------------------------------------------
    polish_deadline = deadline - 150.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.30)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -------------------------------------------------------------
    # 3. Final stochastic hill‑climb
    # -------------------------------------------------------------
    hill_deadline = deadline - 7.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # Safeguard – fallback to the best shifted reverse‑power‑law we
    # discovered during the sweep (if everything else failed).
    # -------------------------------------------------------------
    if best_seq is None:
        # Use the parameters that gave the lowest surrogate score
        a_fallback = shifted_reverse_power_law(n, 2.0, 0.5)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final report (official evaluate_sequence)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
