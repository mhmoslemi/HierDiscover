"""
Improved search for a non‑negative step‑height sequence that minimises
evaluate_sequence().  The algorithm consists of:

1. **Broad stochastic exploration** – generate thousands of candidate
   probability vectors from many parametric families (power‑law,
   exponential, beta, Gaussian, sine‑power, U‑shape, piece‑wise linear)
   and Dirichlet draws.  Keep the best few (by the exact objective).

2. **Projected Adam on a smooth max‑approximation** – for each retained
   candidate run a projected Adam optimiser that minimises the
   τ·log‑sum‑exp surrogate of max(conv).  τ is exponentially annealed
   from 0.3 down to 1e‑9.  The learning‑rate schedule is also annealed.
   The optimisation stops when the allocated budget for this phase is
   exhausted.

3. **Polishing with a line‑search sub‑gradient descent** – using the
   true max‑function we perform a sub‑gradient step with an adaptive
   back‑tracking line search and occasional jitter.

4. **Local random refinement** – many cheap random perturbations
   (Gaussian, zero‑mean) are projected back onto the simplex; improvements
   are kept and the perturbation scale is slowly reduced.

5. **Final simulated‑annealing hill‑climb** – a Metropolis walk with
   gradually cooling temperature is applied as a last‑minute escape
   from shallow local minima.

The whole pipeline respects the 1000 s wall‑clock budget; the best
sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·conv_k(a,a)   w.r.t. a.
    Returns g_i = 2·Σ_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    m = len(w)
    rev_w = w[::-1]
    size = 1 << ((n + m - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) with sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def smooth_max(conv: np.ndarray, tau: float):
    """
    Smooth approximation S = τ·log(∑ exp(conv/τ)).
    Returns the value S and the soft‑max weight vector w (∑ w = 1).
    """
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w

# Parametric families for initial candidates
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (p may be negative)."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) (lam may be negative)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α−1}·(n−i)^{β−1}."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(−(i−mid)²/(2σ²))."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_ushape(n: int, p: float) -> np.ndarray:
    """U‑shaped: a_i ∝ ((i+1)*(n−i))^{p}, p>0 concentrates mass near edges."""
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_piecewise_linear(n: int, knots: int = 9) -> np.ndarray:
    """Random piece‑wise linear shape defined by `knots` control points."""
    ctrl = np.random.rand(knots) + 0.01
    xs = np.linspace(0, 1, knots)
    xs_full = np.linspace(0, 1, n)
    a = np.interp(xs_full, xs, ctrl)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def explore_candidates(n: int, deadline: float, keep: int = 30) -> list[np.ndarray]:
    """Generate many vectors, keep the `keep` best according to compute_score."""
    best = []  # list of (score, vector)

    def consider(v: np.ndarray):
        if not np.isfinite(v).all():
            return
        v = np.maximum(v, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        if len(best) < keep:
            best.append((sc, v.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, v.copy())
            best.sort(key=lambda x: x[0])

    # uniform baseline
    consider(np.full(n, 1.0 / n))

    # power‑law (both decreasing and increasing)
    for p in np.linspace(-10.0, 10.0, 81):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(-1.0, 1.0, 81):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # beta family
    al_vals = np.linspace(0.1, 10.0, 21)
    be_vals = np.linspace(0.1, 10.0, 21)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    # Gaussian widths
    for sigma in np.linspace(0.02 * n, 0.5 * n, 15):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # sine‑power shapes
    for p in np.linspace(0.1, 30.0, 30):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # U‑shape families
    for p in np.linspace(0.1, 10.0, 30):
        if time.time() >= deadline:
            break
        consider(generate_ushape(n, p))

    # piece‑wise linear random shapes
    for _ in range(200):
        if time.time() >= deadline:
            break
        consider(generate_piecewise_linear(n, knots=9))

    # Random Dirichlet draws with many concentrations
    concentrations = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per = max(1, (keep * 150) // len(concentrations))
    for conc in concentrations:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            consider(np.random.dirichlet([conc] * n))

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 0.3,
                  tau_end: float = 1e-9,
                  tau_decay: float = 0.92,
                  lr_start: float = 0.07,
                  lr_decay: float = 0.995,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """
    Projected Adam minimising τ·log‑sum‑exp(conv) with an exponential τ‑schedule.
    Returns the best point seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    lr = lr_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        # update schedules
        if t % 300 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        if t % 500 == 0 and lr > 1e-5:
            lr = max(1e-5, lr * lr_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)          # soft‑max weights
        grad = conv_with_weights(a, w)

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)

        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional clipping of the raw step to avoid huge jumps
        step_norm = np.linalg.norm(step)
        if step_norm > 5.0:
            step = step * (5.0 / step_norm)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (lr * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing with simple back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       init_step: float = 0.30,
                       min_step: float = 1e-6,
                       shrink: float = 0.5) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective with line‑search."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    step = init_step

    while time.time() < deadline and step > min_step:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # build sub‑gradient (sum of gradients of active entries)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]   # derivative of a_i * a_{k-i}

        # --- back‑tracking line search ---
        improved = False
        for _ in range(8):
            cand = a - step * g
            cand = project_to_simplex(cand)
            val = compute_score(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                a = cand
                improved = True
                break
            step *= shrink

        if not improved:
            # if none of the reductions worked, increase step modestly
            step = min(step * 1.1, init_step)

        # occasional jitter to diversify search
        if it % 2000 == 0:
            jitter = np.random.randn(n) * (init_step * 0.005 / np.sqrt(it + 1))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Local random refinement – cheap Gaussian perturbations
# ----------------------------------------------------------------------


def local_random_refine(a0: np.ndarray, deadline: float,
                       init_eps: float = 0.02,
                       min_eps: float = 5e-7) -> np.ndarray:
    """
    Repeatedly add zero‑mean Gaussian perturbations (preserving the simplex)
    and keep any improvement.  The perturbation magnitude is slowly
    annealed.
    """
    a = a0.copy()
    n = a.shape[0]
    best_val = compute_score(a)
    eps = init_eps
    it = 0

    while time.time() < deadline and eps > min_eps:
        it += 1
        # zero‑mean direction
        delta = np.random.randn(n)
        delta -= delta.mean()               # enforce sum(delta)=0
        delta *= eps

        cand = a + delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            a = cand
            # success – allow epsilon to grow a little
            eps = min(eps * 1.02, init_eps)
        else:
            # no gain – shrink epsilon
            eps *= 0.96

        # very occasional larger jump to avoid stagnation
        if it % 5000 == 0:
            big = np.random.randn(n) * (init_eps * 0.5)
            big -= big.mean()
            cand_big = a + big
            cand_big = np.maximum(cand_big, 0.0)
            cand_big = project_to_simplex(cand_big)
            v_big = compute_score(cand_big)
            if v_big < best_val:
                best_val = v_big
                a = cand_big

    return a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-9,
                        cooling_factor: float = 0.9993,
                        step_scale_init: float = 0.08,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-14)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # occasional jitter restart
            jitter = np.random.randn(n) * (step_scale * 0.02)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety buffer for the harness
    deadline = start + budget_s - 5.0

    n = 2048
    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Fallback (if a known good construction is available)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad exploration (≈ 15 % of the budget)
    # -----------------------------------------------------------------
    explore_deadline = min(deadline - 600.0,
                           start + max(30.0, 0.15 * budget_s))
    if time.time() < explore_deadline:
        print("[explore] generating candidates …")
        cand_vecs = explore_candidates(n, explore_deadline, keep=30)
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_score:
                best_score, best_seq = val, vec.tolist()
        print(f"[explore] best score = {best_score:.6f}")
    else:
        cand_vecs = []

    # -----------------------------------------------------------------
    # 2. Adam optimisation on the smooth‑max surrogate
    # -----------------------------------------------------------------
    # allocate roughly half of the remaining time to Adam
    time_left = deadline - time.time()
    adam_deadline = time.time() + 0.5 * time_left
    if best_seq is not None and time.time() < adam_deadline:
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)

        # start points: best so far + a few random simple vectors
        start_points = [best_vec,
                        np.full(n, 1.0 / n)]
        for _ in range(4):
            start_points.append(project_to_simplex(np.random.rand(n)))

        # add the top few explored candidates
        for vec in cand_vecs[:5]:
            start_points.append(vec)

        refined_best = best_vec.copy()
        refined_val = best_score

        for idx, a0 in enumerate(start_points):
            if time.time() >= adam_deadline:
                break
            local_deadline = adam_deadline - (time.time() - start) - 0.3
            if local_deadline <= time.time():
                break
            print(f"[adam {idx}] start")
            a_opt = adam_optimize(a0, local_deadline)
            val = compute_score(a_opt)
            print(f"[adam {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_opt.copy()
            # early exit if we already beat the target
            if refined_val < 1.503:
                print("[adam] target reached, breaking early")
                break

        best_score, best_seq = refined_val, refined_best.tolist()
        print(f"[adam] best after Adam = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 15 % of the remaining budget)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 200.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[polish] start")
        a_polished = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Local random refinement (≈ 10 % of remaining budget)
    # -----------------------------------------------------------------
    refine_deadline = deadline - 120.0
    if best_seq is not None and time.time() < refine_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[refine] start")
        a_refined = local_random_refine(a0, refine_deadline)
        val = compute_score(a_refined)
        print(f"[refine] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 5. Final simulated‑annealing hill‑climb (≈ 5 % of remaining time)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[anneal] start")
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – uniform vector if nothing has been found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final official report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
