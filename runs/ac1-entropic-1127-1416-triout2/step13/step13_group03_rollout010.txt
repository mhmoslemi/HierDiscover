def lp_refine_step(f, M, eps=1e-12):
    n = len(f)
    m = 2*n - 1
    f_clipped = np.maximum(f, eps)
    f_clipped = f_clipped / f_clipped.sum()
    # Build Toeplitz
    col = np.concatenate([f_clipped, np.zeros(n-1)])
    row = np.concatenate([f_clipped[:1], np.zeros(n-1)])
    A = scipy.linalg.toeplitz(col, row)
    # Solve LP
    c = -np.ones(n)
    b_ub = np.full(m, M) # M is from original f_clipped conv?
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)]*n,
                  method='highs')
    if not res.success:
        return None
    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 0:
        return None
    scale = 1.0 / sum_b
    g = b_opt * scale
    return g

def entropic_descent(a0, deadline):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    eta0 = 0.5  # step size param
    while time.time() < deadline:
        t += 1
        # compute conv
        conv = full_convolution(a)
        max_val = conv.max()
        # define active indices within tolerance
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]
        # compute gradient as average over active
        g = np.zeros_like(a)
        idx_arr = np.arange(n)
        for k in active:
            j = k - idx_arr
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        g = g / max(1, len(active))
        # learning rate schedule
        eta = eta0 / np.sqrt(t)
        # exponentiated update
        a = a * np.exp(-eta * g)
        # renormalize to sum=1
        a = a / a.sum()
        # optionally clamp small
        a = np.maximum(a, 0.0)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        if t % 1000 == 0:
            # jitter
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(t))
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    return best_a

def full_convolution(a):
    # compute linear full convolution using FFT.
    n = a.shape[0]
    m = 2*n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a):
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

"""
Improved step‑function optimiser.

Main ideas
-----------
1. **Start from the known construction** (height_sequence_1) – it already
   yields a good bound (≈1.51).  The vector is normalised to sum = 1.

2. **Linear‑programming refinement** – following the idea from the
   statement, each iteration solves

        maximise    Σ b_i
        subject to   (f * b)_k ≤ M   ∀k
                     b_i ≥ 0

   where *f* is the current candidate and *M* = max (f * f).  The
   solution *b* is scaled to a probability vector *g* (Σ g = 1) and then
   mixed with *f*.  With a simple back‑tracking line‑search the mix is
   guaranteed to decrease the objective.  Repeating this step drives the
   convolution towards a flat shape.

3. **Projected sub‑gradient polishing** – after the LP phase a few
   thousand sub‑gradient steps on the exact max‑objective finish the
   optimisation.

All phases respect the wall‑clock budget (≈ 1000 s).  The best vector
found is returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a*a using real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    C = A * A
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient (average over all active shifts)
        grad = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]
        grad = grad / max(1, len(active))

        step = step0 / np.sqrt(it)
        a -= step * grad
        a = np.maximum(a, 0.0)
        a /= a.sum()                     # stay on the simplex

        cur = compute_score(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()

        # occasional jitter to escape plateaus
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(it))
            a += jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best


# ----------------------------------------------------------------------
# LP‑based refinement
# ----------------------------------------------------------------------


def lp_refine(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    One LP‑step:
        maximise Σ b_i   subject to   (f * b)_k ≤ M   ∀k ,  b ≥ 0 .
    Returns the scaled solution g (Σ g = 1) or None on failure.
    """
    n = f.shape[0]
    # guard against zeros – they would make the LP unbounded
    eps = 1e-12
    f = np.maximum(f, eps)
    f /= f.sum()

    # Toeplitz matrix for the convolution constraints
    col = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=float)])
    A = scipy.linalg.toeplitz(col, row)          # shape (2n‑1, n)

    c = -np.ones(n)                               # maximise Σ b_i  → minimise −Σ b_i
    b_ub = np.full(2 * n - 1, M, dtype=float)

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    b = res.x
    s = b.sum()
    if s <= 0:
        return None
    g = b / s                                     # scale to a probability vector
    return g

# Main entry point
# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0               # a few seconds safety margin

    # ------------------------------------------------------------------
    # 0. Initialise from the historic construction (if available)
    # ------------------------------------------------------------------
    try:
        fallback = np.array(height_sequence_1, dtype=float)   # type: ignore

        fallback = None

    else:
        # adapt the historic vector to its own length (no resizing)
        n = fallback.shape[0]

        a = np.maximum(a, 0.0)

    best_vec = a.copy()
    best_val = compute_score(best_vec)

    # ------------------------------------------------------------------

        M = conv.max()

        if g is None:

        # accept it immediately.
