Thus define correlation_fft:

def correlation_fft(w, a):
    """Compute linear cross-correlation (w ⋆ a) of lengths (len(w), len(a)) returning array length len(a)."""
    m = len(w)
    n = len(a)
    N = next_pow2(m + n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)[:n]
    return corr

def smooth_gradient_descent(start_a, deadline, max_iters=20000, step0=0.1, eps_factor=0.01):
    a = start_a.copy()
    best = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)  # length 2n-1
        max_conv = conv.max()
        eps = max_conv * eps_factor  # epsilon proportional to scale
        # compute weights for softmax
        exp_vals = np.exp(conv / eps)
        w_norm = exp_vals / exp_vals.sum()
        # gradient from smooth max
        grad = 2.0 * correlation_fft(w_norm, a)
        # step size decreasing
        step = step0 / np.sqrt(t)
        a = a - step * grad
        # project onto simplex (ensure nonnegative sum=1)
        a = project_to_simplex(a)
        # evaluate real score occasionally
        if t % 50 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()
                # print improvement
                print(f"[{time.time():.1f}] smooth descent iter {t} improved to {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best = final_score, a.copy()
        print(f"[{time.time():.1f}] smooth descent final improved to {best_score:.6f}")
    return best, best_score

"""
Propose a step‑function that lowers the evaluated bound.

Algorithm
---------
0.  If a previous construction (height_sequence_1) is available, start from it.
1.  Scan a small collection of analytic families (U‑shaped, centre‑heavy,
    left‑heavy, Gaussian, triangular) on a moderate grid – keep the best.
2.  Run several restarts of *smooth‑max gradient descent*:
    - Approximate the max‑convolution by a soft‑max (log‑sum‑exp) with a
      scale ε proportional to the current peak.
    - Compute the exact gradient of the smooth objective via an FFT‑based
      correlation.
    - Take a diminishing Euclidean step and project onto the probability
      simplex.
    - Keep the best exact score seen.
3.  Refine the result with a short *exponentiated‑gradient (mirror descent)*
    run on the true max‑convolution (the same sub‑gradient used before,
    but now starting from the smoother optimum).
4.  Finish with a simulated‑annealing hill‑climb (mass swaps / transfers)
    for the remaining time budget.
5.  Return the best sequence (as a plain list).  All steps respect the
    supplied wall‑clock limit (default 1000 s).
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a)  where
    (w ⋆ a)[i] = Σ_t w[i+t]·a[t] .
    Returns the first len(a) entries (i = 0 … len(a)-1).
    """
    m = w.size
    n = a.size
    nfft = next_pow2(m + n - 1)
    W = np.fft.rfft(w, nfft)
    A = np.fft.rfft(a, nfft)
    corr = np.fft.irfft(W * np.conj(A), nfft)[:n]
    return corr


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)

# Simplex helpers

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s

def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred on the middle of the interval."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """Symmetric triangular bump of relative half‑width."""
    half_width = max(1, int(width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return w / w.sum()

def smooth_gradient_descent(start_a: np.ndarray,
                            deadline: float,
                            max_iters: int = 20000,
                            step0: float = 0.3,
                            eps_factor: float = 0.02) -> Tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of the max‑convolution.
    Uses a soft‑max (log‑sum‑exp) with scale ε = eps_factor·max(conv).
    After each Euclidean step the iterate is projected onto the simplex.
    Returns the best vector (and its exact score) encountered.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_conv = conv.max()
        eps = max(eps_factor * max_conv, 1e-12)

        # soft‑max weights (stable version)
        shifted = (conv - max_conv) / eps
        w = np.exp(shifted)
        w_norm = w / w.sum()

        # gradient = 2·(w_norm ⋆ a)
        grad = 2.0 * correlation_fft(w_norm, a)

        # diminishing step size
        step = step0 / math.sqrt(t)
        a = a - step * grad
        a = project_to_simplex(a)

        # occasional exact evaluation
        if t % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] smooth‑grad iter {t} improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] smooth‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def exp_gradient_descent(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 120_000,
                         eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using exponentiated gradients.
    Uses the exact sub‑gradient at the index where the auto‑convolution attains its maximum.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] exp‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] exp‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 8_000_000,
                  temp_factor: float = 1.2e-3,
                  step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    Moves consist of swaps or fractional transfers; acceptance follows a linear cooling schedule.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two coordinates
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
