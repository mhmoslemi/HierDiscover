def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    The vector a is assumed non‑negative and sum‑normalised (sum≈1),
    but the function works for any scale.
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # numerical guard – overflow gives inf, replace by a very large number
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)

    """
    Discrete Beta‑type density.
    p,q>0.  When p,q<1 the shape is U‑shaped; p,q>1 gives a bell.
    """
    i = np.arange(n, dtype=np.float64)
    # (i+0.5)^{p-1} (n-0.5-i)^{q-1}
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 10.0,
                         beta_end: float = 30_000.0,
                         lr: float = 0.008,
                         sym_interval: int = 10) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    ``sym_interval`` forces symmetry every few iterations (a_i ← (a_i+a_{n-1-i})/2).
    """
    n = start_a.shape[0]
    eps = 1e-12
    # log‑softmax re‑parameterisation
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        # enforce symmetry periodically (helps the shape stay U‑like)
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t a of smooth‑max (see derivation in original script)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the log‑softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective check
        if it % 4000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a


def directed_mass_transfer(start_a: np.ndarray,
                          deadline: float,
                          max_iters: int = 1_500_000,
                          init_step: float = 0.20) -> np.ndarray:
    """
    Move a small fraction of mass from the entry that contributes most to the
    current maximal convolution index to a random low‑impact entry.
    Accept only improving moves; adapt the transfer fraction ``step``.
    """
    n = start_a.size
    a = start_a.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    step = init_step

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))   # index of the active constraint

        # contributions of each position i to conv[k]
        i_vals = np.arange(n)
        j = k - i_vals
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j[mask]]

        donor = int(np.argmax(contrib))          # biggest contributor
        receiver = np.random.randint(n)
        if donor == receiver:
            continue

        delta = step * a[donor]                  # amount to move
        if delta <= 0:
            continue
        cand = a.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < best_score:
            a = cand
            best_score = cand_score
            best_a = a.copy()
            # successful – increase step a little
            step = min(step * 1.02, 0.5)
        else:
            # unsuccessful – shrink step
            step = max(step * 0.97, 1e-5)

        if it % 300_000 == 0:
            print(f"[{time.time():.1f}] mass‑transfer it {it} best {best_score:.6f}")

    return best_a

def hill_climb_alpha(alpha0: float, n: int, deadline: float) -> float:
    """One‑dimensional hill‑climb on the product‑family exponent."""
    best = alpha0
    best_score = compute_score(symmetric_product_family(n, best))
    step = 1.0
    while time.time() < deadline and step > 1e-4:
        improved = False
        for sign in (-1, 1):
            cand = max(0.0, best + sign * step)
            sc = compute_score(symmetric_product_family(n, cand))
            if sc < best_score:
                best_score = sc
                best = cand
                improved = True
                print(f"[{time.time():.1f}] α refined to {best:.6f} score {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best


def hill_climb_beta(p0: float, q0: float, n: int, deadline: float) -> (float, float):
    """Simple two‑parameter hill‑climb on the Beta family."""
    best_p, best_q = p0, q0
    best_score = compute_score(beta_family(n, best_p, best_q))
    step = 0.2
    while time.time() < deadline and step > 1e-3:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(1e-4, best_p + dp)
                cand_q = max(1e-4, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_score = sc
                    best_p, best_q = cand_p, cand_q
                    improved = True
                    print(f"[{time.time():.1f}] β refined to p={best_p:.6f}, q={best_q:.6f} score {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
    return best_p, best_q

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under a fixed wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a few seconds safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # Optional fallback to a previously known good sequence
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = np.array([float(x) for x in height_sequence_1], dtype=np.float64)  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # Choose a working length (large enough for a fine shape)
    # -----------------------------------------------------------
    n = 32768   # 2^15 – a good compromise between resolution and speed

    # -----------------------------------------------------------
    # 1. Coarse deterministic scan of analytic families
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] Start coarse scan (n={n})")

    # ----- product family -----
    best_alpha = None
    for alpha in np.linspace(0.0, 30.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best (product α={alpha:.4f}) = {best_score:.6f}")

    # ----- Beta family (two‑parameter, wide range) -----
    # we probe a log‑scale grid for p,q ∈ [0.02, 1.0]
    p_vals = np.logspace(np.log10(0.02), 0.0, 20)
    q_vals = np.logspace(np.log10(0.02), 0.0, 20)
    for p in p_vals:
        for q in q_vals:
            if time.time() >= deadline - 30.0:
                break
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec
                # store the parameters for later refinement
                best_beta_params = (p, q)
                print(f"[{time.time():.1f}] new best (Beta p={p:.4f}, q={q:.4f}) = {best_score:.6f}")

    # -----------------------------------------------------------
    # 2. Random dense exploration (≈3000 trials)
    # -----------------------------------------------------------
    trials = 0
    max_trials = 3000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(3)
        if choice == 0:      # product family
            alpha = np.random.uniform(0.0, 30.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 1:    # Beta family random
            p = np.random.uniform(0.01, 1.2)
            q = np.random.uniform(0.01, 1.2)
            vec = beta_family(n, p, q)
        else:                # mixture of uniform and product (adds a little regularisation)
            lam = np.random.rand()
            alpha = np.random.uniform(0.0, 30.0)
            prod = symmetric_product_family(n, alpha)
            vec = lam * np.full(n, 1.0 / n) + (1.0 - lam) * prod
            vec = normalize(vec)

        # occasional random shift (right cyclic shift then zero‑pad on the left)
        if np.random.rand() < 0.2:
            d = np.random.randint(0, n // 25 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Local refinement of the best product‑α and Beta parameters
    # -----------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining product α (start {best_alpha:.4f})")
        refined_alpha = hill_climb_alpha(best_alpha, n, deadline - 10.0)
        vec = symmetric_product_family(n, refined_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha = refined_alpha
            print(f"[{time.time():.1f}] refined product‑α={refined_alpha:.6f} → {best_score:.6f}")

    if 'best_beta_params' in locals() and time.time() < deadline - 20.0:
        p0, q0 = best_beta_params
        print(f"[{time.time():.1f}] refining Beta (start p={p0:.4f}, q={q0:.4f})")
        refined_p, refined_q = hill_climb_beta(p0, q0, n, deadline - 10.0)
        vec = beta_family(n, refined_p, refined_q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] refined Beta p={refined_p:.6f}, q={refined_q:.6f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Adam (smooth‑max) refinement – symmetry enforced periodically
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Adam refinement (symmetrised)")
        a_start = best_seq.astype(np.float64)
        a_opt = adam_smooth_optimize(
            start_a=a_start,
            deadline=deadline - 150.0,
            max_iters=200_000,
            beta_start=10.0,
            beta_end=30_000.0,
            lr=0.008,
            sym_interval=10,
        )
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Directed mass‑transfer descent (exact max‑objective)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] directed mass‑transfer descent")
        a_mt = directed_mass_transfer(
            start_a=best_seq.astype(np.float64),
            deadline=deadline - 30.0,
            max_iters=1_200_000,
            init_step=0.20,
        )
        sc_mt = compute_score(a_mt)
        if sc_mt < best_score:
            best_score = sc_mt
            best_seq = a_mt
            print(f"[{time.time():.1f}] after mass‑transfer best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Final fallback (uniform) if everything else failed
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 1024
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = compute_score(best_seq)

    # final official evaluation (includes security checks)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq.tolist()
