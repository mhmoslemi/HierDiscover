def optimal_mixture(candidates, deadline):
    """Solve LP to find convex combination that minimizes max convolution."""
    K=len(candidates)
    if K==0:
        raise ValueError("no candidates")
    n=candidates[0].shape[0]
    m = 2*n - 1
    # compute conv matrix:
    convs = np.empty((K, m), dtype=float)
    for i, v in enumerate(candidates):
        convs[i,:] = full_convolution(v)
    # Build LP.
    # Variables order: lambda0...lambda_{K-1}, t
    c = np.zeros(K+1)
    c[-1] = 1.0  # minimize t
    # A_ub rows for each shift: sum_i lambda_i * conv_i[k] - t <= 0
    A_ub = np.hstack([convs.T, -np.ones((m,1))])
    b_ub = np.zeros(m)
    # Equality: sum lambda =1
    A_eq = np.hstack([np.ones((1,K)), np.zeros((1,1))])
    b_eq = np.array([1.0])
    # bounds:
    bounds = [(0, None)] * K + [(0, None)]  # t >=0

    # Solve using linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    lambdas = res.x[:K]
    mix_vec = np.dot(lambdas, np.array(candidates))
    # Ensure numeric errors: project onto simplex
    mix_vec = project_to_simplex(mix_vec)
    return mix_vec

"""Step‑function optimizer for the evaluate_sequence objective.

The algorithm combines several ideas that were not used together before:

* A broad baseline exploration of parametric families (power‑law,
  exponential, beta, Gaussian, sine‑power) and many Dirichlet draws.
* Adam optimisation on a smooth‑max surrogate of the max‑convolution.
* A *linear‑programming mixture* stage: all promising vectors are
  combined in a convex way by solving a small LP that directly
  minimises the peak of the convolution.
* A final polishing stage (projected sub‑gradient descent) and a short
  simulated‑annealing hill‑climb.

All steps respect the 1000 s wall‑clock budget.  The best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families for the global‑search phase
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    tot = a.sum()
    return a / tot if tot > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·vec1 + (1‑λ)·vec2, projected onto the simplex."""
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Global baseline exploration
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 5000,
):
    """Explore a variety of parametric families and Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # Power‑law sweep
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # Exponential sweep
    for lam in np.linspace(0.0001, 0.5, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # Beta grid
    al_vals = np.linspace(0.02, 2.0, 31)
    be_vals = np.linspace(0.02, 2.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # Random Beta draws
    for _ in range(200):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)   # 0.01 … 2
        beta = 10 ** np.random.uniform(-2, 0.3)
        try_candidate(generate_beta(n, alpha, beta))

    # Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # Sine‑power sweep
    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # Mixtures of families
    for _ in range(150):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # Dirichlet draws
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate of the max‑convolution
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-5,
    tau_decay: float = 0.96,
    lr_start: float = 0.12,
    lr_decay: float = 0.99,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """Adam optimiser on the smooth‑max surrogate.
    Returns the best vector (projected on the simplex) seen before deadline."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # Temperature annealing
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)          # length m
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()       # soft‑max distribution

        # Gradient of smooth‑max: 2 * (a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # Learning‑rate annealing
        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-4)

        # Small jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max‑objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # Build sub‑gradient (sum of contributions from all active shifts)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (stochastic hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 2e-3,
    final_temp: float = 1e-6,
    cooling_factor: float = 0.9992,
    step_scale_init: float = 0.06,
    step_scale_min: float = 5e-6,
) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 7000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming mixture of several candidates
# ----------------------------------------------------------------------


def optimal_mixture(candidates, deadline):
    """Find the convex combination of ``candidates`` that minimizes the
    peak of the convolution.  Returns the mixed vector (simplex‑projected)."""
    K = len(candidates)
    if K == 0:
        raise ValueError("no candidates for mixing")
    n = candidates[0].shape[0]
    m = 2 * n - 1

    # Compute all convolutions once
    convs = np.empty((K, m), dtype=np.float64)
    for i, v in enumerate(candidates):
        convs[i] = full_convolution(v)

    # Linear programme:
    # minimise   t
    # subject to sum_i λ_i * conv_i[k] - t <= 0   (∀k)
    #            sum_i λ_i = 1
    #            λ_i >= 0,   t >= 0
    c = np.zeros(K + 1)
    c[-1] = 1.0  # minimise t

    A_ub = np.hstack([convs.T, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    A_eq = np.hstack([np.ones((1, K)), np.zeros((1, 1))])
    b_eq = np.array([1.0])

    bounds = [(0, None)] * K + [(0, None)]

    # Solve (the problem size is modest, so we ignore the deadline here)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    lambdas = res.x[:K]
    mix = np.dot(lambdas, np.vstack(candidates))
    mix = project_to_simplex(mix)
    return mix
