"""
Propose a low‑score step‑function.

The algorithm consists of three phases:
1. **Coarse parametric search** – a mixture of a uniform vector and the
   symmetric “product” family a_i ∝ [(i+½)(n‑½‑i)]^{‑α}.  A small grid over
   (α,λ) is scanned on a moderate size (n≈8 k) to obtain a good starter.
2. **Upsampling** – the best short vector is interpolated to a much larger
   resolution (N≈3·10⁴).  Because the objective 2 n·max(conv)/sum² converges
   to a constant as n→∞, a finer discretisation can improve the bound.
3. **Smooth‑max Adam optimisation** – the large vector is refined by
   minimizing a log‑sum‑exp surrogate of the max‑convolution.  The temperature
   β is increased from 5 up to 5 000 while Adam adapts the log‑probabilities.
   The best iterate (according to the exact objective) is kept.

All stages respect the global 1000 s wall‑clock limit and periodically
print progress.  The returned list is a plain Python list of non‑negative
floats whose sum is 1, ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the competition:
        2·n·max(a*a) / (sum(a)**2)
    The vector a is assumed non‑negative.
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised.
    Works for any α≥0 (α>n leads to vanishing values but is handled).
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    # use np.exp of -α*logprod; for very large α the exponent may underflow → 0
    w = np.exp(-alpha * logprod)
    # take care of possible infinities (α huge)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 5_000.0,
                         lr: float = 0.008) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate of max(conv).
    ``beta`` grows geometrically from ``beta_start`` to ``beta_end``.
    The parameters are kept in log‑space to enforce positivity.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise log‑probabilities
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # current probabilities
        a = np.exp(z)
        a = a / a.sum()

        # convolution and smooth‑max weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)          # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]      # pick the central n entries

        # back‑prop through the soft‑max (log‑softmax) layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 5000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score
