"""
Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.

Algorithm Overview
------------------
1. **Exploratory sweep** – in addition to the classic families (power‑law,
   exponential, beta, Gaussian) we explore several *edge‑biased* families
   (inverse‑beta, symmetric power‑law, edge‑power‑law, double‑exponential,
   V‑valley and discrete arcsine).  A large pool of Dirichlet draws is also
   sampled.  The best few candidates are retained.

2. **Smoothed mirror descent** – we minimise a smooth approximation of the
   ``max`` convolution using Adam on the soft‑max parametrisation
   ``a = softmax(x)``.  The temperature τ is annealed very slowly to obtain
   a near‑exact solution.  Several restarts are performed from the retained
   candidates and from fresh random Dirichlet draws.

3. **Final simulated annealing** – a short Metropolis‑style hill‑climb helps
   to escape any remaining plateaux.

The whole procedure respects the 1000 s wall‑clock budget, automatically
adapts the time spent in each phase and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution utilities (real FFT, full linear convolution)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for real inputs)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (autoconvolution)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (no clipping – all entries already non‑negative)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used during the search.
    The vector is assumed to sum to 1; the returned value equals
    ``evaluate_sequence`` for such a vector.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a non‑negative vector *already* normalised)
# ----------------------------------------------------------------------


def _normalise(v: np.ndarray) -> np.ndarray:
    s = v.sum()
    if s <= 0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=float)) ** (-p)
    return _normalise(a)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    return _normalise(a)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}  (Beta‑type shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        a = np.zeros_like(a)
    return _normalise(a)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the centre."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return _normalise(a)


def generate_sym_powerlaw(n: int, p: float) -> np.ndarray:
    """
    Symmetric edge‑biased power‑law:
    a_i ∝ (i+1)^{-p} + (n-i)^{-p}.
    """
    i = np.arange(n, dtype=float)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    return _normalise(a)


def generate_edge_powerlaw(n: int, p: float) -> np.ndarray:
    """
    Edge‑focused power‑law:
    a_i ∝ ((i+1)·(n‑i))^{-p}.
    """
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n - i + 1.0)) ** (-p)
    return _normalise(a)


def generate_double_exponential(n: int, lam: float) -> np.ndarray:
    """
    Exponential decay from the edges toward the centre:
    a_i ∝ exp(-lam·min(i, n‑1‑i)).
    """
    i = np.arange(n, dtype=float)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return _normalise(a)


def generate_valley_linear(n: int) -> np.ndarray:
    """
    V‑shaped *valley*: low in the centre, high at the ends.
    a_i ∝ distance from the centre.
    """
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.abs(i - centre) + 1e-12   # avoid exact zeros
    return _normalise(a)


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine density:
    a_i ∝ 1/√(i·(n‑i+1)).
    """
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    return _normalise(a)


# ----------------------------------------------------------------------
# Global exploration (keeps up to `max_keep` best vectors)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 10):
    """Sweep many families and random Dirichlet draws; keep the best candidates."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- Classic families -------------------------------------------------
    for p in np.linspace(0.1, 2.5, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    for lam in np.linspace(0.0001, 0.07, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    al_vals = np.concatenate([np.linspace(0.05, 0.9, 12),
                              np.linspace(1.0, 5.0, 8)])
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    for sigma in np.linspace(n / 24.0, n / 2.0, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # ---- New edge‑biased families -----------------------------------------
    for p in np.linspace(0.2, 3.0, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_sym_powerlaw(n, p))

    for p in np.linspace(0.2, 2.0, 180):
        if time.time() >= deadline:
            break
        try_candidate(generate_edge_powerlaw(n, p))

    for lam in np.linspace(0.001, 0.12, 150):
        if time.time() >= deadline:
            break
        try_candidate(generate_double_exponential(n, lam))

    # V‑valley (no parameters)
    try_candidate(generate_valley_linear(n))

    # Arcsine (no parameters)
    try_candidate(generate_arcsine(n))

    # ---- Dirichlet draws --------------------------------------------------
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1500
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.20,
                    init_tau_factor: float = 5.0,
                    min_tau: float = 1e-10,
                    tau_decay: float = 0.9998) -> np.ndarray:
    """
    Minimise the (soft‑max) approximation to max(conv(a,a)) using Adam.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise x = log a (+tiny epsilon)
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature based on current max
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping (simple and stable) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution of current a -----
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- soft‑max weights over the convolution entries -----
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # stable soft‑max
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient wrt a (g_i = 2·∑_k w_k·a_{k-i}) -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain‑rule for the soft‑max parametrisation -----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature slowly -----
        tau = max(min_tau, tau * tau_decay)

        # ----- occasional exact evaluation -----
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-3,
                        final_temp: float = 5e-7,
                        cooling_factor: float = 0.999,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Simple Metropolis‑style annealing that tweaks the simplex coordinates."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move a fraction of mass from i to j
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to keep the walk alive
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 balances flexibility and speed.
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global exploration (≈ 20 % of the budget)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - (budget_s * 0.55)  # reserve most time for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=10)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smoothed Adam optimisation with restarts (≈ 70 % of the budget)
    # -----------------------------------------------------------------
    polish_deadline = deadline - (budget_s * 0.15)
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)

        # primary run starting from the current best
        refined = smooth_optimize(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # additional random restarts while time permits
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            if remaining < 5.0:
                break
            sub_deadline = time.time() + min(30.0, remaining)

            # random Dirichlet start (vary concentration)
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (≈ 10 % of the budget)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 2.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
