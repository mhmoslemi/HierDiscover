import time

# ---------------------------- Linear algebra utilities ----------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D arrays are supported")
    n = v.shape[0]
    # Shift by mean to ensure numerical stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (zero‑padded)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming a sums to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def smooth_max(conv: np.ndarray, tau: float):
    """τ·log‑sum‑exp approximation of max and the soft‑max weights."""
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w

def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    Returns g_i = 2· Σ_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    rev_w = w[::-1]
    size = 1 << ((n + len(w) - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + len(w) - 1]
    idx = (len(w) - 1) - np.arange(n)
    return 2.0 * conv_full[idx]

def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (p may be negative)."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) (lam may be negative)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α−1}·(n−i)^{β−1} (Beta density on the discrete grid)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_inverse_distance(n: int, p: float) -> np.ndarray:
    """Edge‑heavy: a_i ∝ ((i+0.5)*(n-0.5-i))^{-p}  (p>0)."""
    i = np.arange(n, dtype=float) + 0.5
    d = i * (n - i)
    a = d ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw from a symmetric Dirichlet."""
    v = np.random.dirichlet([concentration] * n)
    return v.astype(float)

def generate_random_spectral(n: int) -> np.ndarray:
    """
    Produce a random vector via a random phase spectrum.
    The sequence length is n; we keep the first n samples of the inverse FFT.
    """
    L = 1 << ((2 * n - 1 - 1).bit_length())  # power of two >= 2n-1
    # Random phases for frequencies 1..L/2-1
    phases = np.random.rand(L // 2 - 1) * 2.0 * np.pi
    A = np.empty(L, dtype=complex)
    A[0] = 1.0  # magnitude 1, phase 0
    A[1:L // 2] = np.exp(1j * phases)
    if L % 2 == 0:
        A[L // 2] = 1.0
    A[L // 2 + 1:] = np.conj(A[1:L // 2][::-1])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    a = project_to_simplex(a[:n])
    return a

def explore_candidates(n: int, deadline: float, keep: int = 20) -> list[np.ndarray]:
    """Generate a rich pool of candidates and retain the ``keep`` best."""
    best = []  # list of (score, vector)

    def consider(v: np.ndarray):
        if not np.isfinite(v).all():
            return
        v = np.maximum(v, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        if len(best) < keep:
            best.append((sc, v.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, v.copy())
            best.sort(key=lambda x: x[0])

    # Uniform baseline
    consider(np.full(n, 1.0 / n))

    # Power‑law sweep
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # Exponential sweep
    for lam in np.linspace(-0.7, 0.7, 81):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # Beta sweep (including strong edge‑heavy shapes)
    al_vals = np.concatenate([np.linspace(0.01, 0.3, 10),
                              np.linspace(0.5, 3.0, 12)])
    be_vals = np.concatenate([np.linspace(0.01, 0.3, 10),
                              np.linspace(0.5, 3.0, 12)])
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    # Inverse‑distance sweep (U‑shape)
    for p in np.linspace(0.2, 2.5, 14):
        if time.time() >= deadline:
            break
        consider(generate_inverse_distance(n, p))

    # Random Dirichlet draws with a wide range of concentrations
    concentrations = [0.01, 0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 5.0]
    draws_per_c = max(1, (keep * 250) // len(concentrations))
    for conc in concentrations:
        for _ in range(draws_per_c):
            if time.time() >= deadline:
                break
            consider(generate_random_dirichlet(n, conc))

    # Random spectral candidates
    for _ in range(keep * 5):
        if time.time() >= deadline:
            break
        consider(generate_random_spectral(n))

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 1.0,
                  tau_end: float = 1e-9,
                  tau_decay: float = 0.85,
                  lr: float = 0.06,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Projected Adam minimising the smooth log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        # Anneal temperature periodically
        if t % 150 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        # smooth approximation of the max
        _, w = smooth_max(conv, tau)
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # Occasional jitter to escape plateaus
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (lr * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a

# ---------------------------- Sub‑gradient polishing ----------------------------

def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.25) -> np.ndarray:
    """Projected sub‑gradient descent on the exact L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # Construct sub‑gradient from all indices that are within tolerance
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            # each contributing pair adds 2*a[j]
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # Add a small random perturbation every few thousand iterations
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

# ---------------------------- Simulated annealing ----------------------------

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        cooling_factor: float = 0.9997,
                        step_scale_init: float = 0.06,
                        step_scale_min: float = 2e-6) -> np.ndarray:
    """Metropolis hill‑climb allowing occasional worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.04, 0.15)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.96, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # Occasional global jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.02)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    # 4) Inverse‑distance (U‑shaped) family
    for p in np.linspace(0.2, 2.5, 14):
        if time.time() >= deadline:
            break
        consider(generate_inverse_distance(n, p))

    # 5) Random Dirichlet draws with a spread of concentrations
    concentrations = [0.01, 0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 5.0]
    draws_per_c = max(1, (keep * 250) // len(concentrations))
    for conc in concentrations:
        for _ in range(draws_per_c):
            if time.time() >= deadline:
                break
            consider(generate_random_dirichlet(n, conc))

    # 6) Random spectral candidates
    for _ in range(keep * 5):
        if time.time() >= deadline:
            break
        consider(generate_random_spectral(n))

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 1.0,
                  tau_end: float = 1e-9,
                  tau_decay: float = 0.85,
                  lr: float = 0.06,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Projected Adam minimising τ·log‑sum‑exp(conv) with exponential τ‑schedule."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        if t % 150 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)          # soft‑max weights
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # Small jitter to escape flat plateaus
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (lr * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exact objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.25) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        cooling_factor: float = 0.9997,
                        step_scale_init: float = 0.06,
                        step_scale_min: float = 2e-6) -> np.ndarray:
    """Metropolis hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.04, 0.15)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.96, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.02)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 123456, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    n = 2048
    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously known construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Exploration (≈ 180 s)
    # -----------------------------------------------------------------
    explore_deadline = deadline - 720.0  # leave ~280 s for the later stages
    if time.time() < explore_deadline:
        print("[explore] generating candidates …")
        cand_vecs = explore_candidates(n, explore_deadline, keep=20)
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_score:
                best_score, best_seq = val, vec.tolist()
        print(f"[explore] best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam optimisation (≈ 300 s)
    # -----------------------------------------------------------------
    if best_seq is not None:
        adam_deadline = min(deadline - 300.0,
                            start + 200.0 + (deadline - start) * 0.35)
        start_points = []

        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        start_points.append(best_vec)

        # deterministic starts
        start_points.append(np.full(n, 1.0 / n))
        for _ in range(5):
            start_points.append(project_to_simplex(np.random.rand(n)))

        # best few from exploration
        for vec in cand_vecs[:5]:
            start_points.append(vec)

        refined_best = best_vec.copy()
        refined_val = best_score

        for idx, a0 in enumerate(start_points):
            if time.time() >= adam_deadline:
                break
            local_deadline = adam_deadline - (time.time() - start) - 0.5
            if local_deadline <= time.time():
                break
            print(f"[adam {idx}] start")
            a_opt = adam_optimize(a0, local_deadline)
            val = compute_score(a_opt)
            print(f"[adam {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_opt.copy()
                print(f"[adam {idx}] NEW GLOBAL BEST = {refined_val:.6f}")

        best_score, best_seq = refined_val, refined_best.tolist()
        print(f"[adam] best after Adam = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 350 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 150.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[polish] start")
        a_polished = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()
            print("[polish] NEW GLOBAL BEST")

    # -----------------------------------------------------------------
    # 4. Simulated annealing (final refinement, ≈ 120 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[anneal] start")
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_final.tolist()
            print("[anneal] NEW GLOBAL BEST")

    # -----------------------------------------------------------------
    # Safeguard – uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
