def simulated_annealing(a_start, deadline):
    a = a_start.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = len(a)
    T0 = best_val * 0.1  # starting temperature
    T = T0
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose moving fraction of mass from i to j
        delta = a[i] * 0.01 * (np.random.rand())
        # ensure not negative
        if delta > a[i]:
            delta = a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # project onto simplex? Actually sum unchanged, and both nonnegatives, so okay.
        # compute score
        val = compute_score(cand)
        # accept condition
        if val < best_val or np.random.rand() < np.exp((best_val - val) / T):
            a = cand
            if val < best_val:
                best_val, best_a = val, cand.copy()
        # cooling schedule
        it += 1
        if it % 1000 == 0:
            # exponent per iteration; simple geometrical cooling
            T *= 0.995
        if T < 1e-8:
            T = 1e-8
    return best_a

"""
search_step_function_v3.py

A faster, more explorative optimiser for the ``evaluate_sequence`` objective.

Main ideas
-----------
*  Keep the successful reverse‑power‑law family but search it with a finer
   golden‑section sweep (instead of a coarse grid).
*  Add an exponential decay family – a shape not covered before.
*  Run a longer sub‑gradient polishing phase (the objective is scale‑invariant,
   we keep the simplex constraint).
*  Finish with a simulated‑annealing hill‑climb that can also accept temporary
   deteriorations – this helps to escape the shallow local minima that the
   greedy hill‑climb of the reference often gets stuck in.
*  All work is split in three timed phases so that the 1000 s budget is used
   efficiently even on a single CPU.

The function ``propose_candidate`` is the entry‑point required by the harness.
It returns a plain list of floats – the best sequence found before the budget
expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (Σv = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of parametric shapes
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)         # n, n‑1, …, 1
    a = idx ** p
    a /= a.sum()
    return a


def exponential_decay(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(‑α·i)   (α > 0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Front plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}   (0<q<1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p} (standard decreasing power‑law)."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Parameter sweep – builds a small set of promising seeds
# ----------------------------------------------------------------------
def parametric_sweep(n: int, time_deadline: float) -> list[np.ndarray]:
    """Explores several families, returns the few best vectors discovered."""
    best = []                                 # list of (score, vector)
    def maybe_insert(vec):
        val = compute_score(vec)
        best.append((val, vec))
        best.sort(key=lambda x: x[0])
        # keep only the top 5
        del best[5:]

    # ---- reverse power‑law (golden‑section search) ----
    lo, hi = 0.3, 4.0                         # reasonable interval
    phi = (1 + 5 ** 0.5) / 2
    while time.time() < time_deadline and hi - lo > 1e-4:
        c = hi - (hi - lo) / phi
        d = lo + (hi - lo) / phi
        a_c = reverse_power_law(n, c)
        a_d = reverse_power_law(n, d)
        f_c = compute_score(a_c)
        f_d = compute_score(a_d)
        if f_c < f_d:
            hi = d
            maybe_insert(a_c)
        else:
            lo = c
            maybe_insert(a_d)

    # a few points around the final interval
    for p in np.linspace(lo, hi, 15):
        if time.time() >= time_deadline:
            break
        maybe_insert(reverse_power_law(n, p))

    # ---- exponential decay ----
    for alpha in np.linspace(0.0005, 0.02, 40):
        if time.time() >= time_deadline:
            break
        vec = exponential_decay(n, alpha)
        maybe_insert(vec)

    # ---- geometric ratios (dense near 1) ----
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 100),
            np.linspace(0.990, 0.9999, 150))):
        if time.time() >= time_deadline:
            break
        vec = geometric(n, q)
        if vec is None:
            continue
        maybe_insert(vec)

    # ---- mixed plateau + power‑law ----
    for L_frac in np.linspace(0.02, 0.30, 12):
        for p in np.linspace(0.3, 0.9, 13):
            if time.time() >= time_deadline:
                break
            vec = mixed_plateau_power(n, L_frac, p)
            if vec is None:
                continue
            maybe_insert(vec)

    # ---- random convex combinations of a few basics ----
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    revpl = reverse_power_law(n, 1.8)
    geom = geometric(n, 0.98)
    basics = [uniform, revpl, geom]

    for _ in range(800):
        if time.time() >= time_deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        maybe_insert(a)

    # return only the vectors (sorted by their score)
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (same spirit as the reference)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.14) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient: for each active shift k add 2*a shifted by k
        g = np.zeros_like(a)
        for k in active:
            # k corresponds to index in full convolution (0 … 2n‑2)
            shift = k - (n - 1)               # signed shift
            if shift >= 0:
                g[shift:] += 2.0 * a[:n - shift]
            else:
                g[:n + shift] += 2.0 * a[-shift:]
        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        # occasional jitter
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (can accept temporary worsening moves)
# ----------------------------------------------------------------------
def annealing_hill_climb(a_start: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    Greedy + simulated annealing moves.
    A move transfers a random fraction of mass from i to j.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]

    # initial temperature roughly 10 % of the current score
    T = max(best_val * 0.1, 1e-4)
    it = 0

    while time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a random proportion (≤50 %) of the source entry
        max_move = a[i] * 0.5
        if max_move == 0.0:
            continue
        delta = np.random.rand() * max_move

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # still on the simplex (sum unchanged) and non‑negative by construction
        val = compute_score(cand)

        # accept if improvement or with a Boltzmann probability
        if val < best_val or np.random.rand() < math.exp((best_val - val) / T):
            a = cand
            if val < best_val:
                best_val, best_a = val, cand.copy()

        # cooling
        it += 1
        if it % 3000 == 0:
            T *= 0.97
            if T < 1e-8:
                T = 1e-8

        # occasional random jitter to avoid stagnation
        if it % 12000 == 0:
            jitter = np.random.randn(n) * 5e-4
            a = best_a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Optimises a non‑negative step‑function for the ``evaluate_sequence`` metric.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # safety margin

    # -----------------------------------------------------------------
    # 1.  Quick fallback – a tiny reverse‑power‑law (in case everything else fails)
    # -----------------------------------------------------------------
    n = 2048
    best_seq = reverse_power_law(n, 1.8).tolist()
    best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 2.  Parametric sweep (≈150 s)
    # -----------------------------------------------------------------
    sweep_deadline = start_time + 150.0
    if time.time() < sweep_deadline:
        seed_vectors = parametric_sweep(n, sweep_deadline)
        for vec in seed_vectors:
            val = evaluate_sequence(vec.tolist())
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[sweep] best score so far = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3.  Sub‑gradient polishing (bulk of the remaining time)
    # -----------------------------------------------------------------
    polish_deadline = start_time + 800.0
    if time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.14)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4.  Simulated‑annealing hill‑climb (use whatever time is left)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = annealing_hill_climb(a_start, deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[anneal‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
