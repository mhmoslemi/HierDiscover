    candidates = []

            best_f = v

    # 2. Main iterative improvement using LP mixing

    lp_deadline = deadline - 300.0

        A = np.empty((2*n-1, n), dtype=float)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')

            break
        g0 = res.x

"""
A two‑stage optimiser for the step‑height problem.

Stage 1 – global exploration of simple parametric families (power‑law,
exponential, sine‑power, beta, Gaussian and random Dirichlet draws)
and keeps the best vector found.

Stage 2 – iterative refinement based on the classic “mix‑with‑a‑better
vector” scheme from the statement.  For a current distribution `f`
(let Σf = 1) we compute M = max(conv(f,f)).  A linear programme finds a
vector `g₀` maximising Σg under the constraints conv(f,g) ≤ M.  After
normalising `g = g₀ / Σg₀` we mix `f ← (1‑t) f + t g` with a small step
size `t`.  Repeating this (while time permits) monotonically reduces the
maximum convolution value.  Finally a short Adam optimisation on the
smooth‑max surrogate performs a polishing pass.

The whole routine respects the 1000 s wall‑clock limit and returns a
list of non‑negative floats (the step heights) that minimises
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (fast for n≈2000)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a vector normalised to sum‑1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families for the coarse initial search
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by the final Adam polishing)
# ----------------------------------------------------------------------


def smooth_max(conv: np.ndarray, tau: float):
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    Returns g_i = 2·Σ_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    m = len(w)
    rev_w = w[::-1]
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)
    return 2.0 * conv_full[idx]


def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 0.5,
                  tau_end: float = 1e-9,
                  tau_decay: float = 0.90,
                  lr: float = 0.10,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Projected Adam on the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming based mixing step
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1) × n) such that A·g = conv(f,g).
    For each column j we place f[:] on rows j … j+n‑1.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.empty((m, n), dtype=float)
    # fill column by column (fast vectorised slice copy)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_mix_step(f: np.ndarray, M: float, t_step: float, deadline: float):
    """
    One LP‑based mixing step:
    * maximise Σg under conv(f,g) ≤ M,
    * normalise the result,
    * mix with the current vector using step size t_step.
    Returns the new vector (or None if LP failed).
    """
    n = f.shape[0]
    A = build_conv_matrix(f)
    c = -np.ones(n)                # maximise Σg  → minimise –Σg
    b = np.full(2 * n - 1, M)

    # Solve the LP (highs is fast and memory‑efficient)
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return None

    g0 = res.x
    s = g0.sum()
    if s <= 0:
        return None
    g = g0 / s                     # renormalise to Σg = 1

    # convex combination
    f_new = (1.0 - t_step) * f + t_step * g
    f_new = np.maximum(f_new, 0.0)
    f_new = f_new / f_new.sum()    # protect against numerical drift
    if time.time() > deadline:
        return None
    return f_new

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑height sequence that minimises
    ``evaluate_sequence``.  The routine combines a cheap global scan,
    an LP‑based iterative refinement and a final Adam polishing.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # small safety margin

    n = 2048

    # ------------------------------------------------------------------
    # 0. Optional fall‑back if a known good construction is present
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Coarse global exploration (≈ 60 s)
    # ------------------------------------------------------------------
    def consider(v):
        nonlocal best_seq, best_score
        if not np.isfinite(v).all():
            return
        v = np.clip(v, 0.0, None)
        if v.sum() == 0:
            return
        v = v / v.sum()
        sc = compute_score(v)
        if sc < best_score:
            best_score, best_seq = sc, v.tolist()
            print(f"[explore] new best = {sc:.6f}")

    # a) uniform
    consider(np.full(n, 1.0 / n))

    # b) a few parametric families
    for p in np.linspace(0.0, 4.0, 17):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))
        consider(generate_powerlaw(n, -p))

    for lam in np.linspace(-0.25, 0.25, 17):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    for p in [0.5, 1.5, 3.0, 6.0]:
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    for sigma in np.linspace(0.08 * n, 0.4 * n, 9):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # random Dirichlet draws (different concentration levels)
    for conc in [0.05, 0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(12):
            if time.time() >= deadline:
                break
            consider(np.random.dirichlet([conc] * n))

    if best_seq is None:
        # safety net – uniform
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq))

    f = np.array(best_seq, dtype=float)
    f = f / f.sum()                     # ensure exact normalisation
    best_score = compute_score(f)

    print(f"\n[init] best score after exploration = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. LP‑based mixing iteration (≈ 500 s)
    # ------------------------------------------------------------------
    mix_deadline = deadline - 250.0     # leave time for final polishing
    iteration = 0
    while time.time() < mix_deadline:
        iteration += 1
        conv_ff = full_convolution(f)
        M = conv_ff.max()

        # try a succession of step sizes, accept the first that improves
        improved = False
        for t_step in (0.5, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005):
            f_new = lp_mix_step(f, M, t_step, mix_deadline)
            if f_new is None:
                continue
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-9:
                f = f_new
                best_score = new_score
                improved = True
                print(f"[mix {iteration:3d}] t={t_step:5.3f}  score={new_score:.6f}")
                break
        if not improved:
            # stagnation – add a tiny random jitter to escape flat regions
            jitter = np.random.randn(n) * 1e-5
            f = project_to_simplex(f + jitter)
            # continue loop; we may still improve later

    # ------------------------------------------------------------------
    # 3. Final Adam polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[adam] polishing start")
        f = adam_optimize(f, deadline,
                           tau_start=0.05,
                           tau_end=1e-8,
                           tau_decay=0.92,
                           lr=0.12)
        best_score = compute_score(f)
        print(f"[adam] final polished score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return result (converted to plain list)
    # ------------------------------------------------------------------
    final_seq = f.tolist()
    reported = evaluate_sequence(final_seq)
    print(f"\nSearch finished. reported score = {reported:.6f}")
    return final_seq
