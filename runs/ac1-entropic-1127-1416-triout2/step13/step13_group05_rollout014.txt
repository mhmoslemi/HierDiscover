def gradient_descent_smoothmax(a0, deadline, max_iter=20000):
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    len_w = 2*n - 1
    n_fft = 1 << ((3*n - 2 - 1).bit_length())

    t = 0
    alpha = 0.5
    alpha_max = 200.0
    step0 = 0.5  # maybe adjust
    while time.time() < deadline and t < max_iter:
        # compute conv
        conv = full_convolution(a)
        max_conv = conv.max()
        # compute softmax weights
        # stability
        scale = alpha
        exp_term = np.exp(scale * (conv - max_conv))
        w = exp_term / exp_term.sum()
        # compute correlation gradient via FFT
        a_fft = np.fft.rfft(a, n=n_fft)
        w_fft = np.fft.rfft(w, n=n_fft)
        corr = np.fft.irfft(np.conj(w_fft) * a_fft, n=n_fft)
        start = len_w - 1
        grad = 4.0 * n * corr[start:start + n]

        # step size schedule
        step = step0 / math.sqrt(t + 1)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        a_new = project_to_simplex(a_new)

        # evaluate
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # maybe accept if worse? we can always keep best but continue search. Might incorporate simulated annealing.
        # For simplicity, update a even if worse (to possibly escape)
        a = a_new

        # increase alpha gradually
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.02)

        t += 1

        # occasionally jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.001 / math.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a

"""
search_step_function_v3.py

A new optimiser for ``evaluate_sequence`` that uses a smooth‑max
approximation of the ``max(conv)`` term together with a projected
gradient descent (with an annealing schedule on the smoothness) and a
final simulated‑annealing hill‑climb.  The gradient is computed via
FFT‑based correlation, which is much faster than the explicit
sub‑gradient loops used previously.  The method starts from several
hand‑crafted seeds (reverse power‑law, beta‑shape, mixed plateau …)
and runs independent restarts inside the time budget, keeping the
best vector found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negative entries that sum to one.
    """
    if np.all(v <= 0):
        # all non‑positive → return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using an FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # next power‑of‑two for circular convolution
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The problem is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors (used for seeding)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (n-i)^{p}   (front‑loaded decreasing power‑law, p ≥ 0).
    """
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Front plateau of length L followed by a decreasing power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient‑descent with smooth‑max
# ----------------------------------------------------------------------
def smoothmax_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               max_iter: int = 200_000) -> np.ndarray:
    """
    Projected gradient descent on a smooth approximation of the
    max‑convolution term.
    Uses a soft‑max (log‑sum‑exp) smoothing with an annealing schedule
    on the temperature parameter `alpha`.  The gradient is obtained
    via FFT‑based correlation, which is O(N log N) per iteration.
    Returns the best point encountered.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    n = a.shape[0]
    len_w = 2 * n - 1
    # length needed for a linear correlation of length n and len_w
    n_fft = 1 << ((3 * n - 2 - 1).bit_length())

    t = 0
    alpha = 0.5                # start with a very smooth max
    alpha_target = 200.0       # final temperature (very sharp)
    step0 = 0.03               # initial step size (will decay)

    while time.time() < deadline and t < max_iter:
        t += 1

        # ---------- convolution ----------
        conv = full_convolution(a)

        # ---------- soft‑max weights ----------
        # stability: subtract the maximum before exponentiation
        max_c = conv.max()
        exp_term = np.exp(alpha * (conv - max_c))
        w = exp_term / exp_term.sum()

        # ---------- gradient via correlation ----------
        a_fft = np.fft.rfft(a, n=n_fft)
        w_fft = np.fft.rfft(w, n=n_fft)
        # correlation a ⋆ w = ifft( conj(FFT(w)) * FFT(a) )
        corr = np.fft.irfft(np.conj(w_fft) * a_fft, n=n_fft)
        start = len_w - 1                     # index that corresponds to lag 0
        grad = 4.0 * n * corr[start:start + n]  # factor 4 n from the derivative

        # ---------- step and projection ----------
        step = step0 / math.sqrt(t)           # classic 1/√t decay
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)

        # ---------- keep best ----------
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if t % 1000 == 0:
                print(f"[gd] iter {t:,}  score = {cur_score:.6f}")

        # ---------- anneal the smoothness ----------
        if alpha < alpha_target:
            alpha = min(alpha_target, alpha * 1.02)   # exponential increase

        # ---------- occasional jitter ----------
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.001 / math.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb_sa(a_start: np.ndarray,
                  deadline: float,
                  max_iter: int = 800_000) -> np.ndarray:
    """
    Greedy stochastic moves with a temperature that slowly cools.
    A move is accepted if it improves the objective or with a
    probability exp(-(Δ)/T) otherwise.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    T0 = 0.02           # initial temperature
    T_end = 1e-5        # final temperature
    it = 0

    while time.time() < deadline and it < max_iter:
        # linear cooling
        T = max(T_end, T0 * (1.0 - it / max_iter))

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a fraction of mass from i to j
        delta = 0.02 * a[i]                 # step scale ~2 %
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        Δ = M_cand - best_M

        if Δ < 0 or np.random.rand() < math.exp(-Δ / T):
            # accept (even if worse, with probability)
            a = cand
            if M_cand < best_M:
                best_M = M_cand
                best_a = cand.copy()
        it += 1

        # occasional progress report
        if it % 100_000 == 0:
            print(f"[sa] iter {it:,}  best max‑conv = {best_M:.8f}")

    return best_a
