"""
Propose a low‑autocorrelation step‑function.

This implementation builds on the previous version but adds several
enhancements that have proven useful for pushing the bound below
1.503:

* A larger search length (n=4096) for the final candidate – still fast
  enough for the official O(N²) evaluation but large enough to see
  asymptotic improvement.
* A new “geometric’’ analytic family (a_i ∝ r**i) and an extended
  parameter sweep for the existing families.
* Precise one‑dimensional optimisation of the product‐family exponent
  α and the β‑family parameters (p,q) via golden‑section search.
* Longer Adam optimisation with a warm‑up schedule for the smooth‑max
  temperature and optional periodic symmetrisation.
* A final short sub‑gradient descent and a greedy hill‑climb.
* All heavy work uses FFT‑based convolutions (O(N log N)), while the
  final call to ``evaluate_sequence`` (which uses a naïve ``np.convolve``)
  is performed only once on the modest length n=4096.

The code respects a 1000 s wall‑clock budget, prints progress
information, and returns the best sequence found as a plain list of
floats.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)**2) used for internal
    optimisation.  ``a`` is assumed non‑negative and will be projected
    onto the simplex by the caller.
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def geometric_family(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i (0<r≤1)."""
    if r <= 0:
        r = 1e-12
    i = np.arange(n, dtype=np.float64)
    w = np.power(r, i)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Discrete Beta density.
    p,q>0. When p,q<1 the shape is U‑shaped, otherwise bell‑shaped.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 700_000,
                         beta_start: float = 2.0,
                         beta_end: float = 30_000.0,
                         lr: float = 0.015,
                         sym_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    Temperature ``beta`` follows a geometric schedule.
    Optional symmetrisation every ``sym_interval`` steps.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # Initialise via log‑softmax (guarantees positivity)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        # occasional symmetrisation (helps for symmetric families)
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # sporadic true‑objective check
        if it % 4000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improvement → {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement → {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.05) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improvement → {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_greedy(start_a: np.ndarray,
                      deadline: float,
                      max_steps: int = 6_000_000,
                      init_scale: float = 0.12) -> Tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb that transfers a small fraction of mass
    from a large entry to a random other entry whenever this reduces
    the objective.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    scale = init_scale

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        i = np.argmax(cur_a)          # donor (largest entry)
        j = np.random.randint(n)      # receiver
        if i == j:
            continue

        delta = scale * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            scale = min(scale * 1.03, 0.5)   # expand step on success
        else:
            scale = max(scale * 0.97, 1e-4)  # shrink on failure

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] greedy step {step}, best {best_score:.6f}")

    return best_a, best_score

def golden_section_search(func, lo: float, hi: float, max_iter: int = 15) -> Tuple[float, float]:
    """Standard golden‑section search on a unimodal function."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd


def refine_product_alpha(alpha0: float, n: int, deadline: float) -> float:
    """Refine the product‑family exponent α using golden‑section."""
    def score(alpha):
        return compute_score(symmetric_product_family(n, alpha))
    lo = max(0.0, alpha0 - 2.0)
    hi = alpha0 + 2.0
    lo = max(lo, 0.0)
    best_alpha, best_score = golden_section_search(score, lo, hi, max_iter=14)
    print(f"[{time.time():.1f}] refined product α = {best_alpha:.6f} → {best_score:.6f}")
    return best_alpha


def refine_beta_params(p0: float, q0: float, n: int, deadline: float) -> Tuple[float, float]:
    """Refine the two‑parameter Beta family using a small local grid search."""
    best_p, best_q = p0, q0
    best_score = compute_score(beta_family(n, p0, q0))
    step = 0.3
    while time.time() < deadline and step > 1e-3:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(0.01, best_p + dp)
                cand_q = max(0.01, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_score = sc
                    best_p, best_q = cand_p, cand_q
                    improved = True
                    print(f"[{time.time():.1f}] Beta refined to p={best_p:.6f}, q={best_q:.6f} → {best_score:.6f}")
        if not improved:
            step *= 0.5
    return best_p, best_q

            r = np.random.uniform(0.2, 0.99)
            vec = geometric_family(n, r)
        else:                  # Beta
            p = np.random.uniform(0.05, 2.5)
            q = np.random.uniform(0.05, 2.5)
            vec = beta_family(n, p, q)

        # occasional random shift (right‑ward)
        if np.random.rand() < 0.2:
            d = np.random.randint(0, n // 40 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Precise refinement of the best analytic parameters
    # ------------------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining product α (start {best_alpha:.4f})")
        refined_alpha = refine_product_alpha(best_alpha, n, deadline - 10.0)
        vec = symmetric_product_family(n, refined_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha = refined_alpha
            print(f"[{time.time():.1f}] refined product α → {best_score:.6f}")

    # Rough Beta refinement (starting from a balanced U‑shape)
    if time.time() < deadline - 20.0:
        start_p, start_q = 0.5, 0.5
        refined_p, refined_q = refine_beta_params(start_p, start_q, n, deadline - 10.0)
        vec = beta_family(n, refined_p, refined_q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] refined Beta (p={refined_p:.4f}, q={refined_q:.4f}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Adam optimisation on the smooth‑max surrogate (no enforced symmetry)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Adam optimisation start (n={n})")
        a_start = best_seq.astype(np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=a_start,
            deadline=deadline - 150.0,
            max_iters=700_000,
            beta_start=2.0,
            beta_end=30_000.0,
            lr=0.015,
            sym_interval=500,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Projected sub‑gradient descent on the true max‑objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent start")
        a_sub, sc_sub = subgrad_optimize(
            start_a=best_seq.astype(np.float64),
            deadline=deadline - 60.0,
            max_iters=2_200_000,
            step0=0.05,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub
            print(f"[{time.time():.1f}] after sub‑gradient best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final greedy hill‑climb (mass‑transfer moves)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final greedy hill‑climb")
        a_final, sc_final = hill_climb_greedy(
            start_a=best_seq.astype(np.float64),
            deadline=deadline,
            max_steps=5_000_000,
            init_scale=0.12,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final
            print(f"[{time.time():.1f}] greedy improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Fallback to a uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 9. Official evaluation (runs the security checks and the exact
    #    objective via np.convolve).  This is done only once.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the external driver.
    return best_seq.tolist()
