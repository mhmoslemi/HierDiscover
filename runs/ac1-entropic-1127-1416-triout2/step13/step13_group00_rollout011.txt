"""
Improved search for a non‑negative step‑height sequence that minimises
``evaluate_sequence``.

The algorithm consists of:

  1. Broad generation of candidate vectors from many parametric families
     (including several window functions) and random Dirichlet draws.
     The 20 best candidates (according to the exact objective) are kept.

  2. A large number of independent Adam runs (projected onto the simplex)
     minimise a smooth approximation of the max‑convolution
     (τ·log‑sum‑exp).  The temperature τ and the learning rate are
     decayed continuously so that the optimiser eventually works on the
     true max.

  3. A short sub‑gradient polishing phase on the exact objective.

  4. A simulated‑annealing hill‑climb that performs small mass‑transfer
     moves while accepting occasional uphill moves.

The whole procedure respects the 1000 s wall‑clock budget; intermediate
best scores are printed.  The final sequence is returned as a plain list
of floats.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of   S(a)=∑_k w_k·(a*a)[k]   w.r.t. a.
    Returns   g_i = 2·∑_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    m = w.shape[0]                # should be 2*n-1
    rev_w = w[::-1]
    size = 1 << ((n + m - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)  # positions that give the required sums
    return 2.0 * conv_full[idx]


# ----------------------------------------------------------------------
# Exact objective
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector on the simplex."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient weights
# ----------------------------------------------------------------------


def smooth_max(conv: np.ndarray, tau: float) -> (float, np.ndarray):
    """
    τ·log‑sum‑exp approximation of max(conv) together with the soft‑max
    weight vector w that sums to 1.
    """
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


# ----------------------------------------------------------------------
# Parametric families for the exploration phase
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_ushape(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_hann(n: int, p: float = 1.0) -> np.ndarray:
    a = (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1))) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_hamming(n: int, p: float = 1.0) -> np.ndarray:
    a = (0.54 - 0.46 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_blackman(n: int, p: float = 1.0) -> np.ndarray:
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * np.arange(n) / (n - 1))) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

# Exploration – keep a pool of promising vectors
# ----------------------------------------------------------------------


def consider_candidate(v: np.ndarray, pool: list, keep: int) -> None:
    """Project v onto the simplex, evaluate and possibly insert into pool."""
    if not np.isfinite(v).all():
        return
    v = np.maximum(v, 0.0)
    v = project_to_simplex(v)
    sc = compute_score(v)
    if len(pool) < keep:
        pool.append((sc, v.copy()))
        pool.sort(key=lambda x: x[0])
    elif sc < pool[-1][0]:
        pool[-1] = (sc, v.copy())
        pool.sort(key=lambda x: x[0])


def explore_candidates(n: int, deadline: float, keep: int = 20) -> list[np.ndarray]:
    """Generate many vectors from different families and keep the best ones."""
    pool: list[tuple[float, np.ndarray]] = []

    # uniform
    consider_candidate(np.full(n, 1.0 / n), pool, keep)

    # power‑law, both decreasing and increasing
    for p in np.linspace(0.0, 5.0, 51):
        if time.time() >= deadline:
            break
        consider_candidate(generate_powerlaw(n, p), pool, keep)
        consider_candidate(generate_powerlaw(n, -p), pool, keep)

    # exponential sweep
    for lam in np.linspace(-0.5, 0.5, 81):
        if time.time() >= deadline:
            break
        consider_candidate(generate_exponential(n, lam), pool, keep)

    # beta family
    al_vals = np.linspace(0.1, 5.0, 21)
    be_vals = np.linspace(0.1, 5.0, 21)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider_candidate(generate_beta(n, a, b), pool, keep)

    # Gaussian widths
    for sigma in np.linspace(0.05 * n, 0.5 * n, 15):
        if time.time() >= deadline:
            break
        consider_candidate(generate_gaussian(n, sigma), pool, keep)

    # sine‑power
    for p in np.linspace(0.1, 8.0, 20):
        if time.time() >= deadline:
            break
        consider_candidate(generate_sine_power(n, p), pool, keep)

    # U‑shape
    for p in np.linspace(0.1, 5.0, 20):
        if time.time() >= deadline:
            break
        consider_candidate(generate_ushape(n, p), pool, keep)

    # Window families
    for p in np.linspace(0.1, 5.0, 20):
        if time.time() >= deadline:
            break
        consider_candidate(generate_hann(n, p), pool, keep)
        consider_candidate(generate_hamming(n, p), pool, keep)
        consider_candidate(generate_blackman(n, p), pool, keep)

    # Random Dirichlet draws
    concentrations = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0]
    draws_per = max(1, (keep * 200) // len(concentrations))
    for conc in concentrations:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            consider_candidate(np.random.dirichlet([conc] * n), pool, keep)

    # Return only the vectors, sorted by score (best first)
    pool.sort(key=lambda x: x[0])
    return [vec for _, vec in pool[:keep]]


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  max_iters: int = 20000,
                  tau_start: float = 0.5,
                  tau_end: float = 1e-9,
                  lr_start: float = 0.07,
                  lr_end: float = 0.003,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Projected Adam minimising τ·log‑sum‑exp(conv) with continuous schedules."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedules (continuous exponential decay)
        prog = t / max_iters
        tau = max(tau_end, tau_start * (tau_end / tau_start) ** prog)
        lr = max(lr_end, lr_start * (lr_end / lr_start) ** prog)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)        # soft‑max weights
        grad = conv_with_weights(a, w)

        # Adam moments
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional tiny jitter to escape flat regions
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr * 0.005)
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing on the exact objective
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-9,
                        cooling: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices with non‑zero mass
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    `evaluate_sequence`. Returns a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # -----------------------------------------------------------------
    # 0. Optional fallback (if a historic construction is available)
    # -----------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad exploration (≈ 120 s)
    # -----------------------------------------------------------------
    explore_deadline = deadline - 600.0
    if time.time() < explore_deadline:
        print("[explore] generating candidates …")
        cand_vecs = explore_candidates(2048, explore_deadline, keep=20)
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_score:
                best_score, best_seq = val, vec.tolist()
        print(f"[explore] best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam optimisation from many start points (≈ 400 s)
    # -----------------------------------------------------------------
    if best_seq is not None:
        adam_deadline = min(deadline - 250.0,
                            start_time + 150.0 + (deadline - start_time) * 0.40)
        start_points = []

        # the current champion
        start_points.append(project_to_simplex(np.array(best_seq, dtype=float)))

        # uniform
        start_points.append(np.full(2048, 1.0 / 2048))

        # a handful of pure random points
        for _ in range(5):
            start_points.append(project_to_simplex(np.random.rand(2048)))

        # top candidates from the exploration phase
        if 'cand_vecs' in locals():
            for vec in cand_vecs[:5]:
                start_points.append(vec)

        # keep only distinct points
        uniq = []
        for v in start_points:
            key = tuple(np.round(v, 8))
            if key not in uniq:
                uniq.append(key)
        start_points = [np.array(k) for k in uniq]

        refined_best = None
        refined_val = float("inf")

        for idx, a0 in enumerate(start_points):
            if time.time() >= adam_deadline:
                break
            local_deadline = adam_deadline - (time.time() - start_time) - 0.5
            if local_deadline <= time.time():
                break
            print(f"[adam {idx}] start")
            a_opt = adam_optimize(a0, local_deadline,
                                  max_iters=20000,
                                  tau_start=0.5,
                                  tau_end=1e-9,
                                  lr_start=0.07,
                                  lr_end=0.003)
            val = compute_score(a_opt)
            print(f"[adam {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_opt.copy()

        if refined_best is not None:
            best_score, best_seq = refined_val, refined_best.tolist()
            print(f"[adam] best after Adam = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Mixing of top solutions (simple convex combinations)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print("[mix] trying convex combinations")
        current_best = np.array(best_seq, dtype=float)
        # gather a small pool of good vectors (already have them in cand_vecs)
        pool = [current_best]
        if 'cand_vecs' in locals():
            pool.extend(cand_vecs[:5])
        for i in range(len(pool)):
            for j in range(i + 1, len(pool)):
                for _ in range(3):
                    w = np.random.rand()
                    mix = w * pool[i] + (1.0 - w) * pool[j]
                    mix = project_to_simplex(mix)
                    val = compute_score(mix)
                    if val < best_score:
                        best_score, best_seq = val, mix.tolist()
                        print(f"[mix] improved to {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Sub‑gradient polishing (≈ 150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[polish] start")
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.30)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 5. Simulated annealing (final refinement, ≈ remaining time)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[anneal] start")
        a_final = simulated_annealing(a_start, anneal_deadline,
                                      init_temp=1e-3,
                                      final_temp=1e-9,
                                      cooling=0.9995,
                                      step_scale_init=0.07,
                                      step_scale_min=1e-6)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing has been found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / 2048] * 2048
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final score using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
