def subgrad_polish(a0,...):
    a = a0.copy()
    best_a = a0.copy()
    best_val = compute_score(a)
    t = 0
    while time<deadline:
        t+=1
        conv = full_convolution(a)
        max_val = conv.max()
        # compute subgradient:
        active = np.where(conv >= max_val - tol)[0]
        g = np.zeros_like(a)
        # For each active index k, i + j = k
        # subgradient of (a*a)_k with respect to a_i: 2*a_{k-i} if valid else 0.
        # In our earlier code we aggregated for each active k: sum_{i} 2*a_j (where j = k-i).
        # Equivalent to:
        for k in active:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            i_idx = np.arange(i_start, i_end+1)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]
        # subgradient of max conv is g (since it's gradient of max of convex functions? Actually each (a*a)_k is convex? It may not be convex but subgradient okay).
        # Then descent step:
        step = 0.5 / np.sqrt(t)  # hyperparameter
        a = a - step * g
        # keep nonnegative
        a = np.maximum(a, 0.0)
        # project onto simplex:
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0/len(a))
        else:
            # Project onto simplex: implement algorithm (Duchi et al.)
            u = np.sort(a)[::-1]
            cssv = np.cumsum(u) - 1
            rho = np.nonzero(u - cssv / (np.arange(len(u))+1) > 0)[0][-1]
            theta = cssv[rho] / (rho + 1)
            a = np.maximum(a - theta, 0)
        # compute value:
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
    return best_a

def simplex_projection(v):
    """Project vector v onto probability simplex (non-negative sum to 1)."""
    # From Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Non‑negative step‑function optimiser (≈ 1000 s budget).

The approach improves the previous baseline in three ways:

1. **Richer global exploration** – In addition to power‑law,
   exponential and beta families we also sweep sine‑power,
   half‑sine and discrete Gaussian families, and we draw many random

   *Sub‑gradient polishing* is added: a projected sub‑gradient method that

   single FFT‑based convolution, then projects back onto the probability

3. **Extended random‑restart & annealing** – While time permits new
   random Dirichlet starts are polished (both Adam and sub‑gradient);

All stages respect the wall‑clock limit; the routine returns the best

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _next_pow_two(m: int) -> int:
    """Return the smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·n‑1) using real FFT.
    a must be a 1‑D array.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = _next_pow_two(m)
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def linear_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two arbitrary 1‑D arrays using FFT.
    Returns exact length len(x)+len(y)-1.
    """
    n = x.shape[0] + y.shape[0] - 1
    n_fft = _next_pow_two(n)
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:n]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x≥0, sum x = 1}.
    Returns a new array.
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero; return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the global sweep
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_half_sine(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n). Positive on (0,n)."""
    a = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^{p}."""
    base = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
    a = np.power(base, p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float, mu: float = None) -> np.ndarray:
    """Discrete Gaussian a_i ∝ exp( -(i‑mu)^2 / (2·σ^2) )."""
    if mu is None:
        mu = (n - 1) / 2.0
    idx = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration (baseline)
# ----------------------------------------------------------------------


def _add_candidate(vec: np.ndarray, best: list, max_keep: int):
    """Maintain a sorted list of the best candidates (low score first)."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return
    vec = vec / s
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_search(n: int, deadline: float, max_keep: int = 15):
    """
    Perform a dense sweep over a handful of parametric families and
    many random Dirichlet draws, keeping the ``max_keep`` best vectors.
    """
    best = []   # list of (score, vector)

    # 1) power‑law
    for p in np.linspace(0.05, 6.0, 350):
        if time.time() >= deadline:
            break
        _add_candidate(generate_powerlaw(n, p), best, max_keep)

    # 2) exponential
    for lam in np.linspace(1e-6, 0.3, 350):
        if time.time() >= deadline:
            break
        _add_candidate(generate_exponential(n, lam), best, max_keep)

    # 3) beta family
    al_vals = np.concatenate([np.geomspace(0.01, 0.2, 12),
                              np.linspace(0.2, 4.0, 24)])
    be_vals = np.concatenate([np.geomspace(0.01, 0.2, 12),
                              np.linspace(0.2, 4.0, 24)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _add_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # 4) half‑sine (no parameter)
    if time.time() < deadline:
        _add_candidate(generate_half_sine(n), best, max_keep)

    # 5) sine‑power family
    for p in [0.2, 0.5, 1.0, 2.0, 4.0, 8.0]:
        if time.time() >= deadline:
            break
        _add_candidate(generate_sine_power(n, p), best, max_keep)

    # 6) Gaussian family – sweep σ as a fraction of n
    for sigma_frac in [0.05, 0.10, 0.15, 0.20, 0.30, 0.40]:
        if time.time() >= deadline:
            break
        sigma = sigma_frac * n
        _add_candidate(generate_gaussian(n, sigma), best, max_keep)

    # 7) massive random Dirichlet draws with varied concentration
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 2000   # ≈ 14 000 draws total – fast on modern CPUs
    for conc in concs:
        if time.time() >= deadline:
            break
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _add_candidate(a, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam polishing (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def adam_polish(a0: np.ndarray, deadline: float,
                lr: float = 0.05,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                tol: float = 1e-12) -> np.ndarray:
    """
    Adam‑based optimisation of max_k (a*a)_k on the simplex.
    The variables are stored in log‑space and transformed by soft‑max.
    Returns the best vector observed before the deadline.
    """
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)          # initialise log‑probabilities
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # ---- soft‑max projection ----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ---- convolution and active set ----
        conv = full_convolution(a)
        max_val = conv.max()
        active = np.where(conv >= max_val - max_val * tol)[0]

        # ---- sub‑gradient w.r.t. a (active constraints) ----
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # ---- mirror‑descent step (soft‑max Jacobian) ----
        g_u = a * (g - np.dot(g, a))

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)   # gentle decay
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to avoid stagnation
        if t % 2000 == 0:
            u += np.random.randn(n) * (lr * 0.001)

        # ---- keep the best solution ----
        cur_val = float(2.0 * n * max_val)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # re‑evaluate every few thousand steps (guards against numerical drift)
        if t % 5000 == 0:
            max_u = np.max(u)
            exp_shift = np.exp(u - max_u)
            a = exp_shift / exp_shift.sum()
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()

    return best_a if best_a is not None else a0


# ----------------------------------------------------------------------
# Sub‑gradient polishing (projected gradient descent)
# ----------------------------------------------------------------------


def subgrad_polish(a0: np.ndarray, deadline: float,
                   init_lr: float = 0.8,
                   tol: float = 1e-12) -> np.ndarray:
    """
    Projected sub‑gradient descent on the simplex.
    The sub‑gradient of max(conv) is obtained via a single FFT‑based
    convolution of the active‑mask with the reversed vector.
    Returns the best vector seen before the deadline.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        # compute convolution and active mask
        conv = full_convolution(a)
        max_val = conv.max()
        active_mask = (conv >= max_val - max_val * tol).astype(float)   # length 2n‑1

        # sub‑gradient via convolution of active mask with reversed a
        a_rev = a[::-1]
        grad_full = linear_convolution(active_mask, a_rev)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]                       # slice of length n

        # step size schedule (1/√t)
        step = init_lr / np.sqrt(t)

        # gradient step & projection
        a = a - step * grad
        a = simplex_projection(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional small random jitter to escape shallow plateaux
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step * 0.02)
            a = a + jitter
            a = simplex_projection(a)

    return best_a if best_a is not None else a0


# ----------------------------------------------------------------------
# Light simulated‑annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # problem size – 2048 gives a good balance between expressivity & speed
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previous run
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 450.0   # reserve most time for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=15)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Polishing phase (Adam + Sub‑gradient) with random restarts
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)

        # 2a) primary Adam polish
        refined_adam = adam_polish(a0, polish_deadline)
        val_adam = compute_score(refined_adam)
        print(f"[adam‑polish‑primary] refined score = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, refined_adam.tolist()

        # 2b) primary sub‑gradient polish (takes remaining time)
        subgrad_deadline = polish_deadline - 30.0
        refined_sub = subgrad_polish(a0, subgrad_deadline)
        val_sub = compute_score(refined_sub)
        print(f"[subgrad‑polish‑primary] refined score = {val_sub:.6f}")
        if val_sub < best_val:
            best_val, best_seq = val_sub, refined_sub.tolist()

        # 2c) random‑restart polishing while we still have time
        restart_deadline = polish_deadline
        while time.time() < restart_deadline:
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            # decide randomly which polish to apply
            if np.random.rand() < 0.5:
                refined = adam_polish(a_start, restart_deadline)
            else:
                refined = subgrad_polish(a_start, restart_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[random‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback to uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
