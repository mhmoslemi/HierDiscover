def softmax_adam_polish(a0, deadline, init_tau=0.1, final_tau=1e-5, decay=0.999, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):

    n = a0.shape[0]; u = np.log(a0 + 1e-20)
    m = np.zeros_like(u); v = np.zeros_like(u)
    best_a = a0
    best_val = compute_score(a0)
    t=0
    tau = init_tau

    while time.time() < deadline:
        t+=1
        # compute a via softmax
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()
        # compute convolution
        conv = full_convolution(a)   # length 2n-1
        # compute actual score
        cur_val = 2*n*conv.max()
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
        # compute smooth gradient
        # compute softmax weights
        # For numerical stability, subtract max conv/tau
        max_conv = np.max(conv)
        # Actually compute w = np.exp((conv - max_conv)/tau) to avoid overflow
        exp_conv = np.exp((conv - max_conv)/tau)
        w = exp_conv / exp_conv.sum()
        # compute gradient of smooth loss
        # gradient of loss L = tau * log( sum exp(conv/tau) ) = tau * log( sum exp )
        # g = Σ_k w_k * grad(conv[k]) = weighted sum of grad conv.
        # grad conv[k] w.r.t a: 2 * shift of a
        # compute g via correlation:
        g = correlation_gradient(w, a)
        # mirror descent: gradient w.r.t u
        g_u = a * (g - np.dot(g, a))
        # Adam update with optional step size scaling
        m = beta1 * m + (1 - beta1) * g_u
        v = beta2 * v + (1 - beta2) * (g_u * g_u)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr / (np.sqrt(t) + 1.0) * (1 / (np.sqrt(v_hat)+eps))
        u -= step * m_hat
        # decrease tau
        tau = max(final_tau, tau * decay)
        # occasional jitter
        if t % 3000 == 0:
            u += np.random.randn(n) * (lr * 0.001)
    return best_a

def correlation_gradient(w, a):
    # w length L = 2n-1, a length n
    # compute correlation g_i = 2 * Σ_j w_{i + j} * a_j for i in 0..n-1
    # Equivalent to 2 * (valid part of convolution of w_rev and a) reversed
    w_rev = w[::-1]
    n = a.shape[0]
    L = w.shape[0]   # =2n-1
    size = L + n - 1
    n_fft = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    W = np.fft.rfft(w_rev, n=n_fft)
    conv = np.fft.irfft(A * W, n=n_fft)
    # conv length = size
    start = n - 1    # index in conv where valid part starts
    valid = conv[start:start + n]  # length n
    g = 2 * valid[::-1]
    return g

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

The optimiser works in three stages:

1. **Exploratory sweep** –  dense scans of a few parametric families
   (power‑law, exponential, beta) plus a large pool of *sparse* uniform
   vectors (random support of size m ≈ 2·√n).  The ten best candidates
   (by the exact objective 2·n·max(conv)) are retained.

2. **Soft‑max Adam polishing** – each retained candidate is refined by
   minimising a *smooth* approximation of the max‑convolution
   (log‑sum‑exp with a temperature τ that is annealed to 0).  The
   gradient of the smoothed objective is obtained analytically via a
   correlation of the softmax weights with the current vector.  Adam
   updates are performed on a log‑softmax parametrisation
   ``a_i = exp(u_i) / Σ_j exp(u_j)``.

3. **Local simulated‑annealing** – a short Metropolis‑style hill‑climb
   is applied to the best vector found so far to escape possible
   plateaux.

All stages respect the 1000 s wall‑clock budget, printing intermediate
scores and returning the best sequence (plain ``list[float]``) found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape:
       a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sparse_uniform(n: int, m: int) -> np.ndarray:
    """Uniform mass on a random support of size m (m ≤ n)."""
    if m <= 0:
        return np.full(n, 0.0)
    idx = np.random.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Global exploration / baseline
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 10):
    """Sweeps a few families and many random sparse vectors, keeping the
    best ``max_keep`` vectors (as numpy arrays)."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s                     # project onto the simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) parametric families – same as before
    for p in np.linspace(0.05, 5.0, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    for lam in np.linspace(1e-5, 0.2, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    al_vals = np.concatenate([np.geomspace(0.01, 0.2, 12),
                              np.linspace(0.2, 3.0, 20)])
    be_vals = np.concatenate([np.geomspace(0.01, 0.2, 12),
                              np.linspace(0.2, 3.0, 20)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 2) massive random Dirichlet draws (same concentrations as before)
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1200
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    # 3) sparse uniform vectors – support size around 2·√n
    sqrt_n = int(np.sqrt(n))
    support_sizes = [max(2, int(k * sqrt_n)) for k in (0.8, 1.0, 1.5, 2.0, 3.0, 4.0)]
    draws_per_m = 500
    for m in support_sizes:
        for _ in range(draws_per_m):
            if time.time() >= deadline:
                break
            a = generate_sparse_uniform(n, m)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Correlation‑gradient for the smooth approximation
# ----------------------------------------------------------------------


def correlation_gradient(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g = 2 * Σ_k w_k * ∂conv[k]/∂a as a vector of length n.
    For each k, ∂conv[k]/∂a_i = 2 * a_{k-i} (zero when out of range).
    This can be written as a correlation:
        g_i = 2 * Σ_j w_{i + j} * a_j   (0 ≤ i < n)
    which equals 2 * (valid part of (w_rev * a)) reversed.
    """
    n = a.shape[0]
    L = w.shape[0]               # = 2*n-1
    size = L + n - 1
    n_fft = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    W_rev = np.fft.rfft(w[::-1], n=n_fft)
    conv = np.fft.irfft(A * W_rev, n=n_fft)   # length = size
    start = n - 1
    valid = conv[start:start + n]            # length n
    g = 2.0 * valid[::-1]
    return g


# ----------------------------------------------------------------------
# Soft‑max Adam polishing (smooth surrogate for max‑conv)
# ----------------------------------------------------------------------


def softmax_adam_polish(a0: np.ndarray, deadline: float,
                        init_tau: float = 0.1,
                        final_tau: float = 5e-5,
                        tau_decay: float = 0.9995,
                        lr: float = 0.05,
                        beta1: float = 0.9,
                        beta2: float = 0.999,
                        eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimiser on a smooth log‑sum‑exp approximation of the max‑conv.
    The variables are stored as ``u`` with ``a = softmax(u)``.
    Returns the best vector observed before the deadline.
    """
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = a0.copy()
    best_val = compute_score(best_a)

    t = 0
    tau = init_tau

    while time.time() < deadline:
        t += 1

        # ----- soft‑max projection -----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ----- convolution -----
        conv = full_convolution(a)            # length 2n‑1
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # ----- soft‑max over conv (smooth max) -----
        # use τ‑scaled softmax with numerical safety
        max_c = np.max(conv)
        scaled = (conv - max_c) / max(tau, 1e-12)
        exp_c = np.exp(scaled)
        w = exp_c / exp_c.sum()

        # ----- gradient of smooth loss -----
        g = correlation_gradient(w, a)   # shape (n,)

        # ----- mirror‑descent gradient w.r.t. u -----
        g_u = a * (g - np.dot(g, a))

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- anneal temperature -----
        tau = max(final_tau, tau * tau_decay)

        # occasional jitter to avoid stagnation
        if t % 2500 == 0:
            u += np.random.randn(n) * (lr * 0.001)

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    fallback = globals().get("height_sequence_1")

            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:

        except Exception:

    # -----------------------------------------------------------------

    baseline_deadline = deadline - 350.0   # reserve the bulk for polishing

        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------

    polish_deadline = deadline - 110.0

    # -----------------------------------------------------------------

    anneal_deadline = deadline - 5.0

    # -----------------------------------------------------------------
    if best_seq is None:

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
