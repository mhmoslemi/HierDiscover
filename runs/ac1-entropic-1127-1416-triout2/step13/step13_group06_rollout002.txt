"""
Propose a step‑function sequence with a lower upper bound.

Algorithm
---------
1.  Choose a relatively large discretisation (n = 16384).  All vectors are
    normalised to have ℓ₁‑norm = 1, because the objective is scale‑invariant.
2.  Build a small collection of “seed’’ vectors:
    *   a uniform vector,
    *   a few sparse vectors with a handful of random spikes,
    *   a Dirichlet random vector,
    *   the previously known construction (if the global `height_sequence_1`
        exists) up‑sampled to the new length.
3.  For each seed run a *smooth* mirror‑descent on the simplex:
    *   compute the auto‑convolution `c = a*a`,
    *   replace `max(c)` by a smooth approximation
        `softmax_β(c) = Σ w_k c_k` with `w_k ∝ exp(c_k/β)`.
        The temperature β is annealed from 2·10⁻² to 1·10⁻³.
    *   The exact gradient of the smoothed numerator is
        `g = 2·(a_rev  *  w)` where `a_rev` is `a` reversed and `*` is the
        ordinary aperiodic convolution.
    *   Perform an (additive) step `a ← a – η_t·g` with a diminishing step
        size `η_t = η₀/√t` and re‑project onto the probability simplex.
    *   Keep the best point (true objective, not the smoothed one) seen
        during the run.
4.  After all gradient runs keep the overall best vector and launch a short
    simulated‑annealing hill‑climb on the true objective to escape any
    remaining local minima.
5.  Return the best vector (as a plain Python list) after a final call to
    `evaluate_sequence`, which also applies the required security checks.

The whole routine respects a 1000 s wall‑clock budget, leaving a safety
margin for the final evaluation call.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, scoring, projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution of two real 1‑D arrays using a real FFT.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth mirror‑descent on the simplex
# ----------------------------------------------------------------------


def smooth_gradient_descent(start_a: np.ndarray,
                            deadline: float,
                            max_iters: int = 2000,
                            eta0: float = 0.2,
                            beta_start: float = 0.02,
                            beta_end: float = 0.001,
                            eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Minimises the true objective using a smooth approximation of the max.
    The smoothness parameter β is annealed from *beta_start* to *beta_end*.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.size

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- auto‑convolution -------------------------------------------------
        c = convolution_fft(a, a)                     # length 2n‑1

        # ----- temperature schedule -------------------------------------------
        beta = beta_start + (beta_end - beta_start) * (t - 1) / (max_iters - 1)

        # ----- soft‑max weights ------------------------------------------------
        # stable soft‑max: shift by max(c)
        shift = c - np.max(c)
        exp_c = np.exp(shift / beta)
        w = exp_c / exp_c.sum()                        # also length 2n‑1, sum = 1

        # ----- gradient of the smoothed numerator ------------------------------
        # g_i = 2 * Σ_j a_j w_{i+j}
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)          # length 3n‑2
        g = 2.0 * conv_rev[n - 1:n - 1 + n]          # take the central block

        # ----- stepsize --------------------------------------------------------
        eta = eta0 / math.sqrt(t)

        # ----- update & projection ---------------------------------------------
        a = a - eta * g
        a = project_to_simplex(a)

        # ----- occasional true‑objective evaluation -----------------------------
        if t % eval_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] iter {t:4d}  smooth‑grad score = {best_score:.6f}")

    # final check
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score, best_a = final_sc, a.copy()
        print(f"[{time.time():.1f}] smooth‑grad final improvement = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (same as previous implementation)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 150_000,
                  temp_factor: float = 1.0e-3,
                  step_scale_init: float = 0.3) -> Tuple[np.ndarray, float]:
    """
    Simple SA on the true objective; moves consist of swaps or fractional
    mass transfers between two random entries.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature proportional to current score
    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # simple linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    start_time = time.time()
    # leave a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    n = 16384  # larger discretisation than the previous run

    # ------------------------------------------------------------------
    # 0. Fallback to the previous construction (if it exists)
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` may be defined in the global scope
        fallback = np.array([float(x) for x in height_sequence_1], dtype=np.float64)  # type: ignore
        # up‑sample / down‑sample to the current length with linear interpolation
        if fallback.size != n:
            x_old = np.linspace(0.0, 1.0, fallback.size)
            x_new = np.linspace(0.0, 1.0, n)
            fallback = np.interp(x_new, x_old, fallback)
        fallback = fallback / fallback.sum()
        sc = compute_score(fallback)
        if sc < best_score:
            best_score = sc
            best_seq = fallback.tolist()
            print(f"[{time.time():.1f}] fallback score = {best_score:.6f}")
    except Exception:
        # either the name is missing or something went wrong – ignore
        pass

    # ------------------------------------------------------------------
    # 1. Build a modest set of seed vectors
    # ------------------------------------------------------------------
    seeds: List[Tuple[str, np.ndarray]] = []

    # uniform
    seeds.append(("uniform", np.full(n, 1.0 / n, dtype=np.float64)))

    # a few sparse random vectors
    for k in (5, 10, 20):
        a = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, size=k, replace=False)
        a[idx] = np.random.rand(k)
        a /= a.sum()
        seeds.append((f"sparse_{k}", a))

    # Dirichlet random vectors (lightly concentrated)
    for i in range(3):
        a = np.random.dirichlet(np.ones(n))
        seeds.append((f"dirichlet_{i}", a))

    # add the fallback if we successfully created it
    if best_seq is not None:
        seeds.append(("fallback_up", np.array(best_seq, dtype=np.float64)))

    # ------------------------------------------------------------------
    # 2. Smooth gradient descent on each seed (use ~70 % of the budget)
    # ------------------------------------------------------------------
    grad_deadline = deadline - (deadline - start_time) * 0.3   # keep ~30 % for SA

    for name, vec in seeds:
        if time.time() >= grad_deadline:
            break
        print(f"[{time.time():.1f}] start smooth‑grad on seed '{name}'")
        a_opt, sc_opt = smooth_gradient_descent(
            start_a=vec,
            deadline=grad_deadline,
            max_iters=2000,
            eta0=0.2,
            beta_start=0.02,
            beta_end=0.001,
            eval_interval=200,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after smooth‑grad ({name}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final simulated‑annealing hill climb (use the remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_start = np.array(best_seq, dtype=np.float64)
        # decide a reasonable step budget from the time that is left
        remaining = deadline - time.time()
        # roughly 150 µs per SA step on a 16k vector → 150 µs⁻¹ ≈ 6600 steps/s
        # we are generous and allow up to 150 000 steps (well within the budget)
        max_sa_steps = 150_000
        a_final, sc_final = hill_climb_sa(
            start_a=a_start,
            deadline=deadline,
            max_steps=max_sa_steps,
            temp_factor=1.0e-3,
            step_scale_init=0.3,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Fallback to a safe uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5. Final official evaluation (includes all security checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
