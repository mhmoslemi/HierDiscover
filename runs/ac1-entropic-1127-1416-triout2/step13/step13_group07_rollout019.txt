def best_shift(seq):
    n = len(seq)
    best = seq; best_score = compute_score(seq)
    for d in range(1, n//2):
        # shift left: prepend zeros
        cand = np.concatenate([np.zeros(d), seq[:n-d]])

"""
Hybrid optimiser for the step‑function bounding problem.

The algorithm works in several inexpensive stages that together explore a
large part of the feasible space within the 1000 s wall‑clock limit:

1.  Analytic families (product‑power, left‑heavy power, exponential‑U,
    a uniform–U mixture and the two‑parameter Beta family) are scanned
    on a modest size (n=2048).  The best parameters are recorded.

2.  Random Dirichlet draws with a variety of concentration parameters
    provide a dense, diverse search.  Small right‑ and left‑shifts are
    tried for each sample.

3.  The best Beta‑family parameters found in step 1 are refined by a
    simple coordinate hill‑climb (much finer than the coarse grid).

4.  A systematic shift‑search (up to half the length in both directions)
    is applied to the current best sequence – moving mass away from the
    edges often reduces the maximal convolution value.

5.  Simulated annealing (mass‑transfer moves with Metropolis acceptance)
    finishes the optimisation.  It works directly on the original
    non‑smooth objective and is able to escape local minima that a pure
    greedy hill‑climb would miss.

All stages respect the global deadline; the best sequence encountered
is finally evaluated with the provided ``evaluate_sequence`` and returned
as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using a real FFT (O(N log N)).
    Both inputs must be one‑dimensional real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    a is assumed non‑negative (the optimiser keeps it on the simplex).
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)*(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of a uniform vector and an exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Discrete Beta‑type density (p,q>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)

def hill_climb_alpha(alpha0: float, n: int, deadline: float) -> float:
    """One‑dimensional hill‑climb on the product‑family exponent."""
    best = alpha0
    best_score = compute_score(symmetric_product_family(n, best))
    step = 0.8
    while time.time() < deadline and step > 1e-4:
        improved = False
        for sign in (-1, 1):
            cand = max(0.0, best + sign * step)
            sc = compute_score(symmetric_product_family(n, cand))
            if sc < best_score:
                best_score = sc
                best = cand
                improved = True
                print(f"[{time.time():.1f}] alpha refined → {best:.6f} score {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best


def hill_climb_beta(p0: float, q0: float, n: int, deadline: float) -> Tuple[float, float]:
    """Two‑dimensional hill‑climb on the Beta‑family parameters."""
    best_p, best_q = p0, q0
    best_score = compute_score(beta_family(n, best_p, best_q))
    step = 0.4
    while time.time() < deadline and step > 1e-3:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(0.01, best_p + dp)
                cand_q = max(0.01, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_score = sc
                    best_p, best_q = cand_p, cand_q
                    improved = True
                    print(f"[{time.time():.1f}] Beta refined → p={best_p:.6f} q={best_q:.6f} score {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
    return best_p, best_q

def shift_search(seq: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Try all left/right cyclic‑zero shifts up to n//2.
    Returns the best shifted version.
    """
    n = seq.size
    best_seq = seq.copy()
    best_score = compute_score(seq)

    max_shift = n // 2
    for d in range(1, max_shift + 1):
        if time.time() >= deadline:
            break

        # shift left (prepend zeros)
        left = np.concatenate([np.zeros(d), seq[: n - d]])
        left = normalize(left)
        sc_left = compute_score(left)
        if sc_left < best_score:
            best_score, best_seq = sc_left, left
            print(f"[{time.time():.1f}] left‑shift {d} improved to {best_score:.6f}")

        # shift right (append zeros)
        right = np.concatenate([seq[d:], np.zeros(d)])
        right = normalize(right)
        sc_right = compute_score(right)
        if sc_right < best_score:
            best_score, best_seq = sc_right, right
            print(f"[{time.time():.1f}] right‑shift {d} improved to {best_score:.6f}")

    return best_seq, best_score

def simulated_annealing(start_a: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing.
    At each step a random donor index loses a fraction of its mass
    which is transferred to a random receiver.  Metropolis acceptance
    with a slowly cooling temperature allows uphill moves.
    """
    a = start_a.copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]

    # temperature schedule – start fairly high, decay exponentially
    temp = 0.02
    cooling = 0.99995
    # typical transfer magnitude (as a fraction of donor mass)
    step_scale = 0.12

    it = 0
    while time.time() < deadline:
        it += 1
        # pick donor and receiver (different)
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # propose a transfer; ensure we never empty the donor completely
        delta = step_scale * a[i] * np.random.rand()
        if delta <= 0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        dE = cand_score - cur_score

        if dE < 0 or np.random.rand() < math.exp(-dE / max(temp, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time():.1f}] SA step {it} new best {best_score:.6f}")

        # temperature decay
        temp *= cooling
        # occasional adaptation of step size
        if it % 5000 == 0:
            # keep acceptance rate around 0.3‑0.4 – crude adaptation
            # (here we simply tighten step_scale a little)
            step_scale = max(0.02, step_scale * 0.99)

    return best_a, best_score
