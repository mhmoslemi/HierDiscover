def anneal_hill_climb(start_a, deadline):
    a = start_a.copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    temperature = 0.5
    # schedule: temperature decays exponentially.
    decay = 0.9995

    while time.time() < deadline:
        # pick two random indices i and j.
        i, j = np.random.randint(len(a), size=2)
        if i == j: continue
        # propose moving epsilon fraction from i to j
        eps = temperature * a[i] # max possible delta
        delta = eps * np.random.rand()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        new_score = compute_score(cand)
        delta_score = new_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / (temperature+1e-12)):
            a = cand
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        temperature *= decay
        if temperature < 1e-6: temperature = 1e-6
    return best_a, best_score

"""
Propose a refined step‑function sequence that minimises the competition
objective

    2·n·max(conv(a, a)) / (sum(a)²)

The search uses several complementary stages within the 1000 s wall‑clock
budget:

1. **Parameterised families** – systematic grid scans and random draws from
   a handful of analytic families (product‑power, two‑parameter product,
   sum‑power, left‑heavy power, exponential‑U, mixtures).

2. **Adam optimisation** – smooth‑max (log‑sum‑exp) surrogate of the max‑
   convolution, with a slowly increasing temperature.  Symmetry is enforced
   regularly (averaging the vector with its reversal) and the iterate is
   projected onto the probability simplex after each symmetry step.

3. **Projected sub‑gradient descent** on the exact max‑objective.

4. **Simulated‑annealing hill‑climb** – random mass transfers between
   entries, accepting worsening moves with a temperature schedule.

All stages respect a common deadline; the best sequence found is returned
as a plain ``list[float]``.  The implementation relies only on NumPy and
the supplied ``evaluate_sequence`` for the final score.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact competition objective.
    a is assumed non‑negative; the caller should ensure sum(a)>0.
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (sum = 1, v ≥ 0).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def two_param_product_family(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑p}·(n‑½‑i)^{‑q}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-p * np.log(i + 0.5) - q * np.log(n - 0.5 - i))
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_product_family(n: int, lam: float, alpha: float) -> np.ndarray:
    """Convex mixture of uniform and product‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    prod = symmetric_product_family(n, alpha)
    w = lam * uniform + (1.0 - lam) * prod
    return normalize(w)


def sum_power_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑p} + (n‑½‑i)^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-p * np.log(i + 0.5)) + np.exp(-p * np.log(n - 0.5 - i))
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 2.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.02,
                         sym_interval: int = 10) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    The temperature (beta) is increased geometrically from ``beta_start``
    to ``beta_end``.  Symmetry (a↔a[::-1]) is enforced every ``sym_interval``
    iterations and the iterate is projected onto the simplex.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise in log‑space to keep positivity
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the log‑softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true objective check
        if it % 3000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.05) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def anneal_hill_climb(start_a: np.ndarray,
                      deadline: float) -> Tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb.
    At each step a random mass transfer between two positions is proposed.
    Worsening moves are accepted with probability exp(-Δ / T) where the
    temperature T decays geometrically.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.5                # initial temperature
    decay = 0.9995         # decay per step

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose moving a fraction of the donor mass
        delta_max = cur_a[i]
        delta = T * delta_max * np.random.rand()
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        diff = cand_score - cur_score
        if diff < 0 or np.random.rand() < math.exp(-diff / (T + 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
        T *= decay
        if T < 1e-6:
            T = 1e-6

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` under a fixed wall‑clock budget.
    Returns the best sequence found as a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # Fallback: use any previously known sequence if it exists
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = np.array([float(x) for x in height_sequence_1],
                                dtype=np.float64)  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # Working length – 1024 gives a good trade‑off between speed
    # and expressive power.
    # -----------------------------------------------------------
    n = 1024

    # small helper to test a candidate and keep the best one
    def try_seq(v: np.ndarray):
        nonlocal best_seq, best_score
        sc = compute_score(v)
        if sc < best_score:
            best_score = sc
            best_seq = v.copy()
            print(f"[{time.time():.1f}] new best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 1. Coarse deterministic scan of analytic families
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] starting coarse scan (n={n})")

    # product family
    for alpha in np.linspace(0.0, 6.0, 25):
        if time.time() >= deadline - 30:
            break
        try_seq(symmetric_product_family(n, alpha))

    # two‑parameter product family
    for p in np.linspace(0.1, 2.0, 7):
        for q in np.linspace(0.1, 2.0, 7):
            if time.time() >= deadline - 30:
                break
            try_seq(two_param_product_family(n, p, q))

    # sum‑power family
    for p in np.linspace(0.1, 4.0, 17):
        if time.time() >= deadline - 30:
            break
        try_seq(sum_power_family(n, p))

    # left‑heavy power family
    for beta in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30:
            break
        try_seq(left_heavy_power_family(n, beta))

    # exponential‑U family
    for gamma in np.linspace(0.0, 4.0, 21):
        if time.time() >= deadline - 30:
            break
        try_seq(exp_u_family(n, gamma))

    # mixture uniform + product family
    for lam in np.linspace(0.0, 1.0, 5):
        for alpha in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30:
                break
            vec = mixed_uniform_product_family(n, lam, alpha)
            try_seq(vec)

    # -----------------------------------------------------------
    # 2. Random dense search over families (≈2000 trials)
    # -----------------------------------------------------------
    trials = 0
    max_trials = 2000
    while time.time() < deadline - 30 and trials < max_trials:
        trials += 1
        choice = np.random.randint(6)
        if choice == 0:   # product
            alpha = np.random.uniform(0.0, 8.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 1: # two‑parameter product
            p = np.random.uniform(0.0, 4.0)
            q = np.random.uniform(0.0, 4.0)
            vec = two_param_product_family(n, p, q)
        elif choice == 2: # sum‑power
            p = np.random.uniform(0.0, 6.0)
            vec = sum_power_family(n, p)
        elif choice == 3: # left‑heavy power
            beta = np.random.uniform(0.0, 8.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 4: # exponential‑U
            gamma = np.random.uniform(0.0, 5.0)
            vec = exp_u_family(n, gamma)
        else:             # mixture uniform + product
            lam = np.random.uniform(0.0, 1.0)
            alpha = np.random.uniform(0.0, 8.0)
            vec = mixed_uniform_product_family(n, lam, alpha)

        # occasional random right‑shift
        if np.random.rand() < 0.2:
            d = np.random.randint(0, n // 30 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        try_seq(vec)

    # -----------------------------------------------------------
    # 3. Refined Adam optimisation on a few top candidates
    # -----------------------------------------------------------
    # Gather a small pool of promising starts
    candidates: List[np.ndarray] = []
    if best_seq is not None:
        candidates.append(best_seq.copy())
    # add a few noisy perturbations of the current best
    if best_seq is not None:
        for _ in range(3):
            noise = np.random.normal(scale=0.02, size=n)
            cand = best_seq + noise
            cand = normalize(cand)
            candidates.append(cand)

    for idx, cand in enumerate(candidates):
        if time.time() >= deadline - 20:
            break
        print(f"[{time.time():.1f}] Adam optimisation on candidate {idx}")
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=cand,
            deadline=deadline - 10,
            max_iters=600_000,
            beta_start=2.0,
            beta_end=5000.0,
            lr=0.02,
            sym_interval=10,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Projected sub‑gradient descent on the true max‑objective
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        print(f"[{time.time():.1f}] Sub‑gradient descent phase")
        a_sub, sc_sub = subgrad_optimize(
            start_a=best_seq,
            deadline=deadline - 5,
            max_iters=2_000_000,
            step0=0.05,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.1f}] Sub‑grad improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Simulated‑annealing greedy hill‑climb
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print(f"[{time.time():.1f}] Simulated‑annealing hill‑climb")
        a_final, sc_final = anneal_hill_climb(best_seq, deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Annealed hill‑climb improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Final safety fallback
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Fallback uniform sequence score = {best_score:.6f}")

    # Final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the competition harness
    return best_seq.tolist()
