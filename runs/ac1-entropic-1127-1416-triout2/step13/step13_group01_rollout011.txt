"""
search for a non‑negative step‑function that minimizes
evaluate_sequence(sequence)

Approach
--------
1. **Problem size** – Work with n = 4096 coefficients (larger than the
   previous 2048).  The objective 2·n·max(conv(a,a)) is invariant to
   scaling, so we keep the vector normalised to sum = 1.

2. **Global exploration** – Sweep several parametric families
   (power‑law, exponential, beta, Gaussian) and many Dirichlet draws.
   The best few candidates are retained.

3. **Two‑phase smooth optimisation**  
   *Phase A* – minimise the true max‑convolution by a smooth
   approximation (soft‑max with an annealing temperature τ).  This is
   essentially the same Adam‑based method used before, but with a
   larger temperature‑decay factor and a slightly higher learning rate
   to explore more aggressively.  
   *Phase B* – starting from the result of Phase A, minimise a *mean‑square‑error*
   against the uniform distribution (μ = 1/(2·n‑1)).  Driving the
   convolution towards uniformity reduces the peak further.

   Both phases operate on the simplex via the log‑softmax
   parametrisation (x → a = softmax(x)).  Gradients are computed with
   FFT‑based convolutions for speed.

4. **Final annealing** – a short Metropolis‑style simulated‑annealing
   pass to escape any tiny local plateaux.

All work respects the 1000 s wall‑clock budget.  The best sequence
found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real FFT, works for any two real arrays)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (used for the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (no clipping – all entries already non‑negative)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))  (sum(a) is assumed to be 1).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global baseline exploration – keep up to `max_keep` best vectors
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Sweep families and many Dirichlet draws, retain the best `max_keep`."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        # ensure non‑negative, normalised to simplex
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- power‑law ----------------------------------------------------
    for p in np.linspace(0.05, 3.0, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # ---- exponential -------------------------------------------------
    for lam in np.linspace(0.00005, 0.07, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # ---- beta ---------------------------------------------------------
    al_vals = np.logspace(-2, np.log10(5.0), 30)  # from 0.01 to 5
    be_vals = np.logspace(-2, np.log10(5.0), 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # ---- Gaussian -----------------------------------------------------
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 35)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # ---- Dirichlet draws -----------------------------------------------
    concs = [0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 4.0]
    draws_per_conc = 1500
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Softmax helper (log‑space parametrisation)
# ----------------------------------------------------------------------


def softmax_from_x(x: np.ndarray) -> np.ndarray:
    """Map real vector x to the probability simplex."""
    max_x = np.max(x)
    e = np.exp(x - max_x)
    return e / e.sum()


# ----------------------------------------------------------------------
# Phase‑A : smooth max‑convolution optimisation (Adam on log‑softmax)
# ----------------------------------------------------------------------


def smooth_optimize_max(a0: np.ndarray, deadline: float,
                        init_lr: float = 0.12,
                        init_tau_factor: float = 15.0,
                        min_tau: float = 1e-10,
                        tau_decay: float = 0.9997) -> np.ndarray:
    """
    Adam optimisation of the *soft‑max* approximation of max(conv(a,a)).
    Returns the best vector observed before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise log‑softmax parameters
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initial temperature (smoothness)
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- softmax mapping -----
        a = softmax_from_x(x)

        # ----- convolution of current a -----
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- smooth max weights (soft‑max over conv values) -----
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient w.r.t. a -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)  # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- mirror‑descent (log‑space) gradient -----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature -----
        tau = max(min_tau, tau * tau_decay)

        # ----- occasional exact evaluation -----
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth‑max] iter {it:6d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Phase‑B : minimise mean‑square error to the uniform convolution
# ----------------------------------------------------------------------


def smooth_optimize_mse(a0: np.ndarray, deadline: float,
                        init_lr: float = 0.10,
                        mu: float = None) -> np.ndarray:
    """
    Adam optimiser that minimises ½·∑_k (conv_k – μ)²,
    where μ = 1/(2·n – 1) is the uniform target.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]
    if mu is None:
        mu = 1.0 / (2 * n - 1)

    # initialise log‑softmax parameters
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        a = softmax_from_x(x)

        conv = full_convolution(a)
        err = conv - mu                 # shape (2n‑1,)

        # gradient w.r.t. a : 2·conv_rev * err
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, err)
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional check of the true max‑convolution score
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth‑mse] iter {it:6d}  score {val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # occasional jitter to avoid stagnation
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin for the outer harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 4096 gives more flexibility than the earlier 2048
    # -----------------------------------------------------------------
    n = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously stored solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 150 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 600.0          # reserve bulk for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=8)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Phase‑A : smooth max‑convolution optimisation (≈ 250 s)
    # -----------------------------------------------------------------
    phase_a_deadline = deadline - 300.0
    if best_seq is not None and time.time() < phase_a_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize_max(a0, phase_a_deadline)
        val = compute_score(refined)
        print(f"[phase‑A] refined max‑score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # a few random restarts while time permits
        while time.time() < phase_a_deadline:
            remaining = phase_a_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)
            conc = np.random.choice([0.05, 0.1, 0.3, 0.7, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize_max(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[phase‑A‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Phase‑B : MSE‑to‑uniform optimisation (≈ 200 s)
    # -----------------------------------------------------------------
    phase_b_deadline = deadline - 80.0
    if best_seq is not None and time.time() < phase_b_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize_mse(a0, phase_b_deadline)
        val = compute_score(refined)
        print(f"[phase‑B] refined (mse) score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # a couple of extra restarts with different initializations
        while time.time() < phase_b_deadline:
            remaining = phase_b_deadline - time.time()
            sub_deadline = time.time() + min(20.0, remaining)
            a_start = np.random.dirichlet(np.full(n, 0.5))
            refined = smooth_optimize_mse(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[phase‑B‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Report the official evaluation (uses the provided wrapper)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
