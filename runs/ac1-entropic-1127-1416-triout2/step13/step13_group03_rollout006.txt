"""
Improved non‑negative step‑function optimiser.

What it does
------------
1. **Broad random exploration** – many Dirichlet draws (wide concentration
   range), parametric families (power‑law, exponential, beta, Gaussian,
   sine‑power) and random permutations of those vectors.  The 30 best
   candidates are kept.

2. **Differential‑Evolution (DE)** – a simple DE/rand/1 scheme works directly
   on the simplex (mutants are projected back).  DE quickly discovers
   diverse, asymmetric shapes that a pure gradient method can miss.

3. **Smooth‑max Adam refinement** – the best DE individual is polished with
   the Adam optimiser on the smooth‑max surrogate (τ·log∑exp(conv/τ)).
   The temperature anneals to almost the true max, so the final vector is
   very close to a local optimum of the original problem.

4. **Projected sub‑gradient polish** – a final exact sub‑gradient descent
   on the true max‑objective eliminates any remaining slack.

All phases respect the 1000 s wall‑clock budget.  The returned list of
floats is a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: ensure exact sum==1
    w /= w.sum() if w.sum() > 0 else 1.0
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length power‑of‑2."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Baseline global exploration (wide‑range random draws)
# ----------------------------------------------------------------------


def baseline_exploration(
    n: int,
    deadline: float,
    keep: int = 30,
    dirichlet_draws: int = 12000,
) -> List[np.ndarray]:
    """Return up to ``keep`` best vectors from a very broad random search."""
    best: List[Tuple[float, np.ndarray]] = []

    def consider(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- parametric families ------------------------------------------------
    # power‑law
    for p in np.linspace(0.1, 5.0, 41):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # exponential
    for lam in np.linspace(0.0001, 0.5, 101):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # beta (grid + random)
    al_grid = np.linspace(0.02, 2.0, 31)
    be_grid = np.linspace(0.02, 2.0, 31)
    for a in al_grid:
        for b in be_grid:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    for _ in range(200):
        if time.time() >= deadline:
            break
        a = 10 ** np.random.uniform(-3, 0.5)   # 0.001 … 3.16
        b = 10 ** np.random.uniform(-3, 0.5)
        consider(generate_beta(n, a, b))

    # Gaussian
    for sigma in np.linspace(0.1 * n, 0.45 * n, 25):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # sine‑power
    for p in np.linspace(0.1, 8.0, 31):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # ---- Dirichlet draws + random permutations -----------------------------
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    per_conc = max(1, dirichlet_draws // len(concs))
    for conc in concs:
        for _ in range(per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            consider(a)
            # also try a random permutation (breaks symmetry)
            np.random.shuffle(a)
            consider(a)

    # ---- Mixed families ----------------------------------------------------
    for _ in range(300):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-3, 0.5),
            10 ** np.random.uniform(-3, 0.5),
        )
        lam = np.random.rand()
        consider(mixture(a, b, lam))

    # Return only the vectors (sorted by score)
    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max surrogate of max(conv)
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.15,
    tau_end: float = 1e-6,
    tau_decay: float = 0.93,
    lr_start: float = 0.10,
    lr_decay: float = 0.985,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate:
           fτ(a) = τ·log Σ_k exp(conv_k(a)/τ)

    Returns the best vector (projected onto the simplex) seen before deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        # anneal temperature
        if t % 250 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # smooth‑max weights
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient of τ·log Σ exp = Σ w * ∂conv/∂a
        # ∂conv/∂a = 2 * (a ⋆ w)  (cross‑correlation)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        corr = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = project_to_simplex(a)

        cur = compute_score(a)
        if cur < best_val:
            best_val, best_a = cur, a.copy()

        # anneal learning‑rate
        if t % 500 == 0:
            lr = max(lr * lr_decay, 1e-4)

        # occasional jitter to escape plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.002 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent directly on max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # Sub‑gradient is sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur = compute_score(a)
        if cur < best_val:
            best_val, best_a = cur, a.copy()

        if t % 4000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Differential‑Evolution on the simplex
# ----------------------------------------------------------------------


def differential_evolution(
    init_pop: List[np.ndarray],
    deadline: float,
    F: float = 0.78,
    popsize: int = 40,
) -> Tuple[np.ndarray, float]:
    """
    Simple “DE/rand/1” optimiser.  All individuals live on the simplex.
    Returns the best vector found and its score.
    """
    n = init_pop[0].shape[0]
    # ensure we have exactly popsize individuals
    pop = init_pop[:popsize]
    while len(pop) < popsize:
        pop.append(project_to_simplex(np.random.rand(n)))
    scores = [compute_score(v) for v in pop]

    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(popsize):
            # pick three distinct others
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            trial = project_to_simplex(mutant)

            trial_score = compute_score(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score, best_vec = trial_score, trial.copy()

            # early exit if deadline is near
            if time.time() >= deadline:
                break
        # small random shake of the whole population every few generations
        if np.random.rand() < 0.08 and time.time() < deadline:
            for i in range(popsize):
                noise = np.random.randn(n) * 0.004
                pop[i] = project_to_simplex(pop[i] + noise)

    return best_vec, best_score
