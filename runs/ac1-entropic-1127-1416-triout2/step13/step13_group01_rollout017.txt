def project_to_simplex(v):
    # ensure nonnegative and sum to 1
    # Use Duchi method for arbitrary v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def projected_gradient_descent(a0, deadline, max_iter=5000, step_init=0.1, step_decay=0.99):

    a = a0.copy()
    step = step_init
    best_a = a.copy()
    best_val = compute_score(a)
    n = len(a)
    while time.time() < deadline and max_iter > 0:
         conv = full_convolution(a)
         max_c = conv.max()
         k = conv.argmax()
         # compute gradient for that max conv
         # gradient of max conv (approx) is 2*a[k-i] for valid i
         grad = np.zeros_like(a)
         lo = max(0, k - (n-1))
         hi = min(k, n-1)
         idx = np.arange(lo, hi+1)
         grad[idx] = 2 * a[k - idx]
         # direction to reduce max, negative gradient
         descent = -grad
         # step
         a_new = a + step * descent
         # project onto simplex
         a_new = project_to_simplex(a_new)
         new_val = compute_score(a_new)
         if new_val < best_val:
             best_val = new_val
             best_a = a_new.copy()
         # acceptance policy
         if new_val < compute_score(a):
             a = a_new
         else:
             # maybe reject but reduce step
             step *= 0.5
         step *= step_decay
         max_iter -= 1
    return best_a

def flatten_opt(a0, deadline, step=0.1, decay=0.99):
    a = a0.copy()
    n = len(a)
    target_val = 1.0 / (2 * n - 1)
    target = np.full(2 * n - 1, target_val)
    best_a = a.copy()

            step *= 0.7
        step *= decay
        a = a_new
    return best_a

"""
search_step_function.py

A heuristic optimizer for the “max‑convolution” objective used in
`evaluate_sequence`.  The algorithm consists of three phases:

1. **Heavy random sampling** – many Dirichlet draws (with several
   concentrations) and a few classical parametric families.  The best
   few samples are kept.

2. **Flatten‑convolution gradient descent** – we minimise the
   L₂‑distance between the auto‑convolution `a*a` and the perfectly
   flat target `1/(2n‑1)`.  The gradient of the loss is
   `2·conv(a_rev, conv−target)`.  After each gradient step the vector
   is projected onto the probability simplex (non‑negative, sum = 1).

   This descent rapidly reduces the peak of the convolution and
   therefore the evaluation score.

3. **Optional simulated‑annealing polish** – a short Metropolis walk
   to escape tiny local plateaux.

The whole search respects the 1000 s wall‑clock budget; the last 5 s
are left as a safety margin for the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real 1‑D arrays via FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())        # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (used for the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (no clipping – all entries already non‑negative)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact score 2·n·max(conv(a,a)).  The sum of a is 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto {x : x≥0, sum(x)=1}.
    Return a new array with the same shape as v.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        # all entries are non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / n, dtype=float)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for a cheap global sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Heavy random baseline (keeps a few best vectors)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 6):
    """
    Random Dirichlet draws + a quick sweep of the classical families.
    Returns a list with up to `max_keep` vectors that have the smallest
    compute_score.
    """
    best = []  # list of (score, vector)

    def maybe_insert(vec: np.ndarray):
        """Insert vec into the best list if it improves the current top‑k."""
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) parametric families ------------------------------------------------
    for p in np.linspace(0.1, 2.0, 150):
        if time.time() >= deadline:
            break
        maybe_insert(generate_powerlaw(n, p))

    for lam in np.linspace(0.0001, 0.05, 200):
        if time.time() >= deadline:
            break
        maybe_insert(generate_exponential(n, lam))

    al_vals = np.linspace(0.2, 5.0, 20)
    be_vals = np.linspace(0.2, 5.0, 20)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            maybe_insert(generate_beta(n, alpha, beta))

    sigma_vals = np.linspace(n / 32.0, n / 2.0, 25)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        maybe_insert(generate_gaussian(n, sigma))

    # 2) many Dirichlet draws -----------------------------------------------
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 5000
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            maybe_insert(a)

    # return only the vectors (sorted by ascending score)
    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Gradient descent on the L2 distance between the convolution and a flat target
# ----------------------------------------------------------------------


def flatten_convolution_opt(a0: np.ndarray, deadline: float,
                            init_step: float = 0.25,
                            step_decay: float = 0.995,
                            min_step: float = 1e-6) -> np.ndarray:
    """
    Minimise ½·‖a*a – target‖₂² where target is a flat distribution.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    target_val = 1.0 / (2 * n - 1)
    target = np.full(2 * n - 1, target_val, dtype=float)

    best_a = a.copy()
    best_val = compute_score(a)

    step = init_step
    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        diff = conv - target                     # size 2n‑1
        # gradient w.r.t. a : 2·conv(a_rev, diff)
        grad_full = _linear_convolution(a[::-1], diff)
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # take a small step and project back onto the simplex
        a_new = a - step * grad_a
        a_new = project_to_simplex(a_new)

        val = compute_score(a_new)

        if val < best_val:
            best_val, best_a = val, a_new.copy()
            # if we improve we can be a bit bolder
            step = min(step * 1.05, 1.0)
            print(f"[flatten_opt] iter {it:6d}  score {val:.6f}  step {step:.2e}")

        # simple accept‑if‑better policy (otherwise keep old a)
        if val < compute_score(a):
            a = a_new

        # adapt step size
        if val >= best_val * 0.9999:      # stagnating
            step = max(step * 0.7, min_step)
        else:
            step = max(step * step_decay, min_step)

        # safeguard – never let step become absurdly small
        if step < min_step:
            step = min_step

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (optional final kick)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling: float = 0.9995,
                        step_scale: float = 0.02) -> np.ndarray:
    """Very light Metropolis hill‑climb – may escape tiny plateaux."""
    a = a_start.copy()
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val
    T = init_temp
    it = 0
    n = a.shape[0]

    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        if val < cur_val or np.random.rand() < np.exp(-(val - cur_val) / max(T, 1e-12)):
            a, cur_val = cand, val
            if val < best_val:
                best_val, best_a = val, cand.copy()
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            step_scale = max(step_scale * 0.98, 1e-6)
        T = max(final_temp, T * cooling)

        if it % 4000 == 0:
            # occasional jitter to keep the walk alive
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 gives a good trade‑off between expressivity
    # and speed.
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Heavy random baseline (≈ 250 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 350.0   # reserve the rest for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=6)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Flatten‑convolution gradient descent (≈ 600 s)
    # -----------------------------------------------------------------
    optimize_deadline = deadline - 120.0
    if best_seq is not None and time.time() < optimize_deadline:
        # a few restarts – the first from the current best
        starts = [np.array(best_seq, dtype=float)]
        # additional random starts (different Dirichlet concentrations)
        for _ in range(4):
            conc = np.random.choice([0.1, 0.3, 0.7, 1.5, 3.0])
            starts.append(np.random.dirichlet(np.full(n, conc)))

        for a0 in starts:
            if time.time() >= optimize_deadline:
                break
            sub_deadline = time.time() + min(120.0, optimize_deadline - time.time())
            refined = flatten_convolution_opt(a0, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[flatten‑opt] improved to {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
