"""
Search for a non‑negative coefficient sequence that minimises
``evaluate_sequence`` (the max‑convolution bound).
The algorithm performs

1. **Broad random sampling** (Dirichlet draws, random monotone vectors).
2. **Fine parametric sweeps** for the exponential and power‑law families
   (dense 1‑D searches on the scale parameters).
3. **Local refinement** by Adam optimisation on a soft‑max (smooth max) of
   the convolution, with an annealed temperature.
4. **Short simulated‑annealing polish** to escape possible plateaus.

All stages stop when the wall‑clock budget (≈ 1000 s) is exhausted.
The final list is returned in plain Python format.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (used in the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1 – the scale is irrelevant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact value 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce vectors summing to 1)
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_monotone_random(n: int) -> np.ndarray:
    """Draw a random vector, sort it decreasingly, and normalise."""
    v = np.random.exponential(scale=1.0, size=n)
    a = np.sort(v)[::-1]               # decreasing
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Simple bookkeeping utilities
# ----------------------------------------------------------------------


def _try_candidate(vec: np.ndarray, best):
    """Update the (score, vector) list ``best`` if ``vec`` improves it."""
    if not np.isfinite(vec).all():
        return
    # safety – clip negatives, renormalise
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0:
        return
    vec = vec / s
    sc = compute_score(vec)
    # keep at most 12 best candidates (small enough for later polishing)
    if len(best) < 12:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Global baseline exploration (wide random search)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float):
    """Collect a set of diverse promising candidates."""
    best = []   # list of (score, vector) – smallest scores first

    # 1) Random Dirichlet draws (various concentrations)
    alphas = [0.1, 0.3, 0.7, 1.0, 2.0, 4.0]
    draws_per_alpha = 2000
    for conc in alphas:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_alpha):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            _try_candidate(a, best)

    # 2) Random monotone vectors (sorted exponentials)
    for _ in range(4000):
        if time.time() >= deadline:
            break
        a = generate_monotone_random(n)
        _try_candidate(a, best)

    # 3) Coarse beta family sweep (helps discover asymmetric shapes)
    al_vals = np.linspace(0.2, 8.0, 12)
    be_vals = np.linspace(0.2, 8.0, 12)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            a = generate_beta(n, alpha, beta)
            _try_candidate(a, best)

    # 4) Gaussian family sweep (different widths)
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 25)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        a = generate_gaussian(n, sigma)
        _try_candidate(a, best)

    # Return just the vectors (sorted by score)
    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Fine 1‑D parameter sweeps for the two most promising families
# ----------------------------------------------------------------------


def fine_exponential_search(n: int, deadline: float):
    """Dense scan of the exponential family around the empirically good region."""
    best_sc = float("inf")
    best_vec = None
    # region around 1e‑3 (the known sweet‑spot). Use ~3000 points for high resolution.
    lam_vals = np.linspace(0.0005, 0.0020, 3000)
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        a = generate_exponential(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    return best_sc, best_vec


def fine_powerlaw_search(n: int, deadline: float):
    """Dense scan of the power‑law family (exponents 0.05 … 4)."""
    best_sc = float("inf")
    best_vec = None
    p_vals = np.linspace(0.05, 4.0, 2500)
    for p in p_vals:
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
    return best_sc, best_vec


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.08,
                    init_tau_factor: float = 5.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9997) -> np.ndarray:
    """
    Adam on the soft‑max variables ``x`` (so that a = softmax(x)).
    The smooth max uses a temperature ``tau`` that is gradually annealed.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise logits safely (avoid log(0))
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise temperature from the current max‑conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping ----------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution ----------
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- soft‑max weighting of the max ----------
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient wrt a (2·∑_k w_k·a_{k−i}) ----------
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)    # length = 3 n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for the logits ----------
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal temperature ----------
        tau = max(min_tau, tau * tau_decay)

        # ----- occasional exact check ----------
        if it % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth] it {it:7d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            # occasional jitter to avoid stagnation
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    # give a small safety margin for the harness

    # -----------------------------------------------------------------
    n = 2048

    best_val = float("inf")               # based on ``compute_score`` (sum=1)

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] official score = {val:.6f}")
            # for internal comparison we store the *compute_score* value:
            sc = compute_score(np.array(fb) / np.sum(fb))
            if sc < best_val:
                best_val, best_seq = sc, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Broad random baseline (≈ 200 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 600.0          # leave 10 min for fine & polish
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
                print(f"[baseline] new best (random) = {compute_score(vec):.6f}")

    # -----------------------------------------------------------------
    # 2. Focused 1‑D sweeps (exponential → power‑law) (≈ 150 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    if time.time() < sweep_deadline:
        # exponential sweep
        sc, vec = fine_exponential_search(n, sweep_deadline)
        if sc is not None and sc < best_val:
            best_val, best_seq = sc, vec.tolist()
            print(f"[exp‑sweep] new best = {sc:.6f}")

        # power‑law sweep
        if time.time() < sweep_deadline:
            sc, vec = fine_powerlaw_search(n, sweep_deadline)
            if sc is not None and sc < best_val:

    # -----------------------------------------------------------------
    # 3. Smoothed Adam optimisation (≈ 200 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 150.0

            # start from a fresh Dirichlet draw (mid‑concentration)
            conc = np.random.choice([0.3, 0.7, 1.5, 3.0])

            refined = smooth_optimize(a_start, sub_deadline)
            sc = compute_score(refined)
            if sc < best_val:

                print(f"[smooth‑restart] new best = {sc:.6f}")

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (≈ 100 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:

        final = simulated_annealing(a_start, anneal_deadline)

    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official reported score = {final_score:.6f}")
