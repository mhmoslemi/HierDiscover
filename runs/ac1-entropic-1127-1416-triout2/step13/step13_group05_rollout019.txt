# Helper utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law), p ≥ 0.
    """
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def mixed_reverse_uniform(n: int, mix: float, p: float) -> np.ndarray:
    """
    convex combination   mix * reverse_power_law(p) + (1‑mix) * uniform,
    0 ≤ mix ≤ 1, p ≥ 0.
    """
    if not (0.0 <= mix <= 1.0):
        return None
    rp = reverse_power_law(n, p)
    if rp is None:
        return None
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * rp + (1.0 - mix) * uniform
    a = project_to_simplex(a)
    return a

# Sub‑gradient polishing (same idea as the reference)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # gradient w.r.t. a is the sum of shifted copies of a
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 1500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# Stochastic hill‑climb with occasional restarts
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 200_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size and restarts."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]                 # move a fraction of the source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1

        it += 1
        if no_improve > max_no_improve:
            # restart from current best with a fresh jitter
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

    # -------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
