Pseudo:

def project_to_simplex(v):
    # Returns projection onto simplex {x: sum x = 1, x >= 0}
    # Using algorithm by Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid search for a non‑negative step‑function minimizing ``evaluate_sequence``.
---------------------------------------------------------------------------
Algorithm
---------

1. **Global exploration** – sweep a few classical parametric families
   (power‑law, exponential, beta, Gaussian) and many random Dirichlet draws.
   Keep the best few candidates (they already give a good starting point).

2. **Projected sub‑gradient descent** – minimise the true objective
   ``max(conv(a,a))`` on the probability simplex.
   * The sub‑gradient of ``max(conv)`` is computed analytically:
       grad_i = 2 * Σ_{k ∈ argmax(conv)} a_{k‑i}.
   * After a gradient step the iterate is projected onto the simplex
     (Euclidean projection, O(n log n)).
   * A diminishing step‑size (α_t = α0 / √t) and a simple time‑budget
     controller are used.

3. **Final simulated‑annealing polish** – a short Metropolis hill‑climb
   that can escape tiny plateaux.

The whole routine respects the 1000 s wall‑clock limit; the final
sequence is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (used for the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (no clipping – all entries already non‑negative)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are negative or zero – return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the quick global sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, deadline: float,
                        init_step: float = 0.2,
                        eps_tol: float = 1e-12) -> np.ndarray:
    """
    Projected sub‑gradient descent for  max(conv(a,a)) on the simplex.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- current convolution -----
        conv = full_convolution(a)
        max_val = conv.max()
        # mask of (approximately) active constraints
        mask = (conv >= max_val - eps_tol).astype(float)

        # ----- sub‑gradient w.r.t. a (g_i = 2·∑_k mask_k·a_{k‑i}) -----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, mask)          # length 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- step size (diminishing) -----
        step = init_step / np.sqrt(it)

        # ----- gradient step + simplex projection -----
        a_new = a - step * grad_a
        a = project_to_simplex(a_new)

        # ----- evaluate and keep the best -----
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 2000 == 0:
                print(f"[subgrad] iter {it:6d}  score {val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # occasional jitter to escape shallow plateaux
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Global (fast) baseline sweep – keeps a few best vectors
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 5):
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law
    for p in np.linspace(0.1, 2.0, 150):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential
    for lam in np.linspace(0.0001, 0.05, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta
    al_vals = np.linspace(0.2, 5.0, 15)
    be_vals = np.linspace(0.2, 5.0, 15)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # Gaussian
    sigma_vals = np.linspace(n / 24.0, n / 4.0, 12)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # Dirichlet draws
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 800
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _sc, vec in best]
