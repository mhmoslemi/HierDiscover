def smooth_max_optimize(a_init, time_deadline):
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta1=0.9; beta2=0.999; eps=1e-8
    lr0 = 0.2 # initial learning rate.
    alpha = 5.0

    while time.time() < time_deadline:
        # compute conv
        conv = full_convolution(a)
        max_val = conv.max()
        # compute weights
        # Stabilize
        max_conv = max_val
        # Use scaling factor to ensure numerical stability
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        w = w / w_sum

        # compute gradient = 2 * cross-correlation of a and w
        a_rev = a[::-1]
        conv_rev = np.convolve(a_rev, w)  # full
        grad = 2.0 * conv_rev[len(w)-1 : len(w)-1 + len(a)]  # check indices: len(w)-1 = (2n-1)-1 = 2n-2 = n-1 + (n-1)
        # Actually len(w) - 1 = 2n-2.
        # slice start = len(w) - n = 2n-1 - n = n-1
        # so slice = conv_rev[n-1 : n-1 + n]
        # Use as above.
        # gradient shape: (n,)

        # Adam update
        t += 1
        lr = lr0 * (np.sqrt(1 - beta2**t) / (1 - beta1**t))

"""
search_step_function_v3.py

Optimiser for ``evaluate_sequence``.
- Broad parametric sweep over many functional families (including a new
  arcsine shape).
- Refined optimisation with a *smooth‑max* surrogate of max(conv) using
  an Adam‑style projected gradient descent.  The surrogate is
  ``log‑sum‑exp`` with a temperature that is increased during the run,
  allowing the optimiser to first move on a smooth landscape and then
  focus on the true maximum.
- Final polishing with the original sub‑gradient method and a brief
  stochastic hill‑climb.

The whole search respects the 1000 s budget and prints intermediate
progress.
"""

import time
import math

# ----------------------------------------------------------------------
# Helper utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a*a using an FFT (real)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    Scale invariance allows us to keep Σa = 1 throughout the search.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Families of candidate vectors

def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Front plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1-x_i)^{β}, x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^{p}  (front‑loaded decreasing power‑law)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)       # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def arcsine_shape(n: int) -> np.ndarray:
    """a_i ∝ 1/√(x·(1‑x)), the arcsine density on [0,1] (regularised)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) optimiser – Adam with simplex projection
# ----------------------------------------------------------------------
def smooth_max_optimize(a0: np.ndarray, deadline: float,
                        alpha_start: float = 5.0,
                        alpha_end: float = 200.0,
                        lr0: float = 0.25) -> np.ndarray:
    """
    Minimises the true max(conv) by optimising a smooth surrogate:
        M_α(a) = (1/α)·log Σ exp(α·conv_k(a))

    Adam‑style updates are applied, with periodic increase of α.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0
    alpha = alpha_start

    while time.time() < deadline:
        # ----- convolution and softmax weights -----
        conv = full_convolution(a)                 # length 2n‑1
        max_c = conv.max()
        # stabilised exponentials
        exp_shift = conv - max_c                   # ≤ 0
        w = np.exp(alpha * exp_shift)
        w_sum = w.sum()
        if w_sum == 0.0:                           # numerical safeguard
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate -----
        # grad = 2 * Σ_k w_k * a shifted by k   (see analysis)
        a_rev = a[::-1]
        m_len = n + (2 * n - 1) - 1                 # = 3n‑2
        n_fft = 1 << ((m_len - 1).bit_length())
        A = np.fft.rfft(a_rev, n=n_fft)
        W = np.fft.rfft(w, n=n_fft)
        conv_rev = np.fft.irfft(A * W, n=n_fft)[:m_len]
        # slice index n‑1 … n‑1+n‑1
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ----- Adam update -----
        t += 1
        lr = lr0 * (np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t))
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex (keeps Σa = 1, a≥0)
        a = project_to_simplex(a)

        # ----- schedule for α (sharpen the approximation) -----
        if t % 400 == 0:
            alpha = min(alpha * 1.07, alpha_end)

        # ----- keep the best (true) objective -----
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()

        # occasional jitter to escape shallow plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (lr0 * 0.002 / np.sqrt(t))
            a = best_a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as in the reference implementation)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient: 2 * a shifted for each active index
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M, best_a = M_new, a.copy()

    return best_a

# Simple stochastic hill‑climb (final fine‑tuning)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]                 # move a fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M, best_a = M_cand, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # keep a small safety margin

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback (if a previous construction is present)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Parametric sweep (≈ 250 s, leaves ample time for refinement)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 500.0    # reserve ~500 s for the optimisation phases
    n = 2048
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[parametric sweep] best proxy score = {val:.6f}")
            val_real = evaluate_sequence(a_initial.tolist())
            print(f"[parametric sweep] official score = {val_real:.6f}")
            if val_real < best_val:
                best_val, best_seq = val_real, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Smooth‑max Adam optimisation (the core improvement)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 180.0
    if best_seq is not None and time.time() < smooth_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_smooth = smooth_max_optimize(a0, smooth_deadline)
        val = evaluate_sequence(a_smooth.tolist())
        print(f"[smooth‑max] after optimisation score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_smooth.tolist()

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (fine‑tunes the exact max)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 60.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.15)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Final stochastic hill‑climb (quick local search)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a solid reverse‑power‑law candidate
    # -----------------------------------------------------------------
    if best_seq is None:
        a_fallback = reverse_power_law(n, 1.8)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
