def project_simplex(v):
    # v: np.ndarray
    # returns projection onto simplex: sum =1, v>=0
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def subgradient_optimize(a0, deadline, init_step=0.1, min_step=1e-8):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        # gradient
        a_rev = a[::-1]
        i = np.arange(n)
        j = n - 1 + i - max_idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a_rev[j[mask]]
        # step size
        step = init_step / np.sqrt(it)

        cand = np.maximum(cand, 0.0)
        # project onto simplex
        cand = project_simplex(cand)
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            # optionally increase step factor?
        a = cand
        # maybe occasional random perturbation
        if it % 10000 == 0:
            # small random jitter to escape
            jitter = np.random.randn(n) * (init_step * 0.01)
            a = best_a * np.exp(jitter)
            a = project_simplex(a)
    return best_a

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0
    n = 2048

    best_seq = None
    best_val = float('inf')

    # baseline search
    base_deadline = deadline - 400 # keep 400s for other phases
    if time.time() < base_deadline:
        baseline_vecs = baseline_search(n, base_deadline, max_keep=12)
        for vec in baseline_vecs:
            val = compute_score(vec)
            if val < best_val:
                best_val = val
                best_seq = vec.tolist()
                print(f"[baseline] score={val:.6f}")

    # subgradient optimization
    # allocate time for subgradient (maybe majority)
    sub_deadline = deadline - 150  # reserve for final polish
    if best_seq is not None and time.time() < sub_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = subgradient_optimize(a0, sub_deadline, init_step=0.2)
        val = compute_score(refined)
        print(f"[subgrad] refined score={val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # also try a few random restarts if time left
        while time.time() < sub_deadline:
            remaining = sub_deadline - time.time()
            if remaining < 30:
                break
            # random Dirichlet with random conc
            conc = np.random.choice([0.1,0.2,0.5,1.0,2.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = subgradient_optimize(a_start, time.time() + min(30, remaining), init_step=0.2)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[subgrad‑restart] new best={val:.6f}")

    # final polishing with Adam smooth
    final_deadline = deadline - 5.0
    if best_seq is not None and time.time() < final_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = smooth_optimize(a_start, final_deadline, init_lr=0.08, init_tau_factor=5.0)
        val = compute_score(final)
        print(f"[final‑smooth] score={val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # fallback
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # final evaluation
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished, final_score={final_score:.6f}")
    return best_seq

"""
Improved search for a non‑negative step‑function that minimises
evaluate_sequence.

Algorithm
---------
1. Global exploration – sweep power‑law, exponential, beta and Gaussian
   families and draw many Dirichlet samples.  Keep the best few vectors.

2. Subgradient descent on the exact max‑convolution objective.
   The subgradient of max_k (a*a)_k is 2·a shifted by the index where the
   convolution attains its maximum.  After each step we project onto the
   probability simplex (non‑negative, sum = 1).  The step size is
   proportional to 1/√iteration and we occasionally add a small jitter
   to escape plateaux.

3. Final polishing – a short Adam optimiser on the smooth log‑sum‑exp
   approximation (the same code as before) to fine‑tune the result.

The whole routine respects the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution via real FFT (any two arrays)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b (real FFT)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())     # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (used for the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (no clipping – all entries already non‑negative)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used by the baseline search)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global baseline exploration (keeps up to `max_keep` best vectors)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s                     # project onto the simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- 1) power‑law -------------------------------------------------
    for p in np.linspace(0.1, 2.0, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # ---- 2) exponential ------------------------------------------------
    for lam in np.linspace(0.0001, 0.05, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # ---- 3) beta -------------------------------------------------------
    al_vals = np.linspace(0.2, 5.0, 25)
    be_vals = np.linspace(0.2, 5.0, 25)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # ---- 4) Gaussian ----------------------------------------------------
    sigma_vals = np.linspace(n / 16.0, n / 2.0, 30)   # from narrow to wide
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # ---- 5) Dirichlet draws --------------------------------------------
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1200
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 10.0,
                    min_tau: float = 1e-8,
                    tau_decay: float = 0.9995) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max mapping a = softmax(x)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        conv = full_convolution(a)
        max_c = conv.max()

        # smooth max weights
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient w.r.t. a
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule for the soft‑max (x‑space)
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # occasional exact evaluation
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:6d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Subgradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray, deadline: float,
                         init_step: float = 0.2,
                         min_step: float = 1e-8) -> np.ndarray:
    """
    Projected subgradient descent for   min  max_k (a*a)_k
    under a ≥ 0, sum(a)=1.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and location of the maximum
        conv = full_convolution(a)
        k_max = int(np.argmax(conv))

        # gradient of the max (one‑hot weight)
        a_rev = a[::-1]
        i = np.arange(n)
        j = (n - 1) + i - k_max
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a_rev[j[mask]]

        # diminishing step size
        step = init_step / np.sqrt(it)
        step = max(step, min_step)

        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        cand = project_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            # allow larger steps after improvement
            init_step *= 1.02

        a = cand

        # occasional random jitter to escape plateaux
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (init_step * 0.01)
            a = best_a * np.exp(jitter)
            a = project_simplex(a)

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.  Returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    overall_deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 offers a good trade‑off between expressivity
    # and speed.
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 180 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(overall_deadline, start_time + 180.0)
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Subgradient descent (major part of the budget)
    # -----------------------------------------------------------------
    subgrad_deadline = overall_deadline - 150.0   # reserve ~150 s for polishing
    if best_seq is not None and time.time() < subgrad_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = subgradient_optimize(a0, subgrad_deadline, init_step=0.2)
        val = compute_score(refined)
        print(f"[subgrad] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # Additional random restarts while time permits
        while time.time() < subgrad_deadline:
            remaining = subgrad_deadline - time.time()
            if remaining < 30.0:
                break
            # random Dirichlet start (vary concentration)
            conc = np.random.choice([0.1, 0.2, 0.4, 0.7, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = subgradient_optimize(
                a_start, time.time() + min(30.0, remaining), init_step=0.2
            )
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[subgrad‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final polishing with the smooth Adam optimiser
    # -----------------------------------------------------------------
    final_deadline = overall_deadline - 5.0
    if best_seq is not None and time.time() < final_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = smooth_optimize(a_start, final_deadline,
                                init_lr=0.08,
                                init_tau_factor=5.0,
                                min_tau=1e-8,
                                tau_decay=0.9995)
        val = compute_score(final)
        print(f"[final‑smooth] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
