def project_onto_simplex(v):
    # from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def smooth_max_convolution_grad(a, eps):
    conv = full_convolution(a)
    # compute weighted probs
    max_conv = conv.max()
    # shift for stability
    shifted = conv - max_conv  # negative values
    # compute exp(shifted/eps)
    exp_vals = np.exp(shifted / eps)
    w = exp_vals / exp_vals.sum()
    # compute correlation term:
    w_rev = w[::-1]
    corr = np.convolve(a, w_rev)
    # get g1
    n = a.shape[0]
    g1 = corr[2*n-2 - np.arange(n)]
    # second term:
    w_center = w[2 * np.arange(n)]
    grad = 2 * n * (g1 + a * w_center)
    return grad

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

This implementation replaces the previous sub‑gradient method with a
*smooth‑max* annealing scheme:

1. The true objective  2·n·max(conv)  is approximated by a
   log‑sum‑exp (smooth maximum) with a temperature ``eps``.
2. The gradient of the smoothed objective can be written in closed form
   using one convolution and one correlation, both evaluated quickly with
   the FFT.
3. An Adam optimiser works on the probability vector directly, with a
   projection onto the simplex after each update.
4. ``eps`` is decreased stepwise (simulated‑annealing on the objective)
   allowing the optimiser to first shape the distribution globally and
   then polish the exact max.

The routine respects a 1000 s wall‑clock budget, keeps the best vector
found (evaluated with the exact objective) and finally returns it as a
plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power‑of‑two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    return np.fft.irfft(X * Y, n=n_fft)[:m]


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (same length vectors)."""
    return convolve_fft(a, a)


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1                # zero‑based index
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) for a simplex‑vector a."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed gradient of 2·n·log‑sum‑exp(conv/eps)
# ----------------------------------------------------------------------


def smooth_max_grad(a: np.ndarray, eps: float):
    """
    Returns (grad, conv) where ``grad`` is the gradient of
    2·n·log‑sum‑exp(conv/eps) w.r.t. a, and ``conv`` is a*a.
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    conv = full_convolution(a)               # length 2·n‑1

    # ----- soft‑max weights for the smooth max -----
    max_c = conv.max()
    # shift for numerical stability
    shifted = (conv - max_c) / eps
    # exp may underflow for large negative values – that's fine
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()            # length 2·n‑1, non‑negative, sums to 1

    # ----- term  Σ_j w_{i+j}·a_j  (correlation) -----
    # correlation can be written as a *rest conj w[::-1]
    w_rev = w[::-1]
    corr_full = convolve_fft(a, w_rev)       # length n + (2n‑1)‑1 = 3·n‑2
    idx = 2 * n - 2 - np.arange(n)           # pick entries that correspond to Σ_j w_{i+j} a_j
    term1 = corr_full[idx]

    # ----- term a_i·w_{2i} -----
    term2 = a * w[2 * np.arange(n)]

    # ----- full gradient (2·n factor from the objective) -----
    grad = 2.0 * n * (term1 + term2)

    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser on the simplex with smooth‑max annealing
# ----------------------------------------------------------------------


def adam_smooth_optimize(a0: np.ndarray,
                         deadline: float,
                         eps_schedule=None,
                         lr: float = 0.15,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         adam_eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimiser that minimises the exact max‑convolution through a
    smooth‑max surrogate.  ``eps_schedule`` is a list of temperatures,
    processed from large to small.
    Returns the best vector observed (exact objective).
    """
    if eps_schedule is None:
        eps_schedule = [0.08, 0.04, 0.02, 0.01, 0.006, 0.003, 0.0015, 0.0008]

    n = a0.shape[0]
    a = project_onto_simplex(a0.copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)
    t = 0

    for eps in eps_schedule:
        # a modest iteration budget per temperature; the outer loop stops
        # early if the global deadline is reached.
        max_iters = 2500
        for _ in range(max_iters):
            if time.time() >= deadline:
                return best_a

            t += 1

            # ---- gradient of the smooth surrogate ----
            grad, conv = smooth_max_grad(a, eps)

            # ---- Adam moment updates ----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # ---- learning‑rate schedule (gentle decay) ----
            step = lr / (np.sqrt(t) + 1.0)

            a_temp = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)

            # ---- projection onto the simplex ----
            a = project_onto_simplex(a_temp)

            # ---- exact objective for the new point ----
            cur_val = 2.0 * n * conv.max()   # using the previously computed conv
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

            # occasional jitter to escape shallow plateaus
            if t % 3000 == 0:
                jitter = np.random.randn(n) * (lr * 0.001)
                a = project_onto_simplex(a + jitter)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The routine uses a smooth‑max
    Adam optimisation with an annealing schedule and a few random restarts.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the call to evaluate_sequence
    deadline = start_time + budget_s - 5.0

    n = 2048                     # length of the candidate vector
    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0.  Use the previously discovered sequence, if present
    # --------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Run a few optimisation restarts (random Dirichlet starts)
    # --------------------------------------------------------------
    # we keep one “primary” run that starts from the current best (or
    # uniform if none exists) and then spawn cheap restarts while time
    # permits.
    def run_one(start_vec):
        nonlocal best_val, best_seq
        refined = adam_smooth_optimize(start_vec, deadline)
        val = compute_score(refined)
        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            print(f"[restart] new best = {val:.6f}")

    # primary start – either the fallback or a uniform distribution
    if best_seq is None:
        primary_start = np.full(n, 1.0 / n)
    else:
        primary_start = np.array(best_seq, dtype=float)

    run_one(primary_start)

    # additional random restarts
    conc_levels = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    while time.time() < deadline - 10.0:
        conc = np.random.choice(conc_levels)
        a0 = np.random.dirichlet(np.full(n, conc))
        run_one(a0)

    # --------------------------------------------------------------
    # 2.  Final polishing with a very small epsilon (almost exact max)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a0 = np.array(best_seq, dtype=float)
        # one last run with a tighter schedule
        refined = adam_smooth_optimize(
            a0,
            deadline,
            eps_schedule=[0.0005, 0.0003, 0.0002, 0.0001, 8e-05, 5e-05],
            lr=0.08
        )
        val = compute_score(refined)
        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            print(f"[final‑polish] improved score = {val:.6f}")

    # --------------------------------------------------------------
    # 3.  Fallback to the uniform distribution if nothing succeeded
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 4.  Report the final score using the official evaluator
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
