    n = len(f)

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    # fallback
    best_seq = None
    best_val = float('inf')
    try:
        fallback = [float(x) for x in height_sequence_1]  # global variable
        val = evaluate_sequence(fallback)
        print(f"[fallback] score {val}")
        best_seq = fallback
        best_val = val
    except Exception:
        pass

    # 0. Determine n and initial vector
    if best_seq is not None:
        n = len(best_seq)
        f = np.array(best_seq, dtype=float)
        # Ensure sum=1 for internal optimization
        f = f / np.sum(f)
    else:
        n = 2048
        f = np.full(n, 1.0 / n)

    # 1. Parametric sweep (short)
    sweep_deadline = deadline - 400
    if time.time() < sweep_deadline:
        a_init = parametric_sweep(n, sweep_deadline)
        if a_init is not None:
            f = a_init
            val = compute_score(f)
            if val < best_val:
                best_val = val
                best_seq = f.tolist()
                print(f"[sweep] new best score {val}")

    # 2. LP iterative improvement
    lp_deadline = deadline - 250
    if time.time() < lp_deadline:
        f_improved = lp_iterative_improve(f, lp_deadline)
        val = compute_score(f_improved)
        if val < best_val:
            best_val = val
            best_seq = f_improved.tolist()
            print(f"[lp] improved score {val}")

    # 3. Subgradient polish
    sub_deadline = deadline - 120
    if time.time() < sub_deadline:
        a_polished = subgradient_polish(np.array(best_seq, dtype=float), sub_deadline)
        val = compute_score(a_polished)
        if val < best_val:
            best_val = val
            best_seq = a_polished.tolist()
            print(f"[subgrad] polished score {val}")

    # 4. Anneal/hill
    hill_deadline = deadline
    if time.time() < hill_deadline:
        a_final = hill_climb(np.array(best_seq, dtype=float), hill_deadline)
        val = compute_score(a_final)
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()
            print(f"[hill] final score {val}")

    # Safeguard
    if best_seq is None:
        best_seq = [1.0 / n] * n

    out_score = evaluate_sequence(best_seq)
    print(f"\nFinal reported score {out_score}")
    return best_seq

Optimiser for the self‑convolution L∞ bound.

Strategy
--------
1. Very quick parametric sweep (power‑law, geometric, beta) to obtain a

∑a is fixed to 1).  The routine respects the 1000 s budget and prints
progress messages.  The returned list contains non‑negative floats and
will be fed to ``evaluate_sequence`` by the harness.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a∗a using a real FFT (fast)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1):
       2·n·max(conv(a,a)) ."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Simple parametric families (used for the initial sweep)
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray | None:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def geometric(n: int, q: float) -> np.ndarray | None:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def random_dirichlet(n: int, concentration: float) -> np.ndarray | None:
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    s = y.sum()
    if s == 0.0:
        return None
    return y / s


def parametric_sweep(n: int, deadline: float) -> np.ndarray | None:
    """Very quick sweep over a few families; returns the best vector."""
    best_vec = None
    best_val = float('inf')

    # power‑law
    for p in np.linspace(0.10, 2.00, 81):
        if time.time() >= deadline:
            return best_vec
        a = power_law(n, p)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # geometric
    for q in np.linspace(0.90, 0.999, 111):
        if time.time() >= deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # beta shape
    alphas = np.linspace(-0.5, 2.5, 13)
    betas = np.linspace(-0.5, 2.5, 13)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # random Dirichlet
    for _ in range(300):
        if time.time() >= deadline:
            break
        conc = 10 ** np.random.uniform(-2, 1)   # 0.01 … 10
        a = random_dirichlet(n, conc)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec

# ----------------------------------------------------------------------
# Sub‑gradient polishing (as used previously)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on the infinity norm of a∗a."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient: for each active index add 2·a shifted
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# ----------------------------------------------------------------------
# Simple hill‑climb (deterministic greedy moves)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy mass‑transfer hill‑climb."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M - 1e-12:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# ----------------------------------------------------------------------
# Linear‑programming based improvement
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """Linear programme:
       maximise Σ b_j  subject to  b_j ≥ 0  and  (f∗b)_k ≤ M."""
    n = f.shape[0]
    L = 2 * n - 1
    # Build the convolution matrix A where (A·b)_k = (f∗b)_k
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    b = np.full(L, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σ b_j  → minimise –Σ b_j
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs')
    if res.success:
        return res.x
    # In the rare case the LP fails we simply return None
    print("[LP] failure:", res.message)
    return None


def lp_iterative_improve(f0: np.ndarray, deadline: float,
                         max_iter: int = 30,
                         mix_factor: float = 0.5) -> np.ndarray:
    """Iteratively improve f by solving the LP described above."""
    f = f0.copy()
    n = f.shape[0]
    best_f = f.copy()
    best_M = full_convolution(f).max()
    it = 0
    t = mix_factor

    while time.time() < deadline and it < max_iter:
        it += 1
        M = full_convolution(f).max()
        g = _solve_lp_max_sum(f, M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        # Normalise g to have total mass 1
        g_norm = g / sum_g

        # Convex combination
        f_new = (1.0 - t) * f + t * g_norm
        f_new = np.clip(f_new, 0.0, None)
        f_new = project_to_simplex(f_new)

        M_new = full_convolution(f_new).max()
        print(f"[LP] iter {it:02d}, M before {M:.8f}, M after {M_new:.8f}")

        # Accept if we made progress
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_f = f_new.copy()
            f = f_new
            # slowly increase the mixing factor if things go well
            t = min(t * 1.05, 0.9)
        else:
            # If no improvement, shrink the mixing factor and try again
            t = max(t * 0.5, 0.05)
            # also break after a few unsuccessful attempts
            if t < 0.06:
                break

    return best_f

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Fallback: previously discovered construction (if available)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        best_seq, best_val = fallback, val
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Determine problem size and an initial vector
    # -----------------------------------------------------------------
    if best_seq is not None:
        n = len(best_seq)
        f = np.array(best_seq, dtype=float)
        # Normalise to sum = 1 for internal optimisation
        f = f / f.sum()
    else:
        n = 2048
        f = np.full(n, 1.0 / n, dtype=float)

    # -----------------------------------------------------------------
    # 2. Quick parametric sweep (≈ 150 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    if time.time() < sweep_deadline:
        a_sweep = parametric_sweep(n, sweep_deadline)
        if a_sweep is not None:
            f = a_sweep
            val = compute_score(f)
            if val < best_val:
                best_val = val
                best_seq = f.tolist()
                print(f"[parametric sweep] improved score = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Linear‑programming alternating improvement (≈ 250 s)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 250.0
    if time.time() < lp_deadline:
        f_lp = lp_iterative_improve(f, lp_deadline, max_iter=30, mix_factor=0.5)
        val = compute_score(f_lp)
        if val < best_val:
            best_val = val
            best_seq = f_lp.tolist()
            print(f"[LP improvement] score = {val:.6f}")
        f = f_lp   # keep the refined vector for the next phase

    # -----------------------------------------------------------------
    # 4. Sub‑gradient polishing (≈ 150 s)
    # -----------------------------------------------------------------
    sub_deadline = deadline - 120.0
    if time.time() < sub_deadline:
        a_polished = subgradient_polish(np.array(best_seq, dtype=float),
                                        sub_deadline, step0=0.15)
        val = compute_score(a_polished)
        if val < best_val:
            best_val = val
            best_seq = a_polished.tolist()
            print(f"[sub‑grad] polished score = {val:.6f}")

    # -----------------------------------------------------------------
    # 5. Final hill‑climb (remaining time)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        a_final = hill_climb(np.array(best_seq, dtype=float), deadline)
        val = compute_score(a_final)
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()
            print(f"[hill‑climb] final score = {val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official scoring function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
