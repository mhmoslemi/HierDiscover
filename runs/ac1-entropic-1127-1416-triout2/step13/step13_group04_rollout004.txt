def lp_refine(a, deadline, max_iter=20):
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while it < max_iter and time.time() < deadline:
        # compute conv & M
        conv = full_convolution(a)
        M = conv.max()
        # Build constraint matrix
        # A shape (2n-1, n)
        rows = 2 * n - 1
        A = np.zeros((rows, n))
        for j in range(n):
            A[j:j+n, j] = a
        # Solve LP
        c = -np.ones(n)  # minimize -sum(g) => maximize sum(g)
        bounds = [(0, None)] * n
        b_ub = np.full(rows, M)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print("[lp_refine] LP failed at iteration", it)
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:
            print("[lp_refine] sum_g0 too small")
            break
        g = g0 / sum_g0
        # line search for t
        # start with t=0.5
        t = 0.5
        improved = False
        while t > 1e-5:
            cand = (1 - t) * a + t * g
            cand_score = compute_score(cand)
            if cand_score < best_score - 1e-12:
                a = cand
                best_a = cand.copy()
                best_score = cand_score
                improved = True
                print(f"[lp_refine] iter {it} t {t:.4f} score {best_score:.6f}")
                break
            else:
                t *= 0.5
        if not improved:
            # maybe no improvement: break or continue with reduced step?
            # could try smaller t
            # we just break loop to avoid endless
            print("[lp_refine] No improvement at iteration", it)
            break
        it += 1
    return best_a

```python
        x_new = np.linspace(0, 1, new_n)
        return np.interp(x_new, x_old, a)
    else:
        factor = old_n // new_n
```

The optimizer searches for a non‑negative step‑function (a list of heights) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2},
\]

where `a*a` is the linear (full) self‑convolution.  
The algorithm works in four stages

1. **Fallback / quick start** – use an already known good construction (`height_sequence_1`) and down‑sample it to the working size.
2. **Broad parametric sweep** – evaluate several analytic families (power‑law, geometric, mixed plateau‑power, beta, logistic, Gaussian and random Dirichlet).  
   The best vector from this stage seeds the next phases.
3. **LP‑based refinement** – for a normalized vector `f` (∑f = 1) we solve  

   \[
   \max\_{g\ge0}\;\sum_i g_i
   \qquad\text{s.t.}\qquad
   (f*g)_k\le M\;\;(k=0,\dots,2n-2),
   \]

   where \(M=\max(f*f)\).  
   The solution `g` is renormalised to unit sum and merged with the current
   vector by a small convex step. Repeating this step provably never raises
   `M` and quickly drives the objective down.
4. **Deterministic hill‑climb** – random mass‑transfer moves are performed
   while always staying on the probability simplex.  The move size is
   adaptively controlled; the best vector seen is kept.

All stages respect the 1000 s time budget.  The final vector is returned as a
plain Python list; its sum is 1, all entries are ≥ 0, and it is guaranteed to
pass the security checks of `evaluate_sequence`.

```python
"""
Optimal step‑function search.

The optimiser works in four phases:
  1. optional fallback (down‑sampled `height_sequence_1`);
  2. a fast parametric sweep over several analytic families;
  3. an LP‑based refinement that is provably non‑increasing for the
     max‑self‑convolution;
  4. a deterministic hill‑climb (random mass transfers with simplex
     projection).

All vectors are kept normalised to sum = 1, because the objective
2·n·max(conv)/(sum²) is scale‑invariant.  The code uses only NumPy,
SciPy‑HiGHS (via `scipy.optimize.linprog`) and the standard library.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Geometry utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        # if everything is ≤0 the projection is the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())        # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a normalised np.ndarray)
# ----------------------------------------------------------------------


def power_law(n: int, p: float) -> np.ndarray | None:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    return None if s == 0 else a / s


def geometric(n: int, q: float) -> np.ndarray | None:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return None if s == 0 else a / s


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray | None:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    return None if s == 0 else a / s


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    return None if s == 0 else a / s


def logistic_shape(n: int, center_frac: float, scale_frac: float) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    scale = max(scale_frac * n, 1e-6)
    a = 1.0 / (1.0 + np.exp((i - center) / scale))
    s = a.sum()
    return None if s == 0 else a / s


def gaussian_shape(n: int, gamma: float) -> np.ndarray | None:
    """a_i ∝ exp(-γ·(i/(n-1))²)   (decreasing Gaussian)"""
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.exp(-gamma * (x ** 2))
    s = a.sum()
    return None if s == 0 else a / s


def random_dirichlet(n: int, concentration: float) -> np.ndarray | None:
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    s = y.sum()
    return None if s == 0 else y / s


# ----------------------------------------------------------------------
# Broad parametric sweep (fast, ≈ a few seconds for n=1024)
# ----------------------------------------------------------------------


def parametric_sweep(n: int, deadline: float) -> np.ndarray | None:
    """Explore families and keep the best normalised vector."""
    best_vec = None
    best_val = float('inf')

    # power‑law
    for p in np.linspace(0.10, 2.00, 121):
        if time.time() >= deadline:
            return best_vec
        a = power_law(n, p)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # geometric
    for q in np.concatenate((np.linspace(0.90, 0.99, 81),
                             np.linspace(0.990, 0.9999, 70))):
        if time.time() >= deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # mixed plateau + power
    for L_frac in np.linspace(0.02, 0.35, 12):
        for p in np.linspace(0.30, 0.90, 25):
            if time.time() >= deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # beta shape
    alphas = np.linspace(-0.8, 3.0, 16)
    betas = np.linspace(-0.8, 3.0, 16)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # logistic
    for center_frac in np.linspace(0.2, 0.8, 11):
        for scale_frac in np.linspace(0.01, 0.20, 15):
            if time.time() >= deadline:
                return best_vec
            a = logistic_shape(n, center_frac, scale_frac)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # Gaussian
    for gamma in np.linspace(0.5, 5.0, 20):
        if time.time() >= deadline:
            return best_vec
        a = gaussian_shape(n, gamma)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # random Dirichlet
    for _ in range(800):
        if time.time() >= deadline:

        a = random_dirichlet(n, conc)
        if a is None:

        val = compute_score(a)
        if val < best_val:

# ----------------------------------------------------------------------
# LP‑based refinement (guaranteed non‑increase of max‑self‑convolution)
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray, deadline: float, max_iter: int = 20) -> np.ndarray:
    """
    Iteratively improve a normalised vector `a` by solving
    a linear program that maximises Σg under the constraint
    (a * g)∞ ≤ M, where M = max(a * a).
    The new vector is a convex combination (1‑t)·a + t·g with a small t.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while it < max_iter and time.time() < deadline:
        conv = full_convolution(a)
        M = conv.max()

        # build the convolution matrix A (size (2n‑1)×n)
        rows = 2 * n - 1
        A = np.empty((rows, n), dtype=np.float64)
        # each column j is the vector a placed at offset j
        for j in range(n):
            A[j:j + n, j] = a

        # LP: minimise -Σg   ⇔ maximise Σg
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        b_ub = np.full(rows, M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # fallback: stop refinement
            print("[lp_refine] LP failed at iteration", it, "status:", res.message)
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:
            # nothing useful was found
            break

        g = g0 / sum_g0                # re‑normalise to unit sum

        # try a convex step; start with a moderately small step size
        t = 0.2
        improved = False
        while t > 1e-5:
            cand = (1.0 - t) * a + t * g
            cand_score = compute_score(cand)
            if cand_score < best_score - 1e-12:
                a = cand
                best_a = cand.copy()
                best_score = cand_score
                improved = True
                print(f"[lp_refine] iter {it} t={t:.4f} score={best_score:.6f}")
                break
            t *= 0.5

        if not improved:
            # no improvement – stop early to avoid endless tiny steps
            break

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Deterministic hill‑climb (random mass transfers)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 400_000) -> np.ndarray:
    """
    Greedy mass‑transfer hill‑climb on the simplex.
    Moves a fraction of mass from a source index to a destination index.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1

        # occasional jitter to escape plateaux
        if no_improve > 120_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Helper: down‑sample a longer vector to a target length
# ----------------------------------------------------------------------


def downsample(seq, new_n: int) -> np.ndarray:
    """
    Down‑sample a one‑dimensional sequence to `new_n` points.
    If the original length is a multiple of `new_n` a simple block‑average
    is used; otherwise linear interpolation is performed.
    """
    a = np.asarray(seq, dtype=np.float64)
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    if old_n % new_n == 0:
        factor = old_n // new_n
        return a.reshape(new_n, factor).mean(axis=1)
    # linear interpolation for non‑integer factors
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)
```
