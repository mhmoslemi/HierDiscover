"""
Propose a high‑quality step‑function (non‑negative vector) that minimises

    2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm combines:
* systematic scanning of several analytic families (including new asymmetric
  mixtures);
* a large random search over these families;
* fine 1‑D local refinements of the best parameters;
* permutation search (scrambling the magnitudes of the best shape);
* a smooth‑max Adam optimisation (continuation on the temperature);
* a short projected sub‑gradient descent;
* a final greedy hill‑climb that transfers small masses between entries.

All stages respect a shared 1000 s wall‑clock budget.  The best sequence
found (as a plain Python list) is returned.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a, a)) / (sum(a)**2).
    a must be non‑negative (the optimiser keeps it on the simplex).
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    # avoid log(0) by adding a tiny epsilon
    eps = 1e-12
    logprod = np.log(i + 0.5 + eps) + np.log(n - 0.5 - i + eps)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Mirror of left_heavy_power_family."""
    return left_heavy_power_family(n, beta)[::-1]


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)


def mixed_product_uniform_family(n: int, lam: float, alpha: float) -> np.ndarray:
    """Convex mixture of uniform and the symmetric product family."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    prod = symmetric_product_family(n, alpha)
    w = lam * uniform + (1.0 - lam) * prod
    return normalize(w)


def asymmetric_mix_family(n: int, lam: float, beta_l: float, beta_r: float) -> np.ndarray:
    """
    Asymmetric mixture of a left‑heavy and a right‑heavy power law.
    lam∈[0,1] weights the left part.
    """
    left = left_heavy_power_family(n, beta_l)
    right = right_heavy_power_family(n, beta_r)
    w = lam * left + (1.0 - lam) * right
    return normalize(w)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 5.0,
                         beta_end: float = 20_000.0,
                         lr: float = 0.012,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    The temperature ``beta`` grows geometrically from ``beta_start`` to
    ``beta_end``.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)          # log‑softmax parametrisation
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                     # smooth‑max weights

        # gradient of smooth‑max w.r.t a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 5000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the max entry k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_greedy(start_a: np.ndarray,
                      deadline: float,
                      max_steps: int = 5_000_000,
                      init_scale: float = 0.15) -> Tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb that moves a small fraction of mass from the
    largest entry to a random other entry whenever this reduces the
    objective.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    scale = init_scale

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        i = np.argmax(cur_a)                     # donor
        j = np.random.randint(n)                 # receiver
        if i == j:
            continue

        delta = scale * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            scale = min(scale * 1.03, 0.5)
        else:
            scale = max(scale * 0.97, 1e-4)

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] greedy step {step}, best {best_score:.6f}")

    return best_a, best_score

def hill_climb_alpha(alpha0: float, n: int, deadline: float) -> float:
    """1‑D hill‑climb on the product‑family parameter α."""
    best = max(0.0, alpha0)
    best_score = compute_score(symmetric_product_family(n, best))
    step = 0.8
    while time.time() < deadline and step > 1e-4:
        improved = False
        for sgn in (-1, 1):
            cand = max(0.0, best + sgn * step)
            sc = compute_score(symmetric_product_family(n, cand))
            if sc < best_score:
                best_score = sc
                best = cand
                improved = True
                print(f"[{time.time():.1f}] α refined to {best:.6f} score {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best


def hill_climb_beta(beta0: float, n: int, deadline: float) -> float:
    """1‑D hill‑climb on the left‑heavy (or right‑heavy) power exponent."""
    best = max(0.0, beta0)
    best_score = compute_score(left_heavy_power_family(n, best))
    step = 0.8
    while time.time() < deadline and step > 1e-4:
        improved = False
        for sgn in (-1, 1):
            cand = max(0.0, best + sgn * step)
            sc = compute_score(left_heavy_power_family(n, cand))
            if sc < best_score:
                best_score = sc
                best = cand
                improved = True
                print(f"[{time.time():.1f}] β refined to {best:.6f} score {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best


def hill_climb_beta_params(p0: float, q0: float, n: int, deadline: float) -> Tuple[float, float]:
    """2‑D hill‑climb in the Beta‑family (p,q)."""
    best_p, best_q = max(0.01, p0), max(0.01, q0)
    best_score = compute_score(beta_family(n, best_p, best_q))
    step = 0.4
    while time.time() < deadline and step > 1e-3:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(0.01, best_p + dp)
                cand_q = max(0.01, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_score = sc
                    best_p, best_q = cand_p, cand_q
                    improved = True
                    print(f"[{time.time():.1f}] Beta refined to p={best_p:.6f}, q={best_q:.6f} score {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
    return best_p, best_q


def hill_climb_asym_params(lam0: float, beta_l0: float, beta_r0: float,
                           n: int, deadline: float) -> Tuple[float, float, float]:
    """Hill‑climb for the asymmetric mixture (lam, β_L, β_R)."""
    best = np.array([lam0, beta_l0, beta_r0], dtype=np.float64)
    best_score = compute_score(asymmetric_mix_family(n, *best))
    # initial step sizes
    step_lam = 0.2
    step_beta = 0.8
    while time.time() < deadline and (step_lam > 1e-3 or step_beta > 1e-3):
        improved = False
        # try adjusting each parameter individually
        for i, step in enumerate([step_lam, step_beta, step_beta]):
            for sgn in (-1, 1):
                cand = best.copy()
                cand[i] = cand[i] + sgn * step
                # enforce bounds
                if i == 0:
                    cand[i] = np.clip(cand[i], 0.0, 1.0)
                else:
                    cand[i] = max(cand[i], 0.0)
                sc = compute_score(asymmetric_mix_family(n, *cand))
                if sc < best_score:
                    best_score = sc
                    best = cand
                    improved = True
                    print(f"[{time.time():.1f}] asym refined to lam={best[0]:.4f}, βL={best[1]:.4f}, βR={best[2]:.4f} → {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step_lam *= 0.5
            step_beta *= 0.5
    return best[0], best[1], best[2]

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises the
    evaluation function within the given wall‑clock budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # keep a small safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback using a previously known sequence
    # ------------------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = np.array([float(x) for x in height_sequence_1], dtype=np.float64)  # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a working length (larger sizes give finer control)
    # ------------------------------------------------------------------
    n = 16384

    # ------------------------------------------------------------------
    # 1. Coarse deterministic scan over analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] starting deterministic scans (n={n})")
    # symmetric product family
    best_alpha = None
    for alpha in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best (product α={alpha:.4f}) = {best_score:.6f}")

        # try a few random right‑shifts (adds zeros on the left)
        for _ in range(2):
            d = np.random.randint(0, n // 30 + 1)
            shifted = np.concatenate([vec[d:], np.zeros(d)])
            shifted = normalize(shifted)
            sc2 = compute_score(shifted)
            if sc2 < best_score:
                best_score = sc2
                best_seq = shifted
                print(f"[{time.time():.1f}] shifted product α={alpha:.4f} d={d} → {best_score:.6f}")

    # left‑heavy power family (monotone decreasing)
    best_beta = None
    for beta in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_beta = beta
            print(f"[{time.time():.1f}] new best (left‑heavy β={beta:.4f}) = {best_score:.6f}")

    # exponential‑U family (symmetric U‑shape)
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        i = np.arange(n, dtype=np.float64)
        dist = np.minimum(i, n - 1 - i)
        vec = np.exp(-gamma * dist)
        vec = normalize(vec)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] new best (exp‑U γ={gamma:.4f}) = {best_score:.6f}")

    # mixture uniform + product
    for lam in np.linspace(0.0, 1.0, 6):
        for alpha in np.linspace(0.0, 15.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_product_uniform_family(n, lam, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f}, α={alpha:.2f}) = {best_score:.6f}")

    # Beta family (grid)
    for p in np.linspace(0.2, 2.0, 10):
        for q in np.linspace(0.2, 2.0, 10):
            if time.time() >= deadline - 30.0:
                break
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec
                print(f"[{time.time():.1f}] new best (Beta p={p:.3f}, q={q:.3f}) = {best_score:.6f}")

    # Asymmetric mixture (new family)
    for _ in range(120):
        if time.time() >= deadline - 30.0:
            break
        lam = np.random.rand()
        beta_l = np.random.uniform(0.0, 15.0)
        beta_r = np.random.uniform(0.0, 15.0)
        vec = asymmetric_mix_family(n, lam, beta_l, beta_r)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] new best (asym lam={lam:.3f}, βL={beta_l:.2f}, βR={beta_r:.2f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random dense search (≈4000 evaluations or until time runs out)
    # ------------------------------------------------------------------
    trials = 0
    max_trials = 4000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(6)
        if choice == 0:           # product family
            alpha = np.random.uniform(0.0, 15.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 1:         # left‑heavy power
            beta = np.random.uniform(0.0, 15.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 2:         # beta family
            p = np.random.uniform(0.05, 2.5)
            q = np.random.uniform(0.05, 2.5)
            vec = beta_family(n, p, q)
        elif choice == 3:         # exponential‑U
            gamma = np.random.uniform(0.0, 6.0)
            i = np.arange(n, dtype=np.float64)
            dist = np.minimum(i, n - 1 - i)
            vec = np.exp(-gamma * dist)
            vec = normalize(vec)
        elif choice == 4:         # mixture uniform+product
            lam = np.random.uniform(0.0, 1.0)
            alpha = np.random.uniform(0.0, 15.0)
            vec = mixed_product_uniform_family(n, lam, alpha)
        else:                     # asymmetric mixture
            lam = np.random.rand()
            beta_l = np.random.uniform(0.0, 15.0)
            beta_r = np.random.uniform(0.0, 15.0)
            vec = asymmetric_mix_family(n, lam, beta_l, beta_r)

        # occasional random right‑shift
        if np.random.rand() < 0.25:
            d = np.random.randint(0, n // 30 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local parameter refinements of the best analytic candidates
    # ------------------------------------------------------------------
    # Refine symmetric product exponent
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining product α (start {best_alpha:.4f})")
        refined_alpha = hill_climb_alpha(best_alpha, n, deadline - 10.0)
        vec = symmetric_product_family(n, refined_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha = refined_alpha
            print(f"[{time.time():.1f}] refined product α={refined_alpha:.6f} → {best_score:.6f}")

    # Refine left‑heavy exponent
    if best_beta is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining left‑heavy β (start {best_beta:.4f})")
        refined_beta = hill_climb_beta(best_beta, n, deadline - 10.0)
        vec = left_heavy_power_family(n, refined_beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_beta = refined_beta
            print(f"[{time.time():.1f}] refined left‑heavy β={refined_beta:.6f} → {best_score:.6f}")

    # Refine beta‑family parameters (starting from a modest U‑shape)
    if time.time() < deadline - 20.0:
        start_p, start_q = 0.5, 0.5
        refined_p, refined_q = hill_climb_beta_params(start_p, start_q, n, deadline - 10.0)
        vec = beta_family(n, refined_p, refined_q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] refined Beta p={refined_p:.6f}, q={refined_q:.6f} → {best_score:.6f}")

    # Refine the asymmetric mixture if it ever appeared
    if time.time() < deadline - 20.0:
        # use the current best sequence's magnitudes as a guide for initialization
        # initialise with a modest set of parameters
        lam0, beta_l0, beta_r0 = 0.5, 2.0, 2.0
        lam_opt, beta_l_opt, beta_r_opt = hill_climb_asym_params(
            lam0, beta_l0, beta_r0, n, deadline - 10.0)
        vec = asymmetric_mix_family(n, lam_opt, beta_l_opt, beta_r_opt)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] refined asym lam={lam_opt:.4f}, βL={beta_l_opt:.4f}, βR={beta_r_opt:.4f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Permutation search – scramble the magnitudes of the best shape
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] permutation search (scrambling magnitudes)")
        base_vals = best_seq.copy()
        perm_trials = 2000
        for _ in range(perm_trials):
            if time.time() >= deadline - 30.0:
                break
            perm = np.random.permutation(n)
            vec = base_vals[perm]
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec
                print(f"[{time.time():.1f}] permutation improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Adam (smooth‑max) refinement – no forced symmetry
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] Adam refinement (smooth‑max)")
        a_start = best_seq.astype(np.float64)
        # allocate iterations proportional to remaining time
        remaining = deadline - time.time()
        max_iter = max(100_000, int(min(300_000, remaining * 150)))  # rough 150 it/s
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=a_start,
            deadline=deadline - 100.0,
            max_iters=max_iter,
            beta_start=5.0,
            beta_end=20000.0,
            lr=0.012,
            sym_interval=0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Projected sub‑gradient descent on the true max‑objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            start_a=best_seq.astype(np.float64),
            deadline=deadline - 30.0,
            max_iters=500_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub
            print(f"[{time.time():.1f}] after sub‑gradient best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final greedy hill‑climb (mass‑transfer moves)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final greedy hill‑climb")
        a_final, sc_final = hill_climb_greedy(
            start_a=best_seq.astype(np.float64),
            deadline=deadline,
            max_steps=8_000_000,
            init_scale=0.15,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final
            print(f"[{time.time():.1f}] greedy improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Fallback to a uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    # Final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq.tolist()
