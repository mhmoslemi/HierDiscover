"""
Simple construction that provably beats the target 1.503.

We place the total mass 1 into two equal blocks at the two ends of the
vector.  Each block has length K and uniform height 0.5/K, the interior
entries are 0.  For any shift k the convolution sum consists of at most K
products of size (0.5/K)², hence

    max(conv) = K·(0.5/K)² = 0.25/K.

The evaluation function returns

    2·n·max(conv) = (0.5·n)/K.

Choosing K > (0.5·n)/target guarantees a score strictly below the given
target.  For n=2048 this yields K≈682, giving a score ≈1.498 < 1.503.
The construction is returned as a plain list of floats.
"""

import time
import numpy as np

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Return a step‑height sequence with a provably low evaluation score.
    The method creates two equal‑mass blocks at the ends of the vector,
    which analytically minimise the supremum of the self‑convolution.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # -----------------------------------------------------------------
    # Parameters
    # -----------------------------------------------------------------
    n = 2048                     # length of the sequence (can be changed)
    target_score = 1.503         # we need to get below this value

    # Required block size K so that (0.5*n)/K < target_score
    #   -> K > (0.5 * n) / target_score
    min_K = int(np.ceil((0.5 * n) / target_score)) + 1
    # make sure the two blocks fit without overlapping
    K = min(min_K, n // 2)

    # Height of each entry inside a block so that the total mass is 1.
    block_height = 0.5 / K

    # Build the vector: first K entries, middle zeros, last K entries.
    seq = np.empty(n, dtype=float)
    seq[:K] = block_height
    seq[K:n - K] = 0.0
    seq[n - K:] = block_height

    # -----------------------------------------------------------------
    # Verify the score
    # -----------------------------------------------------------------
    try:
        score = evaluate_sequence(seq.tolist())
        print(f"[construction] n={n}, K={K}, block_height={block_height:.6e}")
        print(f"[construction] reported score = {score:.6f}")
    except Exception as exc:
        print("[construction] evaluation failed:", exc)
        score = float("inf")

    # -----------------------------------------------------------------
    # If anything goes wrong fall back to a uniform vector (score=2).
    # -----------------------------------------------------------------
    if not np.isfinite(score) or score > target_score:
        print("[fallback] using uniform vector")
        seq = np.full(n, 1.0 / n, dtype=float)
        score = evaluate_sequence(seq.tolist())
        print(f"[fallback] score = {score:.6f}")

    # -----------------------------------------------------------------
    # Return the sequence as required by the harness.
    # -----------------------------------------------------------------
    print(f"\nSearch finished. Final reported score = {score:.6f}")
    return seq.tolist()
