def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution using real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    return np.fft.irfft(C, nfft)[:m]

        final_score = evaluate_sequence(a.tolist())
        print(f"[{time.time():.1f}] constructed two‑block vector "
              f"(n={n}, L={L}) → score = {final_score:.6f}")
    except Exception as e:
        # In the unlikely event that `evaluate_sequence` is unavailable,
        # fall back to a fast exact computation.
        conv = convolution_fft(a, a)
        max_conv = float(conv.max())
        final_score = 2.0 * n * max_conv / (np.sum(a) ** 2)
        print(f"[{time.time():.1f}] (fallback) score = {final_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the sequence as plain Python floats.
    # ------------------------------------------------------------------
    return a.tolist()
