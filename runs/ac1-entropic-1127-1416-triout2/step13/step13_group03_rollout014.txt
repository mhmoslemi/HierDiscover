"""
Propose a coefficient sequence for ``evaluate_sequence``.

The optimiser has three stages:

0.  Try a known fallback (if present).
1.  A very wide *global* search:
    - many classic parametric families (power‑law, exponential, beta, …)
    - a large set of signal‑processing windows (Hann, Hamming, Blackman,
      Kaiser, Tukey, Chebwin, DPSS, …) with a sweep over their shaping
      parameters,
    - many Dirichlet draws with concentrations ranging from extremely
      sparse to almost uniform.
    All candidates are projected onto the probability simplex (∑a_i = 1)
    and the best ``max(conv(a,a))`` values are kept.

2.  Local refinement of the best candidates:
    * Adam optimiser on the smooth‑max surrogate of the max‑convolution,
      with an exponential temperature schedule.
    * Projected sub‑gradient descent on the exact max‑objective.
    * Simulated‑annealing hill‑climb (mass transfers and swaps).

All phases respect the 1000 s wall‑clock budget; the best sequence found
is returned as a plain list of ``float``.  The code makes heavy use of
FFT‑based convolution for speed and works with any vector length (here
``n = 4096`` gives a good balance between search freedom and runtime).
"""

import time
import numpy as np
from scipy.signal import windows

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        # all non‑positive -> uniform simplex point
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based full linear convolution a*a
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two >= m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum(a)=1:
        score = 2 * n * max(conv(a,a))
    (this equals ``evaluate_sequence`` up to the invariant scaling).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (non‑negative, summed to 1 after projection)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    a /= a.sum()
    return a


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    a /= a.sum()
    return a


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return a


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    a /= a.sum()
    return a


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·vec1 + (1−λ)·vec2, projected onto the simplex."""
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Window‑based families (all non‑negative, later projected)
# ----------------------------------------------------------------------


def generate_window(name: str, n: int, param=None) -> np.ndarray:
    """Create a window of length n. ``param`` is window specific."""
    # Most windows return non‑negative values; if any negatives appear,
    # they are clipped by the simplex projection later.
    if name == "hann":
        return windows.hann(n, sym=False)
    if name == "hamming":
        return windows.hamming(n, sym=False)
    if name == "blackman":
        return windows.blackman(n, sym=False)
    if name == "nuttall":
        return windows.nuttall(n, sym=False)
    if name == "flattop":
        return windows.flattop(n, sym=False)
    if name == "parzen":
        return windows.parzen(n, sym=False)
    if name == "bartlett":
        return windows.bartlett(n, sym=False)
    if name == "kaiser":
        beta = float(param) if param is not None else 0.0
        return windows.kaiser(n, beta)
    if name == "tukey":
        alpha = float(param) if param is not None else 0.5
        return windows.tukey(n, alpha)
    if name == "chebwin":
        at = float(param) if param is not None else 30.0
        return windows.chebwin(n, at=at)
    if name == "dpss":
        # DPSS is only available in recent SciPy; guard against ImportError.
        try:
            # returns shape (Kmax, n); use the first (principal) taper.
            a = windows.dpss(n, float(param), Kmax=1, sym=False)[0]
            return a
        except Exception:
            # fallback to uniform if DPSS unavailable
            return np.full(n, 1.0 / n)
    # unknown -> uniform
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Global exploration – keep the best ``max_keep`` vectors
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 30,
    random_draws: int = 30000,
) -> list[np.ndarray]:
    """Explore a huge variety of parametric families and Dirichlet draws."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- classic families -------------------------------------------------
    # power‑law
    for p in np.linspace(0.1, 5.0, 101):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential
    for lam in np.linspace(0.0001, 0.8, 140):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta (grid + random draws)
    al_vals = np.linspace(0.02, 2.0, 31)
    be_vals = np.linspace(0.02, 2.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    for _ in range(300):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2, 0.3)   # 0.01 … 2
        beta = 10 ** np.random.uniform(-2, 0.3)
        try_candidate(generate_beta(n, alpha, beta))

    # Gaussian
    for sigma in np.linspace(0.05 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # sine‑power
    for p in np.linspace(0.1, 8.0, 40):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # ---- window families --------------------------------------------------
    window_names = [
        "hann", "hamming", "blackman", "nuttall", "flattop",
        "parzen", "bartlett"
    ]
    for name in window_names:
        if time.time() >= deadline:
            break
        try_candidate(generate_window(name, n))

    # Kaiser (β sweep)
    for beta in np.linspace(0.0, 14.0, 28):
        if time.time() >= deadline:
            break
        try_candidate(generate_window("kaiser", n, beta))

    # Tukey (α sweep)
    for alpha in np.linspace(0.0, 1.0, 21):
        if time.time() >= deadline:
            break
        try_candidate(generate_window("tukey", n, alpha))

    # Chebwin (attenuation sweep)
    for at in np.linspace(20.0, 120.0, 21):
        if time.time() >= deadline:
            break
        try_candidate(generate_window("chebwin", n, at))

    # DPSS (NW sweep) – only if SciPy supplies it
    for NW in np.linspace(2.0, 7.0, 11):
        if time.time() >= deadline:
            break
        try_candidate(generate_window("dpss", n, NW))

    # ---- mixtures ---------------------------------------------------------
    for _ in range(250):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.1, 5.0))
        b = generate_beta(
            n,
            10 ** np.random.uniform(-2, 0.3),
            10 ** np.random.uniform(-2, 0.3),
        )
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # ---- Dirichlet draws --------------------------------------------------
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    # keep only the vectors (sorted by score)
    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-5,
    tau_decay: float = 0.95,
    lr_start: float = 0.15,
    lr_decay: float = 0.98,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # ----- smooth‑max weights (soft‑max over the convolution) -----
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()

        # ----- gradient of the smooth‑max w.r.t. a -----
        #  ∇ fτ(a) = 2 * (a ⋆ w)   where ⋆ denotes cross‑correlation.
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # ----- Adam update -----
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-4)

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (exact max‑objective)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # Sub‑gradient: sum of contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (mass transfers + occasional swaps)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9993,
                        step_scale_init: float = 0.06,
                        step_scale_min: float = 5e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # ----- mass transfer move -----
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        # ----- occasional swap move (helps to re‑order large values) -----
        if it % 500 == 0:
            p = np.random.choice(n, 2, replace=False)
            cand = a.copy()
            cand[p[0]], cand[p[1]] = cand[p[1]], cand[p[0]]
            cand = project_to_simplex(cand)
            val = compute_score(cand)
            dval = val - cur_val
            if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
                a = cand
                cur_val = val
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()

        # ----- temperature decay -----
        T = max(final_temp, T * cooling_factor)

        # ----- periodic jitter to avoid stagnation -----
        if it % 7000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point for the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 4‑second safety margin for the harness
    deadline = start_time + budget_s - 4.0

    # -----------------------------------------------------------------
    # 0. Optional fallback (if the environment provides a known good construction)
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # may be undefined
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (large portion of the time budget)
    # -----------------------------------------------------------------
    n = 4096  # length of the candidate vectors
    baseline_deadline = deadline - 300.0  # leave ~5 min for refinement phases
    if time.time() < baseline_deadline:
        print("[search] starting global baseline exploration")
        pool = baseline_search(
            n=n,
            deadline=baseline_deadline,
            max_keep=30,
            random_draws=40000,
        )
        for vec in pool:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
        print(f"[baseline] best score after global search = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam‑based smooth‑max optimisation (multiple restarts)
    # -----------------------------------------------------------------
    if best_seq is not None:
        grad_deadline = deadline - 120.0
        if time.time() < grad_deadline:
            print("[adam] starting smooth‑max refinement")
            start_vecs = []

            # best vector from baseline
            best_vec = np.array(best_seq, dtype=float)
            best_vec = project_to_simplex(best_vec)
            start_vecs.append(best_vec)

            # uniform starter
            start_vecs.append(np.full(n, 1.0 / n))

            # a few random starters
            for _ in range(4):
                start_vecs.append(project_to_simplex(np.random.rand(n)))

            # a few Dirichlet random starters (different concentration)
            for _ in range(2):
                conc = np.random.choice([0.01, 0.1, 1.0, 5.0])
                start_vecs.append(project_to_simplex(np.random.dirichlet([conc] * n)))

            refined_best = best_vec.copy()
            refined_val = best_val

            for idx, vec in enumerate(start_vecs):
                if time.time() >= grad_deadline:
                    break
                # each Adam run gets an equal slice of the remaining time
                sub_deadline = min(
                    grad_deadline,
                    time.time() + (grad_deadline - time.time()) / max(1, len(start_vecs) - idx)
                )
                a_ref = smooth_max_adam(vec, sub_deadline)
                val = compute_score(a_ref)
                print(f"[adam {idx}] refined score = {val:.6f}")
                if val < refined_val:
                    refined_val, refined_best = val, a_ref.copy()

            best_val, best_seq = refined_val, refined_best.tolist()
            print(f"[adam] best after smooth optimisation = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (exact max objective)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 30.0
    if best_seq is not None and time.time() < polish_deadline:
        print("[polish] starting projected sub‑gradient polishing")
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_refined)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing / stochastic hill‑climb
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        print("[anneal] starting simulated‑annealing hill‑climb")
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported official score = {final_score:.6f}")
    return best_seq
