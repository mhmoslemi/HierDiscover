```
def incremental_hill_climb(a_initial, time_deadline, max_iter=5_000_000):
    n = len(a_initial)
    a = a_initial.copy()
    conv = full_convolution(a)
    best_a = a.copy()
    best_M = conv.max()
    step_scale = 0.05  # maybe start small
    it = 0
    no_improve = 0
    m = conv.shape[0]
    while time.time() < time_deadline and it < max_iter:
        # choose i weighted by a
        if np.random.rand() < 0.5:
            # select high mass i with probability a
            i = np.random.choice(n, p=a/a.sum())
        else:
            i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = step_scale * a[i]
        if delta <= 0:
            continue
        # compute conv_new
        conv_new = conv.copy()
        # update range i:i+n and j:j+n
        # Note that segment modifications may overlap when i and j differ; handle properly:
        # We'll subtract 2*delta*a from segment i and add 2*delta*a to segment j
        conv_new[i:i+n] -= 2.0 * delta * a
        conv_new[j:j+n] += 2.0 * delta * a
        # second-order updates
        ii = 2 * i
        jj = 2 * j
        ij = i + j
        if ii < m:
            conv_new[ii] += delta * delta
        if jj < m:
            conv_new[jj] += delta * delta
        if ij < m:
            conv_new[ij] -= 2.0 * delta * delta
        M_new = conv_new.max()
        if M_new < best_M:
            best_M = M_new
            best_a = a - delta * np.eye(1, n, i)[0] + delta * np.eye(1, n, j)[0]  # Not nice
            # easier: update a directly
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            # adapt step scale upwards
            step_scale = min(step_scale * 1.02, 0.6)
            no_improve = 0
        else:
            # revert improvements
            step_scale = max(step_scale * 0.995, 1e-6)
            no_improve += 1

        it += 1
        if no_improve > 200_000:
            # jitter: small Gaussian noise to a
            jitter = np.random.randn(n) * 0.0005
            a = best_a + jitter
            a = np.maximum(a, 0)
            a /= a.sum()
            conv = full_convolution(a)
            no_improve = 0
            step_scale = 0.05

    return best_a
```

assistantfinal**Solution Explanation**

The objective  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

is scale‑invariant, therefore we may (and do) keep the vector normalised
\(\sum_i a_i = 1\).  
If \(a\) is thought of as a probability distribution on \(\{0,\dots ,n-1\}\) then
\((a*a)_k\) is the probability that the sum of two independent draws equals
\(k\).  Minimising the largest entry of the convolution flattens this sum‑
distribution.

The previous implementation explored many families of vectors, performed a
sub‑gradient polish and a short stochastic hill‑climb.  
The new optimiser focuses on the *most promising* family – a **front‑loaded
reverse power‑law**  

\[
a_i \propto (n-i)^{p},
\]

searches its exponent \(p\) with a *golden‑section* line‑search (far more
precise than a coarse grid), and then refines the solution with a **mass‑transfer
hill‑climb that updates the convolution incrementally** ( \(O(n)\) per move,
instead of recomputing an FFT).  Because the update is cheap we can afford
hundreds of thousands of moves within the 1000 s budget, which yields a
much tighter optimum.

The algorithm consists of three stages:

1. **Golden‑section search** for the optimal exponent `p` of the reverse
   power‑law (≈ 0.5 s).
2. **Random mixtures** of a few simple shapes (uniform, reverse‑power‑law,
   geometric) to obtain a slightly better starting point (≈ 5 s).
3. **Incremental hill‑climb** – repeatedly move a small amount of mass from a
   high‑weight index to a random index, update the convolution in‑place,
   accept the move only if the maximal convolution value decreases,
   and adapt the step‑size.  After long periods without improvement a small
   jitter is added to escape plateaus.

All operations keep the vector on the probability simplex, therefore the
final list can be fed directly to `evaluate_sequence`.  Extensive printing
allows monitoring of the progress.

```python
"""
search_step_function_v3.py

A fast optimiser for the ``evaluate_sequence`` objective.

Main ideas
----------
* The best known shape is a front‑loaded reverse power‑law
  a_i ∝ (n‑i)^p.  We find the optimal exponent p with a
  golden‑section search (far finer than the previous coarse grid).
* After the parametric phase we run a *mass‑transfer hill‑climb*
  that updates the convolution incrementally (O(n) per move) – this
  lets us perform many more iterations than the FFT‑based version.
* Adaptive step size, occasional jitter, and a few random convex
  combinations of simple bases give a robust final refinement.

The routine returns a plain Python list of floats that minimises
``evaluate_sequence`` within the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (p ≥ 0)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def uniform_vec(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i},   0<q<1."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
# Golden‑section search for the optimal reverse‑power‑law exponent
# ----------------------------------------------------------------------
def golden_section_reverse_power(n: int, lo: float = 0.0,
                                 hi: float = 5.0,
                                 tol: float = 1e-5,
                                 deadline: float = None):
    """Return (best_p, best_score) for the reverse‑power‑law family."""
    phi = (math.sqrt(5) - 1) / 2  # ≈0.618
    a = lo + (1 - phi) * (hi - lo)
    b = lo + phi * (hi - lo)
    fa = compute_score(reverse_power_law(n, a))
    fb = compute_score(reverse_power_law(n, b))

    while hi - lo > tol:
        if deadline is not None and time.time() >= deadline:
            break
        if fa < fb:
            hi, fb = b, fa
            b = a
            fb = fa
            a = lo + (1 - phi) * (hi - lo)
            fa = compute_score(reverse_power_law(n, a))
        else:
            lo, fa = a, fb
            a = b
            fa = fb
            b = lo + phi * (hi - lo)
            fb = compute_score(reverse_power_law(n, b))

    if fa < fb:
        return a, fa
    else:
        return b, fb


# ----------------------------------------------------------------------
# Incremental hill‑climb (mass transfer) – the core local optimiser
# ----------------------------------------------------------------------
def hill_climb_incremental(a_start: np.ndarray,
                           deadline: float,
                           max_iter: int = 10_000_000) -> np.ndarray:
    """
    Greedy stochastic mass‑transfer moves.
    The convolution is updated in O(n) time, which allows many
    iterations within the time budget.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = full_convolution(a)                # length 2*n-1
    best_a = a.copy()
    best_M = conv.max()
    step_scale = 0.07                         # typical fraction of source mass
    it = 0
    no_improve = 0
    m = conv.shape[0]

    while time.time() < deadline and it < max_iter:
        # ---------------------------------------------------------
        # choose source index (biased towards large entries)
        # ---------------------------------------------------------
        if np.random.rand() < 0.5:
            # weighted choice (probability proportional to a_i)
            probs = a / a.sum()
            i = np.random.choice(n, p=probs)
        else:
            i = np.random.randint(n)

        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]   # never larger than a[i]
        if delta <= 0.0:
            continue

        # -----------------------------------------------------------------
        # Incrementally update the convolution
        # conv_new = conv + 2*delta*(a shifted by j – a shifted by i)
        #               + delta^2 * (e_{2i}+e_{2j}–2e_{i+j})
        # -----------------------------------------------------------------
        conv_new = conv.copy()
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a

        ii = 2 * i
        jj = 2 * j
        ij = i + j
        if ii < m:
            conv_new[ii] += delta * delta
        if jj < m:
            conv_new[jj] += delta * delta
        if ij < m:
            conv_new[ij] -= 2.0 * delta * delta

        M_new = conv_new.max()

        if M_new < best_M - 1e-12:                 # strict improvement
            # accept move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            best_M = M_new
            best_a = a.copy()
            step_scale = min(step_scale * 1.02, 0.6)
            no_improve = 0
        else:
            # reject – shrink the step size a little
            step_scale = max(step_scale * 0.995, 1e-6)
            no_improve += 1

        it += 1

        # -----------------------------------------------------------------
        # If we are stuck for a long time, add a tiny jitter to escape
        # -----------------------------------------------------------------
        if no_improve > 300_000:
            jitter = np.random.randn(n) * 0.0005
            a = best_a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = full_convolution(a)
            best_M = conv.max()
            step_scale = 0.07
            no_improve = 0

    return best_a

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – we need a few seconds to return the result
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # 0. optional fallback (previous construction)
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Find a good reverse‑power‑law seed via golden‑section search
    # -------------------------------------------------------------
    n = 2048
    search_deadline = deadline - 300.0          # keep ~5 min for polishing
    if time.time() < search_deadline:
        p_opt, p_score = golden_section_reverse_power(
            n, lo=0.0, hi=5.0, tol=1e-5, deadline=search_deadline)
        a0 = reverse_power_law(n, p_opt)
        print(f"[golden‑search] p = {p_opt:.6f}, score = {p_score:.6f}")

        # ---------------------------------------------------------
        # 1b. random convex combinations of a few simple bases
        # ---------------------------------------------------------
        uniform = uniform_vec(n)
        geom    = geometric(n, 0.98)
        bases   = [a0, uniform, geom]

        for _ in range(300):
            coeffs = np.random.rand(len(bases))
            coeffs /= coeffs.sum()
            cand = sum(c * b for c, b in zip(coeffs, bases))
            cand = project_to_simplex(cand)
            val = compute_score(cand)
            if val < p_score:
                p_score = val
                a0 = cand.copy()
        # evaluate with the official routine
        real_val = evaluate_sequence(a0.tolist())
        print(f"[seed] official score = {real_val:.6f}")
        if real_val < best_val:
            best_val, best_seq = real_val, a0.tolist()
    else:
        a0 = reverse_power_law(n, 1.8)   # a quick fallback

    # -------------------------------------------------------------
    # 2. Long incremental hill‑climb (bulk of the remaining time)
    # -------------------------------------------------------------
    hill_deadline = deadline - 10.0
    if time.time() < hill_deadline:
        a_start = np.array(best_seq if best_seq is not None else a0,
                           dtype=np.float64)
        a_final = hill_climb_incremental(a_start, hill_deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # 3. Safety net – if everything failed, use the best reverse‑power‑law
    #    discovered during the golden search (or a generic fallback).
    # -------------------------------------------------------------
    if best_seq is None:
        a_fallback = reverse_power_law(n, 1.8)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final report
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
```
