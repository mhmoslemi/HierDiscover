```
def propose_candidate(...):
    import time, numpy as np
    start = time.time()
    deadline = start+budget_s-5

    n=4096
    # start with random Dirichlet
    a = np.random.dirichlet(np.ones(n))
    best_a = a.copy()
    best_val = compute_score(a)

    eta = 0.5
    decay = 0.9995
    eps = 1e-12
    iter = 0
    while time.time()<deadline:
        conv = full_convolution(a)
        max_val = conv.max()
        # compute active indices
        # tolerance maybe relative
        tol = max_val*1e-12  # small tolerance
        active = np.where(conv >= max_val - tol)[0]
        # compute subgradient
        g = np.zeros_like(a)
        idx_range = np.arange(n)
        for k in active:
            j = k - idx_range
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # exponentiated gradient update
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.random.rand(n)
            a = a / a.sum()
        else:
            a = a / a_sum
        # evaluate
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            # maybe print progress
            print(f"iter {iter} best {best_val:.6f}")

        # update eta
        eta *= decay
        # occasionally add jitter
        if iter % 1000 == 0:
            jitter = np.random.randn(n) * (eta * 0.01)
            a = a + jitter
            a = np.maximum(a, 0)
            a = a / a.sum()
        iter += 1
    return best_a.tolist()
```

assistantfinal**Solution Overview**

The objective is scale‑invariant, therefore we work with probability vectors
(`sum(a)=1`).  
For a vector `a` the score is  

```
score(a) = 2·n·max_k (a * a)[k]               (n = len(a))
```

where `*` is the linear convolution.  
The problem is to **flatten the autocorrelation** of `a`.  We use a
mirror‑descent (exponentiated‑gradient) method on the exact `max` – this
keeps all entries non‑negative and respects the simplex constraint by a
simple normalisation step.

The search consists of three stages

1. **Random restarts** – Dirichlet draws with a wide range of concentration
   parameters give both dense and very sparse starts.
2. **Exponentiated‑gradient optimisation** – at each iteration we
   * compute the full convolution (FFT based, O(n log n)),
   * locate the index/indices attaining the maximal value,
   * form the sub‑gradient `g_i = 2·∑_{j+i=k_max} a_j`,
   * update `a_i ← a_i·exp(-η·g_i)` and renormalise.
   The step size `η` is decayed geometrically; occasional Gaussian jitter
   prevents stagnation.
3. **Final reporting** – the best vector found is returned as a plain list
   of floats; the official `evaluate_sequence` is called once for the final
   score.

All helper routines are top‑level, there are no closures or lambdas and the
code respects the 1000 s wall‑clock budget.

```python
"""
Mirror‑descent optimiser for the convolution‑max problem.

We minimise 2·n·max(conv(a,a)) under a≥0, sum(a)=1 by a projected
exponentiated‑gradient (mirror‑descent w.r.t. KL) on the exact ℓ∞‑norm.
Multiple random Dirichlet restarts are performed; the best vector found
within the time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # next power‑of‑two for efficient FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        2 * n * max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser
# ----------------------------------------------------------------------


def eg_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One exponentiated‑gradient step that reduces the maximum entry of the
    convolution.  Returns the updated and renormalised vector.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    max_val = conv.max()
    # tolerance for “active” indices – usually only one index is active
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]

    # sub‑gradient of max_k conv_k w.r.t a
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        # contribution 2*a_j for every ordered pair (i,j) with i+j=k
        g[mask] += 2.0 * a[j[mask]]

    # exponentiated‑gradient update (mirror descent on the simplex)
    a = a * np.exp(-eta * g)
    # renormalise to stay on the probability simplex
    s = a.sum()
    if s == 0.0:
        # numerical under‑flow – fall back to a tiny uniform perturbation
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final report
    deadline = start_time + budget_s - 5.0

    # Length of the candidate vector – a larger n gives a tighter theoretical bound.
    n = 4096

    best_a = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional cheap parametric seed (uniform) – guarantees a start.
    # -----------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = compute_score(best_a)

    # -----------------------------------------------------------------
    # 1. Random Dirichlet restarts + mirror‑descent.
    # -----------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 2.0:
        # ----------- draw a random start on the simplex -----------------
        # concentration parameter spans several orders of magnitude
        conc = 10 ** np.random.uniform(-2.0, 0.5)   # from 0.01 to ~3.2
        a = np.random.dirichlet(np.full(n, conc))

        cur_score = compute_score(a)
        if cur_score < best_score:

        eta = 0.7                 # initial step size

        iteration = 0

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    # 3. Final evaluation with the official evaluator.
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\nSearch finished. Best raw score = {best_score:.6f}")
    print(f"Reported evaluate_sequence score = {final_score:.6f}")
```
