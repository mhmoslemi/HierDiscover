def smooth_max_gradient_descent(a0, deadline, alpha_initial=2.0, alpha_final=16.0):
    n = len(a0)
    u = np.log(a0 + 1e-20)
    m = v = np.zeros_like(u)
    best_a = a0.copy()
    best_val = compute_score(a0)

    step = 0.1 maybe.

    while time.time() < deadline:
        step += 1
        # compute a
        max_u = np.max(u); exp_shift = np.exp(u - max_u); a = exp_shift / np.sum(exp_shift)

        # convolution for w
        conv = full_convolution(a)
        # schedule alpha
        alpha = alpha_initial + (alpha_final - alpha_initial) * (step / 10000)  # saturates at alpha_final
        # compute w (softmax of alpha * conv)
        tmp = alpha * conv
        max_tmp = np.max(tmp)
        w = np.exp(tmp - max_tmp)
        w_sum = np.sum(w)
        w /= w_sum

        # compute gradient wrt a
        # compute FFT lengths
        M = conv.shape[0]  # 2*n-1
        n_fft = 1 << ((3 * n - 2 - 1).bit_length())
        # Compute FFT of w once
        W = np.fft.rfft(w, n=n_fft)

        A = np.fft.rfft(a, n=n_fft)
        conv1 = np.fft.irfft(A * W, n=n_fft)[:n]

        a_rev = a[::-1]
        Ar = np.fft.rfft(a_rev, n=n_fft)
        conv2_full = np.fft.irfft(Ar * W, n=n_fft)
        conv2 = conv2_full[(n-1):(n-1 + n)]

        g_a = conv1 + conv2
        # gradient w.r.t u
        g_u = a * (g_a - np.dot(g_a, a))

        # Adam update
        beta1 = 0.9; beta2 = 0.999; eps = 1e-8
        m = beta1 * m + (1 - beta1) * g_u
        v = beta2 * v + (1 - beta2) * (g_u * g_u)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        # step size
        lr = 0.05 / np.sqrt(step + 1)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate actual objective at a (maybe at intervals)
        if step % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth] step {step} best {best_val:.6f}")

    return best_a

def smooth_adam_polish(a0, deadline, init_lr=0.2, alpha0=2.0, alpha_max=12.0):
    n = len(a0)
    u = np.log(a0 + 1e-20)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    best_a = a0.copy()
    best_val = compute_score(a0)
    t = 0
    n_fft = 1 << ((3*n - 2 - 1).bit_length())
    while time.time() < deadline:
        t += 1
        # compute a via softmax
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / np.sum(exp_shift)

        # compute conv and w
        conv = full_convolution(a)
        # schedule alpha
        frac = min(1.0, t / 2000.0)  # ramp up to alpha_max in 2000 iterations
        alpha = alpha0 + (alpha_max - alpha0) * frac

        # softmax weights
        tmp = alpha * conv
        max_tmp = np.max(tmp)
        w = np.exp(tmp - max_tmp)
        w_sum = np.sum(w)
        w /= w_sum

        # FFT for gradient
        W = np.fft.rfft(w, n=n_fft)
        A = np.fft.rfft(a, n=n_fft)
        conv1 = np.fft.irfft(A * W, n=n_fft)[:n]
        a_rev = a[::-1]
        Ar = np.fft.rfft(a_rev, n=n_fft)
        conv2_full = np.fft.irfft(Ar * W, n=n_fft)
        conv2 = conv2_full[(n-1):(n-1+n)]

        g_a = conv1 + conv2

        # gradient in u space
        g_u = a * (g_a - np.dot(g_a, a))

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m = beta1 * m + (1 - beta1) * g_u
        v = beta2 * v + (1 - beta2) * (g_u * g_u)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = init_lr * (1.0 / np.sqrt(t + 1))
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # periodic evaluation
        if t % 100 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[smooth_adam] iteration {t} best {best_val:.6f}")
        # occasional jitter
        if t % 2000 == 0:
            u += np.random.randn(n) * (init_lr * 0.001)

    return best_a
```

"""
Non‑negative step‑function optimiser for the supplied ``evaluate_sequence``.
The search consists of three stages:

1. **Global exploration** – a quick sweep of three parametric families
   (power‑law, exponential, beta) together with many Dirichlet draws.
   The ten best candidates (by the exact objective) are kept.

2. **Smooth‑max Adam polishing** – the objective ``max_k (a*a)_k`` is
   approximated by a log‑sum‑exp (smooth maximum).  The gradient of the
   smoothed objective is obtained analytically via two FFT‑based
   convolutions.  An Adam optimiser (mirror‑descent on the soft‑max
   parametrisation) refines each candidate for the rest of the budget.
   Random restarts continue while time permits.

3. **Optional simulated‑annealing finish** – a short Metropolis walk is
   performed on the current best vector to escape possible plateaux.

All stages respect the 1000 s wall‑clock limit.  The routine returns the
best sequence found as a plain list of floats, ready for ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic convolution utilities (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())       # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the global sweep)
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape:
       a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global baseline exploration
# ----------------------------------------------------------------------
def baseline_search(n: int, deadline: float, max_keep: int = 10):
    """Sweep families and a pool of Dirichlet draws, keeping the best."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.clip(vec, 0.0, None)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep
    for p in np.linspace(0.05, 6.0, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(1e-5, 0.25, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep – generous ranges
    al_vals = np.concatenate([np.geomspace(0.01, 0.5, 15),
                              np.linspace(0.5, 5.0, 25)])
    be_vals = np.concatenate([np.geomspace(0.01, 0.5, 15),
                              np.linspace(0.5, 5.0, 25)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) massive Dirichlet draws (fewer per concentration for large n)
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 800
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(np.full(n, conc))
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max Adam polishing
# ----------------------------------------------------------------------
def smooth_adam_polish(a0: np.ndarray, deadline: float,
                       init_lr: float = 0.20,
                       alpha_start: float = 2.0,
                       alpha_end: float = 12.0,
                       max_iters: int = 10_000) -> np.ndarray:
    """
    Adam optimiser on the smoothed maximum of the convolution.

    a0 – initial probability vector (sum = 1, entries ≥ 0)
    deadline – wall‑clock time at which the run must stop
    Returns the best vector encountered.
    """
    n = a0.shape[0]
    # Soft‑max parametrisation
    u = np.log(a0 + 1e-20)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = a0.copy()
    best_val = compute_score(best_a)

    t = 0
    # FFT length sufficient for both a*w and a_rev*w (size 3n‑2)
    n_fft = 1 << ((3 * n - 2 - 1).bit_length())

    while time.time() < deadline and t < max_iters:
        t += 1

        # ---- soft‑max projection (always on the simplex) ----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ---- convolution a*a (needed for the smooth max) ----
        A_full = np.fft.rfft(a, n=n_fft)
        conv_full = np.fft.irfft(A_full * A_full, n=n_fft)[: 2 * n - 1]

        # ---- smooth‑max weights w_k ∝ exp(alpha * conv_k) ----
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / 2000.0)
        tmp = alpha * conv_full
        max_tmp = np.max(tmp)
        w = np.exp(tmp - max_tmp)
        w /= w.sum()

        # ---- FFT of the weight vector (used twice) ----
        W = np.fft.rfft(w, n=n_fft)

        # ---- first convolution term  Σ_j a_j w_{i‑j} ----
        conv1 = np.fft.irfft(A_full * W, n=n_fft)[:n]

        # ---- second convolution term  Σ_j a_j w_{i+j} ----
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=n_fft)
        conv2_full = np.fft.irfft(A_rev * W, n=n_fft)
        conv2 = conv2_full[(n - 1) : (n - 1 + n)]

        # ---- gradient w.r.t. a (sub‑gradient of the smoothed max) ----
        g_a = conv1 + conv2

        # ---- mirror‑descent gradient (soft‑max Jacobian) ----
        g_u = a * (g_a - np.dot(g_a, a))

        # ---- Adam update on the log‑space variables ----
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = init_lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional jitter to leave plateaux ----
        if t % 3000 == 0:
            u += np.random.randn(n) * (init_lr * 0.001)

        # ---- keep the best exact score seen so far ----
        if t % 200 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[smooth‑adam] iter {t:5d} best {best_val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing (final polish)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------ problem size ------------------------------
    n = 2048                     # a good trade‑off between quality and speed

    best_seq = None
    best_val = float("inf")

    # ------------------- optional fallback from previous run ------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # -------------------------- global sweep ---------------------------
    baseline_deadline = deadline - 300.0   # keep most time for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=8)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best score = {best_val:.6f}")

    # ------------------------ smooth‑Adam polishing --------------------
    polish_deadline = deadline - 60.0
    if best_seq is not None and time.time() < polish_deadline:
        # first polish the current best
        a0 = np.array(best_seq, dtype=float)

        # random restarts while time permits

    # ----------------------- final annealing polish --------------------
    anneal_deadline = deadline - 5.0

    # --------------------------- final report ---------------------------

    return best_seq
