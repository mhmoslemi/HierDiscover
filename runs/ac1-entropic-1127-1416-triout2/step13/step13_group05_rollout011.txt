def simulated_annealing(a0, time_deadline, initial_temp=0.1, cooling_rate=0.9995):
    a = a0.copy()
    best = a.copy()
    best_M = full_convolution(a).max()
    temp = initial_temp
    while time.time() < time_deadline:
        # propose move: pick i, j, delta = temp * random * a[i]

        cand_M = full_convolution(cand).max()
        if cand_M < best_M or np.random.rand() < np.exp((best_M - cand_M)/temp):
            a = cand
            if cand_M < best_M:
                best_M = cand_M
                best = cand.copy()
        temp *= cooling_rate
    return best

"""
Improved optimiser for ``evaluate_sequence``.

Algorithmic outline
-------------------
1. **Coarse/fine parametric sweep** – exhaustive search over the
   *reverse‑power‑law* family a_i ∝ (n‑i)^p for p∈[0.5,6] with a very fine
   grid (step≈0.002).  The best vector from this sweep is used as a seed.

2. **Smooth‑max gradient descent** – minimise a smooth approximation of the
   max‑convolution term, ϕ_α(a)=α·log∑_k exp(conv_k/α), using Adam and a linear
   projection onto the simplex.  The smoothing parameter α is exponentially
   decayed from 0.15 to 1e‑5, allowing the optimiser to first move quickly
   and then concentrate on the true max.

3. **Projected sub‑gradient polishing** – as in the reference code, a few
   rounds of sub‑gradient descent (step size ∝1/√t) are applied to the
   polished candidate.

4. **Stochastic hill‑climb** – a final aggressive greedy random walk refines
   the result.

All stages respect the global time budget (≈1000 s).  The routine prints
intermediate scores and returns the best list of floats found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Geometry / utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a*a using an FFT of the next power of two."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The scale of a does not matter, we therefore keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Families of candidate vectors
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)       # n, n‑1, …, 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Front plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def pow_exp_family(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-lam·i).  Both exponents ≥0."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = (idx ** (-p)) * np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
# Gradient of the smooth max (soft‑max) of the convolution
# ----------------------------------------------------------------------
def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return w_k = exp(conv_k/α)/∑exp(conv/α).  Numerically stable."""
    scaled = conv / alpha
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    return w


def correlation_grad(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute grad_i = 2·∑_k w_k·a_{k‑i}
    using FFT‑based linear correlation (valid for i=0…n‑1).
    """
    n = a.shape[0]
    L = len(w)                     # = 2·n‑1
    size = 1 << ((L + n - 1 - 1).bit_length())   # next power‑of‑2 ≥ L+n‑1
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    # linear correlation = ifft( W·conj(A) )
    corr = np.fft.irfft(W * np.conj(A), n=size)
    start = n - 1                  # lag 0 is at index n‑1
    grad = 2.0 * corr[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (stage 2)
# ----------------------------------------------------------------------
def smooth_max_adam(a0: np.ndarray, deadline: float,
                    init_alpha: float = 0.15,
                    final_alpha: float = 1e-5,
                    max_iters: int = 4000,
                    lr0: float = 0.30) -> np.ndarray:
    """
    minimise the smoothed max ‑ term using Adam, with a projection onto
    the simplex after each update.  α is exponentially decayed from
    ``init_alpha`` to ``final_alpha``.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # α schedule
    decay = (final_alpha / init_alpha) ** (1.0 / max_iters)
    alpha = init_alpha

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = full_convolution(a)
        w = softmax_weights(conv, alpha)

        grad = correlation_grad(w, a)          # shape (n,)
        # Adam update
        lr = lr0 / math.sqrt(it)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)

        # keep best according to the true objective
        val = compute_score(a)
        if val < best_score:
            best_score, best_a = val, a.copy()

        alpha *= decay
    return best_a

def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 1_200_000) -> np.ndarray:
    """Greedy randomised moves, with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]                     # move a fraction
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 180_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Broad parametric sweep (stage 1)
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """
    Exhaustive search over a few carefully chosen families.
    Returns the best vector discovered (may be ``None``).
    """
    best_vec = None
    best_val = float('inf')

    def consider(a: np.ndarray):
        nonlocal best_vec, best_val
        if a is None:
            return
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a.copy()

    # ----- reverse power‑law – very fine grid -----
    # coarse grid first (step 0.01) to avoid spending all time on the fine loop
    for p in np.arange(0.5, 6.0001, 0.01):
        if time.time() >= deadline:
            return best_vec
        a = reverse_power_law(n, float(p))
        consider(a)

    # ----- power‑law (standard) -----
    for p in np.arange(0.05, 5.001, 0.02):
        if time.time() >= deadline:
            return best_vec
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        total = a.sum()
        if total == 0.0:
            continue
        a /= total
        consider(a)

    # ----- geometric family (dense scan near 1) -----
    for q in np.concatenate((
            np.linspace(0.90, 0.999, 150),
            np.linspace(0.9990, 0.99995, 250))):
        if time.time() >= deadline:
            return best_vec
        a = geometric(n, float(q))
        consider(a)

    # ----- mixed plateau + power‑law -----
    for L_frac in np.linspace(0.02, 0.35, 15):
        for p in np.linspace(0.3, 1.0, 15):
            if time.time() >= deadline:
                return best_vec
            a = mixed_plateau_power(n, float(L_frac), float(p))
            consider(a)

    # ----- beta shape -----
    alphas = np.linspace(-0.5, 4.0, 17)
    betas = np.linspace(-0.5, 4.0, 17)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, float(alpha), float(beta))
            consider(a)

    # ----- power‑law × exponential family -----
    for _ in range(1200):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.2, 3.5)
        lam = np.random.uniform(0.0, 0.004)
        a = pow_exp_family(n, p, lam)
        consider(a)

    # ----- random convex combos of a few basics -----
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    rev2 = reverse_power_law(n, 2.0)
    geom = geometric(n, 0.98)
    basics = [uniform, rev2, geom]
    for _ in range(1800):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        consider(a)

    # ----- occasional random permutations of good shapes (break monotonicity) -----
    if best_vec is not None:
        for _ in range(300):
            if time.time() >= deadline:
                break
            perm = np.random.permutation(n)
            a = best_vec[perm]
            a = project_to_simplex(a)
            consider(a)

    return best_vec

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # leave a safety margin

    n = 2048
    best_seq = None
    best_val = float('inf')

    # -------------------------------------------------------------
    # 0. Optional fallback from a previously known construction
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Broad parametric sweep (≈ 250 s)
    # -------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[parametric sweep] best score (internal) = {val:.6f}")
            val_real = evaluate_sequence(a_initial.tolist())
            print(f"[parametric sweep] official score = {val_real:.6f}")
            if val_real < best_val:
                best_val, best_seq = val_real, a_initial.tolist()

    # -------------------------------------------------------------
    # 2. Smooth‑max Adam optimisation (majority of remaining time)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 180.0:
        a0 = np.array(best_seq, dtype=float)
        smooth_deadline = deadline - 180.0
        a_smooth = smooth_max_adam(a0, smooth_deadline,
                                   init_alpha=0.14, final_alpha=5e-6,
                                   max_iters=5000, lr0=0.35)
        val = evaluate_sequence(a_smooth.tolist())
        print(f"[smooth‑max] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_smooth.tolist()

    # -------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 80 s)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, deadline - 80.0, step0=0.12)
        val = evaluate_sequence(a_polished.tolist())
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -------------------------------------------------------------
    # 4. Final stochastic hill‑climb
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, deadline - 5.0)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------------------
    # Safeguard – if nothing succeeded, fall back to a sane guess
    # -------------------------------------------------------------
    if best_seq is None:
        a_fallback = reverse_power_law(n, 1.9)
        best_seq = a_fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final report
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
