def full_convolution(a, b):
    m = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

"""
Improved optimisation for the max‑self‑convolution bound.

Algorithm
---------
1.  **Broad parametric sweep** (n = 8192) over several analytic families
    (power‑law, geometric, plateau+power, beta, logistic, random Dirichlet)
    to obtain a decent starting point.
2.  **Smooth‑max Adam descent** – optimise a soft approximation of
    max(conv(a,a)) by the log‑sum‑exp trick.  The smoothing parameter α
    is increased gradually so the optimiser first sees a smooth landscape
    and finally concentrates on the true max‑convolution.
3.  **Projected sub‑gradient polish** – a few passes of the classic
    sub‑gradient method on the exact max‑convolution (as used in the
    reference implementation) for fine‑tuning.
4.  **Deterministic hill‑climb** – mass‑transfer moves on the simplex for
    the remaining time.

All vectors are kept on the probability simplex (∑ a_i = 1); thanks to the
scale‑invariance of `evaluate_sequence` the final score is simply
    2·n·max(conv(a,a)).
The routine respects the 1000 s budget and prints intermediate scores.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * b using real FFT.
    Length of the result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (same as the reference code, but usable for any n)
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def geometric(n: int, q: float) -> np.ndarray:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def logistic_shape(n: int, center_frac: float, scale_frac: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    scale = max(scale_frac * n, 1e-6)
    a = 1.0 / (1.0 + np.exp((i - center) / scale))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def random_dirichlet(n: int, concentration: float) -> np.ndarray:
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    if y.sum() == 0.0:
        return None
    return y / y.sum()

# Broad parametric sweep
# ----------------------------------------------------------------------
def parametric_sweep(n: int, time_deadline: float) -> np.ndarray:
    """Explore many families and keep the best vector found."""
    best_vec = None
    best_val = float('inf')

    # ---- power‑law ----------------------------------------------------
    for p in np.linspace(0.10, 2.00, 191):
        if time.time() >= time_deadline:
            return best_vec
        a = power_law(n, p)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- geometric ----------------------------------------------------
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 91),
            np.linspace(0.990, 0.9999, 110))):
        if time.time() >= time_deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- mixed plateau + power ---------------------------------------
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= time_deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- beta shape --------------------------------------------------
    alphas = np.linspace(-0.8, 3.0, 20)
    betas = np.linspace(-0.8, 3.0, 20)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= time_deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- logistic ----------------------------------------------------
    for center_frac in np.linspace(0.2, 0.8, 13):
        for scale_frac in np.linspace(0.01, 0.20, 20):
            if time.time() >= time_deadline:
                return best_vec
            a = logistic_shape(n, center_frac, scale_frac)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- random Dirichlet samples ------------------------------------
    for _ in range(1500):
        if time.time() >= time_deadline:
            break
        conc = 10 ** np.random.uniform(-2, 1)   # 0.01 … 10
        a = random_dirichlet(n, conc)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Smooth‑max Adam descent (softened max‑convolution)
# ----------------------------------------------------------------------
def smooth_adam_descent(a0: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    Adam optimiser on a smoothed approximation of max(conv(a,a)).
    The smoothing parameter α starts small and is increased to make the
    objective converge to the true max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    step_size = 0.05
    t = 0

    alpha = 1.0
    alpha_max = 200.0
    alpha_factor = 1.004   # gradually tighten the smooth approximation

    while time.time() < time_deadline:
        t += 1
        # 1) full self‑convolution
        conv = full_convolution(a, a)
        max_val = conv.max()

        # 2) soft‑max weights
        shifted = conv - max_val               # ≤ 0
        exp_vals = np.exp(alpha * shifted)    # safe – max entry is 1.0
        sum_exp = exp_vals.sum()
        p = exp_vals / sum_exp                # probability distribution over 2n‑1 elements

        # 3) gradient of the smoothed objective:
        #    g = 2 * (a ⋆ p)  where ⋆ denotes cross‑correlation.
        #    It can be obtained as g_i = 2 * conv_rev[idx - i],
        #    with conv_rev = a * p_rev  (full convolution) and idx = len(p)-1.
        p_rev = p[::-1]
        conv_rev = full_convolution(a, p_rev)          # length = n + len(p) - 1 = 3n‑2
        idx = len(p) - 1                               # = 2n‑2
        grad = 2.0 * conv_rev[idx - np.arange(n)]

        # 4) Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step_size * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # 5) increase smoothing strength slowly
        if alpha < alpha_max:
            alpha = min(alpha * alpha_factor, alpha_max)

        # 6) check real objective (max‑convolution, not the smoothed version)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[smooth] new best {best_score:.6f} it={t} time={time.time():.1f}s")

        # occasional step‑size decay to keep things stable
        if t % 2000 == 0:
            step_size *= 0.9

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (as used in the reference)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a, a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of each active entry: 2 * a shifted
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to escape shallow plateaux
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a, a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simple deterministic hill‑climb (mass‑transfer)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy mass‑transfer moves with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a, a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of the source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand, cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

    try:

    except Exception:

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    smooth_deadline = deadline - 250.0

    # --------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 150 s)
    # --------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.12)
        val = compute_score(a_polished)

        if val < best_val:

    # --------------------------------------------------------------
    # 4. Final hill‑climb (remaining time)
    # --------------------------------------------------------------
    hill_deadline = deadline
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)

    # --------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
