"""
propose_candidate
-----------------
Search for a non‑negative step‑function (vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Coarse analytic scan** – evaluate a few one‑parameter families
   (U‑shaped product, Beta) on a medium size vector (n≈16384) and keep
   the best parameters.

2. **Parameter refinement** – use golden‑section search on the product
   exponent and a simple hill‑climb on the two Beta parameters, both
   guarded by the global time budget.

3. **Fine stochastic local search** – starting from the best shape,
   repeatedly move a small fraction of mass from a random donor entry
   to a random receiver entry, enforce symmetry, renormalise and keep
   any improvement.  The step size is adapted automatically.

All stages respect a 1000 s wall‑clock limit; the best sequence found
is returned as a plain list of floats.  The implementation uses only
NumPy and the provided ``evaluate_sequence`` for the final official
score.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)**2).
    ``a`` is assumed non‑negative and (approximately) normalised to sum=1.
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    # work in log‑space to avoid overflow/underflow
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against infinities due to extreme α
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Discrete Beta‑type density on the grid.
    p,q>0.  When p,q<1 the shape is U‑shaped; p,q>1 gives a bell.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)

# Simple optimisation helpers

def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 30) -> Tuple[float, float]:
    """Minimise ``func`` on [lo, hi] using golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd


def hill_climb_alpha(alpha0: float, n: int, deadline: float) -> float:
    """Refine the product‑family exponent α by simple coordinate hill‑climb."""
    best = max(0.0, alpha0)
    best_score = compute_score(symmetric_product_family(n, best))
    step = 0.8
    while time.time() < deadline and step > 1e-4:
        improved = False
        for sign in (-1, 1):
            cand = max(0.0, best + sign * step)
            sc = compute_score(symmetric_product_family(n, cand))
            if sc < best_score:
                best_score = sc
                best = cand
                improved = True
                print(f"[{time.time():.1f}] α refined to {best:.6f} score {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best


def hill_climb_beta(p0: float, q0: float, n: int, deadline: float) -> Tuple[float, float]:
    """Two‑dimensional hill‑climb for the Beta‑family parameters."""
    best_p, best_q = max(0.01, p0), max(0.01, q0)
    best_score = compute_score(beta_family(n, best_p, best_q))
    step = 0.4
    while time.time() < deadline and step > 1e-3:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(0.01, best_p + dp)
                cand_q = max(0.01, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_score = sc
                    best_p, best_q = cand_p, cand_q
                    improved = True
                    print(f"[{time.time():.1f}] β refined to p={best_p:.6f}, q={best_q:.6f} score {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
    return best_p, best_q

            best_seq = symmetric_product_family(n, a_opt)
            best_alpha = a_opt
            print(f"[{time.time():.1f}] refined α={a_opt:.6f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refine Beta parameters (if we have a candidate)
    # ------------------------------------------------------------------
    if best_beta is not None and time.time() < deadline - 20.0:
        p0, q0 = best_beta
        p_opt, q_opt = hill_climb_beta(p0, q0, n, deadline - 10.0)
        seq = beta_family(n, p_opt, q_opt)
        sc = compute_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq
            best_beta = (p_opt, q_opt)
            print(f"[{time.time():.1f}] refined Beta p={p_opt:.6f}, q={q_opt:.6f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Fine stochastic local search with symmetry enforcement
    # ------------------------------------------------------------------
    if best_seq is None:
        # safe fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = compute_score(best_seq)

    # start from a symmetric copy
    a = (best_seq + best_seq[::-1]) / 2.0
    a = normalize(a)
    cur_score = compute_score(a)
    best_seq = a.copy()
    best_score = cur_score

    epsilon = 0.1   # initial fraction of donor mass to move

    print(f"[{time.time():.1f}] entering fine stochastic search, start score {best_score:.6f}")

    while time.time() < deadline - 5.0:
        # pick donor (any index, bias towards larger entries by rejection)
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = epsilon * a[i]                    # guaranteed non‑negative remainder
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # enforce symmetry
        cand = (cand + cand[::-1]) / 2.0
        # clip possible tiny negatives from floating‑point noise
        cand = np.maximum(cand, 0.0)
        cand = normalize(cand)

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] improved to {best_score:.6f}")
            epsilon = min(epsilon * 1.02, 0.5)   # increase step size slowly on success
        else:
            epsilon = max(epsilon * 0.98, 1e-4)   # shrink step size on failure

    # ------------------------------------------------------------------
    # 6.  Final official evaluation & return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq.tolist()
