```
def linear_program_step(a, M):
    n = len(a)
    # Build A matrix as dense
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    # b vector
    b = np.full(2*n-1, M, dtype=np.float64)
    # Objective: maximize sum(g); minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    g = res.x
    total = g.sum()
    if total <= 0:
        return None
    g_norm = g / total
    return g_norm
```

"""
Proposes a non‑negative step‑function sequence that minimises the
provided `evaluate_sequence`.  The method combines a quick global
exploration of parametric families, a Frank‑Wolfe style refinement that
solves a small linear program at each iteration, and a final
soft‑max/Adam polish.  All work respects the 1000 s wall‑clock budget.
"""
import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# --- Convolution utilities -------------------------------------------------
def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """FFT‑based linear convolution of two real vectors."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    C = A * B
    conv = np.fft.irfft(C, n_fft)[:m]
    return conv

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a*a (linear convolution)."""
    return _linear_convolution(a, a)

def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(full_convolution(a).max())

def compute_score(a: np.ndarray) -> float:
    """Exact evaluation used throughout the search (sum(a)=1)."""
    n = a.shape[0]
    return float(2.0 * n * max_convolution(a))

# --- Parametric families ---------------------------------------------------
def generate_powerlaw(n, p):
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_exponential(n, lam):
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_beta(n, alpha, beta):
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_gaussian(n, sigma):
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)

def generate_triangular(n):
    """Monotonically decreasing linear shape a_i ∝ (n‑i)."""
    a = np.arange(n, 0, -1, dtype=float)
    return a / a.sum()

def generate_cosine(n):

    i = np.arange(n, dtype=float)
    a = np.cos(np.pi * i / (2.0 * (n - 1)))
    a = np.maximum(a, 0.0)
    return a / a.sum()

def baseline_search(n, deadline, max_keep=6):
    best = []  # list of (score, vector)

    def insert_candidate(vec):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # families
    for p in np.linspace(0.05, 3.0, 250):
        if time.time() >= deadline: break
        insert_candidate(generate_powerlaw(n, p))
    for lam in np.linspace(0.0001, 0.1, 400):
        if time.time() >= deadline: break
        insert_candidate(generate_exponential(n, lam))
    al_vals = np.linspace(0.2, 6.0, 30)
    be_vals = np.linspace(0.2, 6.0, 30)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline: break
            insert_candidate(generate_beta(n, a, b))
    for sigma in np.linspace(n / 32.0, n / 2.0, 40):
        if time.time() >= deadline: break
        insert_candidate(generate_gaussian(n, sigma))
    insert_candidate(generate_triangular(n))
    insert_candidate(generate_cosine(n))

    for conc in concs:
        alpha = np.full(n, conc)
        for _ in range(draws):
            if time.time() >= deadline: break
            vec = np.random.dirichlet(alpha)
            insert_candidate(vec)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

# --- Smoothed Adam optimisation --------------------------------------------
def smooth_optimize(a0, deadline,
                    init_lr=0.15,
                    init_tau_factor=12.0,
                    min_tau=1e-9,
                    tau_decay=0.9997):
    """
    Adam optimiser on the soft‑max parametrisation with an annealed
    temperature τ that turns the smooth max into the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initialise temperature
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # soft‑max mapping
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        conv = full_convolution(a)
        max_c = conv.max()

        # smooth max weights
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient w.r.t. a
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule for x
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing
        tau = max(min_tau, tau * tau_decay)

        # periodic exact evaluation
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth] iter {it:6d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

# --- Linear‑programming direction (Frank‑Wolfe step) -----------------------
def lp_direction(a, M):
    """
    Solve     maximise   ∑ g_i
    subject to (a * g)_k ≤ M  for all k,
               g_i ≥ 0 .
    Returns the optimal g normalised to sum(g)=1.
    """
    n = a.shape[0]
    # Build the Toeplitz matrix that encodes the convolution constraints.
    # First column: a followed by zeros to length 2n‑1.
    c = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
    # First row: a_0 then zeros.
    r = np.concatenate([a[:1], np.zeros(n - 1, dtype=np.float64)])
    A = toeplitz(c, r)                     # shape (2n‑1, n)

    b = np.full(2 * n - 1, M, dtype=np.float64)
    obj = -np.ones(n, dtype=np.float64)    # minimise –∑ g_i  → maximise ∑ g_i

    # Using HiGHS (fast interior‑point) solver.
    res = linprog(obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 5.0})
    if not res.success:
        return None
    g = res.x
    total = g.sum()
    if total <= 0:
        return None
    return g / total

def frank_wolfe_refine(a0, deadline, max_iters=80):
    """
    Frank‑Wolfe style refinement.  Each iteration solves the linear
    programme described above, then moves a convex combination
    (step size 2/(t+2)) towards the solution.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = max_convolution(a)
        g = lp_direction(a, M)
        if g is None:
            break
        # Frank‑Wolfe step size
        alpha = 2.0 / (it + 2.0)
        a = (1.0 - alpha) * a + alpha * g
        # re‑normalise (numerical drift)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[FW] iter {it:3d}  score {cur_val:.6f}  M {M:.6e}")
        # early exit if improvement stalls
        if it > 5 and abs(best_val - cur_val) / best_val < 1e-6:
            break
    return best_a

def generate_triangular(n: int) -> np.ndarray:
    """Monotonically decreasing linear shape."""
    a = np.arange(n, 0, -1, dtype=float)
    return a / a.sum()


def generate_cosine(n: int) -> np.ndarray:
    """a_i ∝ cos(π·i/(2·(n‑1))) – a gently decreasing shape."""
    i = np.arange(n, dtype=float)
    a = np.cos(np.pi * i / (2.0 * (n - 1)))
    a = np.maximum(a, 0.0)
    return a / a.sum()


# ----------------------------------------------------------------------
# Global baseline exploration (keeps a few best candidates)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 6):
    best = []  # list of (score, vector)

    def insert_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law sweep
    for p in np.linspace(0.05, 3.0, 250):
        if time.time() >= deadline:
            break
        insert_candidate(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(0.0001, 0.1, 400):
        if time.time() >= deadline:
            break
        insert_candidate(generate_exponential(n, lam))

    # beta sweep
    al_vals = np.linspace(0.2, 6.0, 30)
    be_vals = np.linspace(0.2, 6.0, 30)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            insert_candidate(generate_beta(n, a, b))

    # Gaussian sweep
    for sigma in np.linspace(n / 32.0, n / 2.0, 40):
        if time.time() >= deadline:
            break
        insert_candidate(generate_gaussian(n, sigma))

    # a few extra deterministic families
    insert_candidate(generate_triangular(n))
    insert_candidate(generate_cosine(n))

    # Dirichlet draws
    concs = [0.1, 0.3, 0.7, 1.0, 2.0, 5.0]
    draws = 1200
    for conc in concs:
        alpha = np.full(n, conc)
        for _ in range(draws):
            if time.time() >= deadline:
                break
            insert_candidate(np.random.dirichlet(alpha))

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (soft‑max / annealed τ)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.15,
                    init_tau_factor: float = 12.0,
                    min_tau: float = 1e-9,
                    tau_decay: float = 0.9997) -> np.ndarray:
    """
    Adam optimiser on the soft‑max parametrisation with an annealed
    temperature τ that turns the smooth max into the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-30
    x = np.log(a + eps)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initialise temperature
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- soft‑max mapping ----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ---- convolution and smooth‑max weights ----
        conv = full_convolution(a)
        max_c = conv.max()
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ---- gradient w.r.t. a (∂/∂a  2·w ⊛ a) ----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ---- chain rule for the log‑soft‑max variables ----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature annealing ----
        tau = max(min_tau, tau * tau_decay)

        # ---- periodic exact evaluation ----
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth] iter {it:6d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming direction (Frank‑Wolfe step)
# ----------------------------------------------------------------------


def lp_direction(a: np.ndarray, M: float):
    """
    Solve   max   Σ g_i
    s.t.    (a * g)_k ≤ M  for every k,
            g_i ≥ 0 .

    Returns the optimal g normalised to Σ g_i = 1.
    """
    n = a.shape[0]

    # Toeplitz matrix that represents the convolution constraints.
    c = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])          # length 2n‑1
    r = np.concatenate([a[:1], np.zeros(n - 1, dtype=np.float64)])     # length n
    A = toeplitz(c, r)                                                  # (2n‑1) × n

    b = np.full(2 * n - 1, M, dtype=np.float64)
    obj = -np.ones(n, dtype=np.float64)                                 # minimise –∑ g_i

    # HiGHS interior‑point solver (fast and reliable)
    res = linprog(obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if not res.success:
        return None
    g = res.x
    total = g.sum()
    if total <= 0:
        return None
    return g / total


def frank_wolfe_refine(a0: np.ndarray, deadline: float, max_iters: int = 70):
    """
    Frank‑Wolfe style refinement.  Each iteration solves the LP above,
    then moves a convex combination (step size 2/(t+2)) towards the LP
    solution.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = max_convolution(a)
        g = lp_direction(a, M)
        if g is None:
            break
        alpha = 2.0 / (it + 2.0)                     # standard FW step size
        a = (1.0 - alpha) * a + alpha * g
        a = np.maximum(a, 0.0)
        a /= a.sum()
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[FW] iter {it:3d}  score {cur_val:.6f}  M {M:.6e}")

        # stop early if improvement stalls
        if it > 5 and abs(best_val - cur_val) / best_val < 1e-6:
            break
    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – the harness aborts if we run out of time
    deadline = start_time + budget_s - 5.0

    # problem size – 2048 is a sweet spot for speed / expressivity
    n = 2048

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # ------------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Global baseline exploration (≈ 150 s)
    # ------------------------------------------------------------------
    explore_deadline = deadline - 500.0
    if time.time() < explore_deadline:
        cand_vectors = baseline_search(n, explore_deadline, max_keep=6)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Frank‑Wolfe linear‑programming refinement (≈ 300 s)
    # ------------------------------------------------------------------
    fw_deadline = deadline - 200.0
    if best_seq is not None and time.time() < fw_deadline:
        a_start = np.array(best_seq, dtype=float)
        refined = frank_wolfe_refine(a_start, fw_deadline, max_iters=70)
        val = compute_score(refined)
        print(f"[FW‑final] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # ------------------------------------------------------------------
    # 3. Final soft‑max/Adam polish (remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_start = np.array(best_seq, dtype=float)
        final = smooth_optimize(a_start, deadline,
                                init_lr=0.20,
                                init_tau_factor=15.0,
                                tau_decay=0.9998)
        val = compute_score(final)
        print(f"[smooth‑final] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # ------------------------------------------------------------------
    # 4. Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5. Official evaluation
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
