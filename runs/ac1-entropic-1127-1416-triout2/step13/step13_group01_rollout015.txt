"""
search for a near‑optimal step‑function using a combinatorial
construction of a sparse support (≈2·√n non‑zero entries) and a
follow‑up smooth optimisation.

The idea:
* For a vector a of length n the objective is
        2·n·max_k (a*a)_k / (∑a)^2 .
  If a has w equal non‑zero entries then (∑a)=1 (after normalisation)
  and the objective reduces to 2·n·M / w² where M is the maximal
  ordered pair count of the support (the convolution of the 0/1 mask).
* The theoretical lower bound for w≈2·√n is attained when M=3,
  giving a value of about 1.5.  We therefore try to build a binary
  mask of size w≈2·√n with max‑pair count ≤3.
* Phase 1 – random constructive search for such a mask.
* Phase 2 – simulated‑annealing refinement if the constructive phase
  does not reach M=3.
* Phase 3 – a short Adam‑based smooth optimisation starting from the
  uniform weight on the found support to fine‑tune the real values.
* The whole routine respects the 1000 s wall‑clock budget.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Exact objective (no clipping – all entries already non‑negative)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    conv = np.convolve(a, a)                # linear full convolution
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Helper – fast update of the integer convolution of a binary mask
# ----------------------------------------------------------------------


def _initial_conv(mask: np.ndarray) -> np.ndarray:
    """Full ordered‑pair count for a binary mask (0/1)."""
    # use a simple O(n·w) method – the mask is sparse, so this is cheap
    idx = np.nonzero(mask)[0]
    conv = np.zeros(2 * len(mask) - 1, dtype=np.int32)
    # each unordered pair gives two contributions, self‑pairs give one
    for i in idx:
        conv[2 * i] += 1                     # self‑pair
        for j in idx:
            if j > i:
                conv[i + j] += 2
    return conv


# ----------------------------------------------------------------------
# Phase 1 – random constructive search for a mask with low max‑pair
# ----------------------------------------------------------------------


def constructive_mask_search(
    n: int,
    w: int,
    max_allowed: int,
    deadline: float,
    attempts: int = 30,
) -> tuple[np.ndarray, int]:
    """
    Try to build a binary mask of length n with exactly w ones such that
    the maximal ordered‑pair count does not exceed max_allowed.
    Returns the best mask found and its maximal count.
    """
    best_mask = None
    best_max = 10 ** 9

    while time.time() < deadline and attempts > 0:
        attempts -= 1

        mask = np.zeros(n, dtype=np.int8)
        conv = np.zeros(2 * n - 1, dtype=np.int32)
        cur_set = set()
        # shuffled order of candidate indices
        order = list(range(n))
        random.shuffle(order)

        for idx in order:
            if len(cur_set) >= w:
                break

            # check whether adding idx would keep all counts ≤ max_allowed
            # self‑pair
            if conv[2 * idx] + 1 > max_allowed:
                continue
            violate = False
            for p in cur_set:
                if conv[idx + p] + 2 > max_allowed:
                    violate = True
                    break
            if violate:
                continue

            # safe to add
            mask[idx] = 1
            cur_set.add(idx)
            conv[2 * idx] += 1
            for p in cur_set:
                if p == idx:
                    continue
                conv[idx + p] += 2

        if len(cur_set) == w:
            cur_max = int(conv.max())
            if cur_max < best_max:
                best_max = cur_max
                best_mask = mask.copy()
                print(f"[constructive] found mask with max {best_max}")
                # perfect solution reached – stop early
                if best_max <= max_allowed:
                    break

    # fallback – if we never built a full mask, fallback to a random mask
    if best_mask is None:
        # random Dirichlet‑like mask (still binary)
        mask = np.zeros(n, dtype=np.int8)
        idxs = random.sample(range(n), w)
        mask[idxs] = 1
        conv = _initial_conv(mask)
        best_mask, best_max = mask, int(conv.max())

    return best_mask, best_max


# ----------------------------------------------------------------------
# Phase 2 – simulated‑annealing refinement on a binary mask
# ----------------------------------------------------------------------


def anneal_mask(
    mask: np.ndarray,
    cur_max: int,
    deadline: float,
    init_temp: float = 5.0,
    final_temp: float = 1e-4,
    decay: float = 0.99994,
) -> tuple[np.ndarray, int]:
    """
    Starting from an existing mask, perform a Metropolis hill‑climb
    that swaps a ‘1’ with a ‘0’.  The goal is to lower the maximal
    ordered‑pair count.
    """
    n = mask.shape[0]
    w = mask.sum()
    # list / set representation for fast random access
    ones = [i for i, v in enumerate(mask) if v]
    zeros = [i for i, v in enumerate(mask) if not v]
    in_set = mask.astype(bool)

    # current convolution (integer counts)
    conv = _initial_conv(mask)

    best_mask = mask.copy()
    best_max = cur_max
    T = init_temp
    it = 0

    while time.time() < deadline:
        it += 1
        # pick random swap candidates
        i = random.choice(ones)          # will be removed
        j = random.choice(zeros)         # will be added

        # tentative update (work on a copy of conv)
        conv_new = conv.copy()

        # remove i
        for p in ones:
            s = i + p
            if p == i:
                conv_new[s] -= 1
            else:
                conv_new[s] -= 2

        # new set after removal (i excluded)
        new_ones = [p for p in ones if p != i]

        # add j
        for p in new_ones:
            conv_new[j + p] += 2
        conv_new[2 * j] += 1

        new_max = int(conv_new.max())
        delta = new_max - cur_max

        if delta <= 0 or random.random() < np.exp(-delta / max(T, 1e-12)):
            # accept move
            mask[i] = 0
            mask[j] = 1
            in_set[i] = False
            in_set[j] = True
            conv = conv_new
            cur_max = new_max

            # update helper lists
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)

            if cur_max < best_max:
                best_max = cur_max
                best_mask = mask.copy()
                print(f"[anneal] iter {it:7d}  max {best_max}")

        # temperature schedule
        T = max(final_temp, T * decay)

    return best_mask, best_max


# ----------------------------------------------------------------------
# Phase 3 – smooth optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """FFT‑based linear convolution (real, full length)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    return np.fft.irfft(C, n=n_fft)[:m]


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a (used for the objective)."""
    return _linear_convolution(a, a)


def smooth_optimize(
    a0: np.ndarray,
    deadline: float,
    init_lr: float = 0.15,
    init_tau_factor: float = 12.0,
    min_tau: float = 1e-9,
    tau_decay: float = 0.9995,
) -> np.ndarray:
    """
    Adam optimiser on the soft‑max parametrisation a = softmax(x).
    Minimises the max‑convolution (smoothly approximated) while keeping
    the vector on the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)          # start in the exponentiated‑gradient space

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature from the current max‑convolution
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max mapping (stable)
        mx = np.max(x)
        e = np.exp(x - mx)
        a = e / e.sum()

        # convolution of current a
        conv = full_convolution(a)
        mc = conv.max()

        # smooth‑max weights over the convolution entries
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - mc) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient w.r.t. a : g_i = 2·∑_k w_k·a_{k-i}
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain rule for the soft‑max (x‑space)
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        tau = max(min_tau, tau * tau_decay)

        # occasional exact evaluation
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[smooth_opt] iter {it:6d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # problem size – a modest length gives fast convolution but still
    # enough room for the theoretical lower bound (≈1.5)
    n = 1024
    w = int(round(2.0 * np.sqrt(n)))               # target support size

    print(f"[setup] n={n}, target support size w={w}")

    # -----------------------------------------------------------------
    # Phase 1 – try a pure constructive construction aiming at max=3
    # -----------------------------------------------------------------
    phase1_deadline = deadline - 400.0
    mask, cur_max = constructive_mask_search(
        n=n,
        w=w,
        max_allowed=3,
        deadline=phase1_deadline,
        attempts=30,
    )
    print(f"[phase1] initial mask max count = {cur_max}")

    # -----------------------------------------------------------------
    # Phase 2 – if needed, anneal the mask to lower the max count
    # -----------------------------------------------------------------
    if cur_max > 3 and time.time() < deadline - 200.0:
        phase2_deadline = deadline - 200.0
        mask, cur_max = anneal_mask(mask, cur_max, phase2_deadline)
        print(f"[phase2] refined mask max count = {cur_max}")

    # -----------------------------------------------------------------
    # Build the real-valued vector (uniform weight on the support)
    # -----------------------------------------------------------------
    a = np.zeros(n, dtype=float)
    idxs = np.nonzero(mask)[0]
    if len(idxs) == 0:
        # fallback: uniform distribution (should never happen)
        a[:] = 1.0 / n
    else:
        a[idxs] = 1.0 / len(idxs)                  # sum = 1

    # -----------------------------------------------------------------
    # Phase 3 – smooth Adam optimisation starting from the mask vector
    # -----------------------------------------------------------------
    if time.time() < deadline - 50.0:
        polish_deadline = deadline - 50.0
        a_refined = smooth_optimize(a, polish_deadline)
        a = a_refined
        print("[phase3] smooth optimisation finished")

    # -----------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # -----------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")

    return final_seq
