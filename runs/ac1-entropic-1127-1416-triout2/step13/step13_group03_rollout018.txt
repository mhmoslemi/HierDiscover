"""
A multi‑phase optimiser for the non‑negative step‑function problem.

Phases
------
1. **Baseline exploration** – diverse parametric families (power‑law,
   exponential, beta, Gaussian, mixtures, Dirichlet, and a few handcrafted
   windows).  The best few candidates (projected onto the simplex) are kept.

2. **L2‑flattening** – minimise the squared deviation of the full
   convolution from the constant target 1/(2 n‑1) using an Adam optimiser.
   The loss is smooth, its gradient is obtained by a single FFT‑based
   correlation, and each iterate is projected onto the probability simplex.

3. **Smooth‑max refinement** – the original soft‑max Adam optimiser is
   applied for a short final polish.

4. **Projected sub‑gradient polishing** – exact max‑convolution gradient
   descent (as in the reference implementation).

5. **Stochastic hill‑climb** – simulated‑annealing moves that swap mass
   between coordinates to escape possible local minima.

All phases respect the 1000 s wall‑clock budget; the best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT (exact, no circular wrap)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (additional to the previous implementation)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_inverse_sqrt(n: int, eps: float = 1e-6) -> np.ndarray:
    """a_i ∝ 1 / sqrt(i+eps) · 1 / sqrt(n-i+eps) – high at the ends, low in the centre."""
    i = np.arange(n, dtype=float) + eps
    left = 1.0 / np.sqrt(i)
    right = 1.0 / np.sqrt(n - i + eps)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_arcsine(n: int) -> np.ndarray:
    """Arcsine‑type density: a_i ∝ 1/√(i·(n‑i+1))."""
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_triangle(n: int) -> np.ndarray:
    """A simple triangular shape (peak in the centre)."""
    i = np.arange(n, dtype=float)
    a = np.minimum(i + 1, n - i)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_inv_hann(n: int) -> np.ndarray:
    """1 – Hann window: high at the ends, low in the centre."""
    i = np.arange(n, dtype=float)
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    a = 1.0 - hann
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def mixture(vec1: np.ndarray, vec2: np.ndarray, lam: float) -> np.ndarray:
    mixed = lam * vec1 + (1.0 - lam) * vec2
    return project_to_simplex(mixed)

def baseline_search(
    n: int, deadline: float, max_keep: int = 16, random_draws: int = 8000
):
    """Explore many parametric families and Dirichlet draws; keep the best vectors."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1. Power‑law sweep
    for p in np.linspace(0.05, 5.0, 91):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2. Exponential sweep
    for lam in np.linspace(0.00005, 0.6, 140):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3. Beta grid (extended)
    al_vals = np.geomspace(0.01, 3.0, 25)
    be_vals = np.geomspace(0.01, 3.0, 25)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4. Random Beta draws (explore non‑grid points)
    for _ in range(300):
        if time.time() >= deadline:
            break
        alpha = 10 ** np.random.uniform(-2.5, 0.5)
        beta = 10 ** np.random.uniform(-2.5, 0.5)
        try_candidate(generate_beta(n, alpha, beta))

    # 5. Gaussian
    for sigma in np.linspace(0.05 * n, 0.45 * n, 38):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # 6. Sine‑power
    for p in np.linspace(0.1, 10.0, 40):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # 7. Inverse‑sqrt, arcsine, triangle, inv‑hann
    for maker in (generate_inverse_sqrt, generate_arcsine, generate_triangle, generate_inv_hann):
        if time.time() >= deadline:
            break
        try_candidate(maker(n))

    # 8. Mixtures of two families
    for _ in range(200):
        if time.time() >= deadline:
            break
        a = generate_powerlaw(n, np.random.uniform(0.05, 5.0))
        b = generate_beta(n,
                          10 ** np.random.uniform(-2.5, 0.5),
                          10 ** np.random.uniform(-2.5, 0.5))
        lam = np.random.rand()
        try_candidate(mixture(a, b, lam))

    # 9. Dirichlet draws (wide concentration range)
    concs = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

def l2_flatten_adam(
    a0: np.ndarray,
    deadline: float,
    lr_start: float = 0.18,
    lr_decay: float = 0.98,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser that minimises ½·‖conv(a,a) – avg‖₂²,
    where avg = 1/(2·n‑1).  Returns the best simplex‑projected vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    avg = 1.0 / m

    # Adam moments
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    lr = lr_start

    # pre‑compute FFT size for the correlation (n+m‑1 <= 3n‑2)
    size = 1 << ((n + m - 2).bit_length())

    while time.time() < deadline:
        # full linear convolution
        conv = full_convolution(a)          # length m
        diff = conv - avg                    # centred residual

        # correlation = a (reversed) * diff   → gradient = 2·corr
        A = np.fft.rfft(a, n=size)
        D = np.fft.rfft(diff, n=size)
        corr_full = np.fft.irfft(np.conj(A) * D, n=size)[: n + m - 1]
        offset = m - 1                       # lag 0 at this index
        grad = 2.0 * corr_full[offset: offset + n]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # learning‑rate annealing
        if t % 400 == 0:
            lr = max(lr * lr_decay, 5e-4)

        # occasional jitter to avoid stagnation
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.004 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max (kept from the reference implementation)
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-5,
    tau_decay: float = 0.96,
    lr_start: float = 0.12,
    lr_decay: float = 0.99,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    lr = lr_start

    while time.time() < deadline:
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()

        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 400 == 0:
            lr = max(lr * lr_decay, 1e-4)

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr_start * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (as before)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (stochastic hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9992,
                        step_scale_init: float = 0.06,
                        step_scale_min: float = 5e-6) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 7000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # Length – larger n gives a tighter theoretical bound.
    n = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Fallback (if a known good construction exists)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 200 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 300.0
    if time.time() < baseline_deadline:
        candidates = baseline_search(n, baseline_deadline, max_keep=12, random_draws=7000)
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. L2‑flattening (new phase)
    # -----------------------------------------------------------------
    if best_seq is not None:
        # allocate roughly 300 s for this phase
        l2_deadline = min(deadline - 150.0,
                          start_time + 250.0 + (deadline - start_time) * 0.30)
        start_vectors = []

        # best found so far (projected, just in case)
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        start_vectors.append(best_vec)

        # uniform start
        start_vectors.append(np.full(n, 1.0 / n))

        # a few random starts
        for _ in range(4):
            start_vectors.append(project_to_simplex(np.random.rand(n)))

        refined_best = best_vec.copy()
        refined_val = best_val

        for idx, start_vec in enumerate(start_vectors):
            if time.time() >= l2_deadline:
                break
            local_deadline = l2_deadline - (time.time() - start_time) - 0.5
            if local_deadline <= time.time():
                break
            a_ref = l2_flatten_adam(start_vec, local_deadline)
            val = compute_score(a_ref)
            print(f"[l2‑adam {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_ref.copy()

        best_val, best_seq = refined_val, refined_best.tolist()
        print(f"[l2‑adam] best after L2 phase = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Smooth‑max Adam refinement (short)
    # -----------------------------------------------------------------
    if best_seq is not None:
        refine_deadline = deadline - 150.0
        if time.time() < refine_deadline:
            a0 = np.array(best_seq, dtype=float)
            a_ref = smooth_max_adam(a0, refine_deadline)
            val = compute_score(a_ref)
            print(f"[smooth‑adam] refined score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_ref.tolist()

    # -----------------------------------------------------------------
    # 4. Sub‑gradient polishing
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_refined)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 5. Simulated annealing / stochastic hill‑climb
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
