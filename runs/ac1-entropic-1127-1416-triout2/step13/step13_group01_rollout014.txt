"""
Search for a non‑negative step function minimizing evaluate_sequence.

Algorithm overview
------------------
* Global exploration (≈ 150 s):
  - Sweep several parametric families (power‑law, exponential, beta,
    Gaussian, tent, cosine) over enlarged parameter ranges.
  - Draw many random Dirichlet vectors with various concentration
    parameters.
  - Keep the `max_keep` best candidates (lowest exact score).

* Smoothed mirror‑descent (soft‑max) optimisation (≈ 750 s):
  - Optimise the smooth approximation  `softmax_τ(conv(a,a))`
    using Adam on the log‑variables (`x = log a`).  The temperature τ
    is started high (≈ 10·max conv) and annealed geometrically to a
    very small floor.
  - Run the optimiser from each of the retained candidates and from
    a few fresh random Dirichlet starts while time permits.

* Final simulated‑annealing polish (≈ 80 s) to escape possible tiny
  plateaux.

All steps respect the global wall‑clock limit (default 1000 s).  The
length `n` is set to 4096 – large enough to allow a markedly lower
theoretical bound (`≈ 1.0`) while still fitting comfortably into the
time budget.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution using real FFT (fast and accurate)
# ----------------------------------------------------------------------
def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (self‑convolution)."""
    return _linear_convolution(a, a)

# ----------------------------------------------------------------------
# Exact evaluation (matches evaluate_sequence when sum(a)=1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * full_convolution(a).max())

# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return a / a.sum()

def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    return a / a.sum()

def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    return a / a.sum()

def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a / a.sum()

def generate_tent(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i + 1, n - i) ** p
    return a / a.sum()

def generate_cosine(n: int, width: float) -> np.ndarray:
    """
    Cosine bump centred at the middle.
    width controls the half‑width (as a fraction of n/2); values
    outside the bump are set to zero.
    """
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    half = width * (n / 2.0)
    d = np.abs(i - mu)
    a = np.cos(np.pi * d / (2 * half))
    a[d > half] = 0.0
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return a / a.sum()

# ----------------------------------------------------------------------
# Global exploration – keep the best few vectors
# ----------------------------------------------------------------------
def baseline_search(n: int, deadline: float, max_keep: int = 12):
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        # keep up to max_keep lowest scores
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- 1) power‑law -------------------------------------------------
    for p in np.linspace(0.05, 4.0, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # ---- 2) exponential ------------------------------------------------
    for lam in np.linspace(0.00001, 0.2, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # ---- 3) beta -------------------------------------------------------
    al_vals = np.linspace(0.2, 8.0, 30)
    be_vals = np.linspace(0.2, 8.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # ---- 4) Gaussian ----------------------------------------------------
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # ---- 5) Tent --------------------------------------------------------
    p_vals = np.linspace(0.5, 5.0, 30)
    for p in p_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_tent(n, p))

    # ---- 6) Cosine -------------------------------------------------------
    width_vals = np.linspace(0.1, 1.0, 20)   # half‑width as fraction of n/2
    for w in width_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_cosine(n, w))

    # ---- 7) Dirichlet draws --------------------------------------------
    concs = [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 2000
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max representation)
# ----------------------------------------------------------------------
def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 20.0,
                    min_tau: float = 1e-10,
                    tau_decay: float = 0.9994) -> np.ndarray:
    """Minimise the max‑convolution using a temperature‑annealed soft‑max.

    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    # log‑variables (ensure positivity)
    eps = 1e-30
    x = np.log(a + eps)

    # Adam accumulators
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initialise temperature from the current max‑conv
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- soft‑max mapping a = exp(x)/sum(exp(x)) ----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ---- current convolution and max ----
        conv = full_convolution(a)
        max_c = conv.max()

        # ---- soft‑max weights over the convolution entries ----
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ---- gradient w.r.t. a (g = 2 * conv_{rev} * w) ----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)      # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ---- chain rule through soft‑max (gradient wrt x) ----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                best_a = a.copy()
                print(f"[smooth_opt] iter {it:6d}  score {val:.6f}  τ {tau:.2e}")

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 5e-7) -> np.ndarray:

    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:

        j = np.random.randint(n)

            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-15)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            # occasional jitter to shake out plateaux
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

"""
Search for a non‑negative step function minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Global exploration** (≈150 s)
   - Sweep several parametric families (power‑law, exponential, beta,
     Gaussian, tent, cosine) over wide ranges.
   - Draw many Dirichlet vectors with different concentration values.
   - Keep the `max_keep` best candidates (lowest exact score).

2. **Smoothed mirror‑descent** (≈750 s)
   - Optimise the smooth approximation of the max‑convolution
     using Adam on the log‑variables (soft‑max parametrisation).
   - Temperature τ starts high (≈ 20·max conv) and is geometrically
     annealed to a tiny floor.
   - Run the optimiser from each retained candidate and from a few
     fresh random starts while time permits.

3. **Final simulated‑annealing polish** (≈ 80 s)
   - Small Metropolis‑style hill‑climb that can escape tiny plateaux.

All steps honour the global wall‑clock budget (default 1000 s).  The
vector length `n` is set to 4096 – large enough to push the theoretical
lower bound (`≈ 1.0`) far below the previous best (~1.51) while still
fitting comfortably into the time limit.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution via real FFT (fast and accurate)
# ----------------------------------------------------------------------
def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (self‑convolution)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (matches evaluate_sequence when sum(a)=1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – the same quantity that evaluate_sequence returns
    after normalising sum(a) to 1."""
    n = a.shape[0]
    return float(2.0 * n * full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return a / a.sum()


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    return a / a.sum()


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    return a / a.sum()


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a / a.sum()


def generate_tent(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i + 1, n - i) ** p
    return a / a.sum()


def generate_cosine(n: int, width: float) -> np.ndarray:
    """
    Cosine bump centred at the middle.
    ``width`` is a fraction of n/2 defining the half‑width of the bump.
    Outside the bump the value is set to zero.
    """
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    half = width * (n / 2.0)
    d = np.abs(i - mu)
    a = np.cos(np.pi * d / (2 * half))
    a[d > half] = 0.0
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Global exploration – retain the best few vectors
# ----------------------------------------------------------------------
def baseline_search(n: int, deadline: float, max_keep: int = 12):
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ----- 1) power‑law -------------------------------------------------
    for p in np.linspace(0.05, 4.0, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # ----- 2) exponential ------------------------------------------------
    for lam in np.linspace(0.00001, 0.2, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # ----- 3) beta -------------------------------------------------------
    al_vals = np.linspace(0.2, 8.0, 30)
    be_vals = np.linspace(0.2, 8.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # ----- 4) Gaussian ----------------------------------------------------
    sigma_vals = np.linspace(n / 32.0, n / 2.0, 30)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # ----- 5) Tent --------------------------------------------------------
    for p in np.linspace(0.5, 5.0, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_tent(n, p))

    # ----- 6) Cosine -------------------------------------------------------
    for w in np.linspace(0.1, 1.0, 20):
        if time.time() >= deadline:
            break
        try_candidate(generate_cosine(n, w))

    # ----- 7) Dirichlet draws --------------------------------------------
    concs = [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 2000
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------
def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 20.0,
                    min_tau: float = 1e-10,
                    tau_decay: float = 0.9994) -> np.ndarray:
    """Minimise the max‑convolution using a temperature‑annealed soft‑max.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # temperature init
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- soft‑max mapping a = exp(x)/∑exp(x) ----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ---- current convolution ----
        conv = full_convolution(a)
        max_c = conv.max()

        # ---- soft‑max weights over convolution entries ----
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ---- gradient w.r.t. a (g = 2·rev(a) * w) ----
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)          # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ---- chain rule through soft‑max (gradient w.r.t. x) ----
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- anneal temperature ----
        tau = max(min_tau, tau * tau_decay)

        # ---- occasional exact evaluation ----
        if it % 300 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:6d}  score {val:.6f}  τ {tau:.2e}")

    return best_a
