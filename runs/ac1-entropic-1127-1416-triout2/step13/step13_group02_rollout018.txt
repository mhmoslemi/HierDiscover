def smooth_adam_polish(a0, deadline, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    n = a0.shape[0]
    u = np.log(a0 + 1e-20)   # initial log
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    best_val = float('inf')
    best_a = None
    t = 0
    # schedule for beta
    beta_initial = 1.0
    beta_max = 5000.0   # large but limited.
    while time.time() < deadline:
        t += 1
        # softmax
        exp_u = np.exp(u - np.max(u))
        a = exp_u / exp_u.sum()
        # convolution
        conv = full_convolution(a)
        # smooth weight
        beta = min(beta_max, beta_initial * (t**0.5))
        # subtract max to improve stability
        conv_shifted = conv - conv.max()
        w = np.exp(beta * conv_shifted)
        w_sum = w.sum()
        if w_sum == 0:
            # overflow case, reduce beta
            w = np.exp(beta * conv_shifted)  # already small?
            w_sum = w.sum() + 1e-20
        w = w / w_sum
        # gradient of smooth max w.r.t a
        # grad_a = 2 * np.correlate(w, a, mode='valid')
        # use np.convolve (valid) similar
        grad_a = 2 * np.correlate(w, a, mode='valid')
        # map to u gradient
        grad_u = a * (grad_a - np.dot(grad_a, a))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate actual objective
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # occasional jitter
        if t % 2000 == 0:
            u += np.random.randn(n) * (lr * 0.001)
    return best_a if best_a is not None else np.exp(u) / np.exp(u).sum()
```

"""
Search for a step‑function with a low value of
``evaluate_sequence``.

Approach
========
1. **Exploration** – generate many candidate vectors from a few
   parametric families (exponential, power‑law, Gaussian) and a large pool of
   Dirichlet draws.  The best 30 candidates (by the exact objective
   ``2·n·max(conv)``) are retained.

2. **Smooth Adam polishing** – each retained candidate is refined by minimising
   a smooth approximation of ``max_k (a*a)_k``:
   ```
   f_β(a) = (1/β)·log( Σ_k exp(β·(a*a)_k) )
   ```
   The gradient of ``f_β`` w.r.t. the soft‑max parameters ``u`` (where
   ``a = softmax(u)``) is computed analytically via a weighted convolution.
   An Adam optimiser (mirror‑descent on the simplex) drives the vector toward a
   local optimum.  The temperature β is increased gradually so that the
   optimisation first follows a smooth landscape and later concentrates on the
   true max.

3. **Random‑restart polishing** – as long as time permits new random Dirichlet
   starts are polished in the same way; the best ever vector is kept.

The whole routine respects a 1000 s wall‑clock budget and finally returns the
best sequence found (as a plain Python list) ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i/σ)²)."""
    i = np.arange(n, dtype=float)
    a = np.exp(- (i / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 30):
    """Sweeps a few families and a large pool of Dirichlet draws,
    keeping the ``max_keep`` vectors with the smallest score."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s                      # project onto the simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # --- exponential ---------------------------------------------------
    # fine log‑grid around the region where the earlier run performed well
    for lam in np.logspace(-6, -1, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # --- power‑law -----------------------------------------------------
    for p in np.linspace(0.05, 5.0, 400):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # --- Gaussian ------------------------------------------------------
    # sigma roughly varies from a few points up to half the length
    for sigma in np.linspace(5.0, n / 2.0, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # --- Dirichlet draws -----------------------------------------------
    # a few concentrations give very different shapes
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per = 1200                      # ≈ 8400 draws – still fast
    for conc in concs:
        alpha = np.full(n, conc)
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(alpha)
            try_candidate(vec)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth Adam polishing (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def smooth_adam_polish(a0: np.ndarray, deadline: float,
                       lr: float = 0.05,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimiser on the smoothed max objective
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    Variables are stored as u with a = softmax(u).
    The temperature β is increased slowly (β ≈ sqrt(iter)).
    Returns the best vector observed before the deadline.
    """
    n = a0.shape[0]
    # initialise in log‑space (avoid log(0))
    u = np.log(a0 + 1e-20)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # ----- soft‑max projection -----
        max_u = np.max(u)
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ----- convolution -----
        conv = full_convolution(a)

        # ----- smooth weighting (log‑sum‑exp) -----
        beta = min(5000.0, 1.0 * np.sqrt(t))          # schedule for temperature
        # shift for numerical stability
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:          # should not happen, but guard against overflow
            w = np.exp(beta * (conv - shift) - 1e4)
            w_sum = w.sum() + 1e-20
        w = w / w_sum            # normalised weights, sum to 1

        # ----- gradient of the smooth max w.r.t. a -----
        # grad_a = Σ_k w_k ∂conv_k/∂a = 2 * Σ_k w_k * a_{k‑i}
        # This can be written as a valid correlation
        grad_a = 2.0 * np.correlate(w, a, mode='valid')

        # ----- mirror‑descent gradient (soft‑max Jacobian) -----
        g_u = a * (grad_a - np.dot(grad_a, a))

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- keep the best exact score seen so far -----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # occasional jitter to leave plateaus
        if t % 3000 == 0:
            u += np.random.randn(n) * (lr * 0.001)

    return best_a if best_a is not None else np.exp(u) / np.exp(u).sum()


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``.
    The routine respects the given time budget (default 1000 s) and prints
    progress information.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0            # safety margin

    # --------------------------------------------------------------
    # problem size – 2048 offers a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")

    # ----- 0. fallback from a previous run (if present) -----
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # ----- 1. Global baseline search -----
    baseline_deadline = deadline - 350.0   # reserve the bulk for polishing
    if time.time() < baseline_deadline:
        candidates = baseline_search(n, baseline_deadline, max_keep=30)
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best score after sweep = {best_val:.6f}")

    # ----- 2. Smooth Adam polishing (primary + many restarts) -----
    polish_deadline = deadline - 30.0
    if best_seq is not None and time.time() < polish_deadline:
        # polish the current best vector
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_adam_polish(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[polish‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # random‑restart polishing while time permits
        while time.time() < polish_deadline:
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_adam_polish(a_start, polish_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[polish‑restart] new best = {val:.6f}")

    # ----- 3. Final check (if nothing succeeded) -----
    if best_seq is None:
        # uniform distribution is always feasible (score = 2.0)
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)
        print("[fallback‑uniform] using uniform distribution")

    # ----- 4. Report with the official evaluator -----
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
