"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence``.

Main ideas
----------
1. **Global exploration** – a richer sweep over the same parametric families
   (power‑law, exponential, beta, Gaussian) plus many Dirichlet draws.
2. **Smooth Adam optimisation** – identical to the previous version but we
   keep the best result and also try a few quick restarts with different
   random starts.
3. **Exact sub‑gradient refinement** – after the smooth phase we switch to a
   pure sub‑gradient method for the (hard) max–convolution objective.
   The update is performed on the probability simplex using the classic
   Euclidean projection (Duchi et al., 2008).  This phase is very effective
   at “flattening’’ the convolution peak and typically beats the soft‑max
   approach by a few ‰.
4. **Final simulated annealing** – a short Metropolis hill‑climb that can
   escape tiny plateaux left by the deterministic refinements.

All stages respect the 1000 s wall‑clock budget; the remaining time is split
approximately 15 % baseline, 55 % smooth Adam, 25 % sub‑gradient refinement,
and the last 5 % simulated annealing.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear convolution (real FFT)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real 1‑D arrays using FFT."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Scoring – exact objective (scale invariant, assumes sum(a)≈1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) (the same quantity returned by evaluate_sequence
    when the vector is already normalised)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all are normalised to sum 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global baseline exploration (keeps a small set of the best vectors)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Sweep the families and many Dirichlet draws, retaining up to max_keep."""
    best = []  # list of (score, vector)

    def insert_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- power‑law -------------------------------------------------
    for p in np.linspace(0.05, 2.5, 250):
        if time.time() >= deadline:
            break
        insert_candidate(generate_powerlaw(n, p))

    # ---- exponential ------------------------------------------------
    for lam in np.linspace(1e-4, 0.07, 350):
        if time.time() >= deadline:
            break
        insert_candidate(generate_exponential(n, lam))

    # ---- beta -------------------------------------------------------
    al_vals = np.linspace(0.2, 5.0, 30)
    be_vals = np.linspace(0.2, 5.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            insert_candidate(generate_beta(n, alpha, beta))

    # ---- Gaussian ----------------------------------------------------
    sigma_vals = np.linspace(n / 24.0, n / 2.0, 40)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        insert_candidate(generate_gaussian(n, sigma))

    # ---- Dirichlet draws --------------------------------------------
    for conc in [0.1, 0.2, 0.5, 1.0, 2.0, 4.0]:
        draws = 1300
        alpha_vec = np.full(n, conc)
        for _ in range(draws):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(alpha_vec)
            insert_candidate(vec)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed Adam optimisation (soft‑max parametrisation)
# ----------------------------------------------------------------------


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    ex = np.exp(x - mx)
    return ex / ex.sum()


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 10.0,
                    min_tau: float = 1e-8,
                    tau_decay: float = 0.9995) -> np.ndarray:
    """
    Adam optimiser on the soft‑max parametrisation.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-30
    x = np.log(a + eps)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature (smoothness)
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max mapping
        a = _softmax(x)

        # convolution of the current a
        conv = full_convolution(a)
        max_c = conv.max()

        # soft‑max over the convolution values (temperature τ)
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp((conv - max_c) / tau)
            ws = w.sum()
            if ws == 0 or not np.isfinite(ws):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= ws

        # gradient w.r.t. a
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)      # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # chain‑rule for the log‑space variables (mirror descent)
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal τ
        tau = max(min_tau, tau * tau_decay)

        # periodic exact evaluation
        if it % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:6d}  score {val:.6f}  τ {tau:.2e}")

    return best_a

# Exact sub‑gradient refinement (hard max, simplex projection)
# ----------------------------------------------------------------------


def subgrad_refine(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Deterministic sub‑gradient descent on max(a*a) with Euclidean
    projection onto the simplex.  Uses the exact (hard) max, i.e.
    τ = 0.  Returns the best vector discovered before *deadline*.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = full_convolution(a)
        M = conv.max()
        # active set (indices attaining the max, up to a tiny tolerance)
        tol = max(1e-14, 1e-12 * M)
        active = np.where(conv >= M - tol)[0]

        # build weight vector w that averages over the active constraints
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(1, len(active))

        # sub‑gradient of max at this point
        grad_full = _linear_convolution(a[::-1], w)
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # simple diminishing step size
        eta0 = 0.6
        eta = eta0 / np.sqrt(it)

        # gradient step + projection onto the simplex
        a_new = a - eta * grad_a
        a_new = proj_simplex(a_new)

        a = a_new

        # occasional reporting
        if it % 100 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[subgrad] iter {it:6d}  best {val:.6f}")

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # occasional jitter to shake out plateaux
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
