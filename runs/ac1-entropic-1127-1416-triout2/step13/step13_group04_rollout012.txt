def smooth_max_gradient(a, alpha):
    conv = full_convolution(a)  # length 2n-1
    # compute softmax weight
    # To avoid overflow, subtract max conv.
    max_c = conv.max()
    exps = np.exp(alpha * (conv - max_c))
    w = exps / exps.sum()  # shape same as conv
    # compute gradient:
    a_rev = a[::-1]
    # compute convolution of w with a_rev
    L = len(w) + len(a_rev) - 1
    n_fft = 1 << ((L - 1).bit_length())
    W_fft = np.fft.rfft(w, n=n_fft)
    A_rev_fft = np.fft.rfft(a_rev, n=n_fft)
    conv_rev = np.fft.irfft(W_fft * A_rev_fft, n=n_fft)[:L]
    grad = 2.0 * conv_rev[len(a)-1 : len(a)-1 + len(a)]
    return conv, w, grad

"""...
Search candidate for self-convolution upper bound.
Uses a smooth max (log-sum-exp) surrogate and Adam optimizer
to shape a non‑negative vector on the simplex.
A short parametric sweep provides a good start, then Adam,
followed by a sub‑gradient polishing and a final hill‑climb.
"""

import time, math, numpy as np

# Helper functions

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Parametric families -----------------------------------------------------

def power_law(n: int, p: float):
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a

def geometric(n: int, q: float):
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a

def mixed_plateau_power(n: int, L_frac: float, p: float):
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a

def beta_shape(n: int, alpha: float, beta: float):
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a

def random_dirichlet(n: int, concentration: float):
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    if y.sum() == 0.0:
        return None
    return y / y.sum()

# Smooth‑max surrogate ----------------------------------------------------

def smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Returns (conv, weights, gradient) where
    - conv = a * a (full convolution)
    - weights = softmax(alpha * conv) (sums to 1)
    - gradient = ∂[ (1/α) log Σ exp(α conv) ] / ∂a
    """
    conv = full_convolution(a)                     # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                         # ≤ 0, safe for exp
    exp_scaled = np.exp(alpha * shifted)
    sum_exp = exp_scaled.sum()
    if sum_exp == 0.0:
        w = np.full_like(conv, 1.0 / conv.shape[0])
    else:
        w = exp_scaled / sum_exp                     # soft‑max weights

    # gradient = 2 * (w  * a)  (cross‑correlation)
    a_rev = a[::-1]
    L = w.shape[0] + a_rev.shape[0] - 1
    n_fft = 1 << ((L - 1).bit_length())
    W = np.fft.rfft(w, n=n_fft)
    A_rev = np.fft.rfft(a_rev, n=n_fft)
    conv_rev = np.fft.irfft(W * A_rev, n=n_fft)[:L]
    n = a.shape[0]
    grad = 2.0 * conv_rev[n-1 : n-1 + n]
    return conv, w, grad

# Adam optimiser ---------------------------------------------------------

def adam_optimize(a0: np.ndarray, time_deadline: float):
    """
    Adam optimiser on the smooth‑max surrogate.
    The learning‑rate and the soft‑max temperature α are both
    decayed in time so that the algorithm starts aggressive and
    finishes with a sharper approximation of the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = compute_score(a)
    start = time.time()
    total = time_deadline - start
    while time.time() < time_deadline:
        t += 1
        elapsed = time.time() - start
        # linearly increase α from 5 to 50
        alpha = 5.0 + 45.0 * (elapsed / total)
        _, _, grad = smooth_max_gradient(a, alpha)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning‑rate decays from 0.12 to 0.001
        lr = 0.12 * (1.0 - elapsed / total) + 0.001

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # occasional small jitter to avoid stagnation
        if t % 3000 == 0:
            a += np.random.randn(n) * lr * 0.02
            a = project_to_simplex(a)

        # periodic exact evaluation
        if t % 500 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[adam] iter={t} score={best_val:.6f} time={time.time():.1f}s")
    return best_a

def subgradient_polish(a0: np.ndarray, time_deadline: float, step0: float = 0.12) -> np.ndarray:
    a = a0.copy()
    best_a = a.copy()

"""
Search for a step‑function (non‑negative vector) that minimises
the self‑convolution bound used by ``evaluate_sequence``.
The procedure consists of:

1. Fast parametric sweep (power‑law, geometric, mixed plateau‑power,
   beta, random Dirichlet) to obtain a decent seed.
2. Adam optimisation on a smooth surrogate of the max‑convolution:
   f(a)= (1/α)·log Σ exp(α·(a*a)).
   The softmax temperature α is increased from 5 to ~50 while the
   learning‑rate decays, and the iterates are projected onto the simplex.
3. Short projected sub‑gradient polishing (as in the reference).
4. Optional deterministic hill‑climb for final fine‑tuning.

All vectors are kept normalised (Σa=1), therefore the objective
``evaluate_sequence`` is invariant to scaling and equals
``2·n·max(conv(a,a))``.  The code respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())   # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def power_law(n: int, p: float):
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def geometric(n: int, q: float):
    """a_i ∝ q^{i} (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float):
    """Plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float):
    """a_i ∝ x^{alpha} (1-x)^{beta},   x = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def random_dirichlet(n: int, concentration: float):
    """Draw a random point from Dirichlet(conc,…,conc)."""
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    if y.sum() == 0.0:
        return None
    return y / y.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient
# ----------------------------------------------------------------------
def smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Returns (conv, weights, gradient) where
    - conv   = a * a (full convolution)
    - weights = soft‑max(α·conv)  (sums to 1)
    - gradient = ∂[(1/α)·log Σ exp(α·conv)] / ∂a
    """
    conv = full_convolution(a)                     # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                         # ≤0 ⇒ safe for exp
    exp_scaled = np.exp(alpha * shifted)
    sum_exp = exp_scaled.sum()
    if sum_exp == 0.0:
        w = np.full_like(conv, 1.0 / conv.shape[0])
    else:
        w = exp_scaled / sum_exp                    # soft‑max weights

    # gradient = 2·(w ⋆ a) where ⋆ denotes cross‑correlation.
    a_rev = a[::-1]
    L = w.shape[0] + a_rev.shape[0] - 1
    n_fft = 1 << ((L - 1).bit_length())
    W = np.fft.rfft(w, n=n_fft)
    A_rev = np.fft.rfft(a_rev, n=n_fft)
    conv_rev = np.fft.irfft(W * A_rev, n=n_fft)[:L]
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return conv, w, grad

# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray, time_deadline: float):
    """
    Adam optimiser on the smooth‑max surrogate.
    The temperature α grows from 5 to 50 while the learning rate
    decays from ≈0.12 to 0.001.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = compute_score(a)

    start = time.time()
    total = time_deadline - start

    while time.time() < time_deadline:
        t += 1
        elapsed = time.time() - start

        # temperature schedule (soft‑max sharpness)
        alpha = 5.0 + 45.0 * (elapsed / total)          # 5 → 50

        _, _, grad = smooth_max_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning‑rate schedule
        lr = 0.12 * (1.0 - elapsed / total) + 0.001

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # occasional jitter to escape flat regions
        if t % 3000 == 0:
            a += np.random.randn(n) * lr * 0.02
            a = project_to_simplex(a)

        # periodic exact evaluation
        if t % 500 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[adam] iter={t}  score={best_val:.6f}  time={time.time():.1f}s")

    return best_a
