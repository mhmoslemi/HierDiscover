def gradient_descent_opt(a0, deadline):
    step0 = 0.2
    beta1=0.9
    beta2=0.999
    eps=1e-8
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    alpha = 20.0
    max_alpha = 300.0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        # compute softmax weights
        max_conv = conv.max()
        # adjust alpha schedule
        if t % 2000 == 0 and alpha < max_alpha:
            alpha = min(alpha * 1.5, max_alpha)
        exp_diff = np.exp(alpha * (conv - max_conv))
        w = exp_diff / exp_diff.sum()
        # gradient of weighted sum
        w_rev = w[::-1]
        grad_conv = np.convolve(a, w_rev)
        grad = 2.0 * grad_conv[(2*a.shape[0]-2) - np.arange(a.shape[0])]
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = project_to_simplex(a)
        # evaluate score occasionally
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score

"""
search_step_function_improved.py

A self‑contained optimiser for the ``evaluate_sequence`` objective.

Strategy
--------
1.  Work with a *simplex* representation (Σa = 1, a_i ≥ 0) – the
    objective 2·n·max(conv(a,a)) is scale‑invariant.
2.  Starting points are taken from a handful of analytically‑motivated families
    (reverse‑power‑law, uniform, random Dirichlet, β‑shapes …).
3.  From each start we run a *smooth‑max* Adam optimiser.
    The non‑smooth max is approximated by a soft‑max
        φ_α(a) = log‑sum‑exp(α·conv(a,a))/α,
    whose gradient is a simple convolution (see derivation in the code).
    α is increased gradually so the optimisation first sees a smooth landscape
    and later focuses on the true peak.
4.  The best candidate is polished with a cheap stochastic hill‑climb.
5.  The final sequence is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # tie‑break: uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fft_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (returns full length)."""
    n = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:n]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper – convolution of a with itself (full mode)."""
    return fft_convolution(a, a)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The routine works on the normalised simplex, therefore the scaling
    invariance of ``evaluate_sequence`` is respected.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for seeding the optimiser)
# ----------------------------------------------------------------------
def reverse_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}  (front‑loaded decreasing power‑law)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1
    a = idx ** p
    a /= a.sum()
    return a


def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------
def smooth_max_adam(a0: np.ndarray, deadline: float,
                    step0: float = 0.05,
                    beta1: float = 0.9, beta2: float = 0.999,
                    eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise  max(conv(a,a))  using a soft‑max surrogate.
    Returns the best vector found (still on the simplex) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    alpha = 20.0               # start with a mild smoothing
    max_alpha = 300.0          # final “hard” max

    while time.time() < deadline:
        t += 1
        # 1) full convolution a*a
        conv = full_convolution(a)
        max_c = conv.max()

        # 2) soft‑max weights (stable)
        if t % 2000 == 0 and alpha < max_alpha:
            alpha = min(alpha * 1.3, max_alpha)
        exp_shift = np.exp(alpha * (conv - max_c))
        w = exp_shift / exp_shift.sum()          # w_k ≈ 1 for the maximal index

        # 3) gradient of  Σ_k w_k·conv_k  (see derivation in the docstring)
        w_rev = w[::-1]
        grad_full = fft_convolution(a, w_rev)    # length = 3·n‑2
        # extract the slice that corresponds to indices 0 … n‑1
        grad = 2.0 * grad_full[n - 1:2 * n - 1][::-1]

        # 4) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)                # keep feasibility

        # occasional exact evaluation
        if t % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """Greedy pairwise mass transfers with decreasing step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.03
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]                     # move a fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 120_000:
            # shake out of plateaux
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

        remaining = len(seeds) - idx
        time_per = (deadline - time.time()) / max(remaining, 1)

        a_opt, score_opt = smooth_max_adam(a0, time.time() + time_per)

        # Official evaluation (the only place we call ``evaluate_sequence``)
        val = evaluate_sequence(a_opt.tolist())
        print(f"[run {idx+1}/{len(seeds)}] opt‑score = {score_opt:.6f}, "
              f"official = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = a_opt.tolist()

    # ------------------------------------------------------------------
    # 3. Final stochastic hill‑climb starting from the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, deadline)
        val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] final official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()

    # ------------------------------------------------------------------
    # 4. Fallback – a plain reverse‑power‑law if nothing else succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback = reverse_power_law(n, 1.8)
        best_seq = fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback] score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Report final result
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
