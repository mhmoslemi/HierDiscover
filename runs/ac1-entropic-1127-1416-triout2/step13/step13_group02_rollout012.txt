def l_bfgs_polish(a0, deadline, alpha=80.0, max_iter=5000):
    n = len(a0)
    u0 = np.log(a0 + 1e-20)
    def objective(u):
        a = softmax(u)
        conv = full_convolution(a)
        # Smooth max using log-sum-exp
        # shift for numeric stability
        shift = alpha * conv.max()
        exp_term = np.exp(alpha * conv - shift)
        Z = exp_term.sum()
        smooth_max = (np.log(Z) + shift) / alpha
        f = 2.0 * n * smooth_max
        # gradient
        w = exp_term / Z
        # compute g_a
        w_rev = w[::-1]
        conv_aw = full_convolution_general(a, w_rev)   # length 3n-2
        idx_start = 2 * n - 2
        g_a = conv_aw[idx_start - np.arange(n)]
        # gradient wrt u
        g_u = a * (g_a - np.dot(g_a, a))
        grad = 2.0 * n * g_u
        return f, grad
    # callback
    def callback(u):
        if time.time() >= deadline:
            return True
    # Run minimize
    opts = {'maxfun': max_iter, 'maxiter': max_iter, 'disp': False}
    try:
        res = opt.minimize(fun=lambda u: objective(u), x0=u0, method='L-BFGS-B',
                           jac=True, callback=callback, options=opts)
    except Exception as e:
        # fallback to ADAM
        return a0
    a_opt = softmax(res.x)
    # project onto simplex just to be safe
    a_opt = np.maximum(a_opt, 0.0)
    s = a_opt.sum()
    if s == 0:
        return a0
    a_opt = a_opt / s
    return a_opt

"""
Hybrid global‑search + smooth‑max L‑BFGS optimizer for the
``evaluate_sequence`` objective.

Stage 1 – Global sweep:
    * parametric families: power‑law, exponential, beta,
      valley‑power (product of left/right power‑laws),
      symmetric‑exponential (bell‑shape).
    * many random Dirichlet draws with a wide range of
      concentration parameters.
    The ten (or more) best vectors (normalised to sum = 1) are kept.

Stage 2 – Smooth‑max polishing:
    The non‑smooth objective “max_k (a*a)_k” is replaced by a
    log‑sum‑exp surrogate:
        smooth_max(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    with α≈80.  Its gradient can be expressed via a second
    convolution, so each L‑BFGS iteration costs a single FFT.
    The vector is parametrised as a soft‑max of an unrestricted
    vector u (a_i = exp(u_i)/Σ_j exp(u_j)), guaranteeing non‑negativity
    and the simplex constraint.

Stage 3 – Random‑restart polishing & optional annealing:
    While time permits additional random starts are polished with the
    same L‑BFGS routine; finally a short simulated‑annealing pass may
    escape shallow local minima.

All stages respect a 1000 s wall‑clock budget.  The function
``propose_candidate`` returns a plain list of floats suitable for the
official evaluator.
"""

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT."""
    m = len(a) + len(b) - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return fft_convolve(a, a)


def full_convolution_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution of a (len n) with b (len m) – returns length n+m‑1."""
    return fft_convolve(a, b)


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def softmax(u: np.ndarray) -> np.ndarray:
    """Stable soft‑max; maps ℝⁿ → simplex."""
    m = np.max(u)
    e = np.exp(u - m)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a simplex vector (∑a = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_valley_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p} – a U‑shaped power law."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i * (n - i + 1.0)) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_symmetric_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·|i‑centre|) – bell‑shaped around the centre."""
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(np.arange(n, dtype=float) - centre))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global baseline exploration
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 10):
    """
    Sweep the parametric families and draw many Dirichlet vectors.
    Keeps the ``max_keep`` best (lowest score) vectors.
    """
    best = []   # list of (score, vector)

    def keep_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Power‑law sweep (very wide)
    for p in np.linspace(0.05, 15.0, 500):
        if time.time() >= deadline:
            break
        keep_candidate(generate_powerlaw(n, p))

    # 2) Exponential sweep (broad)
    for lam in np.linspace(1e-5, 2.0, 500):
        if time.time() >= deadline:
            break
        keep_candidate(generate_exponential(n, lam))

    # 3) Beta sweep (U‑shapes + asymmetric)
    al_vals = np.concatenate([
        np.geomspace(0.01, 0.2, 14),
        np.linspace(0.2, 6.0, 28)
    ])
    be_vals = al_vals
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            keep_candidate(generate_beta(n, alpha, beta))

    # 4) Valley power‑law (product of left‑ and right‑hand power laws)
    for p in np.linspace(0.05, 8.0, 200):
        if time.time() >= deadline:
            break
        keep_candidate(generate_valley_powerlaw(n, p))

    # 5) Symmetric exponential (bell‑shape)
    for lam in np.linspace(0.01, 3.0, 200):
        if time.time() >= deadline:
            break
        keep_candidate(generate_symmetric_exponential(n, lam))

    # 6) Massive Dirichlet draws with varied concentrations
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per = 1800   # ≈ 16 200 draws – still fast
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            keep_candidate(a)

    # Return only the vectors, sorted by score (lowest first)
    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS polishing (soft‑max parametrisation)
# ----------------------------------------------------------------------


class SmoothMaxObjective:
    """
    Callable objective for L‑BFGS.
    Given a vector u, defines a = softmax(u) (lies in the simplex) and
    returns the smoothed objective
        f(u) = 2·n·smooth_max(conv(a,a)),
    together with its gradient ∂f/∂u.
    The smooth_max uses a log‑sum‑exp with temperature α.
    """

    def __init__(self, n: int, alpha: float):
        self.n = n
        self.alpha = alpha

    def __call__(self, u: np.ndarray):
        a = softmax(u)                     # a ∈ simplex
        conv = full_convolution(a)         # length 2·n‑1

        # --- log‑sum‑exp (numerically stable) ---
        shift = self.alpha * np.max(conv)
        exp_term = np.exp(self.alpha * conv - shift)
        Z = np.sum(exp_term)
        smooth_max = (np.log(Z) + shift) / self.alpha

        f = 2.0 * self.n * smooth_max      # exact objective we finally report

        # --- gradient ---
        w = exp_term / Z                   # softmax weights, sum to 1
        # g_a = 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]                    # reverse for convolution trick
        conv_aw = full_convolution_general(a, w_rev)   # length 3·n‑2
        idx_start = 2 * self.n - 2
        g_a = conv_aw[idx_start - np.arange(self.n)]

        # Jacobian of soft‑max: ∂a/∂u = diag(a) – a aᵀ
        g_u = a * (g_a - np.dot(g_a, a))
        grad = 2.0 * self.n * g_u
        return f, grad


def l_bfgs_polish(a0: np.ndarray, deadline: float,
                  alpha: float = 80.0,
                  max_iter: int = 2000) -> np.ndarray:
    """
    Refine a simplex vector by minimising the smooth‑max surrogate with
    L‑BFGS.  Returns the (still normalised) vector with the lowest
    exact score observed during optimisation.
    """
    n = a0.shape[0]
    u0 = np.log(a0 + 1e-20)               # initialise in log‑space

    obj = SmoothMaxObjective(n, alpha)

    # ``disp`` disabled – we only need the final point.
    # L‑BFGS will stop after ``max_iter`` function evaluations.
    try:
        res = opt.minimize(fun=obj,
                           x0=u0,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter, 'disp': False})
    except Exception:
        # In the unlikely event that the optimiser crashes, fall back.
        return a0

    a_opt = softmax(res.x)
    a_opt = np.maximum(a_opt, 0.0)
    s = a_opt.sum()
    if s == 0:
        return a0
    a_opt /= s
    # Clip against the small time budget – if we are already past the deadline,
    # just return the last feasible vector.
    if time.time() > deadline:
        return a_opt
    return a_opt


# ----------------------------------------------------------------------
# Light simulated‑annealing (optional final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling)

        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # Problem size – a trade‑off between expressivity and speed.
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previous run
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 350.0   # reserve most budget for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=15)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smooth‑max L‑BFGS polishing (primary + random restarts)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = l_bfgs_polish(a0, polish_deadline,
                                alpha=80.0, max_iter=2500)
        val = compute_score(refined)
        print(f"[L‑BFGS polish‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # Extra random‑restart polishing while time permits
        while time.time() < polish_deadline:
            conc = np.random.choice([0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = l_bfgs_polish(a_start, polish_deadline,
                                    alpha=80.0, max_iter=2000)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[L‑BFGS restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final short simulated‑annealing polish (optional)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback to uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
