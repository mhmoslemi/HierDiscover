def eg_descent(a_init, deadline):
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    # schedule alphas
    alphas = [5.0, 10.0, 20.0, 40.0, 80.0]
    step0 = 0.5
    while time.time() < deadline:
        for alpha in alphas:
            # inner loop limited by time and maybe iteration count
            for _ in range(2000):
                if time.time() >= deadline:
                    break
                t += 1
                # compute conv
                conv = full_convolution_fft(a)
                # compute softmax probabilities p
                # scaling
                scaled = alpha * conv
                max_scaled = np.max(scaled)
                exps = np.exp(scaled - max_scaled)
                p = exps / np.sum(exps)

                # gradient
                p_rev = p[::-1]
                conv2 = fft_convolve(p_rev, a)  # length = 3n-2
                idx = (2*n - 2) - np.arange(n)
                g = 2.0 * conv2[idx]

                # step size
                eta = step0 / (alpha * math.sqrt(t))

                # exponentiated update
                log_a = np.log(a + 1e-30) - eta * g
                # re-normalize via exp stable
                log_a -= np.max(log_a)
                a = np.exp(log_a)
                a /= a.sum()

                # evaluate every 200 steps
                if t % 200 == 0:
                    score = compute_score(a)
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                        # possibly increase step size a bit
                        step0 = min(step0 * 1.02, 2.0)
        # optionally break if no improvement after full cycle?
        # Could check stagnation but for simplicity continue.
    return best_a, best_score

def fft_convolve(x, y):
    n = x.shape[0] + y.shape[0] - 1
    N = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv

def hill_climb(a_start, deadline, max_iter=500_000):
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    step_scale = 0.02
    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] = max(cand[i] - delta, 0.0)
        cand[j] = cand[j] + delta
        cand = project_to_simplex(cand)
        score = compute_score(cand)
        if score < best_score:
            best_score = score
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1
        it += 1
        if no_improve > 100_000:
            # jitter
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0
    return best_a

"""
A fast “exponentiated‑gradient” optimiser for the self‑convolution bound.

The objective for a vector a with Σa = 1 is
    J(a) = 2·n·max (a * a)            (full convolution)
which is scale‑invariant.  By smoothing the max with a soft‑max
(log‑sum‑exp) we obtain a differentiable surrogate.  The gradient of the
smoothed objective can be written as a convolution and is therefore
computed efficiently with FFTs.  Using a multiplicative (exponentiated)
update keeps the iterate on the probability simplex without an explicit
projection.

The optimiser works as follows:
  1. Initialise a from a reasonable seed (fallback, geometric, or a random
     Dirichlet draw) and normalise it.
  2. Run a sequence of smoothness levels – α grows from a few hundred up to
     twenty‑thousand – performing many exponentiated‑gradient steps at each
     level.  The step size is tuned inversely to α.
  3. Keep the best vector seen (according to the exact max‑convolution
     score) and, after the smoothing phase, polish it with a short greedy
     hill‑climb that moves a small amount of mass between two coordinates.
  4. Return the best sequence found before the time budget expires.

The method explores the continuous landscape more aggressively than the
previous sub‑gradient / simulated‑annealing approach and fits easily inside
the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based convolutions)
# ----------------------------------------------------------------------
def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of two real vectors using a real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    N = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    c = np.fft.irfft(X * Y, N)[:n]
    return c


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a (full length 2·n‑1)."""
    return fft_convolve(a, a)               # length = 2·n‑1


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple families for initialisation
# ----------------------------------------------------------------------
def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^i  (0<q<1).  Returned vector is already normalised."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def random_dirichlet(n: int, conc: float) -> np.ndarray:
    """Draw a random point from Dirichlet(conc,…,conc) and normalise."""
    if conc <= 0.0:
        conc = 1.0
    y = np.random.gamma(conc, 1.0, size=n)
    s = y.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return y / s


# ----------------------------------------------------------------------
# Project a vector onto the probability simplex (used by the hill‑climb)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Greedy hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float, max_iter: int = 800_000) -> np.ndarray:
    """Deterministic greedy mass‑transfer hill‑climb."""
    a = a_start.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] = max(cand[i] - delta, 0.0)
        cand[j] = cand[j] + delta
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1
        it += 1
        if no_improve > 150_000:
            # jitter to escape a possible flat region
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on a smoothed max objective
# ----------------------------------------------------------------------
def eg_descent(a_init: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Smooth‑max (log‑sum‑exp) based exponentiated‑gradient descent.
    Returns the best vector seen and its exact score.
    """
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    # smoothing schedule – gradually sharpen the soft‑max
    alphas = [500.0, 2000.0, 5000.0, 10000.0, 20000.0]
    base_step = 0.15   # base step size (will be scaled by 1/α)

    t = 0  # iteration counter used for step‑size decay

    while time.time() < deadline:
        for alpha in alphas:
            # how many inner steps we allow for this α? (time‑driven)
            inner_deadline = deadline - 0.05  # a tiny safety margin
            while time.time() < inner_deadline:
                t += 1
                # ----- convolution of a with itself -----
                conv = full_convolution(a)                     # length = 2n‑1

                # ----- soft‑max probabilities over the convolution entries -----
                scaled = alpha * conv
                max_scaled = np.max(scaled)                    # for numerical stability
                exp_shifted = np.exp(scaled - max_scaled)
                p = exp_shifted / exp_shifted.sum()            # Σ p = 1

                # ----- gradient of the smoothed max -----
                # g = 2 * (p_rev * a)_{(2n‑2)‑i}
                p_rev = p[::-1]
                conv2 = fft_convolve(p_rev, a)                 # length = 3n‑2
                idx = (2 * n - 2) - np.arange(n)               # positions needed
                g = 2.0 * conv2[idx]

                # ----- exponentiated‑gradient update -----
                # step size decays like 1/√t and is inversely proportional to α
                eta = (base_step / math.sqrt(t)) / (alpha + 1.0)
                # multiplicative update (equivalent to a log‑update)
                a = a * np.exp(-eta * g)
                # re‑normalise to the simplex (exactly Σa = 1)
                a_sum = a.sum()
                if a_sum == 0.0:                     # safety net – fall back to uniform
                    a = np.full(n, 1.0 / n)
                else:
                    a /= a_sum

                # ----- occasional exact evaluation -----
                if t % 200 == 0:
                    val = compute_score(a)
                    if val < best_val:
                        best_val, best_a = val, a.copy()
                        # a tiny boost of the step size after a good move
                        base_step = min(base_step * 1.02, 0.5)

                # stop early if we run out of time
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

    return best_a, best_val

    start_time = time.time()
    # leave a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float('inf')

    # -----------------------------------------------------------------
    # 0. Load the previous best construction if it exists
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore

        # normalise to sum = 1 (the optimisation works on the simplex)
        a_fallback = np.array(fallback, dtype=np.float64)

        if a_fallback.sum() == 0.0:

        else:

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    working_n = 4096
    # If the fallback vector is shorter, up‑sample it linearly to the target length
    if best_seq is not None and len(best_seq) != working_n:
        old = np.array(best_seq, dtype=float)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, working_n)

    else:

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
