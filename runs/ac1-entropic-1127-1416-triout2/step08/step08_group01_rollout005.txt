def targeted_hill_climb(a, deadline, max_iter_steps):
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.1
    it = 0
    while time.time() < deadline and it < max_iter_steps:
        # compute conv and max index
        conv = convolution_fft(best, best)
        max_val = conv.max()
        k = int(conv.argmax())
        # compute contributions vector of length n (where both indices valid)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)   # inclusive
        # contributions for i in [i_start,i_end]: a_i * a_{k-i}
        # vector of contributions sized n, zero elsewhere
        contributions = np.zeros(n)
        i_idx = np.arange(i_start, i_end+1)
        j_idx = k - i_idx
        contributions[i_idx] = best[i_idx] * best[j_idx]
        # probability distribution for donor based on contributions
        if contributions.sum() == 0:
            donor = np.random.randint(n)
        else:
            probs = contributions / contributions.sum()
            donor = np.random.choice(n, p=probs)
        # partner index j = k - donor
        partner = k - donor
        # donor may be invalid (partner out of range), check:
        if partner < 0 or partner >= n:
            # fallback random donor
            donor = np.argmax(contributions)  # highest contribution among valid
            partner = k - donor
        # Recipient index selection: low values
        low_mask = best < (best.mean())
        low_indices = np.where(low_mask)[0]
        if low_indices.size == 0:
            recipient = np.random.randint(n)
        else:
            recipient = np.random.choice(low_indices)
        # Determine amount to move
        delta = step_scale * best[donor]
        if delta <= 0:
            delta = step_scale * best.mean()
        # Ensure not moving more than donor has
        delta = min(delta, best[donor])
        # Apply move: decrease donor, increase recipient
        cand = best.copy()
        cand[donor] -= delta
        cand[recipient] += delta
        # Ensure non-negativity (donor may become zero) and sum unchanged.
        cand = np.maximum(cand, 0.0)
        # Slight re-projection onto simplex may adjust small numeric errors
        cand = project_to_simplex(cand)

        sc = compute_score(cand)

        if sc < best_score - 1e-12:  # improvement
            best_score = sc
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97
        it += 1

    return best, best_score

Strategy
--------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) for a vector normalised to sum = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytic family a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return the normalised α‑family vector."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    # guard against overflow when α is negative large
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def extended_alpha_search(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Coarse + fine α grid search."""
    best_a = None
    best_sc = float("inf")
    best_alpha = 0.0

    # ----- coarse grid (step≈0.1) -----
    alphas = np.concatenate([np.linspace(-1.0, 0.0, 11),
                             np.linspace(0.0, 5.0, 51)])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            best_alpha = alpha

    # ----- fine grid around the best coarse α -----
    fine_low = max(best_alpha - 0.25, -2.0)
    fine_high = best_alpha + 0.25
    alphas_fine = np.linspace(fine_low, fine_high, 51)  # step 0.01
    for alpha in alphas_fine:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            best_alpha = alpha

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Random starts (sparse or Dirichlet) for exp‑gradient
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    sparsity = np.random.uniform(0.03, 0.35)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dirichlet_start(n: int) -> np.ndarray:
    """Draw a random vector from a Dirichlet distribution with random concentration."""
    concentration = np.random.uniform(0.1, 5.0)
    a = np.random.dirichlet(np.full(n, concentration))
    return a


def random_start(n: int) -> np.ndarray:
    """Return a random non‑negative probability vector – either sparse or Dirichlet."""
    if np.random.rand() < 0.5:
        return random_sparse_start(n)
    else:
        return random_dirichlet_start(n)


# ----------------------------------------------------------------------
# 3. Exponentiated‑gradient optimisation of the smooth max
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 200_000,
                        beta_start: float = 0.5,
                        beta_end: float = 300.0,
                        eta0: float = 0.7,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent on the log‑sum‑exp surrogate of max(a∗a).
    Returns the best exact score observed.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- smooth soft‑max weights ----
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))        # stabilised
        w = expw / expw.sum()

        # ---- gradient of the smoothed objective ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ---- step size ----
        eta = eta0 / np.sqrt(it)

        # ---- KL‑mirror update ----
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ---- periodic exact evaluation ----
        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Targeted stochastic hill‑climb (exact objective)
# ----------------------------------------------------------------------


def targeted_hill_climb(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 2_500_000) -> tuple[np.ndarray, float]:
    """
    Move mass away from the pairwise products that dominate the current
    convolution peak.  Only moves that strictly improve the exact score
    are accepted.  The step‑size is adapted dynamically.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12  # initial fraction of donor mass to transfer

    steps = 0
    while steps < max_steps and time.time() < deadline:
        # --- compute convolution and locate the peak ---
        conv = convolution_fft(best_a, best_a)
        k = int(conv.argmax())          # shift where max occurs

        # --- contributions to that peak (a_i * a_{k-i}) ---
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        contrib = best_a[idx_i] * best_a[idx_j]

        # choose donor index proportional to its contribution
        if contrib.sum() > 0:
            probs = contrib / contrib.sum()
            donor = np.random.choice(idx_i, p=probs)
        else:
            donor = np.random.randint(n)

        # partner of the donor (might be the same when 2*donor == k)
        partner = k - donor

        # --- pick a recipient with relatively low mass ---
        low_mask = best_a < (best_a.mean())
        low_indices = np.where(low_mask)[0]
        if low_indices.size == 0:
            recipient = np.random.randint(n)
        else:
            recipient = np.random.choice(low_indices)

        # --- amount of mass to move ---
        delta = step_scale * best_a[donor]
        delta = min(delta, best_a[donor])           # cannot go negative
        if delta <= 0:
            delta = step_scale * (best_a.mean())
            delta = min(delta, best_a[donor])

        # --- propose the move ---
        cand = best_a.copy()
        cand[donor] -= delta
        cand[recipient] += delta

        # enforce feasibility (non‑negativity & simplex)
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)

        # --- accept only improving moves ---
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # be a little bolder
        else:
            step_scale *= 0.97                         # get more cautious

        steps += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine follows the multi‑stage plan described in the module docstring
    and respects the 1000 s wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # a small safety margin

    # -------------------------------------------------
    # Choose a workable length (larger n ⇒ better bound but more work)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Extended α‑grid search
    # -------------------------------------------------
    a_alpha, sc_alpha = extended_alpha_search(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Random sparse/Dirichlet restarts with exponentiated gradient
    # -------------------------------------------------
    while time.time() < deadline - 250.0:
        a0 = random_start(n)
        a_opt, sc_opt = exp_smooth_optimize(
            a0,
            deadline - 180.0,
            max_iters=120_000,
            beta_start=0.5,
            beta_end=250.0,
            eta0=0.7,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after random restart = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Long dedicated exponentiated‑gradient run on the current best
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] long exponentiated‑gradient optimisation")
        a_long, sc_long = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=400_000,
            beta_start=0.5,
            beta_end=300.0,
            eta0=0.6,
        )
        if sc_long < best_score:
            best_score = sc_long
            best_seq = a_long.tolist()
            print(f"[{time.time():.1f}] after long exp‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Final targeted hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] targeted hill‑climb")
        a_final, sc_final = targeted_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=2_500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a short uniform vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats as required by the harness.
    return best_seq
