"""
Propose a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`.  The objective equals

    2·n·max(conv(a,a))   with  Σ a_i = 1

where `conv` is the ordinary linear convolution.  This is the
probability that the sum of two independent draws from the distribution
`a` falls into a single point; we want that probability as small as
possible.

The search strategy:

1. **Random restarts** – generate many random probability vectors
   (dense Dirichlet or sparse √n‑support).  For each start run a fast
   **exponentiated‑gradient descent** that directly follows a sub‑gradient
   of the true max‑convolution objective.  Positivity and the simplex
   constraint are built‑in, so only a few hundred iterations are needed.

2. **Simulated‑annealing polish** – starting from the best vector found
   above, perform a long stochastic hill‑climb (mass‑transfer or swap
   moves).  Moves are accepted if they improve the exact objective or,
   with a temperature‑dependent probability, if they worsen it; this
   helps escape shallow local minima.

3. **Fallback** – if nothing is found, return a uniform vector.

All work respects the 1000 s wall‑clock limit; intermediate best scores
are printed for debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, scoring, simplex projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for large arrays)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector already normalised to Σa_i = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    start: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex with KL‑divergence (i.e. exponentiated
    gradient).  At each iteration we move opposite to a sub‑gradient of
    the active convolution entry.
    """
    a = start.astype(float).copy()
    n = a.shape[0]

    # keep the best exact score seen
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current convolution and the index attaining the maximum
        conv = convolution_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of the active entry (2*a_{k-i} where defined)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # step size – decreasing like 1/√t
        eta = eta0 / np.sqrt(t)

        # exponentiated update and renormalisation (implicit projection)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:          # numerical safeguard – fall back to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (mass‑transfer / swap moves)
# ----------------------------------------------------------------------


def simulated_annealing(
    start: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
    init_temp: float = 0.02,
    final_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that accepts worsening moves with a
    temperature that decays linearly in the remaining time.
    """
    a = start.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    # helper – recompute the current exact score
    cur_score = best_score
    start_anneal = time.time()
    total_time = max(deadline - start_anneal, 1e-9)

    step_scale = 0.08  # initial fraction of mass to move

    for step in range(max_steps):
        now = time.time()
        if now >= deadline:
            break

        # linearly decreasing temperature
        elapsed = now - start_anneal
        frac = max(0.0, 1.0 - elapsed / total_time)
        T = init_temp * frac + final_temp

        # propose a candidate by either swapping or transferring mass
        cand = a.copy()
        i, j = np.random.randint(n, size=2)
        if np.random.rand() < 0.5:
            # simple swap
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        # re‑normalise (keeps the sum exactly 1, preserves non‑negativity)
        cand = cand / cand.sum()

        cand_score = compute_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / T):
            a = cand
            cur_score = cand_score
            # make the step size a little bolder after a success
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # become more conservative after a rejection
            step_scale *= 0.96

        # keep the global best
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            # optionally print improvement (very cheap)
            print(f"Anneal new best: {best_score:.6f}")

    return best_a, best_score

            best_val = scr_polished
            best_seq = a_polished.tolist()
            print(f"Polish improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Fallback to a trivial uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = compute_score(np.array(best_seq, dtype=float))

    # final debug print (the harness will call `evaluate_sequence` again)
    final_score = compute_score(np.array(best_seq, dtype=float))
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # plain list is required
    return best_seq
