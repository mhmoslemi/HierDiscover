"""
Non‑negative step‑function optimizer.

Goal
----
Find a sequence `a` (list of non‑negative floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the full linear convolution.
The objective is invariant under scaling, therefore we keep the vector
normalised to the probability simplex (sum = 1) during optimisation.

Method
------
1. **Initialisation** – start from the best known construction
   (`height_sequence_1` if present) and upscale it to a larger size
   (8192 points).  If it is missing we fall back to the truncated
   central‑binomial coefficients.

2. **Random restarts** – a few short Exponentiated‑Gradient runs from
   random simplex points (Dirichlet draws) are performed; the best
   candidate is kept for the long run.

3. **Exponentiated‑gradient on a smooth surrogate** – minimise the
   log‑sum‑exp approximation of `max(conv)`.  The update is

        a_i ← a_i · exp(‑η·∂F/∂a_i)   ,   a ← a / sum(a)

   with an annealing schedule for the smoothing parameter β (5 → 200)
   and for the step size η (1.0 → 1e‑3).  This respects the simplex
   constraints and tends to spread the mass away from the currently
   largest convolution entries.

4. **Projected sub‑gradient descent** on the exact objective
   (`2·n·max(conv)`) for polishing.

5. **Stochastic hill‑climb** – cheap mass‑transfer moves (swap or
   fractional transfer) with simplex re‑normalisation, running until
   the time budget is exhausted.

All stages respect the remaining wall‑clock budget; the best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large vectors)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum == 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def safe_normalise(v: np.ndarray) -> np.ndarray:
    """Make sure the vector sums to one (handles possible under‑flow)."""
    v = np.maximum(v, 1e-16)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# ----------------------------------------------------------------------
# Upscaling utilities
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Random/simplex helpers
# ----------------------------------------------------------------------


def random_simplex_vector(n: int) -> np.ndarray:
    """Draw a random point from the n‑dimensional simplex (Dirichlet(1))."""
    v = np.random.rand(n)
    return safe_normalise(v)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    eta_start: float = 1.0,
    eta_end: float = 1e-3,
    report_every: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth surrogate
       F_β(a) = (1/β)·log Σ exp(β·conv(a,a))

    The gradient is obtained analytically and the update respects the
    simplex via a multiplicative step followed by renormalisation.
    """
    a = safe_normalise(start_a.astype(float).copy())
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothing (β) and learning rate (η)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        eta = eta_start * (eta_end / eta_start) ** t

        # smoothed weights (log‑sum‑exp trick)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()          # soft‑max over convolution entries

        # gradient of the smoothed objective
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # multiplicative (exponentiated) update
        step = -eta * grad
        step = np.clip(step, -20.0, 20.0)            # prevent overflow/underflow
        a = a * np.exp(step)
        a = safe_normalise(a)

        if it % report_every == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[ExpGrad] it {it:6d} – new best = {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
    report_every: int = 1000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑objective."""
    a = safe_normalise(init_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = safe_normalise(a)

        if t % report_every == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[SubGrad] it {t:6d} – new best = {best_score:.6f}")

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
    report_every: int = 5000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = safe_normalise(a)           # ensure it stays on the simplex
    best_score = compute_score(best_a)

    step_scale = 0.10                     # initial mass‑transfer fraction

    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:        # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                             # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = safe_normalise(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        steps += 1
        if steps % report_every == 0:
            print(f"[HillClimb] step {steps:7d} – best = {best_score:.6f}")

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # small safety margin

    # ------------------------------------------------------------------
    # 0. Load a known good construction if it exists
    # ------------------------------------------------------------------
    try:
        init_seq = np.array([float(x) for x in height_sequence_1], dtype=float)
        print(f"Loaded height_sequence_1 (len={len(init_seq)}) as seed.")
    except Exception:
        init_seq = None

    # ------------------------------------------------------------------
    # 1. Decide final length (larger length gives more freedom)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 700:
        n_big = 16384
    else:
        n_big = 8192

    # ------------------------------------------------------------------
    # 2. Build the starting vector
    # ------------------------------------------------------------------
    if init_seq is not None:
        # upscale the known good solution
        a0 = upscale_vector(init_seq, n_big)
    else:
        # fallback: truncated central‑binomial coefficients
        a0 = np.empty(n_big, dtype=float)
        a0[0] = 1.0
        for i in range(n_big - 1):
            a0[i + 1] = a0[i] * (2 * i + 1) / (2.0 * (i + 1))
        a0 = safe_normalise(a0)

    best_a = a0.copy()
    best_score = compute_score(best_a)
    print(f"Initial score (len={n_big}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Random restarts (short exp‑gradient runs) – modest time slice
    # ------------------------------------------------------------------
    restarts = 4
    restart_time = deadline - time.time() - 250.0   # keep ~250 s for the main phase
    if restart_time > 20 and restarts > 0:
        print(f"Running {restarts} random restarts (short runs).")
        for r in range(restarts):
            if time.time() >= deadline - 250:
                break
            a_rand = random_simplex_vector(n_big)
            # very short run – 3000 iterations are enough for a quick probe
            a_tmp, sc_tmp = exp_gradient_optimize(
                a_rand,
                time_deadline=time.time() + min(30.0, restart_time / restarts),
                max_iters=3000,
                beta_start=5.0,
                beta_end=50.0,
                eta_start=0.5,
                eta_end=0.02,
                report_every=1000,
            )
            if sc_tmp < best_score:
                best_score = sc_tmp
                best_a = a_tmp
                print(f"  Restart {r+1}: improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Long exponentiated‑gradient phase
    # ------------------------------------------------------------------
    phase1_deadline = deadline - 150.0   # reserve ~150 s for later polishing
    print(f"Phase‑1 (exp‑gradient) until t={phase1_deadline - start_time:.1f}s")
    best_a, best_score = exp_gradient_optimize(
        best_a,
        time_deadline=phase1_deadline,
        max_iters=200_000,
        beta_start=5.0,
        beta_end=200.0,
        eta_start=1.0,
        eta_end=1e-3,
        report_every=5000,
    )
    print(f"After exp‑gradient best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Sub‑gradient polishing (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 40.0:
        print("Phase‑2 (sub‑gradient refinement).")
        refined_a, refined_score = subgrad_optimize(
            best_a,
            time_deadline=deadline - 40.0,
            max_iter=300_000,
            step0=0.07,
            report_every=2000,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_a = refined_a
            print(f"Sub‑grad improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("Phase‑3 (hill‑climb polishing).")
        polished_seq, polished_score = hill_climb_permutation(
            best_a.tolist(),
            time_deadline=deadline,
            max_steps=800_000,
            report_every=10000,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_a = np.array(polished_seq, dtype=float)
            print(f"Hill‑climb further improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Safeguard – fall back to a uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_a is None or best_a.size == 0:
        n_fallback = 256
        best_a = np.full(n_fallback, 1.0 / n_fallback)
        best_score = compute_score(best_a)

    final_score = compute_score(best_a)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_a.tolist()
