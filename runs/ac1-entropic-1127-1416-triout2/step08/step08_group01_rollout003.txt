"""
A fast Adam‑based optimiser for the “max‑self‑convolution” problem.

The algorithm works on the probability simplex by representing a
sequence `a` as a soft‑max of an unconstrained vector `z`:

    a_i = exp(z_i) / sum_j exp(z_j)

The objective to be minimised is

    f(a) = 2·n·max_k (a * a)_k

which is scale‑invariant.  We smooth the max by a log‑sum‑exp
approximation with a temperature `β` and minimise the smooth
objective by Adam in the `z`‑space.  `β` is increased
exponentially (continuation) so that the final iterate approaches the
true non‑smooth optimum.

The optimiser is run several times from diverse random starts, the
best exact score is kept, and a short stochastic hill‑climb is used
as a final polish.  All steps respect the 1000 s wall‑clock budget
and return a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast for large n)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector with sum(a)=1.
    This matches `evaluate_sequence` under the same normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simple simplex projection (Euclidean) – used only in the final hill‑climb
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic one‑parameter family (α‑grid) – cheap deterministic start
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # avoid infinities for extreme alphas
    w[~np.isfinite(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a modest α‑grid and keep the best vector (exact score)."""
    best_a = None
    best_score = float("inf")
    # grid from -1 to 2.5 with a denser positive side
    alphas = np.concatenate(
        [np.linspace(-1.0, 0.0, 13), np.linspace(0.0, 2.5, 51)]
    )
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Random sparse initialisation – diverse starting points
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Return a random sparse probability vector (simplex)."""
    # sparsity fraction in [0.02, 0.3]
    sparsity = np.random.uniform(0.02, 0.30)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. soft‑max params
# ----------------------------------------------------------------------


def smooth_obj_grad(z: np.ndarray, beta: float) -> tuple[float, np.ndarray, np.ndarray]:
    """
    For a given log‑param vector `z` (soft‑max representation) and temperature
    `beta`, return
        f    – the smoothed objective value,
        g_z  – its gradient w.r.t. `z`,
        a    – the probability vector a = softmax(z).
    """
    n = z.size
    # soft‑max (stable)
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()

    # convolution a*a
    conv = convolution_fft(a, a)                     # length 2n‑1
    conv_max = conv.max()
    # soft‑max weights over the convolution entries (log‑sum‑exp)
    w_raw = np.exp(beta * (conv - conv_max))         # shift for stability
    Z = w_raw.sum()
    w = w_raw / Z

    # gradient ∂f/∂a = 2 * (a_rev ⊛ w) (central part)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1: n - 1 + n]        # size n

    # convert to gradient w.r.t. z (soft‑max Jacobian)
    dot = np.dot(a, grad_a)
    g_z = a * (grad_a - dot)

    # smoothed objective value (log‑sum‑exp / β)
    f = (np.log(Z) + beta * conv_max) / beta

    return f, g_z, a


# ----------------------------------------------------------------------
# Adam optimiser in the `z`‑space (mirrored Adam on the simplex)
# ----------------------------------------------------------------------


def adam_optimize(
    z0: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 0.5,
    beta_end: float = 800.0,
    step_size: float = 0.20,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    Returns the best exact (non‑smoothed) vector seen and its score.
    """
    n = z0.size
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = None
    best_score = float("inf")

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # continuation schedule for β (exponential increase)
        progress = t / max_iters
        beta = beta_start * (beta_end / beta_start) ** progress

        # evaluate smooth objective and its gradient
        f, g_z, a = smooth_obj_grad(z, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z * g_z)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # parameter update
        z = z - step_size * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation (every 250 iterations)
        if t % 250 == 0:
            sc = compute_score(a)          # a already on the simplex
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final evaluation
    _, _, a_final = smooth_obj_grad(z, beta_end)   # use the largest β
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Very light hill‑climb (swap / mass‑transfer moves) – final polishing
# ----------------------------------------------------------------------


def hill_climb(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb that only accepts strictly improving
    exact scores.  Moves are either a swap of two entries or a small
    transfer of mass between two positions.
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # scale of transferred mass (relative to the source entry)
    step_scale = 0.10

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 1e-14:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        # re‑project onto simplex (Euclidean)
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            # successful move – slightly increase step size
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # unsuccessful – shrink step size
            step_scale *= 0.96

        max_steps -= 1

    return best_a, best_sc

    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] after α‑grid best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # (2) Multiple Adam runs from random sparse starts
    # ------------------------------------------------------------------
    # we reserve about half of the remaining budget for Adam runs
    while time.time() < deadline - 250.0:
        # create a random sparse start and initialise soft‑max params
        a0 = random_sparse_start(n)
        # avoid exact zeros in log – add a tiny epsilon before log
        eps = 1e-15
        z0 = np.log(a0 + eps)

        # run Adam with a modest iteration budget (fast early restarts)
        a_opt, sc_opt = adam_optimize(
            z0,
            deadline - 180.0,
            max_iters=120_000,
            beta_start=0.5,
            beta_end=400.0,
            step_size=0.22,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after Adam restart = {best_score:.6f}")

    # ------------------------------------------------------------------
    # (3) Long dedicated Adam optimisation from the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] long Adam optimisation from best")
        a_start = np.array(best_seq, dtype=float)
        z_start = np.log(a_start + 1e-15)
        a_long, sc_long = adam_optimize(
            z_start,
            deadline - 120.0,
            max_iters=350_000,
            beta_start=0.5,
            beta_end=800.0,
            step_size=0.18,
        )
        if sc_long < best_score:
            best_score = sc_long
            best_seq = a_long.tolist()
            print(f"[{time.time():.1f}] after long Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # (4) Final hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=1_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # (5) Safeguard – if nothing succeeded, fall back to uniform short vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (also used by the external harness)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain list of floats as required by the harness
    return best_seq
