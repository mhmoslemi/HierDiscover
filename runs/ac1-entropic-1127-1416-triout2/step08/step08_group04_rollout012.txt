"""
Propose an improved step‑function sequence.

The algorithm consists of five stages:

1️⃣ **Extensive shape sweep** – for several lengths (1024…16384) we
   evaluate many analytic families:
   uniform, arcsine, a parametrised “power’’ shape
   a_i ∝ ((i+½)(n‑½‑i)+ε)^‑p and mixtures of the latter with the
   uniform distribution.  A cheap golden‑section search finds the
   optimal exponent p for each length.  The best surrogate
   (2·n·max(conv(a,a))) is kept.

2️⃣ **Smooth‑max Adam refinement** – the best candidate is further
   optimised by Adam on the smooth‑max approximation of the max‑convolution.
   The temperature τ is annealed from 0.5 up to 2000 to approach the true
   max, while a decreasing learning‑rate and the Adam moment estimates give
   stable progress.

3️⃣ **Exact sub‑gradient descent** – a projected Euclidean sub‑gradient
   method on the true max‑convolution (no smoothing) removes the remaining
   bias of the smooth approximation.

4️⃣ **Exponentiated‑gradient polish** – mirror‑descent w.r.t. the KL‑divergence
   (a.k.a. exponentiated‑gradient) further reduces the max‑convolution.

5️⃣ **Cheap stochastic hill‑climbing** – random transfers of a tiny amount
   of mass between two coordinates accept only improving moves (according
   to the fast surrogate).  This sometimes escapes tiny basins.

All stages respect the global wall‑clock deadline (default 1000 s).  The
final candidate is evaluated with the official `evaluate_sequence` function
and returned as a plain Python list of non‑negative floats.
"""

import time
import numpy as np
from typing import List


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (fast and exact for real data).
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the O(n) algorithm from Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # Everybody non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to sum to 1;
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine law; already normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """
    Generalised arcsine family:
        a_i ∝ ((i+½)(n‑½‑i)+ε)^{-p} .
    The result is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    # Guard against pathological parameters
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def mix_shapes(a: np.ndarray, b: np.ndarray, w: float) -> np.ndarray:
    """Return w·a + (1‑w)·b, normalised to sum = 1."""
    mix = w * a + (1.0 - w) * b
    return mix / mix.sum()

# Optimisers
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 300_000,
                             tau_start: float = 0.5,
                             tau_end: float = 2000.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max approximation of max(conv).
    Optimisation variable are logits w with a = softmax(w).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities (add tiny offset)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # a = softmax(w)
        w_shift = w - np.max(w)          # numeric stabilisation
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a (length L)
        conv = convolution_fft(a, a)

        # smooth‑max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        # conv of a with reversed weight
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)                     # a simple decay
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if it % 100_000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries that are (almost) maximal
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent w.r.t. KL).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 30_000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Very cheap stochastic hill climbing:
    move a tiny amount of mass from one coordinate to another.
    Accept only moves that improve the surrogate score.
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.2 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1

    return a_cur


# ----------------------------------------------------------------------
# One‑dimensional search for the optimal power‑shape exponent
# ----------------------------------------------------------------------


def optimise_power_exponent(n: int,
                           lo: float = 0.05,
                           hi: float = 4.0,
                           iters: int = 30) -> (float, float):
    """
    Golden‑section search for the exponent p that minimises the surrogate
    of the power‑shape family for a fixed length n.
    Returns (best_p, best_score).
    """
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    a, b = lo, hi
    fa = compute_score(generate_power_shape(n, a))
    fb = compute_score(generate_power_shape(n, b))

    for _ in range(iters):
        c = b - phi * (b - a)
        d = a + phi * (b - a)
        fc = compute_score(generate_power_shape(n, c))
        fd = compute_score(generate_power_shape(n, d))

        if fc < fd:
            b, fb = d, fd
        else:
            a, fa = c, fc

    best_p = (a + b) / 2.0
    best_score = compute_score(generate_power_shape(n, best_p))
    return best_p, best_score

        uni = np.full(n, 1.0 / n)
        s = compute_score(uni)
        if s < best_surrogate:
            best_surrogate, best_seq = s, uni.tolist()
            print(f"  New best (uniform) surrogate={s:.6f}")

        # Arcsine
        arc = arcsine_initial(n)
        s = compute_score(arc)
        if s < best_surrogate:
            best_surrogate, best_seq = s, arc.tolist()
            print(f"  New best (arcsine) surrogate={s:.6f}")

        # Power‑shape – optimise exponent p
        p_opt, s_opt = optimise_power_exponent(n, lo=0.05, hi=4.0, iters=25)
        shape_opt = generate_power_shape(n, p_opt)
        if s_opt < best_surrogate:
            best_surrogate, best_seq = s_opt, shape_opt.tolist()
            print(f"  New best (power p={p_opt:.3f}) surrogate={s_opt:.6f}")

        # Mixtures arcsine / uniform
        for w in w_grid:
            mix = mix_shapes(arc, uni, w)
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"  New best (mix arc‑uni w={w:.2f}) surrogate={s:.6f}")

        # Mixtures power‑shape / uniform
        for w in w_grid:
            mix = mix_shapes(shape_opt, uni, w)
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"  New best (mix pow‑uni w={w:.2f}) surrogate={s:.6f}")

        # Random mixtures of several bases (to escape possible local minima)
        for _ in range(20):
            # draw random Dirichlet weights for three bases
            coeffs = rng.dirichlet([1.0, 1.0, 1.0])
            mix = (coeffs[0] * arc +
                   coeffs[1] * shape_opt +
                   coeffs[2] * uni)
            mix = mix / mix.sum()
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"  New best (random mix) surrogate={s:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max Adam refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print("\n--- Stage 2: Adam refinement (smooth‑max) ---")
        a_init = np.array(best_seq, dtype=float)
        a_refined = adam_optimize_smooth_max(
            a_init,
            deadline,
            max_iters=500_000,
            tau_start=0.5,
            tau_end=2000.0,
            base_lr=0.5,
        )
        s = compute_score(a_refined)
        print(f"  After Adam surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a_refined.tolist()
    else:
        print("\nSkipping Adam refinement (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        print("\n--- Stage 3: sub‑gradient descent ---")
        a = np.array(best_seq, dtype=float)
        a = subgradient_descent_exact(
            a,
            deadline,
            max_iter=400_000,
            step0=0.07,
        )
        s = compute_score(a)
        print(f"  After sub‑gradient surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping sub‑gradient descent (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 4 – exponentiated‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Stage 4: exponentiated‑gradient polish ---")
        a = np.array(best_seq, dtype=float)
        a = eg_refine(
            a,
            deadline,
            max_iter=250_000,
            eta0=0.25,
        )
        s = compute_score(a)
        print(f"  After EG surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping EG polish (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 5 – cheap stochastic local search
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Stage 5: local random search ---")
        a = np.array(best_seq, dtype=float)
        a = local_random_search(
            a,
            deadline,
            max_trials=60_000,
            step_scale=0.001,
            rng=rng,
        )
        s = compute_score(a)
        print(f"  After local search surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping local random search (insufficient time)")

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        final_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
