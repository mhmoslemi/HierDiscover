"""
Improved search for a non‑negative step‑function that minimises
`evaluate_sequence`.

The programme follows three stages (all on the probability simplex,
i.e. Σ a_i = 1, which leaves the objective scale‑invariant):

1. **Many random‑restart soft‑max Adam optimisation** – a smoother proxy
   of the max‑convolution (log‑sum‑exp) is minimised.  The annealing
   schedule for the softness β is much longer (β_end = 2000) than in the
   baseline, and the Adam step‑size is slightly larger.  Hundreds
   of short restarts are performed; the best exact score is kept.

2. **Projected sub‑gradient descent** – a classic sub‑gradient method
   on the true max‑objective refines the best candidate from stage 1.

3. **Simulated‑annealing hill‑climb** – the final polishing replaces the
   pure “accept‑only‑improvements’’ hill climb with a temperature‑driven
   Metropolis acceptance rule.  Small mass‑transfer moves (and occasional
   swaps) are accepted with probability exp(−Δscore/T), allowing the
   search to escape shallow local minima.  The temperature cools
   geometrically and the step‑size adapts to the observed acceptance
   ratio.

All heavy work respects the 1000 s wall‑clock limit; the routine finally
returns a plain list of floats (the height sequence).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the lengths we use)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth max optimisation (log‑sum‑exp) with Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 1.5,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a soft‑max surrogate of max(conv(a,a)) with Adam momentum.
    The routine returns the best *exact* score encountered.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β (exponential schedule)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- smoothed objective (log‑sum‑exp) -----
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        # stable log‑sum‑exp weights
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        w = w / w_sum

        # ----- gradient of the soft‑max -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # ----- Adam update -----
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 400 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 800_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the entry attaining the max
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

# 3. Simulated‑annealing hill‑climb polishing
# ----------------------------------------------------------------------


def simulated_annealing(
    start_seq,
    time_deadline,
    init_temp: float = 0.02,
    cooling_rate: float = 0.998,
    min_temp: float = 1e-6,
    max_steps: int = 5_000_000,
) -> tuple[list[float], float]:
    """
    Metropolis‑style annealing on the exact objective.
    Small mass‑transfer moves (or swaps) are proposed; worsening moves
    are accepted with probability exp(−Δscore / T).  The temperature
    decays geometrically.  The best sequence seen is returned.
    """
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]

    # ensure we are on the simplex (tiny numerical drift may have happened)
    a = project_to_simplex(a)

    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    T = init_temp
    step_scale = 0.07          # initial fraction of mass to move

    step = 0
    while step < max_steps and time.time() < time_deadline:
        # --- generate a candidate ---
        cand = a.copy()
        if np.random.rand() < 0.5:
            # simple swap of two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 1e-12:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        new_score = compute_score(cand)

        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > min_temp and np.random.rand() < np.exp(-delta / T):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            # improve step‑scale when we accept
            step_scale = min(step_scale * 1.05, 0.5)
            # cooling is already handled globally
        else:
            # become a little more conservative
            step_scale = max(step_scale * 0.97, 1e-4)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule
        T = max(T * cooling_rate, min_temp)

        step += 1

    return best_a.tolist(), best_score

def random_start(n: int) -> np.ndarray:
    """Return a random probability vector of length `n`."""
    if np.random.rand() < 0.55:
        # dense – Dirichlet(1) = uniform on simplex
        a = np.random.rand(n)
        a = a / a.sum()
    else:
        # sparse – put mass on roughly √n positions
        k = max(2, int(np.sqrt(n)))
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k
        a = project_to_simplex(a)
    return a

    # 2. Random‑restart smooth optimisation (phase A)
    # ------------------------------------------------------------------
    reserve_for_refine = 150.0          # seconds we keep for later phases
    while time.time() < deadline - reserve_for_refine:
        remaining = deadline - time.time() - reserve_for_refine
        # allocate a modest slice for each restart (5–20 s)
        slice_time = max(5.0, min(20.0, remaining * 0.4))
        restart_deadline = time.time() + slice_time

        a0 = random_start(n)
        a_opt, scr = smooth_gradient_optimize(
            a0,
            time_deadline=restart_deadline,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=2000.0,
            step0=1.5,
        )
        exact = compute_score(a_opt)
        if exact < best_val:
            best_val = exact
            best_seq = a_opt.tolist()
            print(f"New best after smooth phase: score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient refinement (phase B)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40.0:
        print("\nStarting sub‑gradient refinement")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 20.0,
            max_iter=800_000,
            step0=0.08,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Improved by sub‑grad: score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing polishing (phase C)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nSimulated‑annealing polishing")
        polished_seq, polished_score = simulated_annealing(
            best_seq,
            time_deadline=deadline,
            init_temp=0.02,
            cooling_rate=0.9975,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Annealing improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback to a simple uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final reporting (the harness will call evaluate_sequence again)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
