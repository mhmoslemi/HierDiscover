def convolution_fft(a, b):
    m = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((m-1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

"""
A fast local‑search optimiser for `evaluate_sequence`.

The objective after normalising the sequence to the probability simplex
(sum = 1) is

    score = 2·n·max(conv(a, a))

where `conv` is the full linear convolution.  The optimiser works in three
phases:

1. **Initialisation** – start from the known “central‑binomial” shape (or
   from the supplied `height_sequence_1`) and linearly up‑sample it to a
   longer vector (4096‑8192 entries).  The vector is projected onto the
   simplex.

2. **Smooth Adam optimisation** – minimise a log‑sum‑exp surrogate of the
   `max` using Adam with a schedule that slowly raises the smoothing
   parameter β (≈ 2 → 400).  After each step the vector is projected back to
   the simplex.

3. **Simulated‑annealing polishing** – a cheap stochastic mass‑transfer
   move (move a random fraction of the mass from one index to another)
   is proposed repeatedly.  Moves that improve the true score are always
   accepted; uphill moves are accepted with probability exp(−Δ/T) where the
   temperature T is linearly cooled to a tiny value.

Both phases keep the best sequence seen so far.  The whole search respects
the overall time‑budget (≈ 1000 s) and finally returns a plain list of
floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (real inputs)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(v: np.ndarray) -> float:
    """Exact objective for a vector on the simplex (sum(v)=1)."""
    n = v.size
    conv = convolution_fft(v, v)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Corner case: all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def central_binomial_seq(n: int) -> np.ndarray:
    """Generate the first n central‑binomial coefficients a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    return a


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.size
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Phase‑1: Smooth Adam optimisation of a log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    start_vec: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 2.0,
    beta_end: float = 400.0,
    step0: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Returns the best vector found and its exact score.
    """
    a = start_vec.astype(float).copy()
    n = a.size
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < time_deadline and it < max_iters:
        it += 1

        # anneal smoothing parameter β (log‑linear schedule)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution of a with itself
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights over the convolution entries
        # (stable: subtract maxc before exponentiation)
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smoothed objective
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]  # length n slice

        # Adam update
        m1 = beta1 * m1 + (1.0 - beta1) * grad
        m2 = beta2 * m2 + (1.0 - beta2) * (grad * grad)

        m1_hat = m1 / (1.0 - beta1 ** it)
        m2_hat = m2 / (1.0 - beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + eps)

        # project back onto the simplex
        a_new = project_to_simplex(a_new)

        # occasionally record a better point
        if it % 200 == 0:
            cur_score = compute_score(a_new)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()

        a = a_new

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: Simulated‑annealing polishing (mass‑transfer moves)
# ----------------------------------------------------------------------


def annealing_optimize(
    start_vec: np.ndarray,
    time_deadline: float,
    init_temp: float = 5e-3,
    final_temp: float = 1e-7,
    max_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climb with a linear cooling schedule.
    Mass is transferred from a random source index to a random target index.
    """
    cur = start_vec.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    n = cur.size

    start = time.time()
    total = max(1e-12, time_deadline - start)

    while time.time() < time_deadline:
        elapsed = time.time() - start
        frac_time = elapsed / total
        # linear cooling
        T = init_temp * (1.0 - frac_time) + final_temp * frac_time

        # propose a transfer
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue

        # fraction of the source mass we are willing to move (scaled by temperature)
        move_fraction = max_frac * (T / init_temp) * np.random.rand()
        delta = cur[i] * move_fraction
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum is preserved, non‑negativity guaranteed by construction

        cand_score = compute_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()

        # occasionally try a pure swap (helps escape plateaus)
        if np.random.rand() < 0.02:
            cand2 = cur.copy()
            cand2[i], cand2[j] = cand2[j], cand2[i]
            cand2_score = compute_score(cand2)
            if cand2_score < cur_score or np.random.rand() < np.exp(-(cand2_score - cur_score) / max(T, 1e-12)):
                cur = cand2
                cur_score = cand2_score
                if cand2_score < best_score:
                    best_score = cand2_score
                    best = cand2.copy()

    return best, best_score
