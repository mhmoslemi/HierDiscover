"""
Search for a low‑value step‑function using a block‑parameterised smoothed
Adam optimiser followed by cheap sub‑gradient refinement and a short
simulated‑annealing hill‑climb.

Key ideas
---------
* The continuous arcsine distribution (density ∝(x(1‑x))⁻¹/²) has a flat
  self‑convolution, i.e. its discrete analogue gives a very low
  max‑convolution.  We discretise it on a grid of *blocks* and optimise the
  block masses.
* The optimisation is performed on the logits of the block probabilities
  (soft‑max parametrisation) with an Adam optimiser on the smooth
  max‑approximation  f_β(a) = (1/β)·log Σ exp(β·(a∗a)_k).  The smoothing
  parameter β is increased exponentially from 5 to ≈500, driving the
  objective ever closer to the true max.
* The block‑level optimisation is cheap (≈256 variables) yet yields a full‑
  length vector (n=16384) that respects the simplex constraint.
* The resulting vector is refined with a very cheap projected sub‑gradient
  method on the true max‑objective, then (optionally) symmetrised and finally
  polished by a simulated‑annealing walk on the simplex.
* All stages respect the global time budget (default 1000 s).  The routine
  returns a plain Python list ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a vector on the probability simplex:
        score = 2·n·max(a∗a)   (since sum(a)=1)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Block‑wise Adam optimiser on the smoothed max‑objective
# ----------------------------------------------------------------------


def block_adam_smooth_optimize(start_p: np.ndarray,
                               block_len: int,
                               deadline: float,
                               max_iters: int = 250_000,
                               beta_start: float = 5.0,
                               beta_end: float = 500.0,
                               lr: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector that is constant on blocks of equal size.
    The variables are the block masses p (∑p=1, p≥0).  We optimise the logits
    z = log(p) with Adam on the smooth max:
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    where a is the full‑length vector obtained by repeating each block entry
    block_len times and dividing by block_len (so that sum(a)=1).
    """
    m = start_p.shape[0]                     # number of blocks
    eps = 1e-12
    z = np.log(start_p + eps)                # initialise logits
    m_adam = np.zeros_like(z)
    v_adam = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = None
    best_score = float("inf")

    n = m * block_len                         # total length of the expanded vector

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # ----- schedule for the smoothing parameter β -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- decode logits → block probabilities → full vector -----
        p = softmax(z)                         # shape (m,)
        a = np.repeat(p, block_len) / block_len   # shape (n,)

        # ----- smoothed max (log‑sum‑exp) -----
        conv = convolution_fft(a, a)           # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                # soft‑max weights over the 2n‑1 entries

        # ----- gradient w.r.t. the full vector a -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 2n‑1
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]   # contribution of each a_i

        # ----- aggregate gradient to block level -----
        grad_blocks = grad_a.reshape(m, block_len).mean(axis=1)  # (1/len)·∑_i∈block grad_a_i

        # ----- back‑prop through the soft‑max -----
        dot_pg = np.dot(p, grad_blocks)
        grad_z = p * grad_blocks - dot_pg * p

        # ----- Adam update -----
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad_z
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m_adam / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional exact evaluation -----
        if it % 1000 == 0:
            a_test = np.repeat(softmax(z), block_len) / block_len
            sc = compute_score(a_test)
            if sc < best_score:
                best_score = sc
                best_a = a_test.copy()
                print(f"[{time.time():.1f}] block‑Adam iter {it} new best = {best_score:.6f}")

    # final check
    a_final = np.repeat(softmax(z), block_len) / block_len
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
        print(f"[{time.time():.1f}] block‑Adam final best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Classic projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method for
        2·n·max(a*a)
    with a diminishing stepsize  ∝ 1/√t.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        # compute full convolution
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                     # index of active max

        # sub‑gradient of the max‑entry (only the active terms)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad iter {t} new best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
        print(f"[{time.time():.1f}] sub‑grad final best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb on the simplex
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Very simple stochastic walk: with probability 0.5 swap two entries,
    otherwise transfer a fraction of mass from one index to another.
    Tiny temperature allows the occasional uphill move.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature (tiny)
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            # accept
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # increase exploration a bit when we improve
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Choice of length and block discretisation
    # ------------------------------------------------------------------
    n = 16384                     # larger n gives a lower theoretical bound
    m = 256                       # number of equal‑size blocks
    block_len = n // m
    print(f"[{time.time():.1f}] using length n = {n}, blocks = {m}, block_len = {block_len}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback to a previously found construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Initialise with a discretised arcsine distribution (theoretically flat)
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = 1.0 / np.sqrt(x * (1.0 - x))               # arcsine density on the grid
    block_sums = w.reshape(m, block_len).sum(axis=1)
    start_p = block_sums / block_sums.sum()        # block probabilities (sum to 1)

    # ------------------------------------------------------------------
    # 2. Block‑wise Adam optimiser on the smooth max‑objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] starting block‑Adam optimisation")
        a_opt, sc_opt = block_adam_smooth_optimize(
            start_p,
            block_len,
            deadline - 120.0,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=500.0,
            lr=0.015,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after block‑Adam = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Fine‑grained projected sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=1_500_000,
            step0=0.06,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional symmetrisation (many optimal solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Safeguard: if everything failed, fall back to a tiny uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] generic fallback score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
