"""
Search for a non‑negative step function that minimises
``evaluate_sequence``.  The method consists of three phases:

1. **Parametric seed generation** – a large random pool of analytically
   prescribed shapes (Beta, power‑law, arcsine, mixtures, Dirichlet) on a
   relatively long grid (default 32768).  The fast surrogate
   ``compute_score`` (FFT‑based convolution, assumes Σa = 1) is used to
   keep only the few most promising seeds.

2. **Smooth‑max Adam optimisation** – starting from the best seed,
   Adam minimises a log‑sum‑exp approximation of ``max(conv(a,a))``.
   The temperature β is increased up to a large value (≈2500) so that
   the surrogate tightly follows the true max.  The optimiser runs for
   the majority of the time budget.

3. **Polishing** – a KL‑mirror (exponentiated‑gradient) step on the
   exact max‑convolution objective, followed by a Euclidean projected
   sub‑gradient descent, both respecting the simplex constraint.

The whole routine respects the 1000 s wall‑clock limit and always
returns a valid (non‑empty) sequence.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score when Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, Σ = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid, Σ = 1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape, Σ = 1."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform, Σ = 1."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def random_dirichlet_shape(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Dirichlet draw with concentration α, Σ = 1."""
    vec = rng.gamma(alpha, 1.0, size=n)
    vec = np.maximum(vec, 1e-16)
    return vec / vec.sum()


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-15, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 2500.0,
    ramp_iters: int = 120_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv).
    The optimisation stops at ``deadline``.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # point on simplex
        conv = convolution_fft(a, a)       # length 2n‑1
        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max via log‑sum‑exp
        shifted = conv - mx                # ≤0
        w = np.exp(beta * shifted)         # relative weights
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of the smooth max w.r.t. a
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output
        if t % 5000 == 0:
            print(
                f"    Adam it={t:,d} β={beta:.0f} surrogate={compute_score(a):.6f}"
            )

    # final point on simplex
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Mirror descent (KL) on the exact max‑convolution objective.
    Multiplicative updates keep the iterates on the simplex.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # active set (indices attaining the max within tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # EG step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact Euclidean projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.09,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Helper: build a pool of diverse analytic seeds for a given length
# ----------------------------------------------------------------------


def build_seed_pool(
    n: int, rng: np.random.Generator, best_sofar: List[float], top_k: int = 10
) -> List[np.ndarray]:
    """
    Randomly generate a large pool of analytic shapes, keep the ``top_k``
    according to the fast surrogate score.
    """
    # min‑heap like list (actually sorted list) of (score, array)
    best = []

    def consider(vec: np.ndarray):
        sc = compute_score(vec)
        if len(best) < top_k:
            best.append((sc, vec))
            best.sort(key=lambda x: x[0])
        else:
            if sc < best[-1][0]:
                best[-1] = (sc, vec)
                best.sort(key=lambda x: x[0])

    # deterministic seeds
    consider(arcsine_initial(n))
    consider(np.full(n, 1.0 / n))

    # random beta shapes
    for _ in range(120):
        a = rng.uniform(0.2, 2.0)
        b = rng.uniform(0.2, 2.0)
        eps = rng.uniform(0.0, 2.0)
        consider(generate_beta_shape(n, a, b, eps))

    # random power‑law shapes
    for _ in range(120):
        p = rng.uniform(0.05, 5.0)
        eps = rng.uniform(0.0, 2.0)
        consider(generate_power_shape(n, p, eps))

    # mixtures of arcsine & uniform
    for _ in range(80):
        w = rng.uniform(0.0, 1.0)
        consider(generate_mixture_shape(n, w))

    # Dirichlet draws
    for _ in range(80):
        alpha = rng.uniform(0.2, 2.0)
        consider(random_dirichlet_shape(n, rng, alpha))

    # up‑sample of the current best (if any)
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        consider(upsample_sequence(old, n))

    # return just the vectors, already sorted by increasing score
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 43,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low
    ``evaluate_sequence`` value.  The routine respects a 1000 s wall‑clock
    budget and always returns a valid (non‑empty) sequence.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a tiny safety margin for the final return
    deadline = start + budget_s - 0.5

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to a previously saved construction, if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose a working length (larger lengths give more freedom)
    # ------------------------------------------------------------------
    # We aim for a fairly large grid but still fast enough for many FFTs.
    # 32768 is a good compromise; increase if we have a very generous budget.
    if budget_s > 800:
        n = 65536
    elif budget_s > 400:
        n = 32768
    else:
        n = 16384

    print(f"\n=== Working with length n={n} ===")

    # ------------------------------------------------------------------
    # 2.  Build a diverse seed pool and keep the most promising ones
    # ------------------------------------------------------------------
    seed_pool = build_seed_pool(n, rng, best_seq, top_k=12)
    print(f"Generated {len(seed_pool)} promising seeds (kept by surrogate score)")

    # ------------------------------------------------------------------
    # 3.  Run a long Adam optimisation on the *best* seed.
    #    Remaining time is split: ~80 % for Adam, the rest for polishing.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining < 20.0:
        print("Not enough time for optimisation – returning current best")
        if not best_seq:
            # cheap uniform fallback
            best_seq = [1.0 / n] * n
        return best_seq

    # pick the best seed according to the fast surrogate
    best_seed = seed_pool[0]
    print("Starting Adam optimisation from the best surrogate seed")

    adam_deadline = time.time() + 0.78 * remaining
    # initialise log‑softmax coordinates
    eps = 1e-15
    z0 = np.log(best_seed + eps)

    a_opt = smooth_adam_optimize(
        z0,
        adam_deadline,
        n=n,
        beta0=5.0,
        beta_max=2500.0,
        ramp_iters=150_000,
        lr=0.07,
    )

    # official evaluation (the official function does clamping, sum‑normalisation, etc.)
    try:
        val = evaluate_sequence(a_opt.tolist())
        print(f"   Official score after Adam = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()
            print(f"   *** New global best: {best_val:.6f} ***")
    except Exception as e:
        print(f"   Evaluation after Adam failed: {e}")

    # ------------------------------------------------------------------
    # 4.  Mirror‑descent polishing (exponentiated‑gradient) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        polish_deadline = deadline - 3.0
        print("\nPolishing with exponentiated‑gradient mirror descent …")
        a_polish = exponentiated_gradient_polish(
            np.array(best_seq, dtype=float),
            polish_deadline,
            max_iter=250_000,
            step=0.018,
        )
        try:
            val2 = evaluate_sequence(a_polish.tolist())
            print(f"   Polished (EG) official score = {val2:.6f}")
            if val2 < best_val:
                best_val, best_seq = val2, a_polish.tolist()
                print(f"   *** New best after EG polish: {best_val:.6f} ***")
        except Exception as e:
            print(f"   EG polishing failed: {e}")

    # ------------------------------------------------------------------
    # 5.  Final Euclidean projected sub‑gradient descent (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\nFinal global sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=300_000,
            step0=0.09,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
