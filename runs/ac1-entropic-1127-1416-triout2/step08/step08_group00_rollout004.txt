    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score

    n = a.shape[0]

    T0 = best_score * 0.02
    # schedule: T = T0 * (1 - step/max_steps)
    for step in range(max_steps):
        if time.time() >= time_deadline:
            break
        # propose move
        cand = a.copy()
        # choose move type
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:

            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = np.random.rand() * 0.02 * a[i]   # relative amount

                cand[j] += delta
                if cand[i] < 0:

            accept = np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12))

            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

    Full linear convolution using a real FFT (fast for large arrays).
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score_vec(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    The function works for any non‑negative vector (the sum is not assumed 1).
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * a.shape[0] * conv.max() / (s * s)


def compute_score(a: np.ndarray) -> float:
    """
    Helper for vectors that already sum to 1.
    Returns 2·n·max(conv(a,a)).
    """
    conv = convolution_fft(a, a)
    return 2.0 * a.shape[0] * conv.max()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth max optimisation (log‑sum‑exp) with Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑accelerated gradient descent on a soft‑max surrogate of max(conv(a,a)).
    Returns the best exact score (using compute_score) encountered.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # soft‑max over the convolution
        conv = convolution_fft(a, a)               # length = 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # gradient of the surrogate (see analysis)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 400 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the entry attaining the maximum
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # tiny random jitter every few thousand steps
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Simulated‑annealing hill‑climb polishing
# ----------------------------------------------------------------------


def simulated_annealing_hill_climb(
    start_a: np.ndarray,
    time_deadline: float,
    max_steps: int = 2_000_000,
) -> tuple[list[float], float]:
    """
    Local stochastic moves (swap or small mass transfer) with Metropolis
    acceptance.  Temperature decreases linearly from an initial value.
    Returns a plain list and its exact score.
    """
    a = start_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)
    cur_score = best_score

    # initial temperature – fraction of the current score
    T0 = max(best_score * 0.03, 1e-4)

    for step in range(max_steps):
        if time.time() >= time_deadline:
            break

        cand = a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = np.random.rand() * 0.02 * cand[i]   # up to 2 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        new_score = compute_score(cand)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            # linear cooling schedule
            T = T0 * (1.0 - step / max_steps)
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects the 1000 s wall‑clock
    budget and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. Choose a workable dimension (larger n usually gives more freedom)
    # --------------------------------------------------------------
    if deadline - time.time() > 800:
        n = 4096
    elif deadline - time.time() > 400:
        n = 3072
    elif deadline - time.time() > 200:
        n = 2048
    else:
        n = 1024
    print(f"Using sequence length n = {n}")

    # --------------------------------------------------------------
    # 1. Evaluate a few simple deterministic baselines
    # --------------------------------------------------------------
    def _maybe_update(v):
        nonlocal best_seq, best_score
        scr = compute_score(v)
        if scr < best_score:
            best_score = scr
            best_seq = v.tolist()
            print(f"Baseline improvement: score = {scr:.6f}")

    # uniform
    uniform = np.full(n, 1.0 / n)
    _maybe_update(uniform)

    # triangular (peak in the middle)
    half = n // 2
    tri = np.arange(1, half + 1, dtype=float)
    if n % 2 == 0:
        tri = np.concatenate([tri, tri[::-1]])
    else:
        tri = np.concatenate([tri, np.arange(half, 0, -1, dtype=float)])
    tri = tri / tri.sum()
    _maybe_update(tri)

    # a few Gaussian‑shaped candidates
    def gaussian(sigma):
        idx = np.arange(n, dtype=float)
        centre = (n - 1) / 2.0
        g = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        return g / g.sum()

    for sigma in (n / 8.0, n / 6.0, n / 5.0, n / 4.0):
        _maybe_update(gaussian(sigma))

    # fallback from a previously discovered sequence, if present
    try:
        fallback = np.array(height_sequence_1, dtype=float)
        if fallback.shape[0] != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, fallback.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            fallback = np.interp(x_new, x_old, fallback)
        fallback = np.clip(fallback, 0.0, None)
        fallback = fallback / fallback.sum()
        _maybe_update(fallback)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2. Random‑restart smooth optimisation (soft‑max surrogate)
    # --------------------------------------------------------------
    # Reserve a generous chunk for the later refinement phases
    reserve_for_refine = 130.0
    smooth_deadline = time.time() + max(0.0, deadline - time.time() - reserve_for_refine)
    restart_id = 0

    while time.time() < smooth_deadline:
        remaining = smooth_deadline - time.time()
        # allocate a few seconds to each restart (never less than 4 s)
        time_slice = min(22.0, max(4.0, remaining * 0.45))

        # random start – dense or slightly perturbed around current best
        if best_seq is not None and np.random.rand() < 0.35:
            a0 = np.array(best_seq, dtype=float)
            a0 = a0 + np.random.randn(n) * 0.015
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
        else:
            a0 = np.random.rand(n)
            a0 = a0 / a0.sum()

        a_opt, scr = smooth_gradient_optimize(
            a0,
            time_deadline=time.time() + time_slice,
            max_iters=180_000,
            beta_start=5.0,
            beta_end=2000.0,
            step0=0.9,
        )

        if scr < best_score:
            best_score = scr
            best_seq = a_opt.tolist()
            print(f"Restart {restart_id}: new best smooth score = {scr:.6f}")

        restart_id += 1

    # --------------------------------------------------------------
    # 3. Sub‑gradient refinement (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print("\nSub‑gradient refinement")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 12.0,
            max_iter=640_000,
            step0=0.07,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_a.tolist()
            print(f"Improved by sub‑grad: {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing hill‑climb polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\nSimulated‑annealing hill‑climb")
        polished_seq, polished_score = simulated_annealing_hill_climb(
            np.array(best_seq, dtype=float),
            time_deadline=deadline,
            max_steps=2_500_000,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Fallback to a tiny uniform vector if absolutely nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback to tiny uniform: score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation (the harness will call evaluate_sequence
    # again, but we report the score here for debugging).
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return a plain list – the harness expects this.
    return best_seq
