"""
propose_candidate – a multi‑stage heuristic optimizer for the non‑negative
step‑function problem.

Algorithm outline
-----------------
1.  Simple fallback to the previously known construction (if available).
2.  Dense grid search over two analytic families (symmetric product,
    left‑heavy power law) on a moderately large length (n = 8192).
3.  A handful of random initialisers (monotone, sparse, sinusoidal,
    exponential) – each evaluated on the exact objective.
4.  For each promising start run a long Adam optimisation on the
    smooth log‑sum‑exp surrogate of the max‑convolution.  The smoothing
    parameter β is exponentially annealed from ~5 to ~500.  The
    simplex‑constraint is enforced via a soft‑max parametrisation.
5.  Refine the current best vector with a *multi‑peak* projected sub‑gradient
    method: instead of using the single active index of the max‑convolution,
    all indices whose value is within a small relative tolerance are
    aggregated into a weighted gradient (effectively a bundle of
    sub‑gradients).  Step size = step0/√t.
6.  Keep the top‑K (K=5) vectors discovered so far, form their
    (simplex‑projected) average, and give it a short Adam polish.
7.  Symmetrise the best candidate (a ↔ a[::-1]) – many optimal solutions
    are symmetric.
8.  A final stochastic hill‑climb with a tiny simulated‑annealing rule
    (swap or mass‑transfer moves) finishes the search.

All stages respect the global wall‑clock budget (default 1000 s) and the
function returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution using a real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the probability simplex:
        score = 2·n·max(conv(a, a))
    The vector is assumed to sum to 1 (or close to it); otherwise the
    returned value is scaled by the square of the sum, which is exactly
    what ``evaluate_sequence`` does.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for dense grid search)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, then projected to the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, then projected to the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers (diverse shapes)
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform distribution – about 5 % non‑zeros."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Decreasing vector obtained by sorting a power‑transformed uniform sample."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]          # decreasing
    return project_to_simplex(a)


def random_exponential_start(n: int) -> np.ndarray:
    """a_i ∝ exp(‑λ i) with a random decay rate."""
    lam = np.random.uniform(0.1, 5.0)
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    return project_to_simplex(w)


def random_sinusoid_start(n: int) -> np.ndarray:
    """Positive sinusoidal shape: a_i ∝ sin(2π·freq·i/n + phase) + 1."""
    freq = np.random.uniform(0.5, 3.0)          # a few full cycles
    phase = np.random.uniform(0, 2 * np.pi)
    i = np.arange(n, dtype=np.float64)
    w = np.sin(2 * np.pi * freq * i / n + phase) + 1.0
    w = np.clip(w, 0.0, None)
    return project_to_simplex(w)


# ----------------------------------------------------------------------
# Adam optimisation of the smooth (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 500.0,
                         lr: float = 0.012) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially from ``beta_start``
    to ``beta_end``.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)               # initialise in the soft‑max space

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- current a (soft‑max) ----
        a = softmax(z)

        # ---- smoothed weights ----
        conv = convolution_fft(a, a)           # length 2n‑1
        cmax = conv.max()
        exp_w = np.exp(beta * (conv - cmax))
        w = exp_w / exp_w.sum()                # ∑_k w_k = 1

        # ---- gradient w.r.t. a ----
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # ---- back‑prop through soft‑max ----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasional exact evaluation ----
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                # small progress report
                print(f"[{time.time():.1f}] Adam iter {it} – new best {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Multi‑peak projected sub‑gradient descent (cheaper per iteration)
# ----------------------------------------------------------------------


def subgrad_multi_optimize(start_a: np.ndarray,
                           deadline: float,
                           max_iters: int = 5_000_000,
                           step0: float = 0.08,
                           rel_eps: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method that averages the sub‑gradients of all
    convolution entries whose value lies within ``rel_eps`` of the current
    maximum.  This yields a bundle of active constraints and typically
    reduces the L∞‑peak faster than the single‑index version.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()
        # active set = all indices with value >= max_val·(1‑rel_eps)
        thresh = max_val * (1.0 - rel_eps)
        active = (conv >= thresh).astype(np.float64)
        cnt = active.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = active / cnt                     # uniform weight on active indices

        # gradient of the weighted sum of entries
        grad = 2.0 * convolution_fft(a[::-1], w)[n - 1:n - 1 + n]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * grad)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad multi iter {t} – new best {best_score:.6f}")

    # final evaluation
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb with simulated annealing (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Long random walk on the simplex – tiny Simulated‑Annealing."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature (tiny fraction of the current score)
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)   # allow slightly larger moves
        else:
            step_scale *= 0.97                               # shrink useless moves

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    # ------------------------------------------------------------------
    rand_initialisers = [
        random_monotone_start,
        random_sparse_start,
        random_exponential_start,
        random_sinusoid_start,
    ]
    for init_fn in rand_initialisers:
        for _ in range(20):            # a few draws per family
            if time.time() >= deadline - 30.0:
                break
            vec = init_fn(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best ({init_fn.__name__}) = {best_score:.6f}")
            _push_top_k(top_k, sc, vec)

    # ------------------------------------------------------------------
    # 3. Adam‑based smoothed optimisation from several random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:
        start_vec = random_sparse_start(n)           # a cheap, diverse seed
        # allocate a slice of the remaining budget for this run
        sub_deadline = min(deadline - 5.0, time.time() + 30.0)
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline=sub_deadline,
            max_iters=180_000,
            beta_start=5.0,
            beta_end=500.0,
            lr=0.012,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] Adam‑smooth new best = {best_score:.6f}")
        _push_top_k(top_k, sc_opt, a_opt)

    # ------------------------------------------------------------------
    # 4. Multi‑peak sub‑gradient refinement on the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting multi‑peak sub‑gradient refinement")
        a_sub, sc_sub = subgrad_multi_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 10.0,
            max_iters=4_000_000,
            step0=0.08,
            rel_eps=0.001,
        )
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad new best = {best_score:.6f}")
        _push_top_k(top_k, sc_sub, a_sub)

    # ------------------------------------------------------------------
    # 5. Ensemble averaging of the K best vectors + brief Adam polish
    # ------------------------------------------------------------------
    if len(top_k) >= 2 and time.time() < deadline - 60.0:
        avg_vec = np.mean([v for _, v in top_k], axis=0)
        avg_vec = project_to_simplex(avg_vec)
        sc_avg = compute_score(avg_vec)
        print(f"[{time.time():.1f}] ensemble average score = {sc_avg:.6f}")
        if sc_avg < best_score:
            best_score, best_seq = sc_avg, avg_vec.tolist()
            print(f"[{time.time():.1f}] ensemble improved best = {best_score:.6f}")

        # short polishing run
        sub_deadline = min(deadline - 5.0, time.time() + 20.0)
        a_polish, sc_polish = adam_smooth_optimize(
            avg_vec,
            deadline=sub_deadline,
            max_iters=80_000,
            beta_start=10.0,
            beta_end=400.0,
            lr=0.01,
        )
        if sc_polish < best_score:
            best_score, best_seq = sc_polish, a_polish.tolist()
            print(f"[{time.time():.1f}] after polishing ensemble best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation (many near‑optimal solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score, best_seq = sc_sym, a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final stochastic hill‑climb (tiny simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_500_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: if nothing succeeded, fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
