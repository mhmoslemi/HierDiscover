"""
propose_candidate
=================

A multi‑phase stochastic optimiser for the “max‑self‑convolution” problem.

* Phase 0 – a small fallback from a previously discovered sequence.
* Phase 1 – many short Adam‑gradient runs on a log‑sum‑exp surrogate.
  β (soft‑max temperature) is annealed from a tiny value up to several
  thousands so that the smooth objective approaches the true max.
  Restarts are diversified (dense Dirichlet, sparse, sinusoidal, logistic).
* Phase 2 – a refinement using projected sub‑gradient descent on the exact
  max‑objective with an AdaGrad‑style step‑size.
* Phase 3 – a “pool” of the best vectors is maintained; random convex
  combinations of pool members are tested (mixing can lie inside the convex
  hull of good points and often reduces the peak autocorrelation).
* Phase 4 – a fast stochastic hill‑climb that moves small amounts of mass
  between entries and accepts only improving moves.

All vectors are kept on the probability simplex (non‑negative, sum = 1) –
the evaluation function is invariant under scaling, so this does not lose
generality.  The routine respects the 1000 s wall‑clock budget and returns
the best list of floats it has found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large arrays)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the objective for a vector satisfying sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet (α=1) – dense probability vector."""
    a = np.random.rand(n)
    return a / a.sum()


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on sqrt(n) random positions (sparse start)."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_sinusoidal(n: int) -> np.ndarray:
    """Sinusoidal shape, optionally raised to a random power."""
    p = np.random.choice([0.5, 1.0, 2.0])
    x = np.sin(np.pi * np.arange(n) / (n - 1))
    a = (x ** p).astype(float)
    return a / a.sum()


def random_start_logistic(n: int) -> np.ndarray:
    """Logistic (sigmoid) shape with a random scale."""
    scale = np.random.uniform(0.02, 0.15) * n
    centre = n // 2
    i = np.arange(n)
    a = 1.0 / (1.0 + np.exp(-(i - centre) / scale))
    a = a - a.min()
    a = a / a.sum()
    return a


def random_start_any(n: int) -> np.ndarray:
    """Pick one of the above generator at random."""
    choice = np.random.randint(4)
    if choice == 0:
        return random_start_dense(n)
    if choice == 1:
        return random_start_sparse(n)
    if choice == 2:
        return random_start_sinusoidal(n)
    return random_start_logistic(n)


# ----------------------------------------------------------------------
# 1. Smooth max optimisation (log‑sum‑exp surrogate) with Adam
# ----------------------------------------------------------------------


def smooth_opt(
    start_a: np.ndarray,
    time_deadline: float,
    beta_start: float = 0.5,
    beta_end: float = 5000.0,
    step0: float = 0.8,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate.
    Returns the best exact score encountered.
    """
    a = start_a.copy()
    n = a.shape[0]

    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ---------- beta schedule (log‑linear) ----------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---------- smoothed objective ----------
        conv = convolution_fft(a, a)  # length 2n‑1
        maxc = conv.max()
        # stable soft‑max weights (they are ≤1)
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ---------- gradient of soft‑max ----------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]    # central n entries

        # ---------- Adam update ----------
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 400 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    # final check
    final = compute_score(a)
    if final < best_score:
        best_score = final
        best_a = a.copy()

    return best_a, best_score

# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.07,
    max_iter: int = 600_000,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(conv(a,a)).
    Uses an AdaGrad‑style step‑size (1/√t) and simplex projection.
    """
    a = start_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the maximal entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur = compute_score(a_new)
        if cur < best_score:
            best_score = cur
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    final = compute_score(a)
    if final < best_score:
        best_score = final
        best_a = a.copy()
    return best_a, best_score

def mix_pool(pool, time_deadline):
    """
    Random convex combinations of the best vectors stored in `pool`.
    Returns (best_vec, best_score) after the time runs out.
    """
    if not pool:
        return None, float("inf")

    # keep pool sorted (ascending score)
    pool = sorted(pool, key=lambda x: x[0])
    best_score, best_vec = pool[0]

    while time.time() < time_deadline:
        # pick two distinct members (or best + fresh random)
        if len(pool) >= 2 and np.random.rand() < 0.6:
            (s1, v1), (s2, v2) = np.random.choice(pool, 2, replace=False)
        else:
            v1 = best_vec
            # generate a fresh start to diversify mixing
            n = best_vec.shape[0]
            v2 = random_start_any(n)
            s2 = compute_score(v2)

        # mixing weight skewed towards small steps (helps stay near good region)
        t = np.random.rand() ** 2
        cand = (1 - t) * v1 + t * v2
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand.copy()
            pool.append((cand_score, cand.copy()))
            pool = sorted(pool, key=lambda x: x[0])[:5]
            print(f"[mix] improved to {best_score:.6f}")

    return best_vec, best_score

def hill_climb(seq, deadline, max_steps=2_000_000):
    """
    Tiny mass‑transfer moves accepted only when they improve the exact score.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # initial step fraction
    step_scale = 0.08

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions (purely permutation)
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur = compute_score(cand)

        if cur < best_score:
            best_score = cur
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                       # become more conservative

        max_steps -= 1

    return best_a.tolist(), best_score
