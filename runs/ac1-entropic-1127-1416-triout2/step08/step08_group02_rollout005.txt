"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1. **Initial shapes** – central‑binomial, arcsine and a random Dirichlet
   distribution are generated on a modest size (256).  All are projected onto
   the probability simplex.

2. **Multi‑scale Adam optimisation** – each candidate is linearly
   interpolated to larger lengths (512, 1024, …, 8192).  At every scale an
   Adam optimiser minimises a smooth approximation of the max‑convolution
   objective using a log‑sum‑exp surrogate.  The smoothing parameter β is
   annealed from a very low value (≈0.5) to a high value (≈500) so that the
   early iterations shape the whole profile while the late iterations focus
   on the actual maximum.

3. **Exact sub‑gradient polishing** – after the largest scale an
   (averaged) sub‑gradient descent on the true max‑objective eliminates the
   remaining bias of the surrogate.

4. **Stochastic hill‑climb** – cheap random mass‑transfer moves with a
   simple Metropolis acceptance rule are performed until the time budget
   expires.

All stages respect the global deadline (≈1000 s).  The best sequence found
throughout the run is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric tools
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)).  ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------


def init_central_binomial(n: int) -> np.ndarray:
    """Truncated central‑binomial coefficients (a_i = C(2i,i)/4^i)."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2.0 * i + 1.0) / (2.0 * (i + 1.0))
    return project_to_simplex(a)


def init_arcsine(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density 1/√(x(1‑x))."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return project_to_simplex(a)


def init_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex (Gamma(1,1) → Dirichlet)."""
    x = rng.exponential(scale=1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 100_000,
    beta_start: float = 0.5,
    beta_end: float = 500.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (best_vector, best_score) seen during the run.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal smoothing parameter β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution a * a
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights
        exp_vals = np.exp(beta * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0:
            w = np.full_like(exp_vals, 1.0 / exp_vals.shape[0])
        else:
            w = exp_vals / sum_exp

        # gradient of the smooth objective (cross‑correlation)
        grad_full = convolution_fft(a[::-1], w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = project_to_simplex(a_new)
        a = a_new

        # occasional bookkeeping
        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.07,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = conv.max()

        # indices where the maximum is attained (within a tiny tolerance)
        active = np.where(conv >= maxc - tol)[0]

        # average sub‑gradient over the active set
        grad = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            if np.any(mask):
                grad[mask] += 2.0 * a[j[mask]]
        if active.size > 0:
            grad /= active.size

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        a = a_new
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill climbing (simulated annealing style)
# ----------------------------------------------------------------------


def hill_climb(
    start_seq,
    deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """Tiny random mass‑transfer moves with a simple Metropolis rule."""
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    # temperature schedule – start fairly warm, cool down linearly
    temp_start = best_score * 0.02 + 1e-6
    temp_end = 1e-8
    temp = temp_start

    step_scale = 0.07  # typical fraction of mass moved each step

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        cand = best_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            # become a bit bolder after a success
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # Metropolis acceptance of uphill moves
            prob = np.exp(-(cand_score - best_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                best_a = cand
                best_score = cand_score
                step_scale = max(step_scale * 0.97, 0.01)

        # linear temperature decay
        temp = max(temp_end, temp_start * (1.0 - step / max_steps))

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling utility
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from ``len(v)`` to ``new_len`` followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Multi‑scale optimisation driver
# ----------------------------------------------------------------------


def optimise_multiscale(start_vec: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Run the multi‑scale optimisation pipeline (Adam → sub‑grad) starting from
    ``start_vec`` (already on the simplex).  Returns the best vector and its
    score seen so far.
    """
    # desired sequence lengths (powers of two)
    scales = [256, 512, 1024, 2048, 4096, 8192]

    cur_vec = start_vec.copy()
    best_vec = cur_vec.copy()
    best_score = compute_score(best_vec)

    for n in scales:
        if time.time() >= deadline:
            break

        # upscale to the current size
        cur_vec = upscale_vector(cur_vec, n)

        # iteration budget inversely proportional to sqrt(size)
        base_iters = 50_000
        max_iters = int(base_iters * np.sqrt(256.0 / max(n, 1)))
        max_iters = max(max_iters, 2_000)

        print(f"[Scale {n}] Adam optimisation – max_iters={max_iters}")

        cur_vec, cur_score = adam_optimize(
            cur_vec,
            deadline,
            max_iters=max_iters,
            beta_start=0.5,
            beta_end=500.0,
            step0=0.8,
        )
        if cur_score < best_score:
            best_score, best_vec = cur_score, cur_vec.copy()
            print(f"   > new best after Adam (size {n}): {best_score:.6f}")

    # final polishing on the largest scale reached
    if time.time() < deadline - 30.0:
        print("Sub‑gradient polishing on final scale")
        cur_vec, cur_score = subgradient_optimize(
            best_vec,
            deadline,
            max_iter=80_000,
            step0=0.07,
        )
        if cur_score < best_score:
            best_score, best_vec = cur_score, cur_vec.copy()
            print(f"   > new best after sub‑grad: {best_score:.6f}")

    return best_vec, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness wrapper
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fall‑back from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. generate a few diverse initial vectors on a tiny support
    # ------------------------------------------------------------------
    tiny_n = 256
    rng = np.random.default_rng(seed + 42)

    init_candidates = [
        init_central_binomial(tiny_n),
        init_arcsine(tiny_n),
        init_random_dirichlet(tiny_n, rng),
    ]

    # ------------------------------------------------------------------
    # 2. run the multi‑scale pipeline for each candidate, keep the best
    # ------------------------------------------------------------------
    for idx, cand in enumerate(init_candidates, 1):
        if time.time() >= deadline:
            break
        print(f"\n=== Starting multi‑scale optimisation from candidate {idx} ===")
        opt_vec, opt_score = optimise_multiscale(cand, deadline)

        if opt_score < best_val:
            best_val = opt_score
            best_seq = opt_vec.tolist()
            print(f"*** New overall best after candidate {idx}: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. final stochastic hill‑climb polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Hill‑climb polishing ---")
        polished_seq, polished_score = hill_climb(
            best_seq,
            deadline,
            max_steps=500_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"*** Best improved by hill‑climb: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. safeguard – return a uniform tiny vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
