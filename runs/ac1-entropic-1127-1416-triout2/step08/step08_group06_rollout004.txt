"""
Propose a non‑negative step sequence that (approximately) flattens its
self‑convolution.  The algorithm runs in three phases:

1. **Broad parametric search** – random samples of several analytically
   defined families (beta, power‑law, γ‑shape, arcsine, mixtures) for a set
   of lengths (up to 8192).  The 10 best candidates (by the fast FFT‑based
   surrogate ``compute_score``) are kept.

2. **Gradient polishing** – each top candidate is refined by three
   complementary optimisers that respect the simplex:
      * exponentiated‑gradient (KL‑mirror) descent,
      * Adam on a smooth log‑sum‑exp surrogate of the max,
      * exact projected sub‑gradient descent on the true max‑convolution.

3. **Local random refinement** – a cheap “hill‑climbing’’ routine that
   repeatedly moves a small amount of mass between two coordinates and
   accepts improvements (or occasional uphill moves) while the wall‑clock
   permits.  This often squeezes a little extra performance out of the
   polished solution.

All steps monitor the remaining time and guarantee that a valid
(non‑empty, non‑negative) sequence is returned before the 1000 s budget
expires.
"""

# ----------------------------------------------------------------------
# Imports and low‑level utilities
# ----------------------------------------------------------------------
import time
from typing import List
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose entries sum to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (all normalise to L1‑norm = 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β)‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape – the denominator is raised to p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_gamma_shape(n: int, gamma: float, delta: float = 0.5,
                         eps: float = 0.0) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{-γ}.
    The classic arcsine corresponds to γ≈0.5, δ≈0.5, ε≈0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the exact sub‑gradient method
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step: float = 0.012,
) -> np.ndarray:
    """
    Mirror‑descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 10_000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimisation (optional sharper polishing)
# ----------------------------------------------------------------------
def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 400.0,
    ramp_iters: int = 80_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam on the log‑sum‑exp smooth approximation of max(conv).
    Operates in the log‑domain (soft‑max enforces the simplex).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ≥ 0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient wrt a : 2·(a ⋆ w_rev)
        rev = convolution_fft(a, w_norm[::-1])
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 10_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    return softmax(z)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # mask of indices attaining the maximum (numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # reverse mask for convolution with a
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5_000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score(a):.6f}")

    return a

# Light‑weight random local refinement (hill‑climbing)
# ----------------------------------------------------------------------
def random_local_refinement(
    a0: np.ndarray,
    deadline: float,
    max_moves: int = 30_000,
    base_step: float = 0.02,
) -> np.ndarray:
    """
    Simple stochastic hill‑climbing: repeatedly move a small amount of mass
    from one coordinate to another and keep an improvement.  The move size
    decays with the remaining time to become more fine‑grained later.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = compute_score(a)
    rng = np.random.default_rng()
    moves = 0

    while time.time() < deadline and moves < max_moves:
        moves += 1

        # pick source index with non‑zero mass
        i = rng.integers(n)
        if a[i] <= 1e-15:
            continue
        # pick destination (different) index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # propose a transfer amount; shrink step as we approach deadline
        remaining = deadline - time.time()
        frac = remaining / (deadline - (deadline - 1000.0))  # linear decay
        step = base_step * frac
        delta = step * a[i] * rng.random()  # ≤ step·a[i]

        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # a_new still sums to 1, stays non‑negative by construction

        score_new = compute_score(a_new)
        if score_new < best_score:
            a = a_new
            best_score = score_new

    print(f"    Random refinement performed {moves} moves, final score={best_score:.6f}")
    return a

                               time.time() + (search_deadline - time.time()) * 0.4)

        iter_cnt = 0
        while time.time() < per_len_deadline:
            iter_cnt += 1
            # ---- choose shape type ----
            shape_type = rng.choice(["beta", "power", "gamma", "mixture", "arcsine", "uniform"])

            if shape_type == "beta":
                alpha = 10 ** rng.uniform(-1.0, 0.7)   # ≈[0.1,5]
                beta = 10 ** rng.uniform(-1.0, 0.7)
                eps = rng.uniform(0.0, max(0.5, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)

            elif shape_type == "power":
                p = rng.uniform(0.2, 0.9)            # around the arcsine exponent 0.5
                eps = rng.uniform(0.0, 2.0)
                cand = generate_power_shape(n, p, eps)

            elif shape_type == "gamma":
                gamma = rng.uniform(0.25, 0.75)      # explore around 0.5
                delta = rng.uniform(0.1, 1.2)
                eps = rng.uniform(0.0, 0.5)
                cand = generate_gamma_shape(n, gamma, delta, eps)

            elif shape_type == "mixture":
                w = rng.uniform(0.0, 1.0)
                cand = generate_mixture_shape(n, w)

            elif shape_type == "arcsine":
                cand = arcsine_initial(n)

            else:   # uniform
                cand = np.full(n, 1.0 / n)

            # Guard against pathological zeros and enforce normalisation
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()

            try_candidate(cand)

        print(f"  Completed {iter_cnt:,d} samples for n={n}, best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Polishing of the best candidates (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if not top_candidates:
        print("No candidates found during random search – falling back to uniform.")
        best_seq = [1.0 / 1024] * 1024
        best_score = compute_score(np.array(best_seq))

    # Sort top candidates by their fast score (ascending)
    top_candidates.sort(key=lambda x: x[0])
    # We'll polish at most the first three (to stay within time)
    polished_up_to = min(3, len(top_candidates))

    for idx in range(polished_up_to):
        cand_score, cand_vec = top_candidates[idx]
        print(f"\n--- Polishing candidate #{idx+1}/{polished_up_to}, init score={cand_score:.6f} ---")
        a_arr = np.array(cand_vec, dtype=float)

        # 2a. Exponentiated‑gradient polishing (longer run)
        a_arr = exponentiated_gradient_polish(
            a_arr,
            deadline=deadline,
            max_iter=int(min(200_000, remaining * 1500)),
            step=0.011,
        )
        val = compute_score(a_arr)
        print(f"    After EG polish score={val:.6f}")
        try_candidate(a_arr)

        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # 2b. Adam on smooth surrogate
        if time.time() < deadline - 5.0:
            eps = 1e-12
            z0 = np.log(np.clip(a_arr, eps, None))
            a_arr = smooth_adam_optimize(
                z0,
                deadline=deadline,
                n=a_arr.size,
                beta0=5.0,
                beta_max=400.0,
                ramp_iters=80_000,
                lr=0.07,
            )
            val = compute_score(a_arr)
            print(f"    After Adam polish score={val:.6f}")
            try_candidate(a_arr)

        remaining = deadline - time.time()
        if remaining < 5.0:
            break

    # ------------------------------------------------------------------
    # 3.  Global polishing of the current best (if any time left)
    # ------------------------------------------------------------------
    if best_seq:
        remaining = deadline - time.time()
        if remaining > 15.0:
            print("\n=== Global polishing phase (final sub‑gradient) ===")
            a_arr = np.array(best_seq, dtype=float)

            a_arr = subgradient_descent_exact(
                a_arr,
                deadline=deadline,
                max_iter=int(min(600_000, remaining * 500)),
                step0=0.07,
            )
            val = compute_score(a_arr)
            print(f"    After final sub‑gradient polish score={val:.6f}")
            try_candidate(a_arr)

        remaining = deadline - time.time()
        if remaining > 10.0:
            print("\n=== Random local refinement (hill‑climbing) ===")
            a_arr = np.array(best_seq, dtype=float)
            a_arr = random_local_refinement(
                a_arr,
                deadline=deadline,
                max_moves=30_000,
                base_step=0.02,
            )
            val = compute_score(a_arr)
            print(f"    After random refinement score={val:.6f}")
            try_candidate(a_arr)

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee that we return something valid
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_score = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # 5.  Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
