import scipy.sparse as sp

def lp_step(a, M):
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        # Determine i range
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            j = k - i
            # coefficient a_i for variable b_j
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # b vector: right-hand side M for each constraint
    b_ub = np.full(2*n-1, M)
    # variable bounds: (0, None)
    bounds = [(0, None) for _ in range(n)]
    # objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:

"""Hybrid LP‑gradient optimizer for the non‑negative step‑function problem.

The algorithm
---------------
1. Initialise a probability vector `a` (fallback to the previously discovered
   `height_sequence_1` or a uniform vector) and normalise it so that
   Σa = 1.
2. Loop while time permits:
   * Compute M = max(a ⊛ a) (aperiodic self‑convolution).
   * Solve the linear programme
         maximise   Σ b_i
         subject to   a ⊛ b ≤ M   (2n‑1 linear constraints)
                     b_i ≥ 0
         (an upper bound 1000 is added to keep the problem bounded)
     The LP finds the “most spread’’ vector `b` that does not increase the
     current maximal convolution value.
   * Mix the current vector with the LP solution:
         a ← (1‑t)·a + t·b   (t starts at 0.25 and decays).
     Project the mixture back onto the probability simplex.
   * Keep the best vector seen (lowest value of 2·n·max(a⊛a)).
3. After the LP phase, perform a cheap projected sub‑gradient descent to
   polish the solution.
4. Symmetrise the final vector (a ← (a + a[::-1]) / 2) and project onto
   the simplex again.
5. Return the resulting list of floats; the harness will call
   `evaluate_sequence` on it.

All steps respect the global wall‑clock limit (default 1000 s)."""

import time
import numpy as np

# optional imports – if SciPy is unavailable we fall back to the earlier
# multi‑stage optimiser.
try:
    import scipy.sparse as sp
    import scipy.optimize as opt
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the competition:
        2·n·max(a*a) / (sum a)² .
    The vector is assumed non‑negative.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_conv = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 1e-12:                # defensive – should never happen
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient refinement (very cheap per iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent for the true max‑objective
          2·n·max(a*a)
    using a diminishing stepsize ∝ 1/√t.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index of the active max

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin – the harness adds its own checks
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------

    #    Prefer the previously discovered construction if it exists.

        n = len(init)

        a = a / a.sum()
        print(f"[{time.time():.1f}] using provided height_sequence_1, n={n}")
    except Exception:
        # fall back to a modest uniform vector
        n = 2048
        a = np.full(n, 1.0 / n, dtype=np.float64)
        print(f"[{time.time():.1f}] fallback initialisation, n={n}")

    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"[{time.time():.1f}] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Linear‑program mixing phase (only if SciPy is available)
    # --------------------------------------------------------------
    if _SCIPY_AVAILABLE:
        t_mix = 0.25                     # initial mixing weight
        iteration = 0
        while time.time() < deadline - 10.0:
            iteration += 1
            # --- current maximal convolution ---
            conv_aa = convolution_fft(a, a)
            M = float(conv_aa.max())

            # --- build A (Toeplitz‑like) so that (A @ b)[k] = (a ⊛ b)[k] ---
            # each row k contains the coefficients a_i for indices i where
            # 0 ≤ i < n and 0 ≤ k-i < n
            rows = []
            cols = []
            data = []
            n2 = 2 * n - 1
            for k in range(n2):
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                for i in range(i_start, i_end + 1):
                    j = k - i
                    rows.append(k)
                    cols.append(j)
                    data.append(a[i])
            A = sp.csr_matrix((data, (rows, cols)), shape=(n2, n))

            b_ub = np.full(n2, M, dtype=np.float64)

            # upper bound for each variable – prevents unboundedness when a_i = 0
            ub = 1000.0
            bounds = [(0.0, ub) for _ in range(n)]

            # maximise sum(b)  <=> minimise -sum(b)
            c = -np.ones(n, dtype=np.float64)

            # solve the LP
            lp_res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                 method='highs', options={'presolve': True})
            if not lp_res.success:
                print(f"[{time.time():.1f}] LP failed (iter {iteration}), exiting LP loop")
                break

            b = lp_res.x
            # Normalise to a probability vector (the objective cares only about shape)
            b = np.maximum(b, 0.0)
            sumb = b.sum()
            if sumb == 0.0:
                print(f"[{time.time():.1f}] LP returned zero vector, aborting LP phase")
                break
            b = b / sumb

            # --- mixing and projection ---
            a_new = (1.0 - t_mix) * a + t_mix * b
            a_new = project_to_simplex(a_new)

            # --- evaluate ---
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"[{time.time():.1f}] LP iter {iteration}: new best = {best_score:.6f}")

            a = a_new
            # gradually shrink the mixing factor – later steps become more conservative
            t_mix = max(0.01, t_mix * 0.95)

            # early exit if we are very close to the theoretical lower bound
            if best_score <= 1.0 + 1.0 / (2 * n - 1) + 1e-5:
                print(f"[{time.time():.1f}] reached near‑theoretical bound, stopping LP")
                break
    else:
        print(f"[{time.time():.1f}] SciPy not available – skipping LP phase")

    # --------------------------------------------------------------
    # 2. Short projected sub‑gradient polishing (very cheap)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        # allocate a fraction of the remaining budget to sub‑gradient steps
        max_iters = int(1_500_000 * remaining / (budget_s - 3))
        a_ref, sc_ref = subgrad_optimize(best_a,
                                          deadline=deadline - 5.0,
                                          max_iters=max_iters,
                                          step0=0.06)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()
            print(f"[{time.time():.1f}] after sub‑gradient polishing: best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Symmetrisation (many good solutions turn out symmetric)
    # --------------------------------------------------------------
    a_sym = (best_a + best_a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        best_a = a_sym.copy()
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return best_a.tolist()
