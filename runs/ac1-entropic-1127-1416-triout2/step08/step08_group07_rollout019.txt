"""
Hybrid search for a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The method combines:
  * a rich pool of analytic seeds (arcsine, beta, power‑law,
    exponential, linear, Dirichlet),
  * a few alternating linear‑programming refinements (exact LP on
    max‑convolution constraints),
  * smooth‑max Adam optimisation with β‑continuation,
  * exact projected sub‑gradient polishing,
  * a final random local‑search neighbourhood walk.

All work is bounded by the supplied time budget (default 1000 s).
The returned list is always a valid non‑negative sequence.

import time
import math
from typing import List, Tuple

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate (exact formula without clipping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute   2·n·max(conv(a,a)) / (sum(a)**2)   (no clipping).
    `a` must be non‑negative.
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic generators for diverse starting shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 0.0) -> np.ndarray:
    """Beta(α,β) shape on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Shape proportional to 1/((i+½)*(n‑½‑i)+eps)^p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_exponential_shape(n: int, decay: float) -> np.ndarray:
    """Monotone exponential a_i ∝ exp(-decay·i)."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-decay * i)
    return vals / vals.sum()


def generate_linear_shape(n: int) -> np.ndarray:
    """Strictly decreasing linear shape (a_i ∝ n‑i)."""
    i = np.arange(n, dtype=float)
    vals = n - i
    return vals / vals.sum()


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with random concentration)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)  # between 0.01 and ~5
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx_i = 1, x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent for  f(a)=2·n·max(conv(a,a))/sum(a)^2."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 15_000 == 0:
            print(f"    Sub‑grad it={it:,d} surrogate≈{fast_score(a):.6f}")

    return a

# Smooth‑max (log‑sum‑exp) with Adam optimisation (β‑continuation)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 2000.0,
    ramp_iters: int = 150_000,
    lr: float = 0.03,
) -> np.ndarray:
    """
    Minimises the smooth approximation L_β(a)=log‑sum‑exp(β·conv(a,a)),
    enforcing a∈simplex via the soft‑max transform.  β is increased
    linearly from ``beta0`` to ``beta_max`` (continuation).
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of log‑sum‑exp w.r.t. a:
        # d/da Σ w_i·(a★e_i) = 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 8_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate≈{fast_score(a):.6f}")

    return _softmax(z)

# Random local search (pairwise mass transfers)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.025,
) -> np.ndarray:
    """
    Inexpensive pairwise mass‑transfer moves while time permits.
    Each move picks i≠j and moves a random fraction (≤max_transfer) of the
    mass from i to j; the move is kept only if it strictly improves the
    surrogate ``fast_score``.
    """
    a = a0.copy()
    best_score = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * rng.random() * max_transfer
        if eps <= 0.0:
            continue
        a[i] -= eps
        a[j] += eps
        a /= a.sum()                     # re‑normalise
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            a[j] -= eps
            a[i] += eps
            a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (exact max‑convolution constraints)
# ----------------------------------------------------------------------


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that  A @ g = conv(f,g) .
    Row k contains the coefficients f_{k‑j} for all admissible j.
    """
    n = f.shape[0]
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=float)
    # each column j contributes the whole vector f shifted by +j
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refine(
    f: np.ndarray,
    iters: int,
    mixing: float,
    deadline: float,
) -> np.ndarray:
    """
    Perform ``iters`` alternating linear‑programming steps.
    At each step the LP

        max Σ g_j   s.t.   (f⋆g)_k ≤ M   ∀k ,   g ≥ 0

    is solved (M = max(f⋆f)).  The solution g is rescaled to the same
    ℓ₁‑norm as f and blended:   f ← (1‑mixing)·f + mixing·g .
    Returns the refined vector.
    """
    n = f.size
    target_sum = f.sum()
    for it in range(iters):
        if time.time() >= deadline:
            break
        M = float(np.max(convolution_fft(f, f)))
        A = _conv_matrix(f)
        b = np.full(A.shape[0], M, dtype=float)
        c = -np.ones(n)                    # maximise Σg_j  → minimise -Σg_j
        bounds = [(0, None)] * n
        # solve with HiGHS (fast for dense medium‑size problems)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            print("    LP failed – terminating refinement")
            break
        g = res.x
        if g.sum() == 0.0:
            break
        # rescale to original sum and blend
        g = g * (target_sum / g.sum())
        f = (1.0 - mixing) * f + mixing * g
    return f

            step0=0.05,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_official:
                best_official, best_seq = final_val, a_arr.tolist()
                best_surrogate = fast_score(a_arr)
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final evaluation error: {e}")

    # ------------------------------------------------------------------
    # 4. Random local search around the polished solution (if time)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:
        print("\nRandom local search around current best …")
        a_start = np.array(best_seq, dtype=float)
        a_refined = random_local_search(
            a_start,
            deadline,
            rng,
            max_transfer=0.022,
        )
        try:
            val_loc = evaluate_sequence(a_refined.tolist())
            print(f"   Local‑search official score = {val_loc:.6f}")
            if val_loc < best_official:
                best_official, best_seq = val_loc, a_refined.tolist()
                best_surrogate = fast_score(a_refined)
                print("*** Best improved by local search ***")
        except Exception as e:
            print(f"   Local‑search evaluation error: {e}")

    # ------------------------------------------------------------------
    # 5. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
