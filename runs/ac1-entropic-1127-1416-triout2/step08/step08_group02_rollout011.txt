"""
Non‑negative step‑function minimiser for `evaluate_sequence`.

Approach
--------
1. **Initial shape** – use the classic central‑binomial coefficients
   (a_i = C(2i,i)/4^i) normalised to the probability simplex.

2. **LP minimax refinement** – for the current vector f solve the linear
   program

        minimise   t
        s.t.       (f * b) ≤ t   (all shift positions)
                   Σ b = 1 , b ≥ 0

   The optimizer returns a new probability vector b with a strictly
   lower worst‑case convolution value.  The procedure is iterated (≈ 10
   times) on a modest size (n≈1024).

3. **Upscaling** – linearly interpolate the improved small vector to a
   larger length (n≈4096) and project back onto the simplex.

4. **Smooth log‑sum‑exp Adam optimisation** – minimise a smooth surrogate
   of `max(conv(a,a))` using Adam momentum; the smoothing parameter β is
   annealed from 5 to 250.  The loop runs until the remaining wall‑clock
   budget expires.

5. **Exact sub‑gradient polishing** – projected sub‑gradient descent on
   the true max‑convolution objective (the same objective used by
   `evaluate_sequence`).

6. **Final stochastic hill‑climb** – inexpensive mass‑transfer moves
   (swap or fractional transfer) with simplex projection finish the
   search.

The best sequence found at any stage is kept and finally returned.
All numbers are non‑negative and the vector is already on the simplex,
so `evaluate_sequence` can be called directly on the result.
"""

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast, low‑memory)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_obj(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` when Σ a = 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        # Uniform vector is the projection of the zero‑vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution matrix (Toeplitz) construction
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Return the (2n‑1)×n Toeplitz matrix A with A @ b = conv(f, b)
    for a given vector f (length n).
    """
    n = f.shape[0]
    # Each column i contains f shifted down by i rows.
    diags = [np.full(n, fi) for fi in f]
    offsets = list(range(n))
    A = sp.diags(diagonals=diags, offsets=offsets,
                 shape=(2 * n - 1, n), format='csr')
    return A


# ----------------------------------------------------------------------
# LP: minimise the worst‑case convolution value for a fixed f
# ----------------------------------------------------------------------


def solve_minimax_lp(f: np.ndarray,
                     warm_start: np.ndarray | None = None) -> tuple[np.ndarray, float] | None:
    """
    Solve the LP:
        minimise   t
        s.t.       conv(f, b) ≤ t   (elementwise)
                   Σ b = 1 , b ≥ 0

    Returns a tuple (b_opt, t_opt) or None on failure.
    """
    n = f.shape[0]
    A = build_convolution_matrix(f)

    b = cp.Variable(n)
    t = cp.Variable()

    constraints = [A @ b <= t,
                   cp.sum(b) == 1,
                   b >= 0,
                   t >= 0]

    prob = cp.Problem(cp.Minimize(t), constraints)
    if warm_start is not None:
        b.value = warm_start

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=3000,
                   abstol=1e-7, reltol=1e-6, feastol=1e-7)
    except Exception:
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None

    b_val = np.maximum(b.value, 0.0)
    # Projection onto the simplex removes tiny numerical drift.
    b_val = project_to_simplex(b_val)
    t_val = float(t.value)
    return b_val, t_val


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation with Adam
# ----------------------------------------------------------------------


def smooth_optimize(start_a: np.ndarray, deadline: float,
                    beta_start: float = 5.0,
                    beta_end: float = 250.0,
                    step0: float = 1.0,
                    adam_beta1: float = 0.9,
                    adam_beta2: float = 0.999,
                    adam_eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam optimiser for a soft‑max surrogate of max(conv(a,a))."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first‑moment estimate
    v = np.zeros_like(a)   # second‑moment estimate

    best_a = a.copy()
    best_score = compute_obj(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # anneal smoothing parameter β
        t_frac = it / (it + 1e5)   # slow annealing, never reaches exactly 1
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective (cross‑correlation)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional progress check
        if it % 500 == 0:
            cur_score = compute_obj(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_obj(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray, deadline: float,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_obj(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index attaining the max

        # Derivative of conv[k] w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_score = compute_obj(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new

        # small jitter every few hundred iterations to avoid plateaus
        if it % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_obj(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (mass transfers)
# ----------------------------------------------------------------------


def hill_climb(start_seq: list[float], deadline: float,
               max_steps: int = 300_000) -> tuple[list[float], float]:
    """
    Tiny stochastic moves (swap or transfer a fraction of mass) with simplex
    projection.  Returns the best sequence encountered.
    """
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_obj(best_a)

    # Initial fraction of total mass to move in each step
    step_frac = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_frac * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_obj(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # gradually enlarge step
        else:
            step_frac *= 0.97                        # shrink when not improving

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling utility (linear interpolation)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Interpolate v to length new_len and project onto simplex."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # Fallback: use any previously discovered construction if available
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback (pre‑known) score = {fallback_val:.6f}")
        if fallback_val < best_score:
            best_score = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Phase 0 – initialise with the central‑binomial shape (n = 1024)
    # -----------------------------------------------------------------
    n_small = 1024
    a_small = np.empty(n_small, dtype=float)
    a_small[0] = 1.0
    for i in range(1, n_small):
        a_small[i] = a_small[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a_small = project_to_simplex(a_small)   # Σ = 1
    cur_seq = a_small
    cur_score = compute_obj(cur_seq)
    best_seq = cur_seq.tolist()
    best_score = cur_score
    print(f"Initial (central‑binomial) score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 1 – LP minimax refinement (small size)
    # -----------------------------------------------------------------
    lp_time = min(250.0, (deadline - time.time()) * 0.30)   # at most ~4 min
    lp_deadline = time.time() + lp_time
    warm_b = None
    lp_iter = 0
    while time.time() < lp_deadline and lp_iter < 20:
        lp_iter += 1
        result = solve_minimax_lp(cur_seq, warm_start=warm_b)
        if result is None:
            print("LP solve failed – exiting LP loop.")
            break
        b_opt, t_opt = result
        warm_b = b_opt               # reuse for next iteration

        new_score = 2.0 * n_small * t_opt
        if new_score < best_score - 1e-8:
            best_score = new_score
            best_seq = b_opt.tolist()
            cur_seq = b_opt
            cur_score = new_score
            print(f"[LP {lp_iter}] improved score = {best_score:.6f}")
        else:
            # No improvement – stop early
            print(f"[LP {lp_iter}] no improvement (score {new_score:.6f}); stopping.")
            break

    # -----------------------------------------------------------------
    # Phase 2 – upscale to a larger resolution for fine optimisation
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        n_big = 8192
    else:
        n_big = 4096

    cur_seq = upscale_vector(np.array(best_seq, dtype=float), n_big)
    cur_score = compute_obj(cur_seq)
    if cur_score < best_score:
        best_score = cur_score
        best_seq = cur_seq.tolist()
        print(f"After upscaling to {n_big} points, score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 3 – smooth Adam optimisation (log‑sum‑exp surrogate)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    if time.time() < smooth_deadline:
        print("\n--- Smooth Adam optimisation ---")
        a_opt, opt_score = smooth_optimize(cur_seq, smooth_deadline,
                                           beta_start=5.0,
                                           beta_end=250.0,
                                           step0=1.2)
        if opt_score < best_score:
            best_score = opt_score
            best_seq = a_opt.tolist()
            print(f"Smooth optimisation improved score to {best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 4 – exact sub‑gradient descent
    # -----------------------------------------------------------------
    subgrad_deadline = deadline - 120.0
    if time.time() < subgrad_deadline:
        print("\n--- Sub‑gradient polishing ---")
        a_sub, sub_score = subgrad_optimize(np.array(best_seq, dtype=float),
                                            subgrad_deadline,
                                            step0=0.07)
        if sub_score < best_score:
            best_score = sub_score
            best_seq = a_sub.tolist()
            print(f"Sub‑gradient step improved score to {best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 5 – stochastic hill‑climb (final polishing)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if time.time() < hill_deadline:
        print("\n--- Hill‑climb polishing ---")
        polished_seq, polished_score = hill_climb(best_seq,
                                                  hill_deadline,
                                                  max_steps=500_000)
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb further improved score to {best_score:.6f}")

    # -----------------------------------------------------------------
    # Safety fallback (uniform tiny vector) – should never be needed.
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    return best_seq
