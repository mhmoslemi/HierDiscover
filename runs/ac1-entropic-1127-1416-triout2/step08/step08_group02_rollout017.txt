def build_convolution_matrix(f):
   n = len(f)
   row_indices = []
   col_indices = []
   data = []
   for j in range(n):
       # column j has f_i placed at rows i+j (i from 0 to n-1)
       rows = np.arange(j, j+n)
       row_indices.extend(rows)
       col_indices.extend([j]*n)
       data.extend(f)
   A = sp.csr_matrix((data, (row_indices, col_indices)), shape=(2*n-1, n))
   return A

"""
Non‑negative step‑function optimiser for `evaluate_sequence`.

Strategy
--------
1. **Initial shape** – start from the classic truncated central‑binomial
   coefficients and, if better, a few discrete Gaussian guesses (all
   projected onto the probability simplex).

2. **Robust LP‑mixing** – repeatedly solve the linear programme

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   (k = 0…2n‑2),   b ≥ 0

   using SciPy’s HiGHS solver (far more reliable than ECOS).  The
   resulting partner `b` is mixed into `f` with a small step size; a
   back‑tracking line‑search guarantees a monotone decrease of the
   objective.  Do as many iterations as the remaining time permits.

3. **Upscaling** – linearly interpolate the improved short vector to a
   larger length (4096 or 8192) and re‑project onto the simplex.

4. **Smooth Adam optimisation** – minimise a log‑sum‑exp surrogate of
   `max(conv(a,a))` with Adam momentum.  The smoothing parameter β is
   annealed from 5 to 400, and the Adam step size decays with 1/√t.

5. **Projected sub‑gradient descent** – final polishing on the exact
   `max` objective (the non‑smooth version).  Step size ∝ 1/√t.

6. **Stochastic hill‑climb** – cheap local moves (swap or fractional mass
   transfer) with simplex projection, accepting only improvements.

All stages respect the global time‑budget (≈1000 s).  The best sequence
found (a plain list of non‑negative floats summing to 1) is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for all lengths)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already projected onto the unit simplex.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        # uniform vector if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# LP utilities (stage 2)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Toeplitz matrix A (size (2n‑1)×n) such that A @ b = conv(f, b).
    Each column j contains f shifted down by j rows.
    """
    n = f.shape[0]
    # create n copies of the same diagonal (all equal to f)
    diags = [f] * n
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = sp.diags(diagonals=diags, offsets=offsets,
                 shape=(2 * n - 1, n), format='csr')
    return A


def solve_lp_max_sum(A: sp.csr_matrix, M: float,
                     warm_start: np.ndarray | None = None) -> np.ndarray | None:
    """
    Solve the LP
        maximise   Σ b_j
        subject to A @ b ≤ M,   b ≥ 0 .
    Uses SciPy's HiGHS interior‑point solver (fast and robust).
    Returns the optimal b (numpy array) or None on failure.
    """
    n = A.shape[1]
    c = -np.ones(n)                       # minimise -Σ b_j  ⇔ maximise Σ b_j
    b_ub = np.full(A.shape[0], M)
    bounds = [(0, None)] * n               # b_j ≥ 0

    # HiGHS accepts a sparse matrix directly
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None

    if not res.success:
        return None

    bv = res.x
    if bv is None:
        return None
    return np.maximum(bv, 0.0)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam momentum
# ----------------------------------------------------------------------


def smooth_optimize(start_a: np.ndarray,
                    time_deadline: float,
                    max_iters: int = 500_000,
                    step0: float = 0.8,
                    beta_start: float = 5.0,
                    beta_end: float = 400.0,
                    adam_beta1: float = 0.9,
                    adam_beta2: float = 0.999,
                    adam_eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed version of max‑convolution.
    The surrogate is  (1/β)·log Σ exp(β·conv_ij) .
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)     # first‑moment estimate
    v = np.zeros_like(a)     # second‑moment estimate

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothing parameter β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # current full convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights (log‑sum‑exp trick for numerical stability)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the surrogate:
        #   ∇_a  (1/β)·log Σ exp(β·conv) = Σ w_k ∂ conv_k / ∂ a
        #   ∂ conv_k / ∂ a = 2·(a reversed) * w   (cross‑correlation)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a = a - update
        a = project_to_simplex(a)

        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score, best_a = cur_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimiser on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 400_000,
                     step0: float = 0.05) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * grad
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()

        # occasional tiny jitter to shake off plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score, best_a = cur_score, a.copy()
    return best_a, best_score

# Final stochastic hill‑climb

def hill_climb(seq, time_deadline, max_steps: int = 500_000) -> tuple[list[float], float]:
    """
    Very cheap stochastic moves (swap or fractional transfer) with simplex projection.
    Accepts only improving moves.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    steps_done = 0
    while steps_done < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        steps_done += 1

    return best_a.tolist(), best_score

# Upscaling utility (small → big)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)

        if compute_score(gauss) < compute_score(a_small):
            a_small = gauss
            print(f"Gaussian init (σ={sigma:.1f}) improved score to {compute_score(a_small):.6f}")

    best_small = a_small.copy()
    best_small_score = compute_score(best_small)

    # -----------------------------------------------------------------
    # 2. LP‑based mixing (monotone descent)
    # -----------------------------------------------------------------
    lp_deadline = deadline - 300.0      # reserve ~5 min for later phases
    warm_b = None
    lp_iter = 0
    while time.time() < lp_deadline:
        lp_iter += 1
        M = float(convolution_fft(best_small, best_small).max())
        A = build_convolution_matrix(best_small)
        b_opt = solve_lp_max_sum(A, M, warm_start=warm_b)
        if b_opt is None:
            print("LP solve failed – exiting LP loop.")
            break
        warm_b = b_opt                                # warm‑start next LP
        b_opt = project_to_simplex(b_opt)

        # adaptive mixing – start with t=0.2 and shrink if no improvement
        t = 0.2
        improved = False
        for _ in range(5):
            cand = (1.0 - t) * best_small + t * b_opt
            cand = project_to_simplex(cand)
            cand_score = compute_score(cand)
            if cand_score < best_small_score:
                best_small = cand
                best_small_score = cand_score
                improved = True
                print(f"[LP iter {lp_iter}] t={t:.3f} score={best_small_score:.6f}")
                break
            t *= 0.5
        if not improved:
            # No progress – stop LP phase early
            print(f"[LP iter {lp_iter}] no improving mix found, terminating LP phase.")
            break

    print(f"LP phase completed after {lp_iter} iterations, best small score = {best_small_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Upscale to a larger length for finer optimisation
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    n_big = 8192 if remaining > 600 else 4096
    a_big = upscale_vector(best_small, n_big)

    # -----------------------------------------------------------------
    # 4. Smooth Adam optimisation (log‑sum‑exp surrogate)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    a_smooth, smooth_score = smooth_optimize(
        a_big,
        time_deadline=smooth_deadline,
        max_iters=600_000,
        step0=0.8,
        beta_start=5.0,
        beta_end=400.0,
    )
    if smooth_score < best_val:
        best_val = smooth_score
        best_seq = a_smooth.tolist()
        print(f"After smooth optimisation: score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Projected sub‑gradient polishing
    # -----------------------------------------------------------------
    sub_deadline = deadline - 80.0
    if time.time() < sub_deadline:
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=sub_deadline,
            max_iter=400_000,
            step0=0.05,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑gradient refinement improved score to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 6. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if time.time() < hill_deadline:
        polished_seq, polished_score = hill_climb(
            best_seq,
            time_deadline=hill_deadline,
            max_steps=500_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb further improved score to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to a simple uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
