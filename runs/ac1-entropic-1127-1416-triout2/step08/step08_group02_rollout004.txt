def propose_candidate(seed=...):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.

    best_seq = None; best_score = inf

    # choose length based on remaining time
    if deadline - time.time() > 200:
        n = 8192
    else:
        n = 4096

    # generate initial vectors; store in list
    candidates = []
    # arcsine
    candidates.append(('arcsine', init_arcsine(n)))
    # central binomial (maybe truncated)
    candidates.append(('central_bin', init_central_binomial(n)))
    # random Dirichlet
    for i in range(3):
        candidates.append(('random', np.random.dirichlet(np.ones(n))))
    # fallback (if exists) projected to length n via interpolation
    try:
        fb = np.array(height_sequence_1, dtype=float)
        fb = upscale_vector(fb, n)
        fb = project_to_simplex(fb)
        candidates.append(('fallback', fb))
    except Exception:
        pass

    # for each candidate, run smooth gradient optimizer for maybe up to few hundred thousand iterations, but limit time
    for name, init_vec in candidates:
        if time.time() >= deadline - 100: # ensure remaining time for hill climb
            break
        # run optimizer
        a_opt, scr = smooth_gradient_optimize(init_vec, deadline - 30, max_iters=500_000,
                                              beta_start=2.0, beta_end=400.0, step0=1.0)
        if scr < best_score:
            best_score = scr
            best_seq = a_opt
            print(f'New best from {name}: score={scr}')

    # after loops, final hill climb:
    if best_seq is not None and time.time() < deadline - 5:
        polished_seq, polished_score = hill_climb(best_seq, deadline)
        if polished_score < best_score:
            best_seq = polished_seq
            best_score = polished_score
            print(f'Polished best: score={polished_score}')
    # final score via evaluate_sequence
    final_score = evaluate_sequence(best_seq.tolist())
    print('Final score', final_score)
    return best_seq.tolist()

def smooth_gradient_optimize(start_a, time_deadline, max_iters=300_000,
                            beta_start=5.0, beta_end=200.0,
                            step0=1.0,
                            adam_beta1=0.9, adam_beta2=0.999,
                            adam_eps=1e-8):
    a = start_a.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    it = 0
    while it < max_iters and time.time() < time_deadline:
        it += 1
        # anneal beta
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        # compute conv
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # compute softmax weights
        # avoid overflow: conv - maxc <=0
        exp_vals = np.exp(beta * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        a_rev = a[::-1]
        # convolve reversed a with w
        grad = 2.0 * convolution_fft(a_rev, w)[n-1:n-1+n]
        # Adam updates
        m = adam_beta1 * m + (1 - adam_beta1) * grad
        v = adam_beta2 * v + (1 - adam_beta2) * (grad * grad)
        m_hat = m / (1 - adam_beta1 ** it)
        v_hat = v / (1 - adam_beta2 ** it)
        step = step0 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)
        # project onto simplex
        a = project_to_simplex(a)
        # track best
        if it % 500 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
    # final check
    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()
    return best_a, best_score

def subgrad_optimize(init_a, time_deadline, max_iter=200_000, step0=0.07):
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # subgradient: g_i = 2 * a_{k-i} for i such that 0<=k-i<n
        # using vectorized
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)
        cur = compute_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
        # occasional jitter
        if t % 2000 == 0:
            jitter = np.random.randn(n) * step0 * 0.01 / np.sqrt(t)
            a = a + jitter
            a = project_to_simplex(a)
    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()

def hill_climb(start_seq, time_deadline, max_steps=400000):
    a = np.array(start_seq, copy=True)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.02  # fraction to move
    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                # enforce non-neg
                if cand[i] < 0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        cur = compute_score(cand)
        if cur < best_score:
            best_score = cur
            best_a = cand
            step_scale = min(step_scale * 1.03, 0.5)
        else:
            step_scale *= 0.98
        max_steps -= 1
    return best_a.tolist(), best_score

"""
Non‑negative step‑function minimiser for `evaluate_sequence`.

Algorithm
---------
1. **Initial shapes** – generate several plausible candidates
   (arcsine, central‑binomial, uniform, random Dirichlet and the provided
   fallback).  All are projected onto the probability simplex so that
   `sum(a)=1`; the objective then is simply `2·n·max(conv(a,a))`.

2. **Smoothed max optimisation** – minimise a log‑sum‑exp surrogate of
   `max(conv)` with Adam.  The smoothing parameter β is annealed from a
   modest value to a large one, so the optimiser first moves the mass
   globally and then focuses on the current worst convolution entry.
   The loop stops automatically when the allocated time expires.

3. **Sub‑gradient polishing** – a short projected sub‑gradient descent
   directly on the true max‑objective refines the solution.

4. **Stochastic hill‑climb** – cheap swaps / fractional mass transfers
   with simplex projection escape tiny local minima.

The whole search respects the 1000 s wall‑clock budget and keeps the best
sequence found.  The returned list contains non‑negative floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum(a)=1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]
    # If all entries are non‑positive the projection is the uniform vector.
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial‑vector factories
# ----------------------------------------------------------------------


def init_arcsine(n: int) -> np.ndarray:
    """Arcsine‑style distribution 1/√(x(1‑x))."""
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    w = 1.0 / np.sqrt(x * (1.0 - x) + eps)
    return project_to_simplex(w)


def init_central_binomial(n: int) -> np.ndarray:
    """Truncated central‑binomial coefficients a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2.0 * i + 1.0) / (2.0 * (i + 1))
    return project_to_simplex(a)


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Smoothed max optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 600_000,
    beta_start: float = 5.0,
    beta_end: float = 300.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the log‑sum‑exp surrogate of max(conv).
    Returns the best vector seen and its exact (unsmoothed) score.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)  # first‑moment estimate
    v = np.zeros_like(a)  # second‑moment estimate

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothing parameter β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()

        # stable soft‑max weights
        exp_vals = np.exp(beta * (conv - maxc))
        w = exp_vals / exp_vals.sum()

        # gradient of the smoothed objective
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a = a - update
        a = project_to_simplex(a)

        # occasional bookkeeping
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimiser on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on 2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()

        # small jitter every few thousand steps
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb(start_seq, time_deadline, max_steps: int = 500_000):
    """
    Tiny stochastic moves (swap or fractional transfer) with simplex projection.
    """
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.02  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:  # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                       # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.04, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # a tiny safety margin

    # -----------------------------------------------------------------
    # Choose a working length (larger → finer granularity, but slower)
    # -----------------------------------------------------------------
    n = 8192 if budget_s > 600 else 4096
    print(f"Working length set to {n}")

    # -----------------------------------------------------------------
    # Build a pool of initial candidates
    # -----------------------------------------------------------------
    candidates = []

    # deterministic shapes
    candidates.append(("arcsine", init_arcsine(n)))
    candidates.append(("central_binomial", init_central_binomial(n)))
    candidates.append(("uniform", np.full(n, 1.0 / n)))

    # a few random Dirichlet draws
    for _ in range(2):
        candidates.append(
            ("random_dirichlet", np.random.dirichlet(np.ones(n)))
        )

    # optional fallback supplied by the harness
    try:
        fb = np.array(height_sequence_1, dtype=float)  # type: ignore
        fb_up = upscale_vector(fb, n)
        candidates.append(("fallback", fb_up))
    except Exception:
        pass

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 1️⃣ Smoothed‑max optimisation on each seed (time‑shared)
    # -----------------------------------------------------------------
    remaining = len(candidates)
    for name, vec in candidates:
        # keep ~80 s for sub‑gradient + hill‑climb, divide the rest equally
        time_for_this = max(0.0, (deadline - time.time() - 80.0) / remaining)
        if time_for_this <= 0.5:
            break
        sub_deadline = time.time() + time_for_this
        print(f"Optimising start '{name}' (allocated {time_for_this:.1f}s)")
        a_opt, scr = smooth_gradient_optimize(
            vec,
            sub_deadline,
            max_iters=500_000,
            beta_start=5.0,
            beta_end=300.0,
            step0=1.0,
        )
        if scr < best_score:
            best_score = scr
            best_seq = a_opt
            print(f"  ► New best after '{name}': score = {scr:.6f}")
        remaining -= 1

    # -----------------------------------------------------------------
    # 2️⃣ Sub‑gradient polishing (if time permits)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40.0:
        print("Running sub‑gradient refinement")
        a_refined, scr_refined = subgrad_optimize(
            best_seq,
            deadline - 30.0,
            max_iter=300_000,
            step0=0.07,
        )
        if scr_refined < best_score:
            best_score = scr_refined
            best_seq = a_refined
            print(f"  ► Sub‑grad improved best to {scr_refined:.6f}")

    # -----------------------------------------------------------------
    # 3️⃣ Final stochastic hill‑climb (tight‑budget polishing)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("Hill‑climb polishing")
        polished_seq, polished_score = hill_climb(
            best_seq,
            deadline,
            max_steps=600_000,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq
            print(f"  ► Hill‑climb improved best to {polished_score:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to a tiny uniform vector if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # Final evaluation with the original security‑checked function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)  # type: ignore
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return list(best_seq)
