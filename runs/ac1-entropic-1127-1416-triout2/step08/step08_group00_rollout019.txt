def project_to_simplex(v):
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def evaluate_sequence(sequence):
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---
    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:

"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` objective.

Algorithm overview
------------------
1. **Coarse exploration** – many cheap random restarts (length 1024)
   using a smooth log‑sum‑exp surrogate of the max‑convolution.
   The surrogate temperature β is annealed to a large value (≈1500)
   to approach the true max. The best vector from this stage is kept.

2. **Upsampling** – the best vector is split into twice as many
   entries (each mass is halved and placed in two adjacent slots).
   This operation reduces the self‑squared term and gives a finer
   search space.

3. **Projected sub‑gradient refinement** – a classic sub‑gradient
   method on the exact max‑objective, with a decreasing step‑size
   and Euclidean projection onto the probability simplex.

4. **Stochastic hill‑climb / simulated‑annealing polishing** –
   random local moves (swap or small mass transfer).  Moves that
   improve the exact score are always accepted; worsening moves
   are accepted with a temperature‑controlled probability that
   decays to zero.  This helps escape shallow local minima.

All stages respect the 1000 s wall‑clock budget, and every vector
is kept on the simplex (non‑negative, unit sum) so that the scaling
invariance of `evaluate_sequence` is honoured.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` score under the constraint
    Σ a_i = 1 (i.e. the denominator of `evaluate_sequence`
    is 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random start generators (different shapes)
# ----------------------------------------------------------------------


def random_start_dense(n: int) -> np.ndarray:
    """Uniform Dirichlet‑(1) start."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_start_sparse(n: int) -> np.ndarray:
    """Place mass on about √n positions."""
    k = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_start_gaussian(n: int) -> np.ndarray:
    """A smooth bell‑shaped start."""
    x = np.arange(n)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_start(n: int) -> np.ndarray:
    """Pick one of the three generators at random."""
    r = np.random.rand()
    if r < 0.4:
        return random_start_dense(n)
    elif r < 0.7:
        return random_start_sparse(n)
    else:
        return random_start_gaussian(n)


# ----------------------------------------------------------------------
# 1. Smooth log‑sum‑exp optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 1500.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a differentiable soft‑max surrogate of
    max(conv(a,a)) using Adam momentum and a simplex projection.
    Returns the best exact (non‑smoothed) score seen.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # linear annealing of β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed objective
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))            # stable weights
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient of the soft‑max (cross‑correlation trick)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on 2·n·max(conv(a,a)).
    The update direction comes from the (unique) maximising offset.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index of the maximal entry

        # sub‑gradient for this index (see analysis in the description)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # tiny random jitter every 2000 iterations to avoid stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = None,
) -> tuple[list[float], float]:
    """
    Random local modifications (swap or mass transfer) with a
    temperature schedule.  Accept improving moves always;
    accept worsening moves with probability exp(-Δ/T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    cur_a = a.copy()
    cur_score = best_score

    start = time.time()
    steps = 0
    # initial temperature as a modest fraction of the score
    T0 = best_score * 0.02
    Tmin = best_score * 1e-6

    while (max_steps is None or steps < max_steps) and time.time() < time_deadline:
        # linear cooling based on elapsed wall‑clock time
        elapsed = time.time() - start
        total = time_deadline - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            # swap two random positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * cand[i]          # 5 % of the source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        # acceptance test
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 4. Upsampling helper (splits each mass into two halves)
# ----------------------------------------------------------------------


def upsample_sequence(seq: list[float]) -> np.ndarray:
    """
    Double the length by replacing each entry a_i with two
    adjacent entries a_i/2, a_i/2.  The total mass is unchanged.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    up = np.empty(2 * n, dtype=float)
    up[0::2] = a * 0.5
    up[1::2] = a * 0.5
    # projection is only a safety net against tiny numerical drift
    up = project_to_simplex(up)
    return up
