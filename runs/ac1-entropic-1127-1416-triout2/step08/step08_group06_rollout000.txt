"""
Propose a step sequence with a low official `evaluate_sequence` score.

Algorithm
---------

1. **Fast baseline** – start from the classic arcsine shape (β=0.5) for a
   moderate length (4096).  Its score is already ~1.511.

2. **Parametric grid search** – sweep a few low‑dimensional families of
   shapes that generalise the arcsine envelope:
      * `γ‑shape`   a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^(-γ)
      * `power‑shape` a_i ∝ ((i+½)(n‑½‑i)+ε)^(-p)
      * `mixture`    w·arcsine + (1‑w)·uniform
   The grids are coarse but cheap; every candidate is evaluated with a
   fast FFT‑based `compute_score` (equivalent to the official objective
   when the vector is normalised to L₁=1).

3. **Local random refinement** – from the current best vector we repeatedly
   pick two indices and move a random fraction of mass from one to the
   other, keeping the L₁‑norm fixed.  Accept only improving moves.
   This simple hill‑climb explores neighbouring points that are not
   reachable by the parametric families.

4. **Multiplicative (KL‑mirror) polishing** – run a few thousand
   exponentiated‑gradient iterations on the true max‑convolution objective
   while staying on the probability simplex.  The step size and number of
   iterations are chosen adaptively from the remaining time budget.

5. **Final check** – evaluate the official `evaluate_sequence` once (just
   for logging) and return the best vector found.  The routine always
   returns a non‑empty sequence before the 1000 s wall‑clock limit.

The whole search stays well within the time budget on a single CPU.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose entries
    sum to 1.  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_gamma_shape(n: int, gamma: float, delta: float = 0.5,
                         eps: float = 0.0) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^(-γ).
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_power_shape(n: int, p: float, eps: float = 0.0) -> np.ndarray:
    """Power‑law shape –  a_i ∝ ((i+½)(n‑½‑i)+ε)^(-p)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, w: float) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = w * arc + (1.0 - w) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the exact sub‑gradient method
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(a0: np.ndarray,
                                 deadline: float,
                                 max_iter: int = 150_000,
                                 step: float = 0.01) -> np.ndarray:
    """
    Mirror‑descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # convolution of a with the mask reversed
        rev = convolution_fft(a, mask[::-1])
        # subgradient of max‑convolution w.r.t. a
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # multiplicative update
        a = a * np.exp(-step * subgrad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# (Optional) Exact projected sub‑gradient descent – used if time permits
# ----------------------------------------------------------------------
def subgradient_descent_exact(init_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score(a):.6f}")

    return a

    Search for a step sequence with a low official ``evaluate_sequence`` score.
    The function respects the 1000 s wall‑clock limit and always returns a
    non‑empty sequence of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Try the fallback sequence if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Baseline arcsine (length 4096)
    # ------------------------------------------------------------------
    n_base = 4096
    base_seq = arcsine_initial(n_base)
    base_val = compute_score(base_seq)
    print(f"Baseline arcsine n={n_base} score={base_val:.6f}")
    if base_val < best_val:
        best_val, best_seq = base_val, base_seq.tolist()

    # ------------------------------------------------------------------
    # 2.  Parametric grid search over several families and lengths
    # ------------------------------------------------------------------
    lengths = [4096, 2048]      # keep a small length for speed, larger later if time
    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n--- Grid search for length {n} ---")

        # -------- gamma‑shape -------------------------------------------------
        print("  Gamma‑shape grid")
        for gamma in np.arange(0.30, 0.71, 0.02):      # 0.30 .. 0.70
            for delta in (0.1, 0.5, 1.0):
                for eps in (0.0, 0.1, 0.5, 1.0):
                    if time.time() >= deadline:
                        break
                    cand = generate_gamma_shape(n, gamma, delta, eps)
                    val = compute_score(cand)
                    if val < best_val:
                        best_val, best_seq = val, cand.tolist()
                        print(f"New best γ={gamma:.2f}, δ={delta:.1f}, ε={eps:.1f}, n={n} → {val:.6f}")

        # -------- power‑shape grid -------------------------------------------
        print("  Power‑shape grid")
        for p in np.arange(0.25, 0.81, 0.05):          # 0.25 .. 0.80
            for eps in (0.0, 0.1, 0.5, 1.0):
                if time.time() >= deadline:
                    break
                cand = generate_power_shape(n, p, eps)
                val = compute_score(cand)
                if val < best_val:
                    best_val, best_seq = val, cand.tolist()
                    print(f"New best p={p:.2f}, ε={eps:.1f}, n={n} → {val:.6f}")

        # -------- mixture (arcsine + uniform) ---------------------------------
        print("  Mixture shape grid")
        for w in np.linspace(0.0, 1.0, 11):            # 0.0 .. 1.0 step 0.1
            if time.time() >= deadline:
                break
            cand = generate_mixture_shape(n, w)
            val = compute_score(cand)
            if val < best_val:
                best_val, best_seq = val, cand.tolist()
                print(f"New best mix w={w:.2f}, n={n} → {val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Simple random local refinement (mass transfers)
    # ------------------------------------------------------------------
    if best_seq:
        a_arr = np.array(best_seq, dtype=float)
        n = a_arr.size
        print("\n--- Local random refinement (mass transfers) ---")
        transfer_iters = 0
        while time.time() < deadline - 15.0 and transfer_iters < 30_000:
            transfer_iters += 1
            i, j = rng.choice(n, size=2, replace=False)
            # move a random fraction (up to 10 %) of the mass from i to j
            frac = rng.uniform(0.0, 0.1)
            amt = a_arr[i] * frac
            if amt <= 0:
                continue
            new_a = a_arr.copy()
            new_a[i] -= amt
            new_a[j] += amt
            # enforce non‑negativity (should already hold)
            if new_a[i] < 0:
                continue
            new_a = new_a / new_a.sum()
            val = compute_score(new_a)
            if val < best_val:
                best_val, best_seq = val, new_a.tolist()
                a_arr = new_a
                print(f"  Transfer improve it={transfer_iters:,d} → {val:.6f}")

    # ------------------------------------------------------------------
    # 4.  KL‑mirror (exponentiated‑gradient) polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq and remaining > 12.0:
        print("\n--- Exponentiated‑gradient polishing ---")
        a_arr = np.array(best_seq, dtype=float)

        # Determine a safe iteration budget based on remaining time.
        # One iteration costs roughly O(n log n); we limit to ~150 k for n≈4k.
        max_iter = int(min(150_000, remaining * 2000))
        a_arr = exponentiated_gradient_polish(
            a_arr,
            deadline=deadline,
            max_iter=max_iter,
            step=0.01,
        )
        val = compute_score(a_arr)
        if val < best_val:
            best_val, best_seq = val, a_arr.tolist()
            print(f"After EG polish score={val:.6f}")

    # ------------------------------------------------------------------
    # 5.  Optional exact sub‑gradient descent (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq and remaining > 6.0:
        print("\n--- Final sub‑gradient descent (optional) ---")
        a_arr = np.array(best_seq, dtype=float)
        max_iter = int(min(200_000, remaining * 1500))
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline=deadline,
            max_iter=max_iter,
            step0=0.07,
        )
        val = compute_score(a_arr)
        if val < best_val:
            best_val, best_seq = val, a_arr.tolist()
            print(f"After sub‑gradient score={val:.6f}")

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee that we return something valid
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # 7.  Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
