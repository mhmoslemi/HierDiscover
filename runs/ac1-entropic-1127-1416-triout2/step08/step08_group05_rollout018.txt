"""
Propose a step‑function (non‑negative sequence) with a very low
evaluate_sequence() value.  The method consists of three phases:

1.  Initialise with the analytically‑motivated arcsine shape, add a tiny
    random perturbation to break symmetry and optimise a smooth log‑sum‑exp
    upper‑bound on max(conv(a,a)) using L‑BFGS‑B.  The temperature β is
    increased in stages (5 → 100 → 500) so that the surrogate gradually
    approaches the true max.

2.  Upsample the best solution to a larger length (2×) and repeat a
    short L‑BFGS refinement.  A finer discretisation reduces the
    discretisation error of the continuous optimal shape and usually
    pushes the bound a little lower.

3.  Final polishing with exponentiated‑gradient (KL‑mirror) descent on the
    exact objective.  This step respects the simplex constraint automatically.

The whole routine respects a 1000 s wall‑clock budget, always returns a
valid sequence and prints intermediate scores for monitoring.
"""

import time
from typing import List
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# FFT‑based convolution utilities
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (no circular wrap).
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# --------------------------------------------------------------
# Simplex helpers
# --------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# --------------------------------------------------------------
# Analytic seed families (only arcsine is needed here)
# --------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Arcsine‑like envelope, normalised to sum = 1.
    This shape is known to give a very flat self‑convolution.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of an existing sequence to a larger size.
    The result is re‑normalised to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# --------------------------------------------------------------
# Smooth‑max surrogate for L‑BFGS
# --------------------------------------------------------------


class SmoothObjective:
    """
    Callable that returns (value, gradient) of the smooth log‑sum‑exp
    surrogate of   f(a)=2·n·max(conv(a,a)).
    Internally a is expressed as a soft‑max of the free variables z.
    """

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, z: np.ndarray):
        # a is always on the simplex
        a = softmax(z)

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth max = (1/β)·log Σ exp(β·conv_i)
        mx = np.max(conv)
        shifted = conv - mx                     # ≤0, improves stability
        exp_beta = np.exp(self.beta * shifted)
        sum_exp = np.sum(exp_beta)
        # log‑sum‑exp (stable)
        smooth_max = mx + (1.0 / self.beta) * np.log(sum_exp)

        # objective value (exact scaling 2·n)
        obj = 2.0 * self.n * smooth_max

        # gradient of smooth_max w.r.t a:
        # weights w_i = exp(β·conv_i) / Σ exp(β·conv)
        w = exp_beta / sum_exp
        # reverse for convolution
        w_rev = w[::-1]
        rev = convolution_fft(a, w_rev)                # length 2n‑1
        rev_center = rev[self.n - 1 : self.n - 1 + self.n]
        grad_a = 2.0 * rev_center                       # d(smooth_max)/da

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = 2.0 * self.n * a * (grad_a - dot)      # d obj / dz
        return obj, grad_z


# --------------------------------------------------------------
# Exact sub‑gradient (KL‑mirror) polishing
# --------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Mirror descent (KL divergence) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        if it % 5_000 == 0:
            # cheap surrogate score
            conv_max = np.max(conv)
            print(f"    EG it={it:,d} surrogate score={2*n*conv_max:.6f}")

    return a

def propose_candidate(seed: int = 43, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises the
    ``evaluate_sequence`` value.  The routine respects the 1000 s time budget,
    always returns a (non‑empty) valid sequence and prints progress.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1.  First optimisation at a moderate length (8192)
    # ------------------------------------------------------------------
    n = 8192
    print(f"\n=== Phase 1 – length {n} ===")
    # analytic seed + tiny random asymmetry
    init = arcsine_initial(n)
    init = init + rng.normal(scale=1e-6, size=n)   # break perfect symmetry
    init = np.clip(init, 1e-16, None)
    init = init / init.sum()
    z0 = np.log(init + 1e-12)                     # log‑softmax coordinates

    # temperature schedule (smooth‑max becomes tighter)
    betas = [5.0, 100.0, 500.0]

    for beta in betas:
        if time.time() >= deadline - 20.0:
            break
        print(f"\n--- L‑BFGS with β={beta:.1f} ---")
        max_iter = 3_000                         # L‑BFGS iterations (≈ a few sec)

        obj_fun = SmoothObjective(n, beta)

        # SciPy minimise (L‑BFGS‑B)
        res = minimize(
            obj_fun,
            z0,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
        a_opt = softmax(res.x)

        # official evaluation
        try:
            val = evaluate_sequence(a_opt.tolist())
            print(f"   Official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                print(f"   *** New best: {best_val:.6f} ***")
        except Exception as e:
            print(f"   Evaluation failed: {e}")

        # use the result as the starting point for the next β
        z0 = res.x.copy()

        # stop early if we are already very close to the target
        if best_val <= 1.505:
            break

    # ------------------------------------------------------------------
    # 2.  Upsample to a larger grid and do a quick refinement
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 30.0:
        n_big = 16384
        print(f"\n=== Phase 2 – up‑sample to {n_big} ===")
        a_big = upsample_sequence(np.array(best_seq, dtype=float), n_big)
        z_big = np.log(a_big + 1e-12)

        # a single L‑BFGS pass with a high β – few iterations are enough
        beta_big = 800.0
        max_iter_big = 1_500
        obj_big = SmoothObjective(n_big, beta_big)

        res_big = minimize(
            obj_big,
            z_big,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter_big, "disp": False},
        )
        a_big_opt = softmax(res_big.x)

        try:
            val_big = evaluate_sequence(a_big_opt.tolist())
            print(f"   Upsampled official score = {val_big:.6f}")
            if val_big < best_val:
                best_val, best_seq = val_big, a_big_opt.tolist()
                print(f"   *** New global best after up‑sample: {best_val:.6f} ***")
        except Exception as e:
            print(f"   Upsampled evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3.  Final KL‑mirror polishing on the exact objective
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Phase 3 – final KL‑mirror polish ===")
        a_arr = np.array(best_seq, dtype=float)
        polished = exponentiated_gradient_polish(
            a_arr,
            deadline,
            max_iter=200_000,
            step=0.018,
        )
        try:
            final_val = evaluate_sequence(polished.tolist())
            print(f"   Final polish score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, polished.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback uniform sequence (guaranteed finite score)
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
