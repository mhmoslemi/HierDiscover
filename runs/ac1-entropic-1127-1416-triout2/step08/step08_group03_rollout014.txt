def greedy_sidon(n, target_k=None):
    if target_k is None:
        target_k = int(np.sqrt(n) * 1.3)  # Slightly above.
    pos = []
    sums = set()
    candidate_indices = list(range(n))
    random.shuffle(candidate_indices)
    for i in candidate_indices:
        # compute new sums if i added
        collision = False
        for p in pos:
            if (i + p) in sums:
                collision = True
                break
        if not collision:
            # also check self sum i+i not in sums
            if (i + i) in sums:
                continue
            pos.append(i)
            for p in pos:
                sums.add(i + p)
            if len(pos) >= target_k:
                break
    # assign equal weight to each selected position; normalize to sum=1
    a = np.zeros(n)
    if pos:
        a[pos] = 1.0 / len(pos)
    return a
```

"""
A fast multi‑restart optimiser for the non‑negative step‑function problem.

Key ideas
---------
1. Work with a modest length (n = 2048).  This keeps each FFT cheap,
   allowing many more optimisation steps than the previous 8192‑length
   run.
2. Initialise from many diverse families (uniform, analytic families,
   random sparse / monotone vectors and a greedy Sidon‑set construction).
3. For every start run a long “smooth‑max” optimisation using Adam on a
   soft‑max parametrisation.  The smoothing parameter β is annealed from
   5 → 500 and the iteration count is limited only by the global deadline.
4. Refine the best vector with a cheap projected sub‑gradient method
   (step‑size ∝ 1/√t) – the most iterations possible before time runs out.
5. Finish with a simulated‑annealing hill‑climb that swaps or moves mass
   while staying on the probability simplex.
6. All phases respect the wall‑clock budget; the best vector encountered
   is finally returned (as a plain Python list).

The objective that we minimise is
        f(a) = 2·n·max( a * a ) / ( Σ a )² .
Because the function is scale‑invariant we normalise all candidates to
Σ a = 1 and simply minimise 2·n·max(a*a), which is computed by
`compute_score`.  The final score is obtained with the provided
`evaluate_sequence` (which applies the same formula plus the required
safety checks).
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the probability simplex:
        2·n·max(a*a)
    (n = len(a), a is assumed to sum to 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for diverse initial guesses)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random / combinatorial initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)        # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector obtained by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


def greedy_sidon_start(n: int, target_k: int = None) -> np.ndarray:
    """
    Greedy construction of a Sidon‑like set (no repeated pair sums).
    The resulting vector places equal mass on the chosen positions.
    """
    if target_k is None:
        target_k = int(1.3 * np.sqrt(n))     # a bit above √n when possible
    positions = []
    sums = set()
    indices = list(range(n))
    random.shuffle(indices)
    for i in indices:
        # test whether adding i would create a duplicate sum
        conflict = False
        # self‑sum check
        if (i + i) in sums:
            continue
        for p in positions:
            if (i + p) in sums:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        positions.append(i)
        for p in positions:
            sums.add(i + p)
        if len(positions) >= target_k:
            break
    a = np.zeros(n, dtype=np.float64)
    if positions:
        a[positions] = 1.0 / len(positions)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth optimisation using Adam on the log‑sum‑exp smoothing of max
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 5.0,
                         beta_end: float = 500.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially from
    beta_start → beta_end.
    """
    n = start_a.shape[0]
    eps = 1e-14
    z = np.log(start_a + eps)          # initialise from current simplex point

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- β schedule (exponential) ---------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # --- current a ------------------------------------------------
        a = softmax(z)

        # --- smoothed weights -----------------------------------------
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        # stable exp
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                    # soft‑max over convolution entries

        # --- gradient w.r.t. a (see derivation in the comment block) --
        a_rev = a[::-1]
        # conv_rev = a_rev * w (full convolution)
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]   # centred part

        # --- back‑prop through soft‑max -------------------------------
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # --- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- occasional exact evaluation -------------------------------
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (very cheap per iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method for the true max‑objective
          2·n·max(a*a)
    using a diminishing stepsize  ∝ 1/√t.  The iterate is projected onto
    the probability simplex after each step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # build sub‑gradient of the active max entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with Simulated Annealing
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Long random walk on the simplex.
    Moves:
        * with prob 0.5 swap two entries,
        * otherwise transfer a fraction of mass from i→j.
    Very small temperature schedule permits occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # tiny temperature – enough to escape shallow plateaus
    T0 = best_score * 1e-4
    T = T0

    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sidon) = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Adam‑based smoothed optimisation from the best candidates found
    # --------------------------------------------------------------
    # We'll reuse the current best as a starting point, but also launch a
    # few fresh random starts to diversify the search.
    starts = []
    if best_seq is not None:
        starts.append(np.array(best_seq, dtype=np.float64))
    # add a handful of fresh random starts
    for _ in range(8):
        starts.append(random_sparse_start(n))

    # optimise each start while we have time
    for idx, start_vec in enumerate(starts):
        if time.time() >= deadline - 20.0:
            break
        print(f"[{time.time():.1f}] Adam run {idx+1}/{len(starts)}")
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 10.0,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=500.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after Adam = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Long projected sub‑gradient refinement (very cheap per iteration)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 5.0,
            max_iters=2_500_000,
            step0=0.08,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Symmetrisation – many good solutions turn out symmetric
    # --------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final hill‑climb with a lightweight annealing schedule
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Safeguard – fallback to a simple uniform vector if nothing else
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
