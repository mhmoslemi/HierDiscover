"""
Propose a non‑negative step sequence that minimizes the official
``evaluate_sequence`` objective.

Algorithm Overview
------------------

1. **Fast score** – ``compute_score`` evaluates the objective for a vector
   whose L1‑norm equals 1 (the true objective is scale‑invariant).

2. **Broad parametric search** – a large pool of families (beta,
   power‑law, generalized arcsine, Gaussian, sine, triangular,
   mixtures and simple uniform) is sampled with random parameters.
   For each candidate the fast score is computed; the best shape seen
   so far (across many lengths) is retained.

3. **Fine refinement** – the current global best is fed to a
   multiplicative (KL‑mirror) exponentiated‑gradient descent that works
   directly on the exact max‑convolution objective.  A few thousand
   iterations are performed until the time budget expires.

4. **Safety net** – if nothing was found a uniform vector of length 512
   is returned.

All steps respect the 1000 s wall‑clock limit and the sequence is always
non‑negative.
"""

# ----------------------------------------------------------------------
# Imports & low‑level utilities
# ----------------------------------------------------------------------
import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast objective for an L1‑normalised vector
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """
    Returns the official objective value for a vector with sum = 1.
    For a general vector the objective is scale‑invariant, so this is
    sufficient for the search.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (all normalised to L1‑norm = 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine–like envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_gamma_shape(n: int, gamma: float, delta: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{‑γ}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_gaussian_shape(n: int, sigma_factor: float) -> np.ndarray:
    """Truncated discrete Gaussian, σ = sigma_factor·n."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = max(sigma_factor * n, 1e-12)
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_sine_shape(n: int) -> np.ndarray:
    """Half‑sine envelope – zeros at the extremes are avoided by a tiny offset."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n)  # >0 for all i
    vals = np.clip(vals, 1e-15, None)
    return vals / vals.sum()


def generate_triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangle (linear rise then fall)."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    vals = np.where(i <= mid, i + 1, n - i)
    return vals / vals.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the exact max‑convolution
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step: float = 0.012,
) -> np.ndarray:
    """Multiplicative (mirror‑descent) optimisation of the true objective."""
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance to collect all indices attaining the maximum
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # convolution of a with the reversed mask gives a sub‑gradient
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if not np.isfinite(s) or s <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d}  score={compute_score(a):.6f}")

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function sequence with a low official
    ``evaluate_sequence`` score.  The routine respects a 1000 s wall‑clock
    limit and always returns a non‑empty list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Warm‑start from a previously stored solution (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback candidate (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Broad random parametric search over several lengths
    # ------------------------------------------------------------------
    lengths = [4096, 3072, 2048, 1536, 1024, 768, 512, 384, 256]
    shape_kinds = ["arcsine", "uniform", "beta", "power", "gamma",
                   "gaussian", "sine", "triangular", "mixture"]

    # Estimate a rough per‑candidate cost (FFT + bookkeeping) – empirically
    # around 5–6 ms for n≈4096, considerably less for smaller n.
    # We'll simply cap the total number of candidates per length.
    max_cand_per_len = 30_000

    for n in lengths:
        if time.time() >= deadline - 15.0:
            break
        print(f"\n--- Random search on length {n} ---")
        # allocate a sub‑deadline for this length (≈30 % of remaining time)
        len_deadline = min(deadline - 2.0, time.time() + (deadline - time.time()) * 0.30)

        it = 0
        while time.time() < len_deadline and it < max_cand_per_len:
            it += 1
            kind = rng.choice(shape_kinds)

            if kind == "arcsine":
                cand = arcsine_initial(n)

            elif kind == "uniform":
                cand = np.full(n, 1.0 / n)

            elif kind == "beta":
                alpha = 10 ** rng.uniform(-1.0, 0.7)   # ≈0.1 … 5
                beta = 10 ** rng.uniform(-1.0, 0.7)
                eps = rng.uniform(0.0, max(0.5, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)

            elif kind == "power":
                p = rng.uniform(0.2, 0.9)
                eps = rng.uniform(0.0, 2.0)
                cand = generate_power_shape(n, p, eps)

            elif kind == "gamma":
                gamma = rng.uniform(0.25, 0.75)
                delta = rng.uniform(0.1, 2.0)
                eps = rng.uniform(0.0, 0.5)
                cand = generate_gamma_shape(n, gamma, delta, eps)

            elif kind == "gaussian":
                sigma_factor = rng.uniform(0.05, 0.30)   # σ = factor·n
                cand = generate_gaussian_shape(n, sigma_factor)

            elif kind == "sine":
                cand = generate_sine_shape(n)

            elif kind == "triangular":
                cand = generate_triangular_shape(n)

            elif kind == "mixture":
                w = rng.uniform(0.0, 1.0)
                cand = generate_mixture_shape(n, w)

            else:
                # should never happen
                cand = np.full(n, 1.0 / n)

            # safeguard against numerical glitches
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()

            # fast evaluation (L1‑norm = 1)
            val = compute_score(cand)

            if val < best_val:
                best_val = val
                best_seq = cand.tolist()
                print(f"  New best (len={n}) it={it:,d}  score={val:.6f}")

        # ------------------------------------------------------------------
        # Brief polishing of the current best for this length (if time left)
        # ------------------------------------------------------------------
        if best_seq and (deadline - time.time() > 12.0):
            print(f"  Polishing best candidate of length {n} with EG …")
            a_arr = np.array(best_seq, dtype=float)
            a_arr = exponentiated_gradient_polish(
                a_arr,
                deadline=deadline,
                max_iter=int(min(80_000, (deadline - time.time()) * 2000)),
                step=0.012,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"  EG polish improved score to {val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Global polishing phase (if we still have time)
    # ------------------------------------------------------------------
    if best_seq and (deadline - time.time() > 20.0):
        print("\n--- Global polishing phase ---")
        a_arr = np.array(best_seq, dtype=float)

        # 2a. Exponentiated‑gradient (longer run)
        a_arr = exponentiated_gradient_polish(
            a_arr,
            deadline=deadline,
            max_iter=int(min(150_000, (deadline - time.time()) * 1500)),
            step=0.010,
        )
        val = compute_score(a_arr)
        if val < best_val:
            best_val, best_seq = val, a_arr.tolist()
            print(f"  After global EG polish score = {val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a valid non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))
        print(f"Fallback to uniform length {n0}, score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final official evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Official evaluate_sequence score = {official:.6f}")
    except Exception as exc:
        print(f"\nSearch finished – official evaluation failed: {exc}")

    return best_seq
