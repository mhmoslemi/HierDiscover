"""
Propose a step‐function (non‑negative sequence) with a low
evaluate_sequence() score.

Strategy
--------

1. **Rich parametric pool** – generate candidates from many classical
   windows (arcsine, beta, power‑law, Gaussian, cosine, Blackman,
   triangular, mixtures).  All candidates are normalised to L1‑norm = 1.

2. **Fast screening** – each candidate is scored by the cheap FFT based
   ``compute_score`` (identical to the official objective when the L1‑norm
   is 1).  The best shape found so far is kept.

3. **Intensive polishing** – the current best shape is refined three
   times, each respecting the simplex:
      * KL‑mirror (exponentiated‑gradient) descent with a decreasing step.
      * Adam on a smooth log‑sum‑exp surrogate of the max‐convolution,
        slowly raising the temperature β to sharpen the approximation.
      * Exact projected sub‑gradient descent on the true max‑convolution
        objective, running until the time budget is exhausted.

4. **Final stochastic tweak** – a tiny simulated‑annealing loop that
   swaps small mass bits between coordinates; the temperature linearly
   decays to zero, guaranteeing a non‑worsening move.

The whole routine monitors the wall‑clock and always returns a non‑empty
sequence before the 1000 s budget expires.
"""

import time
from typing import List
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose entries sum to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def arcsine_eps(n: int, eps: float = 0.25) -> np.ndarray:
    """Arcsine‑like envelope with a smoothing epsilon."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β)‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape – denominator raised to p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_gamma_shape(n: int, gamma: float, delta: float = 0.5, eps: float = 0.0) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{-γ}.
    The classic arcsine corresponds to γ=0.5, δ≈0.5, ε≈0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    arc = arcsine_eps(n, eps=0.25)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_cosine_shape(n: int, power: float = 1.0) -> np.ndarray:
    """Raised cosine (Hann‑like) window."""
    i = np.arange(n, dtype=float)
    vals = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** power
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_blackman_shape(n: int) -> np.ndarray:
    """Blackman window – a smooth taper with zero endpoints."""
    i = np.arange(n, dtype=float)
    a0, a1, a2 = 0.42, 0.5, 0.08
    vals = (a0
            - a1 * np.cos(2.0 * np.pi * i / (n - 1))
            + a2 * np.cos(4.0 * np.pi * i / (n - 1)))
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular (tent) window."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - centre) / centre
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by sub‑gradient method
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for “active” indices
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # reverse mask for convolution with a
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional logging
        if it % 10_000 == 0:
            print(f"    Sub‑grad it={it:,d}  score={compute_score(a):.6f}")

    return a

# KL‑mirror (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step: float = 0.01,
) -> np.ndarray:
    """
    Mirror‑descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 10_000 == 0:
            print(f"    EG it={it:,d}  score={compute_score(a):.6f}")

    return a

def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 500.0,
    ramp_iters: int = 80_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam on the log‑sum‑exp smooth approximation of max(conv).
    Operates in the log‑domain (soft‑max enforces the simplex).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ≥ 0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp (shifted for stability)
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient wrt a : 2·(a ⋆ w_rev)
        rev = convolution_fft(a, w_norm[::-1])
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 10_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.0f}  score={compute_score(a):.6f}")

    return softmax(z)

# Simulated‑annealing style fine‑tuning (tiny random swaps)
# ----------------------------------------------------------------------
def simulated_annealing_refine(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Very light‑weight stochastic refinement.
    Swaps a tiny amount of mass between two random positions.
    Temperature decays linearly to zero; downhill moves are always kept,
    uphill moves accepted with Metropolis probability.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_val = compute_score(best)

    # initial temperature is a small fraction of the current score
    T0 = 0.05 * best_val
    start = time.time()
    total_time = deadline - start
    if total_time <= 0:
        return best

    it = 0
    while time.time() < deadline:
        it += 1
        # linear decay
        elapsed = time.time() - start
        T = max(0.0, T0 * (1.0 - elapsed / total_time))

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # amount to transfer – a tiny fraction of the source mass
        delta = 1e-4 * a[i] * rng.random()
        if delta <= 0:
            continue

        a[i] -= delta
        a[j] += delta

        val = compute_score(a)
        accept = False
        if val < best_val:
            best_val, best = val, a.copy()
            accept = True
        elif T > 0:
            # Metropolis criterion
            prob = np.exp(-(val - best_val) / T)
            if rng.random() < prob:
                accept = True

        if not accept:
            # revert the move
            a[i] += delta
            a[j] -= delta
        else:
            # keep a projected onto the simplex (numerical drift)
            a = project_to_simplex(a)

        if it % 20_000 == 0:
            print(f"    SA it={it:,d}  score={best_val:.6f}")

    return best

            # normalise safely
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()

            # fast evaluation
            val = compute_score(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.tolist()
                print(f"New best (len={n}) it={iter_cnt:,d}  score={val:.6f}")

        # ------------------------------------------------------------------
        # 2. Brief polishing of the current best for this length (if time)
        # ------------------------------------------------------------------
        if best_seq and (deadline - time.time() > 10.0):
            a_arr = np.array(best_seq, dtype=float)
            print(f"  Polishing length {n} (exponentiated‑gradient)…")
            a_arr = exponentiated_gradient_polish(
                a_arr,
                deadline=deadline,
                max_iter=int(min(70_000, (deadline - time.time()) * 1500)),
                step=0.009,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"  Polished (EG) new best score={val:.6f}")

    # ------------------------------------------------------------------
    # 3. Global polishing – three‑stage refinement on the best shape
    # ------------------------------------------------------------------
    if best_seq and (deadline - time.time() > 30.0):
        print("\n--- Global polishing phase ---")
        a_arr = np.array(best_seq, dtype=float)

        # 3a. KL‑mirror (exponentiated‑gradient) – longer run
        a_arr = exponentiated_gradient_polish(
            a_arr,
            deadline=deadline,
            max_iter=int(min(150_000, (deadline - time.time()) * 1500)),
            step=0.008,
        )
        val = compute_score(a_arr)
        if val < best_val:
            best_val, best_seq = val, a_arr.tolist()
            print(f"After EG polish score={val:.6f}")

        # 3b. Adam on smooth surrogate (temperature ↑)
        if time.time() < deadline - 5.0:
            eps = 1e-12
            z0 = np.log(np.clip(a_arr, eps, None))
            a_arr = smooth_adam_optimize(
                z0,
                deadline=deadline,
                n=a_arr.size,
                beta0=5.0,
                beta_max=800.0,
                ramp_iters=100_000,
                lr=0.07,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"After Adam polish score={val:.6f}")

        # 3c. Exact sub‑gradient descent (the most accurate)
        if time.time() < deadline - 5.0:
            a_arr = subgradient_descent_exact(
                a_arr,
                deadline=deadline,
                max_iter=400_000,
                step0=0.06,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"After final sub‑gradient score={val:.6f}")

        # 3d. Tiny stochastic tweak (simulated annealing)
        if time.time() < deadline - 2.0:
            a_arr = simulated_annealing_refine(
                np.array(best_seq, dtype=float),
                deadline=deadline,
                rng=rng,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"After SA tweak score={val:.6f}")

    # ------------------------------------------------------------------
    # 4. Guard – guarantee that we return something valid
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))
        print("Fallback to uniform sequence.")

    # ------------------------------------------------------------------
    # 5. Final official evaluation (pure logging)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
