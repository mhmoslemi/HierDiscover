"""
Propose a high‑quality step‑function sequence.

The search works in three phases:

1. **Exploratory grid / random search** on a collection of analytic families
   (arcsine‑like, power‑law, beta‑shapes, mixtures).  For each candidate
   the fast surrogate `2·n·max(conv(a,a))` is evaluated (the exact
   `evaluate_sequence` value is scaling‑invariant, so the surrogate is
   sufficient as long as the vector is normalised to sum = 1).

2. **Upsampling** – the current best vector is linearly interpolated to a
   larger support size, which often yields a strictly better objective.

3. **Mirror‑descent polishing** – the best candidate is refined with an
   exponentiated‑gradient (KL‑mirror) iteration directly on the true
   max‑convolution objective.  This step respects the remaining time budget
   and stays on the probability simplex automatically.

The routine respects the 1000 s wall‑clock budget, always returns a valid
(non‑negative) sequence, and prints progress information.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (no circular wrap).
    Returns a 1‑D array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for ``evaluate_sequence`` when sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic shape families
# ----------------------------------------------------------------------


def arcsine_exp_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine shape.
    a_i ∝ ((i+0.5)*(n-0.5-i) + eps)^(-p)
    """
    i = np.arange(n, dtype=float)
    denom = (i + 0.5) * (n - 0.5 - i) + eps
    vals = 1.0 / (denom ** p)
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of the standard arcsine shape and the uniform distribution."""
    arc = arcsine_exp_shape(n, p=0.5, eps=0.0)   # standard arcsine
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Mirror descent (KL divergence) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # active set – indices where conv is (almost) maximal
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"      EG polish it={it:,d}  approx‑score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 43, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises ``evaluate_sequence``.
    The routine respects a 1000 s wall‑clock budget and always returns a
    valid sequence.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 0.  Fallback to a previously discovered sequence, if present
    # ------------------------------------------------------------------
    best_seq: List[float] = []
    best_val = float("inf")
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Search across several support sizes (larger n → potentially lower bound)
    # ------------------------------------------------------------------
    lengths = [65536, 32768, 16384, 8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 15.0:
            break
        print(f"\n=== Exploring length {n} ===")

        # --------------------------------------------------------------
        # 1a. Grid search over the arcsine‑like family (very cheap)
        # --------------------------------------------------------------
        p_vals = np.linspace(0.4, 0.7, 7)          # 0.40 … 0.70
        eps_vals = [0.0, 1e-8, 1e-4, 1e-2, 0.1, 1.0]

        for p in p_vals:
            for eps in eps_vals:
                if time.time() >= deadline - 5.0:
                    break
                seq = arcsine_exp_shape(n, p, eps)
                # enforce exact symmetry (helps the convolution)
                seq = (seq + seq[::-1]) / 2.0
                seq /= seq.sum()
                val = compute_score(seq)
                if val < best_val:
                    best_val, best_seq = val, seq.tolist()
                    print(f"  Grid‑search best: p={p:.3f} eps={eps:g} → {val:.6f}")

        # --------------------------------------------------------------
        # 1b. Upsample the current best vector (if it has a different size)
        # --------------------------------------------------------------
        if best_seq and len(best_seq) != n:
            up_seq = upsample_sequence(np.array(best_seq, dtype=float), n)
            val = compute_score(up_seq)
            if val < best_val:
                best_val, best_seq = val, up_seq.tolist()
                print(f"  Upsampled best to n={n} → {val:.6f}")

        # --------------------------------------------------------------
        # 1c. Randomised search in other analytic families
        # --------------------------------------------------------------
        max_random = 1500                     # tune to remaining time later
        for _ in range(max_random):
            if time.time() >= deadline - 5.0:
                break
            shape_type = rng.choice(["power", "beta", "mix"])
            if shape_type == "power":
                p = rng.uniform(0.05, 5.0)
                eps = rng.uniform(0.0, 3.0)
                seq = generate_power_shape(n, p, eps)
            elif shape_type == "beta":
                a = rng.uniform(0.05, 6.0)
                b = rng.uniform(0.05, 6.0)
                eps = rng.uniform(0.0, max(0.5, 0.02 * n))
                seq = generate_beta_shape(n, a, b, eps)
            else:  # mix
                w = rng.uniform(0.0, 1.0)
                seq = generate_mixture_shape(n, w)

            # enforce symmetry, normalise
            seq = (seq + seq[::-1]) / 2.0
            seq /= seq.sum()
            val = compute_score(seq)
            if val < best_val:
                best_val, best_seq = val, seq.tolist()
                print(f"  Random new best ({shape_type}) → {val:.6f}")

        # --------------------------------------------------------------
        # 1d. Polishing the current best for this length (if time permits)
        # --------------------------------------------------------------
        if best_seq and len(best_seq) == n and time.time() < deadline - 10.0:
            print(f"  Polishing best vector (n={n}) with exponentiated gradient …")
            a_arr = np.array(best_seq, dtype=float)
            a_arr = exponentiated_gradient_polish(
                a_arr,
                deadline - 2.0,
                max_iter=120_000,
                step=0.018,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"  Polished best (n={n}) → {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Final official evaluation (guaranteed non‑empty return)
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a simple uniform distribution of moderate size
        n0 = 1024
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform sequence.")
    else:
        try:
            best_val = evaluate_sequence(best_seq)
        except Exception as e:
            print(f"Final evaluation failed: {e}")

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
