"""
Improved search for a non‑negative step sequence that minimises
``evaluate_sequence``.

The strategy is a more exhaustive parameter sweep combined with
a short continuation‑Adam phase and a final exact sub‑gradient polish.

Main ideas
----------

1. **Rich parametric seed pool** – for each length a variety of
   analytic families (arcsine, β‑Beta, power‑law, mixtures) are
   generated.  α,β,p and mixture‑weights are scanned on fine grids
   and also sampled randomly, giving a large diverse set of candidates.

2. **Fast pre‑selection** – the cheap surrogate ``fast_score`` (identical to the
   official objective when the vector is already normalised) is used to keep the
   best few seeds for a given length.

3. **Continuation Adam on the smooth‑max** – the surrogate
   ``L_β(a)=log‑sum‑exp(β·conv(a,a))`` is minimised while the simplex
   constraint is enforced via a soft‑max parametrisation.  The smoothing
   parameter β is increased up to 5000 (larger than before) to approach the
   true max more closely.

4. **Exact sub‑gradient polishing** – the best vector found so far is refined
   with a projected sub‑gradient method on the true objective until the time
   budget expires.

5. **Optional random local search** – a cheap greedy pairwise mass‑transfer
   heuristic is run if time permits.

All work respects the overall budget (default 1000 s).  The routine always
returns a non‑empty, non‑negative list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Cheap surrogate identical to the official objective (no clipping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(a,a))/sum(a)^2.
    ``a`` must be non‑negative and sum>0 (otherwise returns inf).
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float = 1e-12
) -> np.ndarray:
    """Beta(α,β) distribution on {0,…,n‑1}, normalised to sum‑1."""
    i = np.arange(n, dtype=float)
    # protect against overflow/underflow for extreme exponents
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        left = (i + eps) ** (alpha - 1.0)
        right = (n - 1 - i + eps) ** (beta - 1.0)
        base = left * right
    if not np.isfinite(base).all() or base.sum() == 0.0:
        # fallback to uniform if numeric trouble
        return np.full(n, 1.0 / n)
    return base / base.sum()


def generate_power_shape(
    n: int, p: float, eps: float = 1e-12
) -> np.ndarray:
    """Shape proportional to 1/((i+½)*(n‑½‑i)+eps)^p, normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw (concentration uniformly sampled in log‑space)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)  # roughly 0.01 … 5
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx=1, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean simplex projection (algorithm of Wang & Carreira‑Perpiñán)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Continuation Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max producing a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 5000.0,
    ramp_iters: int = 200_000,
    lr: float = 0.03,
) -> np.ndarray:

    Minimises the smooth‑max  L_β(a)=log‑sum‑exp(β·conv(a,a))
    while staying on the simplex via a soft‑max parametrisation.
    β is linearly increased from ``beta0`` to ``beta_max``.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ∈ simplex
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # Linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # Smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # Gradient of log‑sum‑exp w.r.t. a:
        #   ∂L/∂a = 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # Chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if t % 8000 == 0:
            print(
                f"    Adam it={t:,d} β={beta:.1f} surrogate≈{fast_score(a):.6f}"
            )

    return _softmax(z)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Simple sub‑gradient descent on f(a)=2·n·max(conv(a,a))/sum(a)^2.
    The vector is kept on the simplex via Euclidean projection.
    Step size follows η_t = step0 / sqrt(t).
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution:
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 10000 == 0:
            print(
                f"    Sub‑grad it={it:,d} surrogate≈{fast_score(a):.6f}"
            )

    return a


# ----------------------------------------------------------------------
# Simple greedy random local search (pairwise mass transfers)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.025,
) -> np.ndarray:
    """
    Greedy pairwise mass‑transfer moves.  A move is kept only if it
    strictly improves the fast surrogate.
    """
    a = a0.copy()
    best_score = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * rng.random() * max_transfer
        if eps <= 0.0:
            continue
        a[i] -= eps
        a[j] += eps
        # renormalise (should stay ≈1)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
            continue
        a /= s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            a[j] -= eps
            a[i] += eps
            a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.3  # tiny safety margin

    best_seq: List[float] = []
    best_official: float = float("inf")
    best_surrogate: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, cand
                best_surrogate = fast_score(np.array(cand))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives a better theoretical bound
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(
            f"\n=== Optimising length {n} (time left {deadline - time.time():.1f}s) ==="
        )

        # ------------------------------------------------------------------
        # 1. Build a very diverse pool of analytic seeds
        # ------------------------------------------------------------------
        seeds: List[np.ndarray] = []

        # deterministic families
        seeds.append(arcsine_initial(n))
        seeds.append(np.full(n, 1.0 / n))

        # power‑law family (p close to 0.5 is arcsine, we scan a broader range)
        for p in np.linspace(0.25, 0.85, 13):
            seeds.append(generate_power_shape(n, p, eps=1e-12))

        # symmetric Beta(α,α) – scan a wide log‑range
        for α in np.logspace(-0.4, 0.6, 12):  # ≈0.4 … 4.0
            seeds.append(generate_beta_shape(n, α, α, eps=1e-12))

        # asymmetric Beta – random draws
        for _ in range(10):
            α = 10 ** rng.uniform(-1.0, 1.0)
            β = 10 ** rng.uniform(-1.0, 1.0)
            seeds.append(generate_beta_shape(n, α, β, eps=1e-12))

        # mixtures of arcsine and uniform
        for w in np.linspace(0.2, 0.9, 8):
            mix = w * arcsine_initial(n) + (1.0 - w) * (np.full(n, 1.0 / n))
            mix = mix / mix.sum()
            seeds.append(mix)

        # random Dirichlet draws (different concentrations)
        for _ in range(12):
            seeds.append(generate_random_dirichlet(n, rng))

        # ------------------------------------------------------------------
        # 2. Cheap surrogate ranking – keep only the most promising seeds
        # ------------------------------------------------------------------
        scored: List[Tuple[float, np.ndarray]] = [(fast_score(v), v) for v in seeds]
        scored.sort(key=lambda x: x[0])
        top_k = min(8, len(scored))
        top_seeds = [vec for _, vec in scored[:top_k]]
        print(f"  Top‑{top_k} seeds (surrogate scores):")
        for i, (sc, _) in enumerate(scored[:top_k], 1):
            print(f"    #{i}: {sc:.6f}")

        # ------------------------------------------------------------------
        # 3. Allocate time per seed and run a short Adam refinement
        # ------------------------------------------------------------------
        per_seed_budget = max(2.0, (deadline - time.time()) / (len(top_seeds) * 3.0))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 2.0:
                break
            print(
                f"\n  Seed {idx}/{len(top_seeds)} (n={n}) – budget≈{per_seed_budget:.1f}s"
            )
            seed_deadline = time.time() + per_seed_budget

            # initialise Adam on the simplex via log‑softmax
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=5000.0,
                ramp_iters=200_000,
                lr=0.03,
            )

            # ----------------------------------------------------------------
            # 4. Evaluate the Adam result with the official function
            # ----------------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score after Adam = {val:.6f}")
                if val < best_official:
                    best_official, best_seq = val, a_opt.tolist()
                    best_surrogate = fast_score(a_opt)
                    print("   *** New global best! ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 5. Final polishing with projected sub‑gradient descent (exact objective)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\nFinal polishing with exact sub‑gradient descent …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=400_000,
            step0=0.05,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_official:
                best_official, best_seq = final_val, a_arr.tolist()
                best_surrogate = fast_score(a_arr)
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final evaluation error: {e}")

    # ------------------------------------------------------------------
    # 6. Optional random local search around the polished solution
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:
        print("\nRandom local search around current best …")
        a_start = np.array(best_seq, dtype=float)
        a_refined = random_local_search(
            a_start,
            deadline,
            rng,
            max_transfer=0.025,
        )
        try:
            val_loc = evaluate_sequence(a_refined.tolist())
            print(f"   Local‑search official score = {val_loc:.6f}")
            if val_loc < best_official:
                best_official, best_seq = val_loc, a_refined.tolist()
                best_surrogate = fast_score(a_refined)
                print("*** Best improved by local search ***")
        except Exception as e:
            print(f"   Local‑search evaluation error: {e}")

    # ------------------------------------------------------------------
    # 7. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
