def targeted_hill_climb(seq, time_deadline, max_steps=200000):
    a = np.array(seq)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_frac = 0.1
    for step in range(max_steps):
        if time.time() >= time_deadline:
            break
        # compute conv and max index
        conv = convolution_fft(a, a)
        k = int(conv.argmax())
        # contributions
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # slice indices
        idxs = np.arange(i_start, i_end+1)
        partners = k - idxs
        contrib = a[idxs] * a[partners]
        i_max = idxs[np.argmax(contrib)]

        # choose target index j that does not contribute to k (i.e., j > k)
        # if k == n-1 then choose j < 0? not possible; choose j random not within [i_start,i_end] maybe.
        # We'll select j uniformly from range(n) that satisfies j > k or j < i_start (so outside contributing range)
        possible = list(range(0, i_start)) + list(range(k+1, n))
        if not possible:
            # fallback: random j
            j = np.random.randint(n)
        else:
            j = possible[np.random.randint(len(possible))]

        delta = step_frac * a[i_max]
        if delta <= 0:
            continue
        a[i_max] -= delta
        a[j] += delta

        # project onto simplex to avoid tiny negative rounding issues
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # maybe increase step_frac
            step_frac = min(step_frac * 1.1, 0.5)
        else:
            step_frac *= 0.97
    return best_a, best_score

def generate_beta_distribution(n, alpha):
    # create positions at center of bins
    x = (np.arange(n) + 0.5) / n  # in (0,1)
    weights = (x**(alpha-1)) * ((1-x)**(alpha-1))
    # For alpha=0.5 use sqrt of 1/(x*(1-x))
    # weights may be huge near edges for alpha<0.5; we can clip to avoid extreme values but after simplex projection will handle.
    v = weights
    v = project_to_simplex(v)
    return v

"""
Propose a step‑function (non‑negative list) that minimizes
    2·n·max(conv(a,a)) / (sum(a)**2)

Strategy
--------
1. **Parametric start** – use a symmetric Beta(α,α) density on the
   discrete grid (α≈0.5 gives the classic central‑binomial shape).
   Scan a few α values and keep the best one (fast internal score).

2. **Multi‑restart Adam optimisation** – smooth the max‑operator with a
   log‑sum‑exp surrogate and minimise it by Adam on the probability
   simplex.  Several random perturbations of the best seed are run
   until the time budget is exhausted.

3. **Projected sub‑gradient descent** – switch to the exact max‑objective;
   the sub‑gradient is cheap to compute from the most active convolution
   index.

4. **Targeted hill‑climb** – identify which pairs of indices contribute
   most to the maximal convolution entry and move a small fraction of
   mass from those positions to indices that do **not** influence that
   entry.  This last polishing step often yields the final improvement.

All stages respect a global time deadline (≈budget_s – a small safety
margin) and keep the best vector seen so far.  The returned value is a
plain list of floats; `evaluate_sequence` will perform the final
clipping/summing when it is called by the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    a, b : 1‑D arrays of length n, m
    Returns an array of length n+m-1.
    """
    n = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(n)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:n]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already lies on the simplex.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    if np.all(v <= 0):
        # trivial case – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric Beta‑distribution initialisation
# ----------------------------------------------------------------------


def beta_initialisation(n: int, alpha: float) -> np.ndarray:
    """
    Return a symmetric Beta(α,α) density sampled on the mid‑points
    of n equal sub‑intervals of (0,1) and projected onto the simplex.
    """
    # mid‑point grid avoids the singularities at 0 and 1
    x = (np.arange(n) + 0.5) / n          # in (0,1)
    w = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)   # protect against NaNs for tiny α
    w = project_to_simplex(w)
    return w


# ----------------------------------------------------------------------
# Smoothed max optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 400_000,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best vector found and its exact score.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)   # first‑moment
    v = np.zeros_like(a)   # second‑moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ----- anneal smoothing ----------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- convolution & soft‑max ----------
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # safe exponential (conv ≤ maxc, so exponent ≤ 0)
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # ----- gradient of the smooth objective ----------
        a_rev = a[::-1]
        # a_rev (len n) * w (len 2n‑1) -> length 3n‑2, slice the central n entries
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # ----- Adam update ----------
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # ----- keep track of the best ----------
        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑objective (projected sub‑gradient)
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 800_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional small jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (mass transfer from the offending pairs)
# ----------------------------------------------------------------------


def targeted_hill_climb(
    start_seq,
    time_deadline,
    max_steps: int = 600_000,
) -> tuple[np.ndarray, float]:
    """
    Small stochastic moves that actively reduce the current
    maximal convolution entry.  Mass is taken from indices that
    participate heavily in that entry and moved to a location that
    does **not** contribute to it.
    """
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # fraction of the chosen donor mass to move each step
    move_frac = 0.10

    while max_steps > 0 and time.time() < time_deadline:
        conv = convolution_fft(a, a)
        k = int(conv.argmax())

        # indices that actually contribute to the maximal entry
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1)
        partners = k - idxs
        contrib = a[idxs] * a[partners]

        if contrib.sum() == 0.0:
            # fallback to a random move
            i = np.random.randint(n)
        else:
            i = idxs[np.argmax(contrib)]

        # pick a destination index that does NOT affect conv(k)
        # i.e. either before i_start or after k
        if k < n - 1:
            dest_candidates = list(range(k + 1, n))
        else:
            dest_candidates = list(range(0, i_start))
        if not dest_candidates:
            dest = np.random.randint(n)
        else:
            dest = np.random.choice(dest_candidates)

        delta = move_frac * a[i]
        if delta > 0:
            a[i] -= delta
            a[dest] += delta

        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            move_frac = min(move_frac * 1.08, 0.5)   # become bolder when we improve
        else:
            move_frac *= 0.97                       # become more cautious otherwise

        max_steps -= 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds as safety margin for the harness
    deadline = start_time + budget_s - 2.5

    # -----------------------------------------------------------------
    # 0. Choose a reasonable length (2048 works well for the time budget)
    # -----------------------------------------------------------------
    n = 2048

    # -----------------------------------------------------------------
    # 1. Scan a few Beta‑law shapes (α around 0.5) – cheap initialisation
    # -----------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    alphas = np.linspace(0.35, 0.70, 8)          # a small grid around the arcsine law
    for alpha in alphas:
        cand = beta_initialisation(n, alpha)
        val = compute_score(cand)               # fast exact score (sum = 1)
        if val < best_score:
            best_score = val
            best_seq = cand
    print(f"[init] best beta‑shape α={alphas[np.argmin([compute_score(beta_initialisation(n, a)) for a in alphas])]:.3f} → score {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Multi‑restart Adam optimisation (smooth surrogate)
    # -----------------------------------------------------------------
    while time.time() < deadline - 30.0:  # keep ~30 s for polishing later
        # perturb the current best a little to obtain a fresh start
        pert = best_seq + 0.02 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert = project_to_simplex(pert)

        # leave at most 8 min for one Adam run
        run_deadline = min(deadline - 30.0, time.time() + 480.0)

        a_opt, scr = smooth_gradient_optimize(
            pert,
            time_deadline=run_deadline,
            max_iters=500_000,
            beta_start=5.0,
            beta_end=500.0,
            step0=1.0,
        )
        if scr < best_score:
            best_score, best_seq = scr, a_opt
            print(f"[adam] improved → score {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (exact max‑objective)
    # -----------------------------------------------------------------
    if time.time() < deadline - 20.0:
        sub_deadline = deadline - 20.0
        refined_a, refined_score = subgrad_optimize(
            best_seq,
            time_deadline=sub_deadline,
            max_iter=800_000,
            step0=0.07,
        )
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_a
            print(f"[subgrad] improved → score {best_score:.6f}")

    # -----------------------------------------------------------------
    # 4. Final targeted hill‑climb
    # -----------------------------------------------------------------
    if time.time() < deadline - 5.0:
        hill_deadline = deadline - 5.0
        polished_a, polished_score = targeted_hill_climb(
            best_seq,
            time_deadline=hill_deadline,
            max_steps=800_000,
        )
        if polished_score < best_score:
            best_score, best_seq = polished_score, polished_a
            print(f"[hill‑climb] improved → score {best_score:.6f}")

    # -----------------------------------------------------------------
    # 5. Safeguard – fallback to a tiny uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_score(best_seq)

    # Final verification with the harness' evaluation routine
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    return best_seq.tolist()
