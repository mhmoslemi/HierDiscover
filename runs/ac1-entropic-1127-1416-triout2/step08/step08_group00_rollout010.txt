"""
Propose a step‑function (non‑negative list) that minimises the
`evaluate_sequence` objective.

Algorithm (≈1000 s budget)
---------------------------
0. **Fallback** – If a previously discovered sequence `height_sequence_1`
   exists, evaluate it first.

1. **Problem size** – Choose `n` (2048 by default, 4096 if time permits)
   and work exclusively on the probability simplex (∑ a_i = 1).

2. **Random restarts + smooth surrogate** –  
   Repeatedly initialise a random probability vector (Dirichlet α∈{0.1,0.5,1,2,5})
   and run a few seconds of Adam‐optimised log‑sum‑exp (smooth‑max) 
   minimisation.  The surrogate becomes very sharp (β≈2000) so that the
   true max is almost attained.

3. **Heavy sub‑gradient refinement** –  
   From the best vectors obtained above run a long projected sub‑gradient
   descent on the *exact* max‑convolution (the true objective).  An
   optional momentum term helps to escape shallow plateaux.

4. **Stochastic annealed hill‑climb** –  
   A final local search that repeatedly moves a small amount of mass from
   one index to another (or swaps two indices).  Moves are accepted if they
   improve the exact score, otherwise with a temperature‑dependent probability.
   The step‑size automatically adapts.

All phases respect the global wall‑clock deadline; the routine finally
returns the best list of floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (must be top‑level – no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (fast for large arrays).
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (∑ a_i = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth‑max optimisation (log‑sum‑exp surrogate) with Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 400_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑accelerated minimisation of a soft‑max surrogate of max(conv(a,a)).
    Returns the best exact score seen during the run.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- anneal β ----------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- forward pass -------------------------------------------------
        conv = convolution_fft(a, a)                 # length 2n‑1
        maxc = conv.max()
        # numerically stable log‑sum‑exp weights
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # ---- gradient of the smoothed max (cross‑correlation) ----------
        # grad = 2 * (a_rev * w) (restricted to the central n entries)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # ---- Adam update -------------------------------------------------
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.07,
    momentum: bool = True,
    momentum_coeff: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected (optionally momentum‑accelerated) sub‑gradient descent on
    2·n·max(conv(a,a)).  Returns the best exact score encountered.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    if momentum:
        vel = np.zeros_like(a)   # velocity for momentum

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        # all indices attaining the maximum (for a more stable direction)
        max_val = conv.max()
        max_idxs = np.flatnonzero(conv >= max_val - 1e-14)

        # sub‑gradient: for each i sum a_j over all j with i+j in max_idxs
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in max_idxs:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += a[j[mask]]

        g = 2.0 * g

        # step size schedule
        step = step0 / np.sqrt(t)

        if momentum:
            vel = momentum_coeff * vel + (1.0 - momentum_coeff) * g
            direction = vel
        else:
            direction = g

        a_new = a - step * direction
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # small random jitter every 2000 iterations to avoid stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score

# 3. Simulated‑annealing hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    deadline: float,
    max_steps: int = 2_500_000,
    init_temp: float = 1e-3,
    final_temp: float = 1e-6,
) -> tuple[list[float], float]:
    """
    Local stochastic moves (swap or mass transfer) with a simple
    exponential temperature schedule.  Only moves that improve the exact
    score are always accepted; worsening moves are accepted with
    probability exp(-Δ / T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # geometric temperature schedule
    temp = init_temp
    decay = (final_temp / init_temp) ** (1.0 / max_steps)

    step_scale = 0.07  # fraction of the source mass to move in a transfer

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # small transfer from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        Δ = cur_score - best_score
        if Δ < 0 or np.random.rand() < np.exp(-Δ / max(temp, 1e-12)):
            # accept
            if cur_score < best_score:
                best_score = cur_score
                best_a = cand
                # if we made a good move, allow a slightly bolder step
                step_scale = min(step_scale * 1.05, 0.5)
            else:
                # accept worsening move – shrink step size a little
                step_scale *= 0.97

        # temperature decay
        temp *= decay

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Helper – generate random starts on the simplex (different conc. α)
# ----------------------------------------------------------------------


def random_start(n: int, alpha: float) -> np.ndarray:
    """
    Dirichlet(α) sample of length `n`; α controls sparsity
    (α<<1 → sparse, α>>1 → dense).
    """
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.ones_like(x)
    return x / x.sum()


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a short safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose sequence length based on remaining time
    # ------------------------------------------------------------------
    n = 2048
    # If we have a lot of time, move to a larger problem (more freedom)
    if deadline - time.time() > 800:
        n = 4096
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2. Main search loop – several phases
    # ------------------------------------------------------------------
    # Reserve some time for the final polishing
    reserve_polish = 120.0

    # Phase‑A: many short smooth‑max runs with diverse Dirichlet alphas
    alphas = [0.1, 0.3, 0.7, 1.0, 2.0, 5.0]
    while time.time() < deadline - reserve_polish - 20.0:
        # allocate a time slice for this restart (5‑30 s)
        remaining = deadline - time.time() - reserve_polish - 20.0
        slice_time = max(5.0, min(25.0, remaining * 0.4))
        slice_deadline = time.time() + slice_time

        # pick a random concentration parameter
        alpha = np.random.choice(alphas)
        a0 = random_start(n, alpha)

        # smooth optimisation
        a_opt, _ = smooth_gradient_optimize(
            a0,
            deadline=slice_deadline,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=2000.0,
            step0=1.0,
        )
        exact = compute_score(a_opt)
        if exact < best_val:
            best_val = exact
            best_seq = a_opt.tolist()
            print(f"[smooth] New best = {best_val:.6f}")

    # Phase‑B: heavy sub‑gradient refinement(s) on the current best
    if best_seq is not None and time.time() < deadline - reserve_polish - 10.0:
        print("\nStarting sub‑gradient refinement")
        # We may run several independent refinements from perturbed starts
        n_refine = 3
        for r in range(n_refine):
            # perturb the current best slightly before refinement
            pert = np.array(best_seq, copy=True)
            jitter = np.random.randn(n) * 0.02
            pert += jitter
            pert = project_to_simplex(pert)

            ref_deadline = deadline - reserve_polish - 5.0
            refined_a, refined_score = subgrad_optimize(
                pert,
                deadline=ref_deadline,
                max_iter=500_000,
                step0=0.07,
                momentum=True,
                momentum_coeff=0.85,
            )
            if refined_score < best_val:
                best_val = refined_score
                best_seq = refined_a.tolist()
                print(f"[sub‑grad] Improvement #{r+1}: {best_val:.6f}")

    # Phase‑C: final stochastic hill‑climb polishing
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nHill‑climb polishing")
        polished_seq, polished_score = hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=2_500_000,
            init_temp=1e-3,
            final_temp=1e-6,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"[hill‑climb] New best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback to a simple uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will call `evaluate_sequence` again)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
