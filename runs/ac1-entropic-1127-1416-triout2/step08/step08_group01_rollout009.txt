"""
Improved search for low‑max convolution step‑function.
The algorithm:
1. Choose length n based on remaining time (up to 16384).
2. Build a rich set of analytic initial vectors:
   - two‑parameter power‑law a_i ∝ (i+½)^{-αL}·(n-½-i)^{-αR}.
   - symmetric α‑grid (αL=αR) and a few asymmetric variants.
3. Run several short Adam‑based smooth‑max optimisations from
   random sparse starts and from the best analytic vectors.
   The smooth objective is the log‑sum‑exp of the convolution,
   with an exponential β‑schedule up to β≈800.
4. Run a long dedicated Adam optimisation from the current best
   (≈3·10⁵ iterations).
5. Refine with a projected Euclidean sub‑gradient method.
6. Finish with a fast stochastic hill‑climb that only accepts
   improving moves.
All steps respect the 1000 s wall‑clock limit and keep the best
sequence seen so far.  The returned object is a plain list of
floats suitable for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    This matches the value returned by `evaluate_sequence` for the
    same input (the security‐checks in evaluate_sequence only clip
    negative entries and enforce a tiny lower bound on the sum).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families of starting vectors
# ----------------------------------------------------------------------


def power_law_start(n: int, alpha_left: float, alpha_right: float) -> np.ndarray:
    """
    Return a_i ∝ (i+½)^{-α_L}·(n‑½‑i)^{-α_R}  on the simplex.
    """
    i = np.arange(n, dtype=float)
    # to avoid zero at the ends we add 0.5 in the denominator
    left = np.log(i + 0.5)
    right = np.log(n - 0.5 - i)
    log_w = -alpha_left * left - alpha_right * right
    w = np.exp(log_w)
    # numerical safeguard: replace infinities
    w[~np.isfinite(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_power_law_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a modest grid of (α_L,α_R) pairs (including the symmetric line)
    and keep the best vector.
    """
    best_a = None
    best_score = float("inf")
    # sensible ranges: α ∈ [-1, 3]; use a coarse grid
    alphas = np.concatenate([np.linspace(-1.0, 0.0, 7),
                             np.linspace(0.0, 2.0, 13),
                             np.linspace(2.0, 3.0, 5)])
    for al in alphas:
        # symmetric case
        a_sym = power_law_start(n, al, al)
        sc = compute_score(a_sym)
        if sc < best_score:
            best_score = sc
            best_a = a_sym.copy()
        # a few asymmetric offsets
        for delta in (-0.5, 0.5):
            a_asym = power_law_start(n, al + delta, al - delta)
            sc = compute_score(a_asym)
            if sc < best_score:
                best_score = sc
                best_a = a_asym.copy()
        if time.time() >= deadline:
            break
    return best_a, best_score


def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    sparsity = np.random.uniform(0.03, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a  # already on the simplex


# ----------------------------------------------------------------------
# Smooth‑max optimiser – Adam on the log‑softmax parametrisation
# ----------------------------------------------------------------------


def smooth_max_adam_optimize(start_a: np.ndarray,
                             deadline: float,
                             max_iters: int = 300_000,
                             beta_start: float = 0.5,
                             beta_end: float = 800.0,
                             lr0: float = 0.8,
                             eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth log‑sum‑exp approximation of the max‑convolution
    using Adam on the log‑softmax parametrisation of a.
    The iterate is kept on the simplex automatically.
    Returns the best exact score seen.
    """
    n = start_a.shape[0]
    # initialise log‑probabilities (stable)
    Z = np.log(np.maximum(start_a, eps))
    m = np.zeros_like(Z)
    v = np.zeros_like(Z)
    beta1 = 0.9
    beta2 = 0.999
    best_a = softmax(Z)
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- β schedule (exponential) ----------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---------- current a ----------
        a = softmax(Z)

        # ---------- convolution ----------
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max over convolution entries (stabilised)
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # ---------- gradient w.r.t. a ----------
        # grad_a = 2 * (reverse(a) ⊛ w) restricted to length n
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ---------- gradient w.r.t. Z (softmax Jacobian) ----------
        ga_dot_a = np.dot(grad_a, a)
        grad_z = a * (grad_a - ga_dot_a)

        # ---------- Adam update ----------
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        Z -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # ---------- occasional exact evaluation ----------
        if it % 250 == 0:
            a_eval = softmax(Z)
            sc = compute_score(a_eval)
            if sc < best_sc:
                best_sc = sc
                best_a = a_eval.copy()
                # keep the log‑vector that produced it
                # (already in Z)

    # final check
    a_final = softmax(Z)
    sc = compute_score(a_final)
    if sc < best_sc:
        best_sc = sc
        best_a = a_final.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 600_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient method for the exact objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of max‑entry (sub‑gradient) is 2 * (a shifted)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Fast hill‑climb with cheap incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb that only accepts moves that improve
    the exact score.  The convolution is recomputed from scratch for
    simplicity – the number of performed steps is limited so the total
    wall‑clock stay within the budget.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12
    rng = np.random.default_rng()

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if rng.random() < 0.5:
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a, best_sc

        n = 8192
    elif remaining > 250:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback from global variable
    # -------------------------------------------------
    try:

    except Exception:

    # -------------------------------------------------

    if a_start is not None and sc_start < best_score:

    # -------------------------------------------------

    while time.time() < deadline - 250.0:

            beta_end=500.0,

        if sc_opt < best_score:

            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after random Adam = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Long dedicated Adam optimisation from current best
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] long Adam optimisation")
        a_long, sc_long = smooth_max_adam_optimize(

            print(f"[{time.time():.1f}] after long Adam best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Sub‑gradient refinement on the exact max‑objective
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(

    # -------------------------------------------------

    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:

```assistantfinalBelow is a self‑contained implementation of `propose_candidate`.  
It expands on the previous approach by  

* using a **two‑parameter power‑law** family for the analytic start,  
* optimising the **smooth log‑sum‑exp surrogate** with an **Adam** optimiser on the log‑softmax parametrisation (momentum → faster convergence),  
* applying a **more aggressive β‑schedule** (up to β≈800),  
* keeping the best exact score after every few iterations, and  
* finishing with a projected sub‑gradient refinement followed by a stochastic hill‑climb.

```python
"""
Improved search for low‑max convolution step‑function.
The algorithm:
1. Choose length n based on remaining time (up to 16384).
2. Build a rich set of analytic initial vectors:
   - two‑parameter power‑law a_i ∝ (i+½)^{-αL}·(n‑½‑i)^{-αR}.
   - symmetric α‑grid (αL=αR) and a few asymmetric variants.
3. Run several short Adam‑based smooth‑max optimisations from
   random sparse starts and from the best analytic vectors.
   The smooth objective is the log‑sum‑exp of the convolution,
   with an exponential β‑schedule up to β≈800.
4. Run a long dedicated Adam optimisation from the current best
   (≈3·10⁵ iterations).
5. Refine with a projected Euclidean sub‑gradient method.
6. Finish with a fast stochastic hill‑climb that only accepts
   improving moves.
All steps respect the 1000 s wall‑clock limit and keep the best
sequence seen so far.  The returned object is a plain list of
floats suitable for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    This matches the value returned by `evaluate_sequence` for the
    same input (the security‑checks in evaluate_sequence only clip
    negative entries and enforce a tiny lower bound on the sum).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families of starting vectors
# ----------------------------------------------------------------------


def power_law_start(n: int, alpha_left: float, alpha_right: float) -> np.ndarray:
    """
    Return a_i ∝ (i+½)^{-α_L}·(n‑½‑i)^{-α_R}  on the simplex.
    """
    i = np.arange(n, dtype=float)
    left = np.log(i + 0.5)
    right = np.log(n - 0.5 - i)
    log_w = -alpha_left * left - alpha_right * right
    w = np.exp(log_w)
    w[~np.isfinite(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_power_law_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a modest grid of (α_L,α_R) pairs (including the symmetric line)
    and keep the best vector.
    """
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([np.linspace(-1.0, 0.0, 7),
                             np.linspace(0.0, 2.0, 13),
                             np.linspace(2.0, 3.0, 5)])
    for al in alphas:
        # symmetric case
        a_sym = power_law_start(n, al, al)
        sc = compute_score(a_sym)
        if sc < best_score:
            best_score = sc
            best_a = a_sym.copy()
        # a few asymmetric offsets
        for d in (-0.5, 0.5):
            a_asym = power_law_start(n, al + d, al - d)
            sc = compute_score(a_asym)
            if sc < best_score:
                best_score = sc
                best_a = a_asym.copy()
        if time.time() >= deadline:
            break
    return best_a, best_score


def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    sparsity = np.random.uniform(0.03, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a  # already on the simplex


# ----------------------------------------------------------------------
# Smooth‑max optimiser – Adam on the log‑softmax parametrisation
# ----------------------------------------------------------------------


def smooth_max_adam_optimize(start_a: np.ndarray,
                             deadline: float,
                             max_iters: int = 300_000,
                             beta_start: float = 0.5,
                             beta_end: float = 800.0,
                             lr0: float = 0.8,
                             eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth log‑sum‑exp approximation of the max‑convolution
    using Adam on the log‑softmax parametrisation of a.
    The iterate stays on the simplex automatically.
    Returns the best exact score seen.
    """
    n = start_a.shape[0]
    Z = np.log(np.maximum(start_a, eps))
    m = np.zeros_like(Z)
    v = np.zeros_like(Z)
    beta1, beta2 = 0.9, 0.999

    best_a = softmax(Z)
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(Z)

        # convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient wrt a (smooth max)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # gradient wrt Z (softmax Jacobian)
        ga_dot_a = np.dot(grad_a, a)
        grad_z = a * (grad_a - ga_dot_a)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        Z -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if it % 250 == 0:
            a_eval = softmax(Z)
            sc = compute_score(a_eval)
            if sc < best_sc:
                best_sc = sc
                best_a = a_eval.copy()

    a_final = softmax(Z)
    sc = compute_score(a_final)
    if sc < best_sc:
        best_sc = sc
        best_a = a_final.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 600_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient method for the exact objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Fast hill‑climb
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb that only accepts moves that improve
    the exact score.  The convolution is recomputed from scratch for
    simplicity – the number of performed steps is limited so the total
    wall‑clock stay within the budget.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12
    rng = np.random.default_rng()

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if rng.random() < 0.5:          # swap two entries
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # transfer a small packet
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point for the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising `evaluate_sequence`.
    The routine follows a multi‑stage pipeline based on
    a two‑parameter analytic start, Adam‑optimisation of a
    smoothed max‑convolution, a sub‑gradient refinement
    and a final hill‑climb.
    The best vector found is returned as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # -------------------------------------------------
    # Choose length according to remaining budget
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 16384
    elif remaining > 500:
        n = 8192
    elif remaining > 250:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback from a global variable
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytic start (two‑parameter power‑law)
    # -------------------------------------------------
    a_start, sc_start = find_best_power_law_start(n, deadline - 30.0)
    if a_start is not None and sc_start < best_score:
        best_score = sc_start
        best_seq = a_start.tolist()
        print(f"[{time.time():.1f}] best after analytic start = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Random sparse restarts with Adam smoothing
    # -------------------------------------------------
    while time.time() < deadline - 250.0:
        a0 = random_sparse_start(n)
        a_opt, sc_opt = smooth_max_adam_optimize(
            a0,
            deadline - 180.0,
            max_iters=150_000,
            beta_start=0.5,
            beta_end=500.0,
            lr0=0.9,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after random Adam = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Long dedicated Adam optimisation from current best
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] long Adam optimisation")
        a_long, sc_long = smooth_max_adam_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=350_000,
            beta_start=0.5,
            beta_end=800.0,
            lr0=0.8,
        )
        if sc_long < best_score:
            best_score = sc_long
            best_seq = a_long.tolist()
            print(f"[{time.time():.1f}] after long Adam best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Sub‑gradient refinement on the exact max‑objective
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 40.0,
            max_iters=600_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Final hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: fall back to a short uniform vector if everything failed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
```
