def annealing_hill_climb(start_a, deadline, max_steps=500_000):
    a = start_a.copy()
    n = a.size
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score
    # temperature schedule: start_T = 0.01 * cur_score? maybe.
    # We can schedule T decays exponentially.
    start_T = 0.01 * cur_score   # 1% of score
    end_T = 1e-6
    total_steps = max_steps
    step = 0
    while time.time() < deadline and step < max_steps:
        # propose move
        cand = a.copy()
        # choose move type
        if np.random.rand() < 0.5:
            i,j = np.random.randint(n, size=2)
            # swap small mass: propose small delta
            delta = np.random.rand() * 0.05 * cand[i]   # 5% of i mass
            cand[i] -= delta
            cand[j] += delta
        else:
            i, j = np.random.randint(n, size=2)
            # random exchange
            delta = np.random.rand() * 0.02 * (cand[i] + cand[j])
            cand[i] = max(cand[i] - delta, 0)
            cand[j] += delta
        cand = project_to_simplex(cand)   # maintain sum=1
        cand_score = compute_score(cand)
        # acceptance
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cand.copy()
        else:
            # compute temperature
            progress = step / total_steps
            T = start_T * ((end_T/start_T) ** progress)   # exponential decay
            if np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = cand
                cur_score = cand_score
        step += 1
    return best_a, best_score

"""
A multi‑stage optimizer for the bound

    score(a) = 2·n·max(a∗a) / (Σ a)²

where a is a non‑negative step‑function (list of heights).  
The routine respects a 1000 s wall‑clock budget, works entirely with
NumPy (no external LP or CVX solvers) and combines several complementary
heuristics:

1️⃣  Analytic one‑parameter families (α‑family, power‑law) – quick grid
    search to obtain a good initial shape.

2️⃣  Exponentiated‑gradient on a smooth log‑sum‑exp approximation of the
    max‑convolution.  β (temperature) is annealed exponentially and the
    KL‑mirror step keeps the iterate on the probability simplex.

3️⃣  A finer exponentiated‑gradient run with a much larger final β
    (≈1500) to sharpen the approximation of the true max.

4️⃣  Projected sub‑gradient descent on the exact max‑objective.  Instead
    of using a single active index we average the sub‑gradient over all
    convolution entries that are within a small tolerance of the
    maximum, which yields a more stable descent direction.

5️⃣  Simulated‑annealing hill‑climb that occasionally accepts worsening
    moves, allowing the search to escape the basin of the last local
    optimum.

Each stage records the best exact score seen (computed with the
same‑scale `compute_score`).  The whole procedure adapts the problem
size to the remaining time (up to 16384 coefficients) and finally
returns a plain Python list of floats.  If everything fails a uniform
256‑vector is returned as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) for a probability vector (Σa = 1).
    This equals the value returned by `evaluate_sequence` when the input
    already satisfies the sum‑normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector (few non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a  # already on the simplex


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray | None, float]:
    """Grid‑search over α and keep the best vector."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([np.linspace(-1.0, 0.0, 11),
                             np.linspace(0.0, 2.5, 26)])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def powerlaw_start(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p), normalised."""
    i = np.arange(n, dtype=float) + 1.0
    w = i ** (-p)
    a = w / w.sum()
    return a


def find_best_powerlaw_start(n: int, deadline: float) -> tuple[np.ndarray | None, float]:
    """Grid‑search over the power‑law exponent."""
    best_a = None
    best_score = float("inf")
    # include a few negative exponents to allow end‑heavy shapes
    ps = np.concatenate([np.linspace(-0.5, 0.0, 6),
                         np.linspace(0.0, 2.5, 26)])
    for p in ps:
        if time.time() >= deadline:
            break
        a = powerlaw_start(n, p)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (smooth max)
# ----------------------------------------------------------------------


def exp_grad_optimize(start_a: np.ndarray,
                      deadline: float,
                      max_iters: int = 300_000,
                      beta_start: float = 0.5,
                      beta_end: float = 300.0,
                      eta0: float = 0.6,
                      eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective

        f_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

    using a KL‑mirror (exponentiated‑gradient) step.  β is exponentially
    annealed from `beta_start` to `beta_end`.  The best exact score seen
    during the run is returned together with the corresponding vector.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β‑schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed max (log‑sum‑exp)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))            # stabilised
        w = expw / expw.sum()                         # softmax weights

        # gradient of f_β (KL‑mirror uses the same w)
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)         # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]       # slice to length n

        # step size (η_t = η₀/√t)
        eta = eta0 / np.sqrt(it)

        # KL‑mirror / exponentiated‑gradient update
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (exact max, averaged over near‑active indices)
# ----------------------------------------------------------------------


def subgrad_optimize_average(start_a: np.ndarray,
                             deadline: float,
                             max_iters: int = 500_000,
                             step0: float = 0.07,
                             tol_frac: float = 0.002) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        2·n·max(a∗a) .

    The sub‑gradient is averaged over all convolution entries whose value
    lies within `tol_frac` of the current maximum, providing a smoother
    descent direction than a single argmax.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # indices that are “almost active”
        active = np.where(conv >= maxc * (1.0 - tol_frac))[0]
        if active.size == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size

        # gradient of the (non‑smooth) max: average of the gradients at active spots
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * grad)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (polishing)
# ----------------------------------------------------------------------


def annealing_hill_climb(start_a: np.ndarray,
                         deadline: float,
                         max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Random walk that occasionally accepts worsening moves.
    Moves are either a small transfer of mass between two coordinates
    or a plain swap.  Temperature decays exponentially.
    """
    a = start_a.copy()
    n = a.size
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc

    # temperature schedule (geometric decay)
    start_T = best_sc * 0.008   # about 0.8 % of the current score
    end_T = best_sc * 1e-6
    for step in range(max_steps):
        if time.time() >= deadline:
            break

        # propose a candidate
        cand = a.copy()
        if np.random.rand() < 0.5:
            # transfer a random fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if a[i] > 0:
                frac = np.random.uniform(0.001, 0.15)   # up to 15 % of the source mass
                delta = frac * a[i]
                cand[i] -= delta
                cand[j] += delta
        else:
            # swap two entries (useful for symmetry breaking)
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        cand = project_to_simplex(cand)
        cand_sc = compute_score(cand)

        # acceptance test
        if cand_sc < cur_sc:
            a = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = cand.copy()
        else:
            # temperature based Metropolis rule
            progress = step / max_steps
            T = start_T * ((end_T / start_T) ** progress)
            if np.random.rand() < np.exp(-(cand_sc - cur_sc) / max(T, 1e-12)):
                a = cand
                cur_sc = cand_sc

    return best_a, best_sc
