"""
Improved search for a low‑score step‑function.

The algorithm follows a multi‑stage scheme designed to explore the
search space more aggressively than the baseline:

1. **Length selection** – a large even length (8192 or 16384) is used
   when time permits; otherwise a shorter length reduces the cost of
   each iteration.

2. **Diverse deterministic starts** – besides the classic
   α‑family we also try a β‑family (a discrete Beta distribution) and a
   random‑sparse start. The best of these is kept.

3. **Adam‑based smoothed optimisation** – we minimise the
   log‑sum‑exp smooth approximation of the max‑convolution using an
   exponentiated‑gradient update with Adam‑style moments.  This gives
   faster, more stable convergence than the simple decaying stepsize.

4. **Scale‑up** – when enough time is left we up‑sample the best vector
   to a larger length (doubling the number of points) and repeat the
   Adam optimisation for a short budget.  The finer resolution lets
   the auto‑correlation peak be reduced further.

5. **Momentum sub‑gradient descent** – a classic projected sub‑gradient
   method with a heavy‑ball momentum term is applied to the exact
   max‑objective to polish the solution.

6. **Simulated‑annealing hill‑climb** – a stochastic walk that
   occasionally accepts worse moves (temperature decreasing) helps the
   search escape shallow local minima.

All stages respect the global 1000 s wall‑clock limit.  The routine
returns a plain Python list of non‑negative floats (already normalised
to sum = 1, which makes the final `evaluate_sequence` call exact).

The implementation makes heavy use of FFT‑based convolution, which
runs in O(n log n) per iteration and is fast enough for the chosen
vector sizes.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    This equals the value returned by `evaluate_sequence` when the
    sequence already satisfies the normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Families of deterministic starting points

def alpha_start(n: int, alpha: float) -> np.ndarray:
    """a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ   (the classic family)."""
    i = np.arange(n, dtype=float)
    # log of the product (i+0.5)*(n-0.5-i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Guard against overflow / underflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def beta_start(n: int, p: float, q: float) -> np.ndarray:
    """Discrete Beta distribution: a_i ∝ (i+1)^{p-1} (n-i)^{q-1}."""
    i = np.arange(1, n + 1, dtype=float)          # 1 … n
    w = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def random_sparse_start(n: int, min_nonzero: int = 2) -> np.ndarray:
    """Uniform random on a sparse support."""
    sparsity = np.random.uniform(0.02, 0.3)      # 2‑30 % non‑zeros
    k = max(min_nonzero, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def best_initial_vector(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a handful of simple families (α‑grid, β‑grid, random sparse)
    and return the best vector found before the given deadline.
    """
    best_a = None
    best_score = np.inf

    # ----- α‑grid ----------------------------------------------------
    alphas = np.concatenate([np.linspace(-1.0, 0.0, 9),
                             np.linspace(0.0, 2.5, 26)])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # ----- β‑grid (Beta distribution) --------------------------------
    p_vals = np.linspace(0.1, 5.0, 12)
    q_vals = np.linspace(0.1, 5.0, 12)
    for p in p_vals:
        for q in q_vals:
            if time.time() >= deadline:
                break
            a = beta_start(n, p, q)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # ----- Random sparse restarts ------------------------------------
    while time.time() < deadline - 1.0:      # keep a few seconds free
        a = random_sparse_start(n)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        # limit the number of random restarts to avoid endless loops
        if np.random.rand() < 0.01:
            break

    return best_a, best_score

# Adam‑based optimisation of the smooth log‑sum‑exp max
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 0.5,
                         beta_end: float = 300.0,
                         alpha0: float = 0.7,
                         eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective
        f_β(a) = (1/β) log Σ_k exp(β·(a*a)_k)
    using an exponentiated‑gradient step with Adam‑style moments.
    Returns the best exact score (computed intermittently) and the
    corresponding vector.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- β schedule (exponential) ----------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---------- smoothed soft‑max weights ----------
        conv = convolution_fft(a, a)            # length 2n‑1
        maxc = conv.max()
        # Stabilised exponentials
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # ---------- gradient of f_β ----------
        a_rev = a[::-1]
        # convolution of reversed a with the weight vector w
        grad_full = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_full
        v = beta2 * v + (1.0 - beta2) * (grad_full ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Exponentiated‑gradient step with per‑coordinate learning rates
        step = alpha0 / np.sqrt(it)            # base step shrinkage
        a = np.maximum(a * np.exp(-step * m_hat / (np.sqrt(v_hat) + eps)), eps)

        # Normalise back onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ---------- occasional exact evaluation ----------
        if it % 400 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # Final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with heavy‑ball momentum on the exact max
# ----------------------------------------------------------------------


def momentum_subgrad_optimize(start_a: np.ndarray,
                             deadline: float,
                             max_iters: int = 500_000,
                             step0: float = 0.08,
                             momentum: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the true objective
    2·n·max(a*a) with a heavy‑ball momentum term.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    vel = np.zeros_like(a)   # momentum buffer

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of max at the active index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)

        # Heavy‑ball momentum update
        vel = momentum * vel - step * g
        a = a + vel
        a = project_to_simplex(a)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (local polishing)
# ----------------------------------------------------------------------


def annealed_hill_climb(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 800_000,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Random walk that occasionally accepts worsening moves according to
    a linearly cooling temperature schedule.  Only moves that keep the
    sequence on the simplex are considered (the simplex projection is
    applied after each tentative move).
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # Pre‑compute a linear cooling schedule
    temps = np.linspace(init_temp, final_temp, max_steps)

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        T = temps[step]

        # Propose a move: either swap two entries or shift small mass
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]                 # up to 5 % transfer
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        # Accept if better; otherwise accept with prob exp(-(Δ)/T)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
        else:
            # Δ = sc - best_sc (positive)
            if np.random.rand() < np.exp(-(sc - best_sc) / (T + 1e-12)):
                best_sc = sc
                best_a = cand

    return best_a, best_sc


# ----------------------------------------------------------------------
# Helper to up‑sample a vector (doubling length while preserving shape)
# ----------------------------------------------------------------------


def upsample_vector(a: np.ndarray) -> np.ndarray:
    """Linear interpolation to twice the length, keeping sum=1."""
    n = a.size
    # Interpolate between consecutive points, then duplicate the last point
    x = np.arange(n)
    x_new = np.linspace(0, n - 1, 2 * n)
    a_new = np.interp(x_new, x, a)
    a_new /= a_new.sum()          # re‑normalise
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The implementation follows the multi‑stage pipeline described in
    the module docstring and respects the 1000 s wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # -------------------------------------------------
    # Choose a length adapted to the available time
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 750:
        n = 16384          # highest resolution when we have plenty of time
    elif remaining > 450:
        n = 8192
    elif remaining > 250:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = np.inf

    # -------------------------------------------------
    # 0. Optional fallback to a known construction
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Deterministic + random initialisation search
    # -------------------------------------------------
    a0, sc0 = best_initial_vector(n, deadline - 30.0)
    if a0 is not None and sc0 < best_score:
        best_score = sc0
        best_seq = a0.tolist()
        print(f"[{time.time():.1f}] best after initial families = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Adam‑based smooth optimisation (primary phase)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] primary Adam‑smooth optimisation")
        a_smooth, sc_smooth = adam_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 240.0,
            max_iters=250_000,
            beta_start=0.5,
            beta_end=300.0,
            alpha0=0.7,
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.tolist()
            print(f"[{time.time():.1f}] after primary Adam = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Scale‑up (if time permits) and quick refinement
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        # Upsample to double the length (if feasible) and run a short Adam run
        a_big = upsample_vector(np.array(best_seq, dtype=float))
        n_big = a_big.size
        print(f"[{time.time():.1f}] up‑sampling to {n_big} and quick Adam")
        a_up, sc_up = adam_smooth_optimize(
            a_big,
            deadline - 140.0,
            max_iters=80_000,
            beta_start=0.5,
            beta_end=400.0,
            alpha0=0.6,
        )
        if sc_up < best_score:
            best_score = sc_up
            best_seq = a_up.tolist()
            print(f"[{time.time():.1f}] after up‑sample Adam = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Momentum sub‑gradient descent on the true max
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] momentum sub‑gradient polishing")
        a_sub, sc_sub = momentum_subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 70.0,
            max_iters=300_000,
            step0=0.07,
            momentum=0.85,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Simulated‑annealing hill‑climb (final polishing)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final annealed hill‑climb")
        a_final, sc_final = annealed_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=900_000,
            init_temp=2e-3,
            final_temp=5e-7,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # 6. Safety net – fall back to a short uniform vector if needed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
