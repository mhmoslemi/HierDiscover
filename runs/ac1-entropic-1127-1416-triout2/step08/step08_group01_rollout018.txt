def fast_hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    conv = convolution_fft(a, a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    step_scale = 0.12
    inc_factor = 1.08
    decay_factor = 0.96
    recompute_interval = 200000 # maybe adjust

    steps = 0
    while time.time() < deadline:
        # optionally break if steps exceeds max
        # pick move
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # ensure src larger
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            delta = a[src] - a[dst]  # positive
        else:
            src = np.random.randint(n)
            delta = step_scale * a[src]
            if delta == 0.0:
                continue
            dst = np.random.randint(n-1)
            if dst >= src:
                dst += 1

        # candidate conv
        conv_cand = conv.copy()
        # apply linear updates
        # conv[src:src+n] -= 2*delta*a
        conv_cand[src:src+n] -= 2.0 * delta * a
        conv_cand[dst:dst+n] += 2.0 * delta * a
        # quadratic terms
        conv_cand[2*src] += delta*delta
        conv_cand[2*dst] += delta*delta
        conv_cand[src+dst] -= 2.0 * delta*delta

        # compute score
        max_cand = conv_cand.max()
        cand_score = 2.0 * n * max_cand

        if cand_score < best_score:
            # accept
            best_score = cand_score
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand  # replace
            best_a = a.copy()
            step_scale = min(step_scale * inc_factor, 0.5)
        else:
            step_scale *= decay_factor

        steps += 1

        if steps % recompute_interval == 0:
            conv = convolution_fft(a, a)  # re-sync

    return best_a, best_score

"""
Propose a low‑score step‑function sequence.

The search proceeds in four stages:

1. **Analytic start** – a one‑parameter family  
   a_i(α) ∝ [(i+½)(n‑½‑i)]^{‑α} is scanned on a modest α‑grid.
   The best point on the simplex is kept.

2. **Smooth exponentiated‑gradient optimisation** – minimises a
   log‑sum‑exp surrogate of the max‑convolution using KL‑mirror
   updates.  The temperature β is annealed, the step‑size η follows
   η₀/√t.

3. **Projected sub‑gradient descent** on the *exact* objective
   2·n·max(a⊛a) with Euclidean projection onto the probability simplex.

4. **Fast stochastic hill‑climb** – a massive random‑walk that
   transfers a small packet of mass between two positions (or swaps
   two entries).  The convolution is updated *incrementally* using the
   linear formula

       Δconv = 2·δ·(shift_{dest}(a) – shift_{src}(a))
               + δ²·e_{2·src} + δ²·e_{2·dest} – 2·δ²·e_{src+dest},

   avoiding an FFT at every step.  The walk runs until the time budget
   is exhausted, constantly keeping the best exact score seen.

All phases respect the 1000 s wall‑clock limit; the best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalized to sum = 1.
    This is the value returned by `evaluate_sequence` when the input
    already satisfies the sum‑normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytic start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # avoid infinities for extreme α
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a modest α‑grid and keep the best vector."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([
        np.linspace(-1.0, 0.0, 11),   # centre‑heavy
        np.linspace(0.0, 2.5, 26)    # edge‑heavy
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Exponentiated‑gradient optimisation of the smooth max
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 150_000,
                        beta_start: float = 0.5,
                        beta_end: float = 300.0,
                        eta0: float = 0.6,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective f_β using a KL‑mirror (exponentiated‑gradient)
    step.  β is exponentially annealed, η follows η₀/√t.
    Returns the best exact score seen during the run.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smooth max (log‑sum‑exp)
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))    # stabilised
        w = expw / expw.sum()                  # softmax weights

        # gradient of f_β (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # step size
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient method for 2·n·max(a*a)."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Fast stochastic hill‑climb (incremental convolution updates)
# ----------------------------------------------------------------------


def fast_hill_climb(start_a: np.ndarray,
                    deadline: float,
                    inc_factor: float = 1.07,
                    decay_factor: float = 0.97,
                    recompute_interval: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Random‑walk polishing with O(n) incremental convolution updates.
    Only strictly improving moves are accepted.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)               # exact convolution
    best_a = a.copy()
    best_sc = 2.0 * n * float(conv.max())
    step_scale = 0.12

    steps = 0
    while time.time() < deadline:
        # Choose move type
        if np.random.rand() < 0.5:
            # swap (implemented as a full transfer of the difference)
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # source = larger entry, destination = smaller entry
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            delta = a[src] - a[dst]            # positive, results in a swap
        else:
            # transfer a small packet from src -> dst
            src = np.random.randint(n)
            if a[src] == 0.0:
                continue
            delta = step_scale * a[src]         # always ≤ a[src]
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1                         # ensure dst != src

        # Build candidate convolution (copy + cheap updates)
        conv_cand = conv.copy()
        # linear part: –2·δ·shift_src + 2·δ·shift_dst
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[dst:dst + n] += 2.0 * delta * a
        # quadratic correction (exact)
        conv_cand[2 * src] += delta * delta
        conv_cand[2 * dst] += delta * delta
        conv_cand[src + dst] -= 2.0 * delta * delta

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        if cand_score < best_sc:
            # accept move
            best_sc = cand_score
            a[src] -= delta
            a[dst] += delta
            conv = conv_cand
            best_a = a.copy()
            step_scale = min(step_scale * inc_factor, 0.5)
        else:
            # reject – decay step size
            step_scale *= decay_factor

        steps += 1
        if steps % recompute_interval == 0:
            # periodic full recomputation to avoid drift
            conv = convolution_fft(a, a)

    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine follows a four‑stage plan and respects the 1000 s wall‑clock limit.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # -------------------------------------------------
    # Choose a workable length (more length → lower theoretical bound)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytic start (α‑grid)
    # -------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Smooth exponentiated‑gradient optimisation (single run)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] exp‑gradient optimisation")
        a_smooth, sc_smooth = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 150.0,
            max_iters=150_000,
            beta_start=0.5,
            beta_end=300.0,
            eta0=0.6,
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Sub‑gradient refinement
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 80.0,
            max_iters=200_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Fast hill‑climb polishing (majority of remaining budget)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] fast hill‑climb")
        a_final, sc_final = fast_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a uniform short vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
