"""
This search builds a multiscale family of non‑negative step sequences.
It starts with a cheap 1 k‑point seed, discovers a promising shape by
Adam on a smooth log‑sum‑exp surrogate of the max‑convolution, then
progressively upsamples the best shape to longer lengths, polishing each
scale with exponentiated‑gradient and a final projected sub‑gradient
descent.  The whole routine respects the 1000 s budget and always
returns a valid sequence.
"""

import time
import numpy as np
from typing import List

# utility: next power of two
def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """ Fast surrogate score when sum(a)=1 """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# -------------------------- seed families ---------------------------

def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals

def half_sine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float) + 0.5
    vals = np.sin(np.pi * i / (n + 1))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

def parabolic_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float) + 0.5
    vals = i * (n + 0.5 - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total

def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total

def generate_mixture_shape(n: int, w_arc: float) -> np.ndarray:
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = w_arc * arc + (1.0 - w_arc) * uni
    mix /= mix.sum()
    return mix

def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups

# --------------------------- optimisation ---------------------------

def softmax(z: np.ndarray) -> np.ndarray:
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def smooth_adam_optimize(z0: np.ndarray,
                         deadline: float,
                         n: int,
                         beta0: float = 5.0,
                         beta_max: float = 2000.0,
                         ramp_iters: int = 150_000,
                         lr: float = 0.07) -> np.ndarray:
    """Adam on the smooth log‑sum‑exp surrogate of max(conv)."""
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # simplex point
        conv = convolution_fft(a, a)       # a ⋆ a
        mx = float(np.max(conv))

        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    [Adam] it={t:,d} β={beta:.0f} surrogate_score={compute_score(softmax(z)):.6f}")

    return softmax(z)

def exponentiated_gradient_polish(a0: np.ndarray,
                                 deadline: float,
                                 max_iter: int = 200_000,
                                 step: float = 0.02) -> np.ndarray:
    """Mirror descent (KL) on the true max‑convolution objective."""
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    [EG]   it={it:,d} score={compute_score(a):.6f}")
    return a

def subgradient_descent_exact(init_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.09) -> np.ndarray:
    """Projected Euclidean sub‑gradient descent on the exact objective."""
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    [SubG] it={it:,d} score={compute_score(a):.6f}")
    return a

# ----------------- seed construction utilities ---------------------

def build_seed_pool(n: int, rng: np.random.Generator, best_sofar: List[float]) -> List[np.ndarray]:
    seeds = [
        arcsine_initial(n),
        half_sine_initial(n),
        parabolic_shape(n),
        np.full(n, 1.0 / n),
    ]

    # random beta shapes
    for _ in range(6):
        a = rng.uniform(0.05, 7.0)
        b = rng.uniform(0.05, 7.0)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        seeds.append(generate_beta_shape(n, a, b, eps))

    # random power‑law shapes
    for _ in range(6):
        p = rng.uniform(0.05, 4.0)
        eps = rng.uniform(0.0, 2.0)
        seeds.append(generate_power_shape(n, p, eps))

    # mixture of arcsine & uniform
    for _ in range(4):
        w = rng.uniform(0.0, 1.0)
        seeds.append(generate_mixture_shape(n, w))

    # up‑sampled previously best
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        seeds.append(upsample_sequence(old, n))

    # slight perturbations of the up‑sampled best (if any) to escape local minima
    if best_sofar:
        base = upsample_sequence(np.array(best_sofar, dtype=float), n)
        for _ in range(3):
            noise = rng.normal(scale=0.01, size=n)
            pert = np.clip(base + noise, 0.0, None)
            pert /= pert.sum()
            seeds.append(pert)

    return seeds

        print(f"  • allocated {level_budget:.1f}s for optimisation at this scale")

                        print(f"   *** improved after EG polish: {best_val:.6f} ***")

    # -----------------------------------------------------------------

        a = np.array(best_seq, dtype=float)
        # a short run of exponentiated‑gradient with a finer step
        a = exponentiated_gradient_polish(a, deadline - 1.0, max_iter=80_000, step=0.01)
        try:
            final_val = evaluate_sequence(a.tolist())
            if final_val < best_val:

                print(f"   *** final best improved to {final_val:.6f} ***")
        except Exception as e:
            print(f"   final polishing exception: {e}")

    # -----------------------------------------------------------------
    # 3. safeguard – return a trivial uniform vector if nothing found
    # -----------------------------------------------------------------
    if not best_seq:
        n0 = 1024
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best score = {best_val:.6f}")
    return best_seq

"""
Multiscale optimisation of a non‑negative step sequence.

The routine proceeds in five stages:

1. **Fallback** – use a previously found sequence if it exists.
2. **Seed generation** – build a diverse pool of analytic shapes
   (arcsine, half‑sine, parabola, beta / power‑law families,
   mixtures and up‑sampled best‑so‑far).
3. **Smooth‑max Adam** – minimise a log‑sum‑exp surrogate of
   ``max(conv(a,a))`` on the probability simplex (variables are the
   soft‑max of free parameters).  The temperature β is increased
   from 5 to 2000, providing a tight approximation to the true max.
4. **KL‑mirror polishing** – exponentiated‑gradient (mirror descent)
   on the exact objective to further reduce the peak of the
   self‑convolution.
5. **Exact sub‑gradient polishing** – a final Euclidean projected
   sub‑gradient descent for a few thousand iterations.

A multiscale schedule (1024 → 2048 → 4096 → 8192 → 16384) is used:
the best vector from a coarser scale is up‑sampled and refined on the
next scale.  All stages respect the overall wall‑clock budget (default
1000 s) and the function always returns a valid list of non‑negative
heights.
"""

import time
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate score when ``sum(a)=1``.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def half_sine_initial(n: int) -> np.ndarray:
    """Half‑sine envelope."""
    i = np.arange(n, dtype=float) + 0.5
    vals = np.sin(np.pi * i / (n + 1))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def parabolic_shape(n: int) -> np.ndarray:
    """Simple parabola (zero at the ends, maximal in the centre)."""
    i = np.arange(n, dtype=float) + 0.5
    vals = i * (n + 0.5 - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()

        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p

    return vals / total

def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups

def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → point on the simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(z0: np.ndarray,
                         deadline: float,
                         n: int,
                         beta0: float = 5.0,
                         beta_max: float = 2000.0,
                         ramp_iters: int = 150_000,
                         lr: float = 0.07) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of ``max(conv)``.

    Parameters
    ----------
    z0 : ndarray
        Initial log‑softmax coordinates.
    deadline : float
        Absolute time at which the optimiser must stop.
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate for Adam.

    Returns
    -------
    a : ndarray
        Optimised probability vector (sum = 1, a_i ≥ 0).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                      # simplex point
        conv = convolution_fft(a, a)        # a⋆a
        mx = float(np.max(conv))

        # temperature cooling schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx                 # ≤0
        w = np.exp(beta * shifted)
        w_sum = w.sum()

        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    [Adam] it={t:,d} β={beta:.0f} surrogate_score={compute_score(softmax(z)):.6f}")

    return softmax(z)


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(a0: np.ndarray,
                                 deadline: float,
                                 max_iter: int = 200_000,
                                 step: float = 0.02) -> np.ndarray:
    """
    Mirror descent (KL divergence) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # active set – indices where conv is (almost) maximal
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    [EG]   it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(init_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.09) -> np.ndarray:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    [SubG] it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Helper: build a pool of diverse seeds for a given length
# ----------------------------------------------------------------------


def build_seed_pool(n: int,
                    rng: np.random.Generator,
                    best_sofar: List[float]) -> List[np.ndarray]:
    """Return a list of candidate probability vectors of length n."""
    seeds = [
        arcsine_initial(n),
        half_sine_initial(n),
        parabolic_shape(n),
        np.full(n, 1.0 / n),
    ]

    # random beta‑shapes
    for _ in range(6):
        a = rng.uniform(0.05, 7.0)
        b = rng.uniform(0.05, 7.0)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        seeds.append(generate_beta_shape(n, a, b, eps))

    # random power‑law shapes
    for _ in range(6):
        p = rng.uniform(0.05, 4.0)
        eps = rng.uniform(0.0, 2.0)
        seeds.append(generate_power_shape(n, p, eps))

    # mixtures of arcsine & uniform
    for _ in range(4):
        w = rng.uniform(0.0, 1.0)
        seeds.append(generate_mixture_shape(n, w))

    # up‑sampled former best (if any)
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        seeds.append(upsample_sequence(old, n))

        # tiny Gaussian perturbations of the up‑sampled best
        base = upsample_sequence(old, n)
        for _ in range(3):
            noise = rng.normal(scale=0.01, size=n)
            pert = np.clip(base + noise, 0.0, None)
            pert /= pert.sum()
            seeds.append(pert)

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 43,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low
    ``evaluate_sequence`` value.  The routine respects a 1000 s
    wall‑clock budget and always returns a non‑empty valid sequence.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.8    # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0.  Fallback: previously discovered sequence (if present)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"[Fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1.  Multiscale optimisation – start small, finish large
    # -----------------------------------------------------------------
    length_schedule = [1024, 2048, 4096, 8192, 16384]

    for level, n in enumerate(length_schedule, start=1):
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Level {level}/{len(length_schedule)} – length {n} ===")

        # ---- 1a. Build a diverse seed pool --------------------------------
        seed_pool = build_seed_pool(n, rng, best_seq)

        # ---- 1b. Quick surrogate ranking ----------------------------------
        scored = [(compute_score(s), s) for s in seed_pool]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]   # keep only few promising seeds

        # ---- 1c. Allocate time for this scale ---------------------------
        remaining = deadline - time.time()
        # reserve a few seconds per remaining scale
        reserve = max(5.0, (len(length_schedule) - level) * 5.0)
        level_budget = max(5.0, (remaining - reserve) * 0.6)
        level_deadline = time.time() + level_budget
        print(f"  • allocated {level_budget:.1f}s for optimisation at this scale")

        # ---- 1d. Optimise each promising seed ----------------------------
        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 5.0:
                break

            # give each seed a slice of the level budget
            per_seed = level_budget / len(top_seeds)
            seed_deadline = min(time.time() + per_seed, level_deadline, deadline - 4.0)

            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            # ----- Smooth‑max Adam optimisation ---------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=2000.0,
                ramp_iters=200_000,
                lr=0.07,
            )

            # ----- Official evaluation ------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   [Adam] seed {idx} score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** new global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   evaluation error: {e}")

            # ----- KL‑mirror polishing (if time permits) -----------------
            if time.time() < deadline - 4.0:
                polish_deadline = deadline - 2.0
                a_pol = exponentiated_gradient_polish(
                    np.array(best_seq, dtype=float),
                    polish_deadline,
                    max_iter=150_000,
                    step=0.02,
                )
                try:
                    val2 = evaluate_sequence(a_pol.tolist())
                    print(f"   [EG] polished score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, a_pol.tolist()
                        print(f"   *** improved after EG polish: {best_val:.6f} ***")
                except Exception as e:
                    print(f"   EG polish error: {e}")

        # ----- Brief sub‑gradient polishing before moving to the next scale
        if best_seq and time.time() < deadline - 3.0:
            a_arr = np.array(best_seq, dtype=float)
            a_arr = subgradient_descent_exact(
                a_arr,
                deadline - 2.0,
                max_iter=120_000,
                step0=0.09,
            )
            try:
                new_val = evaluate_sequence(a_arr.tolist())
                if new_val < best_val:
                    best_val, best_seq = new_val, a_arr.tolist()
                    print(f"   *** sub‑gradient improved to {best_val:.6f} ***")
            except Exception as e:
                print(f"   final sub‑gradient error: {e}")

    # -----------------------------------------------------------------
    # 2. Final global polishing (if any time remains)
    # -----------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\nFinal polishing on the global best …")
        a = np.array(best_seq, dtype=float)
        a = exponentiated_gradient_polish(a, deadline - 1.0, max_iter=80_000, step=0.01)
        try:
            final_val = evaluate_sequence(a.tolist())
            if final_val < best_val:
                best_val, best_seq = final_val, a.tolist()
                print(f"   *** final best improved to {final_val:.6f} ***")
        except Exception as e:
            print(f"   final polishing exception: {e}")

    # -----------------------------------------------------------------
    # 3. Guard – guarantee a non‑empty return
    # -----------------------------------------------------------------
    if not best_seq:
        n0 = 1024
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
