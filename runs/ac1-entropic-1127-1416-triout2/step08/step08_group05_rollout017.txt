    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
        self.L = 2*n - 1

    def compute_a(self, z):
        e = np.exp(z - np.max(z))
        a = e / e.sum()
        return a

    def compute_conv(self, a):
        return convolution_fft(a, a)

    def value(self, z):
        a = self.compute_a(z)
        conv = self.compute_conv(a)
        mx = np.max(conv)
        # smooth max using log-sum-exp
        c = np.max(beta * conv)  # could use conv.max()
        # compute log-sum-exp
        t = self.beta * conv - self.beta * mx
        max_t = np.max(t)
        # Actually we can just compute using stable method:
        # w = exp(beta*(conv - mx))
        w = np.exp(self.beta * (conv - mx))
        w_sum = w.sum()
        smooth = mx + np.log(w_sum) / self.beta
        return 2.0 * self.n * smooth

    def grad(self, z):
        a = self.compute_a(z)
        conv = self.compute_conv(a)
        mx = np.max(conv)
        beta = self.beta
        # compute softmax weights w
        # stable:
        shifted = beta * (conv - mx)
        max_shift = np.max(shifted)
        exp_shift = np.exp(shifted - max_shift)
        w = exp_shift / exp_shift.sum()

        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[self.n - 1 : self.n - 1 + self.n]
        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

"""
Propose a step‑function (non‑negative sequence) with a low
``evaluate_sequence`` value.

Algorithm
---------
1.  Build a diverse pool of analytic seeds for a chosen length ``n``.
    The pool contains arcsine, uniform, beta‑ and power‑law shapes,
    mixtures and a sparse “spike’’ pattern.  If a good sequence has
    already been found it is up‑sampled and added to the pool.

2.  Rank the seeds quickly with the *smooth* surrogate
    ``2·n·max(conv(a,a))`` (the same value that ``evaluate_sequence``
    returns for a normalised sequence).  Keep the few best seeds.

3.  For every kept seed run a long Adam optimisation on a
    *smooth‑max* surrogate of the true objective.  The surrogate
    replaces ``max`` by a log‑sum‑exp with a temperature that grows
    from ``β0`` to ``βmax``; the gradient is computed analytically via
    FFT convolutions.  Adam keeps the parameters in the log‑softmax
    (i.e. a simplex) domain.

4.  The best Adam result for the length is polished with an
    exponentiated‑gradient (KL‑mirror) step whose step‑size decays as
    ``1/√t`` – this keeps us on the probability simplex without any
    explicit projection.

5.  After all lengths have been processed a final *L‑BFGS* optimisation
    (SciPy) is launched on the globally best candidate.  The optimiser
    works on the same smooth‑max surrogate and uses the exact gradient;
    a modest iteration budget is enforced so that the wall‑clock limit
    is respected.

6.  A short Euclidean projected sub‑gradient descent finishes the
    optimisation (the exact objective).  This guarantees that the
    returned point satisfies all constraints and often yields a tiny
    extra improvement.

The routine respects a 1000 s wall‑clock budget, always returns a
non‑empty valid sequence and prints intermediate scores for
monitoring.
"""

import time
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation for a probability vector (sum(a)=1).
    Returns the exact value of ``evaluate_sequence`` (ignoring clipping).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))

# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_sparse_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Place ``m`` equal spikes at random positions (Dirichlet‑like)."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / m
    return a


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence onto a larger grid."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 800.0,
    ramp_iters: int = 120_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate of
    ``2·n·max(conv)``.  Parameters are chosen for a long, steady
    optimisation.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # simplex point

        # convolution and smooth‑max weights
        conv = convolution_fft(a, a)       # length 2n‑1
        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # log‑sum‑exp weights (stable)
        shifted = beta * (conv - mx)       # ≤ 0
        max_shift = np.max(shifted)        # = 0
        w = np.exp(shifted - max_shift)    # already normalised by subtraction of 0
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central n entries

        # chain‑rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if t % 10_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(softmax(z)):.6f}")

    return softmax(z)


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.03,
) -> np.ndarray:
    """
    Mirror descent on the exact max‑convolution objective.
    Step size decays as ``step0 / sqrt(t)`` which is a good
    schedule for convex non‑smooth objectives.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # active set – indices attaining the max (up to numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # decaying step size
        step = step0 / np.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if not np.isfinite(s) or s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 10_000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.09,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Euclidean projection onto the simplex keeps the iterate feasible.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 20_000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max surrogate for L‑BFGS (SciPy)
# ----------------------------------------------------------------------


class SmoothMaxObjective:
    """
    Callable class for SciPy L‑BFGS.  The objective is the
    log‑sum‑exp surrogate of 2·n·max(conv) with a fixed temperature β.
    Both value() and grad() are provided.
    """

    def __init__(self, n: int, beta: float = 500.0):
        self.n = n
        self.beta = beta
        self.L = 2 * n - 1

    @staticmethod
    def _softmax(z: np.ndarray) -> np.ndarray:
        z_max = np.max(z)
        e = np.exp(z - z_max)
        return e / e.sum()

    def value(self, z: np.ndarray) -> float:
        """Smooth objective value (scalar)."""
        a = self._softmax(z)
        conv = convolution_fft(a, a)

        # log‑sum‑exp of β·conv
        b = self.beta * conv
        b_max = np.max(b)
        logsum = b_max + np.log(np.sum(np.exp(b - b_max)))
        smooth_max = logsum / self.beta
        return float(2.0 * self.n * smooth_max)

    def grad(self, z: np.ndarray) -> np.ndarray:
        """Gradient of the smooth objective w.r.t. logits `z`."""
        a = self._softmax(z)
        conv = convolution_fft(a, a)

        # softmax weights w = exp(β·conv) / sum exp(β·conv)
        b = self.beta * conv
        b_max = np.max(b)
        exp_shift = np.exp(b - b_max)
        w = exp_shift / exp_shift.sum()

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[self.n - 1 : self.n - 1 + self.n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        return grad_z


def lbfgs_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200,
    beta: float = 500.0,
) -> np.ndarray:
    """
    L‑BFGS using SciPy on the smooth‑max surrogate.
    The routine stops early if the wall‑clock budget expires.
    """
    try:
        import scipy.optimize
    except Exception as e:
        print("SciPy not available – skipping L‑BFGS phase.")
        return a0

    z0 = np.log(a0 + 1e-12)
    obj = SmoothMaxObjective(n=a0.size, beta=beta)

    # SciPy's L‑BFGS does not natively support a time‑budget,
    # so we limit the number of function evaluations.
    # The callback terminates early if the deadline is passed.
    def callback(xk):
        # return True to stop – SciPy ignores the return value,
        # so we raise a StopIteration exception.
        if time.time() >= deadline:
            raise StopIteration

    try:
        z_opt, fopt, _ = scipy.optimize.fmin_l_bfgs_b(
            func=obj.value,
            x0=z0,
            fprime=obj.grad,
            maxiter=max_iter,
            maxfun=max_iter,
            callback=callback,
            iprint=-1,  # suppress SciPy's own prints
        )
    except StopIteration:
        # optimisation cut short by time limit
        z_opt = None

    if z_opt is None:
        # we stopped during the iterations – fall back to the last safe point
        a = softmax(z0)
    else:
        a = softmax(z_opt)

    return a


# ----------------------------------------------------------------------
# Seed‑pool construction
# ----------------------------------------------------------------------


def build_seed_pool(
    n: int,
    rng: np.random.Generator,
    best_sofar: List[float],
) -> List[np.ndarray]:
    """
    Assemble a list of diverse candidate vectors (already normalised).
    """
    seeds: List[np.ndarray] = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),                     # uniform
    ]

    # random beta‑shapes (including very peaked versions)
    for _ in range(12):
        a = rng.uniform(0.2, 1.5)   # focus on the region around arcsine (0.5) and uniform (1)
        b = rng.uniform(0.2, 1.5)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        seeds.append(generate_beta_shape(n, a, b, eps))

    # random power‑law shapes
    for _ in range(8):
        p = rng.uniform(0.05, 3.0)
        eps = rng.uniform(0.0, 2.0)
        seeds.append(generate_power_shape(n, p, eps))

    # mixtures of arcsine & uniform
    for _ in range(4):
        w = rng.uniform(0.0, 1.0)
        seeds.append(generate_mixture_shape(n, w))

    # sparse spikes (few equal Dirichlet atoms)
    for m in (int(np.sqrt(n) // 2), int(np.sqrt(n)), int(2 * np.sqrt(n))):
        if 1 < m < n:
            seeds.append(generate_sparse_seed(n, m, rng))

    # up‑sampled version of the current best (if any)
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        seeds.append(upsample_sequence(old, n))

    return seeds

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 43, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low
    ``evaluate_sequence`` value.  The routine respects a 1000 s
    wall‑clock budget and always returns a non‑empty valid sequence.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use a previously discovered sequence if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Iterate over a cascade of increasing lengths
    # ------------------------------------------------------------------
    # Larger n tends to give smaller asymptotic constant, start with the biggest.
    lengths = [16384, 8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 25.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ---- 1a. Build a diverse seed pool ---------------------------
        seed_pool = build_seed_pool(n, rng, best_seq)

        # ---- 1b. Quick surrogate ranking --------------------------------
        scored = [(compute_score(s), s) for s in seed_pool]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]   # keep at most five promising seeds

        # ---- 1c. Allocate time per seed --------------------------------
        remaining = deadline - time.time()
        per_seed_time = max(12.0, remaining / (len(top_seeds) * 2.0))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 12.0:
                break

            print(
                f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_time:.1f}s"
            )
            seed_deadline = min(time.time() + per_seed_time, deadline - 6.0)

            # Initialise Adam on the log‑softmax of the seed
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            # ------------------------------------------------------------
            # 2.  Adam optimisation on the smooth‑max surrogate
            # ------------------------------------------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=800.0,
                ramp_iters=150_000,
                lr=0.07,
            )

            # ------------------------------------------------------------
            # 3.  Official evaluation (fast surrogate = exact at sum=1)
            # ------------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ------------------------------------------------------------
            # 4.  Optional KL‑mirror polishing (if time permits)
            # ------------------------------------------------------------
            if time.time() < deadline - 8.0:
                polish_deadline = deadline - 3.0
                a_polish = exponentiated_gradient_polish(
                    np.array(best_seq, dtype=float),
                    polish_deadline,
                    max_iter=120_000,
                    step0=0.03,
                )
                try:
                    val2 = evaluate_sequence(a_polish.tolist())
                    print(f"   Polished (EG) score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, a_polish.tolist()
                        print(
                            f"   *** New best after EG polish: {best_val:.6f} ***"
                        )
                except Exception as e:
                    print(f"   EG polishing failed: {e}")

        # ------------------------------------------------------------
        # 5.  L‑BFGS polish on the current global best for this length
        # ------------------------------------------------------------
        if best_seq and time.time() < deadline - 6.0:
            a_arr = np.array(best_seq, dtype=float)
            # we may need to up‑sample / down‑sample to the current n
            if a_arr.size != n:
                a_arr = upsample_sequence(a_arr, n)
            lbfgs_deadline = deadline - 1.0
            a_refined = lbfgs_optimize(
                a_arr,
                lbfgs_deadline,
                max_iter=250,
                beta=800.0,
            )
            try:
                val3 = evaluate_sequence(a_refined.tolist())
                print(f"   L‑BFGS polished score = {val3:.6f}")
                if val3 < best_val:
                    best_val, best_seq = val3, a_refined.tolist()
                    print(
                        f"   *** New best after L‑BFGS: {best_val:.6f} ***"
                    )
            except Exception as e:
                print(f"   L‑BFGS evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 6.  Final global polishing (exact sub‑gradient descent)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\nFinal global sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=300_000,
            step0=0.09,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 7.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
