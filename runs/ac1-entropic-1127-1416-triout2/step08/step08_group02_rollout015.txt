This implementation searches for a non‑negative step‑function that minimises  

```
evaluate_sequence(seq) = 2·n·max(conv(seq,seq))/(sum(seq)²)
```

Because the objective is homogeneous we may fix `sum(seq)=1`; then the
objective is simply `2·n·max(conv(seq,seq))`.  The algorithm consists of
four stages that respect the 1000 s wall‑clock budget:

1. **Multiple warm‑starts** – a variety of initial vectors (binomial,
   arcsine‑like, random, uniform) of modest size (`n≈4096`) are created.
2. **Smooth max optimisation** – for each start we run Adam on the
   log‑sum‑exp surrogate of the `max` (parameterised by a temperature
   `β`).  The optimisation is performed on the simplex using Euclidean
   projection.
3. **Upscaling & fine‑tuning** – the best small vector is linearly
   interpolated to a larger size (`n≈8192` or `16384` depending on the
   remaining budget) and refined again with the same smooth optimiser.
4. **Polishing** – a short projected sub‑gradient descent followed by a
   stochastic hill‑climb (swap / fractional mass transfer) removes the
   last overshoots.

All stages stop automatically when the overall deadline is reached, and
the best sequence found is returned as a plain list of floats.

```python
"""
Non‑negative step‑function optimiser for `evaluate_sequence`.

Algorithm
---------
1. Build several diverse initial vectors (binomial, arcsine‑like,
   random, uniform) of moderate length (≈4096).
2. For each start run Adam on a log‑sum‑exp (soft‑max) surrogate of the
   max‑convolution, keeping the best result.  Projection onto the simplex
   enforces non‑negativity and unit sum.
3. Upscale the best vector to a larger length (8192 or 16384) by linear
   interpolation and continue the smooth optimisation.
4. Finish with a brief projected sub‑gradient descent and a stochastic
   hill‑climb (swap / fractional transfer) to polish the solution.
5. Return the best sequence found before the 1000 s deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    Length of result = len(a) + len(b) – 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initial‑vector generators
# ----------------------------------------------------------------------


def init_binomial(n: int) -> np.ndarray:
    """Central binomial coefficients a_i = C(2i,i)/4^i, normalised to sum‑1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2.0 * i + 1.0) / (2.0 * (i + 1))
    return project_to_simplex(a)


def init_arcsine_like(n: int) -> np.ndarray:
    """
    Arcsine‑shaped density: a_i ∝ 1/√((i+½)(n‑½‑i)).
    This mimics the Beta(½,½) law which spreads mass toward the ends.
    """
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return project_to_simplex(a)


def init_random(n: int) -> np.ndarray:
    """Uniform‑Dirichlet random start."""
    a = np.random.rand(n)
    return project_to_simplex(a)


def init_uniform(n: int) -> np.ndarray:
    """Pure uniform distribution."""
    return np.full(n, 1.0 / n, dtype=float)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam on the simplex
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 1_000_000,
    beta_start: float = 5.0,
    beta_end: float = 300.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.

    The surrogate is
        L_β(a) = (1/β)·log Σ_k exp(β·c_k) ,  c = conv(a,a)
    whose gradient is computed analytically (cross‑correlation with the
    soft‑max weights).  After each update the vector is projected onto the
    probability simplex.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)  # first‑moment estimate
    v = np.zeros_like(a)  # second‑moment estimate

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal temperature β (log‑linear schedule)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution a * a
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights (log‑sum‑exp trick)
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient = 2·(a_rev * w)  (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional progress check
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (max‑objective)
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        2·n·max(conv(a,a))
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()
        a = a_new

        # small jitter every 1000 steps to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """
    Simple stochastic local search:
        – with 50 % probability swap two entries,
        – otherwise move a fraction of mass from one index to another.
    After each move the vector is projected onto the simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass transferred

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                       # fractional transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling utility (small → large)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)
```
