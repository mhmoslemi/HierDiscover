def lp_step(current_a, M, deadline):
    # active constraints: indices where conv_f within epsilon of M (e.g., > M - eps)
    conv_f = convolution_fft(current_a, current_a)  # length L
    M = conv_f.max()
    # Identify indices near M
    active_indices = np.where(conv_f >= M * 0.9999)[0]  # tolerance maybe 1e-4
    # Or pick top K
    if len(active_indices) > 30:
        # keep top 30 largest conv values
        topk = np.argpartition(conv_f, -30)[-30:]
        active_indices = topk
    # Build constraint matrix
    L = len(conv_f)
    n = len(current_a)
    m = len(active_indices)
    # Lazy building: for each k, we need row vector row_i where row_i[j] = current_a[k-j] if 0≤k-j<n else 0.
    # We'll build using np.take with out-of-bounds =0.
    # Since n may be large, but m small, the matrix size m x n moderate (30x8192 ~ 250k entries)
    # Use np.zeros((m,n)) and fill rows.
    F = np.zeros((m, n), dtype=np.float64)
    for idx, k in enumerate(active_indices):
        # For j ranging 0..n-1, compute i = k - j
        # Values are a[i] if i in [0,n-1] else 0.
        # Equivalent to: shift array a by -j? This is slower; we can fill using slicing.
        # Compute start = max(0, k - (n - 1)), end = min(k, n-1)
        start_j = max(0, k - (n - 1))
        end_j = min(k, n - 1)
        # For j in [start_j, end_j], i = k - j
        # a_i = a[k - j]
        # Fill: F[idx, start_j:end_j+1] = a[k - np.arange(start_j, end_j+1)]
        j_range = np.arange(start_j, end_j + 1)
        i_vals = k - j_range
        F[idx, start_j:end_j+1] = current_a[i_vals]
    # Solve LP
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, None)] * n
    # Use linprog with method='highs' if available, which is fast.
    res = linprog(c, A_ub=F, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    g0 = res.x
    # Normalize g to sum = sum(current_a) (which is 1)
    g_sum = g0.sum()
    if g_sum == 0:
        return None
    g_norm = g0 / g_sum
    # Compute new mixture (1-t)*a + t*g_norm for some small t maybe 0.1?
    t = 0.2  # tuning.
    new_a = (1 - t) * current_a + t * g_norm
    # Normalize to sum 1 again
    new_a /= new_a.sum()
    return new_a

"""
propose_candidate – a multi‑phase optimizer for the “max‑convolution”
objective used by ``evaluate_sequence``.

The algorithm follows a sequence of increasingly fine‑grained refinements:

1. **Fallback** – use the globally supplied ``height_sequence_1`` if it exists.
2. **Analytic α‑grid** – a one‑parameter family
   a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ is scanned; the best vector is kept.
3. **Random sparse restarts** – many cheap starts are polished with
   an exponentiated‑gradient optimiser that minimises a smooth
   log‑sum‑exp approximation of the max‑convolution.
4. **Long exponentiated‑gradient run** – a dedicated, higher‑budget run
   from the current best vector.
5. **Projected sub‑gradient descent** – classic Euclidean projection onto
   the probability simplex for the exact max‑objective.
6. **Linear‑programming improvement** – given the current vector *f* with
   max‑convolution value *M*, we solve a small LP that maximises the total
   mass of a new vector *g* under the linear constraints *(f∗g) ≤ M*.
   The LP solution is normalised, optionally mixed with *f*, and accepted
   if it improves the exact score.
7. **Final stochastic hill‑climb** – a simple random walk that swaps or
   transfers mass between entries, accepting only strict improvements.

All phases respect a total wall‑clock budget of 1000 s.  The routine
returns a plain ``list[float]`` that the harness will feed to
``evaluate_sequence``.  The implementation is self‑contained; only NumPy
and SciPy (for the LP) are required.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using real FFT.
    The result length is len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) for a vector normalised to sum = 1.
    This matches the value returned by ``evaluate_sequence`` on a
    normalised input.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def enforce_symmetry(a: np.ndarray) -> np.ndarray:
    """Optional symmetry enforcement (mirroring)."""
    b = (a + a[::-1]) * 0.5
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# Analytic start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Guard against overflow for extreme negative alphas
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a modest α‑grid and keep the best vector."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([np.linspace(-1.0, 0.0, 11),
                             np.linspace(0.0, 2.5, 26)])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Random sparse initialisation
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    sparsity = np.random.uniform(0.03, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation of the smooth max
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        beta_start: float = 0.5,
                        beta_end: float = 300.0,
                        eta0: float = 0.6,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective f_β using a KL‑mirror (exponentiated‑gradient)
    step.  β is exponentially annealed, η follows η₀/√t.
    Returns the best exact score seen during the run.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- β schedule (exponential) ---
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # --- smooth weights (log‑sum‑exp) ---
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))    # stabilised
        w = expw / expw.sum()

        # --- gradient of f_β ---
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # --- step size ---
        eta = eta0 / np.sqrt(it)

        # --- KL‑mirror (exponentiated‑gradient) update ---
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # --- occasional exact evaluation ---
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 600_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient method for 2·n·max(a*a)."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming improvement (LP step)
# ----------------------------------------------------------------------


def _build_active_constraint_matrix(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape = len(active_idx) × len(f)) such that
    (A @ g)[m] = (f ∗ g)[active_idx[m]].
    """
    n = f.shape[0]
    m = active_idx.shape[0]
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        # valid j satisfy 0 ≤ k‑j < n → max(0, k‑(n‑1)) ≤ j ≤ min(k, n‑1)
        start_j = max(0, k - (n - 1))
        end_j = min(k, n - 1)
        # slice assignment
        idx_range = np.arange(start_j, end_j + 1, dtype=int)
        i_vals = k - idx_range
        A[row, start_j:end_j + 1] = f[i_vals]
    return A


def lp_improve(start_a: np.ndarray,
               deadline: float,
               max_iters: int = 120,
               active_k: int = 30,
               mix_weights: tuple = (0.1, 0.2, 0.3, 0.4)) -> tuple[np.ndarray, float]:
    """
    Iteratively improve a vector using a small linear program.
    For the current vector f we maximise ∑ g_i subject to (f∗g) ≤ M,
    where M = max(f∗f).  The LP solution g is normalised to sum(f)
    and optionally mixed back into f.  The best exact score found is
    returned.
    """
    a = start_a.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(best_a)

    n = a.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(best_a, best_a)
        M = conv.max()

        # choose a compact set of “active’’ constraints (top values of conv)
        if conv.size <= active_k:
            active_idx = np.arange(conv.size)
        else:
            # indices of the top‑active_k entries
            active_idx = np.argpartition(conv, -active_k)[-active_k:]

        # build the linear constraints (f * g) ≤ M for the chosen indices
        A = _build_active_constraint_matrix(best_a, active_idx)

        # Solve: maximise ∑ g_i  ⇔ minimise −∑ g_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        try:
            res = linprog(c,
                          A_ub=A,
                          b_ub=np.full(A.shape[0], M, dtype=np.float64),
                          bounds=bounds,
                          method='highs')
        except Exception as e:
            print(f"[{time.time():.1f}] LP solver failure: {e}")
            break

        if not res.success:
            # If the LP fails we stop the improvement loop.
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:
            break

        g_norm = g0 / sum_g0  # re‑scale to unit sum

        # evaluate the pure LP solution
        sc_g = compute_score(g_norm)
        improved = False
        if sc_g < best_sc:
            best_sc = sc_g
            best_a = g_norm.copy()
            improved = True

        # try a few convex combinations of current best and g_norm
        for t in mix_weights:
            a_mix = (1.0 - t) * best_a + t * g_norm
            a_mix /= a_mix.sum()
            sc_mix = compute_score(a_mix)
            if sc_mix < best_sc:
                best_sc = sc_mix
                best_a = a_mix.copy()
                improved = True

        if improved:
            print(f"[{time.time():.1f}] LP iter {it}: score improved to {best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Random‑walk polishing: with prob 0.5 swap two entries,
    otherwise transfer a fraction of mass from one index to another.
    Only moves that strictly improve the exact score are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a, best_sc

# Main entry point

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine orchestrates several optimisation phases and respects
    the 1000 s wall‑clock limit.  The returned object is a plain Python
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # -------------------------------------------------
    # Choose a sensible length (larger n → lower theoretical bound)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    elif remaining > 250:
        n = 2048
    else:
        n = 1024
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback using a global variable if present
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytic start (α‑grid)
    # -------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Random sparse restarts with exponentiated‑gradient
    # -------------------------------------------------
    while time.time() < deadline - 250.0:
        a0 = random_sparse_start(n)
        a_opt, sc_opt = exp_smooth_optimize(
            a0,
            deadline - 180.0,
            max_iters=150_000,
            beta_start=0.5,
            beta_end=250.0,
            eta0=0.7,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after random restart = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Long dedicated exponentiated‑gradient run
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] long exponentiated‑gradient optimisation")
        a_smooth, sc_smooth = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=350_000,
            beta_start=0.5,
            beta_end=300.0,
            eta0=0.6,
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Projected sub‑gradient refinement
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 80.0,
            max_iters=600_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Linear‑programming improvement (new phase)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] LP‑based improvement")
        a_lp, sc_lp = lp_improve(
            np.array(best_seq, dtype=float),
            deadline - 50.0,
            max_iters=150,            # a few LP solves are cheap
            active_k=30,
            mix_weights=(0.05, 0.1, 0.2, 0.35),
        )
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.tolist()
            print(f"[{time.time():.1f}] after LP improvement best = {best_score:.6f}")

    # -------------------------------------------------
    # 6. Final hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: fallback to a simple uniform vector if everything failed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness
    return best_seq
