def lp_based_refine(f_init, deadline, max_iters=20, t=0.1):
    f = f_init.copy()
    best = f
    best_val = compute_score(f)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = np.max(convolution_fft(f, f))
        # Build conv matrix once per iteration
        A = build_conv_matrix(f) # shape (2n-1,n)
        # Solve LP
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        g = res.x
        # Normalize to sum 1
        if g.sum() <= 0:
            break
        g_norm = g / g.sum()
        # Mix
        f = (1 - t) * f + t * g_norm
        # Project f to simplex ensure sum 1 (maybe small rounding errors)
        f = project_to_simplex(f)
        val = compute_score(f)
        if val < best_val:
            best_val, best = val, f.copy()
        # possibly reduce t
        t *= 0.9
    return best

"""
Propose a step‑function sequence that (hopefully) beats the previous best
official score ≈ 1.5114.  

Approach
--------
1. **Coarse multi‑scale search** – test several lengths (1024, 2048, 4096)
   and a few parametric families (uniform, arcsine, power‑shaped).  The
   cheap surrogate `2·n·max(conv(a,a))` (identical to the official score
   when Σa = 1) is used to keep only the most promising candidate.

2. **LP‑based refinement** – for the chosen length (restricted to ≤ 2048
   so that the LP stays moderate) we repeatedly solve a linear program
   that maximises the total mass of a secondary distribution `g` under
   the linear constraints `conv(f,g) ≤ M` where `M` is the current worst
   convolution value of `f`.  Mixing a small fraction of `g` into `f`
   strictly reduces the worst convolution and drives the objective down.

3. **Smooth‑max Adam** – a few Adam steps on a soft‑max (log‑sum‑exp)
   approximation of the max‑convolution, using a large temperature
   schedule (`τ` up to 2000) to tighten the approximation.

4. **Exact sub‑gradient descent** – Euclidean projection onto the
   probability simplex together with a diminishing stepsize.

5. **Exponentiated‑gradient polish** – mirror‑descent w.r.t. KL divergence.

6. **Cheap stochastic hill‑climbing** – random mass transfers that
   improve the surrogate.

All stages respect a global wall‑clock deadline (default 1000 s).  The
best sequence found (always non‑negative and summing to 1) is finally
evaluated with `evaluate_sequence` and returned.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score (assuming Σa = 1):
        score = 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """
    Generalised arcsine shape:
        a_i ∝ ((i+½)*(n‑½‑i) + ε)^{-p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Linear‑programming based refinement
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the dense matrix A (shape (2n‑1)×n) where A·g = conv(f,g)."""
    n = f.size
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def lp_refine(f_init: np.ndarray,
              deadline: float,
              max_outer_iters: int = 15,
              mix_factor: float = 0.2) -> np.ndarray:
    """
    Perform a few LP‑based improvement steps.
    Each step solves:
        maximise   Σ g_j
        subject to conv(f, g) ≤ M   (M = max(conv(f,f)))
                  g ≥ 0
    The solution `g` is normalised to sum = 1 and mixed into the current
    distribution `f` with weight `mix_factor`.  The mixing weight decays
    over the iterations.
    """
    from scipy.optimize import linprog

    f = f_init.copy()
    best = f.copy()
    best_val = compute_score(best)
    t = mix_factor

    for it in range(max_outer_iters):
        if time.time() >= deadline:
            break

        # current worst convolution value
        M = np.max(convolution_fft(f, f))

        # build linear constraints: A·g ≤ M
        A = build_conv_matrix(f)
        L = A.shape[0]
        b = np.full(L, M, dtype=np.float64)

        # linear programme: maximise Σ g  → minimise -Σ g
        c = -np.ones(f.size, dtype=np.float64)

        # bounds g_i ≥ 0
        bounds = [(0, None) for _ in range(f.size)]

        # solve
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # If the LP fails we abort the refinement.
            print(f"    LP iteration {it + 1}: solver failed ({res.message})")
            break

        g = np.maximum(res.x, 0.0)          # clip tiny negatives
        s = g.sum()
        if s == 0.0:
            print(f"    LP iteration {it + 1}: got zero‑mass solution")
            break

        # normalise and mix
        g_norm = g / s
        f = (1.0 - t) * f + t * g_norm
        # keep on simplex (numerical drift)
        f = project_to_simplex(f)

        val = compute_score(f)
        if val < best_val:
            best_val, best = val, f.copy()
            print(f"    LP iter {it + 1}: improved surrogate = {val:.6f}")

        # decay mixing weight – smaller steps as we approach a fixed point
        t *= 0.85
        if t < 0.01:
            t = 0.01

    return best

# Optimisers (identical to the previous code but with a few tweaks)
# ----------------------------------------------------------------------
def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 120_000,
                             tau_start: float = 0.5,
                             tau_end: float = 2000.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max approximation of max(conv).
    Variable = logits w, a = softmax(w).  Temperature τ is increased
    linearly from `tau_start` to `tau_end`.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)          # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional diagnostics
        if it % 30000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.05) -> np.ndarray:
    """Euclidean projected sub‑gradient descent on the true max‑convolution."""
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    Sub‑grad it={it:,d} surrogate={surrogate:.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.25) -> np.ndarray:
    """Exponentiated‑gradient polish (mirror descent w.r.t. KL)."""
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    EG it={it:,d} surrogate={surrogate:.6f}")

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 30_000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Very cheap stochastic hill climbing:
    move a tiny amount of mass from one coordinate to another.
    Accept only improving moves (according to the surrogate score).
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    trials = max_trials
    while time.time() < deadline - 0.2 and trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:          # maintain non‑negativity
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        trials -= 1

    return a_cur

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low official score.
    The pipeline consists of:

    1. coarse multi‑scale shape search,
    2. an LP‑based refinement (restricted to n≤2048),
    3. smooth‑max Adam,
    4. exact sub‑gradient descent,
    5. KL‑mirror (exponentiated‑gradient) polishing,
    6. cheap stochastic local search.

    All stages respect a global wall‑clock deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_surrogate: float = float("inf")
    best_official: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = fallback
                best_surrogate = compute_score(np.array(fallback))
        except Exception as e:
            print(f"Fallback evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Stage 1 – coarse multi‑scale shape search
    # ------------------------------------------------------------------
    candidate_ns = [1024, 2048, 4096]
    p_grid = np.linspace(0.35, 0.65, 13)   # power‑shape exponents around 0.5

    print("\n--- Stage 1: coarse grid search ---")
    for n in candidate_ns:
        print(f"Testing length n={n}")

        # uniform
        uni = np.full(n, 1.0 / n)
        s = compute_score(uni)
        if s < best_surrogate:
            best_surrogate, best_seq = s, uni.tolist()
            print(f"  New best (uniform) surrogate={s:.6f}")

        # arcsine
        arc = arcsine_initial(n)
        s = compute_score(arc)
        if s < best_surrogate:
            best_surrogate, best_seq = s, arc.tolist()
            print(f"  New best (arcsine) surrogate={s:.6f}")

        # power shapes
        for p in p_grid:
            shape = generate_power_shape(n, p)
            s = compute_score(shape)
            if s < best_surrogate:
                best_surrogate, best_seq = s, shape.tolist()
                print(f"  New best (p={p:.3f}) surrogate={s:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – LP‑based refinement (only for moderate lengths)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    use_lp = (len(best_seq) <= 2048) and (remaining > 60.0)
    if use_lp:
        print("\n--- Stage 2: LP‑based refinement ---")
        a = np.array(best_seq, dtype=float)
        a = lp_refine(a, deadline, max_outer_iters=12, mix_factor=0.25)
        s = compute_score(a)
        print(f"  After LP surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping LP‑based refinement (either too large n or insufficient time)")

    # ------------------------------------------------------------------
    # Stage 3 – smooth‑max Adam (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print("\n--- Stage 3: Adam refinement (smooth‑max) ---")
        a_init = np.array(best_seq, dtype=float)
        # iteration budget proportional to remaining time
        max_iters = int(min(120_000, remaining * 2000))   # very rough heuristic
        a_refined = adam_optimize_smooth_max(
            a_init,
            deadline,
            max_iters=max_iters,
            tau_start=0.5,
            tau_end=2000.0,
            base_lr=0.5,
        )
        s = compute_score(a_refined)
        print(f"  After Adam surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a_refined.tolist()
    else:
        print("\nSkipping Adam refinement (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 4 – exact sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 25.0:
        print("\n--- Stage 4: sub‑gradient descent ---")
        a = np.array(best_seq, dtype=float)
        a = subgradient_descent_exact(
            a,
            deadline,
            max_iter=int(min(200_000, remaining * 1500)),
            step0=0.05,

        s = compute_score(a)
        print(f"  After sub‑gradient surrogate={s:.6f}")
        if s < best_surrogate:

    else:

    # ------------------------------------------------------------------

    if remaining > 15.0:
        print("\n--- Stage 5: exponentiated‑gradient polish ---")
        a = np.array(best_seq, dtype=float)
        a = eg_refine(

            max_iter=int(min(150_000, remaining * 1000)),
            eta0=0.25,

    else:

    # ------------------------------------------------------------------
    # Stage 6 – cheap stochastic local search
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:

        )

        print(f"  After local search surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping local random search (insufficient time)")

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        final_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
