"""
Simple yet effective multi‑phase search for a step‑function with a low
`evaluate_sequence` score.

Phases
------
1. **Initial seeding** – try a few hand‑crafted shapes (sine‑squared,
   polynomial “beta” families) and a few random Dirichlet draws.  Keep the
   best one (on the exact objective).

2. **Simulated annealing** – a temperature‑controlled random walk that
   moves a small fraction of mass between two positions (local or global)
   or applies a small smoothing filter.  Moves that worsen the score are
   sometimes accepted, allowing the search to escape shallow minima.

3. **Projected sub‑gradient refinement** – once the annealing has cooled,
   perform many sub‑gradient steps on the true max‑objective
   `2·n·max(conv(a,a))`.  The gradient is cheap to compute from the index
   attaining the maximal convolution value; after each step the vector is
   projected back onto the probability simplex.

4. **Greedy hill‑climb polishing** – a short stochastic local‑search that
   only accepts improving moves (swap or small mass transfer) to polish
   the final answer.

All work respects the 1000 s wall‑clock budget; intermediate best scores
are printed for monitoring.  The returned value is a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Phase helpers
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """
    Sine‑squared shape a_i = sin^2(pi*(i+1)/(n+1)), normalised to sum 1.
    """
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p .
    p can be negative (mass at edges) or positive (mass at centre).
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Return a random probability vector drawn from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a

def simulated_annealing(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    total_time: float,
) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk on the simplex.
    Returns the best vector encountered and its exact score.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature – a few percent of the current objective
    T0 = max(1e-9, cur_score * 0.04)
    step_initial = 0.35
    min_step = 5e-6

    # pre‑allocate a small smoothing kernel (used rarely)
    smooth_kernel = np.ones(5) / 5.0

    it = 0
    while time.time() < deadline - 2.0:
        it += 1
        elapsed = time.time() - (deadline - total_time)
        frac = elapsed / total_time
        # linear cooling schedule
        T = max(1e-12, T0 * (1.0 - frac))
        step_scale = max(min_step, step_initial * (1.0 - frac))

        r = np.random.rand()
        if r < 0.65:
            # --- local mass transfer (to a neighbour) ---
            i = np.random.randint(n)
            if cur_a[i] <= 0.0:
                continue
            # neighbour (wrap around)
            if np.random.rand() < 0.5:
                j = (i + 1) % n
            else:
                j = (i - 1) % n
            if i == j:
                continue
            delta = step_scale * np.random.rand() * cur_a[i]
        elif r < 0.85:
            # --- global mass transfer ---
            i = np.random.randint(n)
            if cur_a[i] <= 0.0:
                continue
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = step_scale * np.random.rand() * cur_a[i]
        else:
            # --- smoothing move (rare) ---
            a_smooth = np.convolve(cur_a, smooth_kernel, mode="same")
            s = a_smooth.sum()
            if s == 0.0:
                continue
            a_smooth /= s
            new_score = compute_score(a_smooth)
            if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / T):
                cur_a[:] = a_smooth
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = cur_a.copy()
                continue
            else:
                continue

        # apply the transfer
        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = compute_score(cur_a)

        # acceptance test
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            # revert
            cur_a[i] += delta
            cur_a[j] -= delta

        # occasional progress report (every ~30 s)
        if it % 5000 == 0:
            print(f"[SA] elapsed {elapsed:.1f}s, best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient refinement
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    2·n·max(conv(a,a)).  Stops at the given deadline.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # evaluate convolution and locate a maximising index
        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        # gradient of the active max entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        # step size with sqrt decay
        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # small jitter to avoid stagnation every 1000 iterations
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb polishing (greedy improvement only)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Small stochastic moves (swap or local mass transfer) that are kept
    only when they improve the exact score.  Acts as a final polish.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                # keep non‑negative (numerical safety)
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # re‑normalise – the move conserves total mass, but rounding may drift
        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                       # become more conservative

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects the given time budget and
    returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a manageable length – 2048 works well within the budget
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 800.0:
        n = 4096
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2. Generate a handful of seed vectors (hand‑crafted + random)
    # ------------------------------------------------------------------
    seeds = []

    # a) deterministic sine‑squared shape
    seeds.append(generate_deterministic(n))

    # b) a few beta‑family shapes (p in modest range)
    for p in np.linspace(-2.5, 2.5, 7):
        seeds.append(generate_beta_family(n, p))

    # c) random Dirichlet draws (dense and sparse)
    for _ in range(4):
        seeds.append(random_dirichlet(n, conc=1.0))  # uniform on simplex
    for _ in range(4):
        seeds.append(random_dirichlet(n, conc=0.3))  # sparser

    # d) use the fallback if it matches the current length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float) / np.sum(best_seq))

    # Evaluate each seed on the exact objective and keep the best.
    for s in seeds:
        s = s.astype(float)
        s = project_to_simplex(s)  # guard against tiny numerical drift
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"Initial seed improvement: score = {best_val:.6f}")

    # Prepare the best vector for further optimisation
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    # ------------------------------------------------------------------
    # 3. Simulated annealing (major part of the budget)
    # ------------------------------------------------------------------
    print("\n[Phase 1] Simulated annealing")
    phase1_deadline = start_time + total_time * 0.58  # ~58 % of time
    cur_a, cur_score = simulated_annealing(
        cur_a, cur_score, phase1_deadline, total_time * 0.58
    )
    best_val = cur_score
    print(f"After SA: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n[Phase 2] Sub‑gradient refinement")
        phase2_deadline = deadline - 30.0
        cur_a, cur_score = subgradient_refine(
            cur_a, cur_score, phase2_deadline, max_iter=600_000, step0=0.07
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"After sub‑grad: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 3] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=1_200_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"After polishing: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Convert to list, final official evaluation, and return
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")
    return final_seq
