"""
Propose a non‑negative step sequence that minimizes the official
``evaluate_sequence`` objective.

Algorithm Overview
------------------

1. **Coarse parametric sweep** – systematic search over a few
   parametric families (Beta‑like, Γ‑shape, arcsine, mixtures) for several
   lengths (1024, 2048, 4096).  Every candidate is evaluated with a fast
   FFT‑based surrogate ``compute_score`` (identical to the official
   objective when the L1‑norm equals 1).  The best shape so far is kept.

2. **Continuous polishing** – starting from the best shape we run three
   refinements, each respecting the simplex constraint:
   * Exponentiated‑gradient (mirror descent with KL divergence).
   * Adam optimisation of a smooth log‑sum‑exp surrogate of the max.
   * Exact projected sub‑gradient descent on the true max‑convolution.

3. **Local hill‑climbing** – a cheap random‐move search (mass transfer
   between two coordinates) explores the neighbourhood of the polished
   vector.

All phases stop as soon as the 1000 s wall‑clock budget is exhausted,
guaranteeing that a valid (non‑empty) sequence is always returned.
"""

# ----------------------------------------------------------------------
# Imports and utilities
# ----------------------------------------------------------------------
import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose
    entries sum to 1.  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Parametric shape generators (all return a probability vector)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Discrete Beta(α,β)‑like density."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_gamma_shape(n: int, gamma: float, delta: float = 0.5,
                        eps: float = 0.0) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{-γ}.
    The classic arcsine corresponds to γ=0.5, δ≈0.5, ε≈0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Mirror‑descent with KL (exponentiated‑gradient) on the true max‑conv.
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step: float = 0.01,
) -> np.ndarray:
    """
    KL‑mirror descent (multiplicative updates) on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # mask of indices attaining the maximum (numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # gradient of max w.r.t. a (via reverse convolution)
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # multiplicative update
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d}  score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smooth log‑sum‑exp surrogate of the max.
# ----------------------------------------------------------------------
def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 2000.0,
    ramp_iters: int = 80_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam on the log‑sum‑exp (smooth‑max) approximation of max(conv).
    Operates in the log‑domain (soft‑max ensures the simplex).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ≥ 0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # temperature schedule (smooth‑max sharpening)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient wrt a : 2·(a ⋆ w_rev)
        rev = convolution_fft(a, w_norm[::-1])
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f}  score={compute_score(a):.6f}")

    return softmax(z)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true max‑convolution.
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d}  score={compute_score(a):.6f}")

    return a

            pass

    # ------------------------------------------------------------------
    # 1.  Systematic parametric sweep over a few lengths
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]   # modest sizes – still cheap to explore

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Exploring length {n} ---")

        # arcsine baseline
        cand = arcsine_initial(n)
        val = compute_score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.tolist()
            print(f"  New best (arcsine) score={val:.6f}")

        # --------------------------------------------------------------
        # Γ‑shape family (generalised arcsine)
        # --------------------------------------------------------------
        gammas = np.linspace(0.35, 0.65, 31)   # around 0.5
        deltas = np.linspace(0.0, 1.2, 13)    # shift of singularities
        for gamma in gammas:
            for delta in deltas:
                if time.time() >= deadline - 5.0:
                    break
                cand = generate_gamma_shape(n, gamma=gamma,
                                            delta=delta, eps=0.0)
                cand = np.clip(cand, 1e-15, None)
                cand = cand / cand.sum()
                val = compute_score(cand)
                if val < best_val:
                    best_val, best_seq = val, cand.tolist()
                    print(f"  New best (γ={gamma:.3f},δ={delta:.2f}) score={val:.6f}")

        # --------------------------------------------------------------
        # Symmetric Beta family (α = β)
        # --------------------------------------------------------------
        alphas = np.linspace(0.3, 1.5, 25)
        for a in alphas:
            if time.time() >= deadline - 5.0:
                break
            cand = generate_beta_shape(n, alpha=a, beta=a, eps=0.0)
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()
            val = compute_score(cand)
            if val < best_val:
                best_val, best_seq = val, cand.tolist()
                print(f"  New best (α=β={a:.3f}) score={val:.6f}")

        # --------------------------------------------------------------
        # Mixture of arcsine and uniform
        # --------------------------------------------------------------
        for w in np.linspace(0.0, 1.0, 11):
            if time.time() >= deadline - 5.0:
                break
            cand = generate_mixture_shape(n, weight_arc=w)
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()
            val = compute_score(cand)
            if val < best_val:
                best_val, best_seq = val, cand.tolist()
                print(f"  New best (mix w={w:.2f}) score={val:.6f}")

        print(f"Length {n} best so far: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Polishing phases (if we have a candidate)
    # ------------------------------------------------------------------
    if not best_seq:
        # very basic fallback
        n0 = 1024
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))

    a_arr = np.array(best_seq, dtype=float)

    # ---------- Exponentiated‑gradient polish ----------
    remaining = deadline - time.time()
    if remaining > 12.0:
        print("\n--- Exponentiated‑gradient polishing ---")
        a_arr = exponentiated_gradient_polish(
            a_arr,
            deadline=deadline,
            max_iter=int(min(250_000, remaining * 2000)),
            step=0.009,
        )
        val = compute_score(a_arr)
        if val < best_val:
            best_val, best_seq = val, a_arr.tolist()
            print(f"  After EG polish score={val:.6f}")

    # ---------- Adam on smooth‑max ----------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Adam (smooth‑max) polishing ---")
        eps = 1e-12
        z0 = np.log(np.clip(a_arr, eps, None))
        a_arr = smooth_adam_optimize(
            z0,
            deadline=deadline,
            n=a_arr.size,
            beta0=5.0,
            beta_max=2000.0,
            ramp_iters=100_000,
            lr=0.07,
        )
        val = compute_score(a_arr)
        if val < best_val:
            best_val, best_seq = val, a_arr.tolist()
            print(f"  After Adam polish score={val:.6f}")

    # ---------- Exact sub‑gradient descent ----------
    remaining = deadline - time.time()
    if remaining > 8.0:
        print("\n--- Exact sub‑gradient polishing ---")
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline=deadline,
            max_iter=int(min(350_000, remaining * 2000)),
            step0=0.07,
        )
        val = compute_score(a_arr)
        if val < best_val:
            best_val, best_seq = val, a_arr.tolist()
            print(f"  After sub‑gradient score={val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Local hill‑climbing using tiny mass transfers
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Local hill‑climbing ---")
        a = np.array(best_seq, dtype=float)
        a = a / a.sum()
        best_local_score = compute_score(a)
        best_local_a = a.copy()
        start_hc = time.time()
        while time.time() < deadline:
            # pick two distinct indices
            i, j = rng.integers(0, a.size, size=2)
            if i == j:
                continue
            # amount to move (relative to source mass)
            delta = min(a[i], 1e-4) * rng.random()
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # maintain simplex automatically
            sc_new = compute_score(a_new)
            if sc_new < best_local_score:
                best_local_score = sc_new
                best_local_a = a_new.copy()
                a = a_new
                # occasional progress report
                if int(time.time() - start_hc) % 10 == 0:
                    print(f"    HC improved score={sc_new:.6f}")
            else:
                # reject move (simple hill‑climb)
                pass

        if best_local_score < best_val:
            best_val, best_seq = best_local_score, best_local_a.tolist()
            print(f"  After hill‑climbing best score={best_local_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final official evaluation (logging only)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
