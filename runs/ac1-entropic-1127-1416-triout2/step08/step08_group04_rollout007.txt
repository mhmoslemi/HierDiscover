The evaluation function  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2}
\]

is *scale invariant*: multiplying the whole vector by a constant does not change the
score.  
Therefore we may (and will) keep every candidate on the probability simplex  

\[
\sum_i a_i = 1,\qquad a_i\ge 0 .
\]

On the simplex the official score reduces to the cheap surrogate  

\[
\text{surrogate}(a)=2\,n\;\max (a*a) .
\]

The optimiser works in three phases

1. **Explorative shape search** – a fast random scan over a rich family of
   parameterised step‑functions (Beta‑shapes, mixtures with a uniform part,
   an ``ε``‑regularisation of the singularities).  The best shape found
   is kept as a seed.

2. **Continuous refinement** – starting from the seed we run three
   deterministic optimisers, all respecting the simplex:
   * **smooth‑max Adam** – a soft‑max approximation of the `max` is
     differentiable; we use Adam with a temperature schedule that finally
     reaches a very large value (τ≈2000) so the true maximum is
     approximated accurately.
   * **projected sub‑gradient descent** – the exact (non‑smooth) maximum is
     used; a classical √t‑step‑size and Euclidean projection onto the simplex
     give a reliable polishing stage.
   * **exponentiated‑gradient (mirror descent)** – a KL‑proximal step that
     often removes the remaining ridge‑like artefacts.

3. **Stochastic hill climbing** – tiny random mass transfers between coordinates
   are accepted only if they improve the surrogate.  This cheap local
   search can escape tiny basins that the deterministic stages might get stuck
   in.

All stages stop as soon as the global wall‑clock deadline (default 1000 s)
is reached, guaranteeing that a non‑empty sequence is always returned.
The final sequence is evaluated with the official `evaluate_sequence`
function and printed.

The code below follows the description; only top‑level functions are used,
no closures or lambdas are required, and everything respects the time limit.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex; the surrogate is
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (parameter families used for seeding)
# ----------------------------------------------------------------------


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.full(n, 1.0 / n)


def arcsine_shape(n: int, eps: float = 1e-12) -> np.ndarray:
    """Discrete arcsine density (Beta(0.5,0.5)) with a small epsilon."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def beta_shape(n: int, a: float, b: float, eps: float = 1e-12) -> np.ndarray:
    """
    Discretised Beta(α,β) density on {0,…,n‑1}.
    α,β > 0.  eps prevents division by zero.
    """
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n
    # Guard against x=0 or 1 when α or β < 1
    x = np.clip(x, eps, 1.0 - eps)
    vals = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return uniform_shape(n)
    return vals / total


def mixture_shape(n: int,
                  base: np.ndarray,
                  weight: float) -> np.ndarray:
    """
    Simple convex mixture `weight*base + (1-weight)*uniform`.
    Returned vector is re‑normalised to sum‑to‑1.
    """
    uni = uniform_shape(n)
    mix = weight * base + (1.0 - weight) * uni
    return mix / mix.sum()

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 2000.0,
                             base_lr: float = 0.6) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max approximation of `max(conv)`.
    The optimisation variable are logits w so that a = softmax(w).
    The temperature τ is increased linearly from `tau_start` to `tau_end`.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a (always on the simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)  # length L

        # smooth‑max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 250_000,
                              step0: float = 0.08) -> np.ndarray:
    """
    Projected Euclidean sub‑gradient descent on the *true* max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # mask of (almost) maximal entries – usually just one index
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 180_000,
              eta0: float = 0.28) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent w.r.t. KL) polishing.
    Guarantees that the iterate always stays on the simplex.
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        # renormalise (always on the simplex)
        total = a.sum()
        if total == 0.0:
            a = uniform_shape(n)
        else:
            a /= total

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 30_000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Very cheap stochastic hill climbing:
    move a tiny amount of mass from one coordinate to another,
    accepting only improving (surrogate) moves.
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.2 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1

    return a_cur

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine performs a fast multi‑scale random shape search,
    followed by smooth‑max Adam optimisation, projected sub‑gradient polishing,
    exponentiated‑gradient refinement and a final cheap stochastic local search.
    All stages respect a global 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final official evaluation
    deadline = start_time + budget_s - 1.0

    best_seq: List[float] = []
    best_surrogate: float = float("inf")
    best_official: float = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – random shape search (cheap, many candidates)
    # ------------------------------------------------------------------
    candidate_ns = [1024, 2048, 4096, 8192]
    n_trials_per_n = 1200   # total ~ 5 000 evaluations → a fraction of a second

    print("\n--- Stage 1: random shape search ---")
    for n in candidate_ns:
        print(f"  testing length n={n}")

        # trivial baselines
        for base_name, base_vec in [("uniform", uniform_shape(n)),
                                    ("arcsine", arcsine_shape(n))]:
            s = compute_score(base_vec)
            if s < best_surrogate:
                best_surrogate = s
                best_seq = base_vec.tolist()
                print(f"    new best ({base_name}) surrogate={s:.6f}")

        # random Beta families + mixtures
        for _ in range(n_trials_per_n):
            # draw shape parameters
            a = rng.uniform(0.12, 0.88)          # α
            b = rng.uniform(0.12, 0.88)          # β
            eps = 10 ** rng.uniform(-16, -6)     # regularisation for the singularity
            shape_type = rng.integers(0, 3)     # 0: pure Beta, 1: mix with uniform, 2: mix arcsine+uniform

            if shape_type == 0:
                vec = beta_shape(n, a, b, eps)
            elif shape_type == 1:
                w = rng.random()
                pure = beta_shape(n, a, b, eps)
                vec = mixture_shape(n, pure, w)
            else:
                w = rng.random()
                arc = arcsine_shape(n, eps)
                vec = mixture_shape(n, arc, w)

            s = compute_score(vec)
            if s < best_surrogate:
                best_surrogate = s
                best_seq = vec.tolist()
                print(f"    new best (rand) surrogate={s:.6f}")

        # early exit if we are already out of time
        if time.time() >= deadline - 30.0:
            print("    time is low – break shape search")
            break

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max Adam refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0:
        print("\n--- Stage 2: Adam refinement (smooth‑max) ---")
        a_init = np.array(best_seq, dtype=float)
        a_refined = adam_optimize_smooth_max(
            a_init,
            deadline,
            max_iters=350_000,
            tau_start=0.5,
            tau_end=2000.0,
            base_lr=0.6,
        )
        s = compute_score(a_refined)
        print(f"    after Adam surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a_refined.tolist()
    else:
        print("\nSkipping Adam refinement (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        print("\n--- Stage 3: exact sub‑gradient descent ---")
        a = np.array(best_seq, dtype=float)
        a = subgradient_descent_exact(
            a,
            deadline,
            max_iter=300_000,
            step0=0.08,
        )
        s = compute_score(a)
        print(f"    after sub‑gradient surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping sub‑gradient descent (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 4 – exponentiated‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0:
        print("\n--- Stage 4: exponentiated‑gradient polish ---")
        a = np.array(best_seq, dtype=float)
        a = eg_refine(
            a,
            deadline,
            max_iter=200_000,
            eta0=0.28,
        )
        s = compute_score(a)
        print(f"    after EG surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping EG polish (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 5 – cheap stochastic local search
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        print("\n--- Stage 5: local stochastic hill climbing ---")
        a = np.array(best_seq, dtype=float)
        a = local_random_search(
            a,
            deadline,
            max_trials=40_000,
            step_scale=0.001,
            rng=rng,
        )
        s = compute_score(a)
        print(f"    after local search surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping local search (insufficient time)")

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
    except Exception as exc:
        print(f"Official evaluation failed: {exc}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        final_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
