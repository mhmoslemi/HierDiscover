official ``evaluate_sequence`` objective.

Algorithm
---------

1. **Randomised parametric seeding** – sample thousands of candidate
   shapes (beta, power‑law, γ‑shape, mixtures, Dirichlet, triangular,
   arcsine, uniform).  All candidates are normalised to a probability
   vector (sum = 1) and scored with the cheap ``compute_score`` routine.
   The best candidate seen so far is retained.

2. **Smooth–max refinement with L‑BFGS** – the max‑convolution is
   approximated by a log‑sum‑exp (temperature = β).  Using the analytic
   gradient we run a few L‑BFGS optimisations for an increasing schedule
   of β (5 → 20 → 80 → 300 → 800).  Each stage is limited by the remaining
   wall‑clock time.

3. **Exact projected sub‑gradient descent** – the final shape is polished
   on the true non‑smooth objective with a classic sub‑gradient method

The routine respects the 1000 s budget, always returns a non‑empty

# Convolution utilities (FFT, fast score)

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a probability vector.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def generate_arcsine_shape(n: int, rng: np.random.Generator) -> np.ndarray:  # noqa: D401
    """Arcsine‑like envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Asymmetric beta‑like shape."""
    alpha = 10 ** rng.uniform(-1.0, 0.7)   # ≈[0.1,5]
    beta = 10 ** rng.uniform(-1.0, 0.7)
    eps = rng.uniform(0.0, max(0.5, 0.02 * n))
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        base = np.full(n, 1.0 / n)
    else:
        base = base / total
    return base


def generate_power_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law shape ”1 / ((i+½)(n‑½‑i)+ε)^p”. """
    p = rng.uniform(0.2, 0.9)            # around the arcsine exponent 0.5
    eps = rng.uniform(0.0, 2.0)
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


def generate_gamma_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^‑γ.
    """
    gamma = rng.uniform(0.25, 0.75)      # around 0.5
    delta = rng.uniform(0.1, 1.0)
    eps = rng.uniform(0.0, 0.5)
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


def generate_mixture_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    w = rng.uniform(0.0, 1.0)
    arc = generate_arcsine_shape(n, rng)
    uni = np.full(n, 1.0 / n)
    mix = w * arc + (1.0 - w) * uni
    mix /= mix.sum()
    return mix

    conc = rng.uniform(0.1, 2.0)  # smaller → spikier
    a = rng.dirichlet(np.full(n, conc))
    return a

    i = np.arange(n, dtype=float)

# Simplex utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.08,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on   f(a) = 2·n·max(conv(a,a)),
    keeping a on the probability simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # indices where the max is (within numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max‑convolution
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score(a):.6f}")

    return a

# Smooth‑max objective (for L‑BFGS) and its gradient
# ----------------------------------------------------------------------
def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_obj(z: np.ndarray, beta: float, n: int):
    """
    Smoothed version of the objective using log‑sum‑exp at temperature β.
    Returns (loss, gradient) where gradient is w.r.t. the log‑variables z.
    """
    a = softmax(z)                       # a ≥ 0, Σa = 1
    conv = convolution_fft(a, a)
    maxc = float(np.max(conv))

    shifted = conv - maxc
    w = np.exp(beta * shifted)
    w_sum = np.sum(w)
    # smooth max = maxc + (1/β)·log Σexp(β·(conv‑maxc))
    smooth_max = maxc + (1.0 / beta) * np.log(w_sum)
    loss = 2.0 * n * smooth_max

    w_norm = w / w_sum
    rev = convolution_fft(a, w_norm[::-1])
    grad_a = 2.0 * rev[n - 1 : n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)

    return loss, grad_z

    best_val = float("inf")
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Random parametric search
    # ------------------------------------------------------------------
    n = 2048
    generators = [
        generate_arcsine_shape,
        generate_beta_shape,
        generate_power_shape,
        generate_gamma_shape,
        generate_mixture_shape,
        generate_dirichlet_shape,
        generate_triangular_shape,
        generate_revtriangular_shape,
        generate_uniform_shape,
    ]

    iters = 0
    # allocate roughly 60 % of the budget to raw exploration
    exploration_deadline = start + 0.60 * (deadline - start)
    while time.time() < exploration_deadline:
        iters += 1
        gen = rng.choice(generators)
        cand = gen(n, rng)

        # guard against pathological values
        cand = np.clip(cand, 1e-15, None)
        cand /= cand.sum()

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.tolist()
            print(f"New best after {iters:,d} iters: score = {val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Smooth‑max L‑BFGS refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if best_seq and _SCIPY_AVAILABLE and time.time() < deadline - 10.0:
        print("\n--- L‑BFGS refinement (smooth‑max) ---")
        a = np.array(best_seq, dtype=float)

        # temperature schedule – increasing β tightens the approximation
        beta_schedule = [5.0, 20.0, 80.0, 300.0, 800.0]
        for beta in beta_schedule:
            if time.time() >= deadline - 5.0:
                break
            # initialise log‑variables
            eps = 1e-12
            z0 = np.log(np.clip(a, eps, None))

            # decide a modest number of L‑BFGS iterations (bounded by time)
            remaining = deadline - time.time()
            max_iter = int(min(2500, remaining * 200))  # ~0.005 s per iter

            res = minimize(
                fun=smooth_obj,
                x0=z0,
                args=(beta, n),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter, "disp": False},
            )
            a = softmax(res.x)
            val = compute_score(a)
            if val < best_val:
                best_val, best_seq = val, a.tolist()
                print(f"  L‑BFGS β={beta:.0f} improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final exact projected sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Final sub‑gradient polishing ---")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline=deadline,
            max_iter=300_000,
            step0=0.08,
        )
        val = compute_score(a_arr)
        if val < best_val:
            best_val, best_seq = val, a_arr.tolist()
            print(f"After final sub‑gradient score = {val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty valid sequence
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # 5.  Official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
