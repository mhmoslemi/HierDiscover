"""
Propose a step‑function sequence that minimises

    2·n·max(convolve(a,a)) / (sum(a)**2)

under the non‑negative simplex constraint.

Algorithm
---------
1.  Scan several problem sizes (1024…8192).  For each length try a
    variety of analytically‑motivated families:
        – symmetric product a_i ∝ [(i+½)(n‑½‑i)]^{-α}
        – left‑heavy power‑law a_i ∝ (i+½)^{-β}
        – exponential “U” a_i ∝ exp(-γ·min(i,n‑1‑i))
        – convex mixtures of the uniform vector with the exponential U.
    The parameter grids are deliberately finer (α,β up to 12, γ up to 6)
    than in the reference implementation.

2.  Generate many random initial vectors (monotone, sparse, random U‑shapes)
    and keep the best candidate seen so far (exact score = 2·n·max(conv)).

3.  Refine the current best vector with a smoothed‑max optimisation.
    The objective
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    is differentiable; we optimise it on the simplex via the
    soft‑max parametrisation a = softmax(z) using Adam.
    The smoothing parameter β is increased exponentially from
    β_start to β_end, letting the optimiser focus on the true max
    as the run proceeds.

4.  Perform a cheap projected sub‑gradient descent on the true
    max‑objective (step size ∝ 1/√t).  This further reduces the peak
    of the convolution.

5.  Symmetrise the result (a ← (a+a[::-1])/2) and keep the better of
    the two – many optimal solutions are symmetric.

6.  A final stochastic hill‑climb with a tiny simulated‑annealing
    rule (swap / mass‑transfer moves) is run for the remaining time.

All stages respect the global wall‑clock budget (default 1000 s).  The
function returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a=1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # safety for overflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """
    Convex mixture of the uniform vector and the U‑shape.
    lam∈[0,1] is the weight of the uniform component.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    # Both components already sum to 1 → mixture also sums to 1.
    a = lam * uniform + (1.0 - lam) * u_shape
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform prob. vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Sample a random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 5.0,
                         beta_end: float = 150.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing β is increased exponentially from beta_start to beta_end.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- schedule for β (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- current a and its convolution -----
        a = softmax(z)
        conv = convolution_fft(a, a)            # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                # smooth max‑weights

        # ----- gradient wrt a -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 2n‑1
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ----- back‑prop through soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional exact evaluation -----
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheapest iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method for
          2·n·max(a*a)
    using a diminishing stepsize ∝ 1/√t.  The iterate is projected onto
    the probability simplex after each step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max‑entry (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Long random walk on the simplex.
    – 0.5 probability: swap two entries.
    – 0.5 probability: transfer a fraction of mass from one index to another.
    A tiny temperature schedule permits occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature – order of magnitude of current best
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            # accept
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            max_iters=2_000_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Symmetrisation test
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: if nothing succeeded, fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
