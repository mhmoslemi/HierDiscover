def upsample_sequence(seq, new_n):
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.maximum(ups, 1e-16)
    ups /= ups.sum()
    return ups
```

"""
Propose a step‐function (non‑negative sequence) that minimises
``evaluate_sequence``.  

The algorithm consists of three stages:

1. **Parametric random search** –  
   Sample a few thousand simple symmetric shapes
   `a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}` possibly mixed with a uniform part.
   The parameters (`p, ε, w`) are drawn at random; the best shape
   (according to the exact objective `2·n·max(conv(a,a))`) is kept.

2. **Local refinement** –  
   Starting from the best sampled shape, run a short Adam optimisation
   on the smooth log‑sum‑exp surrogate of the max‑convolution, followed
   by a few hundred thousand exact sub‑gradient steps (projected onto the
   probability simplex).  Both steps respect the simplex constraint and
   improve the objective further.

3. **Upsampling & final polishing** –  
   The refined shape is linearly up‑sampled to a larger length
   (`n = 8192`), where the bound typically improves, and a small amount of
   sub‑gradient descent is performed again.  The best sequence found at
   any stage is returned (as a list of Python floats).  If nothing is
   found, a uniform fallback sequence is used.

The whole routine respects a 1000 s wall‑clock budget, prints progress,
and always returns a valid sequence.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast exact evaluation for a probability vector (sum == 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generation
# ----------------------------------------------------------------------


def shape_parametric(n: int, p: float, eps: float, w: float) -> np.ndarray:
    """
    Symmetric analytic shape:
        base_i = ((i+0.5)*(n-0.5-i) + eps) ** (-p)
        shape = w * base + (1-w) * uniform
    Normalised to sum == 1.
    """
    i = np.arange(n, dtype=float)
    denom = (i + 0.5) * (n - 0.5 - i) + eps
    # avoid negative or zero denom (should never happen)
    denom = np.maximum(denom, 1e-30)
    base = np.power(denom, -p)
    uniform = np.full_like(base, 1.0 / n)
    mix = w * base + (1.0 - w) * uniform
    # Normalise to a probability vector
    total = mix.sum()
    if total <= 0 or not np.isfinite(total):
        return uniform
    return mix / total


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    # Avoid exact zeros (helps later multiplicative updates)
    ups = np.maximum(ups, 1e-16)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser (identical to the reference version)
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 500.0,
    ramp_iters: int = 60000,
    lr: float = 0.08,
) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    z0 : ndarray
        Initial log‑probabilities (log‑softmax coordinates).
    deadline : float
        Absolute time at which the optimiser must stop.
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate for Adam.

    Returns
    -------
    a : ndarray
        Optimised probability vector (sum = 1, a_i ≥ 0).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # simplex point
        conv = convolution_fft(a, a)       # linear conv, length 2n‑1
        mx = float(np.max(conv))

        # temperature β(t)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)         # weights, sum to 1 after normalisation
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central n entries

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output
        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final point on the simplex
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.08,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a

# Main search entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 43, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low evaluate_sequence
    value.  The routine respects a 1000 s wall‑clock budget and always
    returns a valid (non‑empty) sequence.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0.  Use a previously discovered sequence if it exists
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Choose a working length (moderately large but still cheap)
    # --------------------------------------------------------------
    n_small = 4096
    if time.time() > deadline - 300:
        # if we are already close to the end, shrink the problem size
        n_small = 2048

    print(f"\n=== Starting parametric random search (n={n_small}) ===")

    # ------------------------------------------------------------------
    # 1a.  Parametric random sampling of simple symmetric shapes
    # ------------------------------------------------------------------
    # parameter ranges
    p_min, p_max = 0.15, 2.0            # exponent for the denominator
    log_eps_min, log_eps_max = -12.0, 1.0   # epsilon in log10 space: 1e-12 … 10
    w_min, w_max = 0.0, 1.0             # mixture weight with uniform

    # always keep a few deterministic seeds
    seed_shapes = [
        shape_parametric(n_small, p=0.5, eps=0.25, w=1.0),               # classic arcsine
        np.full(n_small, 1.0 / n_small),                               # uniform
    ]
    best_a = None
    best_score = float("inf")
    for a in seed_shapes:
        s = compute_score(a)
        if s < best_score:
            best_score, best_a = s, a

    # allocate about 2 minutes (or however much left) to random search
    param_deadline = min(deadline - 90.0, start + 150.0)
    max_samples = 20_000
    samples = 0

    while time.time() < param_deadline and samples < max_samples:
        p = rng.uniform(p_min, p_max)
        eps = 10 ** rng.uniform(log_eps_min, log_eps_max)
        w = rng.uniform(w_min, w_max)
        a = shape_parametric(n_small, p, eps, w)
        s = compute_score(a)
        if s < best_score:
            best_score, best_a = s, a
            print(f"    New parametric best: score={s:.6f} (p={p:.3f}, eps={eps:.3e}, w={w:.3f})")
        samples += 1

    # store the current best
    if best_a is not None:
        best_seq = best_a.tolist()
        best_val = best_score
        print(f"\nBest after parametric search: score={best_val:.6f}")

    # --------------------------------------------------------------
    # 2.  Local refinement on the small grid with Adam (smooth‑max)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 60.0:
        print("\n=== Adam refinement on small grid ===")
        a0 = np.array(best_seq, dtype=float)
        eps = 1e-12
        z0 = np.log(a0 + eps)
        adam_deadline = deadline - 30.0
        a_adam = smooth_adam_optimize(
            z0,
            adam_deadline,
            n=n_small,
            beta0=5.0,
            beta_max=600.0,
            ramp_iters=100_000,
            lr=0.09,
        )
        val_adam = compute_score(a_adam)
        print(f"    Adam refined score = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.tolist()
            print(f"    *** New global best after Adam: {best_val:.6f} ***")
        else:
            # keep the Adam output as a starting point for the next stage
            a0 = a_adam
    else:
        a0 = np.array(best_seq, dtype=float)

    # --------------------------------------------------------------
    # 3.  Exact sub‑gradient descent on the small grid
    # --------------------------------------------------------------
    if a0.size > 0 and time.time() < deadline - 30.0:
        print("\n=== Sub‑gradient descent on small grid ===")
        subgrad_deadline = deadline - 10.0
        a_sub = subgradient_descent_exact(
            a0,
            subgrad_deadline,
            max_iter=300_000,
            step0=0.09,
        )
        val_sub = compute_score(a_sub)
        print(f"    Sub‑gradient refined score = {val_sub:.6f}")
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.tolist()
            print(f"    *** New global best after sub‑gradient: {best_val:.6f} ***")
        a0 = a_sub

    # --------------------------------------------------------------
    # 4.  Upsample to a larger length and polish again
    # --------------------------------------------------------------
    if a0.size > 0 and time.time() < deadline - 5.0:
        n_big = 8192
        print(f"\n=== Upsampling to length {n_big} and final polishing ===")
        a_big = upsample_sequence(a0, n_big)

        # a quick sub‑gradient polish on the larger grid
        big_deadline = deadline
        a_big = subgradient_descent_exact(
            a_big,
            big_deadline,
            max_iter=200_000,
            step0=0.07,
        )
        val_big = compute_score(a_big)
        print(f"    Upsampled final score = {val_big:.6f}")
        if val_big < best_val:
            best_val, best_seq = val_big, a_big.tolist()
            print(f"    *** New global best after upsample polish: {best_val:.6f} ***")

    # --------------------------------------------------------------
    # 5.  Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    # final official evaluation (just to be safe)
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Official best score = {final_val:.6f}")
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = best_val

    return best_seq
