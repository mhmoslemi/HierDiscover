def solve_lp_conditional_gradient(f, M):
    n = len(f)
    # define variable g of length n
    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    constraints = []
    # conv constraints: for k in 0..2n-2, sum_{i=max(0,k-(n-1))}^{min(n-1,k)} f[i] * g[k-i] <= M
    # Equivalent to linear combination of g entries.
    for k in range(2*n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # Terms: f[i] * g[k-i] where i in range
        # Let indices j = k-i -> variable index within [0, n-1], find mapping.
        # We can build expression using cp.sum(cp.multiply(f_sub, g_sub)) maybe.
        # Determine indices j = k - i for i in range.
        # Use list comprehension
        coeffs = []
        idxs = []
        for i in range(i_start, i_end+1):
            j = k - i
            # coefficient f[i] on g[j]
            coeffs.append(f[i])
            idxs.append(j)
        # Build expression: sum coeffs[j] * g[idxs[j]]
        expr = cp.sum(cp.multiply(np.array(coeffs), g[idxs]))
        constraints.append(expr <= M)
    # objective maximize sum(g)
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    # Solve with ECOS default:
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    # In case of failure, fallback to uniform
    if g.value is None:
        return np.full(n, 1.0 / n)
    else:
        return np.array(g.value).flatten()

def solve_lp_conditional_gradient(f, M):
    n = len(f)
    import cvxpy as cp
    F = np.zeros((2*n-1, n))
    for i in range(n):
        F[i:i+n, i] = f
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [F @ g <= M])
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)  # may add abstol
    if g.value is None:
        return np.full(n, 1.0 / n)
    return np.array(g.value)

"""
Improved search for a low‑score step‑function.

Main ideas
----------

* **Broad initial seeding** – uniform, arcsine, a family of Beta‑shaped
  distributions and a few random Dirichlet draws are examined on several
  lengths (1024…8192).  The best surrogate `2·n·max(conv(a,a))` is kept.

* **Frank–Wolfe (conditional‑gradient) refinement** – for the current
  candidate `f` we solve the linear program  

        maximise   Σ g_i
        subject to  g ≥ 0 ,  conv(f,g) ≤ M   (M = max conv(f,f))

  (the convolution constraints are linear in `g`).  The solution `g`
  points towards a direction that can lower the max‑convolution.
  A short line‑search determines the optimal convex combination
  `f←(1‑α)f+αg`.  The LP is solved with *cvxpy* (ECOS).  This step is
  cheap for n≈1024 and often yields a noticeable drop.

* **Adam on a smooth‑max surrogate** – a soft‑max approximation of the
  max‑convolution is minimised with Adam (logits → softmax
  parametrisation).  The temperature is increased from τ≈0.5 to τ≈500,
  permitting a gradual tightening of the surrogate.

* **Projected sub‑gradient descent** – the true (non‑smooth) objective
  is then polished with a Euclidean projected sub‑gradient method.

* **Exponential‑gradient polish & cheap stochastic hill‑climb** – a few
  passes of mirror‑descent (KL‑projection) followed by tiny random mass
  transfers finish the optimisation.

All stages respect a global wall‑clock budget (default 1000 s).  The
function returns a non‑empty list of non‑negative numbers whose sum is
1; the official score is printed at the end.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate of the official score (assumes Σa = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Generalised arcsine shape: a_i ∝ ((i+½)*(n‑½‑i)+ε)^(‑p)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, eps: float = 1e-12) -> np.ndarray:
    """Symmetric Beta(α,α) pdf discretised on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    x = i / (n - 1)  # in [0,1]
    vals = (x + eps) ** (alpha - 1) * (1 - x + eps) ** (alpha - 1)
    vals[0] = vals[1] * 0.5  # avoid exact zero at the boundaries
    vals[-1] = vals[-2] * 0.5
    return vals / vals.sum()


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if conc <= 0:
        # degenerate case → uniform
        return np.full(n, 1.0 / n)
    draws = rng.gamma(conc, 1.0, size=n)
    return draws / draws.sum()

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 500.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """Adam on a smooth‑max approximation of max(conv)."""
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)          # logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        conv = convolution_fft(a, a)

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp                # shape (L,)

        # gradient wrt a  (∂smoothmax/∂a)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)          # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wrt logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.07) -> np.ndarray:
    """Euclidean projected sub‑gradient descent on the true max‑convolution."""
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.25) -> np.ndarray:
    """Exponentiated‑gradient polish (mirror descent w.r.t. KL)."""
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 20_000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Very cheap stochastic hill climbing: move a tiny amount of mass
    from one coordinate to another, accepting only improving moves.
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.2 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1

    return a_cur

_has_cvxpy = False
try:
    import cvxpy as cp
    _has_cvxpy = True
except Exception:  # pragma: no cover
    pass


def solve_lp_conditional_gradient(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ g_i   s.t.   g ≥ 0 ,   conv(f,g) ≤ M .
    Returns the optimal g (as a NumPy array).  If cvxpy is unavailable
    or the solver fails, a uniform vector is returned.
    """
    n = f.shape[0]
    if not _has_cvxpy:
        return np.full(n, 1.0 / n)

    # Build the Toeplitz‑style matrix F where (F @ g)[k] = Σ_i f_i g_{k-i}
    rows = 2 * n - 1
    F = np.zeros((rows, n), dtype=float)
    for i in range(n):
        F[i:i + n, i] = f

    g = cp.Variable(n, nonneg=True)
    constraints = [F @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:  # pragma: no cover
        return np.full(n, 1.0 / n)

    if g.value is None:
        return np.full(n, 1.0 / n)
    return np.array(g.value).reshape(-1)


def frank_wolfe_refine(f: np.ndarray,

    A few Frank–Wolfe iterations using the LP above.

    """

        M = float(np.max(conv))

        # normalise g to a probability vector
        s = g.sum()
        if s == 0.0:
            break
        g = g / s

        # line‑search on the convex combination (1‑α)·cur + α·g
        best_alpha = 0.0
        best_score = compute_score(cur)
        best_vec = cur
        for α in line_alphas[1:]:
            cand = (1.0 - α) * cur + α * g
            cand = cand / cand.sum()
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_alpha = α
                best_vec = cand
        cur = best_vec
        print(f"    FW step {step+1}: α={best_alpha:.2f} surrogate={best_score:.6f}")

    return cur

# Main entry point
# =============================================================================
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine mixes a wide‑range grid search, a Frank–Wolfe LP step,
    Adam smooth‑max optimisation, projected sub‑gradient descent,
    exponential‑gradient polishing and a final stochastic local search.
    All phases respect a global 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    best_seq: List[float] = []
    best_surrogate: float = float("inf")
    best_official: float = float("inf")

    # ----------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = fallback
                best_surrogate = compute_score(np.array(fallback))
        except Exception as e:
            print(f"Fallback evaluation failed: {e}")

    # ----------------------------------------------------------------------
    # 1. Coarse multi‑scale shape search + random Dirichlet draws
    # ----------------------------------------------------------------------
    candidate_lengths = [1024, 2048, 4096, 8192]
    power_grid = np.linspace(0.15, 0.85, 15)          # exponent p for power‑shape
    beta_grid = [0.3, 0.5, 0.8, 1.5, 3.0]            # α for symmetric Beta
    conc_grid = [0.2, 0.5, 1.0, 2.0]                # Dirichlet concentration

    print("\n--- Stage 0: coarse grid + random Dirichlet search ---")
    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        print(f"Length n = {n}")

        # Uniform
        uniform = np.full(n, 1.0 / n)
        s = compute_score(uniform)
        if s < best_surrogate:
            best_surrogate, best_seq = s, uniform.tolist()
            print(f"  New best (uniform) surrogate={s:.6f}")

        # Arcsine
        arc = arcsine_initial(n)
        s = compute_score(arc)
        if s < best_surrogate:
            best_surrogate, best_seq = s, arc.tolist()
            print(f"  New best (arcsine) surrogate={s:.6f}")

        # Power‑shape family
        for p in power_grid:
            shape = generate_power_shape(n, p)
            s = compute_score(shape)
            if s < best_surrogate:
                best_surrogate, best_seq = s, shape.tolist()
                print(f"  New best (p={p:.3f}) surrogate={s:.6f}")

        # Beta‑shape family
        for a in beta_grid:
            shape = generate_beta_shape(n, a)
            s = compute_score(shape)
            if s < best_surrogate:
                best_surrogate, best_seq = s, shape.tolist()
                print(f"  New best (Betaα={a:.1f}) surrogate={s:.6f}")

        # Random Dirichlet draws
        for conc in conc_grid:
            # limit number of draws per (n,conc) to keep runtime modest
            for _ in range(20):
                cand = random_dirichlet(n, conc, rng)
                s = compute_score(cand)
                if s < best_surrogate:
                    best_surrogate, best_seq = s, cand.tolist()
                    print(f"  New best (Dirichlet α={conc:.1f}) surrogate={s:.6f}")

    # ----------------------------------------------------------------------
    # 2. Frank–Wolfe conditional‑gradient refinement (LP)
    # ----------------------------------------------------------------------
    remaining = deadline - time.time()
    if _has_cvxpy and remaining > 30.0:
        print("\n--- Stage 1: Frank–Wolfe LP refinement ---")
        a_fw = np.array(best_seq, dtype=float)
        a_fw = frank_wolfe_refine(a_fw, deadline, max_steps=6)
        s = compute_score(a_fw)
        print(f"  After Frank‑Wolfe surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a_fw.tolist()
    else:
        print("\nSkipping Frank‑Wolfe (no cvxpy or insufficient time)")

    # ----------------------------------------------------------------------
    # 3. Adam on smooth‑max surrogate (if time permits)
    # ----------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print("\n--- Stage 2: Adam smooth‑max optimisation ---")
        a_init = np.array(best_seq, dtype=float)
        a_adam = adam_optimize_smooth_max(
            a_init,
            deadline,
            max_iters=300_000,
            tau_start=0.5,
            tau_end=500.0,
            base_lr=0.5,
        )
        s = compute_score(a_adam)
        print(f"  After Adam surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a_adam.tolist()
    else:
        print("\nSkipping Adam optimisation (insufficient time)")

    # ----------------------------------------------------------------------
    # 4. Projected sub‑gradient descent (exact max‑conv)
    # ----------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        print("\n--- Stage 3: Sub‑gradient descent (exact max) ---")
        a = np.array(best_seq, dtype=float)
        a = subgradient_descent_exact(a, deadline, max_iter=250_000, step0=0.07)
        s = compute_score(a)
        print(f"  After sub‑gradient surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping sub‑gradient descent (insufficient time)")

    # ----------------------------------------------------------------------
    # 5. Exponentiated‑gradient polish
    # ----------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Stage 4: Exponentiated‑gradient polish ---")
        a = np.array(best_seq, dtype=float)
        a = eg_refine(a, deadline, max_iter=150_000, eta0=0.25)
        s = compute_score(a)
        print(f"  After EG surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping EG polish (insufficient time)")

    # ----------------------------------------------------------------------
    # 6. Cheap stochastic local search (hill climbing)
    # ----------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Stage 5: Local random hill climbing ---")
        a = np.array(best_seq, dtype=float)
        a = local_random_search(a, deadline, max_trials=30_000,
                                step_scale=0.001, rng=rng)
        s = compute_score(a)
        print(f"  After local search surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping local random search (insufficient time)")

    # ----------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ----------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = float("inf")

    # ----------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ----------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        final_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
