        seeds.append(generate_power_shape(n, p=0.5))  # maybe arcsine similar

        # Power law p
        for p in np.linspace(0.3, 1.2, 7):
            seeds.append(generate_power_shape(n, p))

        # random Dirichlet draws
        for _ in range(8):
            seeds.append(generate_random_dirichlet(n, rng))

        # Evaluate fast_score for each
        scored = [(fast_score(v), v) for v in seeds]

        top_k = 4
        top_seeds = [vec for _, vec in scored[:top_k]]

        # allocate per seed budget
        if len(top_seeds) == 0:
            continue
        per_seed_budget = max(10.0, (deadline - time.time()) / (len(top_seeds) * 3.0))
        # Ensure not exceeding remaining time
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline - 2.0:
                break
            print(f"Seed {idx+1}/{len(top_seeds)} (n={n}) budget {per_seed_budget:.1f}s")
            seed_deadline = time.time() + per_seed_budget

            # Initialize z0 from log of seed (avoid zero)

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def subgradient_descent_exact(init_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.07) -> np.ndarray:
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]
        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)
        # optional print
        if it % 20000 == 0:
            print(f"    Subgrad it={it:,d} surrogate≈{fast_score(a):.6f}")
    return a

"""
Heuristic optimisation for the step‑function objective.

The algorithm builds a rich set of simplex seeds (arcsine, Beta‑families,
power‑law, random Dirichlet draws), scores them with a cheap surrogate,
refines the most promising candidates with a continuation‑Adam optimiser
that minimises a smooth‑max (log‑sum‑exp) surrogate of the true objective,
polishes the result with exact sub‑gradient descent and finally performs a
lightweight random local search.  The whole process respects a total time
budget (default 1000 s) and always returns a valid non‑negative sequence.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Cheap surrogate (identical to the official objective without clipping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)^2 – the objective before clipping."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic generators for diverse starting shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape – an approximation of Beta(½,½)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    return vals / vals.sum()


def arcsine_offset_shape(n: int, delta: float) -> np.ndarray:
    """Arcsine‑type shape with a tunable offset δ (δ>0)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + delta) * (n - delta - i) + eps)
    vals = np.maximum(vals, 0.0)
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float = 0.0
) -> np.ndarray:
    """Beta(α,β) density discretised on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Shape proportional to 1/((i+½)*(n‑½‑i)+eps)^p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with random concentration)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)  # roughly 0.01 … 5
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1} (Euclidean projection)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Soft‑max for the simplex‑parameterisation used by Adam
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Continuation Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 8000.0,
    ramp_iters: int = 200_000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    Minimises L_β(a)=log‑sum‑exp(β·conv(a,a)) with a∈simplex via the
    soft‑max transform.  β is increased linearly from ``beta0`` to
    ``beta_max`` (continuation).  Adam is used for the inner optimisation.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a lives on the simplex
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # β schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights (log‑sum‑exp trick)
        shifted = conv - mx                  # ≤ 0
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of L_β w.r.t. a
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)                 # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]           # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(
                f"    Adam it={t:,d} β={beta:.1f} surrogate≈{fast_score(a):.6f}"
            )
    return _softmax(z)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.1,
) -> np.ndarray:
    """
    Direct sub‑gradient descent for f(a)=2·n·max(conv(a,a))/sum(a)^2.
    The iterates are projected onto the simplex after each step.
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution (see notes in the original code)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 50000 == 0:
            print(f"    Sub‑grad it={it:,d} surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Inexpensive random local search around a candidate
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.025,
) -> np.ndarray:
    """
    Pairwise mass‑transfer moves while time permits.
    Moves are kept only if they improve the cheap surrogate.
    """
    a = a0.copy()
    best_sc = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * rng.random() * max_transfer
        if eps <= 0.0:
            continue

        a[i] -= eps
        a[j] += eps
        # re‑normalise for numerical stability
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
            continue
        a /= s

        sc = fast_score(a)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert the change
            a[j] -= eps
            a[i] += eps
            a /= a.sum()
    return a
