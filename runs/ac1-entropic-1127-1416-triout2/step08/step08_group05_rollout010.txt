"""
Propose a low‐value step sequence.

The algorithm proceeds in three stages:

1. **Massive parametric seeding** – generate a few hundred symmetric
   probability vectors of length n (n is 8192 if time permits, otherwise 4096)
   using arcsine, beta, power‑law and mixture families.  All candidates are
   normalised to sum = 1 and their fast surrogate score
   ``2*n*max(conv(a,a))`` (identical to the official objective for sum = 1)
   is computed with an FFT‑based convolution.

2. **Smooth‑max Adam optimisation** – the best few seeds are refined with a
   stochastic Adam optimiser on a log‑sum‑exp (smooth‑max) surrogate of the
   true objective.  The temperature β is ramped from 5 up to 4000, which
   forces the optimisation to focus on the true peak.  Log‑softmax coordinates
   are symmetrised after each Adam update, guaranteeing a symmetric sequence
   (the optimum is known to be symmetric).

3. **KL‑mirror polishing & final sub‑gradient descent** – the best Adam
   result is polished with a multiplicative‑weights (exponentiated‑gradient)
   method on the exact max‑convolution objective, again enforcing symmetry.
   If time remains a few iterations of projected Euclidean sub‑gradient
   descent are performed.

All steps respect the global 1000 s wall‑clock budget and always return a
valid non‑negative sequence (sum ≈ 1, thus the official evaluation equals the
surrogate).  The routine also falls back to a simple uniform sequence if no
candidate is found.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate score for a probability vector (sum(a)=1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all symmetric, sum to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law envelope (inverse of a polynomial)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Helper – symmetry utilities
# ----------------------------------------------------------------------


def symmetrise_vector(v: np.ndarray) -> np.ndarray:
    """Enforce exact mirror symmetry (v_i = v_{n-1-i}) and re‑normalise."""
    v = (v + v[::-1]) / 2.0
    s = v.sum()
    if s > 0:
        v /= s
    return v


# ----------------------------------------------------------------------
# Smooth‑max surrogate optimisation with Adam (log‑softmax coordinates)
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 4000.0,
    ramp_iters: int = 100_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    z0 : ndarray
        Initial log‑probabilities (log‑softmax coordinates).
    deadline : float
        Absolute wall‑clock time when optimisation must stop.
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate.

    Returns
    -------
    a : ndarray
        Optimised probability vector (sum = 1, a_i ≥ 0) and symmetric.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        # ------- simplex point -------
        a = softmax(z)
        # enforce perfect symmetry on the probability vector
        a = symmetrise_vector(a)

        # ------- convolution & smooth max -------
        conv = convolution_fft(a, a)                     # length 2n‑1
        mx = float(np.max(conv))

        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        shifted = conv - mx                               # ≤ 0
        w = np.exp(beta * shifted)                        # weights (unnorm.)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # ------- gradient w.r.t. a -------
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)                  # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # ------- chain rule through soft‑max -------
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce symmetry on logits – guarantees a stays symmetric
        z = (z + z[::-1]) / 2.0

        if t % 5_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate={float(2*n*mx):.6f}")

    # final symmetric probability vector
    a = softmax(z)
    a = symmetrise_vector(a)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the exact objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.03,
) -> np.ndarray:
    """
    Mirror descent (KL divergence) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # active set – indices where conv attains (almost) the max
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # step size schedule (decaying)
        step = step0 / np.sqrt(it)
        a = a * np.exp(-step * subgrad)

        # keep on simplex and enforce symmetry
        a = symmetrise_vector(a)

        if it % 5_000 == 0:
            print(f"    EG it={it:,d} score={float(2*n*mx):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact Euclidean sub‑gradient descent (final polish)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.09,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    """
    a = symmetrise_vector(project_to_simplex(init_a.copy()))
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)
        a = symmetrise_vector(a)

        if it % 5_000 == 0:
            print(f"    SubGrad it={it:,d} score={float(2*n*mx):.6f}")

    return a


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Build a rich pool of analytic seeds
# ----------------------------------------------------------------------


def build_seed_pool(
    n: int,
    rng: np.random.Generator,
    best_sofar: List[float],
) -> List[np.ndarray]:
    """Return a list of candidate symmetric probability vectors of length n."""
    seeds: List[np.ndarray] = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),  # uniform
    ]

    # Random power‑law envelopes
    for _ in range(12):
        p = rng.uniform(0.1, 6.0)
        eps = rng.uniform(0.0, 1.0)
        seeds.append(generate_power_shape(n, p, eps))

    # Random beta‑like envelopes
    for _ in range(12):
        a = rng.uniform(0.3, 6.0)
        b = rng.uniform(0.3, 6.0)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        seeds.append(generate_beta_shape(n, a, b, eps))

    # Mixtures of arcsine & uniform
    for _ in range(6):
        w = rng.uniform(0.0, 1.0)
        seeds.append(generate_mixture_shape(n, w))

    # Upsampled current best (if any)
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        seeds.append(upsample_sequence(old, n))

    # Slight random perturbations of the best (if available)
    if best_sofar:
        for _ in range(4):
            pert = rng.normal(scale=0.02, size=n)
            cand = np.array(best_sofar, dtype=float) + pert
            cand = np.clip(cand, 0.0, None)
            if cand.sum() == 0:
                cand = np.full(n, 1.0 / n)
            else:
                cand /= cand.sum()
            cand = symmetrise_vector(cand)
            seeds.append(cand)

    return seeds

def propose_candidate(seed: int = 43, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value within the allotted 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  If a previous solution is present, try it first
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose problem size (larger size gives finer discretisation)
    # ------------------------------------------------------------------
    # try the larger size first; if time is already low we fall back to 4096
    if time.time() < deadline - 200.0:
        lengths = [8192, 4096]
    else:
        lengths = [4096]

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # a) Build a diverse seed pool
        # --------------------------------------------------------------
        seed_pool = build_seed_pool(n, rng, best_seq)

        # quick surrogate ranking (no optimisation yet)
        scored = [(compute_score(s), s) for s in seed_pool]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:7]]   # keep a handful of promising seeds

        # --------------------------------------------------------------
        # b) Allocate time per seed and run smooth‑max Adam
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        per_seed_time = max(5.0, remaining / (len(top_seeds) * 2.5))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 20.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_time:.1f}s")
            seed_deadline = min(time.time() + per_seed_time, deadline - 5.0)

            # initialise Adam on the log‑softmax of the seed
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=4000.0,
                ramp_iters=120_000,
                lr=0.07,
            )

            # official evaluation
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

        # --------------------------------------------------------------
        # c) KL‑mirror polishing on the best candidate for this length
        # --------------------------------------------------------------
        if best_seq and time.time() < deadline - 8.0:
            print("\nPolishing best candidate with exponentiated gradient...")
            a_arr = np.array(best_seq, dtype=float)
            polish_deadline = deadline - 3.0
            a_polish = exponentiated_gradient_polish(
                a_arr,
                polish_deadline,
                max_iter=180_000,
                step0=0.03,
            )
            try:
                val2 = evaluate_sequence(a_polish.tolist())
                print(f"   Polished (EG) score = {val2:.6f}")
                if val2 < best_val:
                    best_val, best_seq = val2, a_polish.tolist()
                    print(f"   *** New best after EG polish: {best_val:.6f} ***")
            except Exception as e:
                print(f"   EG polishing failed: {e}")

    # ------------------------------------------------------------------
    # 2.  Final global projected sub‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\nFinal global sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=250_000,
            step0=0.09,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty, valid return
    # ------------------------------------------------------------------
    if not best_seq:
        print("No candidate produced – falling back to uniform sequence.")
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
