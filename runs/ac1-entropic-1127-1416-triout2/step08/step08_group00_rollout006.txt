def targeted_hill_climb(a, deadline, max_steps=300000):
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    eps_frac = 0.1   # move up to 10% of source mass each step
    for step in range(max_steps):
        if time.time() >= deadline: break
        conv = convolution_fft(a, a)
        max_val = conv.max()
        k = int(np.argmax(conv))
        # compute gradient
        idx = np.arange(n)
        rev = k - idx
        mask = (rev >=0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]
        # Identify source and target
        source = np.argmax(g)  # largest contribution to max
        target = np.argmin(g)  # smallest
        if g[source] <= 0:  # degenerate (maybe all zero)
            # random move
            i, j = np.random.randint(n, size=2)
            source, target = i, j
        # ensure source != target
        if source == target:
            # pick random distinct
            candidates = np.arange(n)
            candidates = candidates[candidates != source]
            target = np.random.choice(candidates)
        # compute epsilon
        eps = eps_frac * a[source]
        if eps < 1e-12:
            eps = eps_frac * 1e-12  # small
        # compute delta in max conv
        delta = eps * (g[target] - g[source])
        if delta < -1e-12:
            a[source] -= eps
            a[target] += eps
            # clamp negative due to numerical
            if a[source] < 0:
                a[source] = 0.0
            # after move, recompute sum and renormalize small drift
            if step % 5000 == 0:
                s = a.sum()
                if abs(s-1.0) > 1e-12:
                    a /= s
        else:
            # no improvement: reduce step size and maybe random jitter
            eps_frac *= 0.9
            # optional random perturb
            if np.random.rand() < 0.01:
                i, j = np.random.randint(n, size=2)
                delta_rnd = eps_frac * a[i]
                a[i] -= delta_rnd
                a[j] += delta_rnd
        # periodic evaluation
        if step % 1000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # increase step size up to limit after improvement
                eps_frac = min(eps_frac * 1.1, 0.2)
    return best_a, best_score

"""
Optimisation of a non‑negative step‑function for the objective

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm follows three stages within the 1000 s wall‑clock limit:

1. **Smooth‑max (log‑sum‑exp) Adam optimisation** – many short random restarts.
   The surrogate becomes ever sharper (β annealed up to 2000) so the true
   max is approached without sacrificing smooth gradients.

2. **Projected sub‑gradient descent** – classic sub‑gradient on the exact
   max‑objective, useful for escaping the smooth‐max basin.

3. **Directed gradient hill‑climb** – each iteration identifies the current
   maximal convolution entry *k*.  The sub‑gradient at that entry is
   `g_i = 2·a_{k‑i}` (zero if out of range).  Mass is moved from the index
   with the largest `g_i` to the one with the smallest, which provably
   reduces the maximal convolution value when possible.  The step size
   (fraction of source mass) adapts automatically and the best vector
   (lowest exact score) is kept.

All vectors are kept on the probability simplex (non‑negative,
normalised to sum = 1), which respects the scaling‑invariance of the
objective.  The final returned list is a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power‑of‑two greater than or equal to *x*."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the used sizes)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective when *a* already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_start(n: int) -> np.ndarray:
    """
    Return a random probability vector of length *n*.
    With probability 0.5 it is dense (Dirichlet(1));
    otherwise it is sparse on about √n positions.
    """
    if np.random.rand() < 0.5:
        a = np.random.rand(n)
        a /= a.sum()
    else:
        k = max(2, int(np.sqrt(n)))
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k
        a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# 1. Smooth‑max optimisation (log‑sum‑exp) with Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a soft‑max surrogate of `max(conv(a,a))` using Adam.
    Returns the best *exact* score encountered together with the
    corresponding vector (already normalised to sum = 1).
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed surrogate
        conv = convolution_fft(a, a)       # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the log‑sum‑exp objective
        a_rev = a[::-1]
        # convolution of reversed a with weights gives (central) gradient
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Directed gradient hill‑climb (mass transfer based on max‑entry gradient)
# ----------------------------------------------------------------------


def gradient_hill_climb(
    start_a: np.ndarray,
    time_deadline: float,
    max_steps: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Local improvement that repeatedly moves a fraction of mass from the
    index contributing most to the current maximal convolution entry to the
    index contributing least.  The step‑size (fraction of source mass) is
    adapted automatically.  Returns the best vector seen.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    # initial scale – up to 20 % of the source mass may be transferred
    step_frac = 0.20

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        # current convolution and its maximiser
        conv = convolution_fft(a, a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient of the *max* entry (g_i = 2·a_{k‑i} if in range)
        idx = np.arange(n)
        rev = k - idx
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]

        # source = index with largest gradient, target = smallest
        source = int(np.argmax(g))
        target = int(np.argmin(g))

        # safeguard against degenerate cases
        if a[source] <= 0.0:
            nz = np.where(a > 0.0)[0]
            if nz.size == 0:
                break
            source = int(np.random.choice(nz))

        if source == target:
            # pick a random distinct pair
            cand = np.arange(n)
            cand = cand[cand != source]
            target = int(np.random.choice(cand))

        # amount of mass to move
        eps = step_frac * a[source]
        if eps < 1e-15:
            eps = step_frac * 1e-15

        # theoretical change in the maximal entry
        delta = eps * (g[target] - g[source])

        if delta < -1e-12:
            # improvement – accept the move
            a[source] -= eps
            a[target] += eps
            # non‑negativity guard against numerical drift
            if a[source] < 0.0:
                a[source] = 0.0
        else:
            # no improvement – shrink the step size
            step_frac *= 0.90
            # occasional random jitter to escape tiny plateaus
            if step % 2000 == 0:
                i, j = np.random.randint(n, size=2)
                eps_jit = step_frac * a[i]
                a[i] -= eps_jit
                a[j] += eps_jit
                a = project_to_simplex(a)

        # periodic exact evaluation (every few hundred iterations)
        if step % 500 == 0:
            cur_score = 2.0 * n * float(conv.max())  # same as compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # after a successful improvement make the step size a bit bolder
                step_frac = min(step_frac * 1.15, 0.40)

        # occasional renormalisation to counter tiny drift
        if step % 5000 == 0:
            s = a.sum()
            if abs(s - 1.0) > 1e-12:
                a /= s

    return best_a, best_score
