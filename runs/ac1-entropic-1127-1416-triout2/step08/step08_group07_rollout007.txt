    # constraints: A @ g <= M, where A matrix with rows k (0..2n-2), each row length n: A[k,i] = f[k-i] if 0 <= k-i < n else 0.

def lp_refine(a, deadline, max_iters=50, t=0.3):
    a = a.copy()
    a = a / a.sum()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        M = float(conv.max())
        # LP
        n = a.size
        # Build A matrix as described
        row_indices = []
        col_indices = []
        data = []
        for k in range(2*n-1):
            i_start = max(0, k-(n-1))
            i_end = min(k, n-1)
            # i in [i_start, i_end], coefficient for g_i = f[k - i]
            cols = np.arange(i_start, i_end+1, dtype=int)
            rows = np.full_like(cols, k)
            vals = a[k - cols]
            row_indices.append(rows)
            col_indices.append(cols)
            data.append(vals)
        # combine arrays
        import scipy.sparse as sp
        A = sp.csr_matrix((np.concatenate(data), (np.concatenate(row_indices), np.concatenate(col_indices))), shape=(2*n-1, n))
        b = np.full(2*n-1, M)
        # Solve LP
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g  # normalize to sum=1
        # mix
        a = (1 - t) * a + t * g
        # enforce sum=1
        a = a / a.sum()
        # optionally enforce symmetry
        a = (a + a[::-1]) / 2
        a = a / a.sum()
        # Print progress
        est_score = fast_score(a)
        print(f"    LP iter {it+1}, M={M:.6e}, surrogate≈{est_score:.6f}")
    return a
```

"""
Improved search for a non‑negative step sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a blend of three ideas:

1. **Rich analytic seeds** – arcsine, uniform, β‑families and random Dirichlet.
2. **Exact sub‑gradient descent** on the true objective (scale‑invariant, simplex‑projected).
3. **Linear‑programming refinement** – given a current shape f and its peak M,
   we solve the LP

        maximize   Σ g_i
        s.t.       (f * g)_k ≤ M   for all k,
                   g_i ≥ 0

   The LP finds a distribution that can carry *more* total mass under the same
   convolution bound.  Mixing f with the normalised LP‑solution strictly reduces
   the peak M (theoretical guarantee of the “alternating‑LP” method).  A few LP
   iterations quickly push the surrogate below the previous best.

Finally a cheap random local search polishes the result.
All work respects the overall time budget (default 1000 s).  The returned list
is always a valid non‑negative sequence.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, power‑of‑two padding)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Cheap surrogate (exact objective without clipping) – used for pre‑selection.
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic families (used as seeds)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine‑law (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 0.0) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    conc = 10 ** rng.uniform(-2.0, 0.7)        # concentration in [0.01, 5]
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (Σx=1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (scale‑invariant objective)
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Sub‑gradient descent on 2·n·max(conv(a,a))/sum(a)²."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(conv.max())
        # mask of all entries attaining the max (numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)
        # sub‑gradient of max‑convolution
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 20_000 == 0:
            print(f"    Sub‑grad it={it:,d}, surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (alternating‑LP step)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz
except Exception:  # pragma: no cover – SciPy is expected to be present
    linprog = None
    toeplitz = None


def _lp_refine_once(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP  max Σg_i   s.t.  (f*g)_k ≤ M, g ≥ 0.
    Returns the (unnormalised) optimiser g, or None on failure.
    """
    if linprog is None or toeplitz is None:
        return None
    n = f.size
    # Toeplitz matrix that implements (f * g)_k ≤ M
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([np.array([f[0]]), np.zeros(n - 1)])
    A = toeplitz(col, row)                     # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=float)

    # maximise Σg_i  → minimise −Σg_i
    c = -np.ones(n, dtype=float)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 10.0},
    )
    if res.success:
        return res.x
    return None


def lp_refine(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 12,
    mix_weight: float = 0.25,
) -> np.ndarray:
    """
    Repeated LP refinement: each iteration solves the LP described above,
    rescales the optimiser to unit‑sum and takes a convex combination
    (1‑t)·a + t·g.
    """
    a = a0.copy()
    a = _project_to_simplex(a)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        M = float(conv.max())
        g = _lp_refine_once(a, M)
        if g is None:
            break
        # normalise g to have the same ℓ₁‑norm as a (i.e. 1)
        if g.sum() == 0:
            break
        g = g / g.sum()
        # convex mixing
        a = (1.0 - mix_weight) * a + mix_weight * g
        a = _project_to_simplex(a)       # keep exact sum = 1
        # enforce symmetry (the problem is symmetric)
        a = (a + a[::-1]) / 2.0
        a = _project_to_simplex(a)

        print(f"    LP iter {it+1}, peak M={M:.6e}, surrogate≈{fast_score(a):.6f}")
    return a


# ----------------------------------------------------------------------
# Simple random local search around a candidate
# ----------------------------------------------------------------------
def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.025,
) -> np.ndarray:
    """Pairwise mass‑transfer hill‑climb (accept only improvements)."""
    a = a0.copy()
    best = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta
        a /= a.sum()
        sc = fast_score(a)
        if sc < best:
            best = sc
        else:
            # revert
            a[j] -= delta
            a[i] += delta
            a /= a.sum()
    return a

    Search for a non‑negative step sequence that minimises ``evaluate_sequence``.
    The routine combines analytic seeds, exact sub‑gradient descent and a
    small LP‑based alternating refinement.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.4   # a tiny safety margin

    best_seq: List[float] = []

    best_surrogate: float = float("inf")

    # ------------------------------------------------------------------

    #    continuous optimum).  Empirically 2048 works well with the LP stage.
    # ------------------------------------------------------------------
    lengths = [2048, 1024]      # keep the problem size modest for the LP

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} (time left {deadline - time.time():.1f}s) ===")

        # --------------------------------------------------------------
        # 1a. Build a diverse pool of seeds
        # --------------------------------------------------------------
        seeds = [
