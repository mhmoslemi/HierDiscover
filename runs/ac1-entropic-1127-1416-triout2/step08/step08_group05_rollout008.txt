def hill_climb(a_initial, max_iter, step_scale, rng, deadline):
    a = a_initial.copy()
    score = compute_score(a)
    n = len(a)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        # propose moving delta from i to j
        max_move = a[i]
        delta = rng.uniform(0, min(step_scale, max_move))
        # optionally sometimes move from j to i (negative delta)
        # We can move both directions by sign random.
        # Let's pick sign = +1 or -1.
        sign = rng.choice([-1, 1])
        delta = sign * delta
        # If sign is -1, we move mass from j to i; ensure a[j] >= -delta.
        # So adjust max_move accordingly.
        if sign > 0:
            max_delta = a[i]
            delta = rng.uniform(0, min(max_delta, step_scale))
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            max_delta = a[j]
            delta = rng.uniform(0, min(max_delta, step_scale))
            a_new = a.copy()
            a_new[i] += delta
            a_new[j] -= delta
        # Ensure non-negative
        # Already ensured.
        # compute new score
        new_score = compute_score(a_new)
        if new_score < score - 1e-12:
            a = a_new
            score = new_score
    return a, score

def hill_climb_local(a0, max_iter, init_step, rng, deadline):
    a = a0.copy()
    n = a.size
    best_score = compute_score(a)
    step = init_step
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # Decay step
        if it % 5000 == 0 and it > 0:
            step *= 0.9
        # choose two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # propose moving delta from i to j (or reverse)
        direction = rng.choice([1, -1])
        if direction == 1:
            max_delta = a[i]
            delta = rng.uniform(0, min(step, max_delta))
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            max_delta = a[j]
            delta = rng.uniform(0, min(step, max_delta))
            a_new = a.copy()
            a_new[i] += delta
            a_new[j] -= delta
        # evaluate
        new_score = compute_score(a_new)
        if new_score < best_score:
            a = a_new
            best_score = new_score
    return a, best_score

"""
Propose a step‑function (non‑negative sequence) that minimizes the
``evaluate_sequence`` objective.

The algorithm proceeds through several phases:

1.  Build a diverse pool of analytic seed shapes (arcsine, uniform,
    beta‑distribution, power‑law and mixtures).  A fine grid of
    Beta(α,β) parameters is also added.
2.  Rank the seeds with the fast FFT‑based surrogate
    ``compute_score`` = 2·n·max(conv(a,a)) (identical to the official
    score when the sequence is normalised to sum = 1).
3.  For the most promising seeds run a long Adam optimisation on a
    smooth log‑sum‑exp surrogate of the max‑convolution.
4.  Refine the Adam result by a cheap stochastic hill‑climbing that
    repeatedly transfers a small mass between two random coordinates
    while staying on the simplex.
5.  Polish the best candidate with an exponentiated‑gradient (mirror‑
    descent) step on the exact non‑smooth objective.
6.  If time permits perform a final Euclidean projected sub‑gradient
    descent for a last improvement.

The whole routine respects a 1000 s wall‑clock budget, always returns a
valid (non‑empty, non‑negative) sequence and prints intermediate scores.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a normalised sequence (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 1500.0,
    ramp_iters: int = 120_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    z0 : ndarray
        Initial log‑probabilities (log‑softmax coordinates).
    deadline : float
        Absolute stop‑time (wall‑clock seconds since epoch).
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate for Adam.

    Returns
    -------
    a : ndarray
        Optimised probability vector (sum = 1, a_i ≥ 0).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # simplex point
        conv = convolution_fft(a, a)       # full linear convolution
        mx = float(np.max(conv))

        # Temperature schedule (linear ramp)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # Smooth max using log‑sum‑exp
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)         # unnormalised weights
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # Gradient of smooth max w.r.t. a
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central n entries

        # Chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output (optional)
        if t % 10_000 == 0:
            print(f"    [Adam] it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final point on the simplex
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Mirror descent (KL divergence) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # active set – indices where conv attains (almost) the max
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 10_000 == 0:
            print(f"    [EG]  it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 10_000 == 0:
            print(f"    [SubGrad] it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing on the exact score
# ----------------------------------------------------------------------


def hill_climb_local(
    a0: np.ndarray,
    max_iter: int,
    init_step: float,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """
    Very cheap random‑search local optimiser.
    Repeatedly moves a small amount of probability mass between two
    random coordinates while staying on the simplex.
    """
    a = a0.copy()
    n = a.size
    best_score = compute_score(a)
    step = init_step

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # Periodically decay the step size
        if it % 10_000 == 0:
            step *= 0.9

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # Random direction: move mass i→j  or  j→i
        if rng.random() < 0.5:
            max_delta = a[i]
            delta = rng.uniform(0.0, min(step, max_delta))
            if delta == 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            max_delta = a[j]
            delta = rng.uniform(0.0, min(step, max_delta))
            if delta == 0.0:
                continue
            a_new = a.copy()
            a_new[i] += delta
            a_new[j] -= delta

        new_score = compute_score(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_score = new_score

    return a


# ----------------------------------------------------------------------
# Helper: build a pool of diverse seeds for a given length
# ----------------------------------------------------------------------


def build_seed_pool(
    n: int,
    rng: np.random.Generator,
    best_sofar: List[float],
) -> List[np.ndarray]:
    """Return a list of candidate probability vectors of length n."""
    seeds: List[np.ndarray] = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),                     # uniform
    ]

    # random power‑law shapes
    for _ in range(8):
        p = rng.uniform(0.05, 4.0)
        eps = rng.uniform(0.0, 3.0)
        seeds.append(generate_power_shape(n, p, eps))

    # random beta‑shapes
    for _ in range(8):
        a = rng.uniform(0.05, 6.0)
        b = rng.uniform(0.05, 6.0)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        seeds.append(generate_beta_shape(n, a, b, eps))

    # mixtures of arcsine & uniform
    for _ in range(4):
        w = rng.uniform(0.0, 1.0)
        seeds.append(generate_mixture_shape(n, w))

    # systematic Beta grid (fine search around the arcsine region)
    alphas = np.linspace(0.3, 0.9, 9)
    betas = np.linspace(0.3, 0.9, 9)
    for a in alphas:
        for b in betas:
            seeds.append(generate_beta_shape(n, a, b, eps=1e-6))

    # up‑sampled version of the current best (if any)
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        seeds.append(upsample_sequence(old, n))

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 43,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine respects a 1000 s wall‑clock budget and always returns
    a valid (non‑empty) sequence.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback: use any previously discovered sequence if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Optimise a cascade of increasing lengths
    # ------------------------------------------------------------------
    # The most promising lengths (larger lengths give a finer discretisation)
    lengths = [2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 15.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ---- 1a. Build a diverse seed pool -----------------------
        seed_pool = build_seed_pool(n, rng, best_seq)
        scored = [(compute_score(s), s) for s in seed_pool]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]

        print(f"Best seed score (surrogate) = {scored[0][0]:.6f}")

        # ---- 1b. Time allocation per seed ------------------------
        remaining = deadline - time.time()
        per_seed_time = max(5.0, min(remaining / (len(top_seeds) * 2.0), 120.0))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 10.0:
                break

            print(f"\n  Seed {idx}/{len(top_seeds)} – Adam budget ≈{per_seed_time:.1f}s")
            seed_deadline = min(time.time() + per_seed_time, deadline - 5.0)

            # Initialise Adam on the log‑softmax of the seed
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            # ------------------------------------------------------
            # 2. Smooth‑max Adam optimisation
            # ------------------------------------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=1500.0,
                ramp_iters=120_000,
                lr=0.07,
            )

            # ------------------------------------------------------
            # 3. Official evaluation (fast surrogate = exact)
            # ------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Adam result official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ------------------------------------------------------
            # 4. Local stochastic hill‑climbing refinement
            # ------------------------------------------------------
            if time.time() < deadline - 12.0:
                hc_deadline = min(deadline - 2.0, time.time() + 30.0)
                a_hc = hill_climb_local(
                    a_opt,
                    max_iter=80_000,
                    init_step=0.05,
                    rng=rng,
                    deadline=hc_deadline,
                )
                try:
                    val_hc = evaluate_sequence(a_hc.tolist())
                    print(f"   Hill‑climb score = {val_hc:.6f}")
                    if val_hc < best_val:
                        best_val, best_seq = val_hc, a_hc.tolist()
                        print(f"   *** New best after hill‑climb ***")
                except Exception as e:
                    print(f"   Hill‑climb evaluation failed: {e}")

            # ------------------------------------------------------
            # 5. KL‑mirror (exponentiated‑gradient) polishing
            # ------------------------------------------------------
            if time.time() < deadline - 5.0:
                eg_deadline = deadline - 2.0
                a_eg = exponentiated_gradient_polish(
                    np.array(best_seq, dtype=float),
                    eg_deadline,
                    max_iter=120_000,
                    step=0.018,
                )
                try:
                    val_eg = evaluate_sequence(a_eg.tolist())
                    print(f"   EG polish score = {val_eg:.6f}")
                    if val_eg < best_val:
                        best_val, best_seq = val_eg, a_eg.tolist()
                        print(f"   *** New best after EG polish ***")
                except Exception as e:
                    print(f"   EG polishing failed: {e}")

    # ------------------------------------------------------------------
    # 6. Final global polishing (exact sub‑gradient)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\nFinal global sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=200_000,
            step0=0.07,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 7. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 256
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
