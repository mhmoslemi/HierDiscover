"""
Non‑negative step‑function optimiser.

Main ideas
----------
* Initialise with the classic central‑binomial coefficients (the (1‑x)^‑½
  generating function).  After normalisation they already give the
  theoretical bound π/2 ≈ 1.571.
* Refine the shape by an **exponentiated‑gradient** descent on the simplex.
  In each iteration we locate the current worst convolution entry,
  take its sub‑gradient and update the probabilities with
      a ← a ⊙ exp(‑η·g)   (⊙ = element‑wise product)
  followed by a simplex normalisation.  This multiplicative update
  respects non‑negativity and is often more efficient on the simplex
  than plain projected‑gradient steps.
* When the EG phase stalls we inject a tiny random perturbation to
  escape plateaus.
* After the EG phase we run a few more sophisticated refinements:
  – a smooth log‑sum‑exp (soft‑max) optimiser with Adam momentum,
  – a pure sub‑gradient descent on the exact max‑objective,
  – a final stochastic hill‑climb that swaps or shifts small amounts of mass.
* All stages respect the wall‑clock budget (≈1000 s).  The best sequence
  found is returned as a plain list of floats; `evaluate_sequence`
  will apply the final clipping/summing itself.
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the probability simplex.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def central_binomial(n: int) -> np.ndarray:
    """Truncated central‑binomial coefficients a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    return a


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)   # first‑moment
    v = np.zeros_like(a)   # second‑moment
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothing β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smooth surrogate (cross‑correlation)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad ** 2)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score

# Sub‑gradient optimiser on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()
        a = a_new

        # occasional jitter
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

    print(f"EG phase finished after {iter_cnt} iterations, best score = {best_score:.6f}")
    if best_score < best_val:
        best_val = best_score
        best_seq = best_a.tolist()

    # -----------------------------------------------------------------
    # 4. Smooth log‑sum‑exp optimisation (Adam) – fine polishing
    # -----------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("\nStarting smooth Adam optimisation")
        a_opt, scr = smooth_gradient_optimize(
            best_a,
            time_deadline=deadline - 150.0,
            max_iters=400_000,
            beta_start=6.0,
            beta_end=200.0,
            step0=1.0,
        )
        if scr < best_val:
            best_val = scr
            best_a = a_opt
            best_seq = best_a.tolist()
            print(f"[Adam] improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Sub‑gradient polishing (exact max)
    # -----------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print("\nSub‑gradient refinement")
        refined_a, refined_score = subgrad_optimize(
            best_a,
            time_deadline=deadline - 80.0,
            max_iter=400_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_a = refined_a
            best_seq = best_a.tolist()
            print(f"[Sub‑grad] improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 6. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\nHill‑climb polishing")
        polished_seq, polished_score = hill_climb_permutation(
            best_a,
            time_deadline=deadline,
            max_steps=800_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"[Hill‑climb] improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to a tiny uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    return best_seq
