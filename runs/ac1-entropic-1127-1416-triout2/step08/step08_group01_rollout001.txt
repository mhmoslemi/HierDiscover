"""
Propose a low‑score step‑function using a multi‑stage optimisation pipeline.

The pipeline
-------------
1. Choose a length (larger n gives a better theoretical bound, but costs
   FFT work).  8192 is used when enough time remains.
2. Analytic seed vectors:
   - α‑family   a_i ∝ [(i+½)(n‑½‑i)]⁻ᵅ
   - Sine‑power family a_i ∝ sin(π(i+½)/n)^p
   The best of these seeds is kept.
3. Random Dirichlet / sparse seeds are generated and optimised with a
   *mirror‑Adam* optimiser that minimises a smooth approximation of the
   max‑convolution:
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k).
   The Adam update is performed on the logarithmic variables
   (i.e. a_i = exp(x_i)/Σ exp(x)), which automatically respects
   non‑negativity.  β is annealed from a small to a large value.
4. A long dedicated optimisation run on the current best vector.
5. Euclidean projected sub‑gradient descent on the true objective
   2·n·max(a*a) (the objective returned by `evaluate_sequence` when the
   sum is normalised to 1).
6. Stochastic hill‑climbing with simulated‑annealing acceptance to
   escape shallow local minima.

All phases respect the 1000 s wall‑clock budget.  The algorithm prints
progress information and finally returns a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    This equals the value returned by `evaluate_sequence` for the same
    normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def sin_power_start(n: int, p: float) -> np.ndarray:
    """Return a_i ∝ sin(π(i+½)/n)^p on the simplex (p>0)."""
    i = np.arange(n, dtype=float)
    s = np.sin(np.pi * (i + 0.5) / n)
    w = s ** p
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Explore α‑grid and keep the best vector."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([np.linspace(-1.0, 0.0, 11),
                             np.linspace(0.0, 2.5, 26)])
    for alpha in alphas:

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    sparsity = np.random.uniform(0.03, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dirichlet_start(n: int, concentration: float = 1.0) -> np.ndarray:
    """Dirichlet‑distributed start (concentration controls sparsity)."""
    a = np.random.gamma(shape=concentration, scale=1.0, size=n)
    s = a.sum()
    if s == 0.0 or not np.isfinite(s):
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Mirror‑Adam optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------


def exp_adam_optimize(start_a: np.ndarray,
                      deadline: float,
                      max_iters: int = 200_000,
                      beta_start: float = 0.1,
                      beta_end: float = 500.0,
                      lr: float = 0.07,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-14,
                      eps2: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Mirror‑Adam optimisation of the smooth max‑convolution.
    The variables are the logarithms of the coefficients, which guarantees
    positivity.  After each update the vector is normalised onto the simplex.
    """
    n = start_a.shape[0]
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    x = np.log(a)                     # log‑variables
    m = np.zeros_like(a)              # first moment
    v = np.zeros_like(a)              # second moment

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- β schedule (exponential) ---
        t_rel = (it - 1) / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_rel

        # --- compute smoothed max and its softmax weights ---
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))   # stabilised exponentials
        w = expw / expw.sum()

        # --- gradient of the smooth objective w.r.t. a ---
        grad_a = 2.0 * convolution_fft(a[::-1], w)[n - 1:n - 1 + n]

        # --- Adam update on the log‑variables ---
        g = grad_a                           # treat ∂f/∂x ≈ ∂f/∂a
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps2)
        x = x - step

        # --- re‑normalise onto the simplex ---
        x_max = np.max(x)                    # for numerical stability
        a = np.exp(x - x_max)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # --- occasional exact evaluation ---
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final exact evaluation
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Classic projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 600_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient method on 2·n·max(a*a)."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

# Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_simanneal(start_seq: np.ndarray,
                         deadline: float,
                         max_steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Random walk that occasionally accepts worsening moves with a
    temperature schedule.  Only the exact score decides acceptance.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # temperature schedule (exponential decay)
    T0 = 1e-3
    T_end = 1e-7

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.12 * np.random.rand() * best_a[i]   # up to 12 % shift
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        delta_sc = sc - best_sc
        T = T0 * (T_end / T0) ** (step / max_steps)
        accept = False
        if sc < best_sc:
            accept = True
        else:
            # Metropolis acceptance
            if T > 0 and np.exp(-delta_sc / T) > np.random.rand():
                accept = True

        if accept:
            best_sc = sc
            best_a = cand

    return best_a, best_sc

            lr=0.07,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after random Adam = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Long dedicated mirror‑Adam run
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] long mirror‑Adam optimisation")
        a_long, sc_long = exp_adam_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=400_000,
            beta_start=0.05,
            beta_end=800.0,
            lr=0.05,
        )
        if sc_long < best_score:
            best_score = sc_long
            best_seq = a_long.tolist()
            print(f"[{time.time():.1f}] after long Adam best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Projected sub‑gradient refinement
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 40.0,
            max_iters=500_000,
            step0=0.06,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 6. Final stochastic hill‑climb with annealing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (simulated annealing)")
        a_final, sc_final = hill_climb_simanneal(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=1_300_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: fallback to a uniform short vector if everything failed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
