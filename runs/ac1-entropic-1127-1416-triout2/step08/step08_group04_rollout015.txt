    i = np.arange(n, dtype=float) + 0.5
    x = i / n
    # compute beta pdf proportional to x**(alpha-1) * (1-x)**(alpha-1)
    vals = np.power(x, alpha-1) * np.power(1-x, alpha-1)
    # Using epsilon for stability; if alpha <1, near edges singular -> transform to avoid inf.
    # Use np.maximum(x, eps) if needed
    # Write function robustly
    eps = 1e-12

import time
import numpy as np
from typing import List

def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Fast surrogate of official score, assuming sum(a)==1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector onto probability simplex (non‑negative, sum=1)."""
    # implementation of Duchi et al.
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine density (Beta(0.5,0.5)) normalized to sum=1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()

def power_shape(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """Shape proportional to ((i+0.5)*(n-0.5-i)+eps)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total

def mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform, then normalised."""
    arc = arcsine_shape(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 500_000,
                             tau_start: float = 0.5,
                             tau_end: float = 2000.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max approximation.
    a = softmax(w), temperature τ grows linearly from τ_start to τ_end.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-16)  # logits

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp    # shape (L,)

        # gradient w.r.t. a : 2·conv(a, rev(weights))
        rev_w = weights[::-1]
        # conv of a with rev_w (linear conv) produces length n+L-1
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)          # decreasing learning rate
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 100_000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_extended(initial_a: np.ndarray,
                                 deadline: float,
                                 max_iter: int = 400_000,
                                 step0: float = 0.07) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # tight mask of maximal entries (numerical tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, rev(mask))
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 100_000 == 0:
            surrogate = compute_score(a)
            print(f"    Subgrad it={it:,d} surrogate={surrogate:.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent in KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 100_000 == 0:
            surrogate = compute_score(a)
            print(f"    EG it={it:,d} surrogate={surrogate:.6f}")

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 30_000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Simple hill‑climbing: transfer a tiny amount of mass between two coordinates
    and keep any strictly improving move (according to surrogate score).
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    trials = 0
    while time.time() < deadline - 0.2 and trials < max_trials:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            trials += 1
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            trials += 1
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            trials += 1
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        trials += 1

    return a_cur

    Adam optimiser on the smooth‑max approximation of max(conv).
    The optimisation variable are logits w such that a = softmax(w).
    Temperature τ grows linearly from τ_start to τ_end.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-16)  # initialise logits

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·conv(a, rev(weights))
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 100_000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_extended(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 100_000 == 0:
            surrogate = compute_score(a)
            print(f"    Subgrad it={it:,d} surrogate={surrogate:.6f}")

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent w.r.t. KL).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 100_000 == 0:
            surrogate = compute_score(a)
            print(f"    EG it={it:,d} surrogate={surrogate:.6f}")

    return a


def local_random_search(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 40_000,
    step_scale: float = 0.001,
    rng: np.random.Generator = np.random.default_rng(),
) -> np.ndarray:
    """
    Very cheap stochastic hill climbing:
    move a tiny amount of mass from one coordinate to another.
    Accept only improving moves (according to the surrogate score).
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    trials = 0
    while time.time() < deadline - 0.2 and trials < max_trials:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            trials += 1
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            trials += 1
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            trials += 1
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        trials += 1

    return a_cur

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function sequence with a low official score.
    Stages:
      1) Coarse grid search over analytic families,
      2) Random Dirichlet exploration,
      3) Annealed Adam optimisation (smooth‑max, τ up to 2000),
      4) Extended projected sub‑gradient descent,
      5) Exponentiated‑gradient polish,
      6) Cheap stochastic local search.
    The whole procedure respects the supplied wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_surrogate: float = float("inf")
    best_official: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = fallback
                best_surrogate = compute_score(np.array(fallback))
        except Exception as e:
            print(f"Fallback evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Stage 1 – coarse grid search (analytic families)
    # ------------------------------------------------------------------
    candidate_ns = [1024, 2048, 4096, 8192, 16384]
    # extended p‑grid (covers more extreme tails)
    p_grid = np.concatenate(
        [
            np.linspace(0.05, 0.90, 18),  # fine near the classic arcsine
            np.linspace(0.90, 2.50, 17),  # more extreme U‑shapes
        ]
    )
    w_grid = np.linspace(0.0, 1.0, 11)

    print("\n--- Stage 1: coarse grid search ---")
    for n in candidate_ns:
        print(f"Testing length n={n}")

        # uniform
        uni = np.full(n, 1.0 / n)
        s = compute_score(uni)
        if s < best_surrogate:
            best_surrogate, best_seq = s, uni.tolist()
            print(f"  New best (uniform) surrogate={s:.6f}")

        # arcsine (p≈0.5)
        arc = arcsine_shape(n)
        s = compute_score(arc)
        if s < best_surrogate:
            best_surrogate, best_seq = s, arc.tolist()
            print(f"  New best (arcsine) surrogate={s:.6f}")

        # power shapes
        for p in p_grid:
            shape = power_shape(n, p, eps=1e-12)
            s = compute_score(shape)
            if s < best_surrogate:
                best_surrogate, best_seq = s, shape.tolist()
                print(f"  New best (p={p:.3f}) surrogate={s:.6f}")

        # mixtures arcsine / uniform
        for w in w_grid:
            mix = mixture_shape(n, w)
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"  New best (mix w={w:.2f}) surrogate={s:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – large‑scale random Dirichlet exploration
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        n_best = len(best_seq)
        print("\n--- Stage 2: random Dirichlet search ---")
        alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
        draws_per_alpha = 300
        for alpha in alphas:
            if time.time() >= deadline - 5.0:
                break
            print(f"  α={alpha:.2f}")
            for _ in range(draws_per_alpha):
                if time.time() >= deadline - 5.0:
                    break
                a = rng.dirichlet(np.full(n_best, alpha))
                s = compute_score(a)
                if s < best_surrogate:
                    best_surrogate, best_seq = s, a.tolist()
                    print(f"    New best random surrogate={s:.6f}")
    else:
        print("\nSkipping random Dirichlet search (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 3 – annealed Adam refinement on smooth‑max
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print("\n--- Stage 3: annealed Adam refinement ---")
        a_init = np.array(best_seq, dtype=float)
        a_refined = adam_optimize_smooth_max(
            a_init,
            deadline,
            max_iters=600_000,
            tau_start=0.5,
            tau_end=2000.0,
            base_lr=0.5,
        )
        s = compute_score(a_refined)
        print(f"  After Adam surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a_refined.tolist()
    else:
        print("\nSkipping Adam refinement (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 4 – extended projected sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        print("\n--- Stage 4: extended sub‑gradient descent ---")
        a = np.array(best_seq, dtype=float)
        a = subgradient_descent_extended(
            a,
            deadline,
            max_iter=500_000,
            step0=0.07,
        )
        s = compute_score(a)
        print(f"  After sub‑gradient surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping sub‑gradient descent (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 5 – exponentiated‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Stage 5: exponentiated‑gradient polish ---")
        a = np.array(best_seq, dtype=float)
        a = eg_refine(
            a,
            deadline,
            max_iter=200_000,
            eta0=0.25,
        )
        s = compute_score(a)
        print(f"  After EG surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping EG polish (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 6 – cheap stochastic local search
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Stage 6: local random search ---")
        a = np.array(best_seq, dtype=float)
        a = local_random_search(
            a,
            deadline,
            max_trials=40_000,
            step_scale=0.001,
            rng=rng,
        )
        s = compute_score(a)
        print(f"  After local search surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping local random search (insufficient time)")

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        final_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
