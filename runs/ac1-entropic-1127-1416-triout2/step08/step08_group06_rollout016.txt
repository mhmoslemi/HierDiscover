        # ensure positivity

        f = f / f.sum()
        # compute initial score
        M = np.max(convolution_fft(f, f))
        current_score = 2 * n * M
        print(f"Initial score: {current_score:.6f}")

        best_local_seq = f.copy()
        best_local_score = current_score

        # LP iterative improvement
        # set step parameters
        t0 = 0.5  # initial mixing weight
        t_min = 0.01
        # schedule maybe decrease with iteration
        max_iter = 70
        for it in range(1, max_iter+1):
            if time.time() >= deadline - 5:
                break
            # Build A matrix for constraints conv(f,g) <= M
            # using diags
            # Since each column j has coefficient f_i at row i+j => diag offset -i
            data = [np.full(n, f[i]) for i in range(n)]
            offsets = [-i for i in range(n)]
            A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
            b = np.full(2*n-1, M + 1e-12)  # tiny epsilon

            c = -np.ones(n)  # maximize sum
            res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                              method='highs', options={'presolve': True, 'time_limit': max(1.0, (deadline - time.time()))})
            if not res.success:
                print(f"LP failed at iter {it}, status: {res.message}")
                break
            g = res.x
            sum_g = g.sum()
            if sum_g <= 0:
                print("LP gave zero sum, break")
                break
            g_norm = g / sum_g

            # step size schedule: geometric decay
            t = max(t_min, t0 * (0.9 ** (it-1)))  # for instance
            # update f
            f = (1 - t) * f + t * g_norm
            # ensure simplex
            f = np.clip(f, 1e-15, None)
            f = f / f.sum()

            # compute new score
            M = np.max(convolution_fft(f, f))
            current_score = 2 * n * M

            if current_score < best_local_score - 1e-9:
                best_local_score = current_score
                best_local_seq = f.copy()
                print(f"Iter {it}: improved score {current_score:.6f}")

            # optional break if improvement small
            if it % 10 == 0:
                # maybe stop early if not improving
                pass

        # after LP loop, assign best candidate found
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.copy()
            print(f"Best after length {n}: {best_score:.6f}")

        # optionally do final polishing with EG (like earlier)
        if time.time() < deadline - 5:
            # run EG for a few thousand iterations
            a = best_seq.copy()
            a = exponentiated_gradient_polish(a, deadline=deadline, max_iter=20000, step=0.02)
            score = compute_score(a)

                print(f"After EG polishing: {best_score:.6f}")

    if best_seq is None:
        n_fallback = 512
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

        print(f"Final official score: {final_official:.6f}")
    except Exception as e:

    # return as list of floats
    return best_seq.tolist()

"""
Propose a non‑negative step sequence that (almost) minimizes the official
``evaluate_sequence`` objective.

Approach
--------
1. Initialise a reasonably good shape – the discrete arcsine envelope.
2. Apply a *linear‑programming* “balancing” step (the classic LP‑iteration
   described in the statement).  For the current vector `f` we solve

        maximise   Σ g_j
        subject to  (f ⋆ g)_k ≤ M      ∀k
                    g_j ≥ 0

   where `M = max_k (f⋆f)_k`.  The solution `g` is normalised to a simplex
   point and mixed into `f` with a line‑search on the mixing weight `t`.
   The mix is kept only if it strictly reduces the max‑convolution, which
   guarantees a monotone descent of the objective.
3. After a few LP iterations (typically ≤ 15) we hand the best vector to a
   fine‑grained exponentiated‑gradient (mirror‑descent) optimiser that works
   directly on the true max‑convolution objective.
4. Finally a cheap stochastic hill‑climber (pairwise mass transfer) refines
   the result a little more.
5. The routine respects the 1000 s wall‑clock limit, always returns a
   non‑empty sequence, and prints intermediate scores for diagnostics.

All helper functions are top‑level; no closures or lambdas are used.
"""

# ----------------------------------------------------------------------
# Imports and low‑level utilities
# ----------------------------------------------------------------------
import time
from typing import List

import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Simple FFT‑based linear convolution (real, non‑circular)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate of the official objective for L1‑normalised vectors
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(a,a)) under the assumption Σa = 1.
    This equals the official ``evaluate_sequence`` value because the
    evaluation is homogeneous.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Basic shapes – only the arcsine envelope is used as a starter
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """
    Discrete arcsine‑like envelope.
    a_i ∝ 1 / √((i+0.5)(n‑0.5‑i) + eps)  ,   eps = 0.25.
    The vector is already normalised to sum‑one.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (KL‑mirror) polishing – works on the exact max‑conv.
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step: float = 0.015,
) -> np.ndarray:
    """Multiplicative updates (mirror descent) on  max‑convolution."""
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for the active set
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max‑convolution w.r.t. a
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # multiplicative update
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG iter={it:,d}  score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Small stochastic hill‑climber (pairwise mass transfer)
# ----------------------------------------------------------------------
def stochastic_refine(a: np.ndarray, deadline: float, max_steps: int = 5_000) -> np.ndarray:
    """
    Randomly picks two indices i≠j and transfers a tiny amount of mass
    from i to j.  The move is kept if it improves the objective.
    """
    n = a.size
    cur_score = compute_score(a)

    rng = np.random.default_rng()
    step_factor = 0.02  # relative amount moved each trial

    for _ in range(max_steps):
        if time.time() >= deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 0:
            continue
        delta = step_factor * a[i]  # take a small fraction of i's mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalise (should still sum to 1, but guard against tiny drifts)
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        new_score = compute_score(a_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
    return a

        best_local_seq = f.copy()
        best_local_score = cur_score

        # ---- 1. LP‑based balancing iterations --------------------------------
        max_lp_iter = 15                     # keep the total time modest
        t_init = 0.4                         # initial mixing weight
        t_min = 0.04                         # we never go below this
        eps_lp = 1e-12

        for it in range(1, max_lp_iter + 1):
            if time.time() >= deadline - 5:
                break

            # Build the Toeplitz constraint matrix A (size (2n‑1) × n)
            # Row k contains f_i at column j = k‑i whenever 0≤j<n.
            data = [np.full(n, f[i]) for i in range(n)]
            offsets = [-i for i in range(n)]
            A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
            b = np.full(2 * n - 1, M + eps_lp)

            # maximise Σ g_j  <=> minimise −Σ g_j
            c = -np.ones(n)

            res = opt.linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True},
            )

            if not res.success:
                print(f"  LP failed at iter {it}: {res.message}")
                break

            g = res.x
            sum_g = g.sum()
            if sum_g <= 0 or not np.isfinite(sum_g):
                print("  LP produced non‑positive sum, aborting LP phase.")
                break

            # Normalise g to stay on the simplex before mixing
            g_norm = g / sum_g

            # ---- 2. Line‑search on mixing weight t -------------------------
            # We require a strict reduction of the max‑convolution.
            improved = False
            t = max(t_init / (it ** 0.5), t_min)   # gradually shrink t
            while t >= t_min:
                f_candidate = (1.0 - t) * f + t * g_norm
                f_candidate = np.clip(f_candidate, 1e-15, None)
                f_candidate = f_candidate / f_candidate.sum()
                M_new = np.max(convolution_fft(f_candidate, f_candidate))
                if M_new < M - 1e-15:          # genuine improvement
                    f = f_candidate
                    M = M_new
                    improved = True
                    break
                t *= 0.5

            if not improved:
                # No improvement possible – exit the LP loop early.
                print(f"  No improvement at LP iter {it}, terminating LP phase.")
                break

            cur_score = compute_score(f)
            if cur_score < best_local_score - 1e-9:
                best_local_score = cur_score
                best_local_seq = f.copy()
                print(f"  LP iter {it:2d}  score = {cur_score:.6f}")

        # ---- 3. Fine polarisation with exponentiated‑gradient -------------
        if time.time() < deadline - 10:
            print("  Starting exponentiated‑gradient polishing …")
            f_polish = exponentiated_gradient_polish(
                best_local_seq,
                deadline=deadline,
                max_iter=int(min(120_000, (deadline - time.time()) * 1500)),
                step=0.012,
            )
            polish_score = compute_score(f_polish)
            if polish_score < best_local_score:
                best_local_score = polish_score
                best_local_seq = f_polish.copy()
                print(f"  EG polish improved score to {polish_score:.6f}")

        # ---- 4. Stochastic local refinement --------------------------------
        if time.time() < deadline - 5:
            print("  Stochastic hill‑climb refinement …")
            f_refined = stochastic_refine(
                best_local_seq,
                deadline=deadline,
                max_steps=5_000,
            )
            refined_score = compute_score(f_refined)
            if refined_score < best_local_score:
                best_local_score = refined_score
                best_local_seq = f_refined.copy()
                print(f"  Refinement improved score to {refined_score:.6f}")

        # ---- 5. Update global best -----------------------------------------
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.tolist()
            print(f"  *** New global best (len={n}) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Guard – guarantee a valid non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_score = compute_score(np.array(best_seq))
        print("Fallback to uniform distribution.")

    # ------------------------------------------------------------------
    # Final official evaluation (only for logging)
    # ------------------------------------------------------------------
    try:
        official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Official score = {official:.6f}")
    except Exception as exc:
        print(f"\nSearch finished. Official evaluation failed: {exc}")

    return best_seq
