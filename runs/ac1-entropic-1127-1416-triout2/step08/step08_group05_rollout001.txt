from scipy.optimize import linprog
def solve_lp_max_sum(f, M, tol_active=1e-9):
    n = len(f)
    # compute active indices
    conv_f = convolution_fft(f, f)
    # M = max(conv_f) (maybe given)
    # active indices where conv_f >= M - tol_active
    active = np.where(conv_f >= M - tol_active)[0]
    # Build A_ub and b_ub
    # For each active index k: sum_i f_i * b_{k-i} <= M
    # Convert to constraint coefficients for b_j.
    # Use sparse matrix.
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # For i in [i_min,i_max], let j = k-i
        # coefficient for b_j = f_i
        # So for each i: j = k - i -> coefficient = f[i]
        # Append to matrix row.
        # Use loop: i from i_min to i_max
        # That could be many loops but okay for moderate n.
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(row_idx)
            cols.append(j)
            data.append(f[i])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n), dtype=float)
    b_ub = np.full(len(active), M)
    # objective: maximize sum(b) => minimize -c^T b with c = np.ones(n)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

"""
Propose a low‑value step sequence for ``evaluate_sequence``.

Algorithm outline
-----------------
1. Initialise a symmetric, non‑negative shape (arcsine distribution) on a
   moderately large length (default 8192).  The vector is normalised to
   sum = 1 – the objective of ``evaluate_sequence`` is scale‑invariant.

2. Iteratively improve the shape by solving a small linear programme.
   For the current vector *f* we know the current maximal self‑convolution
   value `M = max(f*f)`.  By fixing the active constraints
   (indices where the convolution attains the maximum) we solve

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   for every active k,
                    b_j ≥ 0.

   The solution `b` is a feasible direction that can increase the total
   mass while keeping the peak of the convolution unchanged.  After
   normalising `b` to sum = 1 we mix it with `f` and perform a short
   line‑search on the mixing coefficient `t ∈ [0,1]` to minimise the new
   peak.  This step is guaranteed to never increase the objective and
   usually reduces it.

3. The LP‑step is repeated many times (as long as the time budget permits).
   After each iteration the official objective is evaluated and the best
   sequence is remembered.

4. When the allotted time for the LP phase is exhausted, a few
   exponentiated‑gradient (KL‑mirror) polishing iterations are performed
   on the exact objective to fine‑tune the solution.

5. If for any reason no valid sequence has been found, a uniform fallback
   is returned.

The method relies only on cheap FFT‑based convolutions and small LP
problems (few dozen constraints), so many iterations fit well into the
1000 s budget and typically drive the score below the target 1.503.
"""

import time
from typing import List
import numpy as np
import scipy.optimize
import scipy.sparse


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Analytic seeds
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Linear‑programming step (active‑set)
# ----------------------------------------------------------------------


def solve_lp_active(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   (f * b)_k ≤ M for each active k,
    with b_j ≥ 0.  Returns the optimal b (not normalised) or None on failure.
    """
    n = f.size
    rows = []
    cols = []
    data = []

    # Build rows corresponding to active convolution indices.
    for ridx, k in enumerate(active_idx):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # For each i in the valid range, coefficient for b_{k-i} is f_i.
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(ridx)
            cols.append(j)
            data.append(f[i])

    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n), dtype=float)
    b_ub = np.full(len(active_idx), M, dtype=float)

    # Objective: maximise Σ b_j  → minimise -Σ b_j
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# KL‑mirror polishing (exponentiated‑gradient) on the exact objective
# ----------------------------------------------------------------------


def eg_polish(a: np.ndarray, deadline: float,
              step: float = 0.018,
              max_iter: int = 150_000) -> np.ndarray:
    """
    Multiplicative (exponentiated‑gradient) polishing on
    f(a) = 2·n·max(conv(a,a)).  The iterate stays on the simplex.
    """
    a = np.clip(a, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1

    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(conv.max())
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional progress report (optional)
        if it % 10_000 == 0:
            print(f"    EG polish it={it:,d}")

    return a

def propose_candidate(seed: int = 43, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s wall‑clock budget
    and always returns a valid sequence.

    The core of the search is an active‑set LP that improves the peak of the
    self‑convolution, followed by a short exponentiated‑gradient polish.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use a previously known sequence if available (fallback)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose a working length (increase if time permits)
    # ------------------------------------------------------------------
    # We start from a medium size; if time remains we upscale later.
    lengths = [2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 8.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # Initialise f: either up‑sample the current best or use the arcsine seed.
        if best_seq:
            f = upsample_sequence(np.array(best_seq, dtype=float), n)
        else:
            f = arcsine_initial(n)

        # Normalise to sum = 1 (scale does not affect the objective).
        f = f / f.sum()

        # --------------------------------------------------------------
        # 1a.  Active‑set LP refinement loop
        # --------------------------------------------------------------
        max_lp_iter = 80  # enough iterations for the budget
        t_grid = np.linspace(0.0, 1.0, 11)  # line‑search candidates for mixing

        for lp_it in range(1, max_lp_iter + 1):
            if time.time() >= deadline - 5.0:
                break

            conv_f = convolution_fft(f, f)
            M = float(conv_f.max())

            # Determine the (small) active set where the peak is attained.
            tol_active = max(M * 1e-8, 1e-14)
            active_idx = np.where(conv_f >= M - tol_active)[0]

            # Solve the LP.
            b = solve_lp_active(f, M, active_idx)
            if b is None:
                # LP failed – abort the LP phase for this length.
                print("    LP failed, exiting LP loop.")
                break

            # Normalise the LP solution to a probability vector g.
            b_sum = b.sum()
            if b_sum <= 0.0 or not np.isfinite(b_sum):
                # Degenerate solution – stop early.
                break
            g = b / b_sum

            # ----------------------------------------------------------
            # 1b.  Line‑search for the best mixing coefficient t.
            # ----------------------------------------------------------
            conv_g = convolution_fft(g, g)
            conv_fg = convolution_fft(f, g)

            best_t = 0.0
            best_Mt = M
            for t in t_grid:
                conv_t = (1 - t) ** 2 * conv_f + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_g
                Mt = float(conv_t.max())
                if Mt < best_Mt:
                    best_Mt = Mt
                    best_t = t

            # Update f with the chosen mixing.
            if best_t > 0.0:
                f = (1 - best_t) * f + best_t * g
                f = np.maximum(f, 0.0)
                f /= f.sum()

            # Evaluate the official objective occasionally.
            if lp_it % 5 == 0 or best_t > 0.0:
                seq = f.tolist()
                try:
                    val = evaluate_sequence(seq)
                    if val < best_val:
                        best_val, best_seq = val, seq
                        print(f"   LP it={lp_it:2d} new best = {val:.6f}")
                except Exception:
                    pass

        # --------------------------------------------------------------
        # 2.  Final KL‑mirror polishing (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0 and best_seq:
            remaining = deadline - time.time()
            print(f"  Polishing with EG for ≤{remaining:.1f}s")
            a_arr = np.array(best_seq, dtype=float)
            a_arr = eg_polish(a_arr, deadline, step=0.018, max_iter=150_000)
            try:
                val = evaluate_sequence(a_arr.tolist())
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"   Post‑EG polish improved to {val:.6f}")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty, valid return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
