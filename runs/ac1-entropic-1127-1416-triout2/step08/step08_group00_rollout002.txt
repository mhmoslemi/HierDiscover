"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

The search works in four phases inside the 1000 s wall‑clock budget:

0.  If a previous construction (``height_sequence_1``) is available, use it as a
    warm‑start.

1.  **Fast random‑restart smooth optimisation** – a soft‑max surrogate of the
    true max‑objective is minimised with Adam.  Each restart lasts only a few
    seconds, allowing hundreds of independent starts.  The best exact score
    is retained.

2.  **Simulated‑annealing refinement** – starting from the best candidate,
    stochastic mass‑transfer moves are accepted according to a decreasing
    temperature schedule.  This helps to escape the basin of attraction of
    the gradient descent and to explore new regions of the simplex.

3.  **Deterministic hill‑climb polishing** – small local moves (swap or
    transfer) are performed greedily, keeping every improvement.  The step
    size adapts automatically.

All vectors are kept on the probability simplex (non‑negative, sum = 1);
the evaluation is therefore scale‑invariant.  The routine finally returns
a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using the real FFT (fast for large arrays).
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x_i ≥ 0, ∑x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth‑max optimisation (log‑sum‑exp) with Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 250_000,
    beta_start: float = 5.0,
    beta_end: float = 1500.0,
    step0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Minimise a soft‑max surrogate of  max(conv(a,a)) using Adam.
    ``start_a`` must already lie on the simplex.
    Returns the best exact score seen during the run.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # -- anneal β ----------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # -- smoothed objective -----------------------------------------
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        # stable log‑sum‑exp weights
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # -- gradient of soft‑max (cross‑correlation) -------------------
        a_rev = a[::-1]
        # convolution length = 3n‑2 ; we need the central n entries
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # -- Adam update ------------------------------------------------
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)

        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation (every 300 iterations)
        if it % 300 == 0:
            scr = compute_score(a)
            if scr < best_score:
                best_score = scr
                best_a = a.copy()

    # final check
    final_scr = compute_score(a)
    if final_scr < best_score:
        best_score = final_scr
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Simulated‑annealing refinement
# ----------------------------------------------------------------------


def annealing_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    init_temp: float = 0.05,
    final_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Stochastic annealing on the exact max‑objective.
    ``init_a`` must be a simplex vector.
    """
    a = init_a.astype(float).copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    start = time.time()
    total_time = time_deadline - start
    if total_time <= 0.0:
        return best_a, best_score

    # exponential cooling schedule
    while time.time() < time_deadline:
        elapsed = time.time() - start
        fraction = elapsed / total_time
        # temperature drops exponentially from init_temp to final_temp
        T = init_temp * (final_temp / init_temp) ** fraction

        # propose a small mass‑transfer move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move up to 30 % of the source mass
        delta = (np.random.rand() * 0.3) * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        # accept if improvement or with probability exp(-Δ/T)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Deterministic hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb_optimize(
    init_seq,
    time_deadline: float,
    max_steps: int = 1_500_000,
) -> tuple[list[float], float]:
    """
    Greedy local search: swap or transfer a small amount of mass.
    Only moves that improve the exact score are kept.
    """
    a = np.array(init_seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # start with a modest transfer fraction
    step_scale = 0.07

    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions (does not change the vector, but may change ordering
            # for the convolution due to the reversal – useful when the current
            # vector has a hidden symmetry)
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            # be more daring after a success
            step_scale = min(step_scale * 1.08, 0.5)
        else:
            # become more conservative after a failure
            step_scale *= 0.96

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 4. Helper – generate a random start on the simplex
# ----------------------------------------------------------------------


def random_start_vector(n: int) -> np.ndarray:
    """
    Return a random probability vector of length ``n``.
    With probability 0.5 it is dense (Dirichlet(1)); otherwise it is sparse
    with mass on about √n positions.
    """
    if np.random.rand() < 0.5:
        a = np.random.rand(n)
        a = a / a.sum()
    else:
        k = max(2, int(np.sqrt(n)))
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k
        a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted wall‑clock budget.
    Returns a plain Python list of floats (the sequence).
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a length that balances expressiveness and speed
    # ------------------------------------------------------------------
    # Use a larger n when we have enough time left.
    n = 2048
    if deadline - time.time() > 600.0:
        n = 4096
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2. Random‑restart smooth optimisation (fast phase)
    # ------------------------------------------------------------------
    # Reserve time for later refinements
    reserve_for_refine = 200.0
    while time.time() < deadline - reserve_for_refine:
        remaining = deadline - time.time() - reserve_for_refine
        # each restart gets at most 15 s (but never less than 4 s)
        slice_deadline = time.time() + max(4.0, min(15.0, remaining * 0.4))

        a0 = random_start_vector(n)
        a_opt, scr = smooth_gradient_optimize(
            a0,
            time_deadline=slice_deadline,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=1500.0,
            step0=2.0,
        )
        exact = compute_score(a_opt)
        if exact < best_val:
            best_val = exact
            best_seq = a_opt.tolist()
            print(f"New best after smooth phase: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing refinement (medium phase)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        print("\nSimulated‑annealing refinement")
        refined_a, refined_score = annealing_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 80.0,
            init_temp=0.08,
            final_temp=1e-7,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Improved by annealing: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient fine‑tuning (optional, if time allows)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print("\nSub‑gradient fine‑tuning")
        # a very lightweight sub‑gradient routine (few thousand iter.)
        a = np.array(best_seq, dtype=float).copy()
        n = a.shape[0]
        best_a = a.copy()
        best_score = compute_score(a)
        step0 = 0.07
        for t in range(1, 120_000 + 1):
            if time.time() >= deadline - 40.0:
                break
            conv = convolution_fft(a, a)
            k = int(np.argmax(conv))

            # sub‑gradient for the maximal entry
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g = np.zeros_like(a)
            g[mask] = 2.0 * a[j[mask]]

            step = step0 / np.sqrt(t)
            a_new = a - step * g
            a_new = project_to_simplex(a_new)

            scr = compute_score(a_new)
            if scr < best_score:
                best_score = scr
                best_a = a_new.copy()
            a = a_new

        if best_score < best_val:
            best_val = best_score
            best_seq = best_a.tolist()
            print(f"Sub‑grad improvement: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final hill‑climb polishing (slow but reliable)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nHill‑climb polishing")
        polished_seq, polished_score = hill_climb_optimize(
            best_seq,
            time_deadline=deadline,
            max_steps=1_500_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Fallback to a simple uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will call it again)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
