def adam_smooth_optimize_sym(start_b, deadline, max_iters=150000, beta_start=5.0, beta_end=150.0, lr=0.02):
    half = start_b.shape[0]
    T = 0.5
    # project onto simplex sum= T
    b = project_to_simplex(start_b) * T
    # parameterization z for b: r_i = softmax(z_i)
    eps = 1e-12
    # to avoid zeros in start we may add small jitter
    if np.any(b <= 0):
        b = b + eps
        b = b / b.sum() * T
    z = np.log(b / T + eps)   # since b = T*softmax(z) => softmax(z) = b/T
    # Actually easier: Initialize r = b/T; ensure sum r = 1.
    # So we can set z = np.log(r + eps)
    r = b / T
    z = np.log(r + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1=0.9; beta2=0.999; eps_adam=1e-12
    best_score = np.inf
    best_a = None
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # β schedule:
        t_frac = it / max_iters
        beta = beta_start * (beta_end/beta_start) ** t_frac
        # compute r = softmax(z)
        r = softmax(z) # sum=1
        b = T * r
        a = np.concatenate([b, b[::-1]])
        n = a.size
        # conv of a with a
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smoothed weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()
        # gradient wrt a
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)
        # gradient wrt b
        grad_b = grad_a[:half] + grad_a[half:][::-1] # shape half
        # gradient wrt r (since b = T * r)
        grad_r = T * grad_b
        # backprop to z
        dot = np.dot(r, grad_r)
        grad_z = r * grad_r - dot * r
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # occasionally evaluate exact score
        if it % 500 == 0:
            a_full = np.concatenate([b, b[::-1]])
            sc = compute_score(a_full)
            if sc < best_score:
                best_score = sc
                best_a = a_full.copy()
    # final check
    a_full = np.concatenate([b, b[::-1]])
    sc = compute_score(a_full)
    if sc < best_score:
        best_score = sc
        best_a = a_full.copy()
    return best_a, best_score

"""
Propose a step‑function sequence that (approximately) minimises

    2·n·max(conv(a,a)) / (sum a)^2

using a symmetric representation (a = [b, b[::-1]]), several
parameter families for warm‐starts, an Adam optimiser on the smooth
log‑sum‑exp approximation of the max‑convolution with a high
annealing β, a long projected sub‑gradient descent, and a final
simulated‑annealing hill‑climb.  The code respects the supplied
time‑budget (default 1000 s) and returns a plain list of floats ready
for `evaluate_sequence`.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Basic linear‑algebra / projection utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution using real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_to_simplex_T(v: np.ndarray, total: float) -> np.ndarray:
    """Project onto simplex with prescribed total mass `total`."""
    return project_to_simplex(v) * total


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (sums to 1)."""
    zmax = np.max(z)
    e = np.exp(z - zmax)
    return e / e.sum()


# --------------------------------------------------------------
# Symmetric handling (a = [b, b[::-1]])
# --------------------------------------------------------------

def half_to_full(b: np.ndarray) -> np.ndarray:
    """Mirror a half‑vector to obtain the full symmetric step function."""
    return np.concatenate([b, b[::-1]])


def score_from_half(b: np.ndarray) -> float:
    """Score for the full vector obtained from half‑vector `b`."""
    return compute_score(half_to_full(b))


# --------------------------------------------------------------
# Analytic families (used for quick warm‑starts)
# --------------------------------------------------------------

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α), already symmetric."""
    i = np.arange(n, dtype=np.float64)
    # avoid log(0)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β), monotone decreasing, not forced symmetric."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def random_sparse_start(halflen: int) -> np.ndarray:
    """Sparse uniform probability vector (≈5 % non‑zeros) on half‑length."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * halflen)))
    idx = np.random.choice(halflen, size=k, replace=False)
    a = np.zeros(halflen, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex_T(a, 0.5)


def random_monotone_start(halflen: int) -> np.ndarray:
    """Random decreasing half‑vector (later mirrored)."""
    r = np.random.rand(halflen) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                # decreasing
    a = a / a.sum() * 0.5      # enforce total 0.5
    return a


# --------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution
# --------------------------------------------------------------

def adam_smooth_optimize_sym(start_b: np.ndarray,
                             deadline: float,
                             max_iters: int = 150_000,
                             beta_start: float = 5.0,
                             beta_end: float = 300.0,
                             lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective
        f_β(b) = (1/β)·log Σ_k exp(β·c_k(b))
    where c = conv([b,b[::-1]], [b,b[::-1]]).
    The method works on the half‑vector `b` using the soft‑max
    parametrisation b = 0.5·softmax(z).  β is increased exponentially.
    Returns the best full vector and its exact (non‑smoothed) score.
    """
    half = start_b.shape[0]
    total = 0.5
    # initialise the soft‑max parameters from start_b
    r0 = start_b / total                     # sum to 1
    eps = 1e-12
    z = np.log(r0 + eps)                     # starting log‑odds

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_score = np.inf
    best_full = None

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # current half‑vector and full symmetric vector
        r = softmax(z)                     # sum = 1
        b = total * r
        a = half_to_full(b)                # length n = 2·half

        # convolution and smoothed weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()            # softmax over convolution entries

        # gradient w.r.t. full a (standard for the smoothed log‑sum‑exp)
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)

        # back‑project to half‑vector
        grad_b = grad_a[:half] + grad_a[half:][::-1]   # ∂f/∂b_i = ∂f/∂a_i + ∂f/∂a_{n-1-i}
        grad_r = total * grad_b                       # because b = total·r
        dot = np.dot(r, grad_r)
        grad_z = r * grad_r - dot * r                  # chain rule through soft‑max

        # Adam update in the log‑odds space
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 500 == 0:
            a_full = half_to_full(total * softmax(z))
            sc = compute_score(a_full)
            if sc < best_score:
                best_score = sc
                best_full = a_full.copy()

    # final check
    a_full = half_to_full(total * softmax(z))
    sc = compute_score(a_full)
    if sc < best_score:
        best_score = sc
        best_full = a_full.copy()

    return best_full, best_score


# --------------------------------------------------------------
# Projected sub‑gradient descent (true max‑objective)
# --------------------------------------------------------------

def subgrad_optimize_sym(start_b: np.ndarray,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on the exact max‑convolution.
    Operates on the half‑vector `b`.  The sub‑gradient is the average
    over all active (max‑attaining) convolution indices.
    """
    half = start_b.shape[0]
    total = 0.5
    b = project_to_simplex_T(start_b, total)

    best_b = b.copy()
    best_score = score_from_half(b)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = half_to_full(b)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        active = np.where(conv >= maxc - 1e-12)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size

        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)
        grad_b = grad_a[:half] + grad_a[half:][::-1]

        step = step0 / np.sqrt(t)
        b = project_to_simplex_T(b - step * grad_b, total)

        if t % 2000 == 0:
            sc = score_from_half(b)
            if sc < best_score:
                best_score = sc
                best_b = b.copy()

    # final check
    sc = score_from_half(b)
    if sc < best_score:
        best_score = sc
        best_b = b.copy()

    return half_to_full(best_b), best_score


# --------------------------------------------------------------
# Simple stochastic hill‑climb with simulated annealing
# --------------------------------------------------------------

def hill_climb_sa_sym(start_b: np.ndarray,
                     deadline: float,
                     max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex of the half‑vector.
    Moves: swap two entries or transfer a fraction of mass.
    Accepts occasional uphill steps according to a linear cooling schedule.
    """
    half = start_b.shape[0]
    total = 0.5
    cur_b = start_b.copy()
    cur_score = score_from_half(cur_b)

    best_b = cur_b.copy()
    best_score = cur_score

    # tiny temperature scale – only very rare uphill moves
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_b.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(half, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass transfer
            i, j = np.random.randint(half, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # re‑normalise to the exact total mass
        cand = project_to_simplex_T(cand, total)
        cand_score = score_from_half(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_b = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_b = cur_b.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return half_to_full(best_b), best_score

        if sc < best_score:
            best_score = sc
            best_full = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={b:.2f}) = {best_score:.6f}")

    # a few random monotone vectors (full length)
    for _ in range(60):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_full = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # ----------------------------------------------------------
    # 2. Adam smoothing on the symmetric parametrisation
    # ----------------------------------------------------------
    while time.time() < deadline - 250.0:
        # generate a random sparse half‑vector
        start_b = random_sparse_start(half)
        a_opt, sc_opt = adam_smooth_optimize_sym(
            start_b,
            deadline - 200.0,
            max_iters=180_000,
            beta_start=5.0,
            beta_end=400.0,
            lr=0.017,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_full = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after Adam‑smooth = {best_score:.6f}")

    # ----------------------------------------------------------
    # 3. Long sub‑gradient refinement (still on the symmetric space)
    # ----------------------------------------------------------
    if best_full is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        half_start = np.array(best_full[:half], dtype=np.float64)
        a_sub, sc_sub = subgrad_optimize_sym(
            half_start,
            deadline - 60.0,
            max_iters=3_000_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_full = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ----------------------------------------------------------
    # 4. Final stochastic hill‑climb (simulated annealing)
    # ----------------------------------------------------------
    if best_full is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        half_start = np.array(best_full[:half], dtype=np.float64)
        a_final, sc_final = hill_climb_sa_sym(
            half_start,
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_full = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ----------------------------------------------------------
    # 5. Safety net – fall back to a modest uniform vector
    # ----------------------------------------------------------
    if best_full is None:
        n_small = 256
        best_full = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_full)

    # ----------------------------------------------------------
    # 6. Final official evaluation (the harness will also call it)
    # ----------------------------------------------------------
    final_score = evaluate_sequence(best_full)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # return a plain Python list of floats
    return best_full
