* a greedy hill‑climb that swaps or transfers mass between positions

All phases respect the 1000 s wall‑clock budget.  The best sequence found
throughout is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (both arguments are 1‑D)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` when the vector is already normalised
    to sum = 1 (the usual case inside the optimiser).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # avoid division by zero – fall back to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytical start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a modest α‑grid and keep the best vector."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate(
        [
            np.linspace(-2.0, -0.5, 10),
            np.linspace(-0.5, 0.0, 6),
            np.linspace(0.0, 2.5, 30),
        ]
    )
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Random sparse initialisation
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Return a random sparse probability vector (size n)."""
    sparsity = np.random.uniform(0.02, 0.30)  # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a  # already on the simplex


def random_dirichlet_start(n: int, concentration: float = 0.5) -> np.ndarray:
    """Less sparse Dirichlet draw – useful for larger n."""
    x = np.random.gamma(concentration, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# 3. Exponentiated‑gradient on the smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def exp_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 800_000,
    beta_start: float = 0.5,
    beta_end: float = 800.0,
    eta0: float = 0.6,
    eps: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using a KL‑mirror (exponentiated‑gradient) step.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- convolution and soft‑max weights ----
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        # stabilise the exponentials
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # ---- gradient of the smooth max ----
        a_rev = a[::-1]
        # convolution of a_rev with w (full length 3n‑2)
        grad_full = convolution_fft(a_rev, w)
        # slice the central n entries (correspond to indices 0..n‑1)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ---- step size ----
        eta = eta0 / (it ** 0.55)

        # ---- KL‑mirror update ----
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        # ---- occasional exact evaluation ----
        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 1_200_000,
    step0: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent for objective
        2·n·max(a*a)   (sum is already 1).
    Uses the exact sub‑gradient coming from all maximal convolution slots.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    a_rev = a[::-1]  # will be refreshed each iteration

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_b = conv.max()
        # indicator of where the max is attained
        w = (conv == max_b).astype(float)

        # gradient = 2 * (a_rev * w)_{central block}
        grad_full = convolution_fft(a_rev, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        step = step0 / (t ** 0.55)
        a_new = a - step * grad
        a = project_to_simplex(a_new)

        # keep a_rev up‑to‑date for next iteration
        a_rev = a[::-1]

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 5. Greedy hill‑climb (very cheap local polishing)
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy random walk:
        – with 0.5 probability swap two entries,
        – otherwise move a fraction of mass from one index to another.
    Only moves that improve the exact score are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # start with a modest step size; it will adapt automatically
    step_scale = 0.10

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.08, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# 6. Main entry point for the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising `evaluate_sequence`.
    The routine follows a multi‑phase scheme (analytic start →
    many random exponentiated‑gradient runs → long refinement →
    sub‑gradient → hill‑climb) while respecting the 1000 s budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # 2 s safety margin

    # -------------------------------------------------
    # Choose a length based on the remaining budget
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 32768
    elif remaining > 600:
        n = 16384
    elif remaining > 400:
        n = 8192
    elif remaining > 200:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytical start (α‑grid)
    # -------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Random sparse / Dirichlet restarts with exponentiated‑gradient
    # -------------------------------------------------
    while time.time() < deadline - 250.0:
        # alternate between sparse and Dirichlet starts to increase diversity
        if np.random.rand() < 0.6:
            a0 = random_sparse_start(n)
        else:
            a0 = random_dirichlet_start(n, concentration=0.5)

        a_opt, sc_opt = exp_smooth_optimize(
            a0,
            deadline - 180.0,
            max_iters=200_000,
            beta_start=0.5,
            beta_end=800.0,
            eta0=0.7,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after random restart = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Long dedicated exponentiated‑gradient run
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] long exponentiated‑gradient optimisation")
        a_smooth, sc_smooth = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=600_000,
            beta_start=0.5,
            beta_end=1200.0,
            eta0=0.6,
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.tolist()
            print(f"[{time.time():.1f}] after long exp‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Sub‑gradient refinement (more aggressive)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 45.0,
            max_iters=1_500_000,
            step0=0.15,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Final hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: fall back to a short uniform vector if needed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats.
    return best_seq
