"""
Propose a non‑negative step sequence with a low ``evaluate_sequence`` value.

The search proceeds in four phases (all respecting the same wall‑clock budget):
1.   A fast parametric sweep over Beta‑like shapes (including arcsine,
     uniform, mixtures).  The best seed is kept.
2.   Smooth‑max Adam optimisation of the log‑softmax parameters of the
     seed (continuously approximates the true max‑convolution objective).
3.   Mirror‑descent (exponentiated‑gradient) polishing on the exact max‑convolution.
4.   Projected Euclidean sub‑gradient descent for a final polish.

All phases use FFT‑based convolutions (O(n log n)) and operate on a
probability vector (sum = 1), because the official objective is scale‑invariant.
The routine always returns a valid (non‑empty) sequence before the 1000 s
deadline.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for ``evaluate_sequence`` when ``sum(a) == 1``.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic shape families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (discrete) normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Discrete Beta(α,β) shape on a grid of size n.
    The density is proportional to (i+eps)^{α‑1}·(n‑1‑i+eps)^{β‑1}.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a point on the simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 800.0,
    ramp_iters: int = 120_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of ``max(conv)``.
    The iterate lives in log‑softmax coordinates; the returned vector
    is already normalised (sum = 1, non‑negative).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # simplex point
        conv = convolution_fft(a, a)       # length 2n‑1
        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)         # non‑negative weights
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of smooth max w.r.t. a:   2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if t % 8_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step: float = 0.016,
) -> np.ndarray:
    """
    Mirror descent (KL divergence) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)
    a /= a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if it % 6_000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.085,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 8_000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Parameter sweep for analytic seeds
# ----------------------------------------------------------------------


def sweep_beta_shapes(
    n: int,
    rng: np.random.Generator,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Systematic and random search over Beta‑like families.
    Returns the best sequence found together with its surrogate score.
    """
    # ------------------------------------------------------------------
    # 1) deterministic grid around the arcsine parameters (α≈β≈0.5)
    # ------------------------------------------------------------------
    alphas = np.linspace(0.35, 0.80, 20)          # more weight away from the edges
    eps_vals = [0.0, 0.1, 0.5, 1.0, 2.0]
    for a in alphas:
        for eps in eps_vals:
            if time.time() >= deadline:
                return best_seq, best_score
            seq = generate_beta_shape(n, a, a, eps)
            scr = compute_score(seq)
            if scr < best_score:
                best_seq, best_score = seq, scr
                print(f"    Grid β‑shape α={a:.3f} eps={eps:.2f} → score {scr:.6f}")

    # ------------------------------------------------------------------
    # 2) mixture of arcsine and uniform (search weight)
    # ------------------------------------------------------------------
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    for w in np.linspace(0.0, 1.0, 21):
        if time.time() >= deadline:
            return best_seq, best_score
        mix = w * arc + (1.0 - w) * uni
        mix /= mix.sum()
        scr = compute_score(mix)
        if scr < best_score:
            best_seq, best_score = mix, scr
            print(f"    Mixture weight {w:.2f} → score {scr:.6f}")

    # ------------------------------------------------------------------
    # 3) random exploration (as long as time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a = rng.uniform(0.1, 1.4)
        b = rng.uniform(0.1, 1.4)
        eps = rng.uniform(0.0, 5.0)
        seq = generate_beta_shape(n, a, b, eps)
        scr = compute_score(seq)
        if scr < best_score:
            best_seq, best_score = seq, scr
            print(f"    Random β‑shape α={a:.3f} β={b:.3f} eps={eps:.2f} → score {scr:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 43, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock budget (≈1000 s) and always returns a
    valid sequence.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Choose a working length (power of two for FFT efficiency)
    # ------------------------------------------------------------------
    # The longer the vector the more flexibility we have, but also the slower
    # each iteration.  8192 offers a good trade‑off for the allotted time.
    n = 8192

    best_seq: np.ndarray = np.empty(0)
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0) fallback to an already known construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array([float(x) for x in globals()["height_sequence_1"]], dtype=float)
            cand = np.clip(cand, 0.0, None)
            if cand.sum() == 0:
                cand = np.full_like(cand, 1.0 / cand.size)
            else:
                cand /= cand.sum()
            scr = compute_score(cand)
            print(f"Fallback (height_sequence_1) surrogate score = {scr:.6f}")
            best_seq, best_score = cand, scr
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) Analytic seed sweep (fast)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n=== Phase 1 – analytic seed sweep ===")
        # start from arcsine – already a very good candidate
        init = arcsine_initial(n)
        init_score = compute_score(init)
        if init_score < best_score:
            best_seq, best_score = init, init_score
            print(f"    Initial arcsine surrogate score = {init_score:.6f}")

        best_seq, best_score = sweep_beta_shapes(
            n, rng, deadline - 10.0, best_seq, best_score,
        )
        print(f"Phase 1 best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑max Adam optimisation (continuous proxy)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("\n=== Phase 2 – Adam on smooth‑max proxy ===")
        # Initialise Adam in log‑softmax space
        eps = 1e-12
        z0 = np.log(best_seq + eps)

        adam_deadline = deadline - 140.0
        a_opt = smooth_adam_optimize(
            z0,
            adam_deadline,
            n=n,
            beta0=5.0,
            beta_max=800.0,
            ramp_iters=120_000,
            lr=0.07,
        )
        scr = compute_score(a_opt)
        print(f"    After Adam surrogate score = {scr:.6f}")
        if scr < best_score:
            best_seq, best_score = a_opt, scr

    # ------------------------------------------------------------------
    # 3) KL‑mirror (exponentiated‑gradient) polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print("\n=== Phase 3 – Exponentiated‑gradient polish ===")
        eg_deadline = deadline - 70.0
        a_eg = exponentiated_gradient_polish(
            best_seq,
            eg_deadline,
            max_iter=180_000,
            step=0.016,
        )
        scr = compute_score(a_eg)
        print(f"    After EG surrogate score = {scr:.6f}")
        if scr < best_score:
            best_seq, best_score = a_eg, scr

    # ------------------------------------------------------------------
    # 4) Final Euclidean sub‑gradient descent (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print("\n=== Phase 4 – Final sub‑gradient polishing ===")
        subgrad_deadline = deadline - 5.0
        a_final = subgradient_descent_exact(
            best_seq,
            subgrad_deadline,
            max_iter=250_000,
            step0=0.085,
        )
        scr = compute_score(a_final)
        print(f"    Final surrogate score = {scr:.6f}")
        if scr < best_score:
            best_seq, best_score = a_final, scr

    # ------------------------------------------------------------------
    # 5) Guard – ensure a non‑empty return
    # ------------------------------------------------------------------
    if best_seq.size == 0:
        n0 = 512
        best_seq = np.full(n0, 1.0 / n0)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 6) Report official score (uses the provided ``evaluate_sequence``)
    # ------------------------------------------------------------------
    try:
        official = evaluate_sequence(best_seq.tolist())
        print(f"\nSearch finished. Official score = {official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq.tolist()
