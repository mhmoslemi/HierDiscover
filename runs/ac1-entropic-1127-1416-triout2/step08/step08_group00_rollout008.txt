def simulated_annealing(a0, time_deadline, init_temp=1e-3, min_temp=1e-6, cooling=0.9995):
   a = a0.copy()
   current_score = compute_score(a)
   best_a = a.copy()
   best_score = current_score
   T = init_temp
   n = len(a)
   step_scale = 0.1
   while time.time() < time_deadline and T > min_temp:
       # propose
       i, j = np.random.randint(n, size=2)
       if i == j:
           continue
       delta = step_scale * a[i] * np.random.rand()
       # ensure not exceed a[i]
       if delta > a[i]:
           delta = a[i]
       cand = a.copy()
       cand[i] -= delta
       cand[j] += delta
       # no need to project, sum unchanged; but some small negative may appear due to rounding, fix:
       if cand[i] < 0:
           cand[i] = 0
       conv = convolution_fft(cand, cand)
       new_score = 2.0 * n * float(conv.max())
       # acceptance
       accept = False
       if new_score < current_score:
           accept = True
       else:
           prob = np.exp(-(new_score - current_score) / T)
           if np.random.rand() < prob:
               accept = True
       if accept:
           a = cand
           current_score = new_score
           if new_score < best_score:
               best_score = new_score
               best_a = cand.copy()
               # optionally increase step_scale
               step_scale = min(step_scale * 1.02, 0.5)
           else:
               step_scale = max(step_scale * 0.99, 1e-3)
       T *= cooling
   return best_a, best_score

"""
search for a non‑negative step‑function that minimises the
`evaluate_sequence` objective.

The algorithm consists of several stages:

1. **Many random restarts of a smooth‑max Adam optimiser**.  
   The smooth surrogate of `max(conv(a,a))` is the log‑sum‑exp
   with an annealed temperature β.  Each run projects onto the
   probability simplex (sum = 1) and keeps the best exact score.

2. **A single L‑BFGS refinement** (if SciPy is available).  
   Optimises the same smooth surrogate but with a quasi‑Newton
   method and an exact line search, typically improving the
   solution obtained by Adam.

3. **Simulated‑annealing polishing**.  Small mass‑transfer moves are
   accepted according to a temperature schedule, allowing the
   search to escape shallow local minima.

4. **Deterministic hill‑climb** – the same routine as in the
   reference implementation, accepting only improving moves.

All stages respect the global 1000 s wall‑clock budget.  The final
result is returned as a plain Python list of floats (the heights).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global parameters used by the optional SciPy L‑BFGS optimiser
# ----------------------------------------------------------------------
CURRENT_N = None          # problem dimension (set before each L‑BFGS call)
CURRENT_BETA = None       # smooth‑max temperature (set before each call)

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using the real FFT (fast for large arrays)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑accelerated minimisation of the log‑sum‑exp surrogate of
    `max(conv(a,a))`.  Returns the best exact score seen.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # convolution and soft‑max weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable log‑sum‑exp
        shift = beta * (conv - maxc)
        w = np.exp(shift)
        w_sum = w.sum()
        # protect against numerical issues
        if w_sum == 0.0:
            probs = np.full_like(w, 1.0 / w.shape[0])
        else:
            probs = w / w_sum

        # gradient of the soft‑max (via cross‑correlation)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, probs)[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Simulated‑annealing polishing
# ----------------------------------------------------------------------


def simulated_annealing(
    start_a: np.ndarray,
    time_deadline: float,
    init_temp: float = 1e-3,
    min_temp: float = 1e-6,
    cooling: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Stochastic annealing on the exact objective.
    Moves are simple mass‑transfer operations between two indices.
    """
    a = start_a.copy()
    n = a.shape[0]
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score
    T = init_temp
    step_scale = 0.08  # typical fraction of the source mass to move

    while time.time() < time_deadline and T > min_temp:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # amount to move – a fraction of the source bucket
        delta = step_scale * a[i] * np.random.rand()
        if delta > a[i]:
            delta = a[i]

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # guard against tiny negatives caused by floating point noise
        if cand[i] < 0.0:
            cand[i] = 0.0

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / T)
            accept = np.random.rand() < prob

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                step_scale = min(step_scale * 1.07, 0.5)   # become bolder after success
            else:
                step_scale *= 0.96                       # become more conservative
        else:
            # still adapt step size a little so we can explore different scales
            step_scale *= 0.99

        T *= cooling

    return best_a, best_score

# 4. Stochastic hill‑climb polishing (greedy, improvement‑only)
# ----------------------------------------------------------------------


def hill_climb_polish(
    start_seq,
    time_deadline,
    max_steps: int = 2_000_000,
) -> tuple[list[float], float]:
    """
    Deterministic hill‑climb: random swaps or tiny mass‑transfer moves,
    accepted only if they improve the exact score.
    """
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score

# 5. Optional L‑BFGS refinement (requires SciPy)
# ----------------------------------------------------------------------


def _lbfgs_objective_and_grad(z: np.ndarray):
    """Return (objective, gradient) for the log‑sum‑exp surrogate."""
    global CURRENT_N, CURRENT_BETA
    n = CURRENT_N
    beta = CURRENT_BETA

    # softmax parametrisation (keeps a on the simplex)
    a = np.exp(z)
    a_sum = a.sum()
    a = a / a_sum

    conv = convolution_fft(a, a)
    maxc = conv.max()

    # stable log‑sum‑exp
    shift = beta * (conv - maxc)
    w = np.exp(shift)
    w_sum = w.sum()
    log_sum_exp = maxc + np.log(w_sum) / beta
    obj = 2.0 * n * log_sum_exp

    # soft‑max probabilities
    probs = w / w_sum

    # gradient w.r.t a (same formula as in the Adam optimiser)
    a_rev = a[::-1]
    grad_a = 2.0 * convolution_fft(a_rev, probs)[n - 1 : n - 1 + n]

    # chain rule for the softmax parametrisation
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)

    return obj, grad_z


def lbfgs_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    beta: float = 2000.0,
    maxfun: int = 200,
) -> tuple[np.ndarray, float]:
    """
    One‑shot L‑BFGS optimisation of the smooth surrogate.
    Returns the best exact score encountered.
    """
    global CURRENT_N, CURRENT_BETA
    CURRENT_N = start_a.shape[0]
    CURRENT_BETA = beta

    # initialise the log‑softmax variables (avoid zeros)
    z0 = np.log(np.maximum(start_a, 1e-12))

    # lazy import – SciPy may be unavailable on the platform
    try:
        import scipy.optimize as opt
    except Exception:   # pragma: no cover
        # fall back to the start vector if SciPy cannot be imported
        return start_a, compute_score(start_a)

    # minimise the smooth objective (L‑BFGS‑B respects simple box bounds)
    options = {"maxfun": maxfun, "maxiter": maxfun, "disp": False}
    res = opt.minimize(
        _lbfgs_objective_and_grad,
        z0,
        method="L-BFGS-B",
        jac=True,
        options=options,
    )
    if not res.success:
        a_opt = start_a
    else:
        a_opt = np.exp(res.x)
        a_opt /= a_opt.sum()

    exact = compute_score(a_opt)
    return a_opt, exact

def random_start(n: int) -> np.ndarray:
    """Return a random probability vector of length `n` (dense or structured)."""
    r = np.random.rand()
    if r < 0.3:
        # uniform
        a = np.full(n, 1.0 / n)
    elif r < 0.6:
        # Dirichlet (α=1) – dense random simplex
        a = np.random.rand(n)
        a = a / a.sum()
    elif r < 0.8:
        # triangular (peak in the centre)
        i = np.arange(n)
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(i - centre) / centre
        a = np.clip(a, 0, None)
        a = a / a.sum()
    else:
        # cosine‑shaped (non‑negative)
        i = np.arange(n)
        a = np.sin(np.pi * (i + 1) / (n + 1))
        a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give ourselves a tiny safety margin
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a workable dimension
    # ------------------------------------------------------------------
    n = 2048
    if deadline - time.time() > 800:
        n = 4096
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2. Random‑restart Adam optimisation on the smooth surrogate
    # ------------------------------------------------------------------
    reserve_for_refine = 180.0  # seconds saved for later phases
    while time.time() < deadline - reserve_for_refine:
        remaining = deadline - time.time() - reserve_for_refine
        # allocate a time slice per restart (between 5 s and 30 s)
        slice_time = max(5.0, min(30.0, remaining * 0.4))
        time_limit = time.time() + slice_time

        a0 = random_start(n)
        a_opt, scr = smooth_gradient_optimize(
            a0,
            time_deadline=time_limit,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=2000.0,
            step0=1.0,
        )
        exact = compute_score(a_opt)
        if exact < best_val:
            best_val = exact
            best_seq = a_opt.tolist()
            print(f"New best after Adam phase: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Optional L‑BFGS refinement (if SciPy is present)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        try:
            import scipy  # noqa: F401  # check availability
            print("\nL‑BFGS refinement")
            a_lbfgs, lbfgs_score = lbfgs_optimize(
                np.array(best_seq, dtype=float),
                time_deadline=deadline - 40.0,
                beta=2000.0,
                maxfun=300,
            )
            if lbfgs_score < best_val:
                best_val = lbfgs_score
                best_seq = a_lbfgs.tolist()
                print(f"L‑BFGS improved best to {best_val:.6f}")
        except Exception:
            pass  # SciPy not available – ignore

    # ------------------------------------------------------------------
    # 4. Sub‑gradient refinement (shortened)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print("\nSub‑gradient refinement")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 30.0,
            max_iter=250_000,
            step0=0.04,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑grad improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        print("\nSimulated‑annealing polishing")
        sa_seq, sa_score = simulated_annealing(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 10.0,
            init_temp=1e-3,
            min_temp=1e-6,
            cooling=0.9996,
        )
        if sa_score < best_val:
            best_val = sa_score
            best_seq = sa_seq.tolist()
            print(f"SA improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final deterministic hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nFinal hill‑climb polishing")
        final_seq, final_score = hill_climb_polish(
            best_seq,
            time_deadline=deadline,
            max_steps=2_000_000,
        )
        if final_score < best_val:
            best_val = final_score
            best_seq = final_seq
            print(f"Hill‑climb final best: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Fallback to a simple uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Report the final score (for debugging – the harness calls
    # `evaluate_sequence` again after we return).
    # ------------------------------------------------------------------
    final_report = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_report:.6f}")

    return best_seq
