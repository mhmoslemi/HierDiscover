def solve_lp_max_sum(a, M):
    n = a.shape[0]
    # pattern P
    offsets = np.arange(n)
    diags = [np.ones(n - k) for k in range(n)]
    P = sp.diags(diags, offsets, shape=(2*n-1, n), format='csc')

    if not res.success:
        raise ValueError("LP failed")
    return res.x

    rng = np.random.default_rng(seed)
    start = time.time()

    best_seq = None

        for _ in range(4):

            if time.time() >= deadline - 2:

            a = seed_vec.copy()
            # ensure sum=1
            a = a / a.sum()
            # improvement with LP mixing
            a = lp_mix_improve(a, deadline, max_iter=15)
            # optional polishing
            if time.time() < deadline - 1:

            try:

        print(f"Fallback uniform score = {best_score:.6f}")

    return best_seq

def solve_lp_max_sum(a, M):
    n = a.shape[0]
    m = 2 * n - 1
    # construct A matrix
    A = np.empty((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a[j]
    # scale
    A_ub = A / M
    b_ub = np.ones(m, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x
```

            break
    return a

"""
Search for a non‑negative step sequence that minimises
``evaluate_sequence`` using an LP‑based iterative refinement.

The algorithm works as follows (for a fixed length n ≥ 512):

1. Initialise a on the simplex (sum = 1) with a smooth “U‑shape”
   (arcsine distribution) and optionally a few random seeds.

2. Repeatedly improve the shape:
   * Let M = max(conv(a, a)).
   * Solve the linear programme
         maximise   Σ b_j
         subject to conv(a, b) ≤ M   (component‑wise)
                    b_j ≥ 0.
     The LP has n variables and 2 n‑1 linear constraints; it is solved
     with SciPy’s HiGHS optimiser.
   * Scale the optimal b to have unit sum (b ← b / Σb) – this keeps the
     bound conv(a, b) ≤ M.
   * Mix the current a with the new b:
         a ← (1‑t)·a + t·b   (t starts at 0.5 and is back‑tracked if the
                              max‑convolution does not decrease).
     The mixing step never increases the maximum convolution,
     so the objective 2·n·max(conv(a,a)) strictly decreases or stays
     the same.

3. After the LP‑driven phase a short projected sub‑gradient descent
   (exact max‑convolution sub‑gradient) is applied to polish the result.

4. Finally the best candidate found (according to the official
   ``evaluate_sequence``) is returned.  All numbers are non‑negative,
   sum ≥ 0.01 and ≤ 1000, so the security checks in ``evaluate_sequence``
   never reject the result.

The whole search respects the supplied time‑budget (default 1000 s)
and falls back to a uniform sequence if SciPy is unavailable.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(N log N))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_score(a: np.ndarray) -> float:
    """
    Cheap surrogate: 2·n·max(conv(a,a))  (a is assumed to sum to 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (≈Beta(½,½)), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float) -> np.ndarray:
    """Shape ∝ 1/((i+½)(n‑½‑i)+ε)^p, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + 1e-8) ** p
    vals = 1.0 / denom
    return vals / vals.sum()


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex (Dirichlet with random concentration)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)   # ~0.01 … 5
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent for 2·n·max(conv)/sum²."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        rev_mask = mask[::-1]
        rev = convolution_fft(a, rev_mask)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a -= eta * subgrad
        a = _project_to_simplex(a)

        if it % 10_000 == 0:
            print(f"    Sub‑grad it={it:,d}  surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑program sub‑problem ( maximise Σb  s.t. a★b ≤ M )
# ----------------------------------------------------------------------


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that (A @ b)[k] = (a★b)[k].
    Each column j contains a[j] on the rows j … j+n‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.empty((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a[j]
    return A


def solve_lp_max_sum(a: np.ndarray, M: float):
    """
    Solve:
        maximise Σ b_j
        subject to  conv(a, b) ≤ M  (element‑wise)
                    b_j ≥ 0
    Returns the optimal b (numpy array of length n).
    """
    import scipy.optimize as opt

    n = a.shape[0]
    A = _build_conv_matrix(a)          # (2n‑1) × n
    # Scale to avoid tiny RHS
    A_ub = A / M
    b_ub = np.ones(2 * n - 1, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)   # maximise Σb  ⇔ minimise −Σb
    bounds = [(0, None)] * n

    res = opt.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# LP‑driven refinement of a step sequence
# ----------------------------------------------------------------------


def lp_mix_improve(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 25,
    t_initial: float = 0.5,
) -> np.ndarray:
    """
    Repeatedly:
        - solve the LP (maximise Σb under a★b ≤ M)
        - normalise b to sum = 1
        - mix a ← (1‑t)·a + t·b, with back‑tracking on t if the
          max‑convolution does not decrease.
    Returns the improved a (still on the simplex).
    """
    n = a.shape[0]
    best_obj = fast_score(a)
    t = t_initial

    for it in range(max_iter):
        if time.time() >= deadline - 0.5:
            break

        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        try:
            b = solve_lp_max_sum(a, M)
        except Exception as e:
            print(f"    LP solve failed (iter {it}): {e}")
            break

        sum_b = b.sum()
        if sum_b <= 0:
            break
        b_norm = b / sum_b

        # try mixing; back‑track on t until improvement is seen
        improved = False
        t_cur = t
        while t_cur > 1e-4:
            a_new = (1.0 - t_cur) * a + t_cur * b_norm
            a_new /= a_new.sum()               # keep unit sum
            obj_new = fast_score(a_new)
            if obj_new < best_obj * (1.0 - 1e-12):
                best_obj = obj_new
                a = a_new
                improved = True
                # mildly increase t for next iteration
                t = min(0.9, t_cur * 1.2)
                print(f"    LP‑iter {it:02d} improved to {best_obj:.6f} (t={t_cur:.4f})")
                break
            t_cur *= 0.5

        if not improved:
            # no improvement with this LP solution – stop early
            print(f"    LP‑iter {it:02d} no improvement, terminating LP phase.")
            break

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) minimising
    ``evaluate_sequence``.  The method combines an LP‑based refinement
    with a final exact sub‑gradient polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4      # safety margin

    best_seq: List[float] = []
    best_official: float = float("inf")

    # Small utility to evaluate and possibly update the global best
    def try_update(candidate: np.ndarray):
        nonlocal best_seq, best_official
        try:
            val = evaluate_sequence(candidate.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, candidate.tolist()
                print("    *** New global best! ***")
        except Exception as exc:
            print(f"    Evaluation error: {exc}")

    # --------------------------------------------------------------
    # Choose sequence lengths (larger lengths give lower theoretical bounds)
    # --------------------------------------------------------------
    lengths = [4096, 2048, 1024]   # will try the largest that fits the budget

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} (time left {deadline - time.time():.1f}s) ===")

        # ----- seeding ------------------------------------------------
        seeds: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            generate_power_shape(n, p=0.6),
        ]
        # a few random Dirichlet seeds for diversity
        for _ in range(4):
            seeds.append(generate_random_dirichlet(n, rng))

        # ----- process each seed --------------------------------------
        per_seed_time = max(2.0, (deadline - time.time()) / (len(seeds) * 2.0))

        for idx, seed_vec in enumerate(seeds, start=1):
            if time.time() >= deadline - 2.0:
                break
            print(f"\nSeed {idx}/{len(seeds)} – budget ≈{per_seed_time:.1f}s")
            a = seed_vec.copy()
            a /= a.sum()                     # normalise to unit sum
            seed_deadline = time.time() + per_seed_time

            # 1) LP‑driven refinement
            a = lp_mix_improve(a, seed_deadline, max_iter=20, t_initial=0.5)

            # 2) quick official evaluation after the LP phase
            try_update(a)

            # 3) short exact sub‑gradient polish (if time permits)
            if time.time() < deadline - 1.0:
                print("    Polishing with exact sub‑gradient descent …")
                a = subgradient_descent_exact(
                    a,
                    deadline,
                    max_iter=120_000,
                    step0=0.06,
                )
                try_update(a)

    # --------------------------------------------------------------
    # Fallback – guarantee a non‑empty, valid return
    # --------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence (n={n0}) score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
