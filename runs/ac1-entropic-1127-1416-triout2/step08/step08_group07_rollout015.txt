The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max\limits_{k}(a\star a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the objective is invariant under a positive scaling of the vector, we
restrict ourselves to the probability simplex  

\[
\sum_i a_i = 1,\qquad a_i\ge 0 .
\]

For a fixed length `n` the problem is a non‑convex optimisation of a
max‑convolution.  The programme below

1. **generates a very diverse pool of initial vectors** (arcsine, uniform,
   mixtures, symmetric and asymmetric β‑distributions, power‑law,
   linear ramps and random Dirichlet draws);
2. **keeps the most promising seeds** according to the cheap surrogate  
   `fast_score` (the exact objective without the clipping performed by
   `evaluate_sequence`);
3. **refines each seed with a long projected sub‑gradient descent** on the
   exact max‑convolution objective (the same sub‑gradient that appears in the
   theoretical analysis);
4. **applies a targeted local‑search** that repeatedly moves a small amount
   of probability mass from the coordinate with the largest sub‑gradient
   (the direction that would increase the objective) to the coordinate with
   the smallest sub‑gradient (the most promising direction for decrease);
5. **keeps the best sequence found** and finally returns it.

All phases are bounded by the overall wall‑clock budget (default 1000 s).  
The algorithm never returns an empty list; if everything else fails a uniform
vector of length 1024 is used as a safe fallback.

```python
"""
Search for a non‑negative step sequence that minimises ``evaluate_sequence``.

The procedure is:
1. build a large, diverse set of seed vectors (analytic families + random),
2. keep the best few according to a cheap surrogate (fast_score),
3. for every retained seed run a long projected sub‑gradient descent
   on the exact max‑convolution objective,
4. polish the best result with a deterministic, sub‑gradient‑guided
   pairwise mass–transfer local search,
5. return the best sequence discovered (or a uniform fallback).

The code respects a total time budget (default 1000 s) and contains many
print statements for progress monitoring.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Cheap surrogate (scale invariant)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a))/sum(a)^2 without clipping."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic generators for diverse starting shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine shape (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float,
                       eps: float = 0.0) -> np.ndarray:
    """Beta(α,β) density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float,
                         eps: float = 1e-8) -> np.ndarray:
    """Shape ∝ 1/((i+½)*(n‑½‑i)+eps)^p."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_linear_ramp(n: int, forward: bool = True) -> np.ndarray:
    """Linear ramp (monotonically increasing or decreasing)."""
    if forward:
        vals = np.arange(1, n + 1, dtype=float)
    else:
        vals = np.arange(n, 0, -1, dtype=float)
    return vals / vals.sum()


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with random concentration)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)          # concentration between 0.01 and 5
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σ x_i = 1, x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(init_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 400_000,
                              step0: float = 0.12) -> np.ndarray:
    """
    Projected sub‑gradient descent for
    f(a)=2·n·max(conv(a,a))/sum(a)^2.
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # identify all indices attaining the max (up to numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution (see notes in the initial code)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # diminishing step size η_t = step0 / sqrt(t)
        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 20_000 == 0:
            print(f"    Sub‑grad it={it:,d}  surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Targeted local refinement – move mass from “bad” to “good” coordinates
# ----------------------------------------------------------------------


def targeted_local_refine(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          max_iter: int = 30_000,
                          frac: float = 0.08) -> np.ndarray:
    """
    Guided pairwise mass‑transfer.  In each iteration we compute the sub‑gradient
    of the max‑convolution, move a small fraction of the probability mass from
    the coordinate with the largest sub‑gradient (most harmful) to the one
    with the smallest sub‑gradient (most helpful), keep the change only if the
    surrogate improves.
    """
    a = a0.copy()
    best = a.copy()
    best_score = fast_score(a)
    n = a.size
    L = 2 * n - 1

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        i = int(np.argmax(subgrad))
        j = int(np.argmin(subgrad))
        if a[i] < 1e-14:
            # nothing to move – pick a random pair instead
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

        # amount to move: a fraction of a[i] (but never more than a[i])
        delta = min(a[i] * frac, a[i] * rng.random())
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta
        a = _project_to_simplex(a)   # keep exact sum = 1 and non‑negative

        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
        else:
            # reject – revert
            a[j] -= delta
            a[i] += delta
            a = _project_to_simplex(a)

        if it % 5_000 == 0:
            print(f"    Local‑refine it={it:,d}  best surrogate≈{best_score:.6f}")

    return best

                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) minimising
    ``evaluate_sequence``.  The routine combines a large pool of analytic
    seeds, a long projected sub‑gradient descent, and a final
    sub‑gradient‑guided local search.  All work respects the overall time
    budget (default 1000 s).  A non‑empty list is always returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.4          # tiny safety margin

    best_seq: List[float] = []
    best_official: float = float("inf")
    best_surrogate: float = float("inf")

    # --------------------------------------------------------------
    # Optional fallback – a previously discovered construction
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, cand
                best_surrogate = fast_score(np.array(cand))
        except Exception:
            pass

    # --------------------------------------------------------------
    # Lengths to explore – larger n gives a lower theoretical bound
    # --------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} (time left {deadline - time.time():.1f}s) ===")

        # ----------------------------------------------------------
        # 1. Build a very rich pool of seed vectors
        # ----------------------------------------------------------
        seeds: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            generate_mixture_shape(n, weight_arc=0.6),
            generate_linear_ramp(n, forward=True),
            generate_linear_ramp(n, forward=False),
        ]

        # symmetric Beta(α,α)
        for alpha in np.linspace(0.3, 3.0, 9):
            seeds.append(generate_beta_shape(n, alpha, alpha))

        # asymmetric Beta(α,β)
        for alpha in np.linspace(0.3, 3.0, 5):
            for beta in np.linspace(0.3, 3.0, 5):
                seeds.append(generate_beta_shape(n, alpha, beta))

        # power‑law family
        for p in np.linspace(0.2, 1.8, 8):
            seeds.append(generate_power_shape(n, p))

        # random Dirichlet draws (various concentrations)
        for _ in range(25):
            seeds.append(generate_random_dirichlet(n, rng))

        # ----------------------------------------------------------
        # 2. Keep the most promising seeds according to the cheap score
        # ----------------------------------------------------------
        scored: List[Tuple[float, np.ndarray]] = [(fast_score(v), v) for v in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:10]]   # keep ten best per length

        # ----------------------------------------------------------
        # 3. Refine each retained seed with a long sub‑gradient descent
        # ----------------------------------------------------------
        per_seed_budget = max(2.0, (deadline - time.time()) / (len(top_seeds) * 2.0))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 2.0:
                break
            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_budget:.1f}s")
            seed_deadline = time.time() + per_seed_budget

            # Run a reasonably long projected sub‑gradient descent
            a_opt = subgradient_descent_exact(
                seed_vec,
                seed_deadline,
                max_iter=300_000,
                step0=0.15,
            )

            # --------------------------------------------------
            # 4. Official evaluation of the refined result
            # --------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score after sub‑grad = {val:.6f}")
                if val < best_official:
                    best_official, best_seq = val, a_opt.tolist()
                    best_surrogate = fast_score(a_opt)
                    print("   *** New global best! ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

    # --------------------------------------------------------------
    # 5. Final polishing – long sub‑gradient descent on the current best
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\nFinal polishing with a long sub‑gradient run …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=500_000,
            step0=0.08,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_official:
                best_official, best_seq = final_val, a_arr.tolist()
                best_surrogate = fast_score(a_arr)
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final evaluation error: {e}")

    # --------------------------------------------------------------
    # 6. Targeted local search around the polished solution
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:
        print("\nTargeted local refinement …")
        a_start = np.array(best_seq, dtype=float)
        a_refined = targeted_local_refine(
            a_start,
            deadline,
            rng,
            max_iter=40_000,
            frac=0.07,
        )
        try:
            val_loc = evaluate_sequence(a_refined.tolist())
            print(f"   Local‑search official score = {val_loc:.6f}")
            if val_loc < best_official:
                best_official, best_seq = val_loc, a_refined.tolist()
                best_surrogate = fast_score(a_refined)
                print("*** Best improved by targeted local search ***")
        except Exception as e:
            print(f"   Local‑search evaluation error: {e}")

    # --------------------------------------------------------------
    # 7. Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        n0 = 1024
        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
```
