"""
Propose a step‑function sequence with an improved bound.

The search consists of five phases:

1. **Broad shape exploration** – generate many candidate probability
   vectors (uniform, arcsine, power‑law tails, monotone power‑laws,
   mixtures and random Dirichlet draws) for several lengths.
   The best surrogate (2·n·max(conv)) is memorised.

2. **Smooth‑max Adam refinement** – starting from the best candidate,
   minimise the smooth‑max surrogate  (log‑sum‑exp approximation of
   max(conv)) using a softmax parametrisation.  The temperature is
   driven up to τ=2000 to obtain a very sharp approximation of the
   true maximum.

3. **Exact sub‑gradient descent** – Euclidean‑projected sub‑gradient
   descent on the true max‑convolution (the same update used in the
   reference solution) but with a longer schedule.

4. **Exponentiated‑gradient polish** – mirror‑descent w.r.t. KL to
   converge quickly in the probability simplex.

5. **Permutation hill‑climbing** – random swaps of whole entries
   (instead of tiny mass transfers) can substantially lower the
   max‑convolution for non‑symmetric vectors.  Swaps that improve the
   surrogate are kept.

All phases respect the global 1000 s wall‑clock budget.  The final
candidate is evaluated with the official `evaluate_sequence` function
and returned.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate: assumes Σa = 1, returns 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# Shape generators

def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (regularised) – Σ = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """
    Generalised arcsine shape:
        a_i ∝ ((i+½)*(n‑½‑i)+ε)^(-p) .
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_monotone_power(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """Monotone decreasing power law a_i ∝ (i+1)^(-p)."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / ((i + 1.0) ** p + eps)
    return vals / vals.sum()


def generate_reverse_monotone_power(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """Monotone increasing power law (large values at the right)."""
    return generate_monotone_power(n, p, eps)[::-1]


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Gamma/Dirichlet)."""
    draws = rng.gamma(alpha, 1.0, size=n)
    if draws.sum() == 0:
        draws = np.ones_like(draws)
    return draws / draws.sum()


def alternate_ends_order(values: np.ndarray) -> np.ndarray:
    """Place the sorted values alternating from the left and right ends."""
    sorted_vals = np.sort(values)[::-1]          # descending
    n = len(values)
    out = np.empty_like(values)
    left, right = 0, n - 1
    turn = True
    for v in sorted_vals:
        if turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        turn = not turn
    return out

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 300_000,
                             tau_start: float = 0.5,
                             tau_end: float = 2000.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """Adam optimiser on the smooth‑max approximation of max(conv)."""
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)               # logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # soft‑max to obtain probabilities
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        conv = convolution_fft(a, a)          # length 2n‑1

        # smooth‑max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (2n‑1,)

        # gradient wrt a (see derivation in original code)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # soft‑max Jacobian
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 75000 == 0:
            s = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={s:.6f}")

    # final projection to simplex (soft‑max)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.07) -> np.ndarray:
    """Euclidean projected sub‑gradient descent on the true max‑conv."""
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries attaining the max (within tiny tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.25) -> np.ndarray:
    """Exponentiated‑gradient (mirror descent w.r.t. KL) polish."""
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def permutation_hill_climbing(a: np.ndarray,
                              deadline: float,
                              max_trials: int = 30_000,
                              rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Random full‑value swaps.  Accept only improving moves
    (according to the surrogate `2·n·max(conv)`).
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.2 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        a_new = a_cur.copy()
        a_new[i], a_new[j] = a_new[j], a_new[i]
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1

    return a_cur

            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"  New best (mix w={w:.2f}) surrogate={s:.6f}")

        # random Dirichlet draws (different sparsities)
        for alpha in dirichlet_alphas:
            for _ in range(30):
                cand = random_dirichlet(n, alpha, rng)
                s = compute_score(cand)
                if s < best_surrogate:
                    best_surrogate, best_seq = s, cand.tolist()
                    print(f"  New best (Dirichlet α={alpha}) surrogate={s:.6f}")

        # try alternating‑ends permutation of the current best for this n
        if best_seq:
            perm = alternate_ends_order(np.array(best_seq))
            s = compute_score(perm)
            if s < best_surrogate:
                best_surrogate, best_seq = s, perm.tolist()
                print(f"  New best (alternating ends) surrogate={s:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam smooth‑max refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0:
        print("\n--- Stage 2: Adam smooth‑max refinement ---")
        a_init = np.array(best_seq, dtype=float)
        a_refined = adam_optimize_smooth_max(
            a_init,
            deadline,
            max_iters=350_000,
            tau_start=0.5,
            tau_end=2000.0,
            base_lr=0.5,
        )
        s = compute_score(a_refined)
        print(f"  After Adam surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a_refined.tolist()
    else:
        print("\nSkipping Adam refinement (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 3 – sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        print("\n--- Stage 3: Sub‑gradient descent (exact) ---")
        a = np.array(best_seq, dtype=float)
        a = subgradient_descent_exact(
            a,
            deadline,
            max_iter=350_000,
            step0=0.07,
        )
        s = compute_score(a)
        print(f"  After sub‑gradient surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping sub‑gradient descent (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 4 – exponentiated‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        print("\n--- Stage 4: Exponentiated‑gradient polish ---")
        a = np.array(best_seq, dtype=float)
        a = eg_refine(
            a,
            deadline,
            max_iter=200_000,
            eta0=0.25,
        )
        s = compute_score(a)
        print(f"  After EG surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping EG polish (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 5 – permutation based hill climbing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Stage 5: Permutation hill climbing ---")
        a = np.array(best_seq, dtype=float)
        a = permutation_hill_climbing(a, deadline, max_trials=25_000, rng=rng)
        s = compute_score(a)
        print(f"  After permutation search surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping permutation hill climbing (insufficient time)")

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        final_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
