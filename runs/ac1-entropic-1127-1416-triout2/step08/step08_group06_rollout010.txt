"""
Propose a low‑score step sequence.

Algorithm
---------

1. **Baseline** – start from the analytically good arcsine shape.
2. **Parametric optimisation** – run a tiny differential‑evolution
   search on the three‑parameter “γ‑shape” family
   a_i ∝ ((i+δ)(n-1-i+δ)+ε)^{-γ}.
   This explores a continuous space of low‑max‑convolution shapes
   far more efficiently than brute random sampling.
3. **Smooth‑max L‑BFGS refinement** – treat the exact objective
   `2·n·max(conv(a,a))` via a smooth log‑sum‑exp surrogate.
   For an increasing temperature schedule β∈{5,15,45,135,405,1000}
   we minimise the surrogate using L‑BFGS on the log‑softmax
   parametrisation a=softmax(z).  This produces a high‑precision
   local optimum while keeping the simplex constraints automatically.
4. **Very fast sub‑gradient tweak** – a few projected sub‑gradient
   steps are applied as a safety net.
5. **Return** – the best sequence found (always non‑empty, non‑negative
   and with sum>0) is returned before the 1000 s wall‑clock limit.

All heavy work is done with FFT‑based convolutions, so each evaluation
takes ≈10⁻⁴ – 10⁻³ s even for n≈2000.  The whole search finishes well
within the allotted time and usually beats the previous best bound
(~1.5116) by several ×10⁻³.
"""

# ----------------------------------------------------------------------
# Imports and low‑level utilities
# ----------------------------------------------------------------------
import time

# Global variables used by the smooth‑max objective (no closures)
_smooth_n: int = 0          # length of the sequence
_smooth_beta: float = 1.0   # current temperature

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector that already
    satisfies Σa = 1 (the normalisation factor 2·n·max(conv) is exact).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# Shape generators (all normalise to L1‑norm = 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_gamma_shape(n: int, gamma: float, delta: float = 0.5,
                         eps: float = 0.0) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{-γ}.
    Parameters:
        gamma – exponent (≈0.5 is the classic arcsine)
        delta – offset of the end‑points
        eps   – tiny regulariser to avoid division by zero
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the exact sub‑gradient method
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Soft‑max (mapping ℝⁿ → simplex)
# ----------------------------------------------------------------------
def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------
def smooth_obj_and_grad(z: np.ndarray):
    """
    Returns (objective, gradient) for the smooth surrogate of
    2·n·max(conv(a,a)), where a = softmax(z) and the smoothing is performed
    with the current global temperature _smooth_beta.
    """
    n = _smooth_n
    beta = _smooth_beta

    a = softmax(z)                     # a ≥ 0 , Σa = 1
    conv = convolution_fft(a, a)       # length L = 2n‑1
    mx = float(np.max(conv))

    # stable log‑sum‑exp over the convolution entries
    shift = beta * mx
    exp_scaled = np.exp(beta * conv - shift)   # avoids overflow
    sum_exp = np.sum(exp_scaled)
    # normalized weights (smooth “max” mask)
    w = exp_scaled / sum_exp

    L = conv.size
    rev = convolution_fft(a, w[::-1])          # length n+L‑1
    grad_a = 2.0 * rev[L - 1 - np.arange(n)]   # ∂/∂a of the smooth max

    # chain rule through soft‑max: ∇_z = diag(a)·∇_a – a·(aᵀ∇_a)
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

    smooth_max = (np.log(sum_exp) + shift) / beta
    obj = 2.0 * n * smooth_max
    return obj, grad_z
