"""
Improved step‑function search.

The algorithm builds a non‑negative sequence `a` (sum = 1, scale‑invariant)
that minimises

        2·n·max(conv(a,a)) / (sum a)^2

where `conv` is the aperiodic convolution.
The pipeline is:

1. **Choose length** (larger vectors give a better theoretical bound,
   but we keep them modest to stay inside the time budget).

2. **Analytic α‑grid** – a 1‑parameter family
   a_i(α) ∝ [(i+½)(n‑½‑i)]^{‑α} is scanned; the best vector is kept.

3. **Exponentiated‑gradient smoothing** – minimise a smooth
   log‑sum‑exp approximation of the max‑convolution using a KL‑mirror
   step (mirror‑EG).  Temperature `β` is annealed; the step‑size decays
   as η₀/√t.  The best exact score seen is retained.

4. **Linear‑program refinement** – with the current vector `f`,
   we solve a small LP that **maximises the total mass** of a new vector
   `g` while keeping the (almost) worst convolution entries unchanged:
        maximise   Σ g_j
        subject to (f * g)_i ≤ M   for the `k` largest convolution positions,
                    g_j ≥ 0,
                    Σ g_j ≤ 2.
   The solution `g` is renormalised to a probability vector and mixed
   with `f` (50 % / 50 %).  A few such refinements dramatically lower the
   max‑convolution.

5. **Projected sub‑gradient descent** on the true objective
   `2·n·max(a*a)` (with Euclidean projection onto the simplex).

6. **Stochastic hill‑climb** – random swaps / mass transfers that are
   kept only if they strictly improve the exact score.

All phases respect the global 1000 s wall‑clock limit; the best sequence
found is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, vstack


# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalised to sum = 1.
    This matches the value returned by `evaluate_sequence` for a sum‑1 input.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytical start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # handle possible overflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Coarse scan over α and keep the best vector."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([
        np.linspace(-1.0, 0.0, 11),
        np.linspace(0.0, 2.5, 26)
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Random sparse initialisation
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    sparsity = np.random.uniform(0.03, 0.25)   # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# 3. Exponentiated‑gradient optimisation of the smooth max
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        beta_start: float = 0.5,
                        beta_end: float = 300.0,
                        eta0: float = 0.6,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) minimisation of the smoothed
    log‑sum‑exp surrogate of max(conv(a,a)).
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential annealing)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed max via log‑sum‑exp
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))    # stabilised
        w = expw / expw.sum()

        # gradient of the smooth surrogate
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # step size
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Linear‑program refinement (increase total mass without raising the max)
# ----------------------------------------------------------------------


def lp_refine(a0: np.ndarray,
              deadline: float,
              max_refine: int = 8,
              active_k: int = 10,
              ub_sum: float = 2.0,
              mix_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Given a probability vector a0 (sum = 1), perform up to `max_refine`
    linear‑program iterations.
    In each iteration we:
        * locate the `active_k` largest entries of conv(a,a);
        * maximise Σ g_j under the constraints (a * g)_i ≤ M for those
          indices and Σ g_j ≤ ub_sum;
        * mix the normalised solution g into a using `mix_factor`.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(max_refine):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        M = conv.max()

        # pick indices of the largest convolution values
        if active_k >= conv.size:
            idx = np.arange(conv.size)
        else:
            idx = np.argpartition(-conv, active_k)[:active_k]

        # Build sparse constraint matrix A_ub * g ≤ b_ub
        rows = []
        cols = []
        data = []
        for row_i, i in enumerate(idx):
            # (a * g)_i = Σ_j a_j * g_{i-j}, 0≤j<n, 0≤i-j<n
            j_start = max(0, i - (n - 1))
            j_end = min(i, n - 1)
            js = np.arange(j_start, j_end + 1, dtype=int)
            # coefficients are a[i - js]
            coeffs = a[i - js]
            rows.extend([row_i] * len(js))
            cols.extend(js.tolist())
            data.extend(coeffs.tolist())

        A_ub = csr_matrix((data, (rows, cols)), shape=(len(idx), n))
        # additional bound on total mass Σ g_j ≤ ub_sum
        A_ub = vstack([A_ub, csr_matrix(np.ones((1, n)))])
        b_ub = np.concatenate([np.full(len(idx), M), [ub_sum]])

        # maximise Σ g_j  → minimise −Σ g_j
        c = -np.ones(n)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # if LP fails we abort the refinement
            break

        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            break

        # normalise g to a probability vector
        g_norm = g / sum_g

        # mix and re‑normalise (both already sum to 1, so simple convex combo)
        a_candidate = (1.0 - mix_factor) * a + mix_factor * g_norm
        a_candidate /= a_candidate.sum()

        sc = compute_score(a_candidate)
        if sc < best_sc:
            best_sc = sc
            best_a = a_candidate.copy()
            a = a_candidate  # continue from the improved point
        else:
            # no improvement – shrink mixing for next iteration
            mix_factor *= 0.7
            # continue with the current a (do not replace)
            continue

    return best_a, best_sc


# ----------------------------------------------------------------------
# 5. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 600_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient method for 2·n·max(a*a)."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active quadratic term
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 6. Stochastic hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Random‑walk polishing: swaps or mass transfers.
    Only moves that strictly improve the exact score are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a, best_sc

        print(f"[{time.time():.1f}] LP refinement")
        a_lp, sc_lp = lp_refine(
            np.array(best_seq, dtype=float),
            deadline - 60.0,
            max_refine=8,
            active_k=12,
            ub_sum=2.0,
            mix_factor=0.5,
        )
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.tolist()
            print(f"[{time.time():.1f}] after LP refinement best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Sub‑gradient refinement
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 40.0,
            max_iters=600_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 6. Final hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a tiny uniform vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
