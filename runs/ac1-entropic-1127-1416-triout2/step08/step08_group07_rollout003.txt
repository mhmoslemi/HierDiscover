"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  

The search works in three stages:

1. **Parametric seed generation** –  we sample many Beta(α,β)‑shapes,
   power‑law shapes and a few handcrafted arcsine / uniform seeds.
   The surrogate objective `fast_score` (the exact formula with an FFT‑based
   convolution) is used to keep the most promising candidates.

2. **Local refinement** –  each promising seed is tuned by a cheap
   hill‑climbing on the (log‑)parameters (α,β) of the Beta family and by a
   random pair‑wise mass‑transfer search that directly optimises the surrogate.

3. **Exact polishing** –  the best surrogate candidate is finally refined
   with a mirror‑descent step that uses the true sub‑gradient of the
   max‑convolution (the exact objective).  At the very end we evaluate the
   final sequence with the official ``evaluate_sequence`` routine.

All work respects the overall time budget (default 1000 s).  The routine
always returns a valid non‑negative list; a uniform fallback is used if
nothing else is found.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic FFT convolution (linear, real)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate – scale invariant objective used for pre‑selection
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(conv(a,a))/sum(a)^2 without the clipping
    performed in ``evaluate_sequence``.  ``a`` must be non‑negative.
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float = 1e-12
) -> np.ndarray:
    """Beta(α,β) shape on {0,…,n‑1} (normalised to sum 1)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    vals = left * right
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        # fallback – uniform
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_power_shape(
    n: int, p: float, eps: float = 1e-12
) -> np.ndarray:
    """Shape proportional to 1/((i+½)*(n‑½‑i)+eps)^p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Random local search on the surrogate objective
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.03,
) -> np.ndarray:
    """
    Simple pairwise mass‑transfer moves while time permits.
    Moves are kept only if they improve ``fast_score``.
    """
    a = a0.copy()
    best = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a fraction of the mass at i to j
        eps = a[i] * rng.random() * max_transfer
        if eps <= 0.0:
            continue
        a[i] -= eps
        a[j] += eps
        # renormalise (numerical safety)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
            continue
        a /= s
        sc = fast_score(a)
        if sc < best:
            best = sc
        else:
            # revert the move
            a[j] -= eps
            a[i] += eps
            a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Mirror‑descent with the exact max‑convolution sub‑gradient
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.2,
    decay: float = 0.9995,
) -> np.ndarray:
    """
    Exponential‑gradient (mirror) descent on the true objective.
    The iterate stays on the probability simplex automatically.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # multiplicative (KL) mirror step
        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay
        # occasional progress output
        if int(step * 1e6) % 200_000 == 0:
            print(
                f"    Mirror step={step:.6f} surrogate≈{fast_score(a):.6f}"
            )
    return a


# ----------------------------------------------------------------------
# Hill‑climbing on Beta‑distribution parameters
# ----------------------------------------------------------------------


def hill_climb_beta(
    n: int,
    alpha0: float,
    beta0: float,
    rng: np.random.Generator,
    deadline: float,
) -> Tuple[np.ndarray, float, Tuple[float, float]]:
    """
    Simple hill‑climbing on the log‑parameters (α,β) of a Beta shape.
    Returns the best vector, its surrogate score and the parameters.
    """
    alpha, beta = alpha0, beta0
    a = generate_beta_shape(n, alpha, beta, eps=1e-12)
    best_score = fast_score(a)
    best_vec = a.copy()
    best_params = (alpha, beta)

    # log‑space step size – it will shrink slowly
    log_step = 0.3

    while time.time() < deadline:
        # propose new (log) parameters
        log_alpha = math.log(alpha)
        log_beta = math.log(beta)
        log_alpha_new = log_alpha + rng.normal(scale=log_step)
        log_beta_new = log_beta + rng.normal(scale=log_step)
        alpha_new = math.exp(log_alpha_new)
        beta_new = math.exp(log_beta_new)

        # keep parameters inside a reasonable domain
        if not (0.07 <= alpha_new <= 12.0 and 0.07 <= beta_new <= 12.0):
            continue

        a_new = generate_beta_shape(n, alpha_new, beta_new, eps=1e-12)
        sc = fast_score(a_new)

        if sc < best_score:
            best_score = sc
            best_vec = a_new.copy()
            alpha, beta = alpha_new, beta_new
            best_params = (alpha, beta)
            # make the search a bit more aggressive after a success
            log_step *= 0.98
        else:
            # slowly contract the neighbourhood
            log_step *= 0.995

    return best_vec, best_score, best_params


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42, budget_s: int = 1000, **kwargs
) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence``.  The routine uses a parametric seed search,
    hill‑climbing on Beta‑family parameters, a cheap random local search,
    and a final mirror‑descent polishing step.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.25  # small safety margin

    best_seq: List[float] = []
    best_official: float = float("inf")
    best_surrogate: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, cand
                best_surrogate = fast_score(np.array(cand))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Choose a target length – larger lengths give a tighter theoretical bound
    # ------------------------------------------------------------------
    # We start with the biggest length we can afford; if time runs out we fall
    # back to smaller lengths.
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} (time left {deadline - time.time():.1f}s) ===")

        # --------------------------------------------------------------
        # 1. Build a diverse pool of parametric seeds
        # --------------------------------------------------------------
        seeds: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            generate_mixture_shape(n, weight_arc=0.6),
        ]

        # systematic Beta(α,β) sweep (symmetric)
        for a in np.linspace(0.3, 3.0, 9):
            seeds.append(generate_beta_shape(n, a, a, eps=1e-12))

        # random asymmetric Beta draws
        for _ in range(15):
            alpha = 10 ** rng.uniform(-1.0, math.log10(6.0))
            beta = 10 ** rng.uniform(-1.0, math.log10(6.0))
            seeds.append(generate_beta_shape(n, alpha, beta, eps=1e-12))

        # power‑law family
        for p in np.linspace(0.2, 2.0, 9):
            seeds.append(generate_power_shape(n, p, eps=1e-12))

        # --------------------------------------------------------------
        # 2. Keep the most promising seeds according to the fast surrogate
        # --------------------------------------------------------------
        scored: List[Tuple[float, np.ndarray]] = [(fast_score(v), v) for v in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:6]]   # keep six per length

        # --------------------------------------------------------------
        # 3. Hill‑climb the best Beta seeds and do cheap local search
        # --------------------------------------------------------------
        # Allocate a slice of the remaining budget for this length
        remaining_time = deadline - time.time()
        per_seed_budget = max(1.0, remaining_time / (len(top_seeds) * 2.0))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 2.0:
                break
            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_budget:.1f}s")
            seed_deadline = time.time() + per_seed_budget

            # ---- If the seed originates from a Beta shape, try hill‑climbing ----
            # Heuristic: recognise by checking whether the seed has the exact
            # functional form of a Beta distribution (rough test: the central region
            # should be smooth).  For simplicity we just attempt hill‑climbing on
            # every seed – the generate function will simply return a shape close
            # to the original if the parameters are far from optimum.
            # Estimate initial parameters from the seed (very crude):
            # we start the climb from α=β=0.5 (arcsine) for non‑Beta seeds.
            init_alpha = 0.5
            init_beta = 0.5
            # try a few random initialisations as well
            attempts = 2
            best_local_vec = seed_vec
            best_local_score = fast_score(seed_vec)

            for att in range(attempts):
                # pick a random starting point around the seed's mass centre
                a0 = 10 ** rng.uniform(-1.0, 0.5)
                b0 = 10 ** rng.uniform(-1.0, 0.5)
                vec, sc, _ = hill_climb_beta(
                    n,
                    a0,
                    b0,
                    rng,
                    seed_deadline - 0.1,
                )
                if sc < best_local_score:
                    best_local_score = sc
                    best_local_vec = vec

            # ---- Random local search on the best vector from hill‑climb ----
            if time.time() < deadline - 1.0:
                ls_deadline = time.time() + (deadline - time.time()) * 0.3
                refined = random_local_search(best_local_vec, ls_deadline, rng, max_transfer=0.025)
            else:
                refined = best_local_vec

            surrogate = fast_score(refined)

            # ---- Official evaluation only if surrogate improves significantly ----
            if surrogate < best_surrogate * 0.9995:
                try:
                    val = evaluate_sequence(refined.tolist())
                    print(
                        f"   Official score after refinement = {val:.6f} (surrogate {surrogate:.6f})"
                    )
                    if val < best_official:
                        best_official, best_seq = val, refined.tolist()
                        best_surrogate = surrogate
                        print("   *** New global best! ***")
                except Exception as e:
                    print(f"   Official evaluation failed: {e}")

        # --------------------------------------------------------------
        # 4. Mirror‑descent polishing on the current global best (if any)
        # --------------------------------------------------------------
        if best_seq and time.time() < deadline - 2.0:
            print("\nPolishing global best with mirror‑descent …")
            a_arr = np.array(best_seq, dtype=float)
            md_deadline = deadline - 0.4
            a_polished = mirror_descent_optimize(
                a_arr, md_deadline, step0=0.18, decay=0.9996
            )
            try:
                val_md = evaluate_sequence(a_polished.tolist())
                print(f"   Mirror‑descent official score = {val_md:.6f}")
                if val_md < best_official:
                    best_official, best_seq = val_md, a_polished.tolist()
                    best_surrogate = fast_score(a_polished)
                    print("*** Global best improved by mirror‑descent ***")
            except Exception as e:
                print(f"   Mirror‑descent evaluation error: {e}")

    # ------------------------------------------------------------------
    # 5. Final safety fallback (uniform sequence) if nothing succeeded
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
