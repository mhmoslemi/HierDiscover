"""
search_algorithm.py
-------------------

A time‑bounded (≤ 1000 s) optimizer for the official `evaluate_sequence`
objective.

Strategy
--------

1. **Exploit the known near‑optimal “arcsine” family** – for a unit‑sum
   vector `a` the quantity `2·n·max(conv(a,a))` is approximately
   `2·log(n)/π²`.  For moderate lengths (`n≈1024`) this already yields a
   score ≈ 1.40, well below the target 1.503.

2. **Systematic length sweep** – lengths are examined from small to
   large (512 → 4096).  The best candidate across all lengths is kept.

3. **Local refinement** – each candidate is polished three times,
   always staying on the probability simplex:
   * KL‑mirror (exponentiated‑gradient) descent,
   * Adam on a smooth log‑sum‑exp surrogate of the max‑convolution,
   * Euclidean projected sub‑gradient descent.

   The number of iterations for each phase is adaptively limited by the
   remaining wall‑clock time.

4. **Fallback** – if no candidate is produced (unlikely) a uniform
   distribution of length 512 is returned.

All helper routines are top‑level (no closures) and use only `numpy`,
`time` and the imported `evaluate_sequence`.  The function `propose_candidate`
conforms to the required signature and always returns a non‑empty list of
non‑negative floats before the budget expires.
"""

# ----------------------------------------------------------------------
# Imports and low‑level utilities
# ----------------------------------------------------------------------
import time
from typing import List
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score_normed(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose entries sum to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the exact sub‑gradient method
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators (all normalised to sum = 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int, eps: float = 0.0) -> np.ndarray:
    """Discrete arcsine‑like envelope: a_i ∝ 1/√((i+½)(n‑½‑i)+eps)."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_gamma_shape(
    n: int, gamma: float, delta: float = 0.5, eps: float = 0.0
) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{-γ}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step: float = 0.01,
) -> np.ndarray:
    """
    Mirror‑descent on the exact max‑convolution objective using KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # gradient of max(conv) w.r.t. a (sub‑gradient when several entries tie)
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 10_000 == 0:
            print(f"    EG iter={it:,d} score={compute_score_normed(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser
# ----------------------------------------------------------------------
def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max  → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 400.0,
    ramp_iters: int = 80_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam on the log‑sum‑exp (smooth max) of the convolution.
    Operates in the log‑domain (softmax guarantees simplex feasibility).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ∈ simplex
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # temperature schedule (larger β → sharper approximation of max)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of smooth max wrt a
        rev = convolution_fft(a, w_norm[::-1])
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 10_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score_normed(a):.6f}")

    return softmax(z)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected Euclidean sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 10_000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score_normed(a):.6f}")

    return a
