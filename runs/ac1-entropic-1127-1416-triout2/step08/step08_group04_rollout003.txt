"""
Improved search for a step‑function with a low
evaluate_sequence() score.

The key idea is that the optimal continuous density that
minimises  ∥f∗f∥∞ under the unit‑mass constraint is the
arcsine law (Beta(½,½)).  Discretising this law with a larger
number of points drives the discrete optimum ever closer to the
theoretical lower bound 2·¾ = 1.5.  Therefore we:

* Scan a few powers‑of‑two lengths (4096 … 16384) and a fine grid
  of power‑law shapes a_i ∝ ((i+½)(n‑½‑i))⁻ᵖ (p≈0.5 is the
  arcsine case).
* Pick the best shape according to a fast FFT‑based surrogate
  2·n·max(conv(a,a)) (exact when Σa=1).
* If time permits, perform a short Euclidean projected sub‑gradient
  descent on the true max‑convolution to shave off the remaining
  few‑percent of the error.
* Finally evaluate the chosen sequence with the official
  evaluate_sequence() function and return it.

The routine respects the global 1000 s wall‑clock budget and
always returns a non‑empty, non‑negative, unit‑sum list.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (exact up to floating‑point rounding).
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is exactly 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_shape(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """
    Generalised arcsine shape:
        a_i ∝ ((i+½)(n‑½‑i) + ε)^{-p}   for i = 0,…,n‑1
    Returned vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def subgradient_descent_limited(initial_a: np.ndarray,
                                deadline: float,
                                max_iter: int = 100_000,
                                step0: float = 0.04) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    Stops early when the wall‑clock deadline is reached.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # indices where the current maximum is attained (within tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(a, rev(mask))
        rev_mask = mask[::-1]
        conv_rev = convolution_fft(a, rev_mask)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a

        print("\n--- Stage 2: sub‑gradient descent refinement ---")
        a_refined = subgradient_descent_limited(
            best_seq,
            deadline,
            max_iter=200_000,
            step0=0.04,
        )
        s = compute_score(a_refined)
        print(f"  Surrogate after sub‑gradient = {s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a_refined
    else:
        print("\nSkipping sub‑gradient refinement (insufficient time)")

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\nOfficial final score = {final_score:.6f}")
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_score = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n)
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Falling back to uniform length {fallback_n}, score = {final_score:.6f}")

    print(f"\nSearch finished. Best official score = {final_score:.6f}")
    return best_seq.tolist()
