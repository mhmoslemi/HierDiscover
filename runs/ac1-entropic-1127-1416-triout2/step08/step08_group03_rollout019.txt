def piecewise_constant_family(n, edge_frac, ratio):
    # edge_frac in [0, 0.5], ratio >=1
    k = int(round(edge_frac * n))
    if k == 0:
        # uniform
        a = np.full(n, 1.0/n)
        return a
    # assign weights
    # interior weight d = 1 / (n - 2k + 2k*ratio)
    denom = n - 2*k + 2*k*ratio
    d = 1.0 / denom
    c = ratio * d
    a = np.full(n, d)
    a[:k] = c
    a[-k:] = c
    return a

"""
Optimiser for the non‑negative step‑function problem.

We combine several ideas:
 * a dense grid over analytically‑motivated families (symmetric edge‑heavy,
   left‑heavy power‑law, and a short‑piecewise‑constant “edge‑plateau” family);
 * random monotone starts;
 * a few fast Adam‑optimisation runs on a smooth approximation of the max‑convolution;
 * a long projected sub‑gradient descent;
 * up‑sampling the best vector to a larger length (if time permits) and a final
   simulated‑annealing hill‑climb.
All stages respect a global wall‑clock limit (default 1000 s) and the result
is returned as a plain Python list of floats ready for `evaluate_sequence`.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simple simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against infinities for extreme α
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def piecewise_constant_family(n: int, edge_frac: float, ratio: float) -> np.ndarray:
    """
    Two‑level symmetric shape:
        * a proportion `edge_frac` of the indices (on each side) have weight `c`.
        * the remaining interior points have weight `d`.
      The ratio `ratio = c / d` controls how much heavier the edges are.
      The vector is projected onto the simplex (numerically exact).
    """
    if edge_frac <= 0.0:
        return np.full(n, 1.0 / n)
    if edge_frac >= 0.5:
        edge_frac = 0.5

    k = int(round(edge_frac * n))
    k = max(1, min(k, n // 2 - 1))

    # interior weight d, edge weight c = ratio * d
    denom = (n - 2 * k) + 2 * k * ratio
    d = 1.0 / denom
    c = ratio * d

    a = np.full(n, d, dtype=np.float64)
    a[:k] = c
    a[-k:] = c
    # projection only for safety against rounding errors
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)        # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 5.0,
                         beta_end: float = 150.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially from beta_start to beta_end.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # smoothed max via log‑sum‑exp
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a (see derivation in previous version)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (very cheap per iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method for the true max‑objective
          2·n·max(a*a)
    using a diminishing stepsize  ∝ 1/√t and Euclidean projection onto the simplex.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index giving the active max

        # sub‑gradient of the active max entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb on the simplex
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    Moves:
      – swap two coordinates,
      – or transfer a fraction of mass from one index to another.
    Temperature decays linearly; the step‑scale adapts slowly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4          # tiny temperature
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver – respects the global time budget
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # safety margin

    # ------------------------------------------------------------------
    # Length of the vector – start with a moderate size.
    # ------------------------------------------------------------------
    n = 8192
    print(f"[{time.time():.1f}] Using base length n = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to any previously‑found construction.
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families – dense grid search.
    # ------------------------------------------------------------------
    # a) symmetric product family (edge‑heavy, symmetric)
    alphas = np.linspace(0.0, 4.0, 81)
    for a in alphas:
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, a)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={a:.2f}) = {best_score:.6f}")

    # b) left‑heavy power‑law family (monotone)
    betas = np.linspace(0.0, 4.0, 81)
    for b in betas:
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, b)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={b:.2f}) = {best_score:.6f}")

    # c) two‑level piecewise‑constant family (edge plateau)
    edge_fracs = np.linspace(0.0, 0.45, 46)    # 0 % … 45 % of the indices each side
    ratios = np.geomspace(1.0, 120.0, 30)     # edge‑to‑interior weight ratio
    for ef in edge_fracs:
        for r in ratios:
            if time.time() >= deadline - 30.0:
                break
            vec = piecewise_constant_family(n, ef, r)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (edge={ef:.3f}, r={r:.1f}) = {best_score:.6f}")

    # d) random monotone draws (a few dozen)
    for _ in range(60):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam‑based smoothed optimisation from random sparse starts.
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:
        start_vec = random_sparse_start(n)
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 200.0,
            max_iters=130_000,
            beta_start=6.0,
            beta_end=180.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best (Adam‑smooth) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Long projected sub‑gradient refinement (very cheap per iteration)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting long sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 60.0,
            max_iters=3_000_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Upsample to a larger length (if time permits) and refine again.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        n_big = 16384
        print(f"[{time.time():.1f}] up‑sampling to n = {n_big}")
        old_a = np.array(best_seq, dtype=np.float64)
        # linear interpolation to the larger grid
        x_old = np.linspace(0, 1, old_a.size)
        x_new = np.linspace(0, 1, n_big)
        a_big = np.interp(x_new, x_old, old_a)
        a_big = project_to_simplex(a_big)

        # a short Adam run on the larger grid
        a_opt_big, sc_opt_big = adam_smooth_optimize(
            a_big,
            deadline - 30.0,
            max_iters=100_000,
            beta_start=5.0,
            beta_end=200.0,
            lr=0.015,
        )
        if sc_opt_big < best_score:
            best_score = sc_opt_big
            best_seq = a_opt_big.tolist()
            n = n_big
            print(f"[{time.time():.1f}] improved after up‑sample Adam = {best_score:.6f}")

        # a quick sub‑gradient polish on the bigger vector
        if time.time() < deadline - 20.0:
            a_sub_big, sc_sub_big = subgrad_optimize(
                np.array(best_seq, dtype=np.float64),
                deadline - 10.0,
                max_iters=1_000_000,
                step0=0.05,
            )
            if sc_sub_big < best_score:
                best_score = sc_sub_big
                best_seq = a_sub_big.tolist()
                print(f"[{time.time():.1f}] best after big sub‑grad = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation test (many optima are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if nothing succeeded, fall back to a modest uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call `evaluate_sequence`).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
