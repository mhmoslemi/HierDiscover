"""
Propose a step‑function with minimal evaluate_sequence score.

Algorithm (≈ 1000 s budget)

1. **Parameter sweep** – generate many simple families (exponential,
   power‑law, inverse‑power, linear) on a chosen length *n*
   (≈2048 or 4096).  All vectors are normalised to sum = 1 and the exact
   score (2·n·max(conv)) is computed via a fast FFT‑convolution.
   The best vector from this low‑cost sweep is kept.

2. **Smooth max optimisation** – initialise Adam‑accelerated gradient
   descent from the best sweep result (and a few random restarts) on the
   log‑sum‑exp surrogate of the max‑convolution.  The surrogate is
   annealed (β from 5 to 2000) and the iterates are projected onto the
   probability simplex after each update.

3. **Projected sub‑gradient descent** – a classic sub‑gradient method on
   the true max‑objective refines the solution further.

4. **Stochastic hill‑climb** – simple mass‑transfer / swap moves accepted
   only when they improve the exact score; the step‑size adapts
   automatically.

All phases respect the wall‑clock deadline; the best vector found is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for n up to a few 10⁴)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a) == 1):
    returns 2·n·max(conv(a, a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators (fixed families for the cheap sweep)
# ----------------------------------------------------------------------


def gen_exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i, normalised to sum = 1."""
    a = r ** np.arange(n, dtype=float)
    return a / a.sum()


def gen_power_decay(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (1 - i/(n‑1))**p  (monotonically decreasing, p≥0).
    p==0 gives the uniform vector.
    """
    if p == 0.0:
        return np.full(n, 1.0 / n)
    x = 1.0 - np.arange(n, dtype=float) / (n - 1)
    a = np.power(x, p)
    return a / a.sum()


def gen_inverse_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)**(-p), normalised."""
    if p == 0.0:
        return np.full(n, 1.0 / n)
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=float), p)
    return a / a.sum()


def gen_linear(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# 1. Smooth max optimisation (log‑sum‑exp surrogate) with Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 400_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑accelerated minimisation of the soft‑max surrogate.
    Returns the best exact score (using compute_score) found.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # Anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # Convolution a*a
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # Log‑sum‑exp weights (stable)
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # Gradient of the smooth max (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)                # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]             # central n entries

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)                          # slowly decreasing step
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a = project_to_simplex(a - update)

        # Periodically evaluate the *exact* score
        if it % 500 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    # final check
    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of the maximal entry:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        cur = compute_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    # final check
    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    start_a: np.ndarray,
    time_deadline: float,
    max_steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simple random local moves (swap or small mass transfer) accepted only
    when they improve the exact score.  Step size adapts.
    """
    a = start_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.07  # initial fraction of mass to move

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # small transfer from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur = compute_score(cand)

        if cur < best_score:
            best_score = cur
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns a plain Python list of floats (the sequence).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered sequence (if present)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_score:.6f}")
        if fallback_score < best_score:
            best_score = fallback_score
            best_a = np.array(fallback, dtype=float)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a reasonable length – larger n gives finer granularity
    # ------------------------------------------------------------------
    # Rough heuristic: use larger n if we still have plenty of time.
    n = 2048
    if deadline - time.time() > 800:
        n = 4096
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 0. Cheap parametric sweep (exponential, power, inverse‑power)
    # ------------------------------------------------------------------
    print("=== Parameter sweep ===")
    candidates = []

    # exponential family
    for r in np.linspace(0.92, 0.9999, 30):
        candidates.append(gen_exponential(n, r))

    # power‑law decreasing ( (1‑i/(n‑1))^p )
    for p in np.linspace(0.0, 5.0, 31):
        candidates.append(gen_power_decay(n, p))

    # inverse‑power ( (i+1)^‑p )
    for p in np.linspace(0.0, 5.0, 31):
        candidates.append(gen_inverse_power(n, p))

    # uniform (as a reference)
    candidates.append(gen_linear(n))

    # evaluate all candidates
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        scr = compute_score(cand)
        if scr < best_score:
            best_score = scr
            best_a = cand.copy()
            print(f"Sweep improvement #{idx}: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper – generate a random start on the simplex (dense or sparse)
    # ------------------------------------------------------------------
    def random_start():
        """Return a random probability vector of length n."""
        if np.random.rand() < 0.5:
            a = np.random.rand(n)
            a /= a.sum()
        else:
            # sparse start: put mass on sqrt(n) random positions
            k = max(2, int(np.sqrt(n)))
            idx = np.random.choice(n, size=k, replace=False)
            a = np.zeros(n, dtype=float)
            a[idx] = 1.0 / k
            a = project_to_simplex(a)
        return a

    # ------------------------------------------------------------------
    # 1. Random‑restart smooth optimisation (Adam)
    # ------------------------------------------------------------------
    reserve_for_refine = 150.0  # seconds to keep for later phases
    while time.time() < deadline - reserve_for_refine:
        remaining = deadline - time.time() - reserve_for_refine
        slice_time = time.time() + max(5.0, min(30.0, remaining * 0.4))

        # 30 % chance to start from the current best, otherwise random
        if best_a is not None and np.random.rand() < 0.3:
            start_vec = best_a.copy()
        else:
            start_vec = random_start()

        a_opt, scr_opt = smooth_gradient_optimize(
            start_vec,
            time_deadline=slice_time,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=2000.0,
            step0=1.0,
        )
        exact = compute_score(a_opt)
        if exact < best_score:
            best_score = exact
            best_a = a_opt.copy()
            print(f"Smooth phase new best: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Sub‑gradient refinement (if time permits)
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 30.0:
        print("\n=== Sub‑gradient refinement ===")
        refined_a, refined_score = subgrad_optimize(
            best_a,
            time_deadline=deadline - 10.0,
            max_iter=400_000,
            step0=0.07,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_a = refined_a.copy()
            print(f"Sub‑grad improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final hill‑climb polishing
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 1.0:
        print("\n=== Hill‑climb polishing ===")
        polished_a, polished_score = hill_climb(
            best_a,
            time_deadline=deadline,
            max_steps=2_000_000,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_a = polished_a.copy()
            print(f"Hill‑climb improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a simple uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_a is None:
        n_small = 256
        best_a = np.full(n_small, 1.0 / n_small)
        best_score = compute_score(best_a)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will call `evaluate_sequence`
    # again, but we print the score here for debugging).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_a.tolist()
