def propose_candidate(seed=0, budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5
    best_seq = None
    best_surrogate = math.inf
    ...

    # Stage 0: baseline fallback
    if 'height_sequence_1' in globals():
        try:
            fallback = [float(x) for x in globals()['height_sequence_1']]
            val = evaluate_sequence(fallback)
            best_seq = fallback
            best_surrogate = compute_score(np.array(fallback))
            best_official = val
        except Exception:
            pass

    # Stage 1: grid search
    lengths = [4096, 8192, 16384, 32768, 65536]
    p_vals = [0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65]
    for n in lengths:
        # arcsine
        a = arcsine_initial(n)
        s = compute_score(a)
        if s < best_surrogate: best update
        # power shapes
        for p in p_vals:
            a = generate_power_shape(n, p)
            s = compute_score(a)
            if s < best_surrogate: best update
        # mixture (maybe w=0.25,0.5,0.75)
        arc = arcsine_initial(n)
        uni = np.full(n,1.0/n)
        for w in [0.0,0.25,0.5,0.75,1.0]: # actually w=0 uniform, w=1 arcsine but already tested, maybe others.
            mix = w * arc + (1-w) * uni
            mix = mix / mix.sum()
            s = compute_score(mix)
            if s < best_surrogate: update
    
    # Stage2: Adam
    remaining = deadline - time.time()
    if remaining > 20:
        a_init = np.array(best_seq)
        a_refined = adam_optimize_smooth_max(a_init, deadline, max_iters=200_000, tau_start=0.5, tau_end=400.0, base_lr=0.6)
        s = compute_score(a_refined)
        if s < best_surrogate: update

    # Stage3: subgradient descent
    if deadline - time.time() > 10:
        a = np.array(best_seq)
        a = subgradient_descent_exact(a, deadline, max_iter=150_000, step0=0.07)
        s = compute_score(a)
        if s < best_surrogate: update

    # Stage4: EG
    if deadline - time.time() > 8:
        a = np.array(best_seq)
        a = eg_refine(a, deadline, max_iter=100_000, eta0=0.2)
        s = compute_score(a)
        if s < best_surrogate: update

    # Stage5: local random search
    if deadline - time.time() > 5:
        a = np.array(best_seq)
        a = local_random_search(a, deadline, max_trials=50000, step_scale=0.0005, rng=np.random.default_rng(seed+12345))
        s = compute_score(a)
        if s < best_surrogate: update

    # Finally official evaluation
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"Official final score = {final_val:.6f}")
    except Exception:
        final_val = float('inf')
        print("Official evaluation error")

    # Ensure non-empty
    if not best_seq:
        n = 4096
        best_seq = [1.0/n]*n
        final_val = evaluate_sequence(best_seq)

    return best_seq

def convolution_matrix(f):

def lp_optimize_g(f, M):
    n = f.size
    L = 2*n - 1
    # build convolution matrix of f
    # Using scipy.sparse
    from scipy.sparse import csr_matrix, diags
    # Build toeplitz using sparse: column c = [f; zeros], row r = [f[0], zeros]
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.concatenate([f[:1], np.zeros(n-1)])
    A = csr_matrix(toeplitz(col, row))   # shape L x n
    # inequality: A @ b <= M
    # Upper bound vector
    b_ub = np.full(L, M)
    # objective: maximize sum(b) i.e., minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Use scipy.optimize.linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError('LP failed')
    return res.x
```

"""
Improved step‑function search for the “max‑self‑convolution” problem.

Algorithm (all stages respect a global 1000 s wall‑clock):
1.  Coarse grid search over several lengths (1024…4096) and several
    simple analytic families (uniform, arcsine, power‑shape, mixtures).
    The fast FFT surrogate   s = 2·n·max(conv(a,a))   is evaluated.
2.  Linear‑programming refinement (inspired by the original iterative LP
    scheme).  For a current distribution f (∑f=1) we solve

           max   Σ b_i        (equivalently min −Σb_i)
           s.t.  conv(f,b) ≤ M ,  b ≥ 0 ,

    where M = max(conv(f,f)).  The solution b is normalised to a new
    distribution g (∑g=1) and mixed with f:
           f ← (1−τ)·f + τ·g   (τ = 0.2).

    A few such LP updates already lower the maximal convolution.
3.  Smooth‑max Adam optimisation (log‑softmax parametrisation) to
    polish the solution.
4.  Exact sub‑gradient descent on the true max‑convolution.
5.  Exponentiated‑gradient (KL‑mirror) polishing.
6.  Very cheap stochastic local hill‑climbing.
7.  Final official evaluation and return of the best sequence.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (fast, uses next power‑of‑2 length).
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    Assumes a lies on the simplex (Σa = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def arcsine_initial(n: int, eps: float = 1e-12) -> np.ndarray:
    """Discrete arcsine‑like density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """
    Generalised arcsine shape:
        a_i ∝ ((i+½)*(n‑½‑i) + ε)^{−p}
    """
    i = np.arange(n, dtype=float)
    vals = 1.0 / ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """
    Mixture: weight_arc·arcsine + (1‑weight_arc)·uniform.
    """
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

def build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Return the (2n‑1) × n sparse matrix A such that
        A @ b == conv(f, b)   (full linear convolution).
    Each diagonal k (0 ≤ k < n) is constant f[k].
    """
    n = f.size
    L = 2 * n - 1
    # data for diagonal k is a constant array f[k] of length n
    data = [np.full(n, f[k]) for k in range(n)]
    offsets = np.arange(n)               # row - col = k
    A = sparse.diags(data, offsets, shape=(L, n), format='csr')
    return A


def lp_refine(f: np.ndarray, M: float, t: float = 0.2) -> np.ndarray:
    """
    Perform one LP‑based refinement step.
    * f  – current distribution (Σf = 1).
    * M  – current maximal self‑convolution (max(conv(f,f))).
    * t  – mixing parameter (0 < t ≤ 1).

    Returns the updated distribution f_new.
    """
    n = f.size
    A = build_conv_matrix(f)                     # (2n‑1)×n matrix
    b_ub = np.full(2 * n - 1, M)                # conv(f,b) ≤ M
    c = -np.ones(n)                             # maximise Σb_i  → minimise −Σb_i
    bounds = [(0.0, None)] * n

    # Solve LP with HiGHS (fast interior‑point algorithm)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # If LP fails, fall back to the unchanged distribution
        return f.copy()

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return f.copy()
    g = b / sum_b            # normalise to the simplex

    # Simple convex combination (the mixing parameter t can be tuned)
    f_new = (1.0 - t) * f + t * g
    return f_new

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """Adam optimisation of the smooth‑max surrogate."""
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)                     # logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        conv = convolution_fft(a, a)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()   # soft max weights

        # grad_a = 2 * (a * (weights convolved with reversed a))
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)      # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 50_000 == 0:
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={compute_score(a):.6f}")

    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.07) -> np.ndarray:
    """Euclidean projected sub‑gradient descent on the true max‑convolution."""
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.25) -> np.ndarray:
    """Exponentiated‑gradient polish (mirror descent w.r.t. KL)."""
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 20_000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """Very cheap stochastic hill climbing (only accepting improvements)."""
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.2 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1

    return a_cur

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_surrogate = float("inf")
    best_official = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = fallback
                best_surrogate = compute_score(np.array(fallback))
        except Exception as e:
            print(f"Fallback evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Stage 1 – coarse multi‑scale shape search
    # ------------------------------------------------------------------
    candidate_ns = [1024, 2048, 4096]   # longer vectors become expensive for LP
    p_grid = np.linspace(0.15, 0.85, 15)      # exponent for power shape
    w_grid = np.linspace(0.0, 1.0, 11)       # mixture weight

    print("\n--- Stage 1: coarse grid search ---")
    for n in candidate_ns:
        print(f"Testing length n={n}")

        # Uniform
        uni = np.full(n, 1.0 / n)
        s = compute_score(uni)
        if s < best_surrogate:
            best_surrogate, best_seq = s, uni.tolist()
            print(f"  New best (uniform) surrogate={s:.6f}")

        # Arcsine
        arc = arcsine_initial(n)
        s = compute_score(arc)
        if s < best_surrogate:
            best_surrogate, best_seq = s, arc.tolist()
            print(f"  New best (arcsine) surrogate={s:.6f}")

        # Power shapes
        for p in p_grid:
            shape = generate_power_shape(n, p, eps=1e-12)
            s = compute_score(shape)
            if s < best_surrogate:
                best_surrogate, best_seq = s, shape.tolist()
                print(f"  New best (p={p:.3f}) surrogate={s:.6f}")

        # Mixtures arcsine / uniform
        for w in w_grid:
            mix = w * arc + (1.0 - w) * uni
            mix = mix / mix.sum()
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"  New best (mix w={w:.2f}) surrogate={s:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – LP‑based refinement (only for moderate lengths)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and len(best_seq) <= 2048:
        print("\n--- Stage 2: linear‑programming refinement ---")
        a = np.array(best_seq, dtype=float)
        # Run a few LP updates (t fixed to 0.2).  Each LP is cheap for n≤2048.
        for it in range(5):
            M = float(np.max(convolution_fft(a, a)))
            a_new = lp_refine(a, M, t=0.2)
            new_surrogate = compute_score(a_new)
            print(f"  LP iteration {it+1}, surrogate={new_surrogate:.6f}")
            if new_surrogate < best_surrogate:
                best_surrogate, best_seq = new_surrogate, a_new.tolist()
            a = a_new
            if deadline - time.time() < 5.0:
                break
    else:
        print("\n--- Skipping LP refinement (insufficient time or large n) ---")

    # ------------------------------------------------------------------
    # Stage 3 – smooth‑max Adam refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print("\n--- Stage 3: Adam refinement (smooth‑max) ---")
        a_init = np.array(best_seq, dtype=float)
        a_refined = adam_optimize_smooth_max(
            a_init,
            deadline,
            max_iters=300_000,
            tau_start=0.5,
            tau_end=400.0,
            base_lr=0.5,
        )
        s = compute_score(a_refined)
        print(f"  After Adam surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a_refined.tolist()
    else:
        print("\nSkipping Adam refinement (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 4 – exact sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        print("\n--- Stage 4: sub‑gradient descent ---")
        a = np.array(best_seq, dtype=float)
        a = subgradient_descent_exact(
            a,
            deadline,
            max_iter=250_000,
            step0=0.07,
        )
        s = compute_score(a)
        print(f"  After sub‑gradient surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping sub‑gradient descent (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 5 – exponentiated‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Stage 5: exponentiated‑gradient polish ---")
        a = np.array(best_seq, dtype=float)
        a = eg_refine(
            a,
            deadline,
            max_iter=150_000,
            eta0=0.25,
        )
        s = compute_score(a)
        print(f"  After EG surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping EG polish (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 6 – cheap stochastic local search
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Stage 6: local random search ---")
        a = np.array(best_seq, dtype=float)
        a = local_random_search(a, deadline,
                                max_trials=40000,
                                step_scale=0.001,
                                rng=rng)
        s = compute_score(a)
        print(f"  After local search surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping local random search (insufficient time)")

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        final_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
