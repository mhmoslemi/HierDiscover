"""
Hybrid search for a low‑value step‑function.

1. Build a diverse pool of analytic seeds (arcsine, beta‑,
   power‑law, mixture, Dirichlet, up‑sampled current best).
2. Rank the seeds with a fast FFT‑based surrogate
   `compute_score = 2·n·max(conv(a,a))` (valid when sum(a)=1).
3. Run a short Adam optimiser on the smooth log‑sum‑exp
   surrogate of the max‑convolution for the top seeds.
4. Polish each Adam result with a KL‑mirror (exponentiated‑gradient)
   descent on the true objective.
5. Perform cheap random perturbations (move a small amount of mass
   between two coordinates) while time permits.
6. Finish with a final exponentiated‑gradient polish.
The routine respects a 1000 s wall‑clock budget and always returns
a non‑empty valid sequence.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation when a already satisfies sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# Simple feasibility utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_sym_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric sum‑of‑powers shape:
    a_i ∝ (i+eps)^(-p) + (n‑1‑i+eps)^(-p) .
    """
    i = np.arange(n, dtype=float)
    # avoid division by zero
    left = 1.0 / ((i + eps + 1e-12) ** p)
    right = 1.0 / ((n - 1 - i + eps + 1e-12) ** p)
    vals = left + right
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups

def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,

) -> np.ndarray:

    Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv).

    Parameters

    z0 : ndarray
        Initial log‑probabilities (log‑softmax coordinates).
    deadline : float
        Absolute time at which the optimiser must stop.
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate for Adam.

    Returns
    -------
    a : ndarray
        Optimised probability vector (sum = 1, a_i ≥ 0).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # point on simplex
        conv = convolution_fft(a, a)       # linear conv, length 2n‑1
        mx = float(np.max(conv))

        # temperature β(t)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)         # positive weights
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central n entries

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

    # final point on the simplex
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Mirror descent (KL divergence) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # active set – indices where conv attains (almost) the max
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev) (restricted)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.08,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 43, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises
    `evaluate_sequence`.  The routine respects a 1000 s wall‑clock
    budget and always returns a valid (non‑empty) sequence.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    best_seq: List[float] = []
    best_fast_score = float("inf")   # fast surrogate value
    best_score = float("inf")        # official evaluate_sequence value

    # --------------------------------------------------------------
    # 0.  Fallback: use a previously discovered sequence if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            fast = compute_score(np.array(cand))
            if val < best_score:
                best_score, best_fast_score, best_seq = val, fast, cand
                print(f"Fallback (height_sequence_1) score = {val:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Optimise a cascade of increasing lengths
    # --------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 12.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ---- 1a. Build a diverse seed pool -----------------------
        seed_pool = []

        # analytic families
        seed_pool.append(arcsine_initial(n))
        seed_pool.append(np.full(n, 1.0 / n))                     # uniform

        # random power‑law (product) shapes
        for _ in range(6):
            p = rng.uniform(0.3, 2.5)
            eps = rng.uniform(0.0, 3.0)
            seed_pool.append(generate_power_shape(n, p, eps))

        # symmetric sum‑of‑powers shapes (new family)
        for _ in range(6):
            p = rng.uniform(0.3, 2.5)
            eps = rng.uniform(0.0, 3.0)
            seed_pool.append(generate_sym_power_shape(n, p, eps))

        # random beta‑shapes
        for _ in range(8):
            a = rng.uniform(0.1, 1.5)
            b = rng.uniform(0.1, 1.5)
            eps = rng.uniform(0.0, max(0.1, 0.01 * n))
            seed_pool.append(generate_beta_shape(n, a, b, eps))

        # mixtures of arcsine & uniform
        for _ in range(4):
            w = rng.uniform(0.0, 1.0)
            seed_pool.append(generate_mixture_shape(n, w))

        # random Dirichlet / Gamma draws
        for _ in range(4):
            alphas = rng.gamma(shape=1.0, scale=1.0, size=n) + 1e-6
            seq = alphas / alphas.sum()
            seed_pool.append(seq)

        # up‑sampled version of the current best (if any)
        if best_seq:
            old = np.array(best_seq, dtype=float)
            seed_pool.append(upsample_sequence(old, n))

        # ------------------------------------------------------
        # 1b. Fast ranking of seeds (no optimisation yet)
        # ------------------------------------------------------
        scored = [(compute_score(s), s) for s in seed_pool]
        scored.sort(key=lambda x: x[0])
        top_k = min(8, len(scored))
        top_seeds = [vec for _, vec in scored[:top_k]]
        print(f"Best seed fast score = {scored[0][0]:.6f}")

        # ------------------------------------------------------
        # 1c. Refine each promising seed with Adam + EG polish
        # ------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 6.0:
                break

            # jitter the log‑probabilities a little to explore neighbourhoods
            z0 = np.log(seed_vec + 1e-12) + rng.normal(scale=0.05, size=n)

            # allocate a slice of time for this seed
            remaining = deadline - time.time()
            per_seed = min(remaining / (top_k - idx + 1), 12.0)
            seed_deadline = time.time() + per_seed

            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=800.0,
                ramp_iters=120_000,
                lr=0.07,
            )

            fast_score = compute_score(a_opt)
            # quick check against the best fast score we have
            if fast_score + 1e-9 < best_fast_score:
                # official evaluation (scale‑invariant, but we keep it)
                try:
                    val = evaluate_sequence(a_opt.tolist())
                    if val < best_score:
                        best_score, best_fast_score, best_seq = val, fast_score, a_opt.tolist()
                        print(f"   New global best after Adam seed {idx}: {best_score:.6f}")
                except Exception as e:
                    print(f"   Evaluation failed after Adam: {e}")

            # --------------------------------------------------
            # 1d. Optional KL‑mirror polishing for this candidate
            # --------------------------------------------------
            if time.time() < deadline - 2.0:
                polish_deadline = deadline - 0.5
                a_polish = exponentiated_gradient_polish(
                    a_opt,
                    polish_deadline,
                    max_iter=150_000,
                    step=0.016,
                )
                fast2 = compute_score(a_polish)
                if fast2 + 1e-9 < best_fast_score:
                    try:
                        val2 = evaluate_sequence(a_polish.tolist())
                        if val2 < best_score:
                            best_score, best_fast_score, best_seq = val2, fast2, a_polish.tolist()
                            print(f"   Improved after EG polish: {best_score:.6f}")
                    except Exception as e:
                        print(f"   EG polish evaluation error: {e}")

    # --------------------------------------------------------------
    # 2.  Random local search around the best found sequence
    # --------------------------------------------------------------
    if best_seq:
        a_arr = np.array(best_seq, dtype=float)
        n = a_arr.size
        while time.time() < deadline - 1.0:
            # move a small fraction of mass from i to j
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            if a_arr[i] <= 0:
                continue
            delta = rng.uniform(0.0, a_arr[i] * 0.08)  # at most 8 % of source
            a_new = a_arr.copy()
            a_new[i] -= delta
            a_new[j] += delta
            if a_new[i] < 0:
                continue
            a_new /= a_new.sum()
            fast = compute_score(a_new)
            if fast + 1e-9 < best_fast_score:
                try:
                    val = evaluate_sequence(a_new.tolist())
                    if val < best_score:
                        best_score, best_fast_score, best_seq = val, fast, a_new.tolist()
                        a_arr = a_new
                        print(f"   Random move improved to {best_score:.6f}")
                except Exception:
                    pass

            # occasional smoothing step
            if rng.random() < 0.006:
                kernel = np.array([0.25, 0.5, 0.25])
                smooth = np.convolve(a_arr, kernel, mode='same')
                smooth = np.clip(smooth, 1e-12, None)
                smooth /= smooth.sum()
                fast_s = compute_score(smooth)
                if fast_s + 1e-9 < best_fast_score:
                    try:
                        val_s = evaluate_sequence(smooth.tolist())
                        if val_s < best_score:
                            best_score, best_fast_score, best_seq = val_s, fast_s, smooth.tolist()
                            a_arr = smooth
                            print(f"   Smoothing improved to {best_score:.6f}")
                    except Exception:
                        pass

    # --------------------------------------------------------------
    # 3.  Final global polishing (exact sub‑gradient)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\nFinal exact sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=250_000,
            step0=0.09,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            if final_val < best_score:
                best_score, best_seq = final_val, a_arr.tolist()
                print(f"   Final polish improved to {best_score:.6f}")
        except Exception as e:
            print(f"   Final polish failed: {e}")

    # --------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence, score = {best_score}")

    print(f"\nSearch finished. Best official score = {best_score:.6f}")
    return best_seq
