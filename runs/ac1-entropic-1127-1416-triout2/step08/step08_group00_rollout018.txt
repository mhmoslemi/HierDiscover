def l_bfgs_optimize(a0, time_deadline, beta=1000.0, max_iter=200):
    # Using SciPy if available.
    try:
        from scipy.optimize import minimize
    except Exception:
        return a0, compute_score(a0)
    n = a0.shape[0]

    # Initialize y = log(a0) maybe.
    eps = 1e-12
    y0 = np.log(a0 + eps)  # add epsilon to avoid log(0)
    # Optional shift to zero-mean: y0 = y0 - np.mean(y0)

    def obj_and_grad(y):
        # compute a = softmax(y)
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        a = exp_y / exp_y.sum()
        # convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # compute weights w for smooth max
        # stable LSE
        diff = conv - maxc  # all <=0
        w = np.exp(beta * diff)
        w_sum = w.sum()
        w = w / w_sum
        # gradient wrt a
        a_rev = a[::-1]
        # compute conv_rev = convolution_fft(a_rev, w)
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]  # slice central n entries
        # objective f = maxc + (1/beta) * log(w_sum)  # LSE
        f = maxc + (1.0 / beta) * np.log(w_sum)

        # gradient wrt y = a * (grad_a - dot(grad_a, a))
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)

        # return f, grad_y
        return f, grad_y

    # Use SciPy minimize with Jacobian
    res = minimize(fun=lambda y: obj_and_grad(y)[0],
                   x0=y0,
                   method='L-BFGS-B',
                   jac=lambda y: obj_and_grad(y)[1],
                   bounds=[(-20, 20)] * n,
                   options={'maxiter': max_iter, 'disp': False, 'ftol': 1e-12})
    y_opt = res.x
    # compute final a
    exp_y = np.exp(y_opt - np.max(y_opt))
    a_opt = exp_y / exp_y.sum()
    return a_opt, compute_score(a_opt)

    n = a.shape[0]

    best_a = a.copy()
    best_conv = conv.max()
    best_score = 2.0 * n * best_conv
    step_scale = 0.05

        partner = k - idx

        partner_i = k - i

"""
Propose a step‑function (non‑negative vector on the simplex) that minimises
`evaluate_sequence`.  The search consists of three stages:

1. **Fast L‑BFGS refinement** – a smooth log‑sum‑exp surrogate of the
   max‑convolution is minimised using SciPy's L‑BFGS‑B.  The variables are
   logits `y` with `a = softmax(y)` so that the simplex constraints are
   satisfied automatically.  Several random restarts are performed within the
   time budget.

2. **Targeted hill‑climb** – starting from the best L‑BFGS result we keep the
   exact convolution `c = a*a` and iteratively move a small fraction of mass
   from the index that contributes most to the currently maximal convolution
   entry to an index that contributes the least.  The convolution is updated
   by O(n) slice operations, and the true convolution is recomputed every
   1000 steps to avoid drift.

3. **Fallback** – if nothing is found we return a uniform distribution.

All intermediate vectors stay on the probability simplex, and the routine
returns a plain Python list of floats.  The total running time never exceeds
the 1000 s wall‑clock limit.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, no lambdas as required)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n≈2 k)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` objective for a vector with sum = 1.
    Returns 2·n·max(conv(a,a)), where conv is the full linear convolution.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. L‑BFGS optimisation of a smooth surrogate of max(conv)
# ----------------------------------------------------------------------


def lbfgs_optimize(start_a: np.ndarray,
                   time_deadline: float,
                   beta: float = 1000.0,
                   max_iter: int = 200) -> tuple[np.ndarray, float]:
    """
    Minimise a log‑sum‑exp surrogate of max(conv(a,a)) using SciPy's L‑BFGS‑B.
    Variables are logits y with a = softmax(y), guaranteeing a on the simplex.
    Returns the (exact‑score) best vector found.
    """
    try:
        from scipy.optimize import minimize
    except Exception:
        # SciPy unavailable – fall back to the plain start vector.
        return start_a, compute_score(start_a)

    n = start_a.shape[0]
    eps = 1e-12
    y0 = np.log(start_a + eps)                     # initialise logits

    # bounds are generous – we only need to keep numbers finite.
    bounds = [(-20.0, 20.0)] * n

    def softmax(y):
        y_shift = y - np.max(y)                     # for numerical stability
        exp_y = np.exp(y_shift)
        return exp_y / exp_y.sum()

    def obj_and_grad(y):
        a = softmax(y)
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()

        # log‑sum‑exp (stable) :  f = maxc + (1/β)·log∑exp(β·(conv‑maxc))
        diff = conv - maxc                         # ≤0
        w = np.exp(beta * diff)
        w_sum = w.sum()
        f = maxc + (1.0 / beta) * np.log(w_sum)

        # gradient wrt a :  ∂f/∂a = 2 * (a_rev * w)   (central slice)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # chain rule through softmax: ∂f/∂y = a ⊙ (grad_a – <grad_a,a>)
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)

        return f, grad_y

    # SciPy expects separate callable for value and jacobian; we give two lambdas.
    def fun(y):
        f, _ = obj_and_grad(y)
        return f

    def jac(y):
        _, g = obj_and_grad(y)
        return g

    # Run the optimiser – it terminates after `max_iter` iterations or earlier
    # if it converges.  We do not enforce a hard time‑limit inside SciPy;
    # the call is cheap for the modest `max_iter`.
    res = minimize(fun,
                   y0,
                   method='L-BFGS-B',
                   jac=jac,
                   bounds=bounds,
                   options={'maxiter': max_iter,
                            'gtol': 1e-12,
                            'ftol': 1e-12,
                            'disp': False})
    y_opt = res.x
    a_opt = softmax(y_opt)
    return a_opt, compute_score(a_opt)


# ----------------------------------------------------------------------
# 2. Targeted hill‑climb on the exact max‑convolution
# ----------------------------------------------------------------------


def targeted_hill_climb(seq: list[float],
                        time_deadline: float,
                        max_steps: int = 500_000) -> tuple[list[float], float]:
    """
    Greedy local search: repeatedly move a tiny fraction of mass from the
    index that participates most in the current maximal convolution entry to
    an index that participates the least.  The convolution is updated by O(n)
    slice operations; the exact convolution is recomputed every 1000 steps.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    # exact convolution cache
    conv = convolution_fft(a, a)
    best_a = a.copy()
    best_max = conv.max()
    best_score = 2.0 * n * best_max

    # step‑size as a fraction of the donor mass
    step_scale = 0.05
    it = 0

    while it < max_steps and time.time() < time_deadline:
        # index of the current maximal convolution entry
        k = int(np.argmax(conv))

        # contributions to that entry: a[i] * a[k-i]  (zero where out of range)
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        # donor = index with the largest product
        i = int(np.argmax(contrib))
        # partner of the donor (only needed for debugging)
        # partner_i = k - i

        # victim = index whose partner value is smallest (least impact on max)
        valid = np.where(mask)[0]
        if valid.size == 0:                     # should not happen for realistic k
            break
        partner_vals = a[partner[valid]]
        j = int(valid[np.argmin(partner_vals)])

        if i == j:                               # fallback – pick a random other index
            j = (i + 1) % n

        delta = step_scale * a[i]
        if delta <= 1e-15:                       # nothing left to move
            break
        delta = min(delta, a[i])                # keep non‑negative

        # --- tentative update of the convolution (linearised) -----------------
        # conv_new = conv - 2·δ·a shifted by i  + 2·δ·a shifted by j
        conv_candidate = conv.copy()
        conv_candidate[i:i + n] -= 2.0 * delta * a
        conv_candidate[j:j + n] += 2.0 * delta * a
        cand_max = conv_candidate.max()

        # accept only if we improve the maximal entry
        if cand_max < best_max - 1e-14:
            # accept move
            conv = conv_candidate
            a[i] -= delta
            a[j] += delta
            best_max = cand_max
            best_score = 2.0 * n * best_max
            best_a = a.copy()
            # become bolder
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # become more conservative
            step_scale *= 0.96

        it += 1

        # periodic exact recomputation to avoid accumulated linearisation error
        if it % 1000 == 0:
            conv = convolution_fft(a, a)
            cur_max = conv.max()
            if cur_max < best_max - 1e-14:
                best_max = cur_max
                best_score = 2.0 * n * best_max
                best_a = a.copy()

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a plain list of floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the surrounding harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]      # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a manageable length (larger n gives more freedom)
    # ------------------------------------------------------------------
    n = 2048
    if deadline - time.time() > 800:
        n = 3072                     # a modest increase when time permits
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # Helper – generate random starts on the simplex
    # ------------------------------------------------------------------
    def random_start():
        """Return a random probability vector of length `n` (dense or sparse)."""
        if np.random.rand() < 0.5:
            a = np.random.rand(n)
            a = a / a.sum()
        else:
            k = max(2, int(np.sqrt(n)))                     # sparsity level
            idx = np.random.choice(n, size=k, replace=False)
            a = np.zeros(n, dtype=float)
            a[idx] = 1.0 / k
            a = project_to_simplex(a)
        return a

    # ------------------------------------------------------------------
    # 2. Random‑restart L‑BFGS optimisation (smooth surrogate)
    # ------------------------------------------------------------------
    reserve_refine = 120.0                                 # seconds for later phases
    while time.time() < deadline - reserve_refine:
        remaining = deadline - time.time() - reserve_refine
        # each restart gets at most 20 s (or the whole remaining time if less)
        run_time = min(20.0, remaining * 0.6)
        per_restart_deadline = time.time() + run_time

        a0 = random_start()
        a_opt, scr = lbfgs_optimize(a0,
                                    time_deadline=per_restart_deadline,
                                    beta=1500.0,
                                    max_iter=250)      # cheap, many restarts
        exact = compute_score(a_opt)
        if exact < best_val:
            best_val = exact
            best_seq = a_opt.tolist()
            print(f"New best after L‑BFGS: score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Targeted hill‑climb polishing (exact objective)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nTargeted hill‑climb polishing")
        polished_seq, polished_score = targeted_hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=2_000_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final fallback (very small uniform vector) if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5. Final official evaluation (the harness will call it again)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return a plain Python list as required.
    return best_seq
