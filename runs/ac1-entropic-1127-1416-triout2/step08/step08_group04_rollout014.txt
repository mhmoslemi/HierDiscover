"""
propose_candidate
-----------------
Searches for a non‑negative step‑function (a list of floats) that minimizes the
official evaluation function

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in several fast phases:

1. **Coarse grid search** – tries many lengths (up to 16384) and a dense
   family of symmetric Beta‑shapes
        a_i ∝ ((i+½)(n‑½‑i)+ε)^{-p}
   with a fine grid on the exponent `p` (0.20…0.80) and a few
   regularisation values `ε`.  Uniform and simple arcsine shapes are
   also examined.  The best candidate (lowest surrogate 2·n·max(conv)) is
   kept.

2. **Parameter refinement** – starting from the best power‑shape,
   a tiny hill‑climbing loop perturbs `p` and `ε` (multiplicative steps)
   while the wall‑clock time permits.  This often reduces the surrogate
   by a few 1e‑4.

3. **Simple sub‑gradient descent** – performs a few thousand projected
   sub‑gradient steps on the *exact* max‑convolution objective.
   Projection is approximated by clamping to zero and renormalising,
   which is fast enough even for n≈16384.

4. **Exponentiated‑gradient polish** – a mirror‑descent (KL) step that
   stays on the probability simplex and quickly drives the surrogate
   down further.

5. **Stochastic local search** – cheap random mass transfers between
   coordinates; occasionally accepts a worsening move (simulated annealing)
   to escape tiny basins.

All phases respect a global 1000 s wall‑clock budget.  The final
sequence is returned as a plain list of Python floats; its official score
is printed for verification.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors via FFT.
    Returns vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.size
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a must already sum to 1 (probability simplex).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_shape(n: int, p: float, eps: float = 0.0) -> np.ndarray:
    """
    Symmetric Beta‑type shape:
        a_i ∝ ((i+½)(n‑½‑i) + eps)^{-p}
    Returns a probability vector (sum = 1), all entries ≥0.
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i) + eps
    # avoid negative or zero denom (should never happen)
    denom = np.maximum(denom, 1e-30)
    vals = denom ** (-p)
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        # fallback to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total


def uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def arcsine_shape(n: int) -> np.ndarray:
    """Arcsine law (p = 0.5, eps = 0)."""
    return generate_power_shape(n, p=0.5, eps=0.0)


def mixture_shape(n: int, w: float) -> np.ndarray:
    """
    Convex mixture of arcsine and uniform:
        w·arcsine + (1‑w)·uniform
    """
    arc = arcsine_shape(n)
    uni = uniform_shape(n)
    mix = w * arc + (1.0 - w) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Optimisation utilities (sub‑gradient, exponentiated‑gradient, random walk)
# ----------------------------------------------------------------------


def subgradient_step(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv(a,a)) w.r.t. a.
    Returns a vector of same size.
    """
    n = a.size
    L = 2 * n - 1
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    # tolerance for “active” indices
    tol = max_val * 1e-12 + 1e-14
    mask = (np.abs(conv - max_val) <= tol).astype(np.float64)

    # reverse mask for convolution trick
    mask_rev = mask[::-1]
    conv_rev = convolution_fft(a, mask_rev)          # length n+L‑1
    # pick the relevant slice
    subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return subgrad


def simple_subgradient_descent(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 20000,
    step0: float = 0.1,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Projection is approximated by clipping negatives to zero and renormalising.
    """
    a = a_init.copy()
    n = a.size
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        grad = subgradient_step(a)
        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        # simple projection onto simplex (clamp + renorm)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = uniform_shape(n)
        else:
            a = a / s
    return a


def exponentiated_gradient(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 100000,
    eta0: float = 0.2,
) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) polishing step.
    Guarantees positivity and automatic normalisation.
    """
    a = a_init.copy()
    n = a.size
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        grad = subgradient_step(a)
        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * grad)
        total = a.sum()
        if total == 0.0:
            a = uniform_shape(n)
        else:
            a = a / total
    return a


def stochastic_local_search(
    a_init: np.ndarray,
    deadline: float,
    max_trials: int = 30000,
    step_scale: float = 0.001,
    rng: np.random.Generator = np.random.default_rng(),
) -> np.ndarray:
    """
    Very cheap random hill climbing with a simple simulated‑annealing
    acceptance rule (temperature slowly decays to zero).
    """
    a_cur = a_init.copy()
    n = a_cur.size
    score_cur = compute_score(a_cur)

    # linear temperature schedule
    start_temp = 0.01
    end_temp = 0.0

    trials = 0
    while time.time() < deadline - 0.1 and trials < max_trials:
        # gradually shrink temperature
        progress = trials / max_trials
        temp = start_temp * (1.0 - progress) + end_temp * progress

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur or rng.random() < np.exp(-(s_new - score_cur) / max(temp, 1e-12)):
            a_cur = a_new
            score_cur = s_new
        trials += 1

    return a_cur

# ----------------------------------------------------------------------


def refine_power_params(
    n: int,
    p0: float,
    eps0: float,
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200,
) -> Tuple[float, float, np.ndarray]:
    """
    Simple hill‑climbing on (p, eps).  Starts from (p0, eps0) and
    returns a (potentially) better parameter pair and the corresponding
    shape vector.
    """
    best_p, best_eps, best_a = p0, eps0, a0
    best_score = compute_score(best_a)

    step_p = 0.05
    step_eps_rel = 0.2  # multiplicative factor for eps

    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        improved = False
        # try moving p up/down
        for dp in (-step_p, step_p):
            p_new = max(0.01, min(3.0, best_p + dp))
            # try eps unchanged and scaled up/down
            for factor in (1.0 - step_eps_rel, 1.0, 1.0 + step_eps_rel):
                eps_new = best_eps * factor if best_eps > 0.0 else 0.0
                a_new = generate_power_shape(n, p_new, eps_new)
                s_new = compute_score(a_new)
                if s_new < best_score:
                    best_score = s_new
                    best_p, best_eps, best_a = p_new, eps_new, a_new
                    improved = True
        # shrink the step sizes if nothing improved
        if not improved:
            step_p *= 0.7
            step_eps_rel *= 0.7
        # stop early if steps become too tiny
        if step_p < 1e-4 and step_eps_rel < 1e-4:
            break
    return best_p, best_eps, best_a

    # ------------------------------------------------------------------
    candidate_ns = [1024, 2048, 4096, 8192, 16384]
    p_grid = np.linspace(0.20, 0.80, 31)                # finer exponent grid
    eps_grid = [0.0, 1e-12, 1e-10, 1e-8, 1e-6, 1e-4]   # regularisation
    w_grid = np.linspace(0.0, 1.0, 21)                 # mixture weight grid

    print("\n--- Stage 1: exhaustive shape grid search ---")
    for n in candidate_ns:
        if time.time() >= deadline:
            break
        print(f"\nTesting length n = {n}")

        # ----- uniform -----
        uni = uniform_shape(n)
        s = compute_score(uni)
        if s < best_surrogate:
            best_surrogate, best_seq = s, uni.tolist()
            print(f"  New best (uniform)    surrogate = {s:.6f}")

        # ----- arcsine (p=0.5) -----
        arc = arcsine_shape(n)
        s = compute_score(arc)
        if s < best_surrogate:
            best_surrogate, best_seq = s, arc.tolist()
            print(f"  New best (arcsine)    surrogate = {s:.6f}")

        # ----- power‑shape grid -----
        for p in p_grid:
            for eps in eps_grid:
                shape = generate_power_shape(n, p, eps=eps)
                s = compute_score(shape)
                if s < best_surrogate:
                    best_surrogate, best_seq = s, shape.tolist()
                    print(f"  New best (p={p:.3f}, eps={eps:.0e}) surrogate = {s:.6f}")

        # ----- mixtures arcsine / uniform -----
        for w in w_grid:
            mix = mixture_shape(n, w)
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"  New best (mix w={w:.2f}) surrogate = {s:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – parameter refinement for the best power‑shape (if applicable)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        # Try to infer whether the best sequence originates from a power‑shape
        # by attempting a short hill‑climb starting from its current parameters.
        # We simply run the generic hill‑climb on (p,eps) using the length
        # of the current best sequence.
        n_best = len(best_seq)
        a_best = np.array(best_seq, dtype=np.float64)

        # rough guess for starting parameters:
        # – if the shape looks like arcsine, p≈0.5; – otherwise we keep p=0.5.
        # – eps is guessed from the smallest element of a_best.
        # These guesses are harmless; the refinement will move them if needed.
        p_guess = 0.5
        eps_guess = 0.0
        # Run refinement
        p_ref, eps_ref, a_ref = refine_power_params(
            n=n_best,
            p0=p_guess,
            eps0=eps_guess,
            a0=a_best,
            deadline=deadline,
            max_iter=120,
        )
        s = compute_score(a_ref)
        if s < best_surrogate:
            best_surrogate, best_seq = s, a_ref.tolist()
            print(f"\nAfter power‑shape refinement: p={p_ref:.4f}, eps={eps_ref:.0e}, surrogate={s:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simple sub‑gradient descent (few thousand steps)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Stage 3: simple sub‑gradient descent ---")
        a = np.array(best_seq, dtype=np.float64)
        a = simple_subgradient_descent(a, deadline, max_iter=8000, step0=0.12)
        s = compute_score(a)
        print(f"  After sub‑gradient surrogate = {s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()

    # ------------------------------------------------------------------
    # Stage 4 – exponentiated‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Stage 4: exponentiated‑gradient polish ---")
        a = np.array(best_seq, dtype=np.float64)
        a = exponentiated_gradient(a, deadline, max_iter=50000, eta0=0.18)
        s = compute_score(a)
        print(f"  After EG polish surrogate = {s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()

    # ------------------------------------------------------------------
    # Stage 5 – cheap stochastic local search (with annealing)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        print("\n--- Stage 5: stochastic local search ---")
        a = np.array(best_seq, dtype=np.float64)
        a = stochastic_local_search(a, deadline, max_trials=40000, step_scale=0.001, rng=rng)
        s = compute_score(a)
        print(f"  After random search surrogate = {s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = uniform_shape(fallback_n).tolist()
        final_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
