"""
Improved multi‑stage optimizer for the non‑negative step‑function problem.

Key ideas
---------
1.  Very fine grid on the symmetric product family (α up to 8) and a
    V‑shaped exponential family (γ up to 6).  After a coarse sweep we
    perform a local finer sweep around the best parameters.
2.  Several random initialisers (sparse, monotone) are refined with a
    long Adam optimisation of the smoothed max‑convolution.
3.  A cheap Euclidean projected sub‑gradient descent provides a quick
    reduction of the true max‑objective.
4.  Symmetrisation (averaging with the reversed vector) is tried,
    because many good solutions are almost symmetric.
5.  A final simulated‑annealing hill‑climb with a larger move budget,
    followed by a support‑pruning pass (keep only the largest entries
    and renormalise) to possibly lower the max‑convolution further.
6.  All phases respect the global wall‑clock limit (default 1000 s).

The routine returns a plain Python list of floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the probability simplex:
        2·n·max(a*a)
    (Here sum(a)=1, therefore the denominator of `evaluate_sequence`
    is 1 and the expression reduces to 2·n·max(conv).)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (parameterised shapes)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ [(i+½)(n‑½‑i)]^(‑α) .  Symmetric, normalised to sum=1.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    # avoid overflow/underflow
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """
    a_i ∝ (i+½)^(‑β) .  Monotonically decreasing, sum=1.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def v_exponential_family(n: int, gamma: float) -> np.ndarray:
    """
    V‑shaped exponential: a_i ∝ exp(‑γ·min(i, n‑1‑i)).
    Heavier mass at the two ends, decaying towards the centre.
    """
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def two_param_product_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Generalised product: a_i ∝ (i+½)^{p}·(n‑½‑i)^{q},
    where p and q are (typically negative) exponents.
    """
    i = np.arange(n, dtype=np.float64)
    term1 = (i + 0.5) ** p
    term2 = (n - 0.5 - i) ** q
    w = term1 * term2
    # replace NaNs/Infs by zero
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0.0:
        # fallback to uniform
        return np.full(n, 1.0 / n)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]  # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth optimisation of the log‑sum‑exp surrogate (Adam)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 5.0,
                         beta_end: float = 200.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise f_β(a) = (1/β)·log Σ exp(β·(a*a)_k) using Adam on the
    soft‑max parametrisation a = softmax(z).  The smoothing
    parameter β grows exponentially from `beta_start` to `beta_end`.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)       # initialise from the start vector

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- current a ----
        a = softmax(z)

        # ---- smoothed weights ----
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # numerically stable exponentials
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ---- gradient w.r.t. a (using correlation) ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 800_000,
                     step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on 2·n·max(conv).  Step size = step0/√t.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        # occasional check
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (stochastic local search)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
        – 50 %: swap two entries.
        – 50 %: transfer a fraction of mass from i to j.
    Tiny temperature schedule allows very occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap two coordinates
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            # accept
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Support‑pruning helper
# ----------------------------------------------------------------------


def prune_and_renorm(a: np.ndarray, keep_frac: float) -> np.ndarray:
    """
    Keep only the largest `keep_frac` fraction of the mass,
    set the rest to zero and re‑project onto the simplex.
    """
    n = a.size
    k = max(2, int(n * keep_frac))
    # indices of the largest k entries
    idx = np.argpartition(a, -k)[-k:]
    new = np.zeros_like(a)
    new[idx] = a[idx]
    return project_to_simplex(new)

        if time.time() >= deadline - 250.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Local fine‑grained search around the best analytic parameters
    # ------------------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 200.0:
        lo = max(0.0, best_alpha - 0.3)
        hi = min(8.0, best_alpha + 0.3)
        fine_alphas = np.linspace(lo, hi, 61)          # step ≈0.01
        for a in fine_alphas:
            if time.time() >= deadline - 200.0:
                break
            vec = symmetric_product_family(n, a)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (fine‑sym‑α={a:.3f}) = {best_score:.6f}")

    if best_gamma is not None and time.time() < deadline - 200.0:
        lo = max(0.0, best_gamma - 0.3)
        hi = min(6.0, best_gamma + 0.3)
        fine_gammas = np.linspace(lo, hi, 61)
        for g in fine_gammas:
            if time.time() >= deadline - 200.0:
                break
            vec = v_exponential_family(n, g)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (fine‑v‑γ={g:.3f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random sparse restarts + Adam‑based smoothed optimisation
    # ------------------------------------------------------------------
    while time.time() < deadline - 120.0:
        start_vec = random_sparse_start(n)
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 100.0,
            max_iters=200_000,
            beta_start=6.0,
            beta_end=300.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam‑smooth new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Long projected sub‑gradient refinement (cheapest per iteration)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=1_200_000,
            step0=0.04,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Symmetrisation test (many good solutions are nearly symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final hill‑climb with simulated annealing (more steps than before)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=3_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional support pruning (keep only the largest entries)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] trying support‑pruning")
        a_cur = np.array(best_seq, dtype=np.float64)
        for keep in [0.8, 0.6, 0.4, 0.2, 0.1]:
            if time.time() >= deadline - 2.0:
                break
            cand = prune_and_renorm(a_cur, keep_frac=keep)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.tolist()
                print(f"[{time.time():.1f}] prune keep={keep:.2f} improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: fall back to a modest uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
