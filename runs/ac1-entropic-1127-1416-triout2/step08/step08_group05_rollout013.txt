"""
Search for a non‑negative step sequence that minimises
    evaluate_sequence

The strategy is:
  1. Enumerate a few simple analytic families (arcsine, Beta,
     power‑law, mixtures) on several lengths (up to 32768).  All
     candidates are normalised to sum = 1, which makes the official
     objective 2·n·max(conv) and can be evaluated cheaply with an FFT‑based
     convolution (function ``compute_score``).

  2. Keep the best candidate found so far.

  3. If time permits, up‑sample the current best to a larger grid and
     polish it with a few thousand steps of exponentiated‑gradient (KL‑mirrored)
     descent on the true objective; finally finish with a short Euclidean
     sub‑gradient descent.

The routine respects a 1000 s wall‑clock budget, never returns an empty
sequence and always prints intermediate scores.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective when ``a`` is already
    normalised (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_shape(n: int, eps: float = 0.25) -> np.ndarray:
    """Arcsine‑like envelope (discrete Beta(0.5,0.5)), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    # clip tiny negative due to numeric issues
    vals = np.clip(vals, 0.0, None)
    total = vals.sum()
    if total == 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Inverse‑polynomial envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_shape(n, eps=0.25)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix = np.clip(mix, 0.0, None)
    mix /= mix.sum()
    return mix


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the exact objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Mirror descent (KL divergence) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)      # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for active set
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG polish it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.08,
) -> np.ndarray:
    """
    Projected Euclidean sub‑gradient descent on
        f(a) = 2·n·max(conv(a,a))
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a
