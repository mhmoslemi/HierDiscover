```python
"""
Enhanced search for low‑max‑convolution step sequences.

The algorithm:
1. Choose a target length (usually 4096–8192) based on remaining time.
2. Build a large pool of simple analytic shapes (arcsine, power‑law,
   beta‑like, mixtures) with many random parameters, plus the previous
   best sequence up‑sampled to the target length.
3. Evaluate the fast surrogate ``2·n·max(conv)`` for every seed using an
   FFT‑based convolution and keep the top few.
4. For each promising seed run a long Adam optimisation on a smooth
   Log‑Sum‑Exp approximation of the max‑convolution (temperature is
   increased to a large value).  The optimisation works in the log‑softmax
   space, guaranteeing non‑negativity and a unit sum.
5. If time permits, polish the result with an exponentiated‑gradient
   mirror‑descent on the true objective, followed by a short projected
   sub‑gradient descent.
6. The best sequence found (according to ``evaluate_sequence``) is
   returned.  The routine always respects the 1000 s wall‑clock budget
   and returns a valid non‑empty list.
"""
import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a, a)), assuming sum(a)=1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping onto the simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_shape(n: int, eps: float = 0.0) -> np.ndarray:
    """Arcsine‑like envelope 1/√((i+½)(n‑½‑i)+eps), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    # Prevent division by zero when eps is extremely small
    denom = (i + 0.5) * (n - 0.5 - i) + eps
    vals = 1.0 / np.sqrt(denom)
    vals /= vals.sum()
    return vals


def power_shape(n: int, p: float, eps: float = 0.0) -> np.ndarray:
    """Power‑law shape 1/(((i+½)(n‑½‑i)+eps) ** p), normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def beta_shape(n: int, alpha: float, beta: float, eps: float = 0.0) -> np.ndarray:
    """Beta‑distribution‑like shape (i+eps)^{α‑1} (n‑1‑i+eps)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    vals = left * right
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def mixture_shape(n: int, w_arc: float, eps: float = 0.0) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_shape(n, eps=eps)
    uni = np.full(n, 1.0 / n)
    mix = w_arc * arc + (1.0 - w_arc) * uni
    mix /= mix.sum()
    return mix


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 2000.0,
    ramp_iters: int = 150_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    z0 : ndarray
        Initial log‑softmax coordinates.
    deadline : float
        Absolute time at which the optimiser must stop.
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate for Adam.

    Returns
    -------
    a : ndarray
        Optimised probability vector (sum = 1, a_i ≥ 0).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # point on the simplex
        conv = convolution_fft(a, a)       # length 2n‑1
        mx = float(np.max(conv))

        # temperature β(t)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)         # weights, can be very small
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient wrt a: 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central n entries

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output (optional, throttled)
        if t % 10000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate={compute_score(a):.6f}")

    # final simplex point
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step: float = 0.015,
) -> np.ndarray:
    """
    Mirror descent (KL divergence) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 10000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.09,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 10000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Seed construction utilities
# ----------------------------------------------------------------------


def build_seed_pool(
    n: int,
    rng: np.random.Generator,
    best_sofar: List[float],
    extra: int = 200,
) -> List[np.ndarray]:
    """
    Build a pool of diverse analytic shapes plus a few randomised families.
    The pool size is moderate (~few hundred) to keep the fast surrogate
    evaluation cheap.
    """
    seeds: List[np.ndarray] = []

    # deterministic candidates
    seeds.append(np.full(n, 1.0 / n))                # uniform
    seeds.append(arcsine_shape(n, eps=0.0))          # pure arcsine
    seeds.append(arcsine_shape(n, eps=0.01))
    seeds.append(arcsine_shape(n, eps=0.25))

    # mixtures of arcsine and uniform
    for _ in range(4):
        w = rng.uniform(0.0, 1.0)
        eps = rng.uniform(0.0, 0.5)
        seeds.append(mixture_shape(n, w_arc=w, eps=eps))

    # random power‑law shapes
    for _ in range(20):
        p = rng.uniform(0.05, 3.0)          # exponent
        eps = rng.uniform(0.0, 5.0)         # smoothing
        seeds.append(power_shape(n, p=p, eps=eps))

    # random beta‑like shapes
    for _ in range(20):
        a = rng.uniform(0.05, 6.0)
        b = rng.uniform(0.05, 6.0)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        seeds.append(beta_shape(n, alpha=a, beta=b, eps=eps))

    # add up‑sampled best if available
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        seeds.append(upsample_sequence(old, n))

    # random perturbations of existing seeds to increase diversity
    for _ in range(extra):
        base = seeds[rng.integers(len(seeds))]
        # add small Gaussian noise and re‑project onto simplex
        noise = rng.normal(scale=0.02, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = pert / pert.sum()
        seeds.append(pert)

    return seeds

def propose_candidate(seed: int = 43, budget_s: int = 1000, **kwargs) -> List[float]:

    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously discovered sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Choose target length based on remaining time
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    # Prefer larger n when we have enough budget, otherwise fallback to 4096
    if time_left > 700:
        n = 8192
    elif time_left > 350:
        n = 4096
    else:
        n = 2048
    print(f"Target length = {n}")

    # ------------------------------------------------------------------
    # Build a large pool of analytic seeds and quickly rank them using the fast surrogate
    # ------------------------------------------------------------------
    seed_pool = build_seed_pool(n, rng, best_seq, extra=100)
    scored = [(compute_score(s), s) for s in seed_pool]
    scored.sort(key=lambda x: x[0])
    # Keep the most promising few seeds for expensive optimisation
    top_k = 7
    top_seeds = [vec for _, vec in scored[:top_k]]
    print(f"Fast surrogate best score among seeds = {scored[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Allocate time per seed (roughly equal split of the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    per_seed_time = max(8.0, remaining / (len(top_seeds) + 2))

    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 12.0:

        print(f"\n=== Optimising seed {idx}/{len(top_seeds)} (budget≈{per_seed_time:.1f}s) ===")
        seed_deadline = min(time.time() + per_seed_time, deadline - 5.0)

        # ------------------------------------------------------------------

        try:

            print(f"   Official score after Adam = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                print(f"   *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"   Evaluation failed: {e}")

        # ------------------------------------------------------------------
        # 3. Optional KL‑mirror polishing if time permits
        # ------------------------------------------------------------------
        if time.time() < deadline - 6.0:
            polish_deadline = deadline - 2.0
            a_polish = exponentiated_gradient_polish(
                np.array(best_seq, dtype=float),

                step=0.015,
            )
            try:
                val2 = evaluate_sequence(a_polish.tolist())
                print(f"   Polished (EG) score = {val2:.6f}")
                if val2 < best_val:
                    best_val, best_seq = val2, a_polish.tolist()
                    print(f"   *** New best after EG polish: {best_val:.6f} ***")
            except Exception as e:
                print(f"   EG polishing failed: {e}")

    # ------------------------------------------------------------------
    # 4. Final global polishing (exact sub‑gradient) if time remains

"""
Enhanced search for non‑negative step sequences that minimise
``evaluate_sequence``.

The algorithm proceeds as follows:

1. **Target length** – Choose a large n (8192, 4096 or 2048) depending on the
   remaining wall‑clock budget.
2. **Seed pool** – Generate a diverse collection of analytic shapes
   (arcsine, power‑law, beta‑like, mixtures) with many random parameters and
   a few random perturbations of the current best solution.  All seeds are
   normalised to sum = 1.
3. **Fast surrogate ranking** – Evaluate the cheap surrogate
   ``2·n·max(conv(a,a))`` (computed with FFT) for every seed and keep the
   most promising few.
4. **Smooth‑max Adam optimisation** – For each promising seed run a long
   Adam optimiser on a smooth log‑sum‑exp approximation of the max‑convolution.
   The optimisation works in the log‑softmax space, guaranteeing non‑negativity
   and unit‑sum.
5. **Official evaluation** – After Adam, call ``evaluate_sequence``; keep the
   best sequence found.
6. **KL‑mirror polishing** – If time permits, polish the best candidate with
   exponentiated‑gradient mirror descent on the true objective.
7. **Final sub‑gradient polishing** – A short projected sub‑gradient descent
   is applied as a final refinement.
8. **Guard** – Ensure a non‑empty sequence is always returned.

The routine respects the 1000 s wall‑clock limit and prints intermediate
scores for monitoring.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a, a)), assuming sum(a)=1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping onto the simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_shape(n: int, eps: float = 0.0) -> np.ndarray:
    """Arcsine‑like envelope 1/√((i+½)(n‑½‑i)+eps), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    denom = (i + 0.5) * (n - 0.5 - i) + eps
    vals = 1.0 / np.sqrt(denom)
    vals /= vals.sum()
    return vals


def power_shape(n: int, p: float, eps: float = 0.0) -> np.ndarray:
    """Power‑law shape 1/(((i+½)(n‑½‑i)+eps) ** p), normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def beta_shape(n: int, alpha: float, beta: float, eps: float = 0.0) -> np.ndarray:
    """Beta‑distribution‑like shape (i+eps)^{α‑1} (n‑1‑i+eps)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    vals = left * right
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def mixture_shape(n: int, w_arc: float, eps: float = 0.0) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_shape(n, eps=eps)
    uni = np.full(n, 1.0 / n)
    mix = w_arc * arc + (1.0 - w_arc) * uni
    mix /= mix.sum()
    return mix


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 1500.0,
    ramp_iters: int = 120_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on a smooth log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    z0 : ndarray
        Initial log‑softmax coordinates.
    deadline : float
        Absolute time at which the optimiser must stop.
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate for Adam.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # simplex point
        conv = convolution_fft(a, a)       # length 2n‑1
        mx = float(np.max(conv))

        # temperature β(t)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)         # weights
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central n entries

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 10_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate={compute_score(a):.6f}")

    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 180_000,
    step: float = 0.015,
) -> np.ndarray:
    """
    Mirror descent (KL) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 10_000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.09,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 10_000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Seed construction utilities
# ----------------------------------------------------------------------


def build_seed_pool(
    n: int,
    rng: np.random.Generator,
    best_sofar: List[float],
    extra: int = 100,
) -> List[np.ndarray]:
    """
    Assemble a moderately sized pool of analytic shapes and random perturbations.
    """
    seeds: List[np.ndarray] = []

    # deterministic baselines
    seeds.append(np.full(n, 1.0 / n))                # uniform
    seeds.append(arcsine_shape(n, eps=0.0))
    seeds.append(arcsine_shape(n, eps=0.01))
    seeds.append(arcsine_shape(n, eps=0.25))

    # mixture of arcsine and uniform
    for _ in range(4):
        w = rng.uniform(0.0, 1.0)
        eps = rng.uniform(0.0, 0.5)
        seeds.append(mixture_shape(n, w_arc=w, eps=eps))

    # random power‑law shapes
    for _ in range(20):
        p = rng.uniform(0.05, 3.0)
        eps = rng.uniform(0.0, 5.0)
        seeds.append(power_shape(n, p=p, eps=eps))

    # random beta‑like shapes
    for _ in range(20):
        a = rng.uniform(0.05, 6.0)
        b = rng.uniform(0.05, 6.0)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        seeds.append(beta_shape(n, alpha=a, beta=b, eps=eps))

    # up‑sample current best if we have one
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        seeds.append(upsample_sequence(old, n))

    # random perturbations of existing seeds for extra diversity
    for _ in range(extra):
        base = seeds[rng.integers(len(seeds))]
        noise = rng.normal(scale=0.02, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = pert / pert.sum()
        seeds.append(pert)

    return seeds
