def multi_subgrad_optimize(start_a, deadline, max_iters=400000, step0=0.2, reduction_factor=0.001):
    n = len(start_a)
    a = project_to_simplex(start_a.astype(np.float64))
    best_a = a.copy()
    best_score = compute_score(a)
    idx_slice = 2 * n - 2
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # record best
        cur_score = 2.0 * n * maxc
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # target threshold for violations:
        target = maxc * (1.0 - reduction_factor)  # e.g., 0.001 reduction
        viol = conv - target
        mask = viol > 0
        if not mask.any():
            continue
        w = 2.0 * viol * mask   # derivative of (conv - target)^2 wrt conv
        # gradient
        grad_full = convolution_fft(a, w[::-1])
        grad = 2.0 * grad_full[idx_slice - np.arange(n)]
        # step size
        step = step0 / np.sqrt(it)
        a = project_to_simplex(a - step * grad)
        # optionally enforce symmetry:
        # a = (a + a[::-1]) * 0.5
        # a = project_to_simplex(a)  # reproject to simplex after symmetrization (optional)
        if it % 5000 == 0:
            print(f"[{time.time():.1f}] multi_subgrad iter {it}, cur_score={cur_score:.6f}, best={best_score:.6f}")
    # final best check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

"""
Multi‑stage optimizer for the non‑negative step‑function problem.

Stage summary
-------------
1. Analytic families (symmetric product & left‑heavy power law) – extensive
   grid search on the probability simplex, keep the best.
2. Random decreasing vectors – a larger batch of draws.
3. Adam‑based smoothed optimisation (log‑sum‑exp) – longer runs and a
   higher final smoothing parameter β.
4. Custom projected gradient descent on the squared‑violation penalty
   (uses all active constraints, not just the single maximum).  This
   “multi‑sub‑gradient” step is cheap and empirically reaches lower
   max‑convolution values.
5. Optional cheap sub‑gradient refinement (single‑max variant) for a
   final polish.
6. Symmetrisation (average with the reverse) – many good solutions are
   symmetric.
7. Simulated‑annealing hill‑climb on the simplex.

All stages are bounded by the global wall‑clock limit (default 1000 s).
The routine returns a plain Python list of floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the probability simplex:
        2·n·max(a*a)
    (here sum(a)=1, so the denominator is 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        # Return the centre of the simplex.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)  # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Adam on smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 300.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimises the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially from
    `beta_start` to `beta_end`.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise logits

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- β schedule (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- current a -----
        a = softmax(z)

        # ----- smoothed weights -----
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ----- gradient wrt a -----
        # derivative of (a*a)_k w.r.t. a is 2 * (a reversed convolved with unit vector at k)
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # ----- back‑prop through soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional exact evaluation -----
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam iter {it}, new best = {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Multi‑subgradient descent on the squared‑violation penalty
# ----------------------------------------------------------------------


def multi_subgrad_optimize(start_a: np.ndarray,
                           deadline: float,
                           max_iters: int = 300_000,
                           step0: float = 0.08,
                           reduction: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the penalty
          P(a) = Σ_k max(0, (a*a)_k - t)^2
    where t = (1‑reduction)·max_k (a*a)_k.
    This uses *all* indices that exceed the target, giving a richer
    sub‑gradient than the classic single‑max variant.
    """
    n = start_a.shape[0]
    a = project_to_simplex(start_a.astype(np.float64).copy())

    best_a = a.copy()
    best_score = compute_score(a)

    idx_slice = 2 * n - 2  # index needed for gradient extraction

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = conv.max()
        cur_score = 2.0 * n * maxc
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] Multi‑subgrad iter {it}, new best = {best_score:.6f}")

        # target slightly smaller than current maximum
        target = maxc * (1.0 - reduction)
        viol = conv - target
        mask = viol > 0.0
        if not mask.any():
            # nothing above target – tighten further next round
            continue

        # w = 2 * viol (gradient of (violation)^2 w.r.t. conv)
        w = 2.0 * viol * mask

        # gradient = 2 * (a ⋆ w_rev)[2n‑2‑i]
        grad_full = convolution_fft(a, w[::-1])
        grad = 2.0 * grad_full[idx_slice - np.arange(n)]

        # diminishing step size
        step = step0 / np.sqrt(it)
        a = project_to_simplex(a - step * grad)

        # optional tiny symmetry enforcement every few thousand steps
        if it % 5000 == 0:
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)

    # final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Classic sub‑gradient descent (single‑max) – kept as a final polish
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """Cheap sub‑gradient descent on the true max‑objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # compute sub‑gradient at the active index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad iter {t}, new best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with simulated annealing
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Long random walk on the simplex with a tiny SA component."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature scales with the current best score
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

        if step % 250_000 == 0:
            print(f"[{time.time():.1f}] SA step {step}, best = {best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the global time budget and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so the harness can finish cleanly
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # Choose vector length – 8192 gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 8192
    print(f"[{time.time():.1f}] start search, n = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  fallback to a known construction if available
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families: expanded grid search
    # ------------------------------------------------------------------
    # a) symmetric product family – scan a wider α range
    alphas = np.linspace(0.0, 10.0, 201)  # step ≈0.05
    for a in alphas:
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, a)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={a:.2f}) = {best_score:.6f}")

    # b) left‑heavy power‑law family – extend β up to 10
    betas = np.linspace(0.0, 10.0, 201)
    for b in betas:
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, b)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={b:.2f}) = {best_score:.6f}")

    # c) random monotone draws – more samples than before
    for _ in range(200):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam‑based smoothed optimisation – several restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:
        start_vec = random_sparse_start(n)
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 200.0,
            max_iters=150_000,
            beta_start=6.0,
            beta_end=250.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after Adam‑smooth = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Multi‑subgradient descent (most expensive stage)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting multi‑subgradient refinement")
        a_multi, sc_multi = multi_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 80.0,
            max_iters=300_000,
            step0=0.08,
            reduction=0.001,
        )
        if sc_multi < best_score:
            best_score = sc_multi
            best_seq = a_multi.tolist()
            print(f"[{time.time():.1f}] after multi‑subgrad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional classic sub‑gradient polish (cheap)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:

        a_sub, sc_sub = subgrad_optimize(

        a_sym = (a_arr + a_arr[::-1]) * 0.5

        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_500_000,
        )
        if sc_final < best_score:

            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is None:

        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
