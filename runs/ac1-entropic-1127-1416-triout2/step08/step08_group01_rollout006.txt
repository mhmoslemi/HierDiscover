def exp_smooth_optimize(start_a, deadline, max_iters=200000, beta_start=0.5, beta_end=1000., eta0=0.6):
    a = np.maximum(start_a, eps)
    a/=a.sum()
    n = a.shape[0]
    m = np.zeros_like(a) # first moment
    v = np.zeros_like(a) # second moment
    beta1 = 0.9
    beta2 = 0.999
    epsadam = 1e-8

    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of smooth max
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n-1:n-1+n]
        # Adam update in log-space (multiplicative):
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = eta0 / np.sqrt(it)
        # multiplicative update
        a = a * np.exp(-step * m_hat / (np.sqrt(v_hat) + epsadam))
        # enforce positivity and renormalize
        a = np.maximum(a, eps)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

def hill_climb(start_seq, deadline, max_steps=500000):
    a = start_seq.copy()
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(best_a)
    step_scale = 0.1
    temperature = 1.0
    while max_steps>0 and time.time()<deadline:
        max_sc = best_sc
        # compute conv and max shift only occasionally, maybe every step?
        conv = convolution_fft(best_a, best_a) # compute each iteration
        k = int(np.argmax(conv))
        # find biggest contributor to this k
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # compute contributions:
        i_range = np.arange(i_min, i_max+1)
        j_range = k - i_range
        contrib = best_a[i_range] * best_a[j_range]
        # choose i with max contribution
        idx = np.argmax(contrib)
        i_big = i_range[idx]
        j_big = j_range[idx]
        # random other indices
        p = np.random.randint(n)
        q = np.random.randint(n)
        # ensure distinct from i_big and j_big
        while p == i_big or p == j_big:
            p = np.random.randint(n)
        while q == i_big or q == j_big or q == p:
            q = np.random.randint(n)
        # transfer fraction
        delta_i = step_scale * best_a[i_big]
        delta_j = step_scale * best_a[j_big]
        cand = best_a.copy()
        cand[i_big] -= delta_i
        cand[p] += delta_i
        cand[j_big] -= delta_j
        cand[q] += delta_j
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.3)
            temperature = max(temperature * 0.99, 0.001)
        else:
            step_scale *= 0.97
            temperature *= 0.995
        max_steps -= 1

    return best_a, best_sc

"""
Improved step‑function search.

The code follows the five‑stage plan of the reference implementation but
adds several refinements:

* **Adam‑style KL‑mirror updates** for the smooth max–objective.  The
  exponential‑gradient scheme is replaced by an adaptive first‑ and
  second‑moment estimator which dramatically improves convergence
  when the smoothing parameter β becomes large.

* **Extended α‑grid** (including a few negative exponents) for a more
  diverse analytical start.

* **Two random‑initialisation families** – a sparse start and a dense
  Dirichlet draw – to increase diversity of the search.

* **Higher final smoothing** (β≈2000) and a longer dedicated run.

* **Targeted hill‑climbing** – moves are biased towards the indices that
  contribute most to the current maximal convolution value, allowing
  faster reduction of the true objective.

All phases respect a global 1000 s wall‑clock limit.  The best sequence
found is returned as a plain Python list of floats.

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a⊛a) for a vector with sum = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytical start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Family a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    # avoid log(0) by adding a tiny offset
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow for extreme alphas
    w = np.where(np.isfinite(w), w, np.finfo(np.float64).max)
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a modest α‑grid and keep the best vector."""
    best_a = None
    best_score = float("inf")
    # expanded grid – a few negative alphas (centre‑heavy) and a finer positive range
    alphas = np.concatenate([
        np.linspace(-1.5, -0.5, 7),
        np.linspace(-0.5, 0.0, 7),
        np.linspace(0.0, 3.0, 31)
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Random initialisations
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    sparsity = np.random.uniform(0.03, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dirichlet_start(n: int, concentration: float = 0.8) -> np.ndarray:
    """Dense random draw from a Dirichlet (α=concentration)."""
    # Gamma draws
    g = np.random.gamma(concentration, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    a = g / g.sum()
    return a


# ----------------------------------------------------------------------
# 3. Adam‑mirrored exponentiated gradient on the smooth max
# ----------------------------------------------------------------------


def exp_adam_optimize(start_a: np.ndarray,
                      deadline: float,
                      max_iters: int = 300_000,
                      beta_start: float = 0.5,
                      beta_end: float = 2000.0,
                      eta0: float = 0.6,
                      eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑objective using a KL‑mirror step with
    Adam‑style adaptive moments.  β is exponentially annealed from
    ``beta_start`` to ``beta_end``.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    # Adam moments (in log‑space)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- β schedule (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- smooth weights (soft‑max) -----
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))    # stabilized
        w = expw / expw.sum()

        # ----- gradient of smoothed max -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ----- Adam on the log‑space (mirror) -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # step size – classic 1/√t schedule
        eta = eta0 / np.sqrt(it)

        # multiplicative KL‑mirror update
        a = np.maximum(a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps_adam)), eps)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- record progress -----
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 600_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic Euclidean projected sub‑gradient method for the exact objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for shift ``k``
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 5. Targeted stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_targeted(start_seq: np.ndarray,
                        deadline: float,
                        max_steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Random walk that preferentially moves mass away from the indices
    contributing most to the current maximal convolution value.
    Only strictly improving moves are kept (no worsening acceptance).
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # scale for mass transfers – starts small and adapts
    step_scale = 0.12

    while max_steps > 0 and time.time() < deadline:
        # occasionally recompute the full convolution to know the active shift
        conv = convolution_fft(best_a, best_a)
        k = int(np.argmax(conv))

        # indices that actually participate in shift ``k``
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_range = np.arange(i_min, i_max + 1)
        j_range = k - i_range

        # contribution = a[i] * a[j]
        contrib = best_a[i_range] * best_a[j_range]
        # pick the pair with the largest product
        idx_max = int(np.argmax(contrib))
        i_big = int(i_range[idx_max])
        j_big = int(j_range[idx_max])

        # propose a move: shift a fraction of the two large entries
        # to two random locations that are (hopefully) far from each other.
        delta_i = step_scale * best_a[i_big]
        delta_j = step_scale * best_a[j_big]

        # choose distinct target slots
        p = np.random.randint(n)
        q = np.random.randint(n)
        while p == i_big or p == j_big:
            p = np.random.randint(n)
        while q == i_big or q == j_big or q == p:
            q = np.random.randint(n)

        cand = best_a.copy()
        cand[i_big] -= delta_i
        cand[p] += delta_i
        cand[j_big] -= delta_j
        cand[q] += delta_j

        # project back onto the simplex (keeps sum = 1)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative sequence that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the 1000 s wall‑clock
    budget and logs progress to stdout.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # -------------------------------------------------
    # Choose a problem length (larger N gives more freedom)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Simple fallback (use previously discovered vector if present)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytical α‑grid start
    # -------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] best after α‑grid = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Random restarts with Adam‑mirrored exponentiated‑gradient
    # -------------------------------------------------
    while time.time() < deadline - 250.0:
        # randomise the initialisation method
        if np.random.rand() < 0.5:
            a0 = random_sparse_start(n)
        else:
            a0 = random_dirichlet_start(n)

        a_opt, sc_opt = exp_adam_optimize(
            a0,
            deadline - 180.0,
            max_iters=180_000,
            beta_start=0.5,
            beta_end=250.0,
            eta0=0.7,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after random restart = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Long dedicated Adam‑mirrored run (fine‑tune the best candidate)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] long Adam‑mirrored optimisation")
        a_smooth, sc_smooth = exp_adam_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=400_000,
            beta_start=0.5,
            beta_end=2000.0,
            eta0=0.6,
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.tolist()
            print(f"[{time.time():.1f}] after Adam‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Sub‑gradient refinement (exact max objective)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 40.0,
            max_iters=500_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Final targeted hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb_targeted(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=1_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if everything failed fall back to a uniform short vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
