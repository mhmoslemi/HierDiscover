"""
Step‑function minimiser for ``evaluate_sequence`` (≈1000 s budget).

Algorithm
---------
1. **Start** – truncated central‑binomial coefficients (a_i≈C(2i,i)/4^i)
   are a strong analytical guess.  They are projected onto the probability
   simplex (∑a=1) because the objective is scale‑invariant.

2. **LP refinement (optional)** – With the current vector *f* we compute
   M = max(conv(f,f)).  We then solve a *linear* program

        maximise    ∑ b_j
        subject to   T(f)·b ≤ M ,   b ≥ 0

   where T(f) is the (2n‑1)×n Toeplitz matrix that maps b → conv(f,b).
   The optimal partner *b* carries *more* mass than *f* without increasing
   the worst convolution entry.  Mixing f←(1‑t)f+t·b (t=0.25) provably
   reduces M.  The LP is solved with SciPy’s high‑performance “highs”
   interior‑point solver; each iteration costs O(n²) but for n≈1024 it is
   very fast, allowing dozens of reductions.

3. **Upscaling** – the improved small vector is linearly interpolated to a
   larger length (4096 or 8192) and projected onto the simplex again.
   Larger n gives a finer discretisation and a lower achievable bound.

4. **Smooth max optimisation** – we minimise a log‑sum‑exp surrogate of
   max(conv(a,a)) using Adam momentum.  The smoothing parameter β is
   annealed from 5 to 2000, the learning rate decays as 1/√t, and the
   iterate is always projected onto the simplex.  This stage drives the
   maximum of the convolution down close to the theoretical optimum.

5. **Exact sub‑gradient polishing** – after the smooth phase we switch
   to true max‑conv optimisation.  The sub‑gradient of the active entry
   is 2·a shifted appropriately; a projected sub‑gradient descent with
   a 1/√t step size cleans up the remaining overshoots.

6. **Stochastic hill‑climb** – cheap mass‑transfer moves (swap or
   fractional transfer) + simplex projection explore tiny neighbourhoods
   and occasionally escape shallow plateaus.

All stages respect the remaining wall‑clock budget and keep the best
sequence found.  The returned value is a plain Python list of non‑negative
floats (already normalised); ``evaluate_sequence`` will apply the final
clipping/summing anyway.
"""

# Core numeric utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for n up to ~10 k)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a vector that already
    lies on the probability simplex (∑a = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform fallback – the simplex requires positive mass.
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# LP utilities (stage 2)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n toeplitz matrix T such that T @ b = conv(f,b).
    Column j contains a copy of f shifted down by j rows.
    """
    n = f.shape[0]
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=float)
    for j in range(n):
        T[j:j + n, j] = f
    return T


def solve_lp_max_sum(T: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  maximise Σ b_j   s.t.   T·b ≤ M,   b ≥ 0.
    Returns the optimal b (numpy array) or ``None`` on failure.
    """
    n = T.shape[1]
    c = -np.ones(n)                     # linprog minimises; -Σb ⇔ maximise Σb
    b_ub = np.full(T.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c,
            A_ub=T,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 30.0},
        )
    except Exception:  # pragma: no cover
        return None
    if not res.success or res.x is None:
        return None
    # Numerical noise may produce tiny negatives – clip them.
    return np.maximum(res.x, 0.0)

# ----------------------------------------------------------------------


def smooth_optimize(
    a0: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 1.0,
    max_iters: int = 1_500_000,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.

    Parameters
    ----------
    a0 : np.ndarray
        Starting point (already on the simplex).
    deadline : float
        Absolute time at which the optimisation must stop.
    beta_start, beta_end : float
        Annealing range for the soft‑max temperature.
    step0 : float
        Initial learning‑rate factor (decays as 1/√t).
    max_iters : int
        Upper bound on the number of iterations (will stop earlier
        if the wall‑clock expires).

    Returns
    -------
    best_a : np.ndarray
        The best vector found.
    best_score : float
        Corresponding exact ``compute_score`` value.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first‑moment estimate
    v = np.zeros_like(a)   # second‑moment estimate

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # anneal β (soft‑max temperature)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        # soft‑max over the whole convolution vector
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()          # shape (2n‑1,)

        # gradient of the smooth surrogate:
        #   ∇_a  Σ_k w_k * conv_k = 2 * (a_rev * w)[n‑1:n‑1+n]
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # Adam update (projected onto the simplex afterwards)
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a = project_to_simplex(a)

        # occasional bookkeeping
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[smooth] iter {it:6d} – score {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimiser on the exact max‑objective
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.07,
    max_iter: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = a0.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))           # index of the active entry

        # sub‑gradient of the active entry:
        #   g_i = 2·a_{k‑i}   for indices where 0 ≤ k‑i < n
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(it)
        a = a - step * g
        a = project_to_simplex(a)

        # keep the best seen so far
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[subgrad] iter {it:6d} – score {best_score:.6f}")

        # small jitter every 1000 iterations to escape plateaus
        if it % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / math.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climbing (mass transfers)
# ----------------------------------------------------------------------


def hill_climb(
    a0: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic moves (swap or fractional transfer) with simplex
    projection, kept only while there is remaining wall‑clock budget.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_frac = 0.10   # initial fraction of mass moved in a transfer

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                         # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                               # move a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_frac * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac *= 0.97

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Upscaling utilities (small → large)
# ----------------------------------------------------------------------


def upscale(v: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear interpolation from ``len(v)`` to ``new_len`` followed by a
    simplex projection.
    """
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Central‑binomial‑coefficients initial guess
# ----------------------------------------------------------------------


def central_binomial_initial(n: int) -> np.ndarray:
    """
    Return the first ``n`` terms of the sequence a_i = C(2i,i)/4^i,
    normalised onto the simplex.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence: a_i = a_{i‑1} * (2i‑1) / (2i)
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising ``evaluate_sequence``.
    Returns the best list of floats found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin for the harness

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback (previously discovered construction)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"[fallback] score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Small‑scale LP refinement (if SciPy is available)
    # ------------------------------------------------------------------
    n_small = 1024
    a_small = central_binomial_initial(n_small)
    best_small = a_small.copy()
    best_small_score = compute_score(best_small)

    if _HAS_LP:
        lp_deadline = deadline - 300.0   # keep ~5 min for later stages
        iter_lp = 0
        while time.time() < lp_deadline:
            iter_lp += 1
            conv = convolution_fft(best_small, best_small)
            M = float(conv.max())
            T = build_conv_matrix(best_small)
            b_opt = solve_lp_max_sum(T, M)
            if b_opt is None:            # LP failed – abort this phase
                print("[LP] solve failed, exiting LP loop.")
                break

            # Normalise partner and mix
            b_opt = project_to_simplex(b_opt)
            t_mix = 0.25
            cand = (1.0 - t_mix) * best_small + t_mix * b_opt
            cand = project_to_simplex(cand)

            cand_score = compute_score(cand)
            if cand_score < best_small_score:
                best_small_score = cand_score
                best_small = cand
                print(f"[LP iter {iter_lp:3d}] improved score = {best_small_score:.6f}")

        print(f"[LP] finished after {iter_lp} iterations, best score = {best_small_score:.6f}")
    else:
        print("[LP] SciPy not available – skipping LP phase.")

    # ------------------------------------------------------------------
    # 2. Upscale to a larger discretisation
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        n_big = 8192
    else:
        n_big = 4096
    a_big = upscale(best_small, n_big)

    # ------------------------------------------------------------------
    # 3. Smooth log‑sum‑exp optimisation (Adam)
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    a_smooth, score_smooth = smooth_optimize(
        a_big,
        deadline=smooth_deadline,
        beta_start=5.0,
        beta_end=2000.0,
        step0=1.0,
        max_iters=1_200_000,
    )
    if score_smooth < best_val:
        best_val = score_smooth
        best_seq = a_smooth.tolist()
        print(f"[smooth] new best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        sub_deadline = deadline - 150.0
        a_sub, score_sub = subgradient_optimize(
            np.array(best_seq, dtype=float),
            deadline=sub_deadline,
            step0=0.07,
            max_iter=600_000,
        )
        if score_sub < best_val:
            best_val = score_sub
            best_seq = a_sub.tolist()
            print(f"[subgrad] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        hill_deadline = deadline
        a_hill, score_hill = hill_climb(
            np.array(best_seq, dtype=float),
            deadline=hill_deadline,
            max_steps=1_000_000,
        )
        if score_hill < best_val:
            best_val = score_hill
            best_seq = a_hill.tolist()
            print(f"[hill‑climb] further improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – fallback to a simple uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[final] reported score = {final_score:.6f}")
    return best_seq
