def adam_smooth_optimize(start_a, deadline, max_iters=200_000, beta_start=2.0, beta_end=1000.0, lr=0.02):
    n = len(start_a)
    eps = 1e-12
    z = np.log(start_a + eps)  # shape n

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # schedule beta
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        # compute w = softmax(beta * conv)
        maxc = conv.max()
        # subtract max to improve numeric stability
        exp_w = np.exp(beta * (conv - maxc))
        sum_exp = exp_w.sum()
        w = exp_w / sum_exp

        # gradient wrt a: grad_a = 2 * (a_rev * w) convolution?
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # backprop through softmax
        # grad_z = a * (grad_a - (a dot grad_a))
        dot_ag = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot_ag)

        # optionally adjust step size based on beta
        step_lr = lr / (1 + 0.001 * beta)   # decreasing with beta

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        z -= step_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 2000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_final = softmax(z)
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

"""
Fast multimodal optimizer for the non‑negative step‑function problem.

The search works on two scales:
  1. A small problem (n≈1024) is explored intensively with a
     smoothed‑max Adam optimiser (log‑sum‑exp surrogate).  Many
     analytic seeds, random sparse and monotone vectors are tried;
     the best shape is kept.
  2. The best small‑scale shape is interpolated to the target size
     (n≈8192) and refined again with the same Adam optimiser,
     this time using a much larger smoothing parameter so that the
     surrogate is essentially the true max.  A final symmetrisation
     and a short simulated‑annealing hill‑climb clean up the result.

All vectors are kept on the probability simplex (sum = 1) – the
objective 2·n·max(a*a) is scale‑invariant, so the final sequence can
be fed directly to ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector ``a``:
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used as seeds)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse vector (~5 % non‑zeros) on the simplex."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector on the simplex."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max optimiser (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 2.0,
                         beta_end: float = 800.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‐convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    `beta` is increased exponentially from ``beta_start`` to ``beta_end``.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise from the start vector

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- schedule for β (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- current a on the simplex -----
        a = softmax(z)

        # ----- smoothed weights (log‑sum‑exp) -----
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()                           # for numerical stability
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                     # softmax over the convolution entries

        # ----- gradient wrt a (exact) -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]    # size n

        # ----- back‑prop through soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot_ag)

        # ----- Adam update (step size mildly decays with β) -----
        step_lr = lr / (1.0 + 0.001 * beta)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= step_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional exact evaluation -----
        if it % 2000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb with simulated annealing (kept from the previous code)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex.
    – With prob 0.5 swap two entries.
    – Otherwise transfer a fraction of mass between two indices.
    Temperature decreases linearly; downhill moves are always accepted,
    uphill moves with probability exp(-Δ/T).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
