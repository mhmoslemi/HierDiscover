assistantfinal**Solution Overview**

The objective is  

\[
\text{score}(a)=\frac{2\,n\,\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

which – after normalising the sequence to a probability distribution  
\(\sum_i a_i = 1\) – reduces to minimising  

\[
\Phi(a)=2\,n\;\max_k (a*a)_k .
\]

`evaluate_sequence` already applies this normalisation internally, so the
search works on the simplex \(\{a\ge0,\;\mathbf 1^\top a =1\}\).

The previous code used only smooth‑max Adam and sub‑gradient polishing,
which got stuck at a score of **1.5114**.  
A classic way to improve the bound is the *Frank‑Wolfe / conditional‑gradient*
scheme described in the statement:

* given a current distribution \(f\) with
  \(M=\max (f*f)\) solve the linear programme  

\[
\max_{g\ge0}\;\mathbf 1^\top g
\quad\text{s.t.}\quad (f*g)_k\le M,\;k=0,\dots,2n-2 .
\tag{LP}
\]

* the optimal \(g\) has a larger \(\ell_1\)–norm while keeping the
  convolution bound unchanged.
* after normalising \(g\) back to a probability distribution we move a
  small step toward it.  
  This guarantees a non‑increasing value of \(\Phi\).

The LP is **linear** (convolution is linear in the second argument) and can
be solved with `scipy.optimize.linprog`.  
Because the constraint matrix has a Toeplitz structure we build it once
as a sparse matrix and only update the data vector (the current shape
\(f\)) on every iteration.

The final algorithm is

1. **Coarse seeding** – several analytic shapes (arcsine, power‑law,
   mixtures) on a few lengths (1024, 2048, 4096).  
   The best surrogate score is kept.
2. **Few Adam steps** – a short Adam optimisation with a smooth‑max
   surrogate, merely to escape pathological corners.
3. **LP descent** – 10–15 iterations of the LP described above, each
   followed by a mixing step (step size = 0.2).  The surrogate and the
   real score are monitored.
4. **Exponentiated‑gradient polish** – the same EG routine as before.
5. **Cheap stochastic hill climbing** – tiny mass transfers that accept
   any improvement.
6. **Official evaluation** – the best sequence found is fed to
   `evaluate_sequence`.

All stages respect a global wall‑clock deadline (default 1000 s).  
If the linear programme cannot be solved (e.g. SciPy missing) the code
falls back to the previous pure‑gradient pipeline, guaranteeing a valid

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, accurate)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()

    a_i ∝ ((i+½)*(n‑½‑i) + ε)^{-p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """
    Mixture weight_arc·arcsine + (1‑weight_arc)·uniform,
    normalised to sum = 1.
    """
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

# Optimisers used before the LP phase

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 100_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate of max(conv).
    Variables are logits w with a = softmax(w).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)            # length L

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output
        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent w.r.t. KL).
    """
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 5000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Very cheap stochastic hill climbing: move a tiny amount of mass
    from one index to another and keep improvements.
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.2 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1

    return a_cur

# Linear‑programming descent (the core new ingredient)
# ----------------------------------------------------------------------


# Cache of index arrays needed to build the Toeplitz constraint matrix
_conv_indices_cache = {}


def _conv_indices(n: int):
    """
    Return (i_idx, row_idx, col_idx) that encode the linear map
    g ↦ a * g for a given length n.
    i_idx stores the index of a for each non‑zero entry.
    """
    if n in _conv_indices_cache:
        return _conv_indices_cache[n]

    i_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    j_idx = np.tile(np.arange(n, dtype=np.int32), n)
    row_idx = i_idx + j_idx            # range 0 … 2n‑2
    col_idx = j_idx                    # column = index of g
    _conv_indices_cache[n] = (i_idx, row_idx, col_idx)
    return i_idx, row_idx, col_idx


def _lp_max_sum(a: np.ndarray, M: float):
    """
    Solve  max Σ g_j   s.t.  (a * g)_k ≤ M  (for all k), g ≥ 0.
    Returns the optimal g (numpy array) or None on failure.
    """
    if not SCIPY_AVAILABLE:
        return None

    n = a.shape[0]
    i_idx, row_idx, col_idx = _conv_indices(n)
    data = a[i_idx]                     # coefficient a_i for each entry
    A = sparse.csr_matrix((data, (row_idx, col_idx)),
                          shape=(2 * n - 1, n))

    c = -np.ones(n)                     # maximise Σg → minimise -Σg
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    # linprog may emit a warning if the problem is ill‑conditioned;
    # we simply treat a non‑optimal result as a failure.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        return None
    return res.x


def lp_descent(a: np.ndarray,
               deadline: float,
               max_iters: int = 12,
               step: float = 0.2) -> np.ndarray:
    """
    Perform a few Frank‑Wolfe steps using the LP described above.
    The function returns a (hopefully) improved distribution.
    """
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv_a = convolution_fft(a, a)
        M = float(np.max(conv_a))

        g = _lp_max_sum(a, M)
        if g is None:
            # LP failed – give up the LP refinement
            break

        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        # scale to a probability distribution
        g_scaled = g / sum_g

        # try a convex combination; if it does not improve we fall back
        # to the pure scaled solution.
        candidate = (1.0 - step) * a + step * g_scaled
        candidate /= candidate.sum()          # protect against tiny drift
        new_M = float(np.max(convolution_fft(candidate, candidate)))

        if new_M < M - 1e-12:
            a = candidate
        else:
            # pure replacement might still be better
            pure_M = float(np.max(convolution_fft(g_scaled, g_scaled)))
            if pure_M < M - 1e-12:
                a = g_scaled
            else:
                # no improvement – stop early
                break

        # debugging output
        surrogate = compute_score(a)
        print(f"    LP step {it + 1}/{max_iters}: surrogate={surrogate:.6f}")

    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine combines coarse shape seeding, a short Adam phase,
    iterative linear‑programming descent, exponentiated‑gradient polish
    and a final cheap stochastic hill climb.  All phases respect a
    global 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_official: float = float("inf")
    best_surrogate: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = fallback
                best_surrogate = compute_score(np.array(fallback))
        except Exception as e:
            print(f"Fallback evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Stage 1 – coarse shape search (very cheap)
    # ------------------------------------------------------------------
    candidate_ns = [1024, 2048, 4096]
    p_grid = np.linspace(0.2, 0.8, 9)          # power‑shape exponents
    w_grid = np.linspace(0.0, 1.0, 9)          # mixture weights

    print("\n--- Stage 1: coarse shape search ---")
    for n in candidate_ns:
        if time.time() >= deadline:
            break
        print(f"Exploring length n={n}")

        # uniform
        uni = np.full(n, 1.0 / n)
        s = compute_score(uni)
        if s < best_surrogate:
            best_surrogate, best_seq = s, uni.tolist()
            print(f"  New best (uniform) surrogate={s:.6f}")

        # arcsine
        arc = arcsine_initial(n)
        s = compute_score(arc)
        if s < best_surrogate:
            best_surrogate, best_seq = s, arc.tolist()
            print(f"  New best (arcsine) surrogate={s:.6f}")

        # power shapes
        for p in p_grid:
            shape = generate_power_shape(n, p, eps=1e-8)
            s = compute_score(shape)
            if s < best_surrogate:
                best_surrogate, best_seq = s, shape.tolist()
                print(f"  New best (p={p:.2f}) surrogate={s:.6f}")

        # mixtures arcsine / uniform
        for w in w_grid:
            mix = w * arc + (1.0 - w) * uni
            mix = mix / mix.sum()
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"  New best (mix w={w:.2f}) surrogate={s:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – short Adam refinement (smooth‑max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print("\n--- Stage 2: short Adam refinement ---")
        a = np.array(best_seq, dtype=float)
        a = adam_optimize_smooth_max(
            a,
            deadline,
            max_iters=80_000,
            tau_start=0.5,
            tau_end=150.0,
            base_lr=0.3,
        )
        s = compute_score(a)
        print(f"  After Adam surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping Adam refinement (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 3 – LP‑based Frank‑Wolfe descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and SCIPY_AVAILABLE:
        print("\n--- Stage 3: LP‑based descent ---")
        a = np.array(best_seq, dtype=float)
        a = lp_descent(a, deadline, max_iters=12, step=0.2)
        s = compute_score(a)
        print(f"  After LP surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        if not SCIPY_AVAILABLE:
            print("\nSciPy not available – skipping LP descent")
        else:
            print("\nSkipping LP descent (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 4 – exponentiated‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Stage 4: exponentiated‑gradient polish ---")
        a = np.array(best_seq, dtype=float)
        a = eg_refine(a, deadline, max_iter=120_000, eta0=0.25)
        s = compute_score(a)
        print(f"  After EG surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping EG polish (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 5 – cheap stochastic local search
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Stage 5: local random search ---")
        a = np.array(best_seq, dtype=float)
        a = local_random_search(a, deadline,
                                max_trials=30_000,
                                step_scale=0.001,
                                rng=rng)
        s = compute_score(a)
        print(f"  After local search surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping local random search (insufficient time)")

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        final_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
