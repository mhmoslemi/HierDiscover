def mirror_subgrad_optimize(start_a, deadline, step0=0.5):
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # compute gradient g
        idx = np.arange(a.shape[0])
        j = k - idx
        mask = (j >= 0) & (j < a.shape[0])
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # step size
        eta = step0 / np.sqrt(t)
        # multiplicative update
        a = a * np.exp(-eta * g)
        # renormalize
        sum_a = a.sum()

"""
Propose a step‑function sequence that minimises the evaluation objective.

Algorithm (≈ 1000 s total)
--------------------------------
1. Choose a reasonably large length (default 16384) – more degrees of freedom
   give a lower achievable max‑convolution.
2. Initialise many candidate vectors:
   * Analytic families (symmetric product, left‑heavy power‑law, exponential‑decay)
   * Random sparse / random monotone draws
   The best of these forms the starting point.
3. Smooth optimisation:
   minimise the log‑sum‑exp surrogate of the max‑convolution using Adam on the
   soft‑max parametrisation.  The smoothing parameter β is annealed
   exponentially from a modest value (≈ 5) up to a very large one (≈ 2000),
   which drives the surrogate towards the true max.
4. Mirror‑subgradient descent on the true max‑objective:
   multiplicative (exponentiated) updates on the simplex (entropy mirror map)
   with a 1/√t stepsize.  This is a cheap iteration (single FFT) and
   converges further towards the optimum.
5. Symmetrisation – averaging the vector with its reverse often reduces the
   max‑convolution.
6. Final stochastic hill‑climb with a tiny simulated‑annealing rule to escape
   shallow local minima.
7. Return the best sequence found (plain Python list of floats).

All stages respect the global wall‑clock deadline (default 1000 s).  The
function prints intermediate scores and timestamps for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a) = 1):
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    # avoid log(0) at the very ends – add 0.5 inside the product
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)  # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]  # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp) via Adam on the soft‑max layer
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially.
    """
    eps = 1e-12
    z = np.log(start_a + eps)  # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- schedule for β (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- current a ----
        a = softmax(z)

        # ---- smoothed weights w_k ∝ exp(β·conv_k) ----
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ---- gradient of f_β w.r.t. a ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # ---- back‑prop through soft‑max ----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasional exact evaluation ----
        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.8,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) sub‑gradient descent for the true objective
    2·n·max(a*a).  The step size is η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))  # active diagonal index

        # sub‑gradient of the max‑entry (active terms only)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        # exponentiated update (always stays non‑negative)
        a = a * np.exp(-eta * g)

        # renormalise to the simplex (entropy projection)
        s = a.sum()
        if s == 0.0:
            # numerical disaster – fall back to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with a tiny simulated‑annealing component
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
    – with prob 0.5 swap two entries,
    – otherwise move a fraction of mass from one index to another.
    Very low temperature allows occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule – start small, linearly cool
    T0 = best_score * init_temp_factor
    T = T0

    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # simplex projection (renormalisation)
        cand = cand / cand.sum()

        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)  # slightly enlarge moves
        else:
            step_scale *= 0.97  # shrink after rejection

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
