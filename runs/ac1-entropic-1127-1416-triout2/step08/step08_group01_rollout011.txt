def lp_step(f_vector, M=None):

    # compute M if not provided as max of conv of f

    n = len(f_vector)

    conv = convolution_fft(f_vector, f_vector)

    if M is None: M = conv.max()

    # Build A matrix of shape (2n-1, n) but using sparse representation can help. Using linear constraints A @ g <= M.

    # However, we can let scipy.optimize.linprog handle dense matrix? For n up to 4096, shape approx (8191 x 4096) ~ 33 million elements, too large for dense. Use sparse matrix.

    from scipy.sparse import csr_matrix

    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # for each j index in g vector
        # i ranges such that 0 <= i < n and 0 <= k-i < n => i in [max(0, k-n+1), min(k, n-1)]
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        # for each i, coefficient of g_{k-i} is f_i
        # So for each i, column = k - i.
        for i in range(i_min, i_max+1):
            rows.append(k)
            cols.append(k - i)
            data.append(f_vector[i])

    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # max constraint = M*ones
    b = np.full(2*n-1, M)

    # solve LP: maximize sum(g) -> minimize -sum(g) subject to A g <= b, g >= 0
    c = -np.ones(n)  # minimize -sum.

    # Using scipy.optimize.linprog (highs) with bounds (0, None)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        print("LP failed:", res.message)
        return None, None
    g = res.x
    sum_g = g.sum()
    return g, sum_g

def annealing_hill_climb(a, deadline, max_steps=10_000_000, init_T=0.001, min_T=1e-7, decay=0.9999):
   current = a.copy()
   n = a.shape[0]
   best = current.copy()
   best_score = compute_score(best)
   T = init_T

   steps = 0
   while steps < max_steps and time.time() < deadline:
       steps += 1
       # propose move
       i, j = np.random.randint(n, size=2)
       if i==j:
           continue
       # choose delta: sample fraction from uniform [0, step_scale]
       # step_scale can be set relative to current a[i] maybe.
       delta = (np.random.rand() * 0.02) * current[i] # up to 2% transfer
       if delta > current[i]:
           delta = current[i]
       cand = current.copy()
       cand[i] -= delta
       cand[j] += delta
       # ensure nonneg
       # already maintain sum unchanged.
       # Evaluate
       sc = compute_score(cand)
       delta_sc = sc - best_score
       if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / T):
           # accept
           current = cand
           if sc < best_score:
               best_score = sc
               best = cand.copy()
               # increase step scale slightly
           # update temperature
       T = max(T * decay, min_T)
   return best, best_score

"""
Optimisation of a non‑negative step‑function for the objective

    2·n·max(a*a) / (sum(a)^2)

The programme follows a multi‑stage strategy:

1. Choose a working length `n` (larger n gives a better theoretical bound,
   but is more expensive).

2. Initialise many candidate vectors by analytic families:
      – a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ
      – Beta‑type a_i ∝ (i+1)^{p‑1}(n‑i)^{q‑1}
      – a product of the above with a sinusoidal factor sin(π(i+½)/n)^γ.
   The best of these is kept.

3. Perform a number of short “Adam‑smooth” optimisation runs from random
   sparse starts.  The smooth objective
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
   is minimised by ADAM on the log‑variables (mirror‑step on the simplex).
   The temperature β is annealed from a small to a large value.

4. A long dedicated Adam run refines the current best vector.

5. A classic Euclidean sub‑gradient descent on the exact max‑objective
   (2·n·max) gives a final deterministic push.

6. A simulated‑annealing hill‑climb explores the neighbourhood of the
   current best solution, accepting occasional uphill moves with a
   decaying temperature.

All phases respect the 1000 s wall‑clock budget.  The final sequence
(plain Python list of floats) is returned.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# --- FFT utilities ---------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a vector normalised to sum = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# --- Analytic families -----------------------------------------------------

def alpha_shape(n: int, alpha: float) -> np.ndarray:
    """a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type shape a_i ∝ (i+1)^{p‑1}(n‑i)^{q‑1}."""
    i = np.arange(n, dtype=float)
    w = np.power(i + 1.0, p - 1.0) * np.power(n - i, q - 1.0)
    a = w / w.sum()
    return a


def sinusoid_alpha_shape(n: int, alpha: float, gamma: float) -> np.ndarray:
    """
    Combined shape:
        a_i ∝ [(i+½)(n‑½‑i)]⁻ᵅ · sin(π(i+½)/n)^γ
    """
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    base = np.exp(-alpha * logprod)
    theta = np.pi * (i + 0.5) / n
    sinfac = np.power(np.sin(theta) + 1e-12, gamma)   # keep positivity
    w = base * sinfac
    if w.sum() == 0:
        # fallback to uniform if numerical disaster
        return np.full(n, 1.0 / n)
    a = w / w.sum()
    return a


def analytic_scan(n: int, deadline: float) -> Tuple[np.ndarray, float]:
    """Search a modest grid of analytic families and keep the best vector."""
    best_a = None
    best_score = float("inf")

    # 1) α‑grid (same family as the previous programme)
    alphas = np.concatenate([np.linspace(-0.5, 0.0, 9),
                             np.linspace(0.0, 2.5, 21)])
    for alpha in alphas:
        if time.time() >= deadline:
            return best_a, best_score
        a = alpha_shape(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # 2) Beta‑type grid (p,q)
    ps = np.linspace(0.2, 4.0, 9)
    qs = np.linspace(0.2, 4.0, 9)
    for p in ps:
        for q in qs:
            if time.time() >= deadline:
                return best_a, best_score
            a = beta_shape(n, p, q)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # 3) Combined sinusoid‑α family
    alphas2 = np.linspace(-0.5, 2.0, 7)
    gammas = np.linspace(-2.0, 2.0, 9)
    for alpha in alphas2:
        for gamma in gammas:
            if time.time() >= deadline:
                return best_a, best_score
            a = sinusoid_alpha_shape(n, alpha, gamma)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# --- Random sparse start ---------------------------------------------------

def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    sparsity = np.random.uniform(0.03, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# --- Smoothed optimisation with ADAM ---------------------------------------

def smooth_grad_and_max(a: np.ndarray, beta: float) -> Tuple[np.ndarray, float]:
    """
    Returns (gradient of f_β, max convolution) for a given β.
    Gradient formula:
        ∇f_β = 2 * (rev(a) * w)[n‑1:n‑1+n] ,  w = softmax(β·conv)
    """
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    # soft‑max weights, stabilised
    expw = np.exp(beta * (conv - maxc))
    w = expw / expw.sum()
    # gradient via cross‑correlation
    a_rev = a[::-1]
    rev_conv = convolution_fft(a_rev, w)            # length 3n‑2
    grad = 2.0 * rev_conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return grad, maxc


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 0.5,
                         beta_end: float = 300.0,
                         eta0: float = 0.6,
                         eps: float = 1e-14) -> Tuple[np.ndarray, float]:
    """
    ADAM optimiser on the log‑variables of the smoothed max‑objective.
    The temperature β is exponentially annealed; the learning‑rate follows
    η₀/√t.  Returns the best exact score seen during the run.
    """
    # initialise log‑variables (ensure positivity)
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]
    z = np.log(a)                     # log‑parameters
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac
        a = np.exp(z)
        a /= a.sum()

        grad_a, _ = smooth_grad_and_max(a, beta)      # gradient w.r.t. a
        grad_z = grad_a * a                            # chain rule

        # ADAM moments
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        z -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation
        if it % 300 == 0:
            a = np.exp(z)
            a /= a.sum()
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    a = np.exp(z)
    a /= a.sum()
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# --- Sub‑gradient descent on the true max‑objective -------------------------

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 600_000,
                     step0: float = 0.07) -> Tuple[np.ndarray, float]:
    """Classic projected sub‑gradient method for 2·n·max(a*a)."""
    a = start_a.astype(float).copy()
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# --- Simulated‑annealing hill‑climb -----------------------------------------

def annealing_hill_climb(start_seq: np.ndarray,
                         deadline: float,
                         max_steps: int = 2_000_000,
                         init_T: float = 0.005,
                         min_T: float = 1e-7,
                         decay: float = 0.99995) -> Tuple[np.ndarray, float]:
    """
    Random walk that transfers a small amount of mass between two positions.
    Moves that improve the exact score are always kept; uphill moves are
    accepted with probability exp(-Δ/T) where T is a geometrically cooled
    temperature.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    T = init_T
    steps = 0
    while steps < max_steps and time.time() < deadline:
        steps += 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose moving up to 5 % of the mass at i
        max_delta = min(0.05 * a[i], a[i])
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # exact score
        sc = compute_score(cand)
        dsc = sc - best_sc
        if dsc < 0 or np.random.rand() < math.exp(-dsc / T):
            a = cand
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
        T = max(T * decay, min_T)

    return best_a, best_sc
