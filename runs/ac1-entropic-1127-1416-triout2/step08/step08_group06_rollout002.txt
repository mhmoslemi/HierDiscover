"""
Propose a non‑negative step sequence that minimizes the official
``evaluate_sequence`` objective.

Algorithm
---------

1. **Large‑scale parametric search** – we explore several lengths
   (up to 16384) because the arcsine‑shape limit approaches the
   theoretical optimum (≈1.5) as the discretisation gets finer.
   For each length we generate candidates from a handful of simple
   families (arcsine, symmetric beta, power‑law, gamma‑type and a
   mixture with the uniform distribution).  All candidates are
   normalised to L¹‑norm = 1 and scored with a fast FFT‑based routine
   ``compute_score`` that returns the exact official value when the
   sum equals 1.

2. **Local perturbations** – from the current global best we occasionally
   add small Gaussian noise, re‑project onto the simplex and keep any
   improvement.  This provides a cheap hill‑climbing step.

3. **Mirror‑descent polishing** – after the random phase we run a
   projected exponentiated‑gradient (KL‑mirror) optimisation on the
   true max‑convolution objective for the best sequence found, using the
   remaining time budget.

4. **Safety fallback** – if no candidate is discovered we fall back to a
   uniform distribution of modest length.

The routine respects a 1000 s wall‑clock limit, prints progress information,
and finally returns the best sequence (as a plain Python list) found.
"""

# ----------------------------------------------------------------------
# Imports and low‑level utilities
# ----------------------------------------------------------------------
import time
from typing import List
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose entries sum to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (all normalise to L¹‑norm = 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Symmetric (or not) Beta‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape – the denominator is raised to p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_gamma_shape(n: int, gamma: float, delta: float = 0.5,
                         eps: float = 0.0) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{-γ}.
    The classic arcsine corresponds to γ=0.5, δ≈0.5, ε≈0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the mirror‑descent polish
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true max‑convolution
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    step: float = 0.015,
) -> np.ndarray:
    """
    Mirror‑descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # back‑propagation of the sub‑gradient through the max
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # multiplicative update
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d}  score={compute_score(a):.6f}")

    return a
