"""
Propose a step sequence with a low value of `evaluate_sequence`.

Approach
--------
1.  Build a very diverse pool of analytic “seed” shapes for a given length:
    * arcsine envelope (the classic start‑point)
    * pure uniform
    * power‑law shapes   a_i ∝ 1/((i+½)(n‑½‑i)+ε)^p   with many p‑values
    * beta‑distribution shapes  a_i ∝ (i+ε)^{α‑1}(n‑1‑i+ε)^{β‑1}
    * mixtures of the arcsine envelope and the uniform vector
    * an up‑sampled copy of the current best sequence (if any)

   The families are explored on a dense grid (not just a few random draws),
   because the objective is extremely sensitive to the shape parameters.

2.  Rank the seeds cheaply with the FFT‑based surrogate
        score = 2·n·max(conv(a,a))          (a already normalised, sum=1)
    and keep the best few (≈8) for further optimisation.

3.  For each kept seed run a *smooth‑max Adam* optimiser.
    - The loss is a log‑sum‑exp approximation of the true max‑convolution.
    - Parameters are the log‑softmax vector `z`; after each Adam step the
      probability vector `a = softmax(z)` is **symmetrised**
      (`a = (a + a[::-1])/2`) and renormalised – symmetry is known to be
      optimal for the autocorrelation problem.
    - The temperature β is increased linearly from a small start to a large
      final value (up to 2000) so that the surrogate gradually approaches the
      true max.
    - A modest learning‑rate schedule and Adam‘s moments are used.

4.  The refined candidate is evaluated with the official (slow) `evaluate_sequence`.
    If it improves the best score it becomes the new incumbent.

5.  Whenever time permits a short **KL‑mirror (exponentiated‑gradient) polish**
    on the true objective is performed.  Finally a longer **projected
    sub‑gradient descent** is applied to the global best.

All heavy work respects the global 1000 s wall‑clock budget and the routine
always returns a valid (non‑empty) sequence.

# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate for evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry (a_i = a_{n-1-i})."""
    return (v + v[::-1]) * 0.5

# Analytic seed families

def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups

def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # simplex point
        a = symmetrize(a)                  # keep the sequence symmetric
        a /= a.sum()                       # re‑normalise (tiny drift possible)

        conv = convolution_fft(a, a)       # linear conv, length 2n‑1
        mx = float(np.max(conv))

        # temperature β(t)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)         # weights, sum to 1 after normalisation
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central n entries

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output
        if t % 5_000 == 0:
            print(f"    Adam iter={t:,d} β={beta:.0f} surrogate={compute_score(a):.6f}")

    a = softmax(z)
    a = symmetrize(a)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Mirror descent (KL) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # active set – indices where conv attains (almost) the max
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev) (restricted)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5_000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.09,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5_000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Helper: build a pool of diverse seeds for a given length
# ----------------------------------------------------------------------


def build_seed_pool(
    n: int,
    rng: np.random.Generator,
    best_sofar: List[float],
) -> List[np.ndarray]:
    """Return a list of candidate probability vectors of length n."""

    seeds: List[np.ndarray] = []
    # Classic baselines
    seeds.append(arcsine_initial(n))
    seeds.append(np.full(n, 1.0 / n))

    # Power‑law grid
    for p in np.linspace(0.1, 2.5, 20):
        seeds.append(generate_power_shape(n, p, eps=1e-12))

    # Beta‑grid (symmetric)
    for a in np.linspace(0.05, 0.8, 16):
        seeds.append(generate_beta_shape(n, a, a, eps=1e-12))

    # Asymmetric beta draws (few random choices)
    for _ in range(6):
        a = rng.uniform(0.05, 1.5)
        b = rng.uniform(0.05, 1.5)
        eps = rng.uniform(0.0, 0.5)
        seeds.append(generate_beta_shape(n, a, b, eps))

    # Mixtures of arcsine and uniform
    for w in np.linspace(0.0, 1.0, 7):
        seeds.append(generate_mixture_shape(n, w))

    # Upsampled version of the current best (if any)
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        seeds.append(upsample_sequence(old, n))

    # Random perturbations of the best (if any) – keep diversity
    if best_sofar:
        base = np.array(best_sofar, dtype=float)
        base = upsample_sequence(base, n)
        for _ in range(4):
            noise = rng.normal(scale=0.02, size=n)
            pert = np.clip(base + noise, 0.0, None)
            pert /= pert.sum()
            seeds.append(pert)

    # Ensure all candidates are valid probabilities
    clean = []
    for v in seeds:
        v = np.clip(v, 1e-15, None)
        v /= v.sum()
        clean.append(v)
    return clean


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 43,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low
    `evaluate_sequence` value.  The routine respects a 1000 s wall‑clock
    budget and always returns a non‑empty valid sequence.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a known good sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Run the optimisation on a cascade of increasing lengths.
    # ------------------------------------------------------------------
    lengths = [16384, 8192, 4096, 2048]          # larger lengths give a better limit

    for n in lengths:
        if time.time() >= deadline - 20.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ---- 1a. Build a large and diverse seed pool -----------------
        seed_pool = build_seed_pool(n, rng, best_seq)

        # ---- 1b. Quick surrogate ranking (no optimisation yet) -----
        scored = [(compute_score(s), s) for s in seed_pool]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:8]]   # keep the most promising

        # ---- 1c. Time allocation per seed ---------------------------
        remaining = deadline - time.time()
        per_seed_time = max(8.0, remaining / (len(top_seeds) * 2.0))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 10.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_time:.1f}s")
            seed_deadline = min(time.time() + per_seed_time, deadline - 5.0)

            # initialise Adam on the log‑softmax of the seed
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            # ---------------------------------------------------------
            # 2. Smooth‑max Adam optimisation (long run)
            # ---------------------------------------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=2000.0,
                ramp_iters=80_000,
                lr=0.09,
            )

            # ---------------------------------------------------------
            # 3. Official evaluation (slow)
            # ---------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ---------------------------------------------------------
            # 4. Optional KL‑mirror polish if time permits
            # ---------------------------------------------------------
            if time.time() < deadline - 6.0:
                polish_deadline = deadline - 3.0
                a_polish = exponentiated_gradient_polish(
                    np.array(best_seq, dtype=float),
                    polish_deadline,
                    max_iter=120_000,
                    step=0.018,
                )
                try:
                    val2 = evaluate_sequence(a_polish.tolist())
                    print(f"   Polished (EG) score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, a_polish.tolist()
                        print(f"   *** New best after EG polish: {best_val:.6f} ***")
                except Exception as e:
                    print(f"   EG polishing failed: {e}")

    # ------------------------------------------------------------------
    # 5. Final global polishing (exact sub‑gradient)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\nFinal global sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=300_000,
            step0=0.09,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 6. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
