"""
Non‑negative step‑function optimiser for `evaluate_sequence`.

Approach
--------
* Initialise a large vector (~8192 entries) with the classic
  central‑binomial coefficients (a_i = C(2i,i)/4^i) projected onto the
  probability simplex (∑a_i = 1).

* Perform several *restarts* of a smooth log‑sum‑exp optimisation (Adam
  with projection).  The smoothing parameter β is exponentially annealed
  from a modest value (≈5) up to a very large one (≈2000) so that the
  surrogate converges to the true L∞‑objective.  Each restart receives a
  short time‑budget and a slight random perturbation of the start vector.

* Refine the best candidate with a projected sub‑gradient descent on the
  exact maximum‑convolution objective.

* Finish with a cheap stochastic hill‑climb that swaps or transfers a
  fraction of mass between positions while keeping the vector on the
  simplex.

All stages respect the overall wall‑clock budget (≈1000 s).  The function
`propose_candidate` returns the best list of floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (real‑valued inputs)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1:
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    """
    if np.all(v <= 0):
        # uniform distribution is the projection of the non‑positive orthant
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisation
# ----------------------------------------------------------------------


def central_binomial_vector(n: int) -> np.ndarray:
    """
    Generate the (truncated) central‑binomial coefficients
       a_i = C(2i,i)/4^i  , i = 0…n‑1
    and project onto the simplex (∑a_i = 1).
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2.0 * i + 1.0) / (2.0 * (i + 1.0))
    a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 400_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth surrogate

        φ_β(a) = (1/β) log Σ_k exp(β·conv_k(a,a))

    When β → ∞, φ_β → max(conv).  The routine runs until `time_deadline`
    and returns the best vector encountered (projected onto the simplex)
    together with its exact score (2·n·max(conv)).
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)            # 1st moment
    v = np.zeros_like(a)            # 2nd moment
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β exponentially
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable log‑sum‑exp weighting
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smooth objective:
        # ∇_a φ = 2 * (a_rev * w)  restricted to the original length
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update with bias‑correction
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # periodic exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    The sub‑gradient at the active shift k is 2·a shifted by (k‑i).
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))               # active shift

        # sub‑gradient of entry k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to shake plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 800_000,
) -> tuple[list[float], float]:
    """
    Tiny stochastic moves (swap or fractional transfer) with simplex
    projection, keeping the best vector seen.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12  # initial fraction of mass to relocate

    steps_done = 0
    while steps_done < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                # protect against tiny negatives caused by rounding
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps_done += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name‑defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"[fallback] score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem size based on remaining budget
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 720:
        n = 8192
    elif remaining > 400:
        n = 4096
    else:
        n = 2048
    print(f"[setup] using length n = {n}")

    # ------------------------------------------------------------------
    # 2. Initialise with central‑binomial coefficients (probability simplex)
    # ------------------------------------------------------------------
    a0 = central_binomial_vector(n)
    init_score = compute_score(a0)
    print(f"[init] score = {init_score:.6f}")

    # record as current best
    if init_score < best_val:
        best_val = init_score
        best_seq = a0.tolist()

    # ------------------------------------------------------------------
    # 3. Smooth log‑sum‑exp Adam optimisation – several restarts
    # ------------------------------------------------------------------
    # allocate ~70 % of the remaining budget to this phase
    grad_phase_deadline = deadline - 250.0
    restart_budget = max(0.0, grad_phase_deadline - time.time())
    n_restarts = 3
    if n_restarts > 0:
        per_restart = restart_budget / n_restarts
    else:
        per_restart = 0.0

    for r in range(n_restarts):
        if time.time() >= grad_phase_deadline:
            break
        # perturb the start vector a little to escape possible local minima
        a_start = a0.copy()
        noise = np.random.randn(n) * 0.008
        a_start += noise
        a_start = np.maximum(a_start, 0.0)
        a_start = project_to_simplex(a_start)

        print(f"[grad‑restart {r+1}/{n_restarts}] starting optimisation...")
        a_opt, scr = smooth_gradient_optimize(
            a_start,
            time_deadline=time.time() + per_restart,
            max_iters=500_000,
            beta_start=5.0,
            beta_end=2000.0,
            step0=0.8,
        )
        print(f"    -> restart {r+1} score = {scr:.6f}")

        if scr < best_val:
            best_val = scr
            best_seq = a_opt.tolist()
            # keep the refined vector as the new seed for subsequent restarts
            a0 = a_opt.copy()

    # ------------------------------------------------------------------
    # 4. Exact sub‑gradient refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("[sub‑grad] starting refinement")
        sub_deadline = deadline - 80.0
        a_best_arr = np.array(best_seq, dtype=float)
        a_ref, scr_ref = subgrad_optimize(
            a_best_arr,
            time_deadline=sub_deadline,
            max_iter=600_000,
            step0=0.07,
        )
        print(f"    sub‑grad score = {scr_ref:.6f}")
        if scr_ref < best_val:
            best_val = scr_ref
            best_seq = a_ref.tolist()

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climb (tight‑time polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[hill‑climb] polishing")
        final_seq, final_score = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=900_000,
        )
        print(f"    hill‑climb score = {final_score:.6f}")
        if final_score < best_val:
            best_val = final_score
            best_seq = final_seq

    # ------------------------------------------------------------------
    # Safeguard – if nothing succeeded, fall back to a tiny uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # final evaluation (the harness will also evaluate, this is just for logging)
    final_report = evaluate_sequence(best_seq)
    print(f"\n[final] reported score = {final_report:.6f}")

    # Return plain Python list of floats
    return best_seq
