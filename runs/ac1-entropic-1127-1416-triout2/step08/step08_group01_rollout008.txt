    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast and accurate)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector on the simplex (∑a = 1).
    This is the value returned by evaluate_sequence when the sum is 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation based on the algorithm by Wang & Carreira‑Perpiñán (2013).
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytical α‑family start
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    # Avoid overflow/underflow – work in log‑space then exponentiate.
    w = np.exp(-alpha * logprod)
    # Numerical safety for extreme α.
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> Tuple[np.ndarray, float]:
    """Search a fine α‑grid (including negatives) and keep the best vector."""
    best_a = None
    best_score = float("inf")
    # finer grid – more points near the previously good region (0 … 4)
    alphas = np.concatenate([
        np.linspace(-1.0, 0.0, 11),      # -1…0
        np.linspace(0.0, 4.0, 41)       # 0…4 (step 0.1)
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Random starts
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector (uniform on a random support)."""
    sparsity = np.random.uniform(0.03, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dense_dirichlet_start(n: int, concentration: float = 0.4) -> np.ndarray:
    """Dense Dirichlet‑type start, concentration < 1 gives heavy‑tailed vectors."""
    # Gamma(shape=concentration, scale=1) → Dirichlet when normalised.
    g = np.random.gamma(concentration, 1.0, size=n)
    # Guard against all‑zero draws (practically impossible for concentration>0)
    if g.sum() == 0:
        g = np.ones_like(g)
    a = g / g.sum()
    return a


# ----------------------------------------------------------------------
# 3. Exponentiated‑gradient optimisation (smooth max)
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        beta_start: float = 0.5,
                        beta_end: float = 300.0,
                        eta0: float = 0.6,
                        eps: float = 1e-14) -> Tuple[np.ndarray, float]:
    """
    Minimises the smooth log‑sum‑exp approximation of the max‑convolution
    using a KL‑mirror (exponentiated‑gradient) update.
    Returns the best exact (unsmoothed) score observed.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- temperature schedule (exponential) ---------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # --- smooth max via log‑sum‑exp -----------------------------------------
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))        # stabilised
        w = expw / expw.sum()                      # softmax weights

        # --- gradient of the smooth objective ------------------------------------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)      # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]     # length n

        # --- step size -----------------------------------------------------------
        eta = eta0 / np.sqrt(it)

        # --- exponentiated‑gradient update (KL‑projection) -----------------------
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # --- occasional exact evaluation -----------------------------------------
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Projected Euclidean sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 600_000,
                     step0: float = 0.07) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method for f(a)=2·n·max(a*a) with Euclidean projection
    onto the simplex.  The step size follows ηₜ = step0 / √t.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of max at position k:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 5. Aggressive stochastic hill‑climb (with simulated‑annealing flavour)
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 1_200_000,
               init_scale: float = 0.12) -> Tuple[np.ndarray, float]:
    """
    Random walk that tries three move types:
      – swap two entries,
      – transfer a fraction of mass from i to j,
      – redistribute a tiny packet among three distinct indices.
    Moves that strictly improve the exact score are always accepted.
    Occasionally a non‑improving move is accepted with probability
    exp(−Δ/T) where the temperature T decays linearly.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = init_scale
    # linear temperature schedule
    T0 = 1e-3
    Tmin = 1e-7

    steps_done = 0
    while steps_done < max_steps and time.time() < deadline:
        # temperature linearly decreasing from T0 to Tmin
        T = T0 + (Tmin - T0) * (steps_done / max_steps)

        cand = best_a.copy()
        move_type = np.random.rand()
        if move_type < 0.4:                               # 40 % – swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.8:                             # 40 % – transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        else:                                             # 20 % – three‑way mix
            i, j, k = np.random.choice(n, size=3, replace=False)
            total = best_a[i] + best_a[j] + best_a[k]
            # draw a fresh Dirichlet (α=1) on the three coordinates
            new_weights = np.random.gamma(1.0, 1.0, size=3)
            new_weights /= new_weights.sum()
            cand[i] = total * new_weights[0]
            cand[j] = total * new_weights[1]
            cand[k] = total * new_weights[2]

        # keep on the simplex (round‑off may break it slightly)
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        # acceptance rule
        if sc < best_sc:
            best_sc = sc
            best_a = cand
            # slowly increase step‑scale when things go well
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # simulated‑annealing acceptance of worsening moves
            if np.random.rand() < np.exp(-(sc - best_sc) / max(T, 1e-12)):
                best_a = cand            # we move the current state
                # do *not* update best_sc (it stays the best seen so far)
            # otherwise reject
            step_scale *= 0.97           # slowly shrink step‑scale

        steps_done += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# 6. Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Entry point used by the evaluation harness.
    Returns a list of non‑negative floats that (hopefully) yields a low
    evaluate_sequence() score.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin so we never run over the hard limit
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a length – 8192 gives the best theoretical room,
    # but fall back if time is already scarce.
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 8192
    elif remaining > 500:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] using length n = {n}")

    best_seq: List[float] = []
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Fallback: possibly a previously discovered construction
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytical start (fine α‑grid)
    # -------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] after α‑grid best = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Random restarts – a mixture of sparse and dense starts
    # -------------------------------------------------
    # We allocate roughly the first half of the remaining budget to restarts.
    while time.time() < deadline - 250.0:
        # alternate between sparse and dense initialisers
        if np.random.rand() < 0.5:
            a0 = random_sparse_start(n)
        else:
            a0 = random_dense_dirichlet_start(n, concentration=0.4)
        a_opt, sc_opt = exp_smooth_optimize(
            a0,
            deadline - 180.0,
            max_iters=180_000,
            beta_start=0.5,
            beta_end=400.0,
            eta0=0.7,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after random restart = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Long dedicated exponentiated‑gradient optimisation
    # -------------------------------------------------
    if best_seq and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] long exponentiated‑gradient optimisation")
        a_long, sc_long = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=450_000,
            beta_start=0.5,
            beta_end=800.0,          # higher final temperature → tighter max‑approx
            eta0=0.6,
        )
        if sc_long < best_score:
            best_score = sc_long
            best_seq = a_long.tolist()
            print(f"[{time.time():.1f}] after long exp‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Sub‑gradient refinement (Euclidean projection)
    # -------------------------------------------------
    if best_seq and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=float),
            deadline - 40.0,
            max_iters=700_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Final hill‑climb polishing (with SA flavour)
    # -------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=2_000_000,      # more steps than the original code
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # 6. Fallback to a tiny uniform vector if everything failed
    # -------------------------------------------------
    if not best_seq:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # 7. Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
