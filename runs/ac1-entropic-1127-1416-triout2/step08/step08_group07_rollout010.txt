"""
Propose a step‑function (a non‑negative sequence) that minimises
``evaluate_sequence``.  The search combines a very rich family of
analytic seeds (arcsine, power‑law, Beta, mixtures, Dirichlet draws)
with a fast FFT‑based surrogate ``fast_score`` to prune candidates.
The most promising seeds are refined by a continuation Adam optimiser
on the smooth‑max surrogate, followed by an exact mirror‑descent
sub‑gradient step and a short projected sub‑gradient polish.
A final cheap random local search explores the neighbourhood of the
best solution.  All work respects the 1000 s time budget.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper: FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate used for pre‑selection (scale‑invariant)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Return the value of the objective 2·n·max(conv(a,a))/sum(a)²
    without the clipping performed by ``evaluate_sequence``.
    ``a`` must be non‑negative.
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic families of candidate shapes
# ----------------------------------------------------------------------


def arcsine_shape(n: int, eps: float = 0.25) -> np.ndarray:
    """Discrete arcsine‑type density 1/√((i+½)(n‑½‑i)+eps)."""
    i = np.arange(n, dtype=float)
    denom = np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = 1.0 / denom
    vals /= vals.sum()
    return vals


def power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Shape proportional to 1/(((i+½)(n‑½‑i)+eps)^p)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    vals /= vals.sum()
    return vals


def mixture_arc_uniform(n: int, w: float, eps: float = 0.25) -> np.ndarray:
    """Mixture w·arcsine + (1‑w)·uniform."""
    return w * arcsine_shape(n, eps) + (1.0 - w) * (np.full(n, 1.0 / n))


def beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-12) -> np.ndarray:
    """Beta(α,β) density on {0,…,n‑1} (discretised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    vals = left * right
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with random concentration)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)          # between 0.01 and 5
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx_i = 1, x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) optimiser (Adam with continuation on β)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 5000.0,
    ramp_iters: int = 120_000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    Minimise the smooth approximation L_β(a)=log‑sum‑exp(β·conv(a,a)),
    enforcing a∈simplex via a soft‑max parametrisation.
    β is increased linearly from ``beta0`` to ``beta_max``.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = np.max(conv)

        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of log‑sum‑exp w.r.t. a:
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice length n

        # chain‑rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional cheap progress report
        if t % 8_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponential‑gradient) on the true max‑convolution
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
    step0: float = 0.5,
    decay: float = 0.9999,
) -> np.ndarray:
    """
    Multiplicative mirror descent (KL‑mirror) on f(a)=2·n·max(conv(a,a))/sum(a)².
    The iterate stays on the simplex automatically.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        factor = np.exp(-step * subgrad)
        a = a * factor
        a_sum = a.sum()
        if a_sum == 0.0 or not np.isfinite(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        step *= decay
        it += 1

        if it % 5_000 == 0:
            print(f"    Mirror step={it:,d} surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent using the *exact* max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent for the official objective."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 20_000 == 0:
            print(f"    Sub‑grad it={it:,d} surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Simple random local search around a candidate
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.025,
) -> np.ndarray:
    """
    Pairwise mass‑transfer hill‑climbing.  Each move transfers a random
    fraction (≤ max_transfer) of the mass at i to j, keeping the vector
    on the simplex.  Moves are kept only if they improve the cheap
    surrogate ``fast_score``.
    """
    a = a0.copy()
    best = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * rng.random() * max_transfer
        if eps <= 0.0:
            continue
        a[i] -= eps
        a[j] += eps
        # renormalise (numerical safety)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
            continue
        a /= s
        cur = fast_score(a)
        if cur < best:
            best = cur
        else:
            # revert
            a[j] -= eps
            a[i] += eps
            a /= a.sum()
    return a

    global_promising.sort(key=lambda x: x[0])
    top_candidates = global_promising[:20]

    print(f"\n=== {len(top_candidates)} seeds selected for full optimisation ===")

    # ------------------------------------------------------------------
    # 2. Refine each selected seed
    # ------------------------------------------------------------------
    for idx, (sur, init_vec, n) in enumerate(top_candidates, start=1):
        if time.time() >= deadline - 4.0:
            break
        print(f"\n--- Refining candidate {idx}/{len(top_candidates)} (n={n}) ---")
        per_deadline = time.time() + (deadline - time.time()) / (len(top_candidates) - idx + 1)

        # --------------------------------------------------------------
        # 2a. Adam on smooth‑max (continuation on β)
        # --------------------------------------------------------------
        eps = 1e-12
        z0 = np.log(init_vec + eps)
        a_adam = smooth_adam_optimize(
            z0,
            per_deadline,
            n=n,
            beta0=5.0,
            beta_max=5000.0,
            ramp_iters=int(1e5),
            lr=0.04,
        )

        # --------------------------------------------------------------
        # 2b. Mirror‑descent exact max optimisation (short run)
        # --------------------------------------------------------------
        md_deadline = min(deadline - 0.5, time.time() + 0.65 * (per_deadline - time.time()))
        a_mirror = mirror_descent_optimize(
            a_adam,
            md_deadline,
            max_iter=20_000,
            step0=0.5,
            decay=0.9999,
        )

        # --------------------------------------------------------------
        # 2c. Very short exact sub‑gradient polish (optional)
        # --------------------------------------------------------------
        sg_deadline = min(deadline - 0.3, time.time() + 0.15 * (per_deadline - time.time()))
        a_polished = subgradient_descent_exact(
            a_mirror,
            sg_deadline,
            max_iter=30_000,
            step0=0.05,
        )

        # --------------------------------------------------------------
        # 2d. Official evaluation
        # --------------------------------------------------------------
        try:
            val = evaluate_sequence(a_polished.tolist())
            print(f"   Official score = {val:.6f} (surrogate≈{fast_score(a_polished):.6f})")
            if val < best_official:
                best_official, best_seq = val, a_polished.tolist()
                best_surrogate = fast_score(a_polished)
                print("   *** New global best! ***")
        except Exception as e:
            print(f"   Evaluation error: {e}")

    # ------------------------------------------------------------------
    # 3. Final polishing of the globally best solution (if any)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\n=== Final polishing of global best ===")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=200_000,
            step0=0.04,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_official:
                best_official, best_seq = final_val, a_arr.tolist()
                best_surrogate = fast_score(a_arr)
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final evaluation error: {e}")

    # ------------------------------------------------------------------
    # 4. Random local search around the polished solution
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:
        print("\n=== Random local search around current best ===")
        a_start = np.array(best_seq, dtype=float)
        a_refined = random_local_search(
            a_start,
            deadline,
            rng,
            max_transfer=0.025,
        )
        try:
            val_loc = evaluate_sequence(a_refined.tolist())
            print(f"   Local‑search official score = {val_loc:.6f}")
            if val_loc < best_official:
                best_official, best_seq = val_loc, a_refined.tolist()
                best_surrogate = fast_score(a_refined)
                print("*** Best improved by local search ***")
        except Exception as e:
            print(f"   Local‑search evaluation error: {e}")

    # ------------------------------------------------------------------
    # 5. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
