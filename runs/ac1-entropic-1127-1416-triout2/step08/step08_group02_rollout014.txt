def build_conv_matrix_dense(f):
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i, fi in enumerate(f):

def solve_lp_max_sum_dense(A, M, warm_start=None):
    n = A.shape[1]
    c = -np.ones(n) # maximize sum b → minimize -sum b
    bounds = [(0, None) for _ in range(n)]
    # Constraints: A @ b <= M
    # linprog expects A_ub * x <= b_ub
    res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M), bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        return None

def solve_lp_max_sum_dense(A, M, a):
    n = A.shape[1]
    c = -np.ones(n)
    bounds = []
    eps = 1e-12
    for i in range(n):
        if a[i] <= eps:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((0.0, None))
    res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

"""
Non‑negative step‑function minimiser for `evaluate_sequence`.

Algorithm
---------
1. **Initial shape** – central‑binomial coefficients (a_i = C(2i,i)/4^i)
   truncated to a moderate length (n≈2048) and projected onto the simplex.

2. **Linear‑programming refinement** – repeatedly
      maximise Σ b_j   s.t.   (a * b)_k ≤ M   (M = max(a*a)),
   where the convolution matrix is built explicitly (dense).  The LP is
   solved with `scipy.optimize.linprog` (high‑s).  The resulting partner
   `b` is projected onto the simplex and mixed with the current `a`.
   An inexpensive 1‑D line‑search on the mixing weight α minimises the
   maximal convolution of the mixture.  The loop runs as long as time
   permits (≈20–30 iterations).

3. **Upscaling** – the improved small vector is linearly interpolated to a
   larger length (4096 or 8192) and re‑projected onto the simplex.

4. **Smooth max optimisation** – Adam optimiser on a log‑sum‑exp (soft‑max)
   surrogate of the max‑convolution.  The smoothing parameter β is
   annealed from 5 to 200 and the learning‑rate decays as 1/√t.

5. **Projected sub‑gradient polishing** – direct sub‑gradient descent on
   the exact objective `2·n·max(conv(a,a))`.

6. **Stochastic hill‑climb** – cheap local moves (swap or fractional mass
   transfer) with simplex projection and a simple acceptance rule.

All stages respect the remaining wall‑clock budget and keep the best
sequence found.  The final result is a plain Python list of non‑negative
floats (any scaling is harmless because the objective is homogeneous).
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# LP utilities
# ----------------------------------------------------------------------


def build_convolution_matrix_dense(f: np.ndarray) -> np.ndarray:
    """
    Build dense matrix A (size (2n‑1)×n) such that A @ b = conv(f,b).
    Column i contains the vector f shifted down by i rows.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i, fi in enumerate(f):
        if fi != 0.0:
            A[i:i + n, i] = fi
    return A


def solve_lp_max_sum_dense(A: np.ndarray, M: float, a: np.ndarray) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Variables whose corresponding coefficient in `a` is effectively zero are
    forced to stay zero to avoid an unbounded LP.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = A.shape[1]
    c = -np.ones(n)                     # maximise sum(b)  → minimise -sum(b)
    # Upper‑bound vector for the constraints
    b_ub = np.full(A.shape[0], M, dtype=np.float64)

    # Bounds per variable: zero if a[i] is (almost) zero.
    eps = 1e-12
    bounds = [(0.0, 0.0) if a[i] <= eps else (0.0, None) for i in range(n)]

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 30},
        )
    except Exception:
        return None

    if not res.success:
        return None
    bv = res.x
    if bv is None:
        return None
    # Clip any tiny negatives caused by numerical noise
    return np.maximum(bv, 0.0)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    step0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Uses a log‑sum‑exp surrogate with temperature β annealed from
    `beta_start` to `beta_end`.
    Returns the best vector found and its (unsmoothed) score.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)   # first‑moment estimate
    v = np.zeros_like(a)   # second‑moment estimate

    best_a = a.copy()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smooth objective (cross‑correlation)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1 : n - 1 + n]

        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)

        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Adam] iter {it:6d} – best score = {best_score:.6f}")

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (the k‑th convolution coefficient)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()
            print(f"[Sub‑grad] iter {t:6d} – new best = {best_score:.6f}")

        a = a_new

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """Simple stochastic hill‑climb with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling (linear interpolation) utility
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)
