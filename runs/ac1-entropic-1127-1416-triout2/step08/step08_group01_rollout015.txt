def hill_climb_sa(start_a, deadline, max_steps=1_000_000):
    a_curr = start_a.copy()
    best_a = a_curr.copy()
    best_sc = compute_score(best_a)

    # Temperature schedule: start_temp = (best_sc - lower_bound_estimate) maybe best_sc*0.01?
    # Let's set start_temp = best_sc * 0.02 (2% of score) maybe.

    temp = best_sc * 0.02
    decay = 0.99995  # decays each step.

    step_scale = 0.1

    while time.time() < deadline and max_steps > 0:
        # Propose move: we can do targeted move with probability.
        if np.random.rand() < 0.7:
            # Targeted move: reduce max conv.
            conv = convolution_fft(a_curr, a_curr)
            k_max = int(np.argmax(conv))
            # Find contributions as product above.
            i_range = np.arange(max(0, k_max - (n-1)), min(n-1, k_max) + 1)
            # contributions = a[i_range] * a[k_max-i_range]
            contrib = a_curr[i_range] * a_curr[k_max - i_range]
            # choose high contribution index i1 (with max contrib)
            idx_high = i_range[np.argmax(contrib)]
            # choose low contribution index i2 maybe random among others with low a.
            # pick random index not equal to idx_high.
            i2 = np.random.randint(n)
            while i2 == idx_high:
                i2 = np.random.randint(n)
            # transfer fraction from idx_high to i2.
            delta = step_scale * a_curr[idx_high]
            cand = a_curr.copy()
            cand[idx_high] -= delta
            cand[i2] += delta
        else:
            # Random swap.
            i, j = np.random.randint(n, size=2)
            cand = a_curr.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # Project to simplex
        cand = project_to_simplex(cand)

        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            a_curr = cand.copy()
            # reduce temperature maybe.
            temp *= 0.995

def linear_program_step(a, M, deadline):

    # Using diagonal offset trick.

    cols = []
    data = []
    # for offset d in range(n):
    #   diag entries value a[d] on rows [d, d+n-1] with cols 0..n-1.
    for d in range(n):
        # row indices for column i are i + d
        # value = a[d]

        rows.append(idxs)
        cols.append(np.arange(n))

    cols = np.concatenate(cols)
    data = np.concatenate(data)

    c = -np.ones(n)

    # inequality: A @ b <= M
    b_ub = np.full(2*n-1, M)

    # bounds for b: (0, None)
    bounds = [(0, None)] * n

    if not res.success:
        return None, None
    b = res.x
    total = b.sum()
    if total <= 0:
        return None, None
    g = b / total
    return g, total

def hill_climb_sa(start_a, deadline, max_steps=1_000_000):
    a_curr = start_a.copy()
    best_a = a_curr.copy()
    best_sc = compute_score(best_a)

    n = a_curr.size
    # initial temperature based on best_sc
    temp = best_sc * 0.02 + 1e-6
    decay = 0.9999

    step_scale = 0.1

    # precompute range for random moves
    while max_steps > 0 and time.time() < deadline:
        # random decision:
        if np.random.rand() < 0.6:
            # targeted mass transfer from high contribution to low
            conv = convolution_fft(a_curr, a_curr)
            k_max = int(np.argmax(conv))
            # i_range:
            i_low = max(0, k_max - (n-1))
            i_high = min(n-1, k_max)
            idxs = np.arange(i_low, i_high+1)
            # contributions:
            contrib = a_curr[idxs] * a_curr[k_max - idxs]
            # choose high contrib index:
            high_idx = idxs[np.argmax(contrib)]
            # choose low contribution index randomly (could be any index not high):
            low_idx = np.random.randint(n)
            while low_idx == high_idx:
                low_idx = np.random.randint(n)
            delta = step_scale * a_curr[high_idx]
            cand = a_curr.copy()
            cand[high_idx] -= delta
            cand[low_idx] += delta
        else:
            # random swap
            i, j = np.random.randint(n, size=2)
            cand = a_curr.copy()
            cand[i], cand[j] = cand[j], cand[i]
        # project
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        # acceptance
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            a_curr = cand.copy()
            # maybe decrease temperature faster because improvement
            temp *= 0.99
        else:
            # accept with probability
            if np.random.rand() < np.exp((best_sc - sc) / max(temp, 1e-12)):
                a_curr = cand.copy()
        # decaying temperature
        temp *= decay
        step_scale = max(0.001, step_scale * 0.99999)
        max_steps -= 1

    return best_a, best_sc

"""
Improved search for a step‑function minimizing

    evaluate_sequence(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm combines

* a fine‑grained analytical start (α‑family + local refinement),
* exponentiated‑gradient optimisation of the log‑sum‑exp smooth max,
* a sub‑gradient method with back‑tracking line search,
* a simulated‑annealing hill‑climber that performs targeted mass
  transfers from the positions that contribute most to the current peak.

All phases respect a global 1000 s wall‑clock limit.  The routine returns
a plain ``list[float]`` that the harness will feed to ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from typing import Tuple, List

# ----------------------------------------------------------------------
# Core linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using a real FFT.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = rfft(a, n=nfft)
    B = rfft(b, n=nfft)
    C = A * B
    return irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(a*a)
    (identical to the value returned by ``evaluate_sequence`` for the same a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytical α‑family (a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a symmetric α‑family vector and project to the simplex."""
    i = np.arange(n, dtype=float)
    # log[(i+½)(n‑½‑i)]  – the product never hits zero inside the range.
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Numerical safety: replace infinities by a huge finite number.
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def refine_alpha_around(best_alpha: float, n: int,
                       deadline: float) -> Tuple[np.ndarray, float]:
    """Dense search in a narrow interval around a promising α."""
    lo, hi = best_alpha - 0.2, best_alpha + 0.2
    alphas = np.linspace(lo, hi, 41)
    best_a, best_sc = None, float("inf")
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Random sparse initialisation
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector (few non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# 3. Exponentiated‑gradient on the smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        beta_start: float = 0.5,
                        beta_end: float = 300.0,
                        eta0: float = 0.6,
                        eps: float = 1e-14) -> Tuple[np.ndarray, float]:
    """
    Minimise (1/β)·log Σ exp(β·(a*a)_k) with a KL‑mirror step.
    β is exponentially annealed from ``beta_start`` to ``beta_end``.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smooth max – soft‑max over the convolution
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        # stabilised exponentials
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient w.r.t. a (see reference implementation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # step size (classic 1/√t schedule)
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update + renormalisation
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Projected sub‑gradient descent with back‑tracking line search
# ----------------------------------------------------------------------


def subgrad_optimize_line_search(start_a: np.ndarray,
                                 deadline: float,
                                 max_iters: int = 300_000,
                                 step0: float = 0.08,
                                 min_step: float = 1e-5) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method for the true max‑objective.
    At each iteration a back‑tracking line search (geometric halving)
    is performed to guarantee a decrease whenever possible.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient at the active index k
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1)
        partner = k - idx
        grad[idx] = 2.0 * a[partner]

        # initial step size (≈ 1/√t)
        step = step0 / np.sqrt(t)

        # back‑track until improvement or step becomes tiny
        improved = False
        while step > min_step:
            cand = a - step * grad
            cand = project_to_simplex(cand)
            sc = compute_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand
                improved = True
                break
            step *= 0.5

        if not improved:
            # no improvement – keep the current a
            pass

        # periodic refresh of the best value (even if a didn't move)
        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final evaluation
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 5. Simulated‑annealing hill‑climb with targeted mass transfers
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_200_000) -> Tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that
      – swaps two entries (exploration),
      – moves a fraction of mass from a “high‑contribution” index to a random one
        (exploitation of the current peak),
      – accepts worsening moves with a temperature that decays exponentially.
    Only moves that respect the simplex after projection are kept.
    """
    a_curr = start_a.copy()
    n = a_curr.size
    best_a = a_curr.copy()
    best_sc = compute_score(best_a)

    # start temperature – a few percent of the current objective
    temperature = max(1e-8, best_sc * 0.02)
    decay = 0.99995
    step_scale = 0.12

    while max_steps > 0 and time.time() < deadline:
        # -------------------------------------------------
        # Propose a candidate
        # -------------------------------------------------
        if np.random.rand() < 0.6:
            # targeted mass transfer
            conv = convolution_fft(a_curr, a_curr)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(n - 1, k_max)
            idxs = np.arange(i_low, i_high + 1)
            contrib = a_curr[idxs] * a_curr[k_max - idxs]
            if contrib.size == 0:                     # safety
                i_high = i_low = np.random.randint(n)
                idxs = np.array([i_low])
                contrib = np.array([0.0])

            high_idx = idxs[np.argmax(contrib)]
            low_idx = np.random.randint(n)
            while low_idx == high_idx:
                low_idx = np.random.randint(n)

            delta = step_scale * a_curr[high_idx]
            cand = a_curr.copy()
            cand[high_idx] -= delta
            cand[low_idx] += delta
        else:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand = a_curr.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # project back onto the simplex (ensures non‑negativity and sum = 1)
        cand = project_to_simplex(cand)

        # evaluate candidate
        sc = compute_score(cand)

        # -------------------------------------------------
        # Acceptance rule
        # -------------------------------------------------
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            a_curr = cand.copy()
            temperature *= 0.99                # cool a little faster on improvement
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp((best_sc - sc) / max(temperature, 1e-12)):
                a_curr = cand.copy()

        # anneal temperature and adapt step scale
        temperature *= decay
        step_scale = max(0.001, step_scale * 0.99997)

        max_steps -= 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with minimal ``evaluate_sequence``.
    The routine follows a multi‑phase scheme (analytic start → smooth
    gradient → sub‑gradient → SA hill‑climb) and respects the 1000 s budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # -----------------------------------------------------------------
    # Choose a reasonable dimension (larger dimensions give a larger
    # search space but cost a bit more per convolution).
    # For the allotted time 2048 is a good compromise.
    # -----------------------------------------------------------------
    n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (global variable ``height_sequence_1``)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Coarse analytical start (α‑grid)
    # -----------------------------------------------------------------
    a_alpha, sc_alpha = find_best_alpha_start(n, deadline - 30.0)
    if a_alpha is not None and sc_alpha < best_score:
        best_score = sc_alpha
        best_seq = a_alpha.tolist()
        print(f"[{time.time():.1f}] after coarse α‑grid = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Fine local refinement of the α‑parameter
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        # the best α from the coarse grid lies somewhere; we scan a dense
        # neighbourhood to squeeze a little more out of the same family.
        # Here we simply re‑scan a tight interval around the previously
        # best α (±0.2) with 41 points.
        # We recover the α that produced `a_alpha` by a cheap root‑find:
        # (not strict – we just re‑search the whole interval)
        a_fine, sc_fine = refine_alpha_around(0.0, n, deadline - 10.0)
        if a_fine is not None and sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.tolist()
            print(f"[{time.time():.1f}] after fine α‑search = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Random sparse restarts with exponentiated‑gradient smoothing
    # -----------------------------------------------------------------
    while time.time() < deadline - 250.0:
        a0 = random_sparse_start(n)
        a_opt, sc_opt = exp_smooth_optimize(
            a0,
            deadline - 180.0,
            max_iters=200_000,
            beta_start=0.5,
            beta_end=800.0,
            eta0=0.7,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after random restart = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 4. Long dedicated exponentiated‑gradient optimisation
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] long exponentiated‑gradient run")
        a_smooth, sc_smooth = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=500_000,
            beta_start=0.5,
            beta_end=2000.0,
            eta0=0.6,
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 5. Sub‑gradient refinement with back‑tracking line search
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] sub‑gradient line‑search refinement")
        a_sub, sc_sub = subgrad_optimize_line_search(
            np.array(best_seq, dtype=float),
            deadline - 50.0,
            max_iters=300_000,
            step0=0.09,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=1_500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a tiny uniform vector
    # -----------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
