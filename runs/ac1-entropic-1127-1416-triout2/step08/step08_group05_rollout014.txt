def lp_refine(f, deadline, tolerance=1e-12, max_constraints=300):
    # returns refined vector b (same shape) or None.
    n = f.size
    M = float(np.max(convolution_fft(f, f)))
    # start with constraints near max
    conv_f = convolution_fft(f, f)
    idx = np.where(conv_f >= M - M * 1e-8)[0]  # relative tolerance
    if idx.size == 0:
        idx = np.array([np.argmax(conv_f)])
    # iterate
    while True:
        # limit number of constraints
        if idx.size > max_constraints:
            # keep only the most violated constraints (largest conv_f)
            # Sort idx by conv_f descending and take top max_constraints
            order = np.argsort(conv_f[idx])[::-1][:max_constraints]
            idx = idx[order]
        # Build A matrix
        A = np.zeros((idx.size, n))
        for row, k in enumerate(idx):
            i_min = max(0, k-(n-1))
            i_max = min(n-1, k)
            i_vals = np.arange(i_min, i_max+1)
            j_vals = k - i_vals
            A[row, j_vals] = f[i_vals]
        # solve LP
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=M * np.ones(idx.size), bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # fallback: return None
            return None
        b = res.x
        # check all constraints
        conv_fb = convolution_fft(f, b)
        violations = np.where(conv_fb > M + 1e-12)[0]
        if violations.size == 0:
            return b
        # add new violations to idx
        idx = np.unique(np.concatenate([idx, violations]))
        # if time limit reached, break
        if time.time() > deadline - 0.5:
            return b

"""
Propose a low‑score non‑negative step sequence.

Main ideas
----------
* Start from a diverse pool of analytic seeds (arcsine, beta, power‑law,
  mixtures, up‑sampled previous best).
* Refine each promising seed with a long Adam optimisation on a smooth‑max
  surrogate of the “max‑convolution” objective.
* Polish the Adam result with exponentiated‑gradient (KL‑mirror) descent.
* **New component:** a cutting‑plane linear‑programming refinement.
  For a current sequence f we solve
        maximise   Σ b_i
        subject to  (f * b)_k ≤ M  for all k,
  where M = max_k (f*f)_k.
  The LP pushes mass toward indices where f is small; a convex
  combination (1‑t)f + t·g (with g = b/Σb) typically reduces the true
  objective.
* Finish with a few iterations of projected sub‑gradient descent.

The routine respects a 1000 s wall‑clock budget, always returns a valid
sequence and prints progress information.
"""

import time
import numpy as np
from scipy.optimize import linprog
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast evaluation when a already satisfies sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 1500.0,
    ramp_iters: int = 100_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    z0 : ndarray
        Initial log‑probabilities (log‑softmax coordinates).
    deadline : float
        Absolute time at which the optimiser must stop.
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate for Adam.

    Returns
    -------
    a : ndarray
        Optimised probability vector (sum = 1, a_i ≥ 0).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # point on simplex
        conv = convolution_fft(a, a)       # linear conv, length 2n‑1
        mx = float(np.max(conv))

        # temperature schedule β(t)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)         # weights, non‑negative
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central n entries

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 60_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Mirror descent (KL divergence) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (cutting‑plane)
# ----------------------------------------------------------------------


def lp_refine(f: np.ndarray, deadline: float,
              tol_rel: float = 1e-9, max_constraints: int = 200) -> np.ndarray | None:
    """
    Cutting‑plane LP: maximise Σ b_i  subject to (f * b)_k ≤ M.
    Returns the raw vector b (not normalised).  If time runs out or
    the LP fails, returns None.
    """
    n = f.size
    conv_f = convolution_fft(f, f)
    M = float(np.max(conv_f))

    # initialise active set with constraints that are (almost) tight
    active = np.where(conv_f >= M - max(M * tol_rel, 1e-12))[0]

    # iterative expansion of the active set
    while True:
        if time.time() > deadline - 0.5:
            return None
        if active.size > max_constraints:
            # keep only the most critical constraints
            idx = np.argsort(conv_f[active])[-max_constraints:]
            active = active[idx]

        # build A matrix for A @ b ≤ M
        A = np.zeros((active.size, n))
        for row, k in enumerate(active):
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i_vals = np.arange(i_min, i_max + 1)
            j_vals = k - i_vals
            A[row, j_vals] = f[i_vals]

        # solve LP  maximise Σ b_i   ⇔   minimise -Σ b_i
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=M * np.ones(active.size),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            return None
        b = res.x

        # verify all constraints; add any violators
        conv_fb = convolution_fft(f, b)
        viol = np.where(conv_fb > M + 1e-12)[0]
        if viol.size == 0:
            return b
        # enlarge active set and repeat
        active = np.unique(np.concatenate([active, viol]))


def best_mix(f: np.ndarray, g: np.ndarray, steps: int = 9) -> tuple[np.ndarray, float]:
    """
    Search a few mixing ratios (1‑t)f + t·g (both already sum to 1)
    and return the best vector together with its score.
    """
    best_vec = f
    best_sc = compute_score(f)
    # include the endpoints explicitly
    candidates = np.linspace(0.0, 1.0, steps)
    for t in candidates:
        cand = (1.0 - t) * f + t * g
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand
    return best_vec, best_sc

def build_seed_pool(n: int, rng: np.random.Generator, best_sofar: List[float]) -> List[np.ndarray]:
    """Return a list of candidate probability vectors of length n."""
    seeds: List[np.ndarray] = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),                     # uniform
    ]

    # random power‑law shapes
    for _ in range(8):
        p = rng.uniform(0.05, 4.0)
        eps = rng.uniform(0.0, 3.0)
        seeds.append(generate_power_shape(n, p, eps))

    # random beta‑shapes
    for _ in range(8):
        a = rng.uniform(0.05, 6.0)
        b = rng.uniform(0.05, 6.0)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        seeds.append(generate_beta_shape(n, a, b, eps))

    # mixtures of arcsine & uniform
    for _ in range(4):
        w = rng.uniform(0.0, 1.0)
        seeds.append(generate_mixture_shape(n, w))

    # up‑sampled version of the current best (if any)
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        seeds.append(upsample_sequence(old, n))

    return seeds

                print(f"   LP‑mix score = {cand_score:.6f}")
                if cand_score < best_val:
                    best_val = cand_score
                    best_seq = cand_vec.tolist()
                    print(f"   *** Improved via LP‑mix: {best_val:.6f} ***")
                else:
                    # no improvement → exit LP loop for this seed
                    break

    # ------------------------------------------------------------------
    # 6. Final global polishing (exact sub‑gradient)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal global sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=200_000,
            step0=0.07,
        )
        final_val = evaluate_sequence(a_arr.tolist())
        print(f"   Final official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a_arr.tolist()
            print("*** Global best improved by final polish ***")

    # ------------------------------------------------------------------
    # 7. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform sequence.")

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
