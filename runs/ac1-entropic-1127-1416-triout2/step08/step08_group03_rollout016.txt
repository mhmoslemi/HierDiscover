"""
Propose a candidate sequence minimizing `evaluate_sequence`.

Algorithm:
1. Gather a fallback sequence if available.
2. Exhaustive grid search over several analytic families
   (symmetric edge‑heavy, left‑heavy power, exponential decay,
   asymmetric product, piecewise‑constant + power tail).  The grid
   extends far beyond the ranges used previously.
3. Randomised restarts: generate sparse or monotone starts and run a
   smoothed optimisation (log‑sum‑exp) with Adam.  The smoothing
   parameter β is annealed to a high value (up to 300) and the
   iteration budget is increased.
4. Classic projected sub‑gradient descent for the true max‑objective,
   using a diminishing step‑size.
5. Simple post‑processing:
   * symmetrisation,
   * convex combination with the uniform vector,
   * random permutation of the best vector.
6. Long simulated‑annealing hill‑climb on the simplex.
All stages respect a global wall‑clock budget (default 1000 s) and
periodically print intermediate scores.
The returned value is a plain Python list of floats ready for
`evaluate_sequence`.
"""
import time
import numpy as np
import math

# --------------------------------------------------------------
# Core utilities
# --------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector on the probability simplex: 2·n·max(a*a)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, sum x = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# --------------------------------------------------------------
# Analytic families (new + original)
# --------------------------------------------------------------

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Edge‑heavy symmetric shape a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    # When alpha == 0 we get uniform
    w = np.exp(-alpha * logprod)
    # safety against overflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a

def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a

def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a

def asymmetric_product_family(n: int, gamma_left: float, gamma_right: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑γ_left)·(n‑½‑i)^(‑γ_right)."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma_left * np.log(i + 0.5))
    right = np.exp(-gamma_right * np.log(n - 0.5 - i))
    w = left * right
    a = w / w.sum()
    return a

def piecewise_constant_power_family(n: int, front_frac: float, beta: float) -> np.ndarray:
    """
    First `front_frac` of entries are constant, the remainder decays as a power law.
    front_frac ∈ (0,1).
    """
    k = max(1, int(round(front_frac * n)))
    i = np.arange(n, dtype=np.float64)
    w = np.empty_like(i)
    w[:k] = 1.0
    tail = i[k:] + 0.5
    w[k:] = np.exp(-beta * np.log(tail))
    a = w / w.sum()
    return a

# Random initialisers
# --------------------------------------------------------------

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)        # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)

def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = a / a.sum()
    return a

def random_asymmetric_start(n: int) -> np.ndarray:
    """Random asymmetric product shape with random exponents."""
    gl = np.random.uniform(0.0, 10.0)
    gr = np.random.uniform(0.0, 10.0)
    return asymmetric_product_family(n, gl, gr)

# --------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp) using Adam
# --------------------------------------------------------------

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 300.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β is increased exponentially from beta_start to beta_end.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise logits so that softmax(z)=start_a
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        a = softmax(z)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # Gradient of smoothed max w.r.t a:
        # ∂/∂a_k Σ_i w_i (a * a)_i = 2 * (a_rev * w)_k
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]   # slice of length n

        # Back‑prop through softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Periodic exact evaluation
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()

    # Final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

# --------------------------------------------------------------

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method for the true max‑objective 2·n·max(a*a)
    with diminishing stepsize ∝ 1/√t.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Compute sub‑gradient for the active index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Long random walk on the simplex.
    Moves:
      * swap two entries (probability 0.5)
      * transfer a fraction of mass from i to j (otherwise)
    Temperature decays linearly; occasional uphill moves accepted.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # Initial temperature set as small fraction of current score
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def mix_with_uniform(a: np.ndarray, weight: float) -> np.ndarray:
    """Return (1-weight)*a + weight*uniform, projected onto simplex."""
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)
    mixed = (1.0 - weight) * a + weight * uniform
    return project_to_simplex(mixed)

    """Return a random permutation of the entries, projected onto simplex."""
    perm = np.random.permutation(a.shape[0])
    shuffled = a[perm]
    return project_to_simplex(shuffled)

# ----------------------------------------------------------------------
# Core utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the probability simplex:
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# Analytic families (original + new)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Edge‑heavy symmetric shape a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def asymmetric_product_family(n: int, gl: float, gr: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑gl)·(n‑½‑i)^(‑gr)."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gl * np.log(i + 0.5))
    right = np.exp(-gr * np.log(n - 0.5 - i))
    w = left * right
    a = w / w.sum()
    return a


def piecewise_constant_power_family(n: int, front_frac: float,
                                    beta: float) -> np.ndarray:
    """
    First `front_frac` of entries are constant, the remainder follows a
    power‑law decay.
    """
    k = max(1, int(round(front_frac * n)))
    i = np.arange(n, dtype=np.float64)
    w = np.empty_like(i)
    w[:k] = 1.0
    tail = i[k:] + 0.5
    w[k:] = np.exp(-beta * np.log(tail))
    a = w / w.sum()
    return a

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)        # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = a / a.sum()
    return a

    """Random asymmetric product shape with random exponents."""
    gl = np.random.uniform(0.0, 12.0)
    gr = np.random.uniform(0.0, 12.0)
    return asymmetric_product_family(n, gl, gr)


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp) using Adam
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 300.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).  β is
    increased exponentially from ``beta_start`` to ``beta_end``.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise logits

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule – exponential annealing
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        a = softmax(z)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed max w.r.t a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Small post‑processing utilities
# ----------------------------------------------------------------------


def mix_with_uniform(a: np.ndarray, weight: float) -> np.ndarray:
    """Return (1‑w)*a + w*uniform, projected onto the simplex."""
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)
    mixed = (1.0 - weight) * a + weight * uniform
    return project_to_simplex(mixed)


def random_permute(a: np.ndarray) -> np.ndarray:
    """Random permutation of entries, projected onto simplex."""
    perm = np.random.permutation(a.shape[0])
    return project_to_simplex(a[perm])

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimizes `evaluate_sequence`.
    The routine respects the global time budget and returns a plain Python
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose a length – 8192 gives a good trade‑off between flexibility
    # and speed.
    # --------------------------------------------------------------
    n = 8192
    print(f"[{time.time():.1f}] Using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. Fallback (if a previous construction is available)
    # --------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Analytic families – dense grid search (expanded ranges)
    # --------------------------------------------------------------
    # a) symmetric product (edge‑heavy)
    for a in np.linspace(0.0, 16.0, 81):          # step ≈0.2
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, a)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={a:.2f}) = {best_score:.6f}")

    # b) left‑heavy power law
    for b in np.linspace(0.0, 12.0, 61):          # step ≈0.2
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, b)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={b:.2f}) = {best_score:.6f}")

    # c) exponential decay
    for g in np.linspace(0.0, 0.02, 41):         # step 0.0005
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, g)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={g:.5f}) = {best_score:.6f}")

    # d) asymmetric product (random samples)
    for _ in range(40):
        if time.time() >= deadline - 30.0:
            break
        gl = np.random.uniform(0.0, 12.0)
        gr = np.random.uniform(0.0, 12.0)
        vec = asymmetric_product_family(n, gl, gr)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (asym γL={gl:.2f},γR={gr:.2f}) = {best_score:.6f}")

    # e) piecewise constant + power tail
    for front_frac in np.linspace(0.05, 0.5, 10):
        for beta in np.linspace(0.0, 12.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = piecewise_constant_power_family(n, front_frac, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (pc‑frac={front_frac:.2f},β={beta:.2f}) = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Random starts + Adam‑smooth optimisation (more aggressive)
    # --------------------------------------------------------------
    while time.time() < deadline - 220.0:
        # Choose a random start type
        choice = np.random.choice(3)
        if choice == 0:
            start_vec = random_sparse_start(n)
        elif choice == 1:
            start_vec = random_monotone_start(n)
        else:
            start_vec = random_asymmetric_start(n)

        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 180.0,
            max_iters=180_000,
            beta_start=6.0,
            beta_end=250.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after Adam‑smooth = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Projected sub‑gradient refinement
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 80.0,
            max_iters=2_500_000,
            step0=0.06,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Post‑processing tricks
    # --------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)

        # a) Symmetrisation
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

        # b) Convex mixture with the uniform vector
        for w in np.linspace(0.0, 0.2, 9):
            cand = mix_with_uniform(a_arr, w)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.tolist()
                print(f"[{time.time():.1f}] mix‑uniform w={w:.2f} improved best = {best_score:.6f}")

        # c) Random permutations (a handful)
        for _ in range(8):
            if time.time() >= deadline - 30.0:
                break
            perm = random_permute(a_arr)
            sc = compute_score(perm)
            if sc < best_score:
                best_score = sc
                best_seq = perm.tolist()
                print(f"[{time.time():.1f}] permutation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final hill‑climb (SA) with remaining budget
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Safeguard: if nothing succeeded, fall back to a modest uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
