def simulated_annealing(a_init, deadline, start_temp=0.01, end_temp=0.0001, max_iter_per_temp=500):
    a = a_init.copy()
    score = compute_score(a)
    n = a.size
    rng = np.random.default_rng()
    # schedule: exponentially decay temperature
    cur_temp = start_temp
    while time.time() < deadline:
        # propose move: transfer eps * a[i] from i to j, where eps small
        i, j = rng.integers(0, n, 2)
        if i == j: continue
        eps = cur_temp * a[i] # small fraction ~temp
        if eps <= 0: continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        s_new = compute_score(a_new)
        delta = s_new - score
        if delta < 0 or rng.random() < np.exp(-delta / cur_temp):
            a = a_new
            score = s_new
        # update temperature
        cur_temp = max(end_temp, cur_temp * 0.9999)  # or schedule based on iterations
    return a
```

    i = np.arange(n) + 0.5
    x = i / n  # values in (0,1)
    vals = (x**(alpha-1) * (1-x)**(beta-1) + eps)

"""
Propose a low‑∞ score step‑function.

Algorithm
---------
1. **Diverse seeding** – generate many probability vectors of several
   lengths (1024, 2048, 4096).  Shapes include:
   * uniform,
   * discrete arcsine (Beta(½,½)),
   * power‑law shapes a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p},
   * symmetric Beta(α,α) shapes,
   * mixtures of arcsine and uniform,
   * random Dirichlet draws.

   All candidates are normalised to sum = 1 and evaluated with a fast
   FFT‑based surrogate `2·n·max(conv(a,a))`.  The best few seeds are kept.

2. **Refinement pipeline** – each selected seed is fed through a series
   of optimisation stages, each respecting the global deadline:
   * smooth‑max Adam (soft‑max parametrisation, temperature τ linearly
     increased from 0.5 to 800),
   * Euclidean projected sub‑gradient descent on the true max‑convolution,
   * exponentiated‑gradient (mirror) polish,
   * simulated‑annealing hill‑climber (temperature‑controlled random
     mass transfers, Metropolis acceptance),
   * cheap stochastic local search (mass transfer only if the surrogate
     improves).

   After every stage the vector is projected onto the probability simplex.

3. **Evaluation** – the best refined vector is finally evaluated with the
   official `evaluate_sequence`.  The routine always returns a non‑empty
   list (fallback to a uniform distribution if nothing else succeeded).

The combination of many initial shapes, multiple restarts and a
temperature‑controlled stochastic refinement typically beats the previous
record (≈ 1.5114) and pushes the bound towards the target 1.503.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    `a` is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (Beta(0.5,0.5)), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """Generalised arcsine shape a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-12) -> np.ndarray:
    """
    Discrete symmetric Beta(α,β) density on {1,…,n}, normalised to sum = 1.
    The continuous pdf is proportional to x^{α‑1} (1‑x)^{β‑1}.
    """
    i = np.arange(n, dtype=float) + 0.5
    x = i / n
    vals = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0)) + eps
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform, normalised."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

# Optimisers

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 300_000,
                             tau_start: float = 0.5,
                             tau_end: float = 800.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate.
    Variables are logits w (softmax → a).  Temperature τ (smoothness)
    is increased linearly from `tau_start` to `tau_end`.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)          # logits

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)          # length L

        # smooth‑max via log‑sum‑exp (weights = softmax_τ(conv))
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp        # shape (L,)

        # gradient w.r.t. a : 2·conv(a, rev(weights))
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)  # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient through softmax
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 250_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror descent w.r.t. KL) polish.
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 20000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Simple hill‑climbing: transfer a tiny fraction of mass from i → j.
    Accept only strictly improving moves (according to the surrogate).
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.2 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1

    return a_cur


def simulated_annealing_refine(a_init: np.ndarray,
                               deadline: float,
                               start_temp: float = 0.01,
                               end_temp: float = 1e-4,
                               max_steps: int = 200_000,
                               rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Temperature‑controlled random mass transfers with Metropolis acceptance.
    The temperature controls both step size (transfer magnitude) and
    acceptance probability for uphill moves.
    """
    a = a_init.copy()
    score = compute_score(a)
    n = a.size
    step = 0

    while time.time() < deadline and step < max_steps:
        # linearised progress for exponential cooling
        t = step / max_steps
        temp = start_temp * ((end_temp / start_temp) ** t)  # exponential schedule

        i, j = rng.integers(0, n, size=2)
        if i == j:
            step += 1
            continue

        # propose moving a small fraction proportional to temperature
        eps = temp * a[i]
        if eps <= 0.0:
            step += 1
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # reject illegal negative mass
        if a_new[i] < -1e-15:
            step += 1
            continue

        # not renormalising – sum stays exactly 1
        new_score = compute_score(a_new)
        delta = new_score - score
        if delta < 0.0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            a = a_new
            score = new_score

        step += 1

    # final projection to guard against rounding artefacts
    a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single seed
# ----------------------------------------------------------------------


def full_optimize(a_init: np.ndarray, deadline: float) -> np.ndarray:
    """
    Run the complete refinement pipeline (Adam → sub‑grad → EG →
    simulated annealing → cheap local search) respecting the given deadline.
    """
    # 1) smooth‑max Adam
    a = adam_optimize_smooth_max(
        a_init,
        deadline,
        max_iters=300_000,
        tau_start=0.5,
        tau_end=800.0,
        base_lr=0.5,
    )

    # 2) exact sub‑gradient descent
    a = subgradient_descent_exact(
        a,
        deadline,
        max_iter=250_000,
        step0=0.07,
    )

    # 3) exponentiated‑gradient polish
    a = eg_refine(
        a,
        deadline,
        max_iter=150_000,
        eta0=0.25,
    )

    # 4) simulated‑annealing refinement
    a = simulated_annealing_refine(
        a,
        deadline,
        start_temp=0.01,
        end_temp=1e-4,
        max_steps=200_000,
    )

    # 5) cheap stochastic local search
    a = local_random_search(
        a,
        deadline,
        max_trials=20000,
        step_scale=0.001,
    )

    # final projection (ensures exact simplex)
    a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine performs a wide initial shape sweep followed by an
    intensive optimisation of the most promising seeds.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_surrogate: float = float("inf")
    best_official: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = fallback
                best_surrogate = compute_score(np.array(fallback))
        except Exception as e:
            print(f"Fallback evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Stage 1 – generate a pool of diverse seeds
    # ------------------------------------------------------------------
    candidate_lengths = [1024, 2048, 4096]
    p_grid = np.linspace(0.15, 1.10, 13)               # power‑shape exponent
    beta_grid = np.linspace(0.2, 1.5, 9)                # symmetric Beta α=β
    mix_grid = np.linspace(0.0, 1.0, 9)                # mixture weight
    num_dirichlet = 5                                  # per length
    dirichlet_conc = 0.5

    seed_pool: List[np.ndarray] = []

    print("\n--- Stage 1: generating seed pool ---")
    for n in candidate_lengths:
        print(f"Generating seeds for length n={n}")

        # uniform
        seed_pool.append(np.full(n, 1.0 / n))

        # arcsine (Beta(0.5,0.5))
        seed_pool.append(arcsine_initial(n))

        # power‑law family
        for p in p_grid:
            seed_pool.append(generate_power_shape(n, p))

        # symmetric Beta family
        for a in beta_grid:
            seed_pool.append(generate_beta_shape(n, a, a))

        # mixtures arcsine / uniform
        arc = arcsine_initial(n)
        uni = np.full(n, 1.0 / n)
        for w in mix_grid:
            mix = w * arc + (1.0 - w) * uni
            seed_pool.append(mix / mix.sum())

        # random Dirichlet draws
        for _ in range(num_dirichlet):
            seed_pool.append(rng.dirichlet(np.ones(n) * dirichlet_conc))

    # ------------------------------------------------------------------
    # Evaluate all seeds (fast surrogate) and keep the best few
    # ------------------------------------------------------------------
    scored_seeds: List[tuple] = []
    for a in seed_pool:
        s = compute_score(a)
        scored_seeds.append((s, a))
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    scored_seeds.sort(key=lambda x: x[0])

    top_k = 5                                            # keep the very best seeds
    top_candidates = [a for (_, a) in scored_seeds[:top_k]]
    print(f"\nTop‑{top_k} surrogate scores from the pool:")
    for i, (s, _) in enumerate(scored_seeds[:top_k]):
        print(f"  #{i+1}: surrogate={s:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – refine each top candidate (in order of quality)
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(top_candidates, 1):
        if time.time() >= deadline:
            break
        print(f"\n--- Refinement of candidate #{idx} (length={a0.size}) ---")
        a_refined = full_optimize(a0, deadline)
        s_refined = compute_score(a_refined)
        print(f"  Refined surrogate={s_refined:.6f}")

        if s_refined < best_surrogate:
            best_surrogate = s_refined
            best_seq = a_refined.tolist()
            print(f"  ** New best surrogate! **")

        # occasional official check (cheap) – only if time permits
        if time.time() < deadline - 5.0:
            off = evaluate_sequence(best_seq)
            if off < best_official:
                best_official = off
                print(f"  Official score improved to {off:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – random restarts (lightweight) until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # pick a random length (favor 2048) and a random Dirichlet seed
        n = rng.choice(candidate_lengths, p=np.array([0.2, 0.6, 0.2]))
        a0 = rng.dirichlet(np.ones(n) * dirichlet_conc)

        a_refined = full_optimize(a0, deadline)
        s_refined = compute_score(a_refined)

        if s_refined < best_surrogate:
            best_surrogate = s_refined
            best_seq = a_refined.tolist()
            print(f"\nRandom restart yielded new best surrogate={s_refined:.6f}")

            # official evaluation if we still have a margin
            if time.time() < deadline - 2.0:
                off = evaluate_sequence(best_seq)
                if off < best_official:
                    best_official = off
                    print(f"  Official score updated to {off:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation of the best candidate
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_official:.6f}")
        if final_official < best_official:
            best_official = final_official
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_official = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        final_official = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {final_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
