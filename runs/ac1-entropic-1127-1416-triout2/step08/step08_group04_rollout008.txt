The algorithm proceeds in three logical phases:

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT, length = len(a)+len(b)-1."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.
    Implements Duchi et al. (2008) which runs in O(n log n).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate equals ``2·n·max(conv(a,a))``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (α=β=–0.5) with a tiny regulariser."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α+1,β+1) density on {0,…,n‑1},
    i.e. a_i ∝ ((i+½)/n)^α * (1−(i+½)/n)^β .
    The parameters must satisfy α,β > –1 so that the density is integrable.
    """
    i = np.arange(n, dtype=float) + 0.5
    x = i / n
    # np.power works for negative exponents as long as x∈(0,1)
    vals = np.power(x, alpha) * np.power(1.0 - x, beta)
    # Guard against infinities caused by extreme exponents
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+½)(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def mixture_arc_uniform(n: int, w: float) -> np.ndarray:
    """Mixture w·arcsine + (1‑w)·uniform, normalised."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = w * arc + (1.0 - w) * uni
    return mix / mix.sum()

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 500_000,
                             tau_start: float = 0.5,
                             tau_end: float = 2000.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """Adam on the smooth‑max (log‑sum‑exp) approximation of max(conv)."""
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from a small positive version of a_init
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)          # for numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)     # length L

        # smooth‑max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp   # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)   # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 100_000 == 0:
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={compute_score(a):.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 500_000,
                              step0: float = 0.09) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    The sub‑gradient is 2·conv(a, mask_rev) where mask flags
    (almost) maximal entries of conv(a,a).
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # tolerance to capture all maximisers
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient computation
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 100_000 == 0:
            print(f"    Sub‑grad it={it:,d} surrogate={compute_score(a):.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 300_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent w.r.t. KL) polish.
    Uses the exact sub‑gradient of the max‑convolution.
    """
    n = a.size
    L = 2 * n - 1
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 100_000 == 0:
            print(f"    EG it={it:,d} surrogate={compute_score(a):.6f}")

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 60_000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Very cheap stochastic hill‑climber.
    Moves a tiny fraction of mass from i → j; improved moves are kept.
    A simple simulated‑annealing rule occasionally accepts uphill steps.
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size
    T0 = 1e-4                     # initial temperature for uphill acceptance

    for trial in range(max_trials):
        if time.time() >= deadline - 0.2:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue

        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue

        s_new = compute_score(a_new)
        delta = s_new - score_cur
        # Temperature decays linearly
        T = T0 * (1.0 - trial / max_trials)

        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a_cur = a_new
            score_cur = s_new

    return a_cur


# ----------------------------------------------------------------------
# Core optimisation pipeline for a single candidate
# ----------------------------------------------------------------------


def optimise_candidate(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Run the deterministic + stochastic refinement pipeline on a single
    candidate sequence.  The allocated wall‑clock time is split automatically.
    """
    # Phase fractions (adjusted on‑the‑fly)
    remaining = deadline - time.time()
    if remaining <= 0.0:
        return a_start

    # 40 % Adam, 30 % sub‑gradient, 15 % EG, 15 % local search
    t_adam = remaining * 0.40
    t_subg = remaining * 0.30
    t_eg   = remaining * 0.15
    t_loc  = remaining * 0.15

    # ------------------------------------------------------------------
    # 1) Adam on smooth‑max
    # ------------------------------------------------------------------
    adam_deadline = time.time() + t_adam
    a = adam_optimize_smooth_max(
        a_init=a_start,
        deadline=adam_deadline,
        max_iters=500_000,
        tau_start=0.5,
        tau_end=2000.0,
        base_lr=0.5,
    )
    print(f"    After Adam surrogate={compute_score(a):.6f}")

    # ------------------------------------------------------------------
    # 2) Exact sub‑gradient descent
    # ------------------------------------------------------------------
    subg_deadline = time.time() + t_subg
    a = subgradient_descent_exact(
        initial_a=a,
        deadline=subg_deadline,
        max_iter=500_000,
        step0=0.09,
    )
    print(f"    After sub‑grad surrogate={compute_score(a):.6f}")

    # ------------------------------------------------------------------
    # 3) Exponentiated‑gradient polish
    # ------------------------------------------------------------------
    eg_deadline = time.time() + t_eg
    a = eg_refine(
        a=a,
        deadline=eg_deadline,
        max_iter=300_000,
        eta0=0.25,
    )
    print(f"    After EG polish surrogate={compute_score(a):.6f}")

    # ------------------------------------------------------------------
    # 4) Local stochastic hill climbing
    # ------------------------------------------------------------------
    loc_deadline = time.time() + t_loc
    a = local_random_search(
        a=a,
        deadline=loc_deadline,
        max_trials=60_000,
        step_scale=0.001,
        rng=np.random.default_rng(),
    )
    print(f"    After local search surrogate={compute_score(a):.6f}")

    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """Search for a non‑negative step sequence with a low official score."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    best_seq: List[float] = []
    best_official: float = float("inf")
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = fallback
        except Exception as e:
            print(f"Fallback evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Stage 1 – broad, multi‑scale shape search
    # ------------------------------------------------------------------
    candidate_ns = [1024, 2048, 4096]
    # α,β grid for Beta shapes (covers uniform, arcsine and more extreme U‑shapes)
    alpha_grid = np.linspace(-0.9, 2.0, 15)
    beta_grid  = np.linspace(-0.9, 2.0, 15)

    # keep the top‑K shapes (smaller surrogate is better)
    K = 5
    heap: List[tuple] = []   # max‑heap (neg‑score, array)

    print("\n--- Stage 1: coarse shape exploration ---")
    for n in candidate_ns:
        print(f"Exploring length n={n}")

        # uniform
        uni = np.full(n, 1.0 / n)
        s = compute_score(uni)
        heapq.heappush(heap, (-s, uni))
        if len(heap) > K:
            heapq.heappop(heap)

        # arcsine
        arc = arcsine_initial(n)
        s = compute_score(arc)
        heapq.heappush(heap, (-s, arc))
        if len(heap) > K:
            heapq.heappop(heap)

        # power‑law (symmetric) shapes
        p_grid = np.linspace(0.15, 0.85, 15)
        for p in p_grid:
            shape = generate_power_shape(n, p, eps=1e-8)
            s = compute_score(shape)
            heapq.heappush(heap, (-s, shape))
            if len(heap) > K:
                heapq.heappop(heap)

        # full Beta grid
        for a_val in alpha_grid:
            for b_val in beta_grid:
                shape = beta_shape(n, a_val, b_val)
                s = compute_score(shape)
                heapq.heappush(heap, (-s, shape))
                if len(heap) > K:
                    heapq.heappop(heap)

        # mixtures arcsine / uniform (extra sanity)
        w_grid = np.linspace(0.0, 1.0, 11)
        for w in w_grid:
            mix = mixture_arc_uniform(n, w)
            s = compute_score(mix)
            heapq.heappush(heap, (-s, mix))
            if len(heap) > K:
                heapq.heappop(heap)

    # Extract best candidates (sorted by increasing surrogate)
    best_candidates = [item[1] for item in sorted(heap, key=lambda x: -x[0])]
    print(f"\nBest surrogate after Stage 1: {compute_score(best_candidates[0]):.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deterministic & stochastic refinements
    # ------------------------------------------------------------------
    for idx, cand in enumerate(best_candidates):
        rem = deadline - time.time()
        if rem < 30.0:          # not enough time for another run
            print("Insufficient time left for further refinements.")
            break
        print(f"\n--- Refinement of candidate #{idx + 1} (length {cand.size}) ---")
        refined = optimise_candidate(cand, deadline)
        # Evaluate officially (the official function rescales automatically)
        try:
            official_val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {official_val:.6f}")
            if official_val < best_official:
                best_official = official_val
                best_seq = refined.tolist()
        except Exception as e:
            print(f"    Official evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to uniform of moderate size
        fallback_n = 2048
        best_seq = [1.0 / fallback_n] * fallback_n
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform length {fallback_n}, official score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
