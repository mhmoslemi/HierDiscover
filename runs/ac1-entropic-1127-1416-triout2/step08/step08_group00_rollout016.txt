"""
Search for a non‑negative step‑function that minimises `evaluate_sequence`.

The search follows a multi‑phase pipeline (all phases respect the 1000 s
wall‑clock limit):

1. **Smooth‑max Adam optimisation** – many short random restarts of a
   log‑sum‑exp surrogate for the max‑convolution.  The annealing schedule
   for the soft‑max is extended (β up to 1500) to tighten the surrogate.

2. **Projected sub‑gradient descent** – refines the best candidate from
   phase 1 on the true max‑objective.

3. **LP‑based refinement (optional)** – solves a linear programme that
   finds a vector `g` with maximal ℓ₁‑mass while respecting the current
   convolution bound `M = max(conv(f,f))`.  Mixing the current vector `f`
   with `g` provably reduces `M`.  Requires SciPy; otherwise this phase is
   skipped.

4. **Stochastic hill‑climb** – random swaps or small mass‑transfer moves
   are accepted only when they improve the exact objective.

5. **Fine‑tune polishing** – very small mass‑transfer moves (step‑scale
   ≈ 10⁻³) are repeatedly tried; any improvement is kept, allowing the
   search to settle into a tighter local minimum.

All helper routines are defined at top‑level; the entry point is
`propose_candidate`.  The returned object is a plain Python list of
floats (the step‑function heights).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast for modest sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth‑max optimisation (log‑sum‑exp) with Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑accelerated gradient descent on a soft‑max surrogate of
    max(conv(a,a)).  Returns the best *exact* score seen.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # gradient of the soft‑max (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score

def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the maximal entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 2_000_000,
) -> tuple[list[float], float]:
    """
    Random local moves (swap or small mass transfer) accepted only when
    the exact score improves.  Step size adapts automatically.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score

# 4. Linear‑programming based refinement (requires SciPy)
# ----------------------------------------------------------------------


def build_convolution_matrix(f):
    """
    Build the (2n‑1)×n matrix T_f such that T_f @ g = conv(f, g).
    """
    import scipy.sparse as sp

    n = f.shape[0]
    diags = [f[d] * np.ones(n) for d in range(n)]
    offsets = list(range(n))
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def lp_refine(f, time_deadline, max_iter=15, step0=0.5):
    """
    LP‑based refinement: at each iteration solve

        maximise   Σ g_i
        s.t.       T_f g ≤ M   (component‑wise)
                   g_i ≥ 0

    where M = max(conv(f,f)).  The solution g is normalised to sum 1 and
    mixed with f.  The process monotonically drives down M.
    """
    try:
        import scipy.optimize as opt
    except Exception:
        print("SciPy not available – skipping LP refinement.")
        return f, compute_score(f)

    n = f.shape[0]
    best_f = f.copy()
    best_score = compute_score(f)
    t = step0

    for it in range(max_iter):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(f, f)
        M = conv.max()
        A = build_convolution_matrix(f)
        b = np.full(2 * n - 1, M, dtype=float)
        c = -np.ones(n, dtype=float)          # maximise Σ g_i  ⇒  minimise –Σ g_i

        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"LP solve failed at iteration {it}: {res.message}")
            break

        g = res.x
        if g.sum() <= 0:
            break
        g = g / g.sum()          # renormalise to a probability vector

        # convex combination
        f = (1 - t) * f + t * g
        f = np.maximum(f, 0)
        f = f / f.sum()

        score = compute_score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
            print(f"LP refinement iter {it+1}: score={best_score:.6f}")
            t = min(0.9, t * 1.1)   # become bolder when improvement occurs
        else:
            t = max(0.1, t * 0.7)   # become more cautious otherwise

    return best_f, best_score

            max_steps=5_000_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Very fine‑grained polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nFine‑tune polishing")
        fine_seq, fine_score = fine_tune(best_seq, deadline - 2.0)
        if fine_score < best_val:
            best_val = fine_score
            best_seq = fine_seq
            print(f"Fine‑tune improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Fallback if nothing found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final evaluation (for debugging)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
