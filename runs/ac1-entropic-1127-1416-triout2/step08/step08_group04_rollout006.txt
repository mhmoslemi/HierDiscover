def accelerated_subgradient(initial_a: np.ndarray,
                           deadline: float,
                           max_iter: int = 200_000,
                           step0: float = 0.07) -> np.ndarray:
    """
    Nesterov‑accelerated projected sub‑gradient descent on the true max‑convolution.
    """
    a_prev = project_to_simplex(initial_a.copy())
    a = a_prev.copy()
    t_prev = 1.0

    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # extrapolation
        beta = (t_prev - 1) / (t_prev + 1e-12)
        y = a + beta * (a - a_prev)   # momentum step
        y = project_to_simplex(y)     # keep feasible

        # compute subgradient at y
        conv = convolution_fft(y, y)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(y, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # step size
        eta = step0 / np.sqrt(it)

        a_next = y - eta * subgrad
        a_next = project_to_simplex(a_next)

        # update Nesterov parameters
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0

        # prepare next iter
        a_prev, a = a, a_next
        t_prev = t_next

    return a

The task is to find a non‑negative sequence `a = (a₀,…,aₙ₋₁)` that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a*a` denotes the linear convolution of `a` with itself.  
Because the score is *scale invariant* we may fix `∑a_i = 1`.  
The optimisation problem is therefore

* minimise `2·n·max(conv(a,a))`  
* subject to `a_i ≥ 0` and `∑a_i = 1`.

The program consists of three layers  

1. **Coarse shape enumeration** – a large set of parametric families (uniform, arcsine, β‑shapes, power‑shapes and their mixtures) is sampled for several lengths.  
   The cheap surrogate `2·n·max(conv)` is evaluated with an FFT‑based convolution and the best few candidates are kept.

2. **Continuous optimisation** – each selected candidate is refined by  
   * **Adam** on a smooth‑max approximation of the `max` (log‑sum‑exp),  
   * **Nesterov‑accelerated projected sub‑gradient descent** on the true objective, and  
   * **Exponentiated‑gradient polishing**.  

3. **Stochastic local search** – a cheap hill‑climber moves a tiny amount of mass between two coordinates, accepting any improving move.  

All phases respect a global wall‑clock budget (default 1000 s).  
The routine finally evaluates the best sequence with the official `evaluate_sequence` function and returns it.

The combination of a richer family of initial shapes, longer Adam/​sub‑gradient schedules, Nesterov acceleration and several restarts gives a better upper bound than the previous baseline.

Propose a step‑function sequence that minimises the official evaluation
score.  The algorithm proceeds in three stages:

1.  Enumerate many parametric families (uniform, arcsine, beta‑shapes,
    power‑shapes and mixtures) on several lengths.  The cheap surrogate

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to satisfy Σa = 1;
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (stage 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """
    Generalised arcsine shape:
    a_i ∝ ((i+½)*(n‑½‑i) + ε)^{-p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta_param: float,
                       eps: float = 1e-12) -> np.ndarray:
    """
    Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1.
    Works also for α,β < 1 (U‑shaped) and for α,β > 1 (bell‑shaped).
    """
    i = np.arange(n, dtype=float) + 0.5
    x = (i - 0.5) / n                      # centre of each bin in (0,1)
    # guard against 0⁰ or division by zero
    with np.errstate(divide='ignore', invalid='ignore'):
        vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta_param - 1.0)
    # replace nan / inf with 0
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """
    Mixture weight_arc·arcsine + (1‑weight_arc)·uniform,
    normalised to sum = 1.
    """
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

# Optimisation primitives

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 300_000,
                             tau_start: float = 0.5,
                             tau_end: float = 800.0,
                             base_lr: float = 0.6) -> np.ndarray:
    """
    Adam optimiser on a smooth‑max approximation of max(conv).
    The optimisation variable are logits w such that a = softmax(w).
    """
    n = a_init.size
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)          # initialise logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·conv(a, rev(weights))
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)   # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def accelerated_subgradient(initial_a: np.ndarray,
                           deadline: float,
                           max_iter: int = 250_000,
                           step0: float = 0.08) -> np.ndarray:
    """
    Nesterov‑accelerated projected sub‑gradient descent on the true
    max‑convolution objective.
    """
    a_prev = project_to_simplex(initial_a.copy())
    a = a_prev.copy()
    t_prev = 1.0
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # extrapolation (Nesterov momentum)
        beta = (t_prev - 1.0) / (t_prev + 1e-12)
        y = a + beta * (a - a_prev)
        y = project_to_simplex(y)

        # sub‑gradient at y
        conv = convolution_fft(y, y)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(y, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # step size
        eta = step0 / np.sqrt(it)

        a_next = y - eta * subgrad
        a_next = project_to_simplex(a_next)

        # update Nesterov coefficients
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0

        a_prev, a = a, a_next
        t_prev = t_next

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.2) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent w.r.t. KL).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 30000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Very cheap stochastic hill climbing:
    move a tiny amount of mass from one coordinate to another.
    Accept only improving moves (according to the surrogate score).
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.2 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1

    return a_cur

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with the smallest official
    score.  The routine follows the three‑stage plan described in the
    module docstring and respects the global wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0       # safety margin

    best_seq: List[float] = []
    best_official: float = float("inf")
    best_surrogate: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = fallback
                best_surrogate = compute_score(np.array(fallback))
        except Exception as e:
            print(f"Fallback evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Stage 1 – extensive cheap grid search over shapes
    # ------------------------------------------------------------------
    candidate_ns = [1024, 1536, 2048, 3072, 4096]
    p_grid = np.linspace(0.15, 0.85, 15)          # power‑shape exponent
    beta_grid = np.linspace(0.15, 0.85, 15)       # symmetric beta exponent
    w_grid = np.linspace(0.0, 1.0, 11)           # mixture weight

    print("\n--- Stage 1: exhaustive coarse shape search ---")
    for n in candidate_ns:
        print(f"  Length n = {n}")

        # uniform
        uni = np.full(n, 1.0 / n)
        s = compute_score(uni)
        if s < best_surrogate:
            best_surrogate, best_seq = s, uni.tolist()
            print(f"    New best (uniform) surrogate = {s:.6f}")

        # arcsine
        arc = arcsine_initial(n)
        s = compute_score(arc)
        if s < best_surrogate:
            best_surrogate, best_seq = s, arc.tolist()
            print(f"    New best (arcsine) surrogate = {s:.6f}")

        # power‑shape family
        for p in p_grid:
            shape = generate_power_shape(n, p)
            s = compute_score(shape)
            if s < best_surrogate:
                best_surrogate, best_seq = s, shape.tolist()
                print(f"    New best (power p={p:.3f}) surrogate = {s:.6f}")

        # beta‑shape family (symmetric)
        for a in beta_grid:
            shape = generate_beta_shape(n, a, a)
            s = compute_score(shape)
            if s < best_surrogate:
                best_surrogate, best_seq = s, shape.tolist()
                print(f"    New best (beta a={a:.3f}) surrogate = {s:.6f}")

        # mixtures arcsine / uniform
        for w in w_grid:
            mix = generate_mixture_shape(n, w)
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"    New best (mix w={w:.2f}) surrogate = {s:.6f}")

        # mixtures beta / uniform
        for w in w_grid:
            beta = generate_beta_shape(n, 0.5, 0.5)      # arcsine is β(0.5,0.5)
            mix = w * beta + (1.0 - w) * uni
            mix = mix / mix.sum()
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"    New best (beta‑mix w={w:.2f}) surrogate = {s:.6f}")

    # ------------------------------------------------------------------
    # Collect a few promising starting points for intensive optimisation
    # ------------------------------------------------------------------
    top_candidates = [np.array(best_seq, dtype=float)]

    # add a few random Dirichlet perturbations of the same length
    n_best = len(best_seq)
    for _ in range(3):
        pert = rng.dirichlet(alpha=np.full(n_best, 0.2))
        top_candidates.append(pert)

    # also keep a pure uniform vector as a safety net
    top_candidates.append(np.full(n_best, 1.0 / n_best))

    # ------------------------------------------------------------------
    # Stage 2 – intensive refinement of each top candidate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    per_candidate = remaining / max(1, len(top_candidates))

    for idx, a_start in enumerate(top_candidates):
        cand_deadline = time.time() + per_candidate
        print(f"\n--- Optimising candidate {idx + 1}/{len(top_candidates)} (n={a_start.size}) ---")

        # 2.1 Adam smooth‑max optimisation
        if time.time() < cand_deadline - 5.0:
            a = adam_optimize_smooth_max(a_start, cand_deadline,
                                         max_iters=400_000,
                                         tau_start=0.5,
                                         tau_end=1000.0,
                                         base_lr=0.7)
        else:
            a = a_start.copy()

        # 2.2 Nesterov‑accelerated sub‑gradient descent on the true max
        if time.time() < cand_deadline - 5.0:
            a = accelerated_subgradient(a, cand_deadline,
                                        max_iter=300_000,
                                        step0=0.08)

        # 2.3 Exponentiated‑gradient polish
        if time.time() < cand_deadline - 5.0:
            a = eg_refine(a, cand_deadline,
                          max_iter=200_000,
                          eta0=0.2)

        # 2.4 Cheap stochastic local search
        if time.time() < cand_deadline - 2.0:
            a = local_random_search(a, cand_deadline,
                                    max_trials=30000,
                                    step_scale=0.001,
                                    rng=rng)

        # final official evaluation of this candidate
        seq = a.tolist()
        try:
            val = evaluate_sequence(seq)
            print(f"Candidate {idx + 1} official score = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = seq
                best_surrogate = compute_score(np.array(seq))
        except Exception as e:
            print(f"Official evaluation failed for candidate {idx + 1}: {e}")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
