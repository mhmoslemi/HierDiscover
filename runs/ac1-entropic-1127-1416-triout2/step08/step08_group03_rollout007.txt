def reduce_max_convolution(a, deadline, max_iters=200000):
    n = len(a)
    a = a.copy()

        # For each i such that 0 <= i < n and j=k-i in range:
        indices = np.arange(n)

def targeted_reduce(a, deadline, max_iters=200000):
    n = a.shape[0]
    a = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    step_frac = 0.05  # fraction of source mass moved each iteration

    it = 0
    while time.time() < deadline and it < max_iters:
        # compute conv and active index
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # compute contributions for each i
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        # source idx with largest contribution
        src = int(np.argmax(contrib))

        # choose a destination index with minimal contribution to the same k
        # compute contributions again maybe for dest selection based on a values:
        # simpler: random dest not equal src
        dst = np.random.randint(n)
        while dst == src:
            dst = np.random.randint(n)

        delta = step_frac * a[src]
        if delta <= 0: break
        a[src] -= delta
        a[dst] += delta
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # increase step size modestly
            step_frac = min(step_frac * 1.02, 0.2)
        else:
            # decay step size
            step_frac *= 0.98

        it += 1
        if it % 1000 == 0:
            # occasionally reload conv etc.
            pass

    return best_a, best_score

"""
Propose a low‑score non‑negative step‑function.

The search follows several loosely coupled stages that fit into the
1000 s wall‑clock limit.

0.  Load a fallback solution if one is provided.
1.  Exhaustively scan a few analytic one‑parameter families
    (symmetric product, left‑heavy power, exponential decay,
    symmetric exponential decay) on a relatively large problem size.
2.  Generate a handful of random monotone vectors.
3.  Run a smoothed‑max optimiser (Adam on the soft‑max
    parametrisation) with a very aggressive β‑schedule.
    Several restarts are performed; the best result is kept.
4.  Fine‑tune the best vector with an even larger β (≈ 10³) and a
    small learning rate.
5.  Apply a cheap, directed “max‑convolution reduction” heuristic:
    each iteration transfers a small fraction of mass from the index
    that contributes most to the current maximal convolution entry to a
    random low‑impact index.
6.  Symmetrise the vector (many optimal solutions are symmetric).
7.  Finish with a long simulated‑annealing hill‑climb on the simplex.

All stages respect the global deadline.  The routine finally returns a
plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using a real FFT.  The result length is
    len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector whose entries sum to 1:
        2·n·max(conv(a, a))
    (the same quantity that `evaluate_sequence` returns once the sum
    normalisation is enforced).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all are normalised to sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α), symmetric w.r.t. centre."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against overflow for α≈0
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β), monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, r: float) -> np.ndarray:
    """Pure exponential decay a_i ∝ r**i (0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    w = r ** i
    a = w / w.sum()
    return a


def symmetric_exponential_family(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two exponentials: r**i + r**(n‑1‑i)."""
    i = np.arange(n, dtype=np.float64)
    w = r ** i + r ** (n - 1 - i)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """A tiny random support (≈ 5 % non‑zeros) projected onto the simplex."""
    sparsity = np.random.uniform(0.03, 0.07)  # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (log‑sum‑exp) using Adam
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 5.0,
                         beta_end: float = 150.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β grows exponentially from beta_start
    to beta_end over the allotted iterations.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule β (exponential growth)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # smoothed max via log‑sum‑exp
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                     # softmax weights

        # gradient d f_β / d a = 2 * (a_rev * w)[center : center+n]
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted reduction of the current maximal convolution entry
# ----------------------------------------------------------------------


def targeted_reduce(a: np.ndarray,
                    deadline: float,
                    max_iters: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer heuristic.
    At each iteration the index that contributes most to the
    current maximal convolution entry is identified and a small
    fraction of its mass is transferred to a randomly chosen
    partner.  The simplex constraint is enforced after each move.
    """
    n = a.shape[0]
    a = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    step_frac = 0.05                      # start with 5 % of source mass
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index attaining the maximum

        # contributions c_i = a_i * a_{k-i}
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        src = int(np.argmax(contrib))     # source index with largest product
        if a[src] <= 0.0:
            break

        # choose a destination that is *not* the source
        dst = np.random.randint(n)
        while dst == src:
            dst = np.random.randint(n)

        delta = step_frac * a[src]        # amount moved
        a[src] -= delta
        a[dst] += delta

        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # success – allow a slightly larger step size
            step_frac = min(step_frac * 1.05, 0.2)
        else:
            # no improvement – shrink step size
            step_frac *= 0.97

        it += 1
        if it % 2000 == 0:
            # occasional clean‑up (re‑project just in case)
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with a tiny simulated‑annealing component
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Long random walk on the simplex.
    – With probability 0.5 swap two entries.
    – Otherwise move a fraction of mass from one index to another.
    Simulated‑annealing acceptance permits occasional uphill steps.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4   # tiny temperature scale
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta_score = cand_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
