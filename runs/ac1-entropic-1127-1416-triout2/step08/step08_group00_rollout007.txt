"""
Differential‑Evolution + sub‑gradient / hill‑climb search for the
minimisation of `evaluate_sequence`.

* Work on the probability simplex (∑a_i = 1) – the objective
  `evaluate_sequence` becomes `2·n·max(conv(a,a))`.
* Initialise a modest population of random vectors and run a classic
  DE/rand/1/bin loop (mutation, binomial crossover, selection).
* Keep the best individual and, when time is running out, polish it with a
  projected sub‑gradient descent followed by a stochastic hill‑climb.
* All operations respect the 1000 s wall‑clock budget; intermediate scores
  are printed for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using the real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # all non‑positive – return the centre of the simplex
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    `init_a` must already sum to 1.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the maximal entry (see analysis)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 2_000_000,
) -> tuple[list[float], float]:
    """
    Simple local search: random swaps or small mass transfers.
    Only moves that strictly improve the exact score are kept.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                       # become more conservative

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Differential Evolution core
# ----------------------------------------------------------------------


def differential_evolution(
    n: int,
    time_deadline: float,
    pop_size: int = 20,
    F_range: tuple[float, float] = (0.4, 0.9),
    Cr_range: tuple[float, float] = (0.8, 0.95),
    initial_population: list[np.ndarray] | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simple DE/rand/1/bin on the simplex.
    Returns the best individual found and its score.
    """
    # initialise population
    pop = []
    scores = []
    for i in range(pop_size):
        if initial_population is not None and i < len(initial_population):
            a = initial_population[i].copy()
        else:
            a = np.random.rand(n)
            a = a / a.sum()
        a = project_to_simplex(a)
        pop.append(a)
        s = compute_score(a)
        scores.append(s)

    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    gen = 0
    while time.time() < time_deadline - 5.0:   # keep a few seconds for polishing
        gen += 1
        for i in range(pop_size):
            if time.time() >= time_deadline - 5.0:
                break

            # three distinct indices different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a_idx, b_idx, c_idx = np.random.choice(idxs, size=3, replace=False)

            # DE mutation with random scaling factor
            F = np.random.uniform(*F_range)
            donor = pop[a_idx] + F * (pop[b_idx] - pop[c_idx])
            donor = np.maximum(donor, 0.0)

            # binomial crossover
            Cr = np.random.uniform(*Cr_range)
            mask = np.random.rand(n) < Cr
            if not np.any(mask):
                mask[np.random.randint(n)] = True   # guarantee at least one component
            cand = pop[i].copy()
            cand[mask] = donor[mask]

            cand = project_to_simplex(cand)
            cand_score = compute_score(cand)

            # selection
            if cand_score < scores[i]:
                pop[i] = cand
                scores[i] = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best = cand.copy()

        if gen % 10 == 0:
            print(f"DE generation {gen:3d}, best score = {best_score:.6f}")

    return best, best_score
