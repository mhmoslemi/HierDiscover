"""
search for a low‑scoring non‑negative step sequence.

The objective (official ``evaluate_sequence``) is scale‑invariant, so we
optimize on the probability simplex (∑a_i = 1).  The algorithm proceeds
in three stages:

1.  **Parametric global search** – for a few lengths (1024 … 8192) we
    run a cheap Differential‑Evolution optimiser on three simple families
    (γ‑shape, β‑shape and a linear mixture of arcsine/uniform).  This
    explores the low‑dimensional space of shape parameters and quickly
    yields high‑quality candidates.

2.  **Mirror‑descent polishing** – the best candidate of each length is
    refined by an exponentiated‑gradient step (KL‑mirror descent) and a
    final projected sub‑gradient descent on the true max‑convolution
    objective.  Both updates stay on the simplex.

3.  **Selection** – the globally best sequence seen is returned.  The code
    always produces a non‑empty list before the 1000 s budget expires.
"""

# ----------------------------------------------------------------------
# Imports & low‑level helpers
# ----------------------------------------------------------------------
import time
from typing import List, Tuple
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Fast proxy for ``evaluate_sequence`` when Σa = 1.
    Returns 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by sub‑gradient polishing
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape factories (all output vectors are already L1‑normalised)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β)‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_gamma_shape(n: int, gamma: float, delta: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{‑γ}.
    When γ≈0.5, δ≈0.5 the classic arcsine appears.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Linear mixture of arcsine and the uniform distribution."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Mirror‑descent / sub‑gradient polishing (same as in the reference code)
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step: float = 0.01,
) -> np.ndarray:
    """KL‑mirror (multiplicative) descent on the true max‑convolution."""
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.shape[0]
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return a


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 250_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a

# Simple differential‑evolution optimiser for a few shape parameters

def de_optimize(
    shape_type: str,
    n: int,
    bounds: List[Tuple[float, float]],
    rng: np.random.Generator,
    deadline: float,
    max_evals: int = 3000,
) -> Tuple[np.ndarray, float]:
    """
    Very small DE optimiser (pop≈max(10,10·dim)).  Returns the best
    normalised shape and its (proxy) score.
    """
    dim = len(bounds)
    popsize = max(10, 10 * dim)
    # initialise population uniformly inside bounds
    pop = np.column_stack(
        [rng.uniform(low, high, size=popsize) for (low, high) in bounds]
    )
    scores = np.empty(popsize, dtype=float)

    for i in range(popsize):
        a = _shape_from_params(shape_type, n, pop[i])
        scores[i] = compute_score(a)

    best_idx = int(np.argmin(scores))
    best_score = scores[best_idx]
    best_vec = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    evals = popsize

    while time.time() < deadline and evals < max_evals:
        for i in range(popsize):
            if time.time() >= deadline or evals >= max_evals:
                break

            # choose three distinct indexes different from i
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            donor = pop[a_idx] + F * (pop[b_idx] - pop[c_idx])
            donor = np.clip(
                donor,
                [low for (low, high) in bounds],
                [high for (low, high) in bounds],
            )
            # binomial crossover
            cross = rng.random(dim) < CR
            if not np.any(cross):
                cross[rng.integers(dim)] = True
            trial = np.where(cross, donor, pop[i])

            a_trial = _shape_from_params(shape_type, n, trial)
            score_trial = compute_score(a_trial)
            evals += 1

            if score_trial < scores[i]:
                pop[i] = trial
                scores[i] = score_trial
                if score_trial < best_score:
                    best_score = score_trial
                    best_vec = trial.copy()
        # end of generation
    return best_vec, best_score


def _shape_from_params(shape_type: str, n: int, params: np.ndarray) -> np.ndarray:
    """Dispatch helper for ``de_optimize``."""
    if shape_type == "gamma":
        gamma, delta, eps = params
        return generate_gamma_shape(n, gamma, delta, eps)
    if shape_type == "beta":
        alpha, beta, eps = params
        return generate_beta_shape(n, alpha, beta, eps)
    if shape_type == "mixture":
        (w_arc,) = params
        return generate_mixture_shape(n, w_arc)
    if shape_type == "power":
        p, eps = params
        return generate_power_shape(n, p, eps)
    if shape_type == "arcsine":
        return arcsine_initial(n)
    # fallback – uniform
    return np.full(n, 1.0 / n)

    if "height_sequence_1" in globals():

            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Parametric DE search on a few lengths
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]      # descending – larger first
    # proportional time slices (more for larger n)
    total_len = sum(lengths)
    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        # allocate roughly (n/total_len) of the remaining budget
        remaining = deadline - time.time()
        slice_time = remaining * (n / total_len) * 0.6
        slice_deadline = time.time() + slice_time
        print(f"\n--- DE search for length {n} (time slice ≈{slice_time:.1f}s) ---")

        # --------- gamma‑shape ------------------------------------------------
        gamma_bounds = [(0.2, 0.8), (0.01, 2.0), (0.0, 10.0)]   # γ, δ, ε
        vec, sc = de_optimize(
            "gamma",
            n,
            gamma_bounds,
            rng,
            deadline=slice_deadline,
            max_evals=4000,
        )
        a_gamma = _shape_from_params("gamma", n, vec)
        val = compute_score(a_gamma)
        if val < best_val:
            best_val, best_seq = val, a_gamma.tolist()
            print(f"  New best (gamma) score={val:.6f}")

        # --------- beta‑shape -------------------------------------------------
        beta_bounds = [(0.1, 10.0), (0.1, 10.0), (0.0, 5.0)]   # α, β, ε
        vec, sc = de_optimize(
            "beta",
            n,
            beta_bounds,
            rng,
            deadline=slice_deadline,
            max_evals=4000,
        )
        a_beta = _shape_from_params("beta", n, vec)
        val = compute_score(a_beta)
        if val < best_val:
            best_val, best_seq = val, a_beta.tolist()
            print(f"  New best (beta) score={val:.6f}")

        # --------- mixture (arcsine ↔ uniform) -------------------------------
        mix_bounds = [(0.0, 1.0)]   # weight of arcsine
        vec, sc = de_optimize(
            "mixture",
            n,
            mix_bounds,
            rng,
            deadline=slice_deadline,
            max_evals=2000,
        )
        a_mix = _shape_from_params("mixture", n, vec)
        val = compute_score(a_mix)
        if val < best_val:
            best_val, best_seq = val, a_mix.tolist()
            print(f"  New best (mix) score={val:.6f}")

        # --------- uniform / pure arcsine as cheap baselines -----------------
        a_uni = np.full(n, 1.0 / n)
        val = compute_score(a_uni)
        if val < best_val:
            best_val, best_seq = val, a_uni.tolist()
            print(f"  New best (uniform) score={val:.6f}")

        a_arc = arcsine_initial(n)
        val = compute_score(a_arc)
        if val < best_val:
            best_val, best_seq = val, a_arc.tolist()
            print(f"  New best (arcsine) score={val:.6f}")

        # ------------------------------------------------------------------
        # 2.  Polishing of the current global best for this length
        # ------------------------------------------------------------------
        if not best_seq:
            continue
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        a_arr = np.array(best_seq, dtype=float)

        # EK‑gradient polishing (short run – keep us inside the time budget)
        a_arr = exponentiated_gradient_polish(
            a_arr,
            deadline=deadline,
            max_iter=int(min(80_000, remaining * 1500)),
            step=0.008,
        )
        val = compute_score(a_arr)
        if val < best_val:
            best_val, best_seq = val, a_arr.tolist()
            print(f"  After EG polish new best score={val:.6f}")

        # final exact sub‑gradient (if we still have a few seconds)
        if time.time() < deadline - 3.0:
            a_arr = subgradient_descent_exact(
                a_arr,
                deadline=deadline,
                max_iter=180_000,
                step0=0.05,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"  After final sub‑gradient new best score={val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Global polishing (a last, longer KL‑mirror run)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\n--- Global polishing phase (final EG) ---")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = exponentiated_gradient_polish(
            a_arr,
            deadline=deadline,
            max_iter=int(min(200_000, (deadline - time.time()) * 1500)),
            step=0.006,
        )
        val = compute_score(a_arr)
        if val < best_val:
            best_val, best_seq = val, a_arr.tolist()
            print(f"  Global EG polish improved score={val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty, valid result
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 1024
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # 5.  Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
