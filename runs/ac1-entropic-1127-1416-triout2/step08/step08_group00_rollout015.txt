        # propose a local move:
        i, j = np.random.randint(n, size=2)

        # ensure nonneg

        cand = a.copy()

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score or np.random.rand() < np.exp(-(cur_score - best_score)/temperature):
            a = cand
            if cur_score < best_score:
                best_score = cur_score
                best = cand.copy()
        # update temperature
        temperature = T0 * (1 - step/max_steps)
        # maybe periodic random restart.

def simulated_annealing(a, time_deadline, max_steps):
    best = a.copy()
    best_score = compute_score(best)
    cur = a.copy()
    cur_score = best_score
    step_scale = 0.1

    for step in range(max_steps):
        if time.time() >= time_deadline:
            break
        # propose move
        i, j = np.random.randint(n, size=2)
        delta = step_scale * cur[i] * np.random.uniform(0, 1)  # random proportion
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        d = cand_score - cur_score
        if d < 0 or np.random.rand() < np.exp(-d / (T0 * (1 - step / max_steps))):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                step_scale = min(step_scale * 1.1, 0.5)
        else:
            step_scale *= 0.99
    return best, best_score

def lbfgs_optimize(start_a, time_deadline):
    # convert to u
    a = start_a
    u = np.log(a + 1e-12)  # ensure no zeros
    for beta in [20.0, 80.0, 200.0]:
        if time.time() >= time_deadline: break
        def fun(u_vec):
            a_vec = np.exp(u_vec - np.max(u_vec))
            a_vec = a_vec / a_vec.sum()
            conv = convolution_fft(a_vec, a_vec)
            maxc = conv.max()
            # compute stable weights
            w = np.exp(beta * (conv - maxc))
            w_sum = w.sum()

"""
Propose a non‑negative step‑function (a probability vector) that
minimises `evaluate_sequence`.  The search uses three independent
strategies that respect the 1000 s wall‑clock budget:

1. **L‑BFGS on a smooth max surrogate** – the objective
   `(1/β)·log Σ exp(β·conv(a,a))` is minimized with the variables
   re‑parameterised by a soft‑max (`a = softmax(u)`) to enforce the
   simplex constraints automatically.  A small schedule of β values
   (20 → 80 → 200 → 500) provides a coarse‑to‑fine optimisation.
   Many random restarts are performed in the available time.

2. **Simulated‑annealing polishing** – starting from the best vector
   found so far, random mass‑transfer moves are proposed and accepted
   according to a temperature schedule.  Only moves that improve the
   exact score or are accepted probabilistically are kept; the step
   size adapts automatically.

3. **Fallback** – if nothing is found a simple uniform vector is
   returned.

All helper functions are defined at module level (no closures) and
the code uses only `numpy`, `scipy.optimize` and the provided
`evaluate_sequence`.  The routine respects the 1000 s time limit and
returns a plain Python list of floats.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for large arrays)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact `evaluate_sequence` objective for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(u: np.ndarray) -> np.ndarray:
    """
    Numerically stable soft‑max (produces a probability vector).
    """
    um = u - np.max(u)
    e = np.exp(um)
    return e / e.sum()


# ----------------------------------------------------------------------
# 1. L‑BFGS optimisation of a smooth max surrogate
# ----------------------------------------------------------------------


def lbfgs_optimize(start_a: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    minimise a smoothed max of the convolution using L‑BFGS.
    The vector is parameterised by `u` such that a = softmax(u),
    guaranteeing non‑negativity and Σa = 1.
    """
    # initialise the unconstrained parameters
    u = np.log(start_a + 1e-12)

    # β schedule – larger β gives a tighter approximation to the true max
    betas = [20.0, 80.0, 200.0, 500.0]

    for beta in betas:
        if time.time() >= time_deadline:
            break

        # objective + gradient for a given β
        def fun(u_vec, beta=beta):
            a = softmax(u_vec)                     # probability vector
            conv = convolution_fft(a, a)           # length 2n‑1
            maxc = conv.max()

            # stable weights for the log‑sum‑exp
            w = np.exp(beta * (conv - maxc))
            w_sum = w.sum()
            loss = maxc + np.log(w_sum) / beta      # (1/β)·log‑sum‑exp

            w_norm = w / w_sum

            # gradient w.r.t. a (weighted sum of sub‑gradients)
            grad_a = 2.0 * convolution_fft(a[::-1], w_norm)[len(a) - 1:
                                                            len(a) - 1 + len(a)]

            # chain rule: ∂L/∂u = a ⊙ (∂L/∂a – ⟨a,∂L/∂a⟩)
            grad_u = a * (grad_a - np.dot(a, grad_a))
            return loss, grad_u

        # L‑BFGS‑B – only needs the gradient, bounds are handled by the soft‑max
        res = minimize(fun,
                       u,
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": 120, "ftol": 1e-12, "disp": False})
        u = res.x

    # final probability vector
    a_opt = softmax(u)
    return a_opt

# 2. Simulated‑annealing polishing
# ----------------------------------------------------------------------


def annealing_optimize(start_a: np.ndarray, time_deadline: float,
                       max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a temperature schedule.
    Random mass transfers between two coordinates are proposed;
    moves are accepted if they improve the exact score or with
    probability exp(−Δ/T) where T decays linearly.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    # temperature schedule (linear from T0 to near 0)
    T0 = 0.15
    step = 0
    while step < max_steps and time.time() < time_deadline:
        # propose a small transfer of mass from i to j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # fraction of the source mass to move (adaptive)
        delta = 0.07 * a[i] * np.random.rand()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # stay on the simplex
        cand = project_to_simplex(cand)

        cur_score = compute_score(a)
        cand_score = compute_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            T = T0 * (1.0 - step / max_steps)
            accept = np.random.rand() < np.exp(-(cand_score - cur_score) / (T + 1e-12))

        if accept:
            a = cand
            cur_score = cand_score

        # record the best seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # adapt the transfer magnitude slowly
        if step % 4000 == 0 and step > 0:
            # if we are improving, become bolder
            if best_score < compute_score(best_a):
                pass  # keep current scale
            else:
                # otherwise shrink a bit
                a = project_to_simplex(a)

        step += 1

    return best_a, best_score

def random_start(n: int) -> np.ndarray:
    """Return a random point on the simplex of dimension `n`."""
    if np.random.rand() < 0.5:
        # dense Dirichlet‑(1) draw (uniform on simplex)
        a = np.random.rand(n)
        a = a / a.sum()
    else:
        # a sparse start – mass on √n positions
        k = max(2, int(np.sqrt(n)))
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k
        a = a / a.sum()
    return a
