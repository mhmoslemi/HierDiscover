"""
LBFGS‑based continuation optimiser for the max‑convolution objective.
Starts from the classic central‑binomial (inverse‑square‑root) shape,
optimises a smooth approximation of max(conv) with increasing
temperature β, then polishes the exact L∞ objective with a projected
sub‑gradient descent.  No LP is used – the whole search runs within the
1000 s time‑budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)), with ∑a = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()

def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Return (f, grad) for the smoothed max‑convolution:
        f(·) = max(conv) + (1/β)·log Σ exp(β·(conv‑max(conv)))
    Gradients are taken w.r.t. the logits of the soft‑max parametrisation,
    i.e. a_i = softmax(logits)_i.
    """
    # a lives on the simplex (strictly positive)
    a = softmax_logits(logits)

    # --- 1. full self‑convolution ------------------------------------
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # --- 2. smooth max (log‑sum‑exp) -------------------------------
    # weights w_k  = exp(β·(conv_k‑maxc)) / Σ exp(β·(conv‑maxc))
    exp_weights = np.exp(beta * (conv - maxc))
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # --- 3. gradient wrt a -----------------------------------------
    w = exp_weights / Z                     # shape (2n‑1,)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # --- 4. chain rule back to logits -------------------------------
    # ∂f/∂logits = a * ( ∂f/∂a  –  (∂f/∂a ⋅ a) )
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits

def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 8,
                       maxiter_per_step: int = 120) -> tuple[np.ndarray, float]:
    """
    Optimise the smooth surrogate by a continuation (annealing) on β.
    Returns the best sequence found (still on the simplex) and its exact score.
    """
    # initialise logits from start_a (log‑softmax parametrisation)
    logits = np.log(start_a + 1e-20)

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    # geometrically spaced β values
    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for idx, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break
        time_left = deadline - time.time()
        # allow a small cushion for the next phases
        max_iter = min(maxiter_per_step, max(10, int(time_left * 30)))  # rough 30 iters per sec
        # L‑BFGS‑B on the smooth objective
        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})
        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.2f} (step {idx})] new best score = {best_score:.6f}")
    return best_a, best_score

def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 400_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective 2·n·max(conv(a,a)).
    Simple, reliable polishing after the smooth phase.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    t = 1
    while time.time() < deadline and t <= max_iter:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (the max)

"""
L‑BFGS‑based continuation optimiser for the max‑convolution objective.

The programme proceeds in four phases:
  1. Initialise with the classic inverse‑square‑root (central‑binomial)
     shape, already known to give a good bound ≈ π/2.
  2. Optimise a smooth log‑sum‑exp surrogate of the max‑convolution
     using L‑BFGS‑B while annealing the temperature β (continuation
     method).  The parameters are kept on the probability simplex via a
     soft‑max re‑parametrisation, which eliminates the need for explicit
     projection inside the optimiser.
  3. Polishing with a projected sub‑gradient descent on the exact
     2·n·max(conv) objective.
  4. A tiny stochastic hill‑climb (swap / fractional mass transfer) for
     any remaining time.

All helper functions are top‑level; no closures or lambdas are used.
The routine respects the 1000 s wall‑clock budget and returns the best
sequence found as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (efficient for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a vector already on the
    probability simplex (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically‑stable soft‑max that always sums to 1."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient (combined)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(logits: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = max(conv) + (1/β)·log Σ exp(β·(conv‑max(conv)))
    and g is the gradient of f with respect to the soft‑max logits.
    The parametrisation a = softmax(logits) guarantees a ≥ 0 and ∑a = 1.
    """
    # a lives on the simplex
    a = softmax_logits(logits)

    # 1) self‑convolution
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # 2) smooth max (log‑sum‑exp)
    exp_weights = np.exp(beta * (conv - maxc))  # centred for stability
    Z = exp_weights.sum()
    f = maxc + np.log(Z) / beta

    # 3) gradient w.r.t. a
    w = exp_weights / Z                         # (2n‑1,) soft‑max weights
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]

    # 4) chain rule back to logits
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return f, grad_logits


# ----------------------------------------------------------------------
# Continuation optimiser (β‑annealing) using L‑BFGS‑B
# ----------------------------------------------------------------------


def lbfgs_continuation(start_a: np.ndarray,
                       beta_start: float,
                       beta_end: float,
                       deadline: float,
                       schedule_steps: int = 9,
                       maxiter_per_step: int = 150):
    """
    Anneal the smoothing parameter β while optimising the smooth surrogate
    with L‑BFGS‑B.  Returns the best sequence (still on the simplex) and
    its exact score.
    """
    # initialise logits from the starting simplex vector
    logits = np.log(start_a + 1e-20)

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    betas = np.geomspace(beta_start, beta_end, num=schedule_steps)

    for step, beta in enumerate(betas, 1):
        if time.time() >= deadline:
            break

        # a generous per‑step budget, but never exceed the remaining time
        time_left = deadline - time.time()
        max_iter = min(maxiter_per_step,
                       max(20, int(time_left * 30)))   # ≈30 iters per second

        res = minimize(smooth_obj_and_grad,
                       logits,
                       args=(beta,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        logits = res.x
        a = softmax_logits(logits)
        cur_score = compute_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[β={beta:.2f} (step {step})] new best score = {best_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
