"""
search for a non‑negative step sequence that minimises ``evaluate_sequence``.

The strategy is:

1.  Pick a relatively large length (8192) – larger n reduces the
    theoretical lower bound 2 n/(2 n‑1) ≈ 1.
2.  Build a small, diverse set of analytic “seeds’’:
    – arcsine‑type (Beta(½,½))
    – Beta(α,α) with α∈{0.1,0.2,0.3,0.5,0.8,1.0,1.5,2.0}
    – a few random Dirichlet draws.
3.  For each seed run a continuation‑Adam optimiser on the smooth
    approximation  Lβ(a)=log‑sum‑exp(β·conv(a,a)) .  β is increased up to
    5000, giving a tight approximation of the true max‑convolution.
4.  Keep the best candidate (exact objective via ``evaluate_sequence``).
5.  Refine that candidate with an exponential‑gradient (mirror) descent
    that uses the exact sub‑gradient of the true max‑convolution.
6.  Polish the result with a long projected sub‑gradient descent on the
    exact objective.
7.  A cheap random pairwise mass‑transfer local search explores the
    neighbourhood of the polished solution.

All phases respect the overall time budget (default 1000 s).  The
function returns a list of floats that is always a valid, non‑negative
step function.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Cheap surrogate (scale‑invariant) used for seed ranking and local
# optimisation
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Return 2·n·max(conv(a,a))/sum(a)^2 without the clipping performed by
    ``evaluate_sequence``.  ``a`` must be non‑negative.
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete approximation of Beta(½,½)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float = 1e-12
) -> np.ndarray:
    """Beta(α,β) shape on {0,…,n‑1}.  Normalised to unit sum."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with random concentration)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)   # from 0.01 to ~5
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx_i = 1, x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent for
    f(a)=2·n·max(conv(a,a))/sum(a)^2.
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)          # length = 3·n‑2
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 20_000 == 0:
            print(f"    Sub‑grad it={it:,d} surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) minimisation with Adam, continuation on β
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 5000.0,
    ramp_iters: int = 300_000,
    lr: float = 0.03,
) -> np.ndarray:
    """
    Minimises L_β(a)=log‑sum‑exp(β·conv(a,a)) while staying on the simplex
    via the soft‑max parametrisation of ``z``.  β is linearly increased from
    ``beta0`` to ``beta_max`` (continuation).
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of log‑sum‑exp w.r.t. a:   d/da Σ w_i * (a★e_i) = 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 8_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) with exact max‑sub‑gradient
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.15,
    decay: float = 0.9999,
) -> np.ndarray:
    """
    Exponential‑gradient (KL‑mirror) descent on the exact max‑convolution
    objective.  The iterate stays on the simplex automatically.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay

        # occasional progress report
        if int(step * 1e6) % 200_000 == 0:
            print(f"    Mirror step={step:.6f} surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Simple random local search with pairwise mass transfers
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.025,
) -> np.ndarray:
    """
    Inexpensive pairwise mass‑transfer moves while the time budget permits.
    Each move selects two indices i≠j and transfers a random fraction
    (up to ``max_transfer``) of the mass at i to j, then re‑normalises.
    Only improvements in the cheap surrogate ``fast_score`` are kept.
    """
    a = a0.copy()
    best_score = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * rng.random() * max_transfer
        if eps <= 0.0:
            continue
        a[i] -= eps
        a[j] += eps
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
            continue
        a /= s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            a[j] -= eps
            a[i] += eps
            a /= a.sum()   # re‑normalise rounding errors

    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The routine combines analytic seeds,
    continuation Adam, mirror‑descent, exact sub‑gradient polishing and
    a final random local search.  All work respects the overall
    time budget (default 1000 s).  The function always returns a
    non‑empty, non‑negative list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a reasonably large length – 8192 gives a good trade‑off
    # ------------------------------------------------------------------
    n = 8192
    print(f"Using length n = {n}")

    # ------------------------------------------------------------------
    # 2. Build a compact pool of diverse seeds
    # ------------------------------------------------------------------
    seeds: List[np.ndarray] = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),                     # uniform fallback
    ]

    # Beta(α,α) with a handful of α values (including α<0.5)
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0]:
        seeds.append(generate_beta_shape(n, alpha, alpha, eps=1e-12))

    # a few random Dirichlet draws
    for _ in range(5):
        seeds.append(generate_random_dirichlet(n, rng))

    # ------------------------------------------------------------------
    # 3. Keep only the most promising seeds according to the cheap surrogate
    # ------------------------------------------------------------------
    scored: List[Tuple[float, np.ndarray]] = [(fast_score(v), v) for v in seeds]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:4]]   # at most four seeds

    best_seq: List[float] = []
    best_official: float = float("inf")
    best_surrogate: float = float("inf")

    # ------------------------------------------------------------------
    # 4. Allocate time for the seed optimisation phase
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # reserve roughly 30 % of the budget for polishing / local search
    seed_budget = remaining * 0.7
    per_seed = max(5.0, seed_budget / max(len(top_seeds), 1))

    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Seed {idx}/{len(top_seeds)} (budget ≈{per_seed:.1f}s) ---")
        seed_deadline = time.time() + per_seed

        # initialise Adam via a log‑softmax (avoids log(0))
        eps = 1e-12
        z0 = np.log(seed_vec + eps)

        # --------------------------------------------------------------
        # 4a. Continuation Adam optimisation (smooth‑max)
        # --------------------------------------------------------------
        a_opt = smooth_adam_optimize(
            z0,
            seed_deadline,
            n=n,
            beta0=5.0,
            beta_max=5000.0,
            ramp_iters=300_000,
            lr=0.03,
        )

        # --------------------------------------------------------------
        # 4b. Exact evaluation of the Adam result
        # --------------------------------------------------------------
        try:
            val = evaluate_sequence(a_opt.tolist())
            print(f"   Official score after Adam = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, a_opt.tolist()
                best_surrogate = fast_score(a_opt)
                print("   *** New global best! ***")
        except Exception as e:
            print(f"   Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 5. Mirror‑descent refinement on the current best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nRefining with mirror descent …")
        md_deadline = deadline - 2.0
        a_mirror = mirror_descent_optimize(
            np.array(best_seq, dtype=float),
            md_deadline,
            step0=0.15,
            decay=0.9999,
        )
        try:
            val_m = evaluate_sequence(a_mirror.tolist())
            print(f"   Mirror‑refined official score = {val_m:.6f}")
            if val_m < best_official:
                best_official, best_seq = val_m, a_mirror.tolist()
                best_surrogate = fast_score(a_mirror)
                print("   *** Updated best after mirror step ***")
        except Exception as e:
            print(f"   Mirror evaluation error: {e}")

    # ------------------------------------------------------------------
    # 6. Final polishing with exact sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\nFinal polishing with exact sub‑gradient descent …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=500_000,
            step0=0.07,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_official:
                best_official, best_seq = final_val, a_arr.tolist()
                best_surrogate = fast_score(a_arr)
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final evaluation error: {e}")

    # ------------------------------------------------------------------
    # 7. Random local search around the polished solution (if time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:
        print("\nRandom local search around current best …")
        a_start = np.array(best_seq, dtype=float)
        a_refined = random_local_search(
            a_start,
            deadline,
            rng,
            max_transfer=0.02,
        )
        try:
            val_loc = evaluate_sequence(a_refined.tolist())
            print(f"   Local‑search official score = {val_loc:.6f}")
            if val_loc < best_official:
                best_official, best_seq = val_loc, a_refined.tolist()
                best_surrogate = fast_score(a_refined)
                print("*** Best improved by local search ***")
        except Exception as e:
            print(f"   Local‑search evaluation error: {e}")

    # ------------------------------------------------------------------
    # 8. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 1024
        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
