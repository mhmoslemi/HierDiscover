def subgradient_descent_momentum(initial_a, deadline, max_iter=200_000, step0=0.1, beta=0.9):
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2*n-1
    vel = np.zeros_like(a)
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = conv.max()
        # mask for near maximal entries
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)
        # optionally expand mask to top few entries
        # We'll keep as mask
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
        vel = beta * vel + (1 - beta) * grad  # momentum
        eta = step0 / np.sqrt(it)
        a = a - eta * vel
        a = project_to_simplex(a)
        # optionally track best surrogate
        if it % 10000 == 0:
            s = compute_score(a)

"""
Propose a step‑function sequence via multi‑start momentum sub‑gradient descent.

Algorithm
---------
1. **Generate a diverse pool of seed shapes** (uniform, arcsine, power‑law,
   symmetric Beta, and random Dirichlet draws) on the same length as the
   fallback construction (or 2048 if none is available).  The cheap surrogate
   `2·n·max(conv(a,a))` (identical to the official score when Σa=1) is used to
   rank the seeds.

2. **Select the best few seeds** (default 3) and improve each of them by
   projected sub‑gradient descent with a simple momentum term.  The gradient
   of the true `max(conv)` is obtained from the active set – all convolution
   indices whose value is within a relative tolerance (`rel_tol`) of the
   current maximum.  A decreasing step size `η_t = step0 / sqrt(t)` together
   with momentum (`β≈0.9`) accelerates convergence.  The best refined
   sequence among the runs is kept.

3. **Polish the winner** with exponentiated‑gradient (mirror‑descent) to
   further reduce the surrogate, again respecting the simplex constraint.

4. **Official evaluation** of the final candidate via the provided
   `evaluate_sequence`.  If for any reason no viable sequence was produced,
   a uniform fallback of length 1024 is returned.

All stages respect the global wall‑clock budget (default 1000 s) and print
progress information.  The implementation relies only on NumPy and the
already‑imported `evaluate_sequence` function.
"""

import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """
    Generalised arcsine shape:
    a_i ∝ ((i+½)*(n‑½‑i) + ε)^{-p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def beta_symmetric_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on n points.
    Uses midpoint rule to avoid zero^negative‑exponent problems.
    """
    i = np.arange(n, dtype=float)
    # mid‑point locations in (0,1)
    x = (i + 0.5) / n
    # add a tiny epsilon to keep the ends finite
    eps = 1e-12
    vals = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, alpha - 1.0)
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent w.r.t. KL).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def subgradient_descent_momentum(initial_a: np.ndarray,
                                 deadline: float,
                                 max_iter: int = 500_000,
                                 step0: float = 0.5,
                                 beta: float = 0.9,
                                 rel_tol: float = 0.001) -> np.ndarray:
    """
    Projected sub‑gradient descent with a simple momentum term on the true
    max‑convolution objective.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1
    vel = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)
    stagnation = 0
    log_interval = 25000

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(conv.max())

        # mask of indices whose convolution value is close to the maximum
        mask = (conv >= max_val * (1.0 - rel_tol)).astype(float)
        if mask.sum() == 0:
            mask = (conv == max_val).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # momentum update (exponential smoothing of the sub‑gradient)
        vel = beta * vel + (1.0 - beta) * subgrad

        eta = step0 / np.sqrt(it)
        a = a - eta * vel
        a = project_to_simplex(a)

        # occasional progress report and best‑so‑far tracking
        if it % log_interval == 0:
            cur_score = compute_score(a)
            print(f"    Subgrad‑momentum it={it:,d} surrogate={cur_score:.6f}")

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # If we have not improved for a long stretch, reset momentum
        if stagnation >= 200_000:
            vel.fill(0.0)
            stagnation = 0

    return best_a

# Main entry point

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The procedure builds several seed shapes, refines the most promising
    with momentum sub‑gradient descent, finishes with an EG polish and
    returns the best sequence found before the wall‑clock deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Determine the length to work with (use fallback if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            n = len(fallback)
            fallback_seq = np.array(fallback, dtype=float)
            fallback_seq = fallback_seq / fallback_seq.sum()
        except Exception:
            n = 2048
            fallback_seq = None
    else:
        n = 2048
        fallback_seq = None

    print(f"\nWorking length n = {n}")

    # ------------------------------------------------------------------
    # Stage 1 – build a rich pool of seed sequences
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(("uniform", np.full(n, 1.0 / n)))

    # arcsine
    candidates.append(("arcsine", arcsine_initial(n)))

    # power‑law family
    for p in np.linspace(0.2, 1.5, 7):
        candidates.append((f"power_{p:.2f}", generate_power_shape(n, p)))

    # symmetric Beta family
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        candidates.append((f"beta_{alpha:.1f}", beta_symmetric_shape(n, alpha)))

    # random Dirichlet draws (different concentration parameters)
    for alpha in [0.01, 0.05, 0.1, 0.5, 1.0]:
        for _ in range(3):
            draws = rng.gamma(alpha, 1.0, size=n)
            seq = draws / draws.sum()
            candidates.append((f"dir_alpha_{alpha}", seq))

    # include fallback if we have one
    if fallback_seq is not None:
        candidates.append(("fallback", fallback_seq))

    # evaluate all candidates with the fast surrogate
    print("\n--- Stage 1: evaluating seed pool ---")
    scored = []
    for name, seq in candidates:
        score = compute_score(seq)
        scored.append((score, seq))
        print(f"  {name:>15}: surrogate={score:.6f}")

    # keep the top few seeds for intensive refinement
    scored.sort(key=lambda x: x[0])
    top_k = 3
    top_seeds = [seq for _, seq in scored[:top_k]]
    best_surrogate = scored[0][0]
    best_seq = top_seeds[0].copy()
    print(f"\nBest surrogate after Stage 1: {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – momentum sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        per_seed_time = remaining / (len(top_seeds) + 1)
        print("\n--- Stage 2: momentum sub‑gradient descent ---")
        for idx, seed_seq in enumerate(top_seeds, 1):
            sub_deadline = time.time() + per_seed_time
            print(f"Refining seed {idx}/{len(top_seeds)}")
            refined = subgradient_descent_momentum(
                seed_seq,
                sub_deadline,
                max_iter=500_000,
                step0=0.5,
                beta=0.9,
                rel_tol=0.001,
            )
            s_ref = compute_score(refined)
            print(f"  refined surrogate={s_ref:.6f}")
            if s_ref < best_surrogate:
                best_surrogate = s_ref
                best_seq = refined.copy()
    else:
        print("\nSkipping Stage 2 (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Stage 3: exponentiated‑gradient polish ---")
        polished = eg_refine(best_seq, deadline, max_iter=200_000, eta0=0.3)
        s_polish = compute_score(polished)
        print(f"  after EG polish surrogate={s_polish:.6f}")
        if s_polish < best_surrogate:
            best_surrogate = s_polish
            best_seq = polished.copy()
    else:
        print("\nSkipping EG polish (insufficient time)")

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\nOfficial final score = {final_val:.6f}")
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if best_seq.size == 0:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n)
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Fallback to uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {final_val:.6f}")
    return best_seq.tolist()
