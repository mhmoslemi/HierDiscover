def solve_lp_max_sum(A, M, warm_start=None):
    n = A.shape[1]
    b = cp.Variable(n)
    constraints = [A @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    if warm_start is not None:
        b.value = warm_start
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-8, reltol=1e-6, feastol=1e-8)
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    bv = b.value
    if bv is None:
        return None
    return np.maximum(bv, 0.0)
```

"""
Non‑negative step‑function optimizer for `evaluate_sequence`.

Algorithm
---------
1. **Start** – central‑binomial coefficients (a_i = C(2i,i)/4^i) give a
   near‑optimal shape.  They are projected onto the probability simplex
   (sum = 1).

2. **LP refinement** – for a modest length (n≈1024) we repeatedly
   construct the exact convolution matrix `A` such that `A @ b = conv(f,b)`.
   We then solve the linear program
        maximize   Σ b_j
        subject to A @ b ≤ M   (M = max(conv(f,f)))
                    b ≥ 0
   The solution `b` is projected onto the simplex and mixed with the
   current vector `f ← (1‑α)f + α b` (α=0.25).  Each mixing step strictly
   reduces the worst auto‑convolution entry, and we iterate until the
   time budget for LPs is exhausted.

3. **Upscaling** – the improved small vector is interpolated to a larger
   length (4096 or 8192) and projected onto the simplex.

4. **Smooth optimisation** – minimise the log‑sum‑exp surrogate of
   `max(conv(a,a))` using Adam with an annealed smoothing parameter.
   This stage runs until the allocated time runs out.

5. **Projected sub‑gradient descent** – polishing on the true
   `max(conv)` objective.

6. **Stochastic hill‑climb** – cheap mass‑transfer moves (swap or
   fractional transfer) with simplex projection to escape remaining
   plateaux.

All phases respect the global deadline (default 1000 s).  The best
sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np
import scipy.sparse as sp

# cvxpy is optional – the LP phase will be skipped if unavailable.
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for many calls)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σ x = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution matrix for the LP (exact construction)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Return a sparse Toeplitz matrix A (size (2n‑1) × n) such that
    A @ b = conv(f, b) for any b ≥ 0.
    """
    n = f.shape[0]
    # rows[i, j] = i + j   (i column index, j = 0…n‑1)
    row_idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    rows = row_idx.ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(f, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_max_sum(A: sp.csr_matrix, M: float,
                     warm_start: np.ndarray | None = None) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Returns the optimal b (as a dense numpy array) or None on failure.
    """
    if cp is None:                         # pragma: no cover
        return None
    n = A.shape[1]
    b = cp.Variable(n)
    constraints = [A @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    if warm_start is not None:
        b.value = warm_start
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-8, reltol=1e-6, feastol=1e-8)
    except Exception:                     # pragma: no cover
        return None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    bv = b.value
    if bv is None:
        return None
    return np.maximum(bv, 0.0)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    deadline: float,
    beta_start: float = 6.0,
    beta_end: float = 200.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
    record_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    The surrogate objective is
        (1/β) * log Σ exp(β * conv(a,a))
    which approaches max(conv) as β → ∞.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # anneal smoothing parameter β
        t = it / max(1, int(deadline - time.time()))  # dummy, real anneal below
        # A linear schedule in log‑space:
        frac = min(1.0, it / 200_000)   # cap after ~200k iterations
        beta = beta_start * ((beta_end / beta_start) ** frac)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights (numerically stable)
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smoothed objective:
        # ∇_a  (1/β) log Σ exp(β conv) = Σ w_k * ∂ conv_k / ∂ a
        # where ∂ conv_k / ∂ a = 2 * (a * shifted indicator)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)
        a = a - update
        a = project_to_simplex(a)

        if it % record_every == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.07,
    jitter_interval: int = 1000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if it % jitter_interval == 0:
            # small random jitter to escape flat zones
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simple mass transfers)
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: list[float],
    deadline: float,
    max_steps: int = 600_000,
) -> tuple[list[float], float]:
    """
    Tiny stochastic moves (swap or fractional transfer) with simplex projection.
    """
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:   # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                        # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling utility (linear interpolation + simplex projection)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear interpolation to a longer length, followed by simplex projection.
    """
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)
