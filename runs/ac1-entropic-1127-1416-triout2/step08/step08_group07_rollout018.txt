import time
import numpy as np
import math
from typing import List

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def fast_score(a: np.ndarray) -> float:
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float('inf')
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)

def generate_power_shape(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot

def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-12) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    # Euclidean projection onto simplex
    n = v.size
    if np.all(v <= 0):
        return np.full_like(v, 1.0/n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def mirror_descent_optimize(a0: np.ndarray, deadline: float, step0: float = 0.15, decay: float = 0.9999) -> np.ndarray:
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0
    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance to capture all max entries (possibly multiple)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay
        # occasionally log progress
        if int(step * 1e6) % 200_000 == 0:
            # Use fast_score for cheap logging
            print(f"    Mirror step={step:.6f} surrogate≈{fast_score(a):.6f}")
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step function (non‑negative sequence) that minimises the
    official objective.  The algorithm builds a rich parametric family of
    U‑shaped sequences (power‑law and Beta families) on several lengths,
    performs a dense grid search using the cheap surrogate ``fast_score``,
    then refines the best candidate with mirror‑descent (exact sub‑gradient)
    for the remaining time budget.  The final sequence is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5    # safety margin

    best_seq: List[float] = []
    best_score = float('inf')
    best_fast = float('inf')

    # ------------------------------------------------------------------
    # 1.  Choose sequence lengths – larger lengths give better lower bound.
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]   # descending order, stop early if time runs short

    # Parameter grids
    power_grid = np.linspace(0.25, 0.75, 21)   # includes arcsine (0.5)
    weight_grid = np.linspace(0.0, 1.0, 21)    # mixture weight of the power shape
    beta_grid = np.linspace(0.25, 0.75, 21)    # symmetric beta(α,α)

    for n in lengths:
        if time.time() >= deadline - 2.0:
            break
        print(f"\n--- Exploring length {n} (time left {deadline - time.time():.1f}s) ---")
        uniform = np.full(n, 1.0 / n, dtype=float)

        # ------------------------------------------------------------------
        # 1a. Power‑law family with uniform mixture
        # ------------------------------------------------------------------
        for p in power_grid:
            shape_p = generate_power_shape(n, p)
            for w in weight_grid:
                if time.time() >= deadline - 1.0:
                    break
                # mixture: w * shape_p + (1-w) * uniform, already sums to 1
                cand = w * shape_p + (1.0 - w) * uniform
                sc = fast_score(cand)
                if sc < best_fast:
                    best_fast = sc
                    best_seq = cand.tolist()
                    best_score = sc
                    # optional log
                    print(f"  New best (power) n={n} p={p:.3f} w={w:.2f} fast≈{sc:.6f}")

        # ------------------------------------------------------------------
        # 1b. Symmetric Beta family (α=β) with uniform mixture
        # ------------------------------------------------------------------
        for alpha in beta_grid:
            shape_b = generate_beta_shape(n, alpha, alpha)
            for w in weight_grid:
                if time.time() >= deadline - 1.0:
                    break
                cand = w * shape_b + (1.0 - w) * uniform
                sc = fast_score(cand)
                if sc < best_fast:
                    best_fast = sc
                    best_seq = cand.tolist()
                    best_score = sc
                    print(f"  New best (beta) n={n} α={alpha:.3f} w={w:.2f} fast≈{sc:.6f}")

        # ------------------------------------------------------------------
        # 1c. Random Dirichlet draws (few) – cheap for diversity
        # ------------------------------------------------------------------
        for _ in range(5):
            if time.time() >= deadline - 1.0:
                break
            # random concentration between 0.2 and 3
            conc = 10 ** rng.uniform(-0.7, 0.5)
            a = rng.gamma(shape=conc, scale=1.0, size=n)
            a_sum = a.sum()
            if a_sum == 0:
                a = uniform
            else:
                a /= a_sum
            sc = fast_score(a)
            if sc < best_fast:
                best_fast = sc
                best_seq = a.tolist()
                best_score = sc
                print(f"  New best (rand) fast≈{sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Final refinement with exact mirror‑descent (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:
        a0 = np.array(best_seq, dtype=float)
        print("\nRefining best candidate with mirror descent …")
        a_refined = mirror_descent_optimize(a0, deadline, step0=0.15, decay=0.9995)
        # Evaluate the refined candidate with fast_score and keep if better
        sc_ref = fast_score(a_refined)
        if sc_ref < best_fast:
            best_fast = sc_ref
            best_seq = a_refined.tolist()
            print(f"  Mirror‑descent improved fast_score to {sc_ref:.6f}")

    # ------------------------------------------------------------------
    # 3.  Official evaluation (optional, for logging)
    # ------------------------------------------------------------------
    try:
        official = evaluate_sequence(best_seq)
        print(f"\nOfficial objective of selected sequence = {official:.6f}")
    except Exception as e:
        print(f"\nOfficial evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        print("Fallback to uniform length 512")

    print(f"\nSearch finished. Best fast_score ≈ {best_fast:.6f}")
    return best_seq

"""
Search for a non‑negative step sequence that minimises the official objective

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three phases:

1.  Choose a single (large) length – a larger n gives a lower theoretical
    bound 2·n/(2n‑1) → 1.0.
2.  Initialise several diverse seeds (uniform, the symmetric arcsine‑type
    Beta(½,½) shape and a handful of random Dirichlet draws).
3.  For every seed run a long projected sub‑gradient descent on the exact
    objective (the sub‑gradient of the max‑convolution is computed analytically
    via FFT).  The best iterate seen during the descent is kept.
4.  The final best sequence is returned.  All work is bounded by the supplied
    time budget (default 1000 s).

The sub‑gradient descent is the core optimisation step – it directly minimises
the true objective (no smooth surrogate) and typically pushes the symmetric
arcsine shape into a slightly asymmetric one that attains a lower maximum
convolution value.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Utility – next power of two for zero‑padding in FFT
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Linear convolution via FFT (O(N log N))
# ----------------------------------------------------------------------


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a★b using real FFT."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# A cheap surrogate used for logging (exact objective without clipping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/sum(a)^2 (the exact metric without clipping)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex Σx_i = 1, x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Symmetric arcsine shape ≈Beta(½,½) – the classic starting point."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + 1e-12)
    vals /= vals.sum()
    return vals


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with random concentration)."""
    # concentration ∈ [0.2, 3] (log‑uniform)
    conc = 10 ** rng.uniform(-0.7, 0.5)
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    total = a.sum()
    if total == 0:
        return np.full(n, 1.0 / n)
    return a / total


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgradient_descent(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.1,
    max_iter: int = 10_000_000,
) -> np.ndarray:
    """
    Projected sub‑gradient descent for
        f(a) = 2·n·max(conv(a,a)) / (sum(a)^2) .
    The iterate stays on the simplex via Euclidean projection.
    The function returns the best iterate (lowest objective) seen
    before the deadline expires.
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    best_a = a.copy()
    # because sum(a) == 1 after projection we can use the simpler expression
    conv = convolution_fft(a, a)
    best_score = 2.0 * n * np.max(conv)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # ‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑
        # sub‑gradient of the max‑convolution
        # ------------------------------------------------------------------
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # step‑size schedule (classical 1/√t)
        eta = step0 / math.sqrt(it)

        a = a - eta * subgrad
        a = _project_to_simplex(a)

        # keep the best iterate seen so far
        current_score = 2.0 * n * mx
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            # occasional log
            print(
                f"    iter {it:,d}  surrogate≈{best_score:.6f}  step={eta:.6f}"
            )

        # a tiny safety printout every 50 k iterations
        if it % 50_000 == 0:
            print(f"    –– iter {it:,d} –– surrogate≈{best_score:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises ``evaluate_sequence``.
    The routine:
        1. picks a relatively large length (default 8192);
        2. builds a few diverse seeds (uniform, arcsine, random Dirichlet);
        3. runs a long projected sub‑gradient descent from each seed;
        4. returns the best sequence found (always a valid probability vector).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.4   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length – larger n gives a lower theoretical lower bound.
    # ------------------------------------------------------------------
    n = 8192                     # fits comfortably in memory & time budget
    print(f"Using length n={n}")

    # ------------------------------------------------------------------
    # 2. Build a small pool of seeds
    # ------------------------------------------------------------------
    seeds = [
        np.full(n, 1.0 / n),          # uniform
        arcsine_initial(n),           # symmetric arcsine (Beta(½,½))
    ]

    # add a few random Dirichlet draws with various concentrations
    for _ in range(6):
        seeds.append(generate_random_dirichlet(n, rng))

    # ------------------------------------------------------------------
    # 3. Run sub‑gradient descent from each seed (time‑sliced)
    # ------------------------------------------------------------------
    best_seq: List[float] = []
    best_obj: float = float("inf")
    best_surrogate: float = float("inf")

    for idx, seed_vec in enumerate(seeds, start=1):
        if time.time() >= deadline - 2.0:
            break
        # allocate a slice of the remaining budget per seed
        remaining = deadline - time.time()
        per_seed = remaining / (len(seeds) - idx + 1)
        seed_deadline = time.time() + per_seed

        print(
            f"\nSeed {idx}/{len(seeds)} – allocated {per_seed:.1f}s "
            f"({time.time() - start:.1f}s elapsed)"
        )
        # run the optimiser
        a_opt = subgradient_descent(
            seed_vec,
            seed_deadline,
            step0=0.12,
            max_iter=2_000_000,
        )
        # compute cheap surrogate (identical to the true objective when sum=1)
        surrogate_val = fast_score(a_opt)
        print(f"   Surrogate after optimisation = {surrogate_val:.6f}")

        # official evaluation (optional, for logging)
        try:
            official_val = evaluate_sequence(a_opt.tolist())
            print(f"   Official objective = {official_val:.6f}")
        except Exception as e:
            official_val = float("inf")
            print(f"   Official evaluation failed: {e}")

        # keep the best overall
        if official_val < best_obj:
            best_obj = official_val
            best_seq = a_opt.tolist()
            best_surrogate = surrogate_val
            print("   *** New global best ***")
        elif surrogate_val < best_surrogate:
            # in the unlikely case the official call fails, keep the best surrogate
            best_surrogate = surrogate_val
            best_seq = a_opt.tolist()
            print("   *** New best based on surrogate ***")

    # ------------------------------------------------------------------
    # 4. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a simple uniform vector of moderate size
        n0 = 512
        best_seq = [1.0 / n0] * n0
        print("Fallback to uniform length 512")
        try:
            best_obj = evaluate_sequence(best_seq)
            print(f"Fallback official score = {best_obj:.6f}")
        except Exception:
            pass

    print(f"\nSearch finished. Best official score ≈ {best_obj:.6f}")
    return best_seq
