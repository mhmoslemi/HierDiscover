"""
Propose a low‑score step‑function sequence.

Algorithm
---------

1. Load a fallback sequence (``height_sequence_1``) if it exists,
   otherwise start from a uniform distribution of length 2048.

2. Clean the vector (enforce positivity, normalise to sum = 1).

3. **Linear‑programming improvement**  
   Repeatedly (while time permits) solve

        maximise   ∑ g_i
        subject to  (f * g)(k) ≤ M   for all k
                    g_i ≥ 0

   where ``f`` is the current vector, ``M = max(f*f)``.
   The LP is built with a sparse Toeplitz matrix that encodes the
   aperiodic convolution.  The optimum typically yields
   ``sum(g) > 1``; after normalising ``g`` back to a probability vector
   we move partway towards it (convex blend).  After each blend we
   symmetrise the vector (average with its mirror) and keep the best
   result.

4. **Smooth exponentiated‑gradient refinement** – minimise the smooth
   log‑sum‑exp approximation of the max‑convolution using KL‑mirrored
   updates (the same scheme as the reference implementation, but on the
   current best vector).

5. **Projected sub‑gradient polishing** (optional) – classic Euclidean
   sub‑gradient descent on the exact objective.

6. **Stochastic hill‑climb** – a cheap random walk that only accepts
   strictly improving moves, finalising the answer.

All stages respect the overall 1000 s wall‑clock budget; the best
sequence found is returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (both inputs 1‑d)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector normalised to sum = 1.
    (The harness will apply the same scaling when calling
    ``evaluate_sequence``.)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Linear‑programming step (f‑g iteration)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> scipy.sparse.csc_matrix:
    """
    Construct the (2n‑1) × n matrix A such that  (A @ g)  =  f * g
    (full aperiodic convolution).  All rows correspond to convolution
    indices 0 … 2n‑2.
    """
    n = f.shape[0]
    # Each offset d (0 … n‑1) contributes a diagonal filled with f[d].
    diags = [np.full(n, f[d], dtype=float) for d in range(n)]
    offsets = np.arange(n)               # offset = row – col
    A = scipy.sparse.diags(diags, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def lp_improvement(f: np.ndarray,
                   deadline: float,
                   max_iter: int = 6,
                   blend_factor: float = 0.5,
                   eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvement steps starting from ``f``.
    Returns the best vector and its exact score.
    """
    n = f.shape[0]
    best = f.copy()
    best_score = compute_score(best)

    it = 0
    while it < max_iter and time.time() < deadline - 10.0:
        # 1) current max‑convolution value
        conv = convolution_fft(best, best)
        M = conv.max()

        # 2) construct constraints A @ g ≤ M
        A = build_conv_matrix(best)
        b = np.full(2 * n - 1, M, dtype=float)

        # 3) LP: maximise sum(g)  ⇔ minimise -∑g
        c = -np.ones(n, dtype=float)
        bounds = [(0.0, None)] * n

        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[LP] iteration {it}: linprog failed – {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:          # no improvement possible
            print(f"[LP] iteration {it}: sum(g)≈{sum_g:.6f} (no gain)")
            break

        # 4) normalise g back to a probability vector
        g_norm = g / sum_g

        # 5) convex blend (fixed blend_factor, but we try a fallback if needed)
        for t in (blend_factor, 0.3, 0.1):
            cand = (1.0 - t) * best + t * g_norm
            cand = np.maximum(cand, eps)
            cand = (cand + cand[::-1]) * 0.5            # enforce symmetry
            cand /= cand.sum()
            cand_score = compute_score(cand)

            if cand_score < best_score - 1e-12:
                best = cand
                best_score = cand_score
                print(f"[LP] iteration {it}, blend {t:.2f} → score {best_score:.6f}")
                break
        else:
            print(f"[LP] iteration {it}: no blending improved the score")
            break

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# 2. Smooth exponentiated‑gradient optimisation (log‑sum‑exp)
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 150_000,
                        beta_start: float = 0.5,
                        beta_end: float = 250.0,
                        eta0: float = 0.7,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        f_β(a) = (1/β)·log Σ exp(β·(a*a)_k)
    by KL‑mirrored updates a ← a·exp(−η∇f_β) and renormalisation.
    Returns the best exact score seen.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smooth soft‑max weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()

        # gradient of f_β (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # step size
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Projected sub‑gradient descent (optional, cheap)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classical Euclidean sub‑gradient method for the exact max‑objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max‑conv w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Random‑walk polishing: with prob 0.5 swap two entries,
    otherwise transfer a small fraction of mass between two indices.
    Only moves that improve the exact score are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12
    rng = np.random.default_rng()

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if rng.random() < 0.5:                     # swap
            i, j = rng.integers(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # mass transfer
            i, j = rng.integers(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        # re‑project onto simplex
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a, best_sc
