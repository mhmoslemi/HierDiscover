def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for j in range(n):  # column
        # rows index from j to j + n - 1 inclusive
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        data.extend(f)
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

"""
Propose a step‑function (non‑negative sequence) that minimises the
evaluate_sequence() objective.

Algorithm
---------
1.  Initialise a modest‑size probability vector (sum = 1) using an
    arcsine‑like envelope – a known good analytic seed.
2.  Repeatedly solve a tiny linear programme:
       maximise   Σ g_j
       subject to  (a ∗ g)_k ≤ M   for all k,
                   g_j ≥ 0,
    where M = max (a ∗ a).  The LP finds a distribution g that can
    carry more mass than a while still respecting the current peak M.
    After normalising g to a probability vector we move a towards g
    (convex combination).  This simple “LP‑boost” step lowers the
    convolution peak quickly.
3.  Upsample the refined short vector to a larger length (4096) and
    perform a long Adam optimisation on a smooth log‑sum‑exp surrogate
    of the max‑convolution (the same surrogate used in the reference
    code, but with a higher temperature schedule and more iterations).
4.  Polish the result with exponentiated‑gradient (mirror‑descent) on the
    exact max‑convolution objective.
5.  Finish with a projected sub‑gradient descent for a few extra
    iterations.
6.  Return the final vector (any scaling is irrelevant – the objective
    is homogeneous).

The whole routine respects the 1000 s wall‑clock budget and always
returns a non‑empty valid sequence.
"""

import time
from typing import List

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ------------------------------------------------------------
# FFT‑based convolution utilities (linear convolution via real FFT)
# ------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (a∗a)."""
    conv = convolution_fft(a, a)
    return float(np.max(conv))


# ------------------------------------------------------------
# Simple feasibility utilities
# ------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Analytic seed families
# ------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ------------------------------------------------------------
# Smooth‑max surrogate + Adam optimiser
# ------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 800.0,
    ramp_iters: int = 120_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    z0 : ndarray
        Initial log‑probabilities (log‑softmax coordinates).
    deadline : float
        Absolute time at which the optimiser must stop.
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate for Adam.

    Returns
    -------
    a : ndarray
        Optimised probability vector (sum = 1, a_i ≥ 0).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # simplex point
        conv = convolution_fft(a, a)       # length 2n‑1
        mx = float(np.max(conv))

        # temperature β(t)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)         # weights, sum to 1 after normalisation
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]  # central n entries

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output
        if t % 5000 == 0:
            score = 2.0 * n * mx
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate score={score:.6f}")

    a = softmax(z)
    return a


# ------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step: float = 0.015,
) -> np.ndarray:
    """
    Mirror descent (KL divergence) on the exact max‑convolution objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.shape[0]
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            score = 2.0 * n * mx
            print(f"    EG it={it:,d} score={score:.6f}")

    return a


# ------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            score = 2.0 * n * mx
            print(f"    SubGrad it={it:,d} score={score:.6f}")

    return a


# ------------------------------------------------------------
# Linear‑program “boost” step (LP‑based mass increase)
# ------------------------------------------------------------


def build_conv_matrix_sparse(a: np.ndarray) -> "scipy.sparse.csr_matrix":
    """
    Return the (2n‑1)×n sparse convolution matrix A such that
    (A @ g) = a ∗ g  (linear convolution, zero‑padded).
    """
    n = a.shape[0]
    # each diagonal offset k (0 ≤ k < n) contains the vector a
    offsets = np.arange(n)
    data = [a] * n                     # list of length‑n arrays
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def lp_boost_step(a: np.ndarray, M: float) -> np.ndarray:
    """
    One LP‑boost iteration:
        maximise   Σ g_j
        subject to  (a ∗ g)_k ≤ M   ∀k,
                    g ≥ 0.
    Returns the normalised g (sum = 1) if the LP succeeds,
    otherwise the original a.
    """
    n = a.shape[0]
    A = build_conv_matrix_sparse(a)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)                     # minimise -Σg  ≡ maximise Σg
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        print(f"    LP failed: {res.message}")
        return a.copy()

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1.0 + 1e-12:
        # no extra mass can be carried – stop improving
        return a.copy()

    g_norm = g / sum_g                   # put g back onto the simplex
    return g_norm
