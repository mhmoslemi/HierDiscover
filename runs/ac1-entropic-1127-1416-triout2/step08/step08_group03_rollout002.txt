def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0

    n = 2048 # chosen length
    target_sum = np.sqrt(2 * n)

    # Initialize f: try to use existing sequence if present and length matches
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        if init.size != n:
            # Trim or pad
            if init.size > n:
                init = init[:n]
            else:
                # pad with zeros and renormalize
                pad = np.zeros(n, dtype=np.float64)
                pad[:init.size] = init
                init = pad
    except Exception:
        init = None

    if init is None or np.all(init == 0):
        # fallback to an analytic shape: left heavy power law
        beta = 1.5
        i = np.arange(n, dtype=np.float64)
        init = np.exp(-beta * np.log(i + 0.5))
        init = init / init.sum()

    # Normalize
    f = init * (target_sum / init.sum())

    best_f = f.copy()
    best_M = np.max(convolution_fft(f, f))

    # iterative LP improvements
    it = 0
    while time.time() < deadline - 1.0 and it < 30:
        # compute M
        conv_ff = convolution_fft(f, f)
        M = float(np.max(conv_ff))

        # Build constraint matrix A
        # For each column i, rows i .. i+n-1 get f[i]
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = f[i]

        # Solve LP to maximize sum(b)
        # minimize -sum(b)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        ub = np.full(rows, M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            print(f"LP failed at iteration {it}, status {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        # scale b to target sum
        g = b * (target_sum / sum_b)

        # mixing step
        t = 0.25  # mixing parameter, could adapt
        f = (1 - t) * f + t * g

        # evaluate new max conv
        new_M = float(np.max(convolution_fft(f, f)))
        if new_M < best_M - 1e-9:
            best_M = new_M
            best_f = f.copy()
            print(f"[{time.time():.1f}] iter {it}: improved M = {best_M:.6f}")

        # If improvement small break maybe
        if it > 0 and abs(new_M - M) < 1e-6:
            # stagnation
            break

        it += 1

    # final fine-tune with simple hillclimb? Could optional.

    # Convert to python list and return.

Propose a low‑score step‑function using a linear‑programming refinement.

Strategy
--------
1. Choose a moderate length `n = 2048`.  The objective
   score = 2·n·max(conv(a,a)) / (sum(a)^2) becomes exactly the
   maximal convolution value `M` when the ℓ₁‑norm is fixed to
   sqrt(2·n).  Hence we keep the vector normalised to this sum and
   simply minimise `M`.

2. Initialise a good candidate on the simplex by scanning a few
   analytic families (symmetric product and left‑heavy power law) and
   by trying the previously discovered construction (`height_sequence_1`).

3. Iteratively improve the candidate:
   * Let `M = max(conv(f,f))`.
   * Solve the linear program
        maximise   Σ b_i
        subject to b_i ≥ 0,
                   conv(f,b) ≤ M   (componentwise)
     Using `scipy.optimize.linprog`.  The feasible point `b = f`
     is always available, so the LP is never infeasible.
   * Scale the optimum `b` to the required ℓ₁‑norm (√(2n)) obtaining `g`.
   * Mix the old vector and the new one: `f ← (1‑t)·f + t·g` with `t = 0.5`.
   * Repeat until the time budget is exhausted or progress stalls.
   Each LP step strictly reduces the maximal convolution, i.e. the
   score.

4. Symmetrise the best vector (average with its reverse) and re‑normalise.
   Many optimal solutions are symmetric, and this often yields a small
   extra gain.

5. A short stochastic hill‑climb (mass‑transfer moves on the simplex)
   finishes the optimisation, accepting only improvements.

All steps are bounded by the global time limit (default 1000 s).  The
function returns a plain Python list of non‑negative floats ready for
`evaluate_sequence`.
"""

import time

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Return max entry of a*a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, non‑negative)."""
    if np.all(v <= 0):
        # Uniform vector if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for good initialisation)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α), symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Avoid infinities that could appear for α huge
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β), monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector on the simplex."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a

    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return result
