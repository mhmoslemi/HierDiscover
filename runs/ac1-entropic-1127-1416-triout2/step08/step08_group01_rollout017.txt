""" 
Convex optimisation based step‑function construction.

We solve the convex minimax problem 
    minimize t 
    s.t. a_i >= 0, sum a_i = 1,
         a*a (aperiodic convolution) <= t componentwise.
The problem is a quadratically constrained program (QCP) and is tackled
with CVXPY. The resulting distribution gives a very flat autocorrelation,
hence a low value of `evaluate_sequence`. A short stochastic hill‑climb
polishes the solution.
"""

import time, math, random
import numpy as np

# ------------------- helper utilities -------------------
def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]

def compute_exact_score(a: np.ndarray) -> float:
    # a is assumed to be a probability vector (sum==1)
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto {x >=0, sum x =1}. """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# ------------------- convex optimisation -------------------
def solve_minmax_convex(n: int, time_limit: float):
    """
    Solve   min t
           s.t. a >=0, sum(a)=1, a.T Q_k a <= t   for all k
    where Q_k encodes the aperiodic convolution.
    Returns the probability vector a and the optimal t (if solved).
    """
    import cvxpy as cp
    import scipy.sparse as sp

    a = cp.Variable(n)
    t = cp.Variable()
    constraints = [a >= 0, cp.sum(a) == 1, t >= 0]

    # Build the Toeplitz‑like matrices that represent each convolution entry.
    # Q_k[i,j] = 1 if i + j == k else 0
    Qs = []
    for k in range(2 * n - 1):
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        rows = np.arange(i0, i1 + 1, dtype=int)
        cols = k - rows
        data = np.ones_like(rows, dtype=float)
        Q = sp.csr_matrix((data, (rows, cols)), shape=(n, n))
        # The matrix is already symmetric because the pair (i,j) and (j,i)
        # both appear when the loop runs over the whole range.
        Qs.append(Q)
        constraints.append(cp.quad_form(a, Q) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # Use ECOS first – it is usually faster for small QPs.
    try:
        prob.solve(solver=cp.ECOS, max_iters=2000, feastol=1e-6,
                   reltol=1e-6, abstol=1e-6, verbose=False)
    except Exception as e:
        print(f"ECOS failed ({e}), trying SCS")
        prob.solve(solver=cp.SCS, max_iters=25000, eps=1e-5,
                   verbose=False)

    if a.value is None:
        raise RuntimeError("Convex solve did not return a solution")

    a_vec = np.maximum(np.array(a.value, dtype=float), 0.0)
    if a_vec.sum() == 0:
        a_vec = np.full(n, 1.0 / n)
    else:
        a_vec /= a_vec.sum()
    t_val = float(t.value) if t.value is not None else None
    return a_vec, t_val

# ------------------- simple hill‑climb polish -------------------
def hill_climb(start_seq: np.ndarray, deadline: float,
               max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Very lightweight randomized polishing.  A move is kept only when
    it strictly improves the exact evaluation score.
    """
    n = start_seq.size
    best = start_seq.copy()
    best_sc = compute_exact_score(best)
    step_scale = 0.12
    while max_steps > 0 and time.time() < deadline:
        cand = best.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_exact_score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand
            step_scale = min(step_scale * 1.08, 0.5)
        else:
            step_scale *= 0.97
        max_steps -= 1
    return best, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Main entry point used by the harness.
    The function first tries a small convex QP (n=256) that globally
    minimises the max‑convolution.  The solution is then polished by a
    short stochastic hill‑climb.  If the convex solver is unavailable the
    routine falls back to a simple uniform vector.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Very cheap fallback – uniform short vector (guarantees a finite score)
    # ------------------------------------------------------------------
    try:
        n0 = 64
        uniform = [1.0 / n0] * n0
        sc0 = evaluate_sequence(uniform)
        best_seq = uniform
        best_score = sc0
        print(f"[{time.time():.1f}] uniform fallback score = {sc0:.6f}")
    except Exception as e:
        print("Uniform fallback failed:", e)

    # ------------------------------------------------------------------
    # 1. Convex optimisation for a modest length (256, maybe 512)
    # ------------------------------------------------------------------
    for n in (256, 512):
        if time.time() >= deadline - 120.0:
            break
        remaining = deadline - time.time()
        # Give the convex solve at most half of the remaining budget.
        solve_time = min(remaining * 0.5, 300.0)
        print(f"[{time.time():.1f}] attempting convex solve with n={n}")
        try:
            a_vec, t_val = solve_minmax_convex(n, solve_time)
            sc = evaluate_sequence(a_vec.tolist())
            print(f"[{time.time():.1f}] convex n={n} score = {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_vec.tolist()
        except Exception as e:
            print(f"Convex optimisation for n={n} failed: {e}")

    # ------------------------------------------------------------------
    # 2. Light hill‑climb polishing if we still have some time left.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_arr = np.array(best_seq, dtype=float)
        print(f"[{time.time():.1f}] polishing with hill‑climb")
        a_polished, sc_polished = hill_climb(
            a_arr, deadline - 5.0, max_steps=600_000)
        print(f"[{time.time():.1f}] polished score = {sc_polished:.6f}")
        if sc_polished < best_score:
            best_score = sc_polished
            best_seq = a_polished.tolist()

    # ------------------------------------------------------------------
    # 3. Final sanity check and return.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return best_seq

"""
Step‑function optimisation – a short analytic starter + long exponentiated‑gradient
refinement.

We build a symmetric “binomial‑root” distribution:
    c_k = C(2k,k) / 4^k = Π_{i=1..k} (2i‑1)/(2i)      (c_0 = 1)
and set a_i = c_i + c_{n‑1‑i}, normalised to a probability vector.
For this distribution a∗a is already close to uniform, hence the
evaluation score 2·n·max(a∗a) is near the theoretical lower bound 2n/(2n‑1)≈1.

The vector is then polished by a very long exponentiated‑gradient run on the
smooth log‑sum‑exp surrogate of the max‑convolution, followed by a cheap
stochastic hill‑climb that only accepts moves that strictly improve the exact
score.  All phases respect the 1000 s wall‑clock budget.

The harness will call ``propose_candidate``; it returns a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based aperiodic convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def exact_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a probability vector (sum(a)=1).
    This matches the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x ≥ 0, sum x = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic “binomial‑root” starter
# ----------------------------------------------------------------------


def binomial_start(n: int) -> np.ndarray:
    """
    Symmetric distribution obtained from the discrete square‑root of a uniform
    sequence: a_i ∝ C(2i,i)/4^i + C(2(n‑1‑i), n‑1‑i)/4^{n‑1‑i}.
    The recursion   c_k = c_{k‑1}·(2k‑1)/(2k)   generates the coefficients.
    """
    c = np.empty(n, dtype=float)
    c[0] = 1.0
    for k in range(1, n):
        c[k] = c[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    a = c + c[::-1]               # make it symmetric (U‑shaped)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Smooth‐max (log‑sum‑exp) optimisation via exponentiated gradient
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 800_000,
                        beta_start: float = 0.5,
                        beta_end: float = 500.0,
                        eta0: float = 0.8,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimises the smooth surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using a KL‑mirror (exponentiated‑gradient) update.
    The temperature β is exponentially annealed from beta_start to beta_end.
    Returns the best exact score seen during the run.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- temperature schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- smoothed soft‑max weights ----
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        # stabilised exponentiation
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # ---- gradient of f_β (cross‑correlation) ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ---- step‑size (KL‑mirror) ----
        eta = eta0 / math.sqrt(it)

        # ---- multiplicative update + re‑normalisation ----
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ---- occasional exact evaluation ----
        if it % 300 == 0:
            sc = exact_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final exact check
    sc = exact_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Very cheap stochastic hill‑climb (accepts only improving moves)
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Random walk that either swaps two entries or moves a small packet of mass.
    Only strictly improving moves are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = exact_score(best_a)

    step_scale = 0.12

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = exact_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.08, 0.5)
        else:
            step_scale *= 0.97
        max_steps -= 1
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Returns a non‑negative sequence that (hopefully) minimises `evaluate_sequence`.
    The strategy is:
        1.  Build a symmetric binomial‑root starter (U‑shaped).
        2.  Run a long exponentiated‑gradient optimisation on the smooth max‑objective.
        3.  Finish with a short hill‑climb polishing.
    All phases respect the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 0. Very cheap fallback – uniform short vector (guarantees a finite score)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        n0 = 64
        uniform = [1.0 / n0] * n0
        sc0 = evaluate_sequence(uniform)
        best_seq = uniform
        best_score = sc0
        print(f"[{time.time():.1f}] uniform fallback score = {sc0:.6f}")
    except Exception as e:
        print("Uniform fallback failed:", e)

    # ------------------------------------------------------------------
    # 1. Choose a moderate length (fast FFT, enough degrees of freedom)
    # ------------------------------------------------------------------
    # For a tight bound we do not need a huge n – n≈256 already gives an
    # objective extremely close to 1.  It also makes the long optimisation cheap.
    n = 256
    print(f"[{time.time():.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # 2. Analytic binomial‑root starter
    # ------------------------------------------------------------------
    a0 = binomial_start(n)
    sc_start = evaluate_sequence(a0.tolist())
    print(f"[{time.time():.1f}] starter (binomial) score = {sc_start:.6f}")
    if sc_start < best_score:
        best_score = sc_start
        best_seq = a0.tolist()

    # ------------------------------------------------------------------
    # 3. Long exponentiated‑gradient optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] starting long exp‑gradient")
        a_opt, sc_opt = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 30.0,
            max_iters=800_000,
            beta_start=0.5,
            beta_end=500.0,
            eta0=0.8,
        )
        print(f"[{time.time():.1f}] after exp‑grad score = {sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()

    # ------------------------------------------------------------------
    # 4. Final hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb(
            np.array(best_seq, dtype=float),
            deadline - 2.0,
            max_steps=600_000,
        )
        print(f"[{time.time():.1f}] hill‑climb score = {sc_final:.6f}")
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()

    # ------------------------------------------------------------------
    # 5. Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
