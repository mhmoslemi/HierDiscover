def arcsine_initial(n):
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()

**Beta shape**:

```
def generate_beta_shape(n, alpha, beta, eps=0.0):
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot
```

    M = np.max(conv)
    # stable log-sum-exp
    exp_terms = np.exp(beta * (conv - M))

"""
Search for a non‑negative step sequence that minimizes ``evaluate_sequence``.

Algorithm
---------
* Multi‑start continuation L‑BFGS on a smooth approximation
  `log‑sum‑exp(β·conv(a,a))`.  The iterate lives on the probability
  simplex via a soft‑max parametrisation `a = softmax(z)`.  The smoothness
  parameter β is increased (5 → 2·10⁴) in several stages; each stage runs a
  short L‑BFGS optimisation (few hundred gradient evaluations).  This
  quasi‑Newton method converges much faster than the pure sub‑gradient
  descent used previously.

* After the continuation phase the best candidate is polished with a
  projected sub‑gradient descent on the *exact* objective and a cheap
  random local search (pairwise mass transfers).

* The whole search respects the 1000 s budget and always returns a
  valid, non‑negative sequence.

The implementation is self‑contained, uses only NumPy and SciPy and prints
progress information.
"""

import time
import math
from typing import List, Tuple

import numpy as np

# SciPy may be unavailable in some environments – the code degrades gracefully.
try:
    import scipy.optimize
except Exception:      # pragma: no cover
    scipy = None


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def softmax_logits(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def fast_score(a: np.ndarray) -> float:
    """
    Cheap surrogate: 2·n·max(conv(a,a))/sum(a)².
    Assumes a is non‑negative.
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 0.0) -> np.ndarray:
    """Beta(α,β) shape on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Shape proportional to 1/((i+½)*(n‑½‑i)+eps)^p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex (Dirichlet with random concentration)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)   # between 0.01 and ≈5
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Projection onto the simplex (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx_i = 1, x_i ≥ 0."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (used for a final polish)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent for f(a)=2·n·max(conv(a,a))/sum(a)²."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

    return a


# ----------------------------------------------------------------------
# Simple random local search (pairwise mass transfers)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.025,
) -> np.ndarray:
    """Cheap stochastic pairwise mass‑transfer moves."""
    a = a0.copy()
    best_score = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * rng.random() * max_transfer
        if eps <= 0.0:
            continue
        a[i] -= eps
        a[j] += eps
        # re‑normalise (numerical stability)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
            continue
        a /= s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
        else:
            a[j] -= eps
            a[i] += eps
            a /= a.sum()   # restore sum=1 (tiny rounding errors)
    return a


# ----------------------------------------------------------------------
# Smooth‑max objective and continuation L‑BFGS
# ----------------------------------------------------------------------


def obj_and_grad(z: np.ndarray, beta: float) -> Tuple[float, np.ndarray]:
    """
    Returns (f,∇f) where f = log‑sum‑exp(β·conv(a,a)) and a = softmax(z).
    The gradient is with respect to the logit vector z.
    """
    a = softmax_logits(z)
    conv = convolution_fft(a, a)
    mx = np.max(conv)

    # stable log‑sum‑exp
    exp_terms = np.exp(beta * (conv - mx))
    sum_exp = np.sum(exp_terms)
    f = beta * mx + np.log(sum_exp)          # scalar

    w_norm = exp_terms / sum_exp            # softmax weights (sum to 1)

    # gradient w.r.t. a : 2·β·(a ★ w_rev)  (★ = convolution)
    rev = convolution_fft(a, w_norm[::-1])   # length = 3·n‑2
    n = a.size
    grad_a = 2.0 * beta * rev[n - 1 : n - 1 + n]

    # chain rule through soft‑max
    grad_z = a * (grad_a - np.dot(grad_a, a))
    return float(f), grad_z


def make_time_callback(deadline: float):
    """Callback for SciPy L‑BFGS; aborts if time limit is exceeded."""
    def callback(xk):
        if time.time() >= deadline:
            raise TimeoutError()
    return callback


def lbfgs_continuation(z0: np.ndarray,
                       beta_schedule: List[Tuple[float, int]],
                       deadline: float) -> np.ndarray:
    """
    Continuation L‑BFGS: run a sequence of L‑BFGS optimisations with
    increasing smoothness parameters β.  Returns a probability vector a.
    """
    if scipy is None:   # pragma: no cover
        # SciPy not available – fall back to the initial soft‑max.
        return softmax_logits(z0)

    z = np.array(z0, copy=True)

    for beta, max_iter in beta_schedule:
        if time.time() >= deadline:
            break
        # the lambda captures the current beta value
        fun = lambda zz, b=beta: obj_and_grad(zz, b)

        try:
            res = scipy.optimize.minimize(
                fun,
                z,
                method='L-BFGS-B',
                jac=True,
                callback=make_time_callback(deadline),
                options={'maxiter': max_iter, 'disp': False},
            )
            z = res.x
        except TimeoutError:
            break
        except Exception as e:   # pragma: no cover
            print(f"    L‑BFGS failed at β={beta}: {e}")
            break

    return softmax_logits(z)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The routine combines analytic seeds,
    a continuation L‑BFGS optimiser on a smooth‑max approximation,
    a short projected sub‑gradient polish and a cheap random local
    search.  All work respects the overall time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2   # safety margin

    best_seq: List[float] = []
    best_official: float = float("inf")
    best_surrogate: float = float("inf")

    # --------------------------------------------------------------
    # Optional fallback to a previously discovered construction
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, cand
                best_surrogate = fast_score(np.array(cand))
        except Exception:
            pass

    # --------------------------------------------------------------
    # Lengths to explore – larger n gives a better theoretical lower bound
    # --------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Optimising length {n} (time left {deadline - time.time():.1f}s) ===")

        # ---- 1. Build a diverse pool of seeds ---------------------------------
        seeds: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            generate_mixture_shape(n, weight_arc=0.6),
        ]

        # symmetric beta family
        for alpha in np.linspace(0.3, 3.0, 9):
            seeds.append(generate_beta_shape(n, alpha, alpha, eps=0.0))

        # power‑law family
        for p in np.linspace(0.2, 1.8, 8):
            seeds.append(generate_power_shape(n, p, eps=1e-8))

        # random Dirichlet draws
        for _ in range(12):
            seeds.append(generate_random_dirichlet(n, rng))

        # ---- 2. Keep the most promising seeds (cheap surrogate) -----------------
        scored: List[Tuple[float, np.ndarray]] = [(fast_score(v), v) for v in seeds]
        scored.sort(key=lambda x: x[0])
        # keep at most 4 seeds per length – enough diversity but cheap enough
        top_seeds = [vec for _, vec in scored[:4]]

        # ---- 3. Allocate a generous slice of the remaining budget per seed -----
        remaining_seeds = sum(len(s) for s in lengths[ lengths.index(n): ]) * 4
        per_seed_budget = max(5.0, (deadline - time.time()) / max(1, remaining_seeds))

        for seed_idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"\nSeed {seed_idx}/{len(top_seeds)} for n={n} – budget ≈{per_seed_budget:.1f}s")
            seed_deadline = time.time() + per_seed_budget

            # Initialise logits – add a tiny epsilon to avoid log(0)
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            # ---- 4. Continuation L‑BFGS on smooth‑max -------------------------
            # β schedule – more stages for larger n, but keep it short enough.
            beta_schedule = [
                (5.0,   80),
                (20.0,  80),
                (80.0,  80),
                (300.0, 80),
                (1200.0, 60),
                (5000.0, 40),
                (20000.0, 30),
            ]
            a_opt = lbfgs_continuation(z0, beta_schedule, seed_deadline)

            # ---- 5. Official evaluation of the L‑BFGS result -------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score after L‑BFGS = {val:.6f}")
                if val < best_official:
                    best_official, best_seq = val, a_opt.tolist()
                    best_surrogate = fast_score(a_opt)
                    print("   *** New global best! ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ---- 6. Very short projected sub‑gradient polish (if time permits) --
            if time.time() < deadline - 2.0:
                # a tiny polishing run – 10 k iterations is enough to shave a bit
                polish_deadline = deadline - 0.5
                a_polish = subgradient_descent_exact(
                    np.array(best_seq, dtype=float),
                    polish_deadline,
                    max_iter=30_000,
                    step0=0.04,
                )
                try:
                    val_polish = evaluate_sequence(a_polish.tolist())
                    print(f"   After sub‑grad polish official score = {val_polish:.6f}")
                    if val_polish < best_official:
                        best_official, best_seq = val_polish, a_polish.tolist()
                        best_surrogate = fast_score(a_polish)
                        print("   *** Updated best after polishing ***")
                except Exception:
                    pass

    # ------------------------------------------------------------------
    # 7. Final cheap random local search around the best solution found
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:
        print("\nFinal random local search around the current best …")
        a_start = np.array(best_seq, dtype=float)
        a_refined = random_local_search(
            a_start,
            deadline,
            rng,
            max_transfer=0.02,
        )
        try:
            val_loc = evaluate_sequence(a_refined.tolist())
            print(f"   Local‑search official score = {val_loc:.6f}")
            if val_loc < best_official:
                best_official, best_seq = val_loc, a_refined.tolist()
                best_surrogate = fast_score(a_refined)
                print("*** Best improved by final local search ***")
        except Exception as e:
            print(f"   Local‑search evaluation error: {e}")

    # ------------------------------------------------------------------
    # 8. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
