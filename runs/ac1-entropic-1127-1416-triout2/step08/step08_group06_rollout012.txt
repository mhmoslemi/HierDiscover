def propose_candidate(seed=42, budget_s=1000):

    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = []
    best_val = float('inf')

    # Check fallback
    if 'height_sequence_1' in globals():
        try: cand= list(map(float, globals()['height_sequence_1']))
             val = evaluate_sequence(cand)
             if val < best_val:
                  best_val, best_seq = val, cand
             print('Fallback', val)
        except: pass

    # Define parameter for lengths
    lengths = [1024, 512]  # start with moderate length
    for n in lengths:
        if time.time() >= deadline - 5: break
        sum_target = np.sqrt(2 * n)
        # Initialize f
        f = arcsine_initial(n)
        # Ensure sum=1
        f = f / f.sum()
        # Scale to sum_target (maybe not needed for compute_score)
        f = f * sum_target
        # Evaluate current
        score = compute_score(f / sum_target)  # normalized
        if score < best_val:
            best_val = score
            best_seq = f.tolist()
            print(f'Initial arcsine n={n} score={score:.6f}')
        # Iterate LP
        # Determine iteration budget: allocate maybe 30% of remaining time for LP
        it = 0
        while time.time() < deadline - 5:
            it += 1
            # compute current M
            conv = convolution_fft(f, f)
            M = np.max(conv)
            # Build A
            # For each j col, fill slice
            # Use np.zeros then fill
            rows = 2 * n - 1
            A = np.zeros((rows, n), dtype=float)
            for j in range(n):
                A[j:j+n, j] = f
            # Solve LP
            try:
                from scipy.optimize import linprog
                res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(rows, M), bounds=[(0, None)] * n, method='highs')
                if not res.success:
                    print(f'LP iteration {it} failed: {res.message}')
                    break
                g0 = res.x
            except Exception as e:
                print('LP solve error', e)
                break
            sum_g0 = np.sum(g0)
            if sum_g0 == 0:
                # can't scale, break
                break
            # Scale to target sum
            g = g0 * (sum_target / sum_g0)
            # Choose step size t
            t = 0.2 / (it + 1)  # decreasing
            f = (1 - t) * f + t * g
            # Re-normalize to sum_target
            f = f * (sum_target / np.sum(f))
            # Compute score
            cur_score = compute_score(f / sum_target)
            if cur_score < best_val:
                best_val = cur_score
                best_seq = f.tolist()
                print(f'LP it={it} new best score={cur_score:.6f}')
            # Optional break if improvement small
            if it >= 50:  # limit number of LP iterations
                break
        # After LP loop, perform polishing (if time)
        if best_seq and time.time() < deadline - 10:
            a_arr = np.array(best_seq, dtype=float)
            # Exponentiated gradient polish
            a_arr = exponentiated_gradient_polish(a_arr, deadline=deadline, max_iter=50000, step=0.01)

            if score < best_val:
                best_val, best_seq = score, a_arr.tolist()
                print(f'Post EG polish best score={score:.6f}')
            # Subgradient final
            if time.time() < deadline - 3:
                a_arr = subgradient_descent_exact(a_arr, deadline=deadline, max_iter=200000, step0=0.05)

                if score < best_val:
                    best_val, best_seq = score, a_arr.tolist()
                    print(f'Post subgrad polish best score={score:.6f}')
        # Keep best across lengths
    # If still empty, fallback to uniform
    if not best_seq:
        n0 = 256
        best_seq = [1./n0]*n0
    # Final evaluation via official function (only for logging)
    try:
        off = evaluate_sequence(best_seq)
        print('Final official score', off)
    except Exception as e:
        print('Official evaluation failed', e)
    return best_seq

"""
Propose a non‑negative step sequence that minimizes the official
``evaluate_sequence`` objective.

Algorithm
---------

1. **Diverse warm‑start** – generate a few parametric shapes
   (arcsine, Gamma‑like) and pick the one with the smallest fast score.

2. **Linear‑programming refinement** – for a fixed length *n* (≤ 2048)
   and a current shape *f* (∑ f = 1) compute *M* = max (f∗f).  Solve the
   LP

        maximise   Σ g_j
        subject to Σ_i f_i·g_{k‑i} ≤ M   for every convolution index k
                   g_j ≥ 0

   using ``scipy.optimize.linprog``.  Rescale the LP solution to
   Σ g = 1 and replace the current shape by the convex combination

        f ← (1‑t)·f + t·g   (t≈0.1)

   The bound *M* never increases; repeating a few iterations drives the
   objective downward.

3. **Mirror‑descent polishing** – after the LP phase the shape is
   refined with three fast simplex‑constrained optimisers that have
   proved useful in the previous state‑of‑the‑art:
   * exponentiated‑gradient (KL‑mirror) descent,
   * Adam on a smooth log‑sum‑exp surrogate,
   * exact projected sub‑gradient descent.

   All three work on the normalised simplex (∑ a = 1) and keep the
   sequence non‑negative.

4. **Return** – the best sequence seen (as a plain ``list``) is returned.
   The routine respects the 1000 s wall‑clock limit and always produces
   a non‑empty result.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a normalised vector
    (∑ a = 1).  Returns ``2·n·max(conv(a,a))`` which equals the official
    value because the scaling cancels out.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (all normalised to L1‑norm = 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (β = ½)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_gamma_shape(n: int, gamma: float,
                         delta: float = 0.5,
                         eps: float = 0.0) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{-γ}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the exact sub‑gradient method
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(a0: np.ndarray,
                                 deadline: float,
                                 max_iter: int = 120_000,
                                 step: float = 0.012) -> np.ndarray:
    """
    Mirror‑descent on the original max‑convolution objective using KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # gradient of 2·n·max(conv) w.r.t. a (sub‑gradient)
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5_000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a

# Smooth‑max surrogate + Adam optimisation (optional)
# ----------------------------------------------------------------------
def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(z0: np.ndarray,
                         deadline: float,
                         n: int,
                         beta0: float = 5.0,
                         beta_max: float = 300.0,
                         ramp_iters: int = 50_000,
                         lr: float = 0.07) -> np.ndarray:
    """
    Adam on the log‑sum‑exp smooth approximation of max(conv).
    Operates in the log‑domain (soft‑max enforces the simplex).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥ 0 , Σa = 1
        conv = convolution_fft(a, a)

        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        rev = convolution_fft(a, w_norm[::-1])
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    return softmax(z)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------
def subgradient_descent_exact(init_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5_000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (core of the new approach)
# ----------------------------------------------------------------------
def lp_refine_step(f: np.ndarray,
                   t: float,
                   deadline: float) -> np.ndarray:
    """
    Perform a single LP‑based refinement step.

    Parameters
    ----------
    f : ndarray, shape (n,)
        Current distribution (∑ f = 1, non‑negative).
    t : float
        Mixing coefficient (0 < t ≤ 1).
    deadline : float
        Absolute time after which the function should abort.

    Returns
    -------
    f_new : ndarray
        Updated distribution (still on the simplex).
    """
    if not SCIPY_AVAILABLE:      # pragma: no cover
        return f

    n = f.size
    # 1. current bound M = max(f∗f)
    M = float(np.max(convolution_fft(f, f)))

    # 2. build Toeplitz constraint matrix A where
    #    (A @ g)[k] = (f * g)[k] for k = 0 … 2n‑2
    rows = 2 * n - 1
    # using a dense matrix – still modest for n ≤ 2048
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f

    # 3. solve LP: maximise Σ g_j  (i.e. minimise -Σ g_j)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M, dtype=float)

    # solve (highs works with dense arrays as well)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    if not res.success:
        # If LP fails we simply return the original vector.
        return f
    g0 = res.x
    sum_g0 = np.sum(g0)
    if sum_g0 == 0.0:
        return f

    # 4. Renormalise g to the simplex (∑ g = 1)
    g = g0 * (1.0 / sum_g0)

    # 5. Convex combination with the current iterate
    f_new = (1.0 - t) * f + t * g

    # 6. Renormalise to avoid drifting due to numerical error
    f_new = f_new * (1.0 / np.sum(f_new))
    return f_new

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee that we return something valid
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 256
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # 5.  Official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as e:   # pragma: no cover
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
