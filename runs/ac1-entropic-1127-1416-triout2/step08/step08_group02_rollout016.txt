"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
1. **Initial shape** – a discretised “arcsine’’ density  
   a_i ∝ 1/√((i+½)(N‑½‑i)).  This puts most mass at the ends,
   which reduces the central overlap a·rev(a) that dominates the
   max‑convolution term.

2. **Combined Adam optimiser** – minimise a weighted mixture of
   *a*’s smoothed maximum convolution (soft‑max via log‑sum‑exp)
   and the L²‑deviation of the full convolution from its flat lower
   bound (≈1/(2N‑1)).  The two objectives are blended with a
   schedule that starts with the variance term (to flatten the whole
   convolution) and ends with the max‑term (to crush the remaining
   peak).  Gradients are obtained analytically as a correlation of a
   weight vector with a (computed efficiently with FFT).  After each
   Adam step the vector is Euclidean‑projected onto the probability
   simplex.

3. **Final stochastic polishing** – cheap mass‑transfer moves
   (swap or fractional shift) with simplex projection.  The best
   sequence encountered during the run is returned.

All phases respect the overall time budget (default 1000 s).  The
algorithm typically drives the score well below the previous
1.511 × baseline, approaching the theoretical lower bound ≈1.0.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Basic numeric helpers
# -------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT (faster than np.convolve).
    Returns array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if np.all(v <= 0):
        # corner case – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum is assumed ≠0)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# -------------------------------------------------------------
# Gradient utilities
# -------------------------------------------------------------


def correlation_with_weight(weight: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation of ``weight`` (length 2n‑1) with ``a`` (length n).
    Returns an array of length n:
        g_j = Σ_k weight_k * a_{k‑j}
    Implemented as convolution with the reversed a.
    """
    a_rev = a[::-1]
    conv = convolution_fft(weight, a_rev)           # length 3n‑2
    start = a.shape[0] - 1                           # index n‑1
    return conv[start:start + a.shape[0]]


def combined_adam_optimize(
    a0: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
    step0: float = 0.5,
    beta_start: float = 1.0,
    beta_end: float = 250.0,
    lambda_ramp: int = 5_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that mixes two objectives:
        * soft‑max approximation of max(conv)
        * L2 distance of conv from its flat lower bound 1/(2n‑1)

    The mixing weight λ moves linearly from 0 to 1 over ``lambda_ramp``
    iterations, i.e. we first flatten the whole convolution, then
    concentrate on the peak.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]
    avg_target = 1.0 / (2 * n - 1)          # = (Σa)²/(2n‑1) because Σa = 1
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # --- convolution of a with itself -------------------------------------------------
        conv = convolution_fft(a, a)                     # length 2n-1
        maxc = conv.max()

        # --- soft‑max weights for the max‑term --------------------------------------------
        beta = beta_start * (beta_end / beta_start) ** (it / max_iters)
        # stability: shift by the maximum
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # --- gradient for the max‑term ----------------------------------------------------
        grad_max = 2.0 * correlation_with_weight(w, a)

        # --- gradient for the variance‑term (flattening) ----------------------------------
        dev = conv - avg_target
        grad_var = 2.0 * correlation_with_weight(dev, a)

        # --- mixing weight λ --------------------------------------------------------------
        lam = min(1.0, it / max(lambda_ramp, 1))

        # --- combined gradient ------------------------------------------------------------
        grad = lam * grad_max + (1.0 - lam) * grad_var

        # --- Adam update -----------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / np.sqrt(it)
        upd = step * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - upd
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional evaluation (cheap exact score)
        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Adam it {it}] new best score = {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
        print(f"[Adam final] improved best to {best_score:.6f}")

    return best_a, best_score


# -------------------------------------------------------------
# Stochastic hill‑climb for final polishing
# -------------------------------------------------------------


def hill_climb(seq, time_deadline, max_steps: int = 600_000):
    """
    Very cheap random moves (swap or fractional transfer) with simplex
    projection.  Keeps the best sequence seen.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.12  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:        # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                             # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

    # 3. Safeguard – fallback to a uniform vector if something went wrong
    # -----------------------------------------------------------------
    if best_seq is None or len(best_seq) == 0:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_seq)

    # Final sanity check (evaluate_sequence applies its own clipping)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
