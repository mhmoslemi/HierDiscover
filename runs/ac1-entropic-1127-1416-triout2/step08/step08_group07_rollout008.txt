"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  The search proceeds as follows:

1.  Choose a single target length (default 2048 – a good compromise
    between expressive power and speed).  The search could be
    extended to larger lengths if time permits.

2.  Build a small but diverse pool of *analytic* seeds:
    * arcsine‑type (Beta(½,½));
    * a power‑law family  a_i ∝ 1/((i+½)(n‑½‑i)+ε)^p  (the exponent *p*
      is searched on a fine grid around the arcsine exponent 0.5);
    * symmetric Beta(α,α) families (α searched on a grid);
    * a uniform vector and a simple mixture of arcsine + uniform.

3.  Keep the few best seeds according to the cheap surrogate
    ``fast_score`` (the exact objective without clipping).

4.  For each retained seed run a *continuation Adam* optimiser that
    minimises the smooth‑max approximation
        L_β(a)=log‑sum‑exp(β·conv(a,a))
    while staying on the probability simplex via a soft‑max
    parametrisation.  β is increased from 5 to 5000.

5.  Refine the Adam result with a long projected sub‑gradient descent
    on the exact max‑convolution (the true objective).  The step size
    follows  η_t = η₀ / sqrt(t) and the iteration count is driven only
    by the remaining wall‑clock time.

6.  Perform a cheap random‑pairwise mass‑transfer local search
    (accepting only moves that improve the surrogate).  This often
    squeezes a few extra tenths of a percent out of the objective.

All phases respect the global time budget (default 1000 s).  The
function returns a list of non‑negative floats; if the search fails,
a uniform fallback is provided.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Cheap surrogate (the exact objective without clipping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(a,a))/sum(a)^2.
    ``a`` must be non‑negative and sum>0.  This is used for quick
    comparisons while searching.
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 0.0) -> np.ndarray:
    """Beta(α,β) shape on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """
    Shape ∝ 1/((i+½)(n‑½‑i)+ε)^p .
    For p=0.5 this coincides (up to a constant) with the arcsine shape.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with random concentration)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)          # ≈[0.01,5]
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx_i = 1, x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Reference: Duchi et al., “Efficient Projections onto the L1‑Ball…”.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) optimiser with Adam and continuation on β
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max producing a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 5000.0,
    ramp_iters: int = 300_000,
    lr: float = 0.03,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate.
    The iterate lives in the log‑space (z) and the actual probabilities
    are obtained via a soft‑max.  β is increased linearly from ``beta0``
    to ``beta_max`` (continuation) over ``ramp_iters`` steps.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # continuation schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)          # w_i ∝ exp(β·(conv_i – mx))
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of L_β w.r.t. a:
        # d/da Σ w_i·conv_i = 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain‑rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report (silently avoid spamming)
        if t % 20000 == 0:
            print(
                f"    [Adam] it={t:,d} β={beta:.1f} surrogate≈{fast_score(_softmax(z)):.6f}"
            )

    return _softmax(z)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *exact* max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 1_500_000,
    step0: float = 0.10,
) -> np.ndarray:
    """
    Projected sub‑gradient descent for   f(a)=2·n·max(conv(a,a))/sum(a)^2 .
    The iterate stays on the simplex via Euclidean projection.
    The step size follows η_t = step0 / sqrt(t).
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # tolerance to capture all (numerically) maximal entries
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max‑convolution (d/d a of max_{k} conv_k)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        # logging every 50 k iterations
        if it % 50_000 == 0:
            print(
                f"    [Sub‑grad] it={it:,d} surrogate≈{fast_score(a):.6f}"
            )

    return a


# ----------------------------------------------------------------------
# Simple random local search (pairwise mass transfers)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.025,
    min_improve: float = 1e-9,
) -> np.ndarray:
    """
    Pairwise mass‑transfer hill‑climbing.
    Each move picks i≠j and transfers a random fraction (up to
    ``max_transfer``) of a[i] to a[j].  The move is kept only if the
    surrogate objective improves by at least ``min_improve``.
    """
    a = a0.copy()
    best_score = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to move (relative to a[i])
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta
        sc = fast_score(a)
        if sc + min_improve < best_score:
            best_score = sc
        else:
            # revert
            a[j] -= delta
            a[i] += delta
    return a

# Helper: fine grid search for power‑law exponent and Beta‑parameter
# ----------------------------------------------------------------------


def best_power_shape(n: int, low: float = 0.30, high: float = 0.70, steps: int = 41) -> Tuple[np.ndarray, float]:
    """Return the power‑law shape with minimal fast_score on a uniform grid."""
    best_val = float("inf")
    best_vec = None
    for p in np.linspace(low, high, steps):
        vec = generate_power_shape(n, p)
        val = fast_score(vec)
        if val < best_val:
            best_val, best_vec = val, vec
    return best_vec, best_val


def best_beta_shape(n: int, low: float = 0.30, high: float = 2.00, steps: int = 41) -> Tuple[np.ndarray, float]:
    """Return the symmetric Beta(α,α) shape with minimal fast_score on a uniform grid."""
    best_val = float("inf")
    best_vec = None
    for a in np.linspace(low, high, steps):
        vec = generate_beta_shape(n, a, a)
        val = fast_score(vec)
        if val < best_val:
            best_val, best_vec = val, vec
    return best_vec, best_val

        except Exception:
            pass

    # --------------------------------------------------------------
    # We concentrate on a single length that historically gives good

    for n in target_lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} (time left {deadline - time.time():.1f}s) ===")

        # ------------------------------------------------------------------
