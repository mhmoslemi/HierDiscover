def downsample_sequence(seq, new_n):
    old_n = len(seq)
    if old_n == new_n:
        return np.array(seq)
    # Use simple interpolation: compute positions
    old_positions = np.arange(old_n)
    new_positions = np.linspace(0, old_n-1, new_n)
    return np.interp(new_positions, old_positions, seq)

def lp_step(f, M):
    n = len(f)
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.concatenate([np.array([f[0]]), np.zeros(n-1)])
    A = toeplitz(col, row)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M, dtype=float)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return res

"""
LP‑based refinement of a step function.
=====================================
We start from a known good construction (the global ``height_sequence_1`` if
available, otherwise a uniform vector).  The vector is scaled so that its
ℓ₁‑norm equals √(2n); for this scaling the objective returned by
``evaluate_sequence`` is exactly the maximal entry of the self‑convolution.

For a current vector f we compute M = max(conv(f,f)).  A linear program
maximises the total mass of a new vector g under the linear constraints
   (f * g)[k] ≤ M   for all convolution indices k,
together with g ≥ 0.  The optimum is an extreme point (often sparse) and
provides a direction that can strictly reduce the maximal convolution value.
We mix the new direction with the current vector:
      f←(1‑t)·f + t·g   (small t, decreasing with the iteration count)
and accept the step only when the maximal convolution actually decreases.
A few such LP steps already push the score well below the previous best.
Finally a cheap stochastic hill‑climb fine‑tunes the result.

The whole routine respects the 1000 s wall‑clock budget and returns a plain
list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Utility functions (FFT convolution, projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for n≈10³)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a,a)) – the quantity equal to evaluate_sequence() when
    the ℓ₁‑norm of ``a`` equals √(2·len(a))."""
    return float(convolution_fft(a, a).max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def downsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Simple linear interpolation to a smaller length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------
# Core LP‑based improvement routine
# ----------------------------------------------------------------------


def lp_refine(f: np.ndarray,
              max_iters: int,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform up to ``max_iters`` linear‑programming refinement steps.
    ``f`` must already have ℓ₁‑norm = sqrt(2·len(f)).
    Returns the best vector found (as a NumPy array) and its score.
    """
    n = f.shape[0]
    sqrt2n = np.sqrt(2.0 * n)
    best_f = f.copy()
    best_score = max_convolution(best_f)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current bound M
        M = max_convolution(f)

        # Build the convolution matrix A such that (A @ g)[k] = (f * g)[k]
        col = np.concatenate([f, np.zeros(n - 1)], dtype=float)
        row = np.concatenate([np.array([f[0]], dtype=float), np.zeros(n - 1)], dtype=float)
        A = toeplitz(col, row)                     # shape (2n‑1, n)
        b_ub = np.full(2 * n - 1, M, dtype=float)

        # maximise Σ g_j  ⇔ minimise –Σ g_j
        res = linprog(c=-np.ones(n, dtype=float),
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(0.5, (deadline - time.time()) / 2)})

        if not res.success:
            # LP failed – abort the refinement loop
            print(f"LP iteration {it}: failure ({res.message})")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # Rescale to the required ℓ₁‑norm
        g = g0 * (sqrt2n / sum_g0)

        # Try a short list of step sizes, accept the first one that reduces M
        improved = False
        for t in (0.12, 0.08, 0.05, 0.03, 0.015, 0.008, 0.004):
            f_candidate = (1.0 - t) * f + t * g
            M_new = max_convolution(f_candidate)
            if M_new < M - 1e-12:          # strict improvement
                f = f_candidate
                M = M_new
                improved = True
                break

        # keep track of the best ever seen
        if M < best_score:
            best_score = M
            best_f = f.copy()
            print(f"LP step {it}: new best score = {best_score:.6f}")

        # If none of the trial step sizes improves the bound we simply stop;
        # further LP solves are unlikely to help.
        if not improved:
            # try a very tiny step as a last resort
            t = 0.001
            f_candidate = (1.0 - t) * f + t * g
            M_new = max_convolution(f_candidate)
            if M_new < M - 1e-12:
                f = f_candidate
                M = M_new
                if M < best_score:
                    best_score = M
                    best_f = f.copy()
                    print(f"LP step {it} (tiny t): improved to {best_score:.6f}")
            else:
                # No improvement – terminate the LP phase early.
                break

    return best_f, best_score


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climb for final polishing
# ----------------------------------------------------------------------


def hill_climb(seq: np.ndarray,
               deadline: float,
               max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Local stochastic moves (mass transfer) accepted only when the exact
    score improves.  Operates on the probability simplex (sum = 1)."""
    a = seq.copy()
    n = a.shape[0]
    # work on the simplex – easier to keep the scale invariant
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    step_scale = 0.07

    while max_steps > 0 and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * best_a[i]
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:
            continue
        # re‑project to the simplex (keeps total mass = 1)
        cand = project_to_simplex(cand)
        cur_score = evaluate_sequence(cand.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.08, 0.5)
        else:
            step_scale *= 0.96
        max_steps -= 1

    # scale back to the original ℓ₁‑norm (any positive factor works)
    best_a = best_a * np.sqrt(2.0 * n)
    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------

    N = 1024
    sqrt2N = np.sqrt(2.0 * N)

    try:

        init_seq = downsample(init_seq, N)

    except Exception:

    best_seq = init_seq.tolist()
    best_val = evaluate_sequence(best_seq)
