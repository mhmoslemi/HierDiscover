def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are large
    sequence = [min(1000.0, x) for x in sequence]

"""
Algorithm
---------

1. Choose a vector length according to the remaining budget.
2. Search two simple one‑parameter families:
   • exponential decay   a_i ∝ exp(‑λ·i)
   • power‑law decay     a_i ∝ (i+1)^(‑p)
   Both families are normalised to sum = 1.  A fine grid of parameters
   is scanned; the best vector (smallest exact objective) is kept.
3. Starting from the best vector, run a long exponentiated‑gradient
   optimisation of a smooth log‑sum‑exp approximation of the max‑convolution.
   The schedule for the temperature β is exponential, ending at a very
   large value so that the smooth objective tightly approximates the true
   max.
4. Finish with a greedy stochastic hill‑climb that repeatedly transfers a
   small packet of mass between two positions (or swaps two entries) and
   keeps only improving moves.
5. Return the best sequence found.  All steps respect the 1000 s wall‑clock
   limit; intermediate scores are printed for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector normalised to sum = 1.
    (Same as the value returned by `evaluate_sequence` for such a vector.)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


# ----------------------------------------------------------------------
# Scan the simple families
# ----------------------------------------------------------------------


def scan_exponential_and_power(
    n: int, deadline: float
) -> tuple[np.ndarray | None, float]:
    """
    Exhaustively scan exponential and power‑law families.
    Returns the best vector and its exact score (or (None, inf) if time ran out).
    """
    best_a = None
    best_sc = float("inf")

    # exponential λ grid (including λ=0 → uniform)
    lam_grid = np.concatenate(
        (np.array([0.0]), np.logspace(-5, -2, 200))
    )
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # power‑law p grid (p=0 → uniform)
    p_grid = np.linspace(0.0, 6.0, 200)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation of the smooth max
# ----------------------------------------------------------------------


def exp_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 0.5,
    beta_end: float = 300.0,
    eta0: float = 0.6,
    eps: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed log‑sum‑exp approximation of the max‑convolution
    using a KL‑mirror (exponentiated‑gradient) step.  β is annealed
    exponentially; the step size follows η₀/√t.
    """
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed weights (log‑sum‑exp)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))  # stabilised
        w = expw / expw.sum()

        # gradient of f_β
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # step size
        eta = eta0 / np.sqrt(it)

        # KL‑mirror update and simplex renormalisation
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Long stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy random walk:
    - with prob 0.5 swap two entries,
    - otherwise transfer a fraction of mass from one index to another.
    Only moves that improve the exact score are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_sc
