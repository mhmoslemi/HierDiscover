"""
Propose a step‑function height vector that minimises
`evaluate_sequence`.  The routine combines several cheap
search ideas – analytic families, Dirichlet draws,
smooth Adam optimisation of a log‑sum‑exp surrogate,
a long sub‑gradient descent (with an active‑set
gradient), elite mixing and a final simulated‑annealing
hill‑climb.  All stages respect the overall 1000 s wall‑clock
budget.

The code is deliberately written with only NumPy (no heavy
dependencies) and uses the same cheap “compute_score”
helper (which is exactly `2·n·max(conv)` for a vector on the
probability simplex).  Because the objective used by the
harness is scale‑invariant we keep all candidates on the
simplex.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the probability simplex:
        2·n·max(a*a)
    (the same ordering as `evaluate_sequence` when sum(a)=1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (new + old)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(- (i‑μ)² / (2σ²) ), centred, normalised."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = w / w.sum()
    return a


def cosine_bump_family(n: int, width: float) -> np.ndarray:
    """a_i ∝ max(0, cos(π·(i‑μ)/width))."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    z = np.cos(np.pi * (i - mu) / width)
    w = np.clip(z, 0, None)
    if w.sum() == 0.0:
        w = np.ones_like(w)          # fall back to uniform
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """≈5 % non‑zeros uniformly spread."""
    sparsity = np.random.uniform(0.03, 0.07)        # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp) using Adam on the soft‑max layer
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 800.0,
                         lr: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).

    The smoothing factor β is increased exponentially from `beta_start`
    to `beta_end`.  The routine returns the best vector found during
    the run together with its exact `compute_score`.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)                     # initialise logits

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- current a ----
        a = softmax(z)

        # ---- smoothed weights ----
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ---- gradient w.r.t. a (exact) ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ---- back‑prop through soft‑max ----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasional exact evaluation ----
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Long sub‑gradient descent – active‑set version
# ----------------------------------------------------------------------


def subgrad_optimize_active(start_a: np.ndarray,
                            deadline: float,
                            max_iters: int = 3_000_000,
                            step0: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method for the true max‑objective
          2·n·max(a*a)
    using an active‑set gradient (all indices whose convolution
    entry is within 1e‑12 of the current maximum).  Diminishing
    stepsize 1/√t and Euclidean projection after each step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()
        # indices where the convolution is essentially maximal
        tol = max_val * 1e-12 if max_val > 0 else 1e-12
        active = np.where(conv >= max_val - tol)[0]

        # gradient = sum_{k∈active} 2·a shifted by k
        grad = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]

        # diminishing step size
        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * grad)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with a tiny simulated‑annealing component
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_800_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex.
    – 50 % swap two entries,
    – otherwise transfer a fraction of mass from one index to another.
    A gentle cooling schedule permits occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # tiny temperature (only a few % of the best score)
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    return best_seq
