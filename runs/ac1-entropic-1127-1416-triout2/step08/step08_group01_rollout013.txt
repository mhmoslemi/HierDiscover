def smooth_objective(u, n, beta):
    a = np.exp(u)
    a = a / a.sum()
    conv = convolution_fft(a, a)
    maxc = conv.max()
    expbeta = np.exp(beta * (conv - maxc))
    Z = expbeta.sum()
    f_beta = (np.log(Z) + beta * maxc) / beta
    # gradient w.r.t a:
    w = expbeta / Z
    grad_a = 2.0 * convolution_fft(a[::-1], w)[n-1:n-1+n]
    # map to grad_u:
    dot = np.dot(a, grad_a)
    grad_u = a * (grad_a - dot)
    return f_beta, grad_u

"""
Improved search for a low‑value step‑function.

The algorithm:
* Pick a length n (typically 8192 or 16384) depending on the remaining time.
* Build a family a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α) and scan a modest α‑grid.
* From the best α‑vector start a smooth‑max optimisation.
  The smooth objective fβ(a) = (1/β)·log Σ_k exp(β·(a∗a)_k) is minimised
  with a L‑BFGS quasi‑Newton method on the soft‑max parametrisation
  a_i = e^{u_i}/∑_j e^{u_j}.  The temperature β is increased
  geometrically (0.5 → … → 1200) so that the smooth problem converges
  towards the true max‑convolution.
* A short projected sub‑gradient refinement (exact max‑convolution) tightens
  the last few digits.
* Finally a lightweight stochastic hill‑climb (swap / mass‑transfer moves)
  polishes the result while staying within the time budget.

All heavy work consists of FFT‑based convolutions; the schedule is tuned
to stay well inside the 1000 s wall‑clock limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global placeholders for the current optimisation size / temperature.
# ----------------------------------------------------------------------
CURRENT_N = 0          # length of the vector (set before each L‑BFGS run)
CURRENT_BETA = 1.0     # temperature for the smooth‑max

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, projection, score)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) for a vector a that already satisfies
    ∑a = 1.  This equals the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1] if np.any(cond) else 0
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic family a_i(α) and grid search
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]^(‑α) on the simplex."""
    i = np.arange(n, dtype=np.float64)
    # log of the product (i+½)(n‑½‑i) for stability
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against infinities at the ends
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search an α‑grid and keep the best vector."""
    best_a = None
    best_score = float("inf")
    # a relatively fine grid that concentrates around the empirically good region
    alphas = np.concatenate([
        np.linspace(-0.5, 0.0, 13),   # a few centre‑heavy shapes
        np.linspace(0.0, 1.2, 31),    # the promising zone
        np.linspace(1.2, 3.0, 19)     # larger exponents for completeness
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max objective (soft‑max) and its gradient on the soft‑max parametrisation
# ----------------------------------------------------------------------


def smooth_obj_and_grad(u: np.ndarray):
    """
    Returns fβ(u) and ∇_u fβ(u) where a = softmax(u) and fβ is the
    log‑sum‑exp approximation of the max‑convolution.
    Uses the global CURRENT_N and CURRENT_BETA.
    """
    n = CURRENT_N
    beta = CURRENT_BETA

    # soft‑max (numerically stable)
    u_shift = u - np.max(u)
    exp_u = np.exp(u_shift)
    a = exp_u / exp_u.sum()

    # convolution a∗a
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # soft‑max weights for the smooth max
    exp_beta = np.exp(beta * (conv - maxc))          # stabilised
    Z = exp_beta.sum()
    f_beta = (np.log(Z) + beta * maxc) / beta

    w = exp_beta / Z                                 # probability vector

    # gradient w.r.t. a (see derivation in analysis)
    grad_a = 2.0 * convolution_fft(a[::-1], w)[n - 1:n - 1 + n]

    # chain rule to obtain ∇_u (soft‑max Jacobian)
    dot = np.dot(a, grad_a)
    grad_u = a * (grad_a - dot)
    return f_beta, grad_u


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth objective
# ----------------------------------------------------------------------


def lbfgs_optimize(u0: np.ndarray,
                   n: int,
                   beta: float,
                   max_iter: int) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Run L‑BFGS on fβ starting from u0.
    Returns:
        a_opt   – the probability vector after optimisation,
        u_opt   – the final soft‑max parameters,
        score   – exact evaluate_sequence score of a_opt.
    """
    global CURRENT_N, CURRENT_BETA
    CURRENT_N = n
    CURRENT_BETA = beta

    # Import here – SciPy might be missing on some platforms.
    from scipy.optimize import minimize

    res = minimize(fun=smooth_obj_and_grad,
                   x0=u0,
                   jac=True,
                   method='L-BFGS-B',
                   options={'maxiter': max_iter, 'disp': False})

    u_opt = res.x
    # Re‑compute a from the final u
    u_shift = u_opt - np.max(u_opt)
    exp_u = np.exp(u_shift)
    a_opt = exp_u / exp_u.sum()
    score = compute_score(a_opt)
    return a_opt, u_opt, score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 20000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient method for the exact objective."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of max‑convolution (exact): 2·a shifted by the arg‑max index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Small random‑walk polishing (hill‑climb)
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very light stochastic hill‑climb.  Moves that improve the exact score
    are always kept; occasional non‑improving moves are rejected.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a, best_sc
