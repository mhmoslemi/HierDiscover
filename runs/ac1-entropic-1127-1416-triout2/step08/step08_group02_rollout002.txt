def solve_convex_opt(n, time_deadline, max_iters=200000):
    # builds constraints and solves
    a = cp.Variable(n)
    t = cp.Variable()
    constraints = [a >= 0, cp.sum(a) == 1]

    # pre-generate constraint matrices
    # to reduce memory, we could generate on-the-fly inside loop: for k in range(2*n-1): define idx = ... then create a matrix representing sum_{i} a_i a_{k-i}
    # Use cp.quad_form with sparse matrix.

    for k in range(2*n-1):
        # compute row indices and col indices and values
        rows = []
        cols = []
        # indices i from max(0,k-(n-1)) to min(n-1,k)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # generate pairs
        # We'll generate (i, k-i) for each i
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(i)
            cols.append(j)
        # Build sparse matrix
        data = np.ones(len(rows))
        Ck = sp.csr_matrix((data, (rows, cols)), shape=(n, n))
        # Ensure matrix symmetric (it should be)
        constraints.append(cp.quad_form(a, Ck) <= t)
        # Possibly early break if time runs out; but building all constraints may be slower.
    # solve
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-5, verbose=False)
    except Exception:
        return None

    if a.value is None:
        return None
    a_opt = np.maximum(a.value, 0.0) # clip tiny negatives
    a_opt = project_to_simplex(a_opt)
    return a_opt

"""
Non‑negative step‑function optimiser for `evaluate_sequence`.

Strategy
--------
1. **Convex QCQP warm‑start** –  
   For a modest size `n0` (default 256) we solve the convex problem  

       minimise   t
       subject to Σ a_i = 1 , a_i ≥ 0
                  aᵀ C_k a ≤ t   for every convolution index k  

   where `(C_k)_{ij}=1` iff `i+j=k`.  Each constraint is a convex
   quadratic form, so the whole problem is a second‑order‑cone program.
   CVXPY+SCS finds a globally optimal shape for this size.

2. **Upscaling** – the small optimal vector is linearly interpolated to a
   much larger length (`n_big` = 4096 – 8192 depending on remaining time),
   forced onto the probability simplex and symmetrised.

3. **Smooth log‑sum‑exp optimisation** –  
   We minimise a smooth approximation of `max(conv(a,a))` using Adam
   momentum.  The smoothness parameter β is annealed from 3 to 250,
   the step size decays as `1/√t`, and the best iterate is remembered.

4. **Projected sub‑gradient descent** – polishing on the true
   `max(conv(a,a))` objective.

5. **Stochastic hill‑climb** – cheap mass‑transfer moves (swap or
   fractional shift) with simplex projection to escape tiny plateaus.

All stages respect a global wall‑clock budget of 1000 s.  The function
`propose_candidate` returns a plain list of non‑negative floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector already on the simplex (sum=1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrise(v: np.ndarray) -> np.ndarray:
    """Make the vector symmetric (v_i = v_{n-1-i}) and re‑project."""
    w = (v + v[::-1]) * 0.5
    return project_to_simplex(w)


# ----------------------------------------------------------------------
# Convex QCQP for a small problem size
# ----------------------------------------------------------------------


def convex_optimize_small(n: int, deadline: float):
    """Solve   minimise t   s.t. aᵀC_k a ≤ t   (k=0…n‑1)   with sum(a)=1, a≥0."""
    import cvxpy as cp

    a = cp.Variable(n)
    t = cp.Variable()
    constraints = [a >= 0,
                   cp.sum(a) == 1,
                   t >= 0]

    # Only half of the convolution indices are needed because the
    # full convolution is symmetric.
    for k in range(n):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        rows = np.arange(i_start, i_end + 1, dtype=int)
        cols = k - rows
        data = np.ones_like(rows, dtype=float)
        Ck = sp.csr_matrix((data, (rows, cols)), shape=(n, n))
        constraints.append(cp.quad_form(a, Ck) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    try:
        prob.solve(solver=cp.SCS,
                   max_iters=15000,
                   eps=1e-5,
                   verbose=False,
                   warm_start=True)
    except Exception:
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None

    a_opt = np.maximum(a.value, 0.0)
    a_opt = project_to_simplex(a_opt)
    return a_opt


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(start_a: np.ndarray,
                             time_deadline: float,
                             max_iters: int = 400_000,
                             beta_start: float = 3.0,
                             beta_end: float = 250.0,
                             step0: float = 1.5,
                             adam_beta1: float = 0.9,
                             adam_beta2: float = 0.999,
                             adam_eps: float = 1e-8):
    """Adam optimiser on the log‑sum‑exp surrogate of max conv."""
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights (log‑sum‑exp trick)
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smoothed objective:
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    final = compute_score(a)
    if final < best_score:
        best_score, best_a = final, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 300_000,
                     step0: float = 0.07):
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur = compute_score(a_new)
        if cur < best_score:
            best_score, best_a = cur, a_new.copy()
        a = a_new

        if t % 1000 == 0:
            # tiny jitter to escape plateaus
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final = compute_score(a)
    if final < best_score:
        best_score, best_a = final, a.copy()
    return best_a, best_score

# Stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq,
               time_deadline: float,
               max_steps: int = 500_000):
    """Simple hill‑climb with swap / fractional transfer moves."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # start moving about 10 % of the source mass

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # shift a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)

    else:
        a_small = np.full(n_small, 1.0 / n_small)

    # --------------------------------------------------------------
    # 2. Upscale to a larger length for fine‑grained optimisation
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        n_big = 8192
    elif remaining > 300:
        n_big = 4096
    else:
        n_big = 2048

    print(f"\n[Stage 2] Upscaling to n = {n_big}")
    a_big = upscale_vector(a_small, n_big)
    a_big = symmetrise(a_big)                # enforce symmetry – usually beneficial

    # --------------------------------------------------------------
    # 3. Smooth log‑sum‑exp optimisation (Adam)
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("\n[Stage 3] Smooth Adam optimisation")
        smooth_deadline = deadline - 100.0
        a_opt, scr = smooth_gradient_optimize(
            a_big,
            time_deadline=smooth_deadline,
            max_iters=600_000,
            beta_start=3.0,
            beta_end=250.0,
            step0=1.5,
        )
        print(f"  Smooth optimisation best score = {scr:.6f}")
        if scr < best_val:
            best_val = scr
            best_seq = a_opt.tolist()

    # --------------------------------------------------------------
    # 4. Projected sub‑gradient polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        print("\n[Stage 4] Sub‑gradient polishing")
        sub_deadline = deadline - 50.0
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=sub_deadline,
            max_iter=400_000,
            step0=0.07,
        )
        print(f"  Sub‑grad score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()

    # --------------------------------------------------------------
    # 5. Final stochastic hill‑climb
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n[Stage 5] Hill‑climb polishing")
        hill_seq, hill_score = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=800_000,
        )
        print(f"  Hill‑climb score = {hill_score:.6f}")
        if hill_score < best_val:
            best_val = hill_score
            best_seq = hill_seq

    # --------------------------------------------------------------
    # 6. Safeguard – fall back to a trivial uniform vector if needed
    # --------------------------------------------------------------
    if best_seq is None:
        print("\n[Fallback] Using uniform tiny vector")
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will call `evaluate_sequence` again,
    # but we also print it here for diagnostics)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
