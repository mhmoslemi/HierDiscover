"""
Propose a low‑score step sequence for ``evaluate_sequence``.

Algorithm
---------
1.   Choose a series of increasing lengths (8192 → 16384).  Larger
     sequences tend to have a lower bound, but they are also more costly.
2.   For each length build a small, diverse pool of probability vectors:
     * the analytic arcsine envelope (known good shape);
     * a uniform vector;
     * several random power‑law shapes;
     * the previously best vector up‑sampled to the new length.
3.   Rank these seeds by the cheap surrogate score
     ``2·n·max(conv(a,a))`` and keep the few best.
4.   Starting from each seed run a **smooth‑max Adam** optimisation.
     The max in the objective is approximated by a log‑sum‑exp with a
     temperature β that is increased from 5 to ≈2000, giving a very sharp
     approximation in the later stage.
5.   The Adam output is evaluated with the official ``evaluate_sequence``.
     If it improves the global best we keep it.
6.   A short **exponentiated‑gradient (KL‑mirror) polish** is applied
     after each Adam run – it works directly on the true non‑smooth
     objective while staying on the simplex.
7.   After all lengths have been processed, the current best vector is
     refined by a long **projected sub‑gradient descent** on the exact
     objective (``2·n·max(conv)``).  The step size decays as 1/√t.
8.   If any time is still left a quick **random local hill‑climb** tries
     to move a tiny amount of mass between two coordinates; improvements
     are kept.  This often nudges the solution a little further.
9.   The routine always returns a non‑empty valid sequence; if nothing
     succeeded a uniform vector of length 512 is used as a safe fallback.

All heavy work respects the 1000 s wall‑clock budget and uses only
numpy (FFT‑based convolutions) and the supplied ``evaluate_sequence``.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: assumes sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 2000.0,
    ramp_iters: int = 80000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    z0 : ndarray
        Initial log‑probabilities (log‑softmax coordinates).
    deadline : float
        Absolute time at which the optimiser must stop.
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate for Adam.

    Returns
    -------
    a : ndarray
        Optimised probability vector (sum = 1, a_i ≥ 0).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # simplex point
        conv = convolution_fft(a, a)       # length 2n‑1
        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp (weights sum to 1)
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)         # may under‑flow → zeros, fine
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of the smooth max w.r.t. a is 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central n entries

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output (optional, cheap)
        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate={compute_score(a):.6f}")

    # final point on the simplex
    a = softmax(z)
    return a


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 60_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Mirror‑descent (KL) polishing on the exact objective.
    Multiplicative updates keep the iterate on the simplex automatically.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} surrogate={compute_score(a):.6f}")

    return a


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} surrogate={compute_score(a):.6f}")

    return a


def hill_climb_local_search(
    a0: np.ndarray,
    deadline: float,
    init_eps: float = 0.005,
    decay: float = 0.999,
) -> np.ndarray:
    """
    Very cheap stochastic local improvement:
    move a tiny amount of mass between two coordinates, accept if it
    strictly reduces the true objective.
    """
    a = a0.copy()
    n = a.size
    eps = init_eps
    best_val = evaluate_sequence(a.tolist())
    print(f"    Hill‑climb start score={best_val:.6f}")

    rng = np.random.default_rng()

    while time.time() < deadline and eps > 1e-9:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # amount to transfer; cannot exceed current mass at i
        delta = eps * rng.random()
        if a[i] < delta:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # simplex is preserved automatically
        val = evaluate_sequence(a_new.tolist())
        if val < best_val:
            a = a_new
            best_val = val
            # keep the improvement
        # decay epsilon slowly
        eps *= decay

    print(f"    Hill‑climb finished score={best_val:.6f}")
    return a

def build_seed_pool(
    n: int,
    rng: np.random.Generator,
    best_sofar: List[float],
) -> List[np.ndarray]:
    """Return a list of candidate probability vectors of length n."""
    seeds = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),                     # uniform
    ]

    # random power‑law shapes
    for _ in range(5):
        p = rng.uniform(0.05, 4.0)
        eps = rng.uniform(0.0, 3.0)
        seeds.append(generate_power_shape(n, p, eps))

    # up‑sampled version of the current best (if any)
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        seeds.append(upsample_sequence(old, n))

    return seeds
