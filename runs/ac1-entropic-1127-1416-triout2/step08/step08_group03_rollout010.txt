"""
Improved step‑function optimizer.

Key ideas
---------
* Use a larger problem size (n = 16384) – the extra degrees of
  freedom allow a flatter self‑convolution and thus a smaller
  objective.
* Very fast dense‑grid search over two analytic families to obtain
  good initial points.
* Multi‑restart Adam optimisation of the smooth approximation
  f_β(a) = (1/β)·log Σ exp(β·(a*a)_k).  The smoothing parameter β
  is increased exponentially up to 400, and each run performs up
  to 150 k iterations (≈ 10 s on a modern CPU).  The best result of
  all restarts is kept.
* An improved sub‑gradient method: instead of using a single
  active index we collect *all* indices that tie for the maximal
  convolution entry (within a tiny tolerance) and compute the exact
  sub‑gradient for the max‑norm.  This gives a considerably more
  directed descent.
* Final stochastic hill‑climb with a tiny simulated‑annealing
  component and an optional symmetrisation step (the optimum is
  often symmetric).

The whole driver respects the global wall‑clock limit (default
1000 s).  The returned object is a plain Python list of floats ready
for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (two useful parameterisations)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) ; symmetric, then projected to simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) (monotone decreasing), then projected to simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse random probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)  # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector obtained by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]          # decreasing
    a = a / a.sum()
    return a


def random_dirichlet_start(n: int) -> np.ndarray:
    """Draw from a symmetric Dirichlet(α=1) – completely random on the simplex."""
    a = np.random.dirichlet(np.ones(n))
    return a


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp) using Adam on the soft‑max layer
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 3.0,
                         beta_end: float = 400.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially from
    beta_start to beta_end.
    """
    n = start_a.shape[0]
    eps = 1e-14
    # initialise z from start_a (log‑softmax)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- current a ----
        a = softmax(z)

        # ---- smoothed weights (soft‑max over the convolution) ----
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        # for numerical stability subtract maxc before exp
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # size 2n‑1, sums to 1

        # ---- gradient w.r.t. a ----
        a_rev = a[::-1]
        # convolution of reversed a with the weight vector gives Σ_k w_k a_{k‑j}
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n] + w.sum() * a  # w.sum() = 1

        # ---- back‑prop through soft‑max ----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot_ag)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasional exact check (every 400 iters) ----
        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()

    # final exact evaluation
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Improved projected sub‑gradient descent (uses full active set)
# ----------------------------------------------------------------------


def subgrad_optimize_improved(start_a: np.ndarray,
                              deadline: float,
                              max_iters: int = 2_000_000,
                              step0: float = 0.05,
                              tol_rel: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the true max‑objective
          2·n·max(a*a)
    using a diminishing stepsize  ∝ 1/√t.  The gradient is computed w.r.t.
    the *active set* of indices attaining the maximal convolution entry.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()
        # active set: all entries within tolerance of the max
        mask_active = conv >= max_val - max(tol_rel, max_val * 1e-15)
        active_idx = np.nonzero(mask_active)[0]
        t_active = active_idx.size

        # compute g = Σ_{k∈active} (a_{k‑j}+a_j)
        # term Σ a_j appears t_active times
        a_rev = a[::-1]
        # mask as float for convolution
        m = mask_active.astype(np.float64)
        conv_rev = convolution_fft(a_rev, m)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n] + t_active * a

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * grad)

        # occasional exact objective check
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with a tiny simulated‑annealing component
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex.
    – With probability 0.5 swap two entries.
    – Otherwise move a fraction of mass from one index to another.
    Simple temperature schedule permits occasional uphill steps.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature scale (tiny, we only want very occasional uphill moves)
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            # accept move
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # slowly increase step scale when we improve
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            # rejected – slightly shrink the transfer magnitude
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
