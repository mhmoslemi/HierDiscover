"""
Propose a non‑negative step sequence that minimises the official
``evaluate_sequence`` objective.

Strategy
--------
1. **Diverse parametrised sampling** – generate candidates from a
   large family of analytic shapes (arcsine, Beta, power‑law,
   Gamma‑type, mixtures with a uniform component and Dirichlet draws).
   All shapes are normalised to L¹‑norm = 1 and scored with the cheap
   FFT based ``compute_score`` which is exactly the official objective
   for ‑sum‑normalised vectors.

2. **Local refinement** – the best candidate from the random phase is
   fed to a long projected sub‑gradient descent on the exact
   max‑convolution (the same descent used in the reference code).
   This drives the solution to a nearby stationary point.

3. **Mirror‑descent polish** – a short exponentiated‑gradient (KL‑mirror)
   run with a tiny step-size further lowers the max‑convolution.

4. **Stochastic hill‑climbing** – a cheap Monte‑Carlo move set that
   transfers a small amount of mass between two coordinates while
   staying on the probability simplex.  Accepted moves strictly improve
   the score, providing tiny tweaks that are hard to obtain with smooth
   gradient methods.

All phases are guarded by the global ``deadline`` (1000 s – 0.5 s safety)
so the routine always returns a valid sequence before termination.
"""

# ----------------------------------------------------------------------
# Imports and low‑level utilities
# ----------------------------------------------------------------------
import time
from typing import List
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast exact evaluation of ``evaluate_sequence`` for a vector whose
    entries sum to 1.  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (all normalise to L¹‑norm = 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (β = ½)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β)‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape – the denominator is raised to p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_gamma_shape(
    n: int, gamma: float, delta: float = 0.5, eps: float = 0.0
) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{-γ}.
    Classic arcsine corresponds to γ=½, δ≈½, ε≈0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_arc_uniform(n: int, w: float) -> np.ndarray:
    """Convex mixture of arcsine and the uniform distribution."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = w * arc + (1.0 - w) * uni
    mix /= mix.sum()
    return mix


def generate_mixture_beta_uniform(
    n: int, alpha: float, beta: float, eps: float, w: float
) -> np.ndarray:
    """Mixture of a Beta‑like shape and the uniform distribution."""
    base = generate_beta_shape(n, alpha, beta, eps)
    uni = np.full(n, 1.0 / n)
    mix = w * base + (1.0 - w) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the exact sub‑gradient method
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # indices attaining the maximum (tiny tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # reverse mask for convolution with a
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional progress print (quiet enough)
        if it % 50000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step: float = 0.008,
) -> np.ndarray:
    """
    Mirror‑descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)          # multiplicative update
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 50000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Stochastic hill‑climbing on the simplex (tiny local tweaks)
# ----------------------------------------------------------------------
def local_hill_climb(
    a0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_moves: int = 30_000,
) -> np.ndarray:
    """Transfer a tiny amount of mass between two coordinates while staying on the simplex."""
    a = a0.copy()
    n = a.shape[0]
    best_score = compute_score(a)

    for mv in range(max_moves):
        if time.time() >= deadline:
            break

        # pick source index proportional to current mass (so we move from a 'large' entry)
        src = rng.choice(n, p=a)
        # pick a destination uniformly; if it already has a lot of mass we keep it small
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a random fraction (up to 20 %) of the source mass
        frac = rng.random() * 0.20
        eps = a[src] * frac
        if eps < 1e-15:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # re‑project onto the simplex to cure floating‑point drift
        a_new = project_to_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_score - 1e-12:          # strict improvement
            a = a_new
            best_score = sc

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low official ``evaluate_sequence`` score.
    The routine respects the 1000 s wall‑clock limit and always returns a
    non‑empty sequence of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to a previously saved construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Random parametric exploration (large lengths first)
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Random exploration for length {n} ===")
        # give each length roughly 30 % of the remaining budget
        length_deadline = min(deadline - 2.0,
                              time.time() + (deadline - time.time()) * 0.30)

        while time.time() < length_deadline:
            shape_type = rng.choice(
                [
                    "arcsine",
                    "beta",
                    "power",
                    "gamma",
                    "mix_arc_uni",
                    "mix_beta_uni",
                    "dirichlet",
                    "uniform",
                ]
            )

            if shape_type == "arcsine":
                cand = arcsine_initial(n)

            elif shape_type == "beta":
                alpha = 10 ** rng.uniform(-2.0, 1.0)       # ∈[0.01,10]
                beta = 10 ** rng.uniform(-2.0, 1.0)
                eps = rng.uniform(0.0, max(0.5, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)

            elif shape_type == "power":
                p = rng.uniform(0.1, 1.0)
                eps = rng.uniform(0.0, 2.0)
                cand = generate_power_shape(n, p, eps)

            elif shape_type == "gamma":
                gamma = rng.uniform(0.1, 1.0)
                delta = rng.uniform(0.0, 1.5)
                eps = rng.uniform(0.0, 0.5)
                cand = generate_gamma_shape(n, gamma, delta, eps)

            elif shape_type == "mix_arc_uni":
                w = rng.uniform(0.0, 1.0)
                cand = generate_mixture_arc_uniform(n, w)

            elif shape_type == "mix_beta_uni":
                alpha = 10 ** rng.uniform(-1.5, 0.5)
                beta = 10 ** rng.uniform(-1.5, 0.5)
                eps = rng.uniform(0.0, 0.5)
                w = rng.uniform(0.0, 1.0)
                cand = generate_mixture_beta_uniform(n, alpha, beta, eps, w)

            elif shape_type == "dirichlet":
                conc = rng.uniform(0.1, 2.0)
                cand = rng.dirichlet([conc] * n)

            else:   # uniform
                cand = np.full(n, 1.0 / n)

            # safeguards
            cand = np.clip(cand, 1e-15, None)
            cand /= cand.sum()

            score = compute_score(cand)
            if score < best_val:
                best_val = score
                best_seq = cand.tolist()
                print(f"  New global best len={n} score={score:.6f}")

        # ------------------------------------------------------------------
        # 2.  Polishing the current best of this length (if any)
        # ------------------------------------------------------------------
        if best_seq and len(best_seq) == n:
            remaining = deadline - time.time()
            if remaining > 10.0:
                print(f"  Polishing length {n} with sub‑gradient descent …")
                a_arr = np.array(best_seq, dtype=float)
                a_arr = subgradient_descent_exact(
                    a_arr,
                    deadline=deadline,
                    max_iter=int(min(120_000, remaining * 1500)),
                    step0=0.07,
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"  Polished (sub‑grad) new best score={val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Global polishing – mirror‑descent, sub‑gradient, hill‑climb
    # ------------------------------------------------------------------
    if best_seq:
        remaining = deadline - time.time()
        if remaining > 25.0:
            print("\n=== Global polishing phase ===")
            a_arr = np.array(best_seq, dtype=float)

            # 3a. KL‑mirror (exponentiated‑gradient)
            a_arr = exponentiated_gradient_polish(
                a_arr,
                deadline=deadline,
                max_iter=int(min(120_000, remaining * 1300)),
                step=0.008,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"After EG polish score={val:.6f}")

            # 3b. Exact sub‑gradient descent (finer tuning)
            remaining = deadline - time.time()
            if remaining > 12.0:
                a_arr = subgradient_descent_exact(
                    a_arr,
                    deadline=deadline,
                    max_iter=int(min(200_000, remaining * 1800)),
                    step0=0.07,
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"After final sub‑gradient score={val:.6f}")

            # 3c. Stochastic local hill‑climbing for tiny tweaks
            remaining = deadline - time.time()
            if remaining > 5.0:
                a_arr = local_hill_climb(
                    a_arr,
                    rng,
                    deadline=deadline,
                    max_moves=int(min(30_000, remaining * 5000)),
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"After local hill‑climb score={val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty, valid return value
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # 5.  Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
