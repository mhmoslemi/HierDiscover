def convolution_fft(a,b):
    m = a.shape[0]+b.shape[0]-1
    n_fft = 1 << ((m-1).bit_length())  # next power of 2
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A*B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

   the optimisation runs for as many iterations as the remaining wall‑clock
   budget permits, first on the modest size and then on an up‑scaled vector
   (n≈8192).

3. **Stochastic hill‑climb polishing** – inexpensive moves (swap or a
   fractional mass transfer between two positions) are attempted while
   maintaining the simplex constraint.  The move that improves the exact
   objective `2·n·max(conv(a,a))` is kept, and the step size is adapted.

The whole search is bounded by the supplied `budget_s` (default 1000 s).  The
function returns a plain list of non‑negative floats (already normalised to
sum‑to‑one).  The harness will invoke `evaluate_sequence` on the returned
list to obtain the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using the FFT.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the probability simplex:
        2 * n * max(conv(a, a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x : x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # Uniform vector if all entries are non‑positive
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear interpolation from `len(v)` to `new_len`, followed by a simplex
    projection to restore the sum‑to‑one property.
    """
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Smoothed max optimisation – projected Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 2_000_000,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    step0: float = 2.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth log‑sum‑exp surrogate of
    `max(conv(a,a))`.  The vector is projected onto the simplex after every
    update.  The smoothing parameter β is annealed exponentially.
    Returns the best vector seen and its exact score.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)  # first‑moment estimate
    v = np.zeros_like(a)  # second‑moment estimate

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # full convolution a*a
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max over the entries of conv
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smoothed objective:
        #   ∂L/∂a = 2 * (a_rev * w) on the window of length n
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)          # length = 2n‑1 + (2n‑1) -1 = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]      # slice of length n

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        # decreasing step size (roughly 1/√it) multiplied by Adam's adaptive term
        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a = a - update
        a = project_to_simplex(a)

        # occasional jitter to escape plateaus
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / np.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

        # keep the best exact score seen so far
        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simple moves) – exact objective
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 400_000,
) -> tuple[list[float], float]:
    """
    Tiny stochastic moves (swap or fractional transfer) with simplex
    projection, accepting only improvements of the exact objective
    `2·n·max(conv)`.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    steps_done = 0
    while steps_done < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        steps_done += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    final_deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Small‑scale Adam optimisation (n≈2048)
    # -----------------------------------------------------------------
    n_small = 2048
    # Analytic central‑binomial start
    a_small = np.empty(n_small, dtype=float)
    a_small[0] = 1.0
    for i in range(n_small - 1):
        a_small[i + 1] = a_small[i] * (2 * i + 1) / (2.0 * (i + 1))
    a_small = project_to_simplex(a_small)

    # time allocation: ~40 % of the budget for the small stage
    small_deadline = start_time + int(0.40 * budget_s) - 5.0
    if small_deadline > final_deadline:
        small_deadline = final_deadline

    print("Starting small‑scale Adam optimisation...")
    a_small_opt, small_score = smooth_gradient_optimize(
        a_small,
        time_deadline=small_deadline,
        max_iters=2_000_000,
        beta_start=5.0,
        beta_end=300.0,
        step0=2.0,
    )
    print(f"Small stage best score = {small_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Upscale to a larger dimension (n≈8192)
    # -----------------------------------------------------------------
    n_big = 8192
    a_big = upscale_vector(a_small_opt, n_big)

    # -----------------------------------------------------------------
    # 3. Large‑scale Adam optimisation (remaining time minus a tiny margin)
    # -----------------------------------------------------------------
    big_deadline = final_deadline - 10.0
    print("Starting large‑scale Adam optimisation...")
    a_big_opt, big_score = smooth_gradient_optimize(
        a_big,
        time_deadline=big_deadline,
        max_iters=3_000_000,
        beta_start=6.0,
        beta_end=500.0,
        step0=1.5,
    )
    print(f"Large stage best score = {big_score:.6f}")

    if big_score < best_val:
        best_val = big_score
        best_seq = a_big_opt.tolist()

    # -----------------------------------------------------------------
    # 4. Final stochastic hill‑climb polishing
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < final_deadline - 2.0:
        print("Starting hill‑climb polishing...")
        polished_seq, polished_score = hill_climb_permutation(
            best_seq,
            time_deadline=final_deadline,
            max_steps=600_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5. Safeguard – fall back to a tiny uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final verification (the harness will also call evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
