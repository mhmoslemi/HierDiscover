def random_shape(n, rng):
    # Choose shape type.
    shape_type = rng.choice(['beta','power','gamma','mixture','arcsine','cosine','sin2','uniform'])
    if shape_type == 'beta':
        alpha = 10**rng.uniform(-1.5, 0.8)  # adjust range
        beta = 10**rng.uniform(-1.5, 0.8)
        eps = rng.uniform(0.0, max(0.5, 0.02*n))
        base = generate_beta_shape(n, alpha, beta, eps)
    elif shape_type == 'gamma':
        gamma = rng.uniform(0.2, 0.8)
        delta = rng.uniform(0.0, 2.0)
        eps = rng.uniform(0.0, 2.0)
        base = generate_gamma_shape(n, gamma, delta, eps)
    elif shape_type == 'power':
        p = rng.uniform(0.2, 0.9)
        eps = rng.uniform(0.0, 2.0)
        base = generate_power_shape(n, p, eps)
    elif shape_type == 'mixture':
        w = rng.uniform(0, 1)
        arc = arcsine_initial(n)
        uni = np.full(n, 1.0/n)
        base = w * arc + (1-w) * uni
    elif shape_type == 'arcsine':
        base = arcsine_initial(n)
    elif shape_type == 'cosine':
        # raised cosine (Hann)
        i = np.arange(n)
        base = 0.5 - 0.5 * np.cos(2*np.pi*i/(n-1))
    elif shape_type == 'sin2':
        i = np.arange(n)
        base = np.sin(np.pi*(i+0.5)/n)**2
    else:
        base = np.full(n, 1.0/n)
    # Now maybe mix with uniform component
    if rng.random() < 0.3:
        w_uni = rng.uniform(0.0, 0.5)
        base = (1 - w_uni) * base + w_uni * np.full(n, 1.0/n)
    base = np.clip(base, 1e-15, None)
    base = base / base.sum()
    return base

   edge‑power and mixtures).  All candidates are normalised to have
   L1‑norm = 1 and are evaluated with a fast FFT based surrogate
   ``compute_score`` (which equals the official objective when the
   L1‑norm is 1).  The best few candidates are retained.

2. **Local polishing** –   Each retained candidate is refined by three
   complementary optimisation phases, all respecting the simplex
   (non‑negative and sum = 1):
   * KL‑mirror (exponentiated‑gradient) descent on the **exact**
     max‑convolution objective.
   * Adam optimiser on a smooth log‑sum‑exp surrogate of the max,
     with a temperature schedule that drives the surrogate to the true
     max.
   * Projected (Euclidean) sub‑gradient descent on the true max‑convolution.

   The polishing runs until the 1000 s wall‑clock budget expires,
   always keeping the best sequence seen so far.

3. **Final check** –   The best sequence is evaluated with the official
   ``evaluate_sequence`` function and returned as a plain ``list`` of
   floats.

The routine continuously monitors the deadline and guarantees that a
valid (non‑empty) sequence is returned before the time limit.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose entries
    sum to 1.  Returns   2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (all produce vectors that sum to 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape – denominator raised to p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_gamma_shape(
    n: int, gamma: float, delta: float = 0.5, eps: float = 0.0
) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^(-γ).
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_edge_power_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law singular at both ends."""
    i = np.arange(n, dtype=float)
    vals = (i + 1) ** (-p) + (n - i) ** (-p)
    vals /= vals.sum()
    return vals


def generate_exponential_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    vals /= vals.sum()
    return vals


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_cosine_shape(n: int) -> np.ndarray:
    """Raised‑cosine (Hann) window."""
    i = np.arange(n, dtype=float)
    vals = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    vals /= vals.sum()
    return vals


def generate_sine_sq_shape(n: int) -> np.ndarray:
    """Sin² window."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n) ** 2
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by sub‑gradient descent
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # gradient of max is 2 * (a ⋆ mask_rev)
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a -= eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d}  score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 250_000,
    step: float = 0.01,
) -> np.ndarray:
    """
    Mirror‑descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d}  score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max surrogate + Adam optimisation
# ----------------------------------------------------------------------
def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; maps ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 500.0,
    ramp_iters: int = 100_000,
    lr: float = 0.06,
) -> np.ndarray:
    """
    Adam on the log‑sum‑exp smooth approximation of max(conv).
    Operates in the log‑domain (soft‑max enforces the simplex).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)  # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        rev = convolution_fft(a, w_norm[::-1])
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f}  score={compute_score(a):.6f}")

    return softmax(z)


# ----------------------------------------------------------------------
# Random shape generator used during the exploratory phase
# ----------------------------------------------------------------------
def random_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a symmetric non‑negative shape and normalise it to sum = 1."""
    shape_type = rng.choice(
        [
            "beta",
            "power",
            "gamma",
            "edge_power",
            "exponential",
            "mixture",
            "arcsine",
            "cosine",
            "sine_sq",
            "uniform",
        ]
    )

    if shape_type == "beta":
        alpha = 10 ** rng.uniform(-1.5, 0.8)   # roughly [0.03, 6]
        beta = 10 ** rng.uniform(-1.5, 0.8)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        cand = generate_beta_shape(n, alpha, beta, eps)

    elif shape_type == "power":
        p = rng.uniform(0.2, 0.9)
        eps = rng.uniform(0.0, 2.0)
        cand = generate_power_shape(n, p, eps)

    elif shape_type == "gamma":
        gamma = rng.uniform(0.25, 0.75)
        delta = rng.uniform(0.1, 1.5)
        eps = rng.uniform(0.0, 0.5)
        cand = generate_gamma_shape(n, gamma, delta, eps)

    elif shape_type == "edge_power":
        p = rng.uniform(0.3, 1.2)
        cand = generate_edge_power_shape(n, p)

    elif shape_type == "exponential":
        lam = rng.uniform(1e-4, 5e-3)  # gentle decay
        cand = generate_exponential_shape(n, lam)

    elif shape_type == "mixture":
        w = rng.uniform(0.0, 1.0)
        cand = generate_mixture_shape(n, w)

    elif shape_type == "arcsine":
        cand = arcsine_initial(n)

    elif shape_type == "cosine":
        cand = generate_cosine_shape(n)

    elif shape_type == "sine_sq":
        cand = generate_sine_sq_shape(n)

    else:  # uniform
        cand = np.full(n, 1.0 / n)

    # Occasionally blend a little uniform mass – helps escaping plateaus
    if rng.random() < 0.2:
        w_uni = rng.uniform(0.0, 0.4)
        cand = (1.0 - w_uni) * cand + w_uni * np.full(n, 1.0 / n)

    # Guard against numerical pathologies
    cand = np.clip(cand, 1e-15, None)
    cand = cand / cand.sum()
    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low official ``evaluate_sequence``
    score.  The function respects the 1000 s wall‑clock limit and always
    returns a non‑empty sequence of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Random parametric search over several lengths
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]   # larger lengths explored first
    for n in lengths:
        if time.time() >= deadline - 20.0:
            break
        print(f"\n--- Random search for length {n} ---")
        # allocate a fraction of remaining time to this length
        length_deadline = min(deadline - 5.0, time.time() + (deadline - time.time()) * 0.25)
        it = 0
        # rough estimate of time per candidate for this n
        est_per = 0.0007
        max_iters = int((length_deadline - time.time()) / est_per) + 1
        max_iters = min(max_iters, 80_000)

        while time.time() < length_deadline and it < max_iters:
            it += 1
            cand_arr = random_shape(n, rng)
            val = compute_score(cand_arr)
            if val < best_val:
                best_val = val
                best_seq = cand_arr.tolist()
                print(f"New best (len={n}) it={it:,d} score={val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Polishing of the current global best (if any)
    # ------------------------------------------------------------------
    if best_seq:
        remaining = deadline - time.time()
        if remaining > 30.0:
            print("\n--- Global polishing phase ---")
            a_arr = np.array(best_seq, dtype=float)

            # 2a. KL‑mirror (exponentiated‑gradient) – long run
            a_arr = exponentiated_gradient_polish(
                a_arr,
                deadline=deadline,
                max_iter=int(min(250_000, remaining * 2000)),
                step=0.008,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val = val
                best_seq = a_arr.tolist()
                print(f"  After EG polish score={val:.6f}")

            # 2b. Adam on smooth surrogate (sharpen max)
            remaining = deadline - time.time()
            if remaining > 15.0:
                eps = 1e-12
                z0 = np.log(np.clip(a_arr, eps, None))
                a_arr = smooth_adam_optimize(
                    z0,
                    deadline=deadline,
                    n=a_arr.size,
                    beta0=5.0,
                    beta_max=800.0,
                    ramp_iters=120_000,
                    lr=0.06,
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val = val
                    best_seq = a_arr.tolist()
                    print(f"  After Adam polish score={val:.6f}")

            # 2c. Final exact sub‑gradient descent
            remaining = deadline - time.time()
            if remaining > 6.0:
                a_arr = subgradient_descent_exact(
                    a_arr,
                    deadline=deadline,
                    max_iter=400_000,
                    step0=0.07,
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val = val
                    best_seq = a_arr.tolist()
                    print(f"  After final sub‑gradient score={val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a valid return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))
        print("Fallback to simple uniform sequence.")

    # ------------------------------------------------------------------
    # 4.  Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
