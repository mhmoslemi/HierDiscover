"""Propose a step‑function sequence with a low evaluate_sequence score.

The strategy is a hybrid of:
* a systematic grid‑search over several analytic families (arcsine,
  symmetric beta, power‑law, mixtures) at increasingly large lengths;
* a fast FFT‑based evaluation of the objective (2·n·max(conv));
* a cheap local‑search (random perturbations) and an optional
  exponentiated‑gradient polish, all respecting the 1000 s wall‑clock
  budget.

The code is self‑contained, uses only numpy and the provided
evaluate_sequence, and always returns a non‑empty valid list of floats.
"""

import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (linear convolution via real FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for evaluate_sequence when sum(a) == 1:
        score = 2·n·max(conv(a,a))

    The function does **not** clip or rescale – it assumes a is already
    a probability vector, which is the case for all candidates we
    generate.
    """
    n = a.shape[0]
    conv = _convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _arcsine_initial_eps(n: int, eps: float) -> np.ndarray:
    """
    Discrete arcsine‑like envelope with a tiny regularising epsilon.
    """
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    Symmetric (or not) Beta‑distribution‐like shape on a discrete grid.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def _generate_power_shape(
    n: int, p: float, eps: float
) -> np.ndarray:
    """1 / ((i+½)(n‑½‑i)+eps)^p shape."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def _generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """
    Convex mixture of the arcsine envelope and the uniform distribution.
    """
    arc = _arcsine_initial_eps(n, eps=1e-6)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def _exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    Multiplicative mirror descent (KL divergence) on the exact
    max‑convolution objective.  The iteration stops when either the
    iteration limit or the time budget expires.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = _convolution_fft(a, a)
        mx = float(np.max(conv))

        # active set (indices where the convolution is (almost) maximal)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = _convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step (multiplicative update)
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG polish it={it:,d} score={_compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 43,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.

    The routine performs a fast grid search over several analytic
    families and a set of lengths, then applies a cheap local‑search
    (random perturbations) and an optional exponentiated‑gradient polish.
    It always returns a valid (non‑empty) list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback: use a previously discovered sequence if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"[fallback] height_sequence_1 score = {val:.6f}")
            if val < best_score:
                best_score = val
                best_seq = np.array(cand, dtype=float)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Systematic grid search over several lengths
    # ------------------------------------------------------------------
    # Lengths are tried from large to small; larger n usually yields a
    # smaller asymptotic constant but costs a bit more per evaluation.
    lengths = [32768, 16384, 8192, 4096]

    # Parameter grids (coarse – we have plenty of time for refinement later)
    eps_vals = [1e-6, 1e-4, 1e-2]
    beta_alphas = np.linspace(0.2, 4.0, 19)   # includes arcsine (≈0.5)
    power_ps = np.linspace(0.1, 2.0, 20)      # exponent in power‑law family
    mix_weights = np.linspace(0.0, 1.0, 11)   # arcsine–uniform mixture

    for n in lengths:
        if time.time() >= deadline - 8.0:
            break
        print(f"\n=== Exploring length {n} ===")

        # ----- arcsine family -------------------------------------------------
        for eps in eps_vals:
            seq = _arcsine_initial_eps(n, eps)
            score = _compute_score(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"  New best: arcsine eps={eps:.0e} → {score:.6f}")

        # ----- symmetric beta family -----------------------------------------
        for a in beta_alphas:
            for eps in eps_vals:
                seq = _generate_beta_shape(n, a, a, eps)
                score = _compute_score(seq)
                if score < best_score:
                    best_score, best_seq = score, seq.copy()
                    print(f"  New best: beta a={a:.2f} eps={eps:.0e} → {score:.6f}")

        # ----- power‑law family -----------------------------------------------
        for p in power_ps:
            for eps in eps_vals:
                seq = _generate_power_shape(n, p, eps)
                score = _compute_score(seq)
                if score < best_score:
                    best_score, best_seq = score, seq.copy()
                    print(f"  New best: power p={p:.2f} eps={eps:.0e} → {score:.6f}")

        # ----- mixture arcsine / uniform ---------------------------------------
        for w in mix_weights:
            seq = _generate_mixture_shape(n, w)
            score = _compute_score(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"  New best: mixture w={w:.2f} → {score:.6f}")

        # ----- random Dirichlet‑like seeds ------------------------------------
        for _ in range(6):
            # Gamma(1,1) → exponential – a simple positive random vector
            vec = rng.gamma(shape=1.0, scale=1.0, size=n)
            vec = np.clip(vec, 1e-12, None)
            vec /= vec.sum()
            score = _compute_score(vec)
            if score < best_score:
                best_score, best_seq = score, vec.copy()
                print(f"  New best: random seed → {score:.6f}")

        # ----- up‑sample previous best to current length (if any) --------------
        if best_seq is not None and best_seq.size != n:
            # linear interpolation, then renormalise
            x_old = np.linspace(0.0, 1.0, best_seq.size)
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, best_seq)
            up = np.clip(up, 1e-15, None)
            up /= up.sum()
            score = _compute_score(up)
            if score < best_score:
                best_score, best_seq = score, up.copy()
                print(f"  New best: up‑sampled previous best → {score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Optional exponentiated‑gradient polishing (time‑aware)
    # ------------------------------------------------------------------
    if (
        best_seq is not None
        and time.time() < deadline - 4.0
        and best_seq.size <= 16384   # larger sizes would be too slow to polish
    ):
        print("\nPolishing best sequence with exponentiated‑gradient (EG)...")
        a_polish = _exponentiated_gradient_polish(
            best_seq, deadline - 0.5, max_iter=200_000, step=0.018
        )
        score_polish = _compute_score(a_polish)
        if score_polish < best_score:
            best_score, best_seq = score_polish, a_polish.copy()
            print(f"  EG polish improved score to {score_polish:.6f}")

    # ------------------------------------------------------------------
    # 3.  Local stochastic refinement (random perturbations)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.size
        while time.time() < deadline - 1.0:
            # scale of the perturbation shrinks as we approach the deadline
            elapsed = time.time() - start
            remaining = deadline - time.time()
            rel = remaining / (deadline - start)
            sigma = 0.04 * rel  # start ~4 % noise, fade to zero
            noise = rng.normal(scale=sigma, size=n)
            cand = best_seq * (1.0 + noise)
            cand = np.clip(cand, 1e-15, None)
            cand_sum = cand.sum()
            if cand_sum == 0.0 or not np.isfinite(cand_sum):
                continue
            cand /= cand_sum
            cand_score = _compute_score(cand)
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                print(f"  Stochastic tweak improved to {cand_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        # a simple uniform fallback (length 512)
        n0 = 512
        best_seq = np.full(n0, 1.0 / n0, dtype=float)
        best_score = _compute_score(best_seq)

    # Convert to plain Python list for the final output
    result = best_seq.tolist()

    # One final official evaluation for logging (may be a bit slow for huge n)
    try:
        official = evaluate_sequence(result)
        print(f"\nSearch finished. Final official score = {official:.6f}")
    except Exception as exc:
        print(f"\nSearch finished, but official evaluation failed: {exc}")
        official = None

    return result
