"""
Propose a non‑negative step sequence that minimizes the official
``evaluate_sequence`` objective.

Algorithm
---------

1. **Parametric random search** – generate candidate shapes from a
   Beta‑integral family (the exact distribution of a Beta(α,β) on each
   discrete cell), power‑law families and simple mixtures.  All candidates
   are normalised to L¹‑norm = 1, their “fast” score
   `2·n·max(conv(a,a))` is computed via an FFT‑based convolution and the
   best shape seen so far is retained.

2. **Upscaling & refinement** – the best shape is evaluated on a larger
   length (up to 4096 points) to reduce discretisation error.  The shape is
   then further improved by three polishing phases that respect the simplex:
   * Exponentiated‑gradient (KL‑mirror) descent on the true max‑convolution.
   * Adam optimisation of a smooth log‑sum‑exp surrogate of the max.
   * Exact projected sub‑gradient descent for a final squeeze.

   All phases stop when the global time budget (≈ 1000 s) is exhausted.

3. **Safety & fallback** – if no candidate is produced a uniform
   distribution of length 512 is returned.  The function always returns a
   non‑empty list of non‑negative numbers.

The routine prints progress information, respects the wall‑clock limit and
returns the best sequence found.
"""

# ----------------------------------------------------------------------
# Imports and low‑level utilities
# ----------------------------------------------------------------------
import time
from typing import List
import numpy as np

# scipy is allowed and provides the exact beta CDF (regularised incomplete beta)
import scipy.special as sp

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (no circular wrap).
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose entries sum to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (all produce a vector with L1‑norm = 1)
# ----------------------------------------------------------------------
def beta_integral_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Exact discretisation of a Beta(α,β) density:
    a_i = CDF((i+1)/n) - CDF(i/n),  i = 0..n‑1.
    """
    # endpoints of the n bins
    x = np.linspace(0.0, 1.0, n + 1, dtype=float)
    cdf = sp.betainc(alpha, beta, x)   # regularised incomplete beta
    vals = np.diff(cdf)
    # numerical safety
    vals = np.clip(vals, 1e-18, None)
    vals /= vals.sum()
    return vals


def power_shape(n: int, p: float, eps: float = 0.0) -> np.ndarray:
    """
    Power‑law envelope a_i ∝ ((i+0.5)*(n-0.5-i)+eps)^(-p).
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    vals = np.clip(vals, 1e-18, None)
    vals /= vals.sum()
    return vals


def mixture_shape(n: int, weight_beta: float, alpha: float, beta: float) -> np.ndarray:
    """
    Linear mixture of a Beta‑integral shape and the uniform distribution.
    """
    beta_vec = beta_integral_shape(n, alpha, beta)
    uni = np.full(n, 1.0 / n)
    mix = weight_beta * beta_vec + (1.0 - weight_beta) * uni
    mix = np.clip(mix, 1e-18, None)
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the exact sub‑gradient method
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Polishing phases
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step: float = 0.01,
) -> np.ndarray:
    """
    KL‑mirror (multiplicative) updates on the true max‑convolution.
    The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # gradient of max‑convolution w.r.t. a
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if not np.isfinite(s) or s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d}  score={compute_score(a):.6f}")

    return a


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 600.0,
    ramp_iters: int = 80_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam on the log‑sum‑exp (smooth max) surrogate.
    Operates in the log‑domain; the simplex constraint is enforced by a
    soft‑max mapping.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ∈ simplex
        conv = convolution_fft(a, a)

        # temperature schedule – slowly sharpen the smooth max
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max: log‑sum‑exp
        shifted = conv - np.max(conv)          # avoid overflow
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        w_norm = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

        # gradient of smooth max w.r.t. a
        rev = convolution_fft(a, w_norm[::-1])
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max (see e.g. “gradient of softmax”)
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d}  β={beta:.1f}  score={compute_score(a):.6f}")

    return softmax(z)


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the probability simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d}  score={compute_score(a):.6f}")

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official
    ``evaluate_sequence`` score.  The function respects a ~1000 s wall‑clock
    limit and always returns a valid (non‑empty) list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5        # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")
    best_len = 0

    # ------------------------------------------------------------------
    # 0.  Load fallback if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
                best_len = len(cand)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Random parametric search (multiple lengths)
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024, 512]   # try larger lengths first
    # roughly 30 % of the total budget for the broad random search
    search_deadline = start + budget_s * 0.3

    for n in lengths:
        if time.time() >= search_deadline:
            break
        print(f"\n--- Random search, length={n} ---")
        # adapt local time quota
        local_deadline = min(search_deadline, time.time() + (search_deadline - time.time()) * 0.5)

        # rough estimate of time per candidate (FFT + gen); used only to cap loops
        est_time_per = 0.0007
        max_iters = int((local_deadline - time.time()) / est_time_per)
        max_iters = max(1, max_iters)

        for it in range(max_iters):
            if time.time() >= local_deadline:
                break

            # ----- pick a shape type and parameters -----
            shape_type = rng.choice(["beta", "power", "mix", "uniform"])

            if shape_type == "beta":
                # log‑uniform sampling of α,β in [0.2,5]
                log_a = rng.uniform(-0.7, 0.7)
                log_b = rng.uniform(-0.7, 0.7)
                alpha = 10 ** log_a
                beta = 10 ** log_b
                cand = beta_integral_shape(n, alpha, beta)

            elif shape_type == "power":
                p = rng.uniform(0.2, 0.9)          # arcsine≈0.5
                eps = rng.uniform(0.0, 1.0)
                cand = power_shape(n, p, eps)

            elif shape_type == "mix":
                # mixture of beta‑integral and uniform
                w = rng.uniform(0.0, 1.0)
                log_a = rng.uniform(-0.7, 0.7)
                log_b = rng.uniform(-0.7, 0.7)
                alpha = 10 ** log_a
                beta = 10 ** log_b
                cand = mixture_shape(n, w, alpha, beta)

            else:   # uniform
                cand = np.full(n, 1.0 / n)

            # ensure positivity and proper normalisation (numerical safety)
            cand = np.clip(cand, 1e-18, None)
            cand = cand / cand.sum()

            # ----- fast evaluation -----
            val = compute_score(cand)
            if val < best_val:
                best_val, best_seq, best_len = val, cand.tolist(), n
                print(f"New best len={n} it={it+1:,d}  score={val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Optional upscale to a larger length (if time permits)
    # ------------------------------------------------------------------
    if best_seq:
        remaining = deadline - time.time()
        # try to double the length while preserving the shape (Beta‑integral)
        target_n = min(8192, best_len * 2)
        if target_n > best_len and remaining > 5.0:
            print(f"\n--- Upscaling best shape from {best_len} → {target_n} ---")
            # Infer parameters from the current best by fitting a Beta distribution
            # (simple heuristic: match first two moments)
            a_arr = np.array(best_seq, dtype=float)
            # moments
            mu = np.dot(np.arange(best_len), a_arr) / a_arr.sum()
            # avoid division by zero
            if mu > 0 and mu < 1:
                # crude method of moments for symmetric beta
                var = np.dot((np.arange(best_len) / best_len - mu) ** 2, a_arr)
                # estimate α=β from var = 1/(4(2α+1))
                est_alpha = max(0.01, (1.0 - 4 * var) / (8 * var))
                est_beta = est_alpha
            else:
                est_alpha = est_beta = 0.5
            cand_up = beta_integral_shape(target_n, est_alpha, est_beta)
            val_up = compute_score(cand_up)
            if val_up < best_val:
                best_val, best_seq, best_len = val_up, cand_up.tolist(), target_n
                print(f"Upscaled best score = {val_up:.6f}")

    # ------------------------------------------------------------------
    # 3.  Global polishing – three‑stage refinement
    # ------------------------------------------------------------------
    if best_seq:
        remaining = deadline - time.time()
        if remaining > 25.0:
            print("\n--- Global polishing phase ---")
            a_arr = np.array(best_seq, dtype=float)

            # 3a. Exponentiated‑gradient (KL‑mirror)
            a_arr = exponentiated_gradient_polish(
                a_arr,
                deadline=deadline,
                max_iter=int(min(150_000, remaining * 1200)),
                step=0.009,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"After EG polish score={val:.6f}")

            # 3b. Adam on smooth max
            if time.time() < deadline - 5.0:
                z0 = np.log(np.clip(a_arr, 1e-12, None))
                a_arr = smooth_adam_optimize(
                    z0,
                    deadline=deadline,
                    n=a_arr.size,
                    beta0=5.0,
                    beta_max=800.0,
                    ramp_iters=120_000,
                    lr=0.07,
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"After Adam polish score={val:.6f}")

            # 3c. Exact sub‑gradient descent (final squeeze)
            if time.time() < deadline - 2.0:
                a_arr = subgradient_descent_exact(
                    a_arr,
                    deadline=deadline,
                    max_iter=300_000,
                    step0=0.07,
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"After final sub‑gradient score={val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))
        print("Fallback to uniform sequence (no candidate found).")

    # ------------------------------------------------------------------
    # 5.  Final official evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
