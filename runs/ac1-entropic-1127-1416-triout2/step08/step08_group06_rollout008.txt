def exponentiated_gradient_polish(a0, deadline, max_iter=None, init_step=0.02):
    a = a0.copy()
    n = a.size
    L = 2*n-1
    it = 0
    while time.time() < deadline and (max_iter is None or it < max_iter):
        it += 1
        conv = convolution_fft(a, a)
        mx = np.max(conv)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]
        step = init_step / np.sqrt(it)
        a = a * np.exp(-step * subgrad)
        total = a.sum()
        if total <= 0 or not np.isfinite(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total
        if it % 20000 == 0:
            print(f"    EG step={it} score={compute_score(a):.6f}")
    return a

"""
Propose a non‑negative step sequence that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Random parametric seeding** – generate many candidates from a
   rich family of U‑shaped distributions (beta, power‑law, generalized
   arcsine, mixtures, cosine‑inverse).  All candidates are normalised to
   ℓ₁‑norm = 1 and scored with a fast FFT‑based surrogate
   ``compute_score`` (identical to the official objective under this
   normalisation).

2. **Gradient polishing** – the current global best is refined by three
   complementary optimisation phases, all staying on the probability
   simplex:
   * KL‑mirror (exponentiated‑gradient) descent with a decaying step‑size.
   * Adam on the smooth log‑sum‑exp approximation of the max‑convolution,
     using a temperature schedule that drives the approximation to the
     true max.
   * Exact projected sub‑gradient descent on the true max‑convolution
     (the same gradient as the smooth version, but without smoothing).

3. **Local hill‑climbing** – a cheap stochastic neighbour‑search that
   transfers a tiny amount of mass between random coordinates while
   preserving the ℓ₁‑norm.  The best neighbour is kept; the process runs
   until the time budget is exhausted.

The routine monitors the wall‑clock and always returns a non‑empty
sequence before the 1000 s budget expires.  Progress and intermediate
bounds are printed for debugging.

All helper functions are defined at module level (no closures) and only
standard scientific packages are used.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose entries
    sum to 1.  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (all normalise to ℓ₁‑norm = 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine envelope (β = ½)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β)‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape – denominator raised to p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_gamma_shape(n: int, gamma: float, delta: float = 0.5,
                         eps: float = 0.0) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{-γ}.
    The classic arcsine corresponds to γ=0.5, δ≈0.5, ε≈0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_spike_mixture(n: int, w_arc: float, w_spike: float) -> np.ndarray:
    """
    Mixture of arcsine and two endpoint spikes.
    Total weight = 1, split as w_arc for arcsine,
    the rest (1‑w_arc) equally put on the first and last index.
    """
    arc = arcsine_initial(n)
    mix = w_arc * arc
    spike_weight = (1.0 - w_arc) * w_spike / 2.0
    mix[0] += spike_weight
    mix[-1] += spike_weight
    mix /= mix.sum()
    return mix


def generate_cosine_inverse_shape(n: int, eps: float = 1e-12) -> np.ndarray:
    """
    U‑shaped envelope generated by 1/(1+cos).
    The denominator vanishes at the ends, creating a strong singularity.
    """
    i = np.arange(n, dtype=float)
    theta = np.pi * i / (n - 1)
    denom = 1.0 + np.cos(theta) + eps
    vals = 1.0 / denom
    total = vals.sum()
    return vals / total


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the exact sub‑gradient method
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Polishers (all stay on the simplex)
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(a0: np.ndarray, deadline: float,
                                 max_iter: int = 200_000,
                                 init_step: float = 0.02) -> np.ndarray:
    """
    KL‑mirror (exponentiated‑gradient) descent on the true max‑convolution.
    Step size decays as 1/√t.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        step = init_step / np.sqrt(it)
        a = a * np.exp(-step * subgrad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if it % 20_000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(z0: np.ndarray, deadline: float, n: int,
                         beta0: float = 5.0, beta_max: float = 800.0,
                         ramp_iters: int = 80_000, lr: float = 0.07) -> np.ndarray:
    """
    Adam on the log‑sum‑exp smooth approximation of max(conv).
    The temperature β is increased linearly to ``beta_max``.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)
        conv = convolution_fft(a, a)

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp
        mx = float(np.max(conv))
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient wrt a : 2·(a ⋆ w_rev)
        rev = convolution_fft(a, w_norm[::-1])
        grad_a = 2.0 * rev[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 20_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    return softmax(z)


def subgradient_descent_exact(init_a: np.ndarray, deadline: float,
                             max_iter: int = 300_000,
                             step0: float = 0.07) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses a decreasing step size ~1/√t.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 50_000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing
# ----------------------------------------------------------------------
def hill_climb(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Very cheap stochastic neighbour search.  Mass is transferred between two
    random coordinates while preserving the ℓ₁‑norm.  Accept only improvements.
    """
    a = a0.copy()
    best_score = compute_score(a)
    n = a.size
    it = 0
    # initial perturbation scale (a typical entry is ~1/n)
    scale = max(1e-4, 5.0 / n)

    while time.time() < deadline:
        it += 1
        # geometric decay of the perturbation magnitude
        scale *= 0.999999
        i, j = rng.choice(n, size=2, replace=False)

        # propose a transfer delta from i to j
        delta = (rng.random() - 0.5) * scale
        # clamp to keep a non‑negative
        if delta > a[i]:
            delta = a[i]
        if -delta > a[j]:
            delta = -a[j]
        if abs(delta) < 1e-16:
            continue

        a[i] -= delta
        a[j] += delta

        val = compute_score(a)
        if val < best_score:
            best_score = val
            # keep the move
        else:
            # revert
            a[i] += delta
            a[j] -= delta

        # occasional larger random perturbation to escape plateaus
        if it % 30_000 == 0:
            idx = rng.choice(n, size=5, replace=False)
            noise = rng.normal(scale=scale * 10, size=5)
            a_tmp = a.copy()
            a_tmp[idx] += noise
            a_tmp = np.clip(a_tmp, 0.0, None)
            a_tmp /= a_tmp.sum()
            val_tmp = compute_score(a_tmp)
            if val_tmp < best_score:
                a = a_tmp
                best_score = val_tmp

        if it % 100_000 == 0:
            print(f"    Hill‑climb it={it:,d} score={best_score:.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence (non‑negative, ℓ₁‑normalised) that minimises
    the official ``evaluate_sequence`` objective.
    The function respects a 1000 s wall‑clock limit and always returns a
    non‑empty list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.3               # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Random parametric search over several lengths
    # ------------------------------------------------------------------
    lengths = [16384, 8192, 4096, 2048, 1024, 512]

    for n in lengths:
        if time.time() >= deadline - 15.0:   # keep a few seconds for polishing
            break
        print(f"\n=== Random search for length {n} ===")
        # allocate roughly 25 % of the remaining budget to this length,
        # but cap to a few minutes.
        remaining = deadline - time.time()
        length_deadline = time.time() + min(0.25 * remaining, 120.0)

        # estimate time per evaluation (empirical constant)
        est_time = 0.0006 * (n / 2048.0)   # rough scaling
        max_iters = int((length_deadline - time.time()) / est_time)
        max_iters = min(max_iters, 120_000)   # hard cap per length

        iter_cnt = 0
        while time.time() < length_deadline and iter_cnt < max_iters:
            iter_cnt += 1

            # ------------------------------------------------------------------
            # pick a generator and random parameters
            # ------------------------------------------------------------------
            shape_type = rng.choice(
                ["beta", "power", "gamma", "mixture", "spike", "cosine", "arcsine", "uniform"]
            )

            if shape_type == "beta":
                # U‑shaped beta: alpha, beta < 1 are most promising
                alpha = 10 ** rng.uniform(-2.0, 0.0)   # [0.01, 1]
                beta = 10 ** rng.uniform(-2.0, 0.0)
                eps = rng.uniform(0.0, 0.5)
                cand = generate_beta_shape(n, alpha, beta, eps)

            elif shape_type == "power":
                p = rng.uniform(0.2, 0.9)               # around 0.5
                eps = rng.uniform(0.0, 2.0)
                cand = generate_power_shape(n, p, eps)

            elif shape_type == "gamma":
                gamma = rng.uniform(0.25, 1.0)           # include >0.5
                delta = rng.uniform(0.05, 1.0)
                eps = rng.uniform(0.0, 0.5)
                cand = generate_gamma_shape(n, gamma, delta, eps)

            elif shape_type == "mixture":
                w = rng.uniform(0.0, 1.0)
                cand = generate_mixture_shape(n, w)

            elif shape_type == "spike":
                w_arc = rng.uniform(0.0, 1.0)
                w_spike = rng.uniform(0.0, 1.0)
                cand = generate_spike_mixture(n, w_arc, w_spike)

            elif shape_type == "cosine":
                eps = rng.uniform(0.0, 1e-6)
                cand = generate_cosine_inverse_shape(n, eps)

            elif shape_type == "arcsine":
                cand = arcsine_initial(n)

            else:   # uniform
                cand = np.full(n, 1.0 / n)

            # normalise and guard against zeros
            cand = np.clip(cand, 1e-15, None)
            cand = cand / cand.sum()

            # --------------------------------------------------------------
            # fast evaluation
            # --------------------------------------------------------------
            val = compute_score(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.tolist()
                print(f"New best (len={n}) it={iter_cnt:,d} score={val:.6f}")

        # ------------------------------------------------------------------
        # brief polish of the current best for this length (if any time left)
        # ------------------------------------------------------------------
        if not best_seq:
            continue
        remaining = deadline - time.time()
        if remaining > 12.0:
            print(f"  Polishing length {n} with EG...")
            a_arr = np.array(best_seq, dtype=float)
            a_arr = exponentiated_gradient_polish(
                a_arr,
                deadline=deadline,
                max_iter=int(min(150_000, remaining * 1800)),
                init_step=0.018,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"  EG polish new best score={val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Global polishing phase (if we have a candidate)
    # ------------------------------------------------------------------
    if best_seq:
        remaining = deadline - time.time()
        if remaining > 25.0:
            print("\n=== Global polishing phase ===")
            a_arr = np.array(best_seq, dtype=float)

            # a) Exponentiated‑gradient (long run)
            a_arr = exponentiated_gradient_polish(
                a_arr,
                deadline=deadline,
                max_iter=int(min(300_000, remaining * 1200)),
                init_step=0.015,
            )
            val = compute_score(a_arr)
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print(f"After EG polish score={val:.6f}")

            # b) Adam on smooth surrogate with higher temperature
            if time.time() < deadline - 8.0:
                eps = 1e-12
                z0 = np.log(np.clip(a_arr, eps, None))
                a_arr = smooth_adam_optimize(
                    z0,
                    deadline=deadline,
                    n=a_arr.size,
                    beta0=5.0,
                    beta_max=800.0,
                    ramp_iters=100_000,
                    lr=0.07,
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"After Adam polish score={val:.6f}")

            # c) Exact sub‑gradient descent (final squeeze)
            if time.time() < deadline - 4.0:
                a_arr = subgradient_descent_exact(
                    a_arr,
                    deadline=deadline,
                    max_iter=400_000,
                    step0=0.08,
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"After sub‑grad polish score={val:.6f}")

            # d) Cheap stochastic hill‑climb for fine‑grained tweaks
            if time.time() < deadline - 2.0:
                a_arr = hill_climb(
                    np.array(best_seq, dtype=float),
                    deadline=deadline,
                    rng=rng,
                )
                val = compute_score(a_arr)
                if val < best_val:
                    best_val, best_seq = val, a_arr.tolist()
                    print(f"After hill‑climb score={val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # 4.  Final official evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {official:.6f}")
    except Exception as e:
        print(f"\nSearch finished – official evaluation failed: {e}")

    return best_seq
