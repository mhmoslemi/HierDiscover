"""
Propose a step‑function (non‑negative sequence) that minimizes the
``evaluate_sequence`` objective.

Algorithm outline
-----------------
* Build a rich pool of analytic “seed” shapes (uniform, arcsine,
  beta‑distribution, power‑law, triangular, sine, mixtures and an
  up‑sampled copy of the current best).
* For each chosen length (large → small) rank the seeds by a fast
  surrogate score (FFT‑based convolution, see ``compute_score``).
* Optimise the top few seeds with three stages:
    1. Adam on a smooth‐max surrogate of the max‑convolution.
    2. Mirror‑descent (exponentiated gradient) polishing on the exact
       objective.
    3. A cheap random‑local‑search (hill‑climbing) that swaps tiny
       amounts of mass between coordinates.
* Keep the best sequence seen so far; after all lengths have been
  processed, apply a final exponentiated‑gradient polish followed by
  Euclidean projected sub‑gradient descent.
* All heavy work uses FFT‑convolution (``O(n log n)``) and respects the
  1000 s wall‑clock budget.  The routine always returns a valid
  non‑empty sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective when sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) / (sum(a)**2).  Used for
    sequences that are (approximately) normalised to sum 1.
    """
    a = np.asarray(a, dtype=float)
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular bump (peak in the centre)."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - mid) / (mid + 1.0)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def sine_shape(n: int) -> np.ndarray:
    """Positive sine envelope."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n) + 1e-12
    vals /= vals.sum()
    return vals


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups

# Smooth‑max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 800.0,
    ramp_iters: int = 100_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    z0 : ndarray
        Initial log‑probabilities (log‑softmax coordinates).
    deadline : float
        Absolute time at which the optimiser must stop.
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate for Adam.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0
    best_val = float("inf")
    best_a = None

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # simplex point, sum(a)=1
        conv = convolution_fft(a, a)       # length 2n‑1
        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights (log‑sum‑exp trick)
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of the smooth‑max (2·a ⋆ reverse(w))
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central n entries

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional monitoring (optional, cheap)
        if t % 10_000 == 0:
            cur = compute_score(a)   # because sum(a)=1, this equals the objective
            if cur < best_val:
                best_val, best_a = cur, a.copy()
                print(f"    Adam it={t:,d} β={beta:.1f} score={cur:.6f}")

    # Return best seen (or final point if none better)
    if best_a is not None:
        return best_a
    return softmax(z)


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step: float = 0.018,
) -> np.ndarray:
    """Mirror descent (KL) on the true max‑convolution objective."""
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance to collect the active set
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if not np.isfinite(s) or s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5_000 == 0:
            cur = compute_score(a)
            print(f"    EG it={it:,d} score={cur:.6f}")

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.09,
) -> np.ndarray:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5_000 == 0:
            cur = compute_score(a)
            print(f"    SubGrad it={it:,d} score={cur:.6f}")

    return a


# ----------------------------------------------------------------------
# Random local hill‑climbing (mass‑transfer) search
# ----------------------------------------------------------------------


def random_local_search(a0: np.ndarray, deadline: float, rng: np.random.Generator,
                        step_scale: float = 0.04, max_trials: int = 200_000) -> np.ndarray:
    """
    Simple hill‑climbing: repeatedly transfer a tiny amount of mass
    from a random index i to a random index j (keeping the simplex).
    Accept only improving moves.
    """
    a = a0.copy()
    n = a.size
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_trials:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # keep a[i] non‑negative after transfer
        max_delta = a[i]
        if max_delta <= 1e-15:
            continue
        delta = max_delta * rng.random() * step_scale
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = compute_score(a_new)
        if cur < best_score:
            a = a_new
            best_score = cur
            # occasional progress report
            if it % 25_000 == 0:
                print(f"    Random LS it={it:,d} best_score={best_score:.6f}")

    return a


# ----------------------------------------------------------------------
# Helper: build a diverse pool of seeds for a given length
# ----------------------------------------------------------------------


def build_seed_pool(
    n: int,
    rng: np.random.Generator,
    best_sofar: list[float] | None,
) -> list[np.ndarray]:
    """Return a list of candidate probability vectors of length n."""
    seeds: list[np.ndarray] = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),
        triangular_shape(n),
        sine_shape(n),
    ]

    # random beta‑shape grid + random draws
    alphas = [0.2, 0.5, 1.0, 2.0, 4.0]
    betas = [0.2, 0.5, 1.0, 2.0, 4.0]
    for a in alphas:
        for b in betas:
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(generate_beta_shape(n, a, b, eps))

    for _ in range(12):
        a = rng.uniform(0.05, 5.0)
        b = rng.uniform(0.05, 5.0)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        seeds.append(generate_beta_shape(n, a, b, eps))

    # random power‑law shapes
    for _ in range(12):
        p = rng.uniform(0.05, 4.0)
        eps = rng.uniform(0.0, 2.0)
        seeds.append(generate_power_shape(n, p, eps))

    # mixtures of arcsine & uniform
    for _ in range(6):
        w = rng.random()
        seeds.append(generate_mixture_shape(n, w))

    # up‑sample current best (if any)
    if best_sofar:
        old = np.array(best_sofar, dtype=float)
        seeds.append(upsample_sequence(old, n))

    # ensure all seeds are normalised to sum = 1
    normed = []
    for s in seeds:
        s = np.maximum(s, 0.0)
        tot = s.sum()
        if tot <= 0.0 or not np.isfinite(tot):
            continue
        normed.append(s / tot)
    return normed


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 43, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low
    ``evaluate_sequence`` value. The routine respects a 1000 s wall‑clock
    budget and always returns a non‑empty valid sequence.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq: list[float] = []
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0.  Fallback: use a previously discovered sequence if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Optimise a cascade of increasing lengths (large → small)
    # --------------------------------------------------------------
    # Upper bound on length: stay below 32768 to keep per‑iteration cost reasonable.
    lengths = [32768, 16384, 8192, 4096, 2048, 1024, 512]

    for n in lengths:
        remaining = deadline - time.time()
        if remaining < 12.0:               # not enough time for another length
            break
        print(f"\n=== Optimising length {n} (remaining ≈ {remaining:.1f}s) ===")

        # ---- 1a. Build a diverse pool of seeds -----------------------
        seed_pool = build_seed_pool(n, rng, best_seq)

        # ---- 1b. Rank seeds by the fast surrogate --------------------
        scored = [(compute_score(s), s) for s in seed_pool]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]   # keep at most five

        # ---- 1c. Allocate time per seed -------------------------------
        remaining = deadline - time.time()
        # give a generous share to each of the promising seeds
        per_seed_time = max(6.0, remaining / (len(top_seeds) * 2.0 + 1.0))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 6.0:
                break
            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_time:.1f}s")
            seed_deadline = min(time.time() + per_seed_time, deadline - 4.0)

            # initialise Adam on the log‑softmax of the seed
            eps = 1e-14
            z0 = np.log(seed_vec + eps)

            # ------------------------------------------------------
            # 2. Smooth‑max Adam optimisation
            # ------------------------------------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=800.0,
                ramp_iters=120_000,
                lr=0.07,
            )

            # ------------------------------------------------------
            # 3. Official evaluation (fast surrogate, sum=1)
            # ------------------------------------------------------
            cand_val = compute_score(a_opt)      # because sum(a_opt)=1
            print(f"   After Adam surrogate score = {cand_val:.6f}")
            if cand_val < best_val:
                best_val, best_seq = cand_val, a_opt.tolist()
                print(f"   *** New global best (surrogate) = {best_val:.6f} ***")

            # ------------------------------------------------------
            # 4. KL‑mirror polishing (if time permits)
            # ------------------------------------------------------
            if time.time() < deadline - 5.0:
                polish_deadline = deadline - 2.5
                a_pol = exponentiated_gradient_polish(
                    np.array(best_seq, dtype=float),
                    polish_deadline,
                    max_iter=120_000,
                    step=0.018,
                )
                cand_val2 = compute_score(a_pol)
                print(f"   After EG polish surrogate score = {cand_val2:.6f}")
                if cand_val2 < best_val:
                    best_val, best_seq = cand_val2, a_pol.tolist()
                    print(f"   *** New global best (EG) = {best_val:.6f} ***")

            # ------------------------------------------------------
            # 5. Random local search (tiny hill‑climb)
            # ------------------------------------------------------
            if time.time() < deadline - 4.0:
                ls_deadline = deadline - 1.5
                a_ls = random_local_search(
                    np.array(best_seq, dtype=float),
                    ls_deadline,
                    rng,
                    step_scale=0.02,
                )
                cand_val3 = compute_score(a_ls)
                print(f"   After random LS surrogate score = {cand_val3:.6f}")
                if cand_val3 < best_val:
                    best_val, best_seq = cand_val3, a_ls.tolist()
                    print(f"   *** New global best (LS) = {best_val:.6f} ***")

    # --------------------------------------------------------------
    # 6. Final global polishing (exact sub‑gradient + EG)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal polishing: exact sub‑gradient descent")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=300_000,
            step0=0.09,
        )
        best_val = compute_score(a_arr)
        best_seq = a_arr.tolist()
        print(f"   After sub‑gradient polish surrogate score = {best_val:.6f}")

    if best_seq and time.time() < deadline - 1.5:
        print("\nFinal polishing: exponentiated‑gradient")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = exponentiated_gradient_polish(
            a_arr,
            deadline,
            max_iter=180_000,
            step=0.016,
        )
        best_val = compute_score(a_arr)
        best_seq = a_arr.tolist()
        print(f"   After final EG surrogate score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))

    # Quick verification with the official evaluator (optional)
    try:
        official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Official score = {official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Could not evaluate officially: {e}")

    return best_seq
