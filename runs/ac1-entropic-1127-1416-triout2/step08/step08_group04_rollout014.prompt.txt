def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # all non‑positive, return uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:

    the surrogate is 2·n·max(conv(a,a)).

    return 2.0 * n * float(np.max(conv))

def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """
    Generalised arcsine shape:
    a_i ∝ ((i+½)*(n‑½‑i) + ε)^{-p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max approximation of max(conv).
    The optimisation variable are logits w such that a = softmax(w).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)                 # stabilise
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)            # length L

        # smooth max via log‑sum‑exp
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output
        if it % 50000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries that are (almost) maximal
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent w.r.t. KL).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 5000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Very cheap stochastic hill climbing:
    move a tiny amount of mass from one coordinate to another.
    Accept only improving moves (according to the surrogate score).
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.2 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a small fraction of the source mass
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1

    return a_cur
