"""
Non‑negative step‑function minimiser for `evaluate_sequence`.

Strategy
--------
* **Analytic seed** – start from the normalized central‑binomial coefficients
  (a_i = C(2i,i)/4^i), a well‑known near‑optimal shape.

* **Linear‑programming smoothing** – for a modest size (n≈1024) we repeatedly
  solve the LP  

        maximise   Σ b_j
        s.t.       (a * b)_k ≤ M   (k = 0…2n‑2)
                    b ≥ 0 ,

  where M = max(a*a).  The solution adds as much mass as possible without
  raising the worst convolution entry.  Mixing a←(1‑t)a + t b (t≈0.2) is
  guaranteed to decrease the objective.  The LP is solved with SciPy’s
  high‑performance “highs” interior‑point method.

* **Upscaling** – the refined small vector is interpolated to a longer length
  (≈16384) and projected back onto the probability simplex.

* **Smooth optimisation** – minimise a log‑sum‑exp surrogate of
  max(a*a) with Adam momentum.  The smoothing parameter β is annealed
  exponentially from 5 to 300, while the step size decays as 1/√t.

* **Exact sub‑gradient polishing** – projected sub‑gradient descent on the
  true max‑objective (the active convolution shift supplies the sub‑gradient).

* **Stochastic hill‑climbing** – cheap mass‑transfer moves (swap or shift a
  fraction of mass) with simplex projection finish the search.

All stages respect the remaining wall‑clock budget.  The best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    N = next_pow2(n)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:n]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(convolution_fft(a, a).max())


def score_from_max(a: np.ndarray) -> float:
    """Exact objective for a vector already on the simplex (sum = 1)."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        # uniform vector is the projection of the all‑zero point
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# LP utilities (stage 1)
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Toeplitz‑like matrix A (size (2n‑1)×n) such that A @ b = conv(a, b).
    Column i contains the vector a shifted down by i rows.
    """
    n = a.shape[0]
    # each diagonal i carries the entire a‑vector
    diags = [a for _ in range(n)]
    offsets = np.arange(n)  # 0,…,n‑1
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def solve_lp_max_sum(A: sp.csr_matrix, M: float, warm_start: np.ndarray = None) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   A @ b ≤ M,   b ≥ 0.
    Returns optimal b (numpy array) or None on failure.
    """
    n = A.shape[1]
    # SciPy minimises c·x, so we minimise -1·x to maximise Σ x.
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M, dtype=float)

    bounds = [(0, None) for _ in range(n)]

    # Highs accepts sparse A_ub directly.
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "integrality": False},
        )
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    # Numerical noise may produce tiny negatives
    return np.maximum(res.x, 0.0)


def lp_refine(a: np.ndarray, deadline: float, max_iter: int = 30, t_mix: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑based smoothing on a modest‑size vector.
    Returns the best vector and its score.
    """
    best = a.copy()
    best_score = score_from_max(best)
    warm = None
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        M = max_convolution(best)
        A = build_conv_matrix(best)
        b_opt = solve_lp_max_sum(A, M, warm_start=warm)
        if b_opt is None:
            # should be rare; break to avoid endless failures.
            break
        warm = b_opt
        b_opt = project_to_simplex(b_opt)  # enforce sum=1
        candidate = (1.0 - t_mix) * best + t_mix * b_opt
        candidate = project_to_simplex(candidate)
        cand_score = score_from_max(candidate)
        if cand_score < best_score:
            best_score = cand_score
            best = candidate
            print(f"[LP iter {it}] improved score = {best_score:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
# Upscaling utility (small → big)
# ----------------------------------------------------------------------


def upscale_vector(v: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation to a longer length, followed by simplex projection."""
    if target_len == v.shape[0]:
        return v.copy()
    old_x = np.linspace(0.0, 1.0, v.shape[0])
    new_x = np.linspace(0.0, 1.0, target_len)
    v_new = np.interp(new_x, old_x, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 250_000,
    beta_start: float = 5.0,
    beta_end: float = 300.0,
    step0: float = 1.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Returns (best_vector, best_score).
    """
    a = a0.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)  # first moment
    v_mom = np.zeros_like(a)  # second moment
    best_a = a.copy()
    best_score = score_from_max(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal smoothing parameter β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights (log‑sum‑exp surrogate)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()  # shape (2n‑1,)

        # gradient of the surrogate: 2 * (a_rev * w) restricted to length n
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)  # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]  # central slice length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v_mom = beta2 * v_mom + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_mom / (1.0 - beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:
            cur_score = score_from_max(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Adam] iter {it:6d}  score = {best_score:.6f}")

    # final check
    final_score = score_from_max(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑objective."""
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_score = score_from_max(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))  # active shift

        # sub‑gradient for the active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = score_from_max(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[Sub‑grad] iter {it:6d}  score = {best_score:.6f}")

        a = a_new

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

    final_score = score_from_max(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb(
    a0: np.ndarray,
    deadline: float,
    max_steps: int = 600_000,
) -> tuple[np.ndarray, float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = score_from_max(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = score_from_max(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback (if a previous construction is present)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic start (central binomial coefficients)
    # ------------------------------------------------------------------
    n_small = 1024
    a_small = np.empty(n_small, dtype=float)
    a_small[0] = 1.0
    for i in range(n_small - 1):
        a_small[i + 1] = a_small[i] * (2.0 * i + 1.0) / (2.0 * (i + 1))
    a_small = project_to_simplex(a_small)

    # ------------------------------------------------------------------
    # 2. LP‑based refinement (modest size)
    # ------------------------------------------------------------------
    lp_deadline = deadline - 350.0  # keep ample time for later stages
    a_small, score_small = lp_refine(a_small, lp_deadline, max_iter=30, t_mix=0.20)
    print(f"LP phase finished. Best small‑size score = {score_small:.6f}")

    # ------------------------------------------------------------------
    # 3. Upscale to a larger length for fine‑grained optimisation
    # ------------------------------------------------------------------
    # Choose final length based on remaining budget
    remaining = deadline - time.time()
    if remaining > 650:
        n_big = 16384
    elif remaining > 450:
        n_big = 12288
    else:
        n_big = 8192

    a_big = upscale_vector(a_small, n_big)
    print(f"Upscaled to length {n_big}. Current score = {score_from_max(a_big):.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth Adam optimisation (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 260.0
    a_big, smooth_score = smooth_adam_optimize(
        a_big,
        time_deadline=smooth_deadline,
        max_iters=500_000,
        beta_start=6.0,
        beta_end=300.0,
        step0=1.2,
    )
    if smooth_score < best_val:
        best_val = smooth_score
        best_seq = a_big.tolist()
        print(f"[Adam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Exact sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 180.0:
        sub_deadline = deadline - 120.0
        a_big, sub_score = subgradient_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=sub_deadline,
            max_iter=600_000,
            step0=0.07,
        )
        if sub_score < best_val:
            best_val = sub_score
            best_seq = a_big.tolist()
            print(f"[Sub‑grad] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        hc_deadline = deadline
        a_big, hc_score = hill_climb(
            np.array(best_seq, dtype=float),
            time_deadline=hc_deadline,
            max_steps=800_000,
        )
        if hc_score < best_val:
            best_val = hc_score
            best_seq = a_big.tolist()
            print(f"[Hill‑climb] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – fall back to a simple uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_val = evaluate_sequence(best_seq)

    # Final verification (the harness will also call evaluate_sequence)
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch completed. Final reported score = {final_score:.6f}")

    # Return a plain list of floats as required by the harness
    return best_seq
