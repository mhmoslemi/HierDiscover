"""
Improved search for a low‑value step‑function.

The algorithm proceeds in three stages, each carefully timed
to respect the 1000 s wall‑clock limit.

1. **Many random restarts with a smooth‑max (log‑sum‑exp) surrogate.**
   The surrogate is annealed from β≈5 to β≈2000, and Adam
   optimisation is performed on the probability simplex.
   The best exact score of all restarts is retained.

2. **Projected sub‑gradient descent on the true max‑objective.**
   Starting from the best vector of stage 1 we run a classic
   sub‑gradient scheme with a diminishing step‑size and simple
   Euclidean projection onto the simplex.

3. **Simulated‑annealing hill‑climb.**
   Small mass‑transfer or swap moves are proposed; a move is
   accepted if it improves the exact score, otherwise it may be
   accepted with probability exp(−Δ/τ) where the temperature τ
   decays linearly with the remaining time.  The step‑size
   (fraction of mass moved) adapts automatically.

All heavy work uses FFT‑based convolution for O(n log n) time.
A modest dimension (n≈1024) is chosen so that many cheap
iterations fit into the budget, yet the search space is still
large enough to reach scores well below the previous best
≈ 1.5113.  The routine finally returns a plain Python list of
non‑negative floats (the heights of the step function).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for moderate n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def exact_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already sums to 1:
        2·n·max(conv(a,a))
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth‑max optimisation (log‑sum‑exp) with Adam
# ----------------------------------------------------------------------


def smooth_max_adam(start: np.ndarray,
                    deadline: float,
                    max_iters: int = 200_000,
                    beta_start: float = 5.0,
                    beta_end: float = 2000.0,
                    step0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Minimises a soft‑max surrogate of max(conv(a,a)) using Adam.
    Returns the best exact score encountered together with the vector.
    """
    a = start.astype(float).copy()
    n = a.size
    m1 = np.zeros_like(a)      # first moment
    m2 = np.zeros_like(a)      # second moment

    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- anneal β -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- compute soft‑max weights -----
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:                      # numerical safeguard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                         # now w is a probability vector

        # ----- gradient of soft‑max (via cross‑correlation) -----
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)   # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]  # central n entries

        # ----- Adam update -----
        m1 = 0.9 * m1 + 0.1 * grad
        m2 = 0.999 * m2 + 0.001 * (grad * grad)

        m1_hat = m1 / (1.0 - 0.9 ** it)
        m2_hat = m2 / (1.0 - 0.999 ** it)

        step = step0 / math.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 500 == 0:
            cur_sc = exact_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    final_sc = exact_score(a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgradient_descent(start: np.ndarray,
                        deadline: float,
                        max_iters: int = 500_000,
                        step0: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method for 2·n·max(conv(a,a)).
    Returns the best vector found.
    """
    a = start.astype(float).copy()
    n = a.size
    best_a = a.copy()
    best_sc = exact_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active (maximum) convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # self‑pair (i=j) appears only once; the above already handles it

        step = step0 / math.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = exact_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # small random jitter every thousand iterations to keep the dynamics alive
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_sc = exact_score(a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def annealed_hill_climb(start: np.ndarray,
                        deadline: float,
                        max_steps: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that occasionally accepts uphill moves.
    The temperature decays linearly with the remaining time.
    """
    a = start.copy()
    n = a.size
    cur_sc = exact_score(a)
    best_a = a.copy()
    best_sc = cur_sc

    # initial fraction of the mass that may be moved in one step
    step_scale = 0.07

    # pre‑compute total allowed steps as a safety bound
    steps = 0
    while steps < max_steps and time.time() < deadline:
        steps += 1
        # linearly decaying temperature
        remaining = max(deadline - time.time(), 1e-6)
        total_rem = max(deadline - (deadline - (deadline - time.time())), 1e-6)
        # proportion of total remaining time vs. original budget
        temp = (remaining / (deadline - (deadline - remaining)))  # ∈(0,1]
        # simple linear schedule: start at T0≈0.01 and end near 0
        T = 0.01 * temp

        cand = a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        new_sc = exact_score(cand)

        delta = new_sc - cur_sc
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis rule
            if T > 0 and math.exp(-delta / T) > np.random.rand():
                accept = True

        if accept:
            a = cand
            cur_sc = new_sc
            # adapt step size upward when we accept a move
            step_scale = min(step_scale * 1.04, 0.5)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        else:
            # down‑scale after a rejection
            step_scale = max(step_scale * 0.96, 1e-4)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Helper – generate random starts on the simplex
# ----------------------------------------------------------------------


def random_start(n: int) -> np.ndarray:
    """Random probability vector of length n (dense or sparse)."""
    if np.random.rand() < 0.5:
        # dense Dirichlet‑(1) – uniform on the simplex
        a = np.random.rand(n)
        a /= a.sum()
    else:
        # sparse – mass on √n positions
        k = max(2, int(math.sqrt(n)))
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k
    return a

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,

    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  All work respects the 1000 s time‑budget.
    Returns a plain Python list of floats (the sequence).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the external harness
    deadline = start_time + budget_s - 2.0

    best_vec = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  Optional “fallback” from a previously discovered sequence.
    # --------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:

            fb_score = evaluate_sequence(fb)
            print(f"Fallback evaluation = {fb_score:.6f}")
            if fb_score < best_score:

                best_vec = np.array(fb, dtype=float)
                best_vec /= best_vec.sum()               # ensure exact simplex for later stages
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1. Choose a manageable dimensionality.
    # --------------------------------------------------------------
    # 1024 gives cheap FFTs and still enough degrees of freedom.
    n = 1024
    if deadline - time.time() > 750:          # if we still have lots of time we can enlarge a bit
        n = 1536
    print(f"Working dimension n = {n}")

    # --------------------------------------------------------------
    # 2. Random‑restart smooth optimisation (Adam + log‑sum‑exp).
    # --------------------------------------------------------------
    reserve_refine = 180.0                    # seconds saved for later phases
    while time.time() < deadline - reserve_refine:

            deadline=deadline,
            max_steps=8_000_000,

    # --------------------------------------------------------------
    if best_vec is None:

    # --------------------------------------------------------------

    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_vec.tolist()
