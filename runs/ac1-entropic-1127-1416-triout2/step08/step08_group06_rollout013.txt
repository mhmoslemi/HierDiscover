"""
Propose a non‑negative step sequence that (approximately) minimizes the
official ``evaluate_sequence`` objective.

Algorithm
---------

1. **Random parametric seeding** – generate many candidate shapes
   (beta, power‑law, inverse‑sqrt, Gaussian, mixture, arcsine, uniform).
   All candidates are normalised to L1‑norm = 1 and scored with a cheap
   FFT‑based surrogate ``compute_score``.  The best *K* candidates are kept.

2. **Intensive polishing** – each kept candidate is refined by three
   complementary optimisation phases that respect the simplex:
   * exponentiated‑gradient (KL‑mirror) descent on the *exact* max‑convolution,
   * Adam optimisation of a smooth log‑sum‑exp approximation of the max
     (temperature is slowly increased up to β≈2000),
   * projected sub‑gradient descent on the true max‑convolution.

3. **Final check** – the best sequence found is evaluated with the official
   ``evaluate_sequence`` routine and returned.  The routine is written so
   that it always returns a non‑empty sequence before the 1000 s wall‑clock
   budget expires.

All helper functions are top‑level; no closures or lambdas are used.
"""

# ----------------------------------------------------------------------
# Imports and low‑level utilities
# ----------------------------------------------------------------------
import time
from typing import List, Tuple
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose L1‑norm = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals

def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β)‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total

def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape – denominator raised to p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total

def generate_gamma_shape(n: int, gamma: float, delta: float = 0.5,
                         eps: float = 0.0) -> np.ndarray:
    """
    Generalised inverse‑sqrt shape:
    a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{‑γ}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total

def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

def generate_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Symmetric discrete Gaussian, centred in the middle."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total

# Simplex utilities

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def exponentiated_gradient_polish(a0: np.ndarray, deadline: float,
                                 max_iter: int = 200_000,
                                 step: float = 0.01) -> np.ndarray:
    """
    Mirror‑descent on the original max‑convolution objective using the KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the active max‑conv term
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        a = a * np.exp(-step * subgrad)
        # renormalise (simplex projection is just normalisation for positive vec)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d}  score={compute_score(a):.6f}")

    return a

def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def smooth_adam_optimize(z0: np.ndarray, deadline: float, n: int,
                         beta0: float = 5.0, beta_max: float = 2000.0,
                         ramp_iters: int = 200_000, lr: float = 0.07) -> np.ndarray:
    """
    Adam optimisation of the log‑sum‑exp (smooth‑max) surrogate of the
    max‑convolution.  Optimises directly in the log‑probability (soft‑max)
    domain; the returned vector lies on the simplex.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ≥ 0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # temperature schedule (linear ramp)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp with parameter β
        shifted = conv - mx                  # ≤ 0
        w = np.exp(beta * shifted)           # max entry -> 1, others → 0
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a (see derivation in analysis)
        rev = convolution_fft(a, w_norm[::-1])
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # back‑prop through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f}  score={compute_score(a):.6f}")

    return softmax(z)

def subgradient_descent_exact(init_a: np.ndarray, deadline: float,
                              max_iter: int = 400_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score(a):.6f}")

    return a

            )
            cur_val = compute_score(a)
            print(f"   After final sub‑gradient score={cur_val:.6f}")
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.tolist()
                print(f"   *** New overall best via sub‑grad ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee we return something valid
    # ------------------------------------------------------------------
    if not best_seq:
        # fall back to a modest length uniform vector
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq, dtype=float))
        print(f"Fallback uniform sequence used, score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final official evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as e:
        print(f"\nSearch finished. Official evaluation failed: {e}")

    return best_seq
