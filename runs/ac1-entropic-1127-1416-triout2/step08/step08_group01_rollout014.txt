def alternating_lp_optimize(a0, deadline, max_iters=200, t=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute convolution and M
        conv = convolution_fft(a, a)
        M = conv.max()
        # Build convolution matrix A from a
        A = build_convolution_matrix(a)
        # Solve LP
        g = cp.Variable(len(a), nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except cp.error.SolverError:
            break
        g_val = g.value
        if g_val is None:
            break
        sum_g = np.sum(g_val)
        if sum_g <= 0:
            break
        # Scale to have same sum as a
        scale = a.sum() / sum_g
        g_scaled = g_val * scale
        # Combine
        a_new = (1 - t) * a + t * g_scaled
        # Ensure non-negative
        a_new = np.maximum(a_new, 0)
        # Evaluate new score
        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time():.1f}] LP iter {it} improved score {best_score}")
        # Update for next iteration
        a = a_new
    return best_a, best_score
```

from scipy.optimize import linprog

def solve_lp_max_sum(a, M):
    n = a.shape[0]
    # Build A matrix
    offsets = np.arange(n)
    # Build diag matrices stretched to shape (2n-1, n)
    # In SciPy, we can use diags([...], offsets, shape=(2*n-1, n))
    A = diags([a]*n, offsets, shape=(2*n-1, n), format='csr')
    # constraint: A @ g <= M
    b_ub = np.full(2*n-1, M, dtype=float)
    # No equality constraints
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve':True})
    if res.success:
        return res.x
    else:
        # fallback maybe return uniform
        return np.full(n, 1.0/n)
```

def lp_refine(start_a, deadline):
    a = start_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute M
        conv = convolution_fft(a, a)
        M = conv.max()
        # solve LP
        g = solve_lp_max_sum(a, M)
        sum_a = a.sum()
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # scale to sum_a
        scale = sum_a / sum_g
        g_scaled = g * scale
        # candidate combos
        candidates = []
        candidates.append(g_scaled)
        for t in (0.2,0.4,0.6,0.8):
            cand = (1-t)*a + t*g_scaled
            candidates.append(cand)
        # evaluate each

        for cand in candidates:

                improved = True
                # print improvement
                print(f"[{time.time():.1f}] LP iteration {it} improved to {best_sc}")

            # attempt with smaller t

"""
Optimisation of the non‑negative step‑function bound.

The algorithm consists of several stages:

1.  A cheap analytical start based on the family
    a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ (α‑grid search).
2.  An exponentiated‑gradient run that minimises a smooth
    log‑sum‑exp approximation of the max‑convolution.
3.  Classical Euclidean sub‑gradient descent on the exact
    max‑convolution objective.
4.  **New stage** – alternating linear‑programming refinement.
    For a current vector *a* we solve

          maximise  Σ g_j
          subject  (a * g)_k ≤ M   for all k,
                     0 ≤ g_j ≤ 1000,

    where *M* = max(a∗a).  The LP pushes the mass of *g*
    as far as possible while keeping the convolution bound.
    After scaling *g* to the same ℓ₁‑norm as *a* we either
    replace *a* by *g* or form a convex combination
    (1‑t)·a + t·g.  Repeating this step quickly drives the
    maximal convolution down, improving the final score.
5.  A short stochastic hill‑climb for polishing.

All phases respect the global 1000 s wall‑clock budget, and the
best sequence (a plain Python list of floats) is returned.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import diags
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core utilities – no closures, pure functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = rfft(a, n=nfft)
    B = rfft(b, n=nfft)
    C = A * B
    return irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) assuming `a` is normalised to sum = 1.
    This is exactly what `evaluate_sequence` returns for the same vector.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Analytical start (α‑grid)
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """Return a_i(α) ∝ [(i+½)(n‑½‑i)]⁻ᵅ on the simplex."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # avoid infinities when alpha is large and logprod is 0
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def find_best_alpha_start(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a modest α‑grid and keep the best vector."""
    best_a = None
    best_score = float("inf")
    alphas = np.concatenate([np.linspace(-1.0, 0.0, 11),
                             np.linspace(0.0, 2.5, 26)])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = alpha_start(n, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Random sparse initialisation (used for restarts)
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Create a random sparse probability vector."""
    sparsity = np.random.uniform(0.03, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# 3. Exponentiated‑gradient optimisation of the smooth max
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 250_000,
                        beta_start: float = 0.5,
                        beta_end: float = 300.0,
                        eta0: float = 0.6,
                        eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """KL‑mirror descent on the log‑sum‑exp smoothing of the max‑convolution."""
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed max – log‑sum‑exp
        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))      # stabilised
        w = expw / expw.sum()                     # soft‑max weights

        # gradient of the smoothed objective
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)     # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # step size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update (KL‑projection)
        a = np.maximum(a * np.exp(-eta * grad), eps)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional exact evaluation
        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] exp‑grad it {it} best {best_sc:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print(f"[{time.time():.1f}] exp‑grad final best {best_sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 400_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Classic Euclidean sub‑gradient descent on 2·n·max(a*a)."""
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad it {t} best {best_sc:.6f}")

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print(f"[{time.time():.1f}] sub‑grad final best {best_sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# 5. Alternating linear‑programming refinement (new stage)
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> diags:
    """Sparse matrix representing the convolution (a * x)."""
    n = a.shape[0]
    # each offset s contributes a on the diagonal placed at row i+s, column i
    offsets = np.arange(n)                     # 0 … n‑1
    data = [a] * n                             # n copies of the same diagonal
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def solve_lp_max_sum(a: np.ndarray, M: float, max_val: float = 1000.0) -> np.ndarray:
    """
    Linear programme:
        maximise Σ g_j
        s.t.   (a * g)_k ≤ M   (for all k)
               0 ≤ g_j ≤ max_val
    Returns the optimal g (numpy array).  If the problem is infeasible
    or unbounded, a uniform vector is returned as a safe fallback.
    """
    n = a.shape[0]
    A = build_conv_matrix(a)

    # RHS vector
    b = np.full(2 * n - 1, M, dtype=float)

    # Objective: minimise -Σ g  (i.e. maximise Σ g)
    c = -np.ones(n, dtype=float)

    # Variable bounds
    bounds = [(0.0, max_val) for _ in range(n)]

    # Use HiGHS via scipy.optimize.linprog – fast for large sparse problems
    res = linprog(c,
                  A_ub=A,
                  b_ub=b,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})  # a few seconds cap

    if res.success and res.x is not None:
        return res.x
    # fallback – uniform distribution (still respects the simplex)
    return np.full(n, 1.0 / n)


def lp_refine(start_a: np.ndarray,
              deadline: float,
              max_iters: int = 200) -> tuple[np.ndarray, float]:
    """
    Alternating LP refinement.
    Repeatedly (i) solves the LP that pushes mass as far as possible while
    keeping the current max‑convolution bound, (ii) rescales the solution
    to the same ℓ₁‑norm, and (iii) moves towards it if the exact score
    improves.  The loop terminates when no improvement is found or the
    time budget is exhausted.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current bound
        conv = convolution_fft(a, a)
        M = conv.max()

        # solve LP
        g = solve_lp_max_sum(a, M)

        sum_a = a.sum()                 # should be 1.0 (simplex)
        sum_g = g.sum()
        if sum_g == 0.0:
            break

        # Scale g back to the simplex
        scale = sum_a / sum_g
        g_scaled = g * scale

        # Prepare a few candidate mixtures
        cand_list = [g_scaled]                 # replace outright
        for t in (0.2, 0.4, 0.6, 0.8):
            cand = (1.0 - t) * a + t * g_scaled
            cand_list.append(cand)

        improved = False
        for cand in cand_list:
            cand = np.maximum(cand, 0.0)        # safety
            cand = project_to_simplex(cand)    # enforce exact sum = 1
            sc = compute_score(cand)
            if sc + 1e-12 < best_sc:           # strict improvement
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
                print(f"[{time.time():.1f}] LP iter {it} improved score {best_sc:.6f}")
                break

        if not improved:
            # No candidate lowered the objective → stop early
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
# 6. Final stochastic hill‑climb (light polish)
# ----------------------------------------------------------------------


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Random walk that swaps or shifts a small packet of mass.
    Only strictly improving moves are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_sc:
            best_sc = sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a, best_sc

    # -------------------------------------------------

    return best_seq
