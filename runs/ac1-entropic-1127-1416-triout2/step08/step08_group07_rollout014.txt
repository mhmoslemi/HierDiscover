def lp_step(a, M):
    n = len(a)
    # Build sparse matrix A shape (2n-1, n)
    row_idx = []
    col_idx = []
    data = []
    for k in range(2*n-1):
        # i range such that 0 <= k-i < n => i from max(0, k-(n-1)) to min(k, n-1)
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # Add entries for each i
        for i in range(i_start, i_end+1):
            row_idx.append(k)
            col_idx.append(i)
            data.append(a[i])
    A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    # bounds for g: (0, None)
    bounds = [(0, None)] * n
    # RHS for constraints
    b = np.full(2*n-1, M)
    # objective: maximize sum(g) -> minimize -sum(g)
    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'time_limit': time_remaining})
    if res.success:
        g = res.x
        return g
    else:
        return None

def build_conv_matrix(a):
    n = len(a)
    L = 2*n-1
    A = np.zeros((L, n), dtype=np.float64)
    for i, ai in enumerate(a):
        A[i:i+n, i] = ai
    return A

Now solving LP:

def lp_min_max_convolution(a):
    n = len(a)
    A = build_conv_matrix(a)
    L = 2*n-1
    # augment A with -1 column for M.
    A_ub = np.hstack([A, -np.ones((L, 1))])
    b_ub = np.zeros(L)
    # equality sum(g) = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    # bounds
    bounds = [(0, None)] * n + [(0, None)] # g_i >=0, M>=0
    # objective: minimize M (coefficient 1 for last variable)
    c = np.zeros(n+1)
    c[-1] = 1.0
    result = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                    bounds=bounds, method='highs')
    if not result.success:
        return None, None
    g = result.x[:n]
    M_opt = result.x[-1]
    return g, M_opt

"""
Improved search for a non‑negative step sequence that minimises
``evaluate_sequence``.

The optimiser works in several stages:
1.  Build a very diverse set of seed vectors (symmetric arcsine,
    asymmetric Beta, power‑law, mixtures, random Dirichlet and a
    random Fourier construction) for a single length n (default 2048).
2.  Keep the few best seeds according to a cheap surrogate
    ``fast_score``.
3.  For each seed run a cascade of optimiser phases, all bounded by
    the global time budget:
       • L2‑flattening (Adam on the squared‑error of the auto‑convolution
         to the uniform target – makes the convolution flat).
       • Smooth‑max Adam (continuation on the log‑sum‑exp of the
         convolution, a standard surrogate for the max‑operator).
       • Exact projected sub‑gradient descent on the true objective.
       • Linear‑programming refinement: solve
         ``minimise M``  s.t.  ``A·g ≤ M`` , ``Σg = 1``, ``g≥0`` where
         ``A`` encodes the convolution constraints (so that max(a*g) is
         as small as possible).  Replace the current vector with the LP
         solution when it improves the true objective.
       • Cheap random pairwise mass‑transfer local search.
4.  Keep the best vector ever found (official score from
    ``evaluate_sequence``) and return it.

All heavy work is performed with FFT‑based convolutions; the LP
uses ``scipy.optimize.linprog`` with a dense constraint matrix – for
n=2048 the matrix is only ∼64 MiB and can be solved in a few seconds.

The routine respects the overall 1000 s budget and always returns a
valid non‑negative sequence.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Cheap surrogate (scale‑invariant) used for pre‑selection
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(conv(a,a)) / (sum(a)**2) without
    clipping.  ``a`` must be non‑negative.  This is cheap (FFT) and used.
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Seed generators – a wide variety of analytic families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-12) -> np.ndarray:
    """Beta(α,β) density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Shape ∝ 1/((i+½)*(n‑½‑i)+eps)^p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw with a random concentration."""
    conc = 10 ** rng.uniform(-2.0, 0.7)          # between 0.01 and 5
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def generate_random_fourier_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Take the absolute value of an inverse FFT of random phases."""
    phases = rng.random(n) * 2.0 * np.pi
    A = np.exp(1j * phases)
    a = np.abs(np.fft.ifft(A).real)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto Σx_i = 1, x_i ≥ 0."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Adam optimiser with smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 3000.0,
    ramp_iters: int = 200_000,
    lr: float = 0.03,
) -> np.ndarray:
    """Minimise log‑sum‑exp(β·conv) via Adam, β is increased linearly."""
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0
    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))
        shifted = conv - mx
        w = np.exp(beta * shifted)          # unnormalised smooth‑max weights
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of Lβ w.r.t. a = 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 10_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate≈{fast_score(a):.6f}")
    return _softmax(z)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective (max‑convolution)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Exact sub‑gradient descent for 2·n·max(conv)/sum²."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)  # which k's are active
        # sub‑gradient of max‑conv (see derivation)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 20_000 == 0:
            print(f"    Sub‑grad it={it:,d} surrogate≈{fast_score(a):.6f}")
    return a


# ----------------------------------------------------------------------
# L2‑flattening optimiser (Adam on Σ(conv‑avg)²)
# ----------------------------------------------------------------------


def l2_flatten_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    lr: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> np.ndarray:
    """Minimise Σ (conv - avg)² via Adam on the soft‑max parametrisation."""
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    eps = 1e-12
    t = 0
    avg_target = 1.0 / (2 * n - 1)   # because sum(a)=1 ⇒ average convolution = 1/(2n‑1)

    while time.time() < deadline:
        t += 1
        a = _softmax(z)
        conv = convolution_fft(a, a)
        diff = conv - avg_target                     # length 2n‑1
        # gradient w.r.t a : 4·(diff ★ a_rev)   (derivation in analysis)
        a_rev = a[::-1]
        grad_a_full = 4.0 * convolution_fft(diff, a_rev)
        grad_a = grad_a_full[n - 1 : n - 1 + n]       # length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 8_000 == 0:
            loss = np.mean(diff ** 2)
            print(f"    L2‑flatten it={t:,d} loss≈{loss:.2e} surrogate≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Linear‑programming refinement (minimise max(a★g))
# ----------------------------------------------------------------------


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Dense matrix A where (A·g)[k] = (a★g)[k]  (full linear convolution)."""
    n = a.size
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for i, ai in enumerate(a):
        A[i : i + n, i] = ai
    return A


def lp_min_max_convolution(a: np.ndarray, deadline: float):
    """
    Solve   minimise M   s.t.   A·g ≤ M·1,
    Σg = 1, g ≥ 0.
    Returns (g, M_opt) or (None, None) on failure/timeout.
    """
    from scipy.optimize import linprog
    n = a.size
    A = _build_conv_matrix(a)               # shape (2n‑1, n)
    L = A.shape[0]

    # inequality: A·g - M ≤ 0   →  [A  -1]·[g;M] ≤ 0
    A_ub = np.hstack([A, -np.ones((L, 1))])
    b_ub = np.zeros(L)

    # equality: Σg = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])

    # bounds for variables g_i ≥0, M ≥0
    bounds = [(0, None)] * n + [(0, None)]

    # objective = minimise M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # ``highs`` works with dense matrices; we enforce a modest time limit
    # by breaking before the LP call if we are already out of budget.
    remaining = deadline - time.time()
    if remaining < 5.0:
        return None, None

    # SciPy's ``linprog`` does not expose a time‑limit flag; we rely on the
    # outer deadline to guard us.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None, None
    g = res.x[:n]
    M_opt = res.x[-1]
    return g, M_opt


def lp_refine(a: np.ndarray, deadline: float, max_iter: int = 5) -> np.ndarray:
    """
    Apply the LP refinement a few times.  Each iteration solves the LP
    to obtain a new vector g that minimises max(a★g).  If the self‑
    convolution of g improves the official score we keep it.
    """
    best = a.copy()
    best_val = fast_score(best)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        g, M_opt = lp_min_max_convolution(best, deadline)
        if g is None:
            break
        # re‑normalise (LP already returns Σg=1)
        new_val = fast_score(g)
        if new_val < best_val:
            best, best_val = g, new_val
            print(f"    LP refine iter {it+1}: improved surrogate≈{best_val:.6f}")
        else:
            # no improvement – stop early
            break
    return best


# ----------------------------------------------------------------------
# Simple random local search around a candidate
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.025,
) -> np.ndarray:
    """Pairwise mass‑transfer moves, kept only if the cheap surrogate improves."""
    a = a0.copy()
    best_score = fast_score(a)
    n = a.size
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * rng.random() * max_transfer
        if eps <= 0.0:
            continue
        a[i] -= eps
        a[j] += eps
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
            continue
        a /= s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
        else:
            a[j] -= eps
            a[i] += eps
            a /= a.sum()
    return a

    """

    best_official: float = float("inf")

    # --------------------------------------------------------------

            seeds.append(generate_beta_shape(n, a_val, b_val))

        seeds.append(generate_power_shape(n, p))

    for _ in range(12):
        seeds.append(generate_random_dirichlet(n, rng))

    # random Fourier seeds
    for _ in range(8):
        seeds.append(generate_random_fourier_shape(n, rng))

    # --------------------------------------------------------------
    # 3️⃣  Pre‑select a handful of promising seeds (cheap surrogate)
    # --------------------------------------------------------------
    scored: List[Tuple[float, np.ndarray]] = [(fast_score(v), v) for v in seeds]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:4]]   # keep at most four seeds

    # --------------------------------------------------------------
    # 4️⃣  Optimisation loop for each seed
    # --------------------------------------------------------------
    per_seed_budget = max(2.0, (deadline - time.time()) / (len(top_seeds) * 4.0))

    for s_idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 2.0:
            break
        print(f"\n--- Seed {s_idx}/{len(top_seeds)} (budget ≈{per_seed_budget:.1f}s) ---")
        sub_deadline = time.time() + per_seed_budget

        eps = 1e-12
        z0 = np.log(seed_vec + eps)

        t1 = time.time() + per_seed_budget * 0.25

        # 4b. Smooth‑max Adam (tightens the max‑convolution)
        t2 = time.time() + per_seed_budget * 0.25
        a_adam = smooth_adam_optimize(
            np.log(a_l2 + eps), t2, n,
            beta0=5.0, beta_max=3000.0, ramp_iters=200_000, lr=0.03
        )

        # 4c. Exact projected sub‑gradient descent (focus on the true max)
        t3 = time.time() + per_seed_budget * 0.20
        a_sub = subgradient_descent_exact(a_adam, t3, max_iter=200_000, step0=0.07)

        # 4d. LP‑based polishing (may provide a noticeable jump)
        t4 = time.time() + per_seed_budget * 0.15
        a_lp = lp_refine(a_sub, t4, max_iter=5)

        # 4e. Random local search around the polished solution
        t5 = max(time.time(), deadline - 0.5)
        a_final = random_local_search(a_lp, t5, rng, max_transfer=0.025)

        # ----------------------------------------------------------
        #  Official evaluation of the candidate
        # ----------------------------------------------------------
        try:
            val = evaluate_sequence(a_final.tolist())
            print(f"   Seed {s_idx} official score = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, a_final.tolist()
                best_surrogate = fast_score(a_final)
                print("   *** New global best! ***")
        except Exception as e:
            print(f"   Evaluation failed: {e}")

    # --------------------------------------------------------------
    # 5️⃣  Final global polishing (if time permits)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\nFinal polishing (exact sub‑gradient) …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(a_arr, deadline,
                                          max_iter=300_000, step0=0.05)
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_official:
                best_official, best_seq = final_val, a_arr.tolist()
                best_surrogate = fast_score(a_arr)
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final evaluation error: {e}")

    # --------------------------------------------------------------
    # 6️⃣  Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
