def lp_mix_step(a, M, t):
    n = len(a)
    A = np.empty((2*n-1, n), dtype=float)
    # fill with zeros then assign
    for i in range(n):
        A[i:i+n, i] = a[i]
    b = cp.Variable(n)
    constraints = [b >= 0, A @ b <= M * np.ones(2*n-1)]
    problem = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        problem.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-7, reltol=1e-6)
    except Exception as e:
        # fallback to uniform
        return a
    if b.value is None:
        # fallback uniform
        g = np.full_like(a, 1.0 / len(a))
    else:
        b_opt = np.array(b.value).reshape(-1)
        sum_b = b_opt.sum()
        if sum_b <= 0:
            g = np.full_like(a, 1.0 / len(a))
        else:
            g = b_opt / sum_b
    a_new = (1 - t) * a + t * g
    # ensure non-negative and normalization
    a_new = np.maximum(a_new, 0.0)
    a_sum = a_new.sum()
    if a_sum == 0:
        a_new = np.full_like(a_new, 1.0 / len(a_new))
    else:
        a_new = a_new / a_sum
    return a_new

def lp_mix_step(a, M, t, solver='ECOS'):
    n = len(a)
    A = build_conv_matrix(a)
    b = cp.Variable(n)
    constraints = [b >= 0, A @ b <= M * np.ones(2*n-1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=solver, warm_start=True, max_iters=2000)
    except Exception as e:
        # fallback uniform
        g = np.full_like(a, 1.0/n)
        a_new = (1-t)*a + t*g
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum > 0:
            a_new = a_new / a_sum
        return a_new
    if b.value is None:
        g = np.full_like(a, 1.0/n)
    else:
        b_opt = np.array(b.value).reshape(-1)
        sum_b = b_opt.sum()
        if sum_b <= 0:
            g = np.full_like(a, 1.0/n)
        else:
            g = b_opt / sum_b
    # mix
    a_new = (1 - t) * a + t * g
    a_new = np.maximum(a_new, 0.0)

        a_new = a_new / s
    return a_new

"""
Propose a step‑function that minimises ``evaluate_sequence``.

The search combines:
* a rich family of analytic seed shapes (arcsine, Beta, power‑law,
  mixtures and random Dirichlet draws);
* a cheap scale‑invariant surrogate ``fast_score`` to pre‑select the most
  promising seeds;
* a small continuation‑Adam run (smooth‑max) to obtain a better local
  optimum;
* an exact linear‑programming refinement:
  for the current distribution a compute M = max(conv(a,a)),
  solve   maximise 1ᵀb   subject to   b≥0,  conv(a,b) ≤ M,
  obtain g = b / Σb and mix a←(1‑t)a + t g (t is slowly decayed).
  This step is a conditional‑gradient move that provably never increases
  the objective and typically reduces the maximal convolution value.
* a final projected sub‑gradient descent on the true objective and a cheap
  random local‑search.

All phases respect a 1000 s wall‑time budget; the routine always returns a
valid non‑negative sequence (fallback uniform if nothing else succeeded).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional CVXPY import (used for the LP refinement)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _cvxpy_ok = True
except Exception:  # pragma: no cover
    _cvxpy_ok = False

# ----------------------------------------------------------------------
# Utility FFT‑based convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate (scale invariant) – used for cheap pre‑selection
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Return 2·n·max(conv(a,a))/sum(a)².
    ``a`` must be non‑negative, the function does not clip or rescale.
    """
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic families of seed shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 0.0) -> np.ndarray:
    """Beta(α,β) shape on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Shape proportional to 1/((i+½)*(n‑½‑i)+eps)^p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex (Dirichlet with random concentration)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)          # concentration between 0.01 and 5
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σxᵢ = 1, xᵢ ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth‑max optimisation (Adam) – cheap local polishing
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 3000.0,
    ramp_iters: int = 100_000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    Minimise the smooth approximation Lβ(a)=log‑sum‑exp(β·conv(a,a))
    using Adam on the soft‑max parametrisation.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # probability vector
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        w_norm = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # gradient of log‑sum‑exp w.r.t. a  (see derivation in original code)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d}   β={beta:.0f}   surrogate≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Linear‑programming refinement (conditional‑gradient step)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that A @ b = conv(f,b).
    Entry A[k,i] = f[i] if 0 ≤ k‑i < n else 0.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i : i + n, i] = f[i]
    return A


def _lp_refine_step(a: np.ndarray, M: float, t: float) -> np.ndarray:
    """
    One conditional‑gradient refinement step:
    maximise Σb   subject to   b≥0,   conv(a,b) ≤ M.
    The optimiser returns g = b/Σb and we mix a←(1‑t)a + t g.
    """
    if not _cvxpy_ok:          # pragma: no cover
        # CVXPY not available – fall back to a no‑op
        return a

    n = a.size
    A = _build_conv_matrix(a)
    b = cp.Variable(n)
    constraints = [b >= 0, A @ b <= M * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-8,
                   reltol=1e-6)
    except Exception:        # pragma: no cover
        # If the solver crashes we simply skip the refinement.
        return a

    if b.value is None:
        g = np.full_like(a, 1.0 / n)
    else:
        b_opt = np.array(b.value).reshape(-1)
        sum_b = b_opt.sum()
        if sum_b <= 0 or not np.isfinite(sum_b):
            g = np.full_like(a, 1.0 / n)
        else:
            g = b_opt / sum_b

    # Mix and re‑normalise (the mix automatically stays on the simplex
    # but numerical drift can appear).
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0 or not np.isfinite(s):
        a_new = np.full_like(a_new, 1.0 / n)
    else:
        a_new = a_new / s
    return a_new


def lp_refine(a: np.ndarray, deadline: float,
              max_iter: int = 15, t0: float = 0.30,
              decay: float = 0.85) -> np.ndarray:
    """
    Repeated conditional‑gradient (LP) refinement.
    The step‑size t is geometrically decayed.
    """
    a_cur = a.copy()
    t = t0
    for it in range(max_iter):
        if time.time() >= deadline - 0.1:
            break
        conv = convolution_fft(a_cur, a_cur)
        M = float(np.max(conv))
        a_cur = _lp_refine_step(a_cur, M, t)
        t *= decay
        if (it + 1) % 5 == 0:
            print(f"    LP refine it={it+1:,d}   surrogate≈{fast_score(a_cur):.6f}")
    return a_cur


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent for
    f(a)=2·n·max(conv(a,a))/sum(a)².
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max‑convolution
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 30_000 == 0:
            print(f"    Sub‑grad it={it:,d}   surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Simple random pairwise mass‑transfer local search
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.025,
) -> np.ndarray:
    """
    Pairwise moves: move a random fraction (≤max_transfer) of the mass
    of a randomly chosen index i to another index j.
    Keep the move only if the cheap surrogate improves.
    """
    a = a0.copy()
    best = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta
        # re‑normalise (just in case of rounding error)
        a /= a.sum()
        sc = fast_score(a)
        if sc < best:
            best = sc
        else:
            # revert
            a[j] -= delta
            a[i] += delta
            a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence``.  The routine mixes analytic seeds,
    a short smooth‑max Adam run, LP‑based conditional‑gradient
    refinement, a final exact sub‑gradient polish and a random
    local search.  All work respects the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.4          # safety margin

    best_seq: list[float] = []
    best_official: float = float("inf")
    best_surrogate: float = float("inf")

    # --------------------------------------------------------------
    # Optional fallback to a previously discovered construction
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, cand
                best_surrogate = fast_score(np.array(cand))
        except Exception:
            pass

    # --------------------------------------------------------------
    # Lengths to explore (larger n gives a smaller theoretical constant)
    # --------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} (time left {deadline - time.time():.1f}s) ===")

        # 1. Build a diverse pool of seed vectors
        seeds: list[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),                      # uniform
            generate_mixture_shape(n, weight_arc=0.6),  # arcsine‑uniform mix
        ]

        # beta(α,α) symmetric sweep – fine grid
        for alpha in np.linspace(0.35, 2.0, 18):
            seeds.append(generate_beta_shape(n, alpha, alpha))

        # power‑law sweep
        for p in np.linspace(0.25, 0.90, 14):
            seeds.append(generate_power_shape(n, p))

        # Random Dirichlet draws
        for _ in range(20):
            seeds.append(generate_random_dirichlet(n, rng))

        # ----------------------------------------------------------
        # 2. Cheap surrogate ranking, keep a handful of promising seeds
        # ----------------------------------------------------------
        scored = [(fast_score(v), v) for v in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]

        # ----------------------------------------------------------
        # 3. Allocate a slice of the remaining budget per seed
        # ----------------------------------------------------------
        per_seed_budget = max(1.5, (deadline - time.time()) / (len(top_seeds) * 2.0))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 1.5:
                break
            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_budget:.1f}s")
            seed_deadline = time.time() + per_seed_budget

            # Normalise to a probability vector (sum = 1)
            a = seed_vec.copy()
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0 or not np.isfinite(a_sum):
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum

            # ---- (a) short smooth‑max Adam optimisation -----------------
            # light‑weight run – 10 k iterations are usually enough for a
            # decent local improvement while staying fast.
            z0 = np.log(a + 1e-12)               # soft‑max initialise
            a = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=2500.0,
                ramp_iters=80_000,
                lr=0.03,
            )

            # ---- (b) conditional‑gradient (LP) refinement -------------
            # Use LP only for moderate lengths (≤1024) – larger n would
            # cause a huge matrix.  For n>1024 the LP step is simply skipped.
            if n <= 1024 and _cvxpy_ok:
                a = lp_refine(a, seed_deadline, max_iter=12,
                              t0=0.30, decay=0.85)

            # ---- (c) official evaluation -------------------------------
            try:
                val = evaluate_sequence(a.tolist())
                print(f"   Official score after refinement = {val:.6f}")
                if val < best_official:
                    best_official, best_seq = val, a.tolist()
                    best_surrogate = fast_score(a)
                    print("   *** New global best! ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_official:
                best_official, best_seq = final_val, a_arr.tolist()
                best_surrogate = fast_score(a_arr)
                print("*** Global best improved by final polish ***")

            print(f"   Final evaluation error: {e}")

    # 5. Random local search around the polished solution
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:

        a_start = np.array(best_seq, dtype=float)
        a_refined = random_local_search(
            a_start,

            rng,
            max_transfer=0.025,
        )

            val_loc = evaluate_sequence(a_refined.tolist())

            if val_loc < best_official:
                best_official, best_seq = val_loc, a_refined.tolist()
                best_surrogate = fast_score(a_refined)
                print("*** Best improved by local search ***")
        except Exception as e:
            print(f"   Local‑search evaluation error: {e}")

    # --------------------------------------------------------------
    # 6. Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:

        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
