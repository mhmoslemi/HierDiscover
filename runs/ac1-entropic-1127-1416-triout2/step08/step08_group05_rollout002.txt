"""
Propose a non‑negative step sequence that (empirically) minimises
``evaluate_sequence``.

Strategy
--------
1.  For a set of lengths (8192 … 512) generate a large, diverse
    collection of analytic “shapes” (beta‑like, power‑law, mixtures,
    uniform, arcsine) and a few random perturbations of the current
    best.  The fast surrogate ``compute_score`` (2·n·max(conv) for a
    unit‑sum vector) ranks the candidates.

2.  Keep the top few candidates for each length and improve them with
    a smooth‑max Adam optimiser.  The smooth max is realised with a
    log‑sum‑exp surrogate; a temperature β is increased linearly from
    a modest start to a large final value (≈2000) so that the gradient
    focuses on the true maximum.

3.  After each Adam run, the official ``evaluate_sequence`` is called.
    The best vector seen so far is stored.

4.  Whenever time permits a short “exponentiated‑gradient” (KL‑mirror)
    polishing step is applied to the current global best.  This step
    works on the exact max‑convolution objective and keeps the iterate
    on the simplex automatically.

5.  When the wall‑clock budget is almost exhausted a final Euclidean
    projected sub‑gradient descent (exact max) is performed.

The routine respects the 1000 s wall‑clock limit, always returns a
valid (non‑empty, non‑negative) sequence and prints progress information.
"""

import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score when ``sum(a)==1``.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple feasibility utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger size."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    ups = np.interp(x_new, x_old, seq)
    ups = np.clip(ups, 1e-16, None)
    ups /= ups.sum()
    return ups


# ----------------------------------------------------------------------
# Optimisers (smooth‑max Adam, KL‑mirror EG, exact sub‑gradient)
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 500.0,
    ramp_iters: int = 60000,
    lr: float = 0.08,
) -> np.ndarray:
    """
    Adam on the smooth log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    z0 : ndarray
        Initial log‑probabilities (log‑softmax coordinates).
    deadline : float
        Absolute time at which the optimiser must stop.
    n : int
        Length of the underlying sequence.
    beta0, beta_max : float
        Temperature schedule for the smooth max.
    ramp_iters : int
        Number of iterations over which β is increased linearly.
    lr : float
        Base learning rate for Adam.

    Returns
    -------
    a : ndarray
        Optimised probability vector (∑a_i=1, a_i≥0).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # point on the simplex
        conv = convolution_fft(a, a)       # linear convolution, length 2n‑1
        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights (log‑sum‑exp trick)
        shifted = conv - mx                # ≤0
        w = np.exp(beta * shifted)         # may under‑flow for far‑away entries
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient ∇_a  (2·(a ⋆ w_rev))
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length 2n‑1
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central n entries

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

    # final simplex point
    a = softmax(z)
    return a


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    step: float = 0.018,
) -> np.ndarray:
    """
    KL‑mirror descent on the exact max‑convolution objective.
    The multiplicative update keeps the iterate on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max(conv) = 2·(a ⋆ mask_rev)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return a


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.08,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex via Euclidean projection.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a

def propose_candidate(seed: int = 43, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine respects the 1000 s wall‑clock budget and always
    returns a valid (non‑empty) sequence.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate on the official scorer (fast path)
    # ------------------------------------------------------------------
    def official_score(vec: np.ndarray) -> float:
        try:
            return float(evaluate_sequence(vec.tolist()))
        except Exception:
            return float("inf")

    # ------------------------------------------------------------------
    # 0.  If a historic solution is available, try it first
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = official_score(np.array(cand, dtype=float))
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Explore a cascade of increasing lengths
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024, 512]

    for n in lengths:
        if time.time() >= deadline - 15.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # Build a diverse pool of analytic seeds
        # --------------------------------------------------------------
        pool: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),                 # uniform
        ]

        # random beta‑shapes
        for _ in range(300):
            a = rng.uniform(0.02, 2.0)          # allow very light tails
            b = rng.uniform(0.02, 2.0)
            eps = rng.uniform(0.0, 1e-3)
            pool.append(generate_beta_shape(n, a, b, eps))

        # random power‑law shapes
        for _ in range(300):
            p = rng.uniform(0.2, 2.5)
            eps = rng.uniform(0.0, 1e-2)
            pool.append(generate_power_shape(n, p, eps))

        # mixtures of arcsine & uniform
        for _ in range(100):
            w = rng.uniform(0.0, 1.0)
            pool.append(generate_mixture_shape(n, w))

        # perturb the current best (if it has the same length)
        if best_seq and len(best_seq) == n:
            base = np.array(best_seq, dtype=float)
            for _ in range(150):
                noise = rng.normal(scale=0.01, size=n)
                pert = base + noise
                pert = np.clip(pert, 0.0, None)
                if pert.sum() == 0.0:
                    continue
                pert /= pert.sum()
                pool.append(pert)

        # up‑sample a shorter best if any
        if best_seq and len(best_seq) != n:
            pool.append(
                upsample_sequence(np.array(best_seq, dtype=float), n)
            )

        # --------------------------------------------------------------
        # Fast ranking of all candidates
        # --------------------------------------------------------------
        scored = [(compute_score(vec), vec) for vec in pool]
        scored.sort(key=lambda x: x[0])
        print(f"Best surrogate score for length {n}: {scored[0][0]:.6f}")

        # keep a handful of the most promising seeds
        top_seeds = [vec for _, vec in scored[:12]]

        # --------------------------------------------------------------
        # 2.  Refine each top seed with a longer Adam run
        # --------------------------------------------------------------
        # allocate roughly equal time to each seed, but guarantee a
        # minimum of a few seconds per seed.
        remaining = deadline - time.time()
        per_seed = max(4.0, remaining / (len(top_seeds) * 2.5))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 10.0:
                break
            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget≈{per_seed:.1f}s")
            seed_deadline = min(time.time() + per_seed, deadline - 5.0)

            # initialise Adam on log‑softmax coordinates
            eps0 = 1e-12
            z0 = np.log(seed_vec + eps0)

            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=2.0,
                beta_max=2000.0,      # high final temperature
                ramp_iters=200_000,
                lr=0.10,
            )

            # surrogate score after Adam
            sur_score = compute_score(a_opt)
            print(f"   Surrogate score after Adam: {sur_score:.6f}")

            # official evaluation
            val = official_score(a_opt)
            print(f"   Official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                print(f"   *** New global best: {best_val:.6f} ***")

            # ---------------------------------------------------------
            # Optional KL‑mirror polishing (if time permits)
            # ---------------------------------------------------------
            if time.time() < deadline - 6.0:
                polish_deadline = deadline - 2.5
                a_pol = exponentiated_gradient_polish(
                    np.array(best_seq, dtype=float),
                    polish_deadline,
                    max_iter=120_000,
                    step=0.019,
                )
                val2 = official_score(a_pol)
                print(f"   Polished (EG) score = {val2:.6f}")
                if val2 < best_val:
                    best_val, best_seq = val2, a_pol.tolist()
                    print(f"   *** New best after EG polish: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3.  Final exact sub‑gradient polishing (if any time remains)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\nFinal exact sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=300_000,
            step0=0.09,
        )
        final_val = official_score(a_arr)
        print(f"   Final official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a_arr.tolist()
            print("*** Global best improved by final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = official_score(np.array(best_seq, dtype=float))
        print(f"Fallback uniform sequence, score = {best_val:.6f}")

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
