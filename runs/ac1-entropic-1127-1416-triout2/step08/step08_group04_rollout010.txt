        b_opt = res.x
        sum_b = b_opt.sum()

"""
Propose a non‑negative step‑function sequence with a low official score.

Algorithm
---------
1. **Coarse shape search** – generate several elementary shapes:
   * discrete arcsine density with a tunable smoothing parameter ε,
   * generalized power‑law shapes a_i ∝ ((i+½)(n‑½‑i)+ε)^{-p},
   * simple mixtures of arcsine and uniform.
   All candidates are normalised to sum = 1 and evaluated with a fast
   FFT based surrogate `2·n·max(conv(a,a))`.  The best candidate is kept.

2. **Linear‑programming refinement** – given the current probability vector
   `f`, compute `M = max(conv(f,f))`.  Solve the LP
       maximise   Σ b_j
       subject to conv(f,b) ≤ M  (componentwise),   b ≥ 0 .
   If the LP yields a vector `b` whose total mass exceeds that of `f`,
   form the normalised vector `g = b / Σb` and replace `f` by a convex
   combination `(1‑t)·f + t·g`.  The mixing weight `t` is selected from a
   small grid (0,0.1,…,1) to minimise the new maximum convolution.
   This step is repeated while time permits; each LP uses a sparse
   Toeplitz‑like matrix and SciPy's high‑performance “highs” solver.

3. **Polishing** – a few hundred thousand iterations of Euclidean
   projected sub‑gradient descent, exponentiated‑gradient (mirror) descent
   and a cheap stochastic hill‑climber are applied, all respecting the
   simplex constraint `Σa = 1`.

The routine respects a global 1000 s wall‑clock budget, prints intermediate
scores and finally returns the best sequence found (as a list of floats).
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1} (Duchi et al. 2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    Assumes Σa = 1, returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used for seeding the optimiser)
# ----------------------------------------------------------------------


def arcsine_initial(n: int, eps: float = 0.25) -> np.ndarray:
    """Discrete arcsine density with smoothing ε, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """
    Generalised arcsine shape:
        a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}.
    Returns a normalised probability vector.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float, eps: float = 0.25) -> np.ndarray:
    """
    Mixture weight_arc·arcsine + (1‑weight_arc)·uniform,
    normalised to sum = 1.
    """
    arc = arcsine_initial(n, eps=eps)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Convolution matrix for the LP (Toeplitz‑like, sparse)
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray):
    """
    Build the (2n‑1)×n matrix A such that A @ b = conv(a,b)
    (linear convolution with zero padding).  Implemented as a sparse
    diagonal matrix with constant diagonals a_i.
    """
    n = a.shape[0]
    L = 2 * n - 1
    offsets = np.arange(n)                     # 0 … n‑1
    data = [np.full(n, ai, dtype=np.float64) for ai in a]
    A = diags(data, offsets, shape=(L, n), format='csr')
    return A


# ----------------------------------------------------------------------
# Optimisers for polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.07) -> np.ndarray:
    """Euclidean projected sub‑gradient descent on the true max‑convolution."""
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, reversed mask)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.25) -> np.ndarray:
    """Exponentiated‑gradient polish (mirror descent w.r.t. KL)."""
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 30_000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Very cheap stochastic hill climbing:
    move a tiny amount of mass from one coordinate to another.
    Accept only improving moves (according to the surrogate score).
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.2 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1

    return a_cur

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    Returns the best sequence (list of floats) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: list[float] = []
    best_surrogate: float = float("inf")
    best_official: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"[fallback] official score = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = fallback
                best_surrogate = compute_score(np.array(fallback))
        except Exception as e:
            print(f"[fallback] evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 1. Coarse shape search (arcsine, power‑law, mixtures)
    # ------------------------------------------------------------------
    print("\n--- Stage 1: coarse shape search ---")
    n = 2048                     # fixed length for the LP stage
    eps_grid = [1e-6, 1e-4, 1e-2, 0.1, 0.25, 0.5, 1.0]
    p_grid = np.linspace(0.15, 0.85, 15)
    w_grid = np.linspace(0.0, 1.0, 11)

    # Uniform baseline
    uni = np.full(n, 1.0 / n)
    s = compute_score(uni)
    if s < best_surrogate:
        best_surrogate, best_seq = s, uni.tolist()
        print(f"[baseline] uniform surrogate={s:.6f}")

    for eps in eps_grid:
        # pure arcsine with this eps
        arc = arcsine_initial(n, eps=eps)
        s = compute_score(arc)
        if s < best_surrogate:
            best_surrogate, best_seq = s, arc.tolist()
            print(f"[arc] eps={eps:.1e} surrogate={s:.6f}")

        # power‑law family
        for p in p_grid:
            shape = generate_power_shape(n, p, eps=eps)
            s = compute_score(shape)
            if s < best_surrogate:
                best_surrogate, best_seq = s, shape.tolist()
                print(f"[pow] eps={eps:.1e}, p={p:.3f} surrogate={s:.6f}")

        # mixtures with the current arcsine
        arc = arcsine_initial(n, eps=eps)    # recompute (cheap)
        for w in w_grid:
            mix = generate_mixture_shape(n, w, eps=eps)
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"[mix] eps={eps:.1e}, w={w:.2f} surrogate={s:.6f}")

    # ------------------------------------------------------------------
    # 2. Linear‑programming refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print("\n--- Stage 2: LP refinement ---")
        a = np.array(best_seq, dtype=float)          # already normalised to sum=1
        max_lp_iters = 30
        for lp_it in range(max_lp_iters):
            if time.time() >= deadline - 10.0:
                break

            M = np.max(convolution_fft(a, a))
            # Build the constraint matrix once per iteration
            A = build_conv_matrix(a)                  # shape (2n‑1, n)

            n_vars = a.shape[0]
            c = -np.ones(n_vars)                     # maximise Σb  ⇔ minimise −Σb
            b_ub = np.full(A.shape[0], M, dtype=np.float64)
            bounds = [(0, None)] * n_vars

            try:
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True})
            except Exception as e:
                print(f"[LP] failure at iter {lp_it}: {e}")
                break

            if not res.success:
                print(f"[LP] not successful (iter {lp_it}): {res.message}")
                break

            b_opt = res.x
            sum_b = b_opt.sum()

            # no improvement possible?
            if sum_b <= 1.000001:
                print(f"[LP] iteration {lp_it}: LP returned sum≈1 ({sum_b:.6f}) – stop")
                break

            g = b_opt / sum_b                         # normalised candidate

            # simple grid search for the best mixing weight t ∈ [0,1]
            t_grid = np.linspace(0.0, 1.0, 11)
            best_t = 0.0
            best_Mt = M
            for t in t_grid:
                a_new = (1.0 - t) * a + t * g
                Mt = np.max(convolution_fft(a_new, a_new))
                if Mt < best_Mt - 1e-14:
                    best_Mt = Mt
                    best_t = t

            if best_t == 0.0:
                print(f"[LP] iteration {lp_it}: no improving t found")
                break

            a = (1.0 - best_t) * a + best_t * g
            cur_surrogate = compute_score(a)
            print(f"[LP] iter {lp_it:2d}: M improved to {cur_surrogate:.6f} (t={best_t:.2f})")
            if cur_surrogate < best_surrogate:
                best_surrogate = cur_surrogate
                best_seq = a.tolist()
    else:
        print("\nSkipping LP refinement (insufficient time)")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient descent polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        print("\n--- Stage 3: sub‑gradient descent polish ---")
        a = np.array(best_seq, dtype=float)
        a = subgradient_descent_exact(a, deadline, max_iter=250_000, step0=0.07)
        cur = compute_score(a)
        print(f"[sub‑grad] surrogate={cur:.6f}")
        if cur < best_surrogate:
            best_surrogate = cur
            best_seq = a.tolist()
    else:
        print("\nSkipping sub‑gradient descent (insufficient time)")

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Stage 4: exponentiated‑gradient polish ---")
        a = np.array(best_seq, dtype=float)
        a = eg_refine(a, deadline, max_iter=150_000, eta0=0.25)
        cur = compute_score(a)
        print(f"[EG] surrogate={cur:.6f}")
        if cur < best_surrogate:
            best_surrogate = cur
            best_seq = a.tolist()
    else:
        print("\nSkipping EG polish (insufficient time)")

    # ------------------------------------------------------------------
    # 5. Cheap stochastic local search
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Stage 5: stochastic local search ---")
        a = np.array(best_seq, dtype=float)
        a = local_random_search(a, deadline, max_trials=50000,
                                step_scale=0.001, rng=rng)
        cur = compute_score(a)
        print(f"[local] surrogate={cur:.6f}")
        if cur < best_surrogate:
            best_surrogate = cur
            best_seq = a.tolist()
    else:
        print("\nSkipping local search (insufficient time)")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_official:.6f}")
        if final_official < best_official:
            best_official = final_official
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_official = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n_fallback = 1024
        best_seq = [1.0 / n_fallback] * n_fallback
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {n_fallback}, official = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
