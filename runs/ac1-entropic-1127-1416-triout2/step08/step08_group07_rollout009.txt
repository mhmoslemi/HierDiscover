"""
A compact but aggressive search for a non‑negative step sequence
that minimises ``evaluate_sequence``.

The method concentrates on a single, reasonably large length (default
2048) and performs:

1.  A rich, deterministic seed pool (uniform, arcsine, power‑law,
    symmetric / asymmetric Beta, mixtures and random Dirichlet draws).
2.  Cheap pre‑selection with the scale–invariant surrogate
    ``fast_score``.
3.  Continuation Adam optimisation of the smooth‐max surrogate
    ``log‑sum‑exp(β·conv)`` with a very large final β (≈5·10⁴) so that
    the optimiser is driven almost directly by the true ℓ∞‑norm.
4.  Exact projected sub‑gradient descent on the true objective for a
    short polishing phase.
5.  A cheap random pairwise‑mass‑transfer local search to squeeze the
    final few tenths of a percent.

All work respects the overall time budget (``budget_s`` seconds).  The
function returns a valid list of non‑negative floats; if nothing is
found a uniform fallback is returned.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate (scale‑invariant) – used for cheap ranking
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Return 2·n·max(conv(a,a))/sum(a)²  (no clipping, no caps).  ``a``
    must be non‑negative.
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 0.0) -> np.ndarray:
    """Beta(α,β) shape on {0,…,n‑1} (unnormalised, then normalised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Shape proportional to 1/((i+½)*(n‑½‑i)+eps)^p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet point (concentration between 0.1 and 5)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)      # 0.01 … 5.0
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) optimiser – continuation Adam with softmax
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 50000.0,
    ramp_iters: int = 300_000,
    lr: float = 0.02,
) -> np.ndarray:
    """
    Minimise log‑sum‑exp(β·conv(a,a)) with a = softmax(z).
    ``beta`` is increased linearly from ``beta0`` to ``beta_max``.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                 # a ≥ 0, Σ a = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max (softmax) weights
        shifted = conv - mx
        w = np.exp(beta * shifted)          # ≤ 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of log‑sum‑exp w.r.t. a:
        #   d/da Σ w_i·(a★e_i) = 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)          # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]     # slice length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 8000 == 0:
            # occasional progress report
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.15,
) -> np.ndarray:
    """
    Projected sub‑gradient descent for
        f(a) = 2·n·max(conv(a,a))/sum(a)²
    (with sum(a)=1 enforced by the simplex projection).
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution (see notes)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a -= eta * subgrad
        a = _project_to_simplex(a)

        if it % 20_000 == 0:
            print(f"    Sub‑grad it={it:,d} surrogate≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Simple random local search – pairwise mass transfers
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.02,
) -> np.ndarray:
    """
    inexpensive pairwise mass‑transfer moves while time permits.
    Moves are kept only if they improve the cheap ``fast_score``.
    """
    a = a0.copy()
    best_score = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * rng.random() * max_transfer
        if eps <= 0.0:
            continue
        a[i] -= eps
        a[j] += eps
        # renormalise (numerical stability)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
            continue
        a /= s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            a[j] -= eps
            a[i] += eps
            a /= a.sum()
    return a

            if val < best_official:
                best_official, best_seq = val, a_opt.tolist()
                best_surrogate = sur
                print("   *** New global best! ***")
        except Exception as e:
            print(f"   Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 5. Final polishing with exact sub‑gradient descent (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nPolishing with projected sub‑gradient descent …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=400_000,
            step0=0.2,
        )
        try:
            val = evaluate_sequence(a_arr.tolist())
            sur = fast_score(a_arr)
            print(f"   After polishing: official {val:.6f} (surrogate {sur:.6f})")
            if val < best_official:
                best_official, best_seq = val, a_arr.tolist()
                best_surrogate = sur
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Polishing evaluation error: {e}")

    # ------------------------------------------------------------------
    # 6. Random local search around the best solution (tiny remaining budget)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:
        print("\nRandom local search …")
        a_start = np.array(best_seq, dtype=float)
        a_refined = random_local_search(
            a_start,
            deadline,
            rng,
            max_transfer=0.02,
        )
        try:
            val = evaluate_sequence(a_refined.tolist())
            sur = fast_score(a_refined)
            print(f"   Local‑search official {val:.6f} (surrogate {sur:.6f})")
            if val < best_official:
                best_official, best_seq = val, a_refined.tolist()
                best_surrogate = sur
                print("*** Best improved by local search ***")
        except Exception as e:
            print(f"   Local‑search evaluation error: {e}")

    # ------------------------------------------------------------------
    # 7. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
