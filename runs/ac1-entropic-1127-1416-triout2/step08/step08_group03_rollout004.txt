"""
Propose a high‑quality step‑function by combining three ideas:

1. **Broad analytic search** – symmetric product, left‑heavy power‑law and a
   Gaussian family are scanned on the full length (8192).  The best vector
   serves as a warm start.

2. **Linear‑programming refinement** – the current vector *f* (down‑sampled
   to 2048 entries) is improved by solving, repeatedly,
   a small LP that maximises the ℓ₁‑norm of a new vector *b* under the
   constraint  (f * b) ≤ max(f * f).  Mixing *f* and the normalised solution
   reduces the maximal self‑convolution while keeping the sum fixed.

3. **Final local search** – the refined shape is up‑sampled back to 8192,
   projected to the simplex and tuned with a cheap projected sub‑gradient
   walk followed by a short simulated‑annealing hill climb.

All stages respect the global wall‑clock limit (default 1000 s).  The
returned object is a plain Python list ready for `evaluate_sequence`.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (large length)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """Discretised Gaussian centred in the middle, projected to the simplex."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random starters (small length)
# ----------------------------------------------------------------------


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector and normalise."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    return a / a.sum()


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Down‑ / up‑sampling helpers
# ----------------------------------------------------------------------


def downsample(arr: np.ndarray, factor: int) -> np.ndarray:
    """Mean‑aggregate by integer factor."""
    n = len(arr)
    new_n = n // factor
    trimmed = arr[:new_n * factor]
    return trimmed.reshape(new_n, factor).mean(axis=1)


def upsample(arr: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation to a larger length."""
    n = len(arr)
    x_src = np.linspace(0.0, 1.0, n)
    x_tgt = np.linspace(0.0, 1.0, target_len)
    return np.interp(x_tgt, x_src, arr)


# ----------------------------------------------------------------------
# Linear‑programming refinement (small dimension)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n convolution matrix A where (A @ b)[k] = (f*b)[k].
    Each column i contains the vector f shifted down by i places.
    """
    n = f.shape[0]
    # each diagonal i (offset i) consists of n copies of f[i]
    diags = [np.full(n, fi) for fi in f]
    offsets = np.arange(n)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Maximise Σ b_j subject to b_j≥0 and (f*b)_k ≤ M for all k.
    Returns the optimal vector b (may be None on failure).
    """
    n = f.shape[0]
    A_ub = build_conv_matrix(f)                     # (2n‑1)×n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # objective: maximise sum(b) → minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)

    # Highs is fast and accepts sparse matrices
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method="highs")
    if res.success:
        return res.x
    return None


def lp_refine(f: np.ndarray, deadline: float, max_iters: int = 12) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP above and mix the solution into f.
    Returns the best vector seen and its exact score.
    """
    best_f = f.copy()
    best_score = compute_score(best_f)
    n = f.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(f, f)
        M = conv.max()
        b = solve_lp_max_sum(f, M)
        if b is None:
            print(f"[{time.time():.1f}] LP failed at iteration {it}")
            break

        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        g = b / sum_b                         # normalise to simplex
        # convex combination (t = 0.5) – keeps sum = 1 and stays non‑negative
        f = 0.5 * f + 0.5 * g
        f = project_to_simplex(f)           # safety

        sc = compute_score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
            print(f"[{time.time():.1f}] LP iter {it}: improved score = {best_score:.6f}")

    return best_f, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (large dimension, cheap)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 400_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient method on the true max‑objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] subgrad iter {t}: new best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (large dimension)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 600_000) -> tuple[np.ndarray, float]:
    """Simple SA walk on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    # small temperature – only occasional uphill steps
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                      # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Global parameters
    # ------------------------------------------------------------------
    n_large = 8192                 # final length
    n_small = 2048                 # dimension for LP refinement
    factor = n_large // n_small    # 4

    # ------------------------------------------------------------------
    # 0. fallback (if a previous solution is present)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families on the full length – keep the best
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] scanning analytic families (n={n_large})")
    alphas = np.linspace(0.0, 4.0, 81)
    for a in alphas:
        if time.time() >= deadline - 20.0:
            break
        vec = symmetric_product_family(n_large, a)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={a:.2f}) = {best_score:.6f}")

    betas = np.linspace(0.0, 4.0, 81)
    for b in betas:
        if time.time() >= deadline - 20.0:
            break
        vec = left_heavy_power_family(n_large, b)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={b:.2f}) = {best_score:.6f}")

    # Gaussian sweep (new family)
    sigmas = np.linspace(0.05 * n_large, 0.30 * n_large, 30)
    for sigma in sigmas:
        if time.time() >= deadline - 20.0:
            break
        vec = gaussian_family(n_large, sigma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (gauss‑σ={sigma:.0f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random monotone starts on the large length (few draws)
    # ------------------------------------------------------------------
    for _ in range(30):
        if time.time() >= deadline - 20.0:
            break
        vec = random_monotone_start(n_large)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Prepare a small‑dimensional working vector for LP refinement
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety fallback – uniform tiny vector
        best_seq = [1.0 / n_large] * n_large

    a_large = np.array(best_seq, dtype=np.float64)
    a_large = project_to_simplex(a_large)

    # Down‑sample to the LP dimension
    a_small = downsample(a_large, factor)
    a_small = project_to_simplex(a_small)

    # ------------------------------------------------------------------
    # 4. Linear‑programming refinement on the small dimension
    # ------------------------------------------------------------------
    lp_deadline = deadline - 80.0
    if time.time() < lp_deadline:
        print(f"[{time.time():.1f}] starting LP refinement (n={n_small})")
        a_small_ref, sc_small = lp_refine(a_small, lp_deadline, max_iters=12)
        print(f"[{time.time():.1f}] LP refinement best small‑score = {sc_small:.6f}")
        # Keep the refined small vector if it improved the large‑scale score
        a_up = upsample(a_small_ref, n_large)
        a_up = project_to_simplex(a_up)
        sc_up = compute_score(a_up)
        if sc_up < best_score:
            best_score = sc_up
            best_seq = a_up.tolist()
            print(f"[{time.time():.1f}] after upsample improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Long sub‑gradient descent on the full size (cheap per iteration)
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 30.0
    if best_seq is not None and time.time() < subgrad_deadline:
        print(f"[{time.time():.1f}] starting long sub‑gradient refinement (n={n_large})")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            subgrad_deadline,
            max_iters=400_000,
            step0=0.06,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb (short budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=600_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Symmetrisation test (many optimum vectors are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
