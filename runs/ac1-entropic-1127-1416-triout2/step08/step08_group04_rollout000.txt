"""
Propose a low‑score step‑function.

The algorithm is deliberately simple and fast:
1.  Try a set of lengths (up to 32768).  For each length evaluate
    a family of probability densities on {0,…,n‑1}:
       – uniform,
       – the discrete arcsine law (β(½,½)),
       – a parametric beta family a_i ∝ (i+½)^{α‑1}(n‑½‑i)^{α‑1}
         (implemented as a power‐shape with exponent p = 1‑α),
       – a mixture of the arcsine law with the uniform law.
    The surrogate objective 2·n·max(conv(a,a)) is computed with an
    FFT‑based convolution, which is exact when the vector sums to 1.
    The best shape (lowest surrogate) is remembered.
2.  With the surviving candidate, perform a cheap stochastic hill‑climb:
    repeatedly move a tiny amount of mass from one coordinate to another
    and keep the move if it lowers the surrogate.  The move respects the
    probability‑simplex (projection onto the simplex after each step).
3.  Finally call the official ``evaluate_sequence`` on the best sequence
    found and return it.  All work respects the overall wall‑clock budget
    (default 1000 s) and guarantees a non‑empty list.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex Σa = 1.
    The surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_shape(n: int, eps: float = 1e-12) -> np.ndarray:
    """
    Discrete arcsine law (Beta(½,½)) on n points,
    normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def power_shape(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """
    Generalised beta family a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p},
    normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def mixture_arc_uniform(n: int, w: float) -> np.ndarray:
    """
    Convex combination w·arcsine + (1-w)·uniform, normalised.
    """
    arc = arcsine_shape(n)
    uni = np.full(n, 1.0 / n)
    mix = w * arc + (1.0 - w) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Simple stochastic hill climbing
# ----------------------------------------------------------------------


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 5000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Very cheap stochastic hill climbing in the simplex.
    Move a tiny amount of mass from i to j and keep the move if it
    decreases the surrogate score.
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.1 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # safeguard against rounding errors
        if a_new[i] < 0.0:
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1
    return a_cur


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low official score.
    The routine explores several lengths, parameterised beta‑type shapes,
    refines the best candidate with a tiny stochastic hill‑climb and finally
    evaluates it with ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq: List[float] = []
    best_surrogate: float = float("inf")
    best_official: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = fallback
                best_surrogate = compute_score(np.array(fallback))
        except Exception as e:
            print(f"Fallback evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Stage 1 – systematic coarse search over shapes and lengths
    # ------------------------------------------------------------------
    candidate_ns = [2048, 4096, 8192, 16384, 32768]
    # finer grid for the beta‑exponent (p = 1‑α)
    p_grid = np.linspace(0.40, 0.60, 41)      # step 0.005
    # mixture weight grid (arcsine ↔ uniform)
    w_grid = np.linspace(0.0, 1.0, 11)       # step 0.1

    print("\n--- Stage 1: systematic shape search ---")
    for n in candidate_ns:
        print(f"\nTesting length n = {n}")

        # Uniform
        uni = np.full(n, 1.0 / n)
        s = compute_score(uni)
        if s < best_surrogate:
            best_surrogate, best_seq = s, uni.tolist()
            print(f"  New best (uniform) surrogate = {s:.6f}")

        # Pure arcsine
        arc = arcsine_shape(n)
        s = compute_score(arc)
        if s < best_surrogate:
            best_surrogate, best_seq = s, arc.tolist()
            print(f"  New best (arcsine) surrogate = {s:.6f}")

        # Power‑shape (beta family) sweep
        for p in p_grid:
            shape = power_shape(n, p)
            s = compute_score(shape)
            if s < best_surrogate:
                best_surrogate, best_seq = s, shape.tolist()
                print(f"  New best (p={p:.3f}) surrogate = {s:.6f}")

        # Mixtures arcsine / uniform
        for w in w_grid:
            mix = mixture_arc_uniform(n, w)
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"  New best (mix w={w:.2f}) surrogate = {s:.6f}")

        # time guard
        if time.time() >= deadline - 5.0:
            print("Time running low – stopping Stage 1 early.")
            break

    # ------------------------------------------------------------------
    # Stage 2 – cheap stochastic local refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Stage 2: stochastic local refinement ---")
        a = np.array(best_seq, dtype=float)
        a = local_random_search(
            a,
            deadline,
            max_trials=20000,
            step_scale=0.001,
            rng=rng,
        )
        s = compute_score(a)
        print(f"  After local search surrogate = {s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping Stage 2 (insufficient time)")

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        final_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
