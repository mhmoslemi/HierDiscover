    start = time.time()
    deadline = start + budget_s - 0.2

    best_seq = None
    best_score = float('inf')  # official

    for a in alphas:
        for b in betas:
            seeds.append(generate_beta_shape(n, a, b, eps=1e-12))

    # Power shape
    for p in np.linspace(0.2, 2.0, 9):
        seeds.append(generate_power_shape(n, p, eps=1e-8))

    for _ in range(20):
        seeds.append(generate_random_dirichlet(n, rng))

    # Evaluate fast_score and keep top 5-8 seeds
    scores = [(fast_score(v), v) for v in seeds]
    scores.sort(key=lambda x: x[0])
    top = [v for (_, v) in scores[:7]]

    # Use each top seed with a longer Adam.
    remaining = deadline - time.time()

"""
Propose a high‑quality step‑function sequence.

The routine follows a “search → polish → refine” strategy.

1.  Choose a single large length (8192) – the optimal constant does not
    depend on n, but a finer discretisation reduces the discretisation
    error of the continuous optimal shape.

2.  Build a very diverse pool of analytic “seed” shapes:
        • arcsine (≈Beta(½,½))
        • uniform
        • Hann window
        • triangular (tent)
        • a grid of symmetric Beta(α,α) with α∈[0.2,5]
        • power‑law families 1/((i+½)(n‑½‑i))^p
        • discrete Gaussian families with several σ
        • random Dirichlet draws with random concentration

    All seeds are normalised to sum‑to‑one (the objective is scale‑invariant).

3.  Rank the seeds by the cheap surrogate ``fast_score`` (the true
    objective without the clipping performed by ``evaluate_sequence``)
    and keep the best few (5).

4.  Each kept seed is refined by a **continuation Adam** optimiser that
    minimises the smooth‑max surrogate
        Lβ(a)=log∑exp(β·conv(a,a)).
    β is linearly increased from 5 to 5000, the learning‑rate is a
    modest 0.04 and the optimisation runs until the per‑seed time budget
    expires.

5.  The best candidate obtained so far is polished with a **projected
    sub‑gradient descent** on the exact max‑convolution objective.
    The step size follows a 1/√t schedule and the iterate is repeatedly
    projected onto the probability simplex.

6.  A short **mirror‑descent** (exponential‑gradient) pass on the exact
    sub‑gradient further drives down the maximum convolution value.

7.  Finally a cheap **random local search** (pairwise mass transfers)
    explores the neighbourhood of the polished solution, accepting only
    moves that improve the cheap surrogate.

All phases respect the global time budget (default 1000 s).  The function
always returns a non‑negative list (the simplex constraint guarantees a
positive sum, so ``evaluate_sequence`` never returns ``inf``).

The algorithm differs from the previous baseline by:
    – richer initial seed families (Gaussian, triangular, Hann, finer
      Beta grid);
    – a higher β‑max and longer Adam schedule;
    – a more aggressive sub‑gradient polish;
    – allocation of the time budget to a few very promising seeds rather
      than many shallow runs.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate (scale invariant) – used for pre‑selection
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    2·n·max(conv(a,a))/sum(a)^2  (without clipping).
    Lower is better.
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic families of candidate shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (≈Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-12) -> np.ndarray:
    """Beta(α,β) density on {0,…,n‑1} (discretised, normalised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """Shape ∝ 1/((i+½)(n‑½‑i)+eps)^p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle, normalised."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    tot = vals.sum()
    if tot == 0.0 or not np.isfinite(tot):
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_hann_shape(n: int) -> np.ndarray:
    """Standard Hann (raised‑cosine) window, normalised."""
    i = np.arange(n, dtype=float)
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    return hann


def generate_triangular_shape(n: int) -> np.ndarray:
    """Symmetric tent (triangular) shape."""
    i = np.arange(n, dtype=float)
    vals = np.minimum(i + 1, n - i)
    vals /= vals.sum()
    return vals


def generate_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet point (concentration drawn uniformly in log‑space)."""
    conc = 10 ** rng.uniform(-2.0, 0.7)   # approx 0.01 … 5
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    total = a.sum()
    if total == 0.0 or not np.isfinite(total):
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx_i=1, x_i≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent for 2·n·max(conv)/sum^2."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of max‑convolution
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a -= eta * subgrad
        a = _project_to_simplex(a)

        if it % 10_000 == 0:
            print(f"    Sub‑grad it={it:,d} fast≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) with Adam optimisation (continuation on β)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 5000.0,
    ramp_iters: int = 200_000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    Minimise the smooth approximation Lβ(a)=log‑sum‑exp(β·conv(a,a))
    via Adam on the soft‑max parametrisation.
    β is increased linearly from ``beta0`` to ``beta_max``.
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # simplex point
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of log‑sum‑exp w.r.t. a
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 8_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.0f} fast≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) on the exact max‑convolution
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.2,
    decay: float = 0.99995,
) -> np.ndarray:
    """
    Multiplicative‑weights (KL‑mirror) descent using the exact
    sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        factor = np.exp(-step * subgrad)
        a *= factor
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay
        if int(step * 1e6) % 200_000 == 0:
            print(f"    Mirror step={step:.6f} fast≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Random local search (pairwise mass transfers)
# ----------------------------------------------------------------------


def random_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_transfer: float = 0.03,
) -> np.ndarray:
    """
    Simple stochastic hill‑climbing: move a random fraction of the mass
    from one index to another and keep the move if it improves the cheap
    surrogate ``fast_score``.
    """
    a = a0.copy()
    best = fast_score(a)
    n = a.size

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * rng.random() * max_transfer
        if eps <= 0.0:
            continue
        a[i] -= eps
        a[j] += eps
        # renormalise (numerical safety)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
            continue
        a /= s
        sc = fast_score(a)
        if sc < best:
            best = sc
        else:
            # revert move
            a[j] -= eps
            a[i] += eps
            a /= a.sum()
    return a

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence``.
    The algorithm builds a large pool of analytic seeds, keeps the most
    promising, refines each with Adam on a smooth‑max surrogate, then
    polishes the current best with exact sub‑gradient, mirror descent,
    and a final stochastic local search.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.3    # small safety margin

    best_seq: List[float] = []
    best_official: float = float("inf")
    best_surrogate: float = float("inf")

    # --------------------------------------------------------------
    # 1. Choose a (large) length – discretisation error matters most.
    # --------------------------------------------------------------
    n = 8192

    # --------------------------------------------------------------
    # 2. Generate a rich pool of analytic seeds.
    # --------------------------------------------------------------
    seeds: List[np.ndarray] = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),          # uniform
        generate_hann_shape(n),
        generate_triangular_shape(n),
    ]

    # symmetric Beta(α,α) grid
    alphas = np.concatenate([
        np.linspace(0.2, 0.8, 7),
        np.linspace(0.9, 2.5, 9),
        np.linspace(2.6, 5.0, 6),
    ])
    for a in alphas:
        seeds.append(generate_beta_shape(n, a, a))

    # power‑law family
    for p in np.linspace(0.2, 2.0, 9):
        seeds.append(generate_power_shape(n, p))

    # discrete Gaussians
    for sigma_factor in (0.12, 0.15, 0.20, 0.25):
        sigma = sigma_factor * n
        seeds.append(generate_gaussian_shape(n, sigma))

    # random Dirichlet draws
    for _ in range(20):
        seeds.append(generate_random_dirichlet(n, rng))

    # --------------------------------------------------------------
    # 3. Keep only the top few seeds according to the cheap surrogate.
    # --------------------------------------------------------------
    scored: List[Tuple[float, np.ndarray]] = [(fast_score(v), v) for v in seeds]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:5]]   # keep the 5 best

    print(f"\n=== Length {n}, {len(top_seeds)} top seeds selected ===")
    for i, sc in enumerate([s for s, _ in scored[:5]], 1):
        print(f"   Seed {i} cheap score = {sc:.6f}")

    # --------------------------------------------------------------
    # 4. Adam refinement (smooth‑max) for each top seed.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate a decent slice per seed (but never less than 4 s)
    per_seed = max(4.0, remaining / (len(top_seeds) * 1.2))

    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 2.0:
            break
        print(f"\n--- Adam on seed {idx}/{len(top_seeds)} (budget ≈{per_seed:.1f}s) ---")
        sub_deadline = time.time() + per_seed

        # initialise soft‑max parameters safely
        eps = 1e-12
        z0 = np.log(seed_vec + eps)

        a_opt = smooth_adam_optimize(
            z0,
            sub_deadline,
            n=n,
            beta0=5.0,
            beta_max=5000.0,
            ramp_iters=200_000,
            lr=0.04,
        )

        # official evaluation
        try:
            val = evaluate_sequence(a_opt.tolist())
            print(f"   Official score after Adam = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, a_opt.tolist()
                best_surrogate = fast_score(a_opt)
                print("   *** New global best! ***")
        except Exception as e:
            print(f"   Evaluation failed: {e}")

    # --------------------------------------------------------------
    # 5. Exact sub‑gradient polishing (long run).
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.5:
        print("\n--- Exact sub‑gradient polishing ---")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=400_000,
            step0=0.07,
        )
        try:
            val = evaluate_sequence(a_arr.tolist())
            print(f"   After polishing official = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, a_arr.tolist()
                best_surrogate = fast_score(a_arr)
                print("*** Updated best after exact polish ***")
        except Exception as e:
            print(f"   Polishing evaluation error: {e}")

    # --------------------------------------------------------------
    # 6. Mirror‑descent refinement.
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\n--- Mirror‑descent refinement ---")
        a_md = mirror_descent_optimize(
            np.array(best_seq, dtype=float),
            deadline,
            step0=0.2,
            decay=0.99995,
        )
        try:
            val = evaluate_sequence(a_md.tolist())
            print(f"   Mirror refinement official = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, a_md.tolist()
                best_surrogate = fast_score(a_md)
                print("*** Best improved by mirror descent ***")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 7. Final random local search.
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 0.5:
        print("\n--- Random local search ---")
        a_loc = random_local_search(
            np.array(best_seq, dtype=float),
            deadline,
            rng,
            max_transfer=0.02,
        )
        try:
            val = evaluate_sequence(a_loc.tolist())
            print(f"   Local‑search official = {val:.6f}")
            if val < best_official:
                best_official, best_seq = val, a_loc.tolist()
                best_surrogate = fast_score(a_loc)
                print("*** Best improved by local search ***")
        except Exception as e:
            print(f"   Local search evaluation error: {e}")

    # --------------------------------------------------------------
    # 8. Guard – guarantee a non‑empty return.
    # --------------------------------------------------------------
    if not best_seq:
        print("\nFallback to uniform sequence (should never happen).")
        n0 = 1024
        best_seq = [1.0 / n0] * n0
        best_official = evaluate_sequence(best_seq)
        print(f"   Fallback official score = {best_official:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
