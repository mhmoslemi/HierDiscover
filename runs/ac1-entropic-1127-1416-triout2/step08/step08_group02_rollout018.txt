            break

        conv = convolution_fft(a, a)

            best_a = cand
            step_scale = min(step_scale * 1.03, 0.5)
        else:
            step_scale *= 0.97
        steps += 1
    return best_a.tolist(), best_score

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

The optimiser works on the probability simplex (∑a_i = 1, a_i ≥ 0) – the
objective is scale‑invariant, so this normalisation is harmless.
It combines three complementary stages:

1. **Smooth surrogate optimisation** – minimise a log‑sum‑exp
   approximation of the max‑convolution using Adam.  The smoothing
   parameter β is annealed from a small value (≈5) to a large one
   (≈2000) so that the optimisation smoothly approaches the true
   max‑objective.

2. **Exact sub‑gradient descent** – a projected sub‑gradient method on
   the genuine max‑convolution objective refines the solution obtained
   by the smooth stage.

3. **Stochastic hill‑climb** – cheap mass‑transfer moves (swap or
   fractional shift) with simplex projection escape tiny local minima
   and squeeze out the last bits of improvement.

Multiple initialisations (central‑binomial, power‑law, random Dirichlet
and the optionally supplied ``height_sequence_1``) are tried; the best
result found before the time‑budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast for our sizes)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a vector that already
    satisfies ∑a_i = 1 (the scale is irrelevant because the objective is
    homogeneous)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)  # denominator = 1²


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initial‑vector factories
# ----------------------------------------------------------------------


def central_binomial(n: int) -> np.ndarray:
    """Truncated central‑binomial coefficients a_i = C(2i,i)/4^i,
    normalised onto the simplex."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2.0 * i + 1) / (2.0 * (i + 1))
    return project_to_simplex(a)


def power_law(n: int, exponent: float = 0.55) -> np.ndarray:
    """a_i = (i+1)^{-exponent}, simplex‑normalised."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-exponent)
    return project_to_simplex(a)


def random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Gamma(1,1) components)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return project_to_simplex(x)


def upscale_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length followed by simplex projection."""
    old_len = v.shape[0]
    if new_len == old_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    return project_to_simplex(v_new)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimiser – Adam
# ----------------------------------------------------------------------


def smooth_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = a0.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)          # first‑moment estimate
    v = np.zeros_like(a)          # second‑moment estimate
    best_a = a.copy()
    best_score = compute_objective(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β grows exponentially from beta_start to beta_end
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights (numerically stable)
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # gradient of the smoothed objective:
        #   ∂/∂a [ (1/β) log Σ exp(β·conv) ] = 2 * (a_rev * w)_{relevant slice}
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional progress check
        if it % 250 == 0:
            cur = compute_objective(a)
            if cur < best_score:
                best_score, best_a = cur, a.copy()

    # final check
    cur = compute_objective(a)
    if cur < best_score:
        best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (max‑convolution)
# ----------------------------------------------------------------------


def subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 250_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the genuine max‑convolution."""
    a = a0.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_objective(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (k)
        start_i = max(0, k - (n - 1))
        end_i = min(k, n - 1)
        g = np.zeros_like(a)
        if start_i <= end_i:
            idx = np.arange(start_i, end_i + 1)
            g[idx] = 2.0 * a[k - idx]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        cur = compute_objective(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()

        # tiny jitter every 1000 steps to avoid plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    cur = compute_objective(a)
    if cur < best_score:
        best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    deadline,
    max_steps: int = 400_000,
) -> tuple[list[float], float]:
    """Local stochastic moves (swap or fractional shift) with simplex
    projection, accepting only improvements."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_objective(best_a)

    step_scale = 0.10  # initial fraction of mass that may be moved

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur = compute_objective(cand)

        if cur < best_score:
            best_score, best_a = cur, cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain ``list[float]`` (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small margin for the harness that will call us
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose a working length (larger => potentially lower bound,
    # but also more expensive per iteration).
    # --------------------------------------------------------------
    remaining_before = deadline - time.time()
    # Rough rule: if we still have > 600 s we can afford n=8192,
    # otherwise we fall back to n=4096.
    n = 8192 if remaining_before > 600 else 4096
    print(f"[INFO] Working length set to n = {n}")

    # --------------------------------------------------------------
    # Build a palette of starting points
    # --------------------------------------------------------------
    init_vectors = []

    init_vectors.append(central_binomial(n))
    init_vectors.append(power_law(n, exponent=0.55))
    for _ in range(3):
        init_vectors.append(random_simplex(n))

    # optional fallback – a previously discovered construction
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback = np.asarray(fallback, dtype=float)
        if fallback.shape[0] != n:
            fallback = upscale_vector(fallback, n)
        else:
            fallback = project_to_simplex(fallback)
        init_vectors.append(fallback)
        print("[INFO] Added supplied fallback sequence.")
    except Exception:
        pass

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Iterate over the initial vectors
    # --------------------------------------------------------------
    for vidx, a0 in enumerate(init_vectors):
        if time.time() >= deadline:
            break

        a = project_to_simplex(a0.copy())
        cur_score = compute_objective(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[INIT {vidx}] baseline score = {cur_score:.6f}")

        # ---- Smooth Adam optimisation ---------------------------------
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        # devote roughly 55 % of the remaining time to the smooth stage
        smooth_deadline = time.time() + time_left * 0.55
        a_s, s_s = smooth_optimize(
            a,
            smooth_deadline,
            max_iters=500_000,
            beta_start=5.0,
            beta_end=2000.0,
            step0=0.9,
        )
        if s_s < best_score:
            best_score, best_seq = s_s, a_s.copy()
            print(f"[SMOOTH {vidx}] improved to {s_s:.6f}")

        # ---- Exact sub‑gradient refinement ----------------------------
        if time.time() < deadline - 2.0:
            sub_deadline = deadline - 2.0
            a_sub, s_sub = subgrad_optimize(best_seq, sub_deadline, max_iter=300_000)
            if s_sub < best_score:
                best_score, best_seq = s_sub, a_sub.copy()
                print(f"[SUBGRAD {vidx}] improved to {s_sub:.6f}")

        # ---- Final hill‑climb polishing -------------------------------
        if time.time() < deadline - 1.0:
            a_hc, s_hc = hill_climb(best_seq, deadline, max_steps=500_000)
            if s_hc < best_score:
                best_score, best_seq = s_hc, np.array(a_hc)
                print(f"[HILLCLIMB {vidx}] improved to {s_hc:.6f}")

        # ---- Light random perturbations for extra squeeze ------------
        # Each perturbation gets about 1 s; we perform as many as fit.
        while time.time() < deadline - 0.5:
            perturb = np.random.randn(n) * 0.001
            cand = best_seq + perturb
            cand = project_to_simplex(cand)
            pert_deadline = time.time() + 1.0
            a_pt, s_pt = smooth_optimize(
                cand,
                pert_deadline,
                max_iters=30_000,
                beta_start=10.0,
                beta_end=1500.0,
                step0=0.5,
            )
            if s_pt < best_score:
                best_score, best_seq = s_pt, a_pt.copy()
                print(f"[PERTURB] improved to {s_pt:.6f}")

    # --------------------------------------------------------------
    # Safety net – if nothing succeeded return a uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_objective(best_seq)
        print("[FALLBACK] No improvement – returning uniform vector.")

    # Final verification using the provided evaluate_sequence (it also
    # clips and rescales, but our vector already respects those limits).
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[FINAL] Length = {len(best_seq)}  score = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return best_seq.tolist()
