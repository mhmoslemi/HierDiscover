#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
search_step_function.py

A self‑contained optimisation routine for the “minimise max‑self‑convolution”
challenge used by the `evaluate_sequence` judge.

Main ideas
-----------
1. Choose a problem size (vector length) based on the remaining budget.
2. Initialise from a small family of analytically‑motivated shapes
   (α‑power law, sine‑power law and a random sparse start).
3. Refine the candidate with a fast Adam‑style mirror descent on the
   smooth log‑sum‑exp approximation of the max‑convolution.
   The temperature β is exponentially annealed while the Adam step‑size
   follows the usual 1/√t schedule.
4. Perform a short simulated‑annealing / hill‑climb phase that accepts
   occasional worsening moves to escape local minima.
5. Return the best sequence seen; the harness will call `evaluate_sequence`
   on the returned list to obtain the final score.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Core numerical utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using a real FFT (FFT length is the next power
    of two).  Returned array has length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a⊛a) for a vector normalized to sum = 1.
    This is the quantity returned by `evaluate_sequence` when the sum already
    equals 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for quick starts
# ----------------------------------------------------------------------


def alpha_start(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]⁻ᵅ   (symmetric power‑law family)."""
    i = np.arange(n, dtype=float)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow when alpha<0
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def sine_start(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π (i+1)/(n+1))ᵖ   (smooth bell‑shaped family)."""
    i = np.arange(1, n + 1, dtype=float)
    s = np.sin(np.pi * i / (n + 1))
    w = np.power(s, power, where=s > 0, out=np.zeros_like(s))
    a = w / w.sum()
    return a


def random_sparse_start(n: int) -> np.ndarray:
    """Uniform mass on a random subset (2‑%–40 % density)."""
    sparsity = np.random.uniform(0.02, 0.4)          # fraction of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a  # already on the simplex


# ----------------------------------------------------------------------
# Adam‑style optimisation of the smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def exp_adam_optimize(start_a: np.ndarray,
                      deadline: float,
                      max_iters: int = 200_000,
                      beta_start: float = 0.5,
                      beta_end: float = 400.0,
                      eta0: float = 0.6,
                      eps: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max of the convolution using an Adam optimiser on the
    log‑variables `v = log a`.  The temperature β is exponentially annealed.
    Returns the best exact score seen during the run.
    """
    n = start_a.shape[0]
    # initialise log‑variables safely
    v = np.log(np.maximum(start_a, eps))
    m = np.zeros_like(v)          # first moment
    v2 = np.zeros_like(v)         # second moment
    beta1, beta2 = 0.9, 0.999
    best_a = None
    best_score = float("inf")
    for it in range(1, max_iters + 1):
        now = time.time()
        if now >= deadline:
            break

        # ---- Normalise a from the current log‑variables ----
        v_shift = v - np.max(v)                 # avoid overflow
        a = np.exp(v_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ---- smoothed max (log‑sum‑exp) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        conv = convolution_fft(a, a)            # length 2n‑1
        maxc = conv.max()
        # stabilised soft‑max
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # ---- gradient of f_β w.r.t. a (vector of length n) ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)    # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ---- gradient w.r.t. log‑variables (chain rule) ----
        grad_v = a * grad_a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        eta = eta0 / np.sqrt(it)                # simple decay
        v -= eta * m_hat / (np.sqrt(v2_hat) + 1e-8)

        # ---- occasional exact evaluation (every 200 iters) ----
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climb polishing
# ----------------------------------------------------------------------


def simulated_annealing(start_a: np.ndarray,
                        deadline: float,
                        max_moves: int = 60_000) -> tuple[np.ndarray, float]:
    """
    A cheap stochastic local optimiser: propose a random transfer of mass
    between two entries (or a swap) and accept according to a temperature
    schedule that linearly cools to zero.  The candidate vector is always
    re‑projected onto the simplex.
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(best_a)

    # initial temperature – a fraction of the current excess over the theoretical bound 1.0
    T0 = max(0.01, (best_score - 1.0) * 0.3)
    move = 0
    while move < max_moves and time.time() < deadline:
        # linear cooling
        T = T0 * (1.0 - move / max_moves)

        cand = best_a.copy()
        if np.random.rand() < 0.45:                     # a swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                           # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # amount proportional to the source magnitude, perturbed a little
            delta = 0.05 * best_a[i] * (np.random.rand() - 0.5)
            delta = np.clip(delta, -best_a[i], best_a[j])  # keep non‑negative
            cand[i] -= delta
            cand[j] += delta

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        # Metropolis acceptance
        if sc < best_score or np.random.rand() < np.exp(-(sc - best_score) / max(T, 1e-12)):
            best_a = cand
            best_score = sc

        move += 1

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The routine proceeds through a series of stages, each respecting the
    global time budget:
      * quick analytic starts (α‑grid, sine‑power)
      * several Adam‑optimised refinements from different seeds
      * a final simulated‑annealing polish
    The best sequence seen is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin – we must return before the harness kills us
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a workable length (larger n → potentially lower bound)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 16384
    elif remaining > 600:
        n = 8192
    elif remaining > 300:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback (global variable `height_sequence_1`)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]      # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Analytic starts (α‑grid + sine‑power grid)
    # -------------------------------------------------
    # α‑grid (dense)
    alphas = np.linspace(-0.8, 2.2, 61)   # 61 points – cheap enough
    for alpha in alphas:
        if time.time() >= deadline - 30.0:
            break
        a = alpha_start(n, float(alpha))
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] new best α={alpha:.3f}  score={best_score:.6f}")

    # sine‑power grid
    powers = np.linspace(0.4, 3.0, 27)    # sin^p with p≈1 gives a bell‑shape
    for p in powers:
        if time.time() >= deadline - 30.0:
            break
        a = sine_start(n, float(p))
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] new best sin‑p={p:.3f}  score={best_score:.6f}")

    # -------------------------------------------------
    # 2. Adam‑optimised refinements from a few diverse starts
    # -------------------------------------------------
    # we will afford at most three independent runs (shorter each)
    runs = 0
    while time.time() < deadline - 250.0 and runs < 3:
        # pick a random start: either sparse or a small perturbation of the current best
        if best_seq is not None and np.random.rand() < 0.6:
            # perturb the current best a little before feeding to Adam
            base = np.array(best_seq, dtype=float)
            perturb = np.random.normal(scale=0.02, size=base.shape)
            start = base + perturb
            start = np.maximum(start, 0.0)
            start = project_to_simplex(start)
        else:
            start = random_sparse_start(n)

        # allocate a slice of the remaining time for this run
        time_left = deadline - time.time()
        # Aim for about 100k iterations or less if time is tight
        max_iters = int(min(120_000, max(20_000, time_left / 0.006)))   # crude estimate
        # randomise hyper‑parameters a little
        beta_end = np.random.uniform(300.0, 600.0)
        eta0 = np.random.uniform(0.5, 0.9)

        a_opt, sc_opt = exp_adam_optimize(
            start,
            deadline - 180.0,
            max_iters=max_iters,
            beta_start=0.5,
            beta_end=beta_end,
            eta0=eta0,
        )
        runs += 1
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] run {runs} improved score to {best_score:.6f}")

    # -------------------------------------------------
    # 3. Long dedicated Adam run from the best candidate so far
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] long Adam refinement")
        remaining_time = deadline - time.time()
        # more iterations thanks to the larger remaining budget
        max_iters = int(min(300_000, max(80_000, remaining_time / 0.006)))
        a_long, sc_long = exp_adam_optimize(
            np.array(best_seq, dtype=float),
            deadline - 120.0,
            max_iters=max_iters,
            beta_start=0.5,
            beta_end=500.0,
            eta0=0.55,
        )
        if sc_long < best_score:
            best_score = sc_long
            best_seq = a_long.tolist()
            print(f"[{time.time():.1f}] after long Adam best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Final simulated‑annealing polish
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final simulated‑annealing")
        a_final, sc_final = simulated_annealing(
            np.array(best_seq, dtype=float),
            deadline,
            max_moves=70_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] annealing improved best to {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: if nothing succeeded fall back to a modest uniform vector
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
