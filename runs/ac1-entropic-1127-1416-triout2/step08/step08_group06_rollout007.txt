   (arcsine, gamma‑type, power‑law, beta, uniform‑mixture and a

   All candidates are normalised to L1‑norm = 1 and scored with a fast

   complementary optimisation phases that respect the simplex:

     iterations with a decreasing stepsize.  This handles the
     non‑smooth max‑convolution directly and keeps the iterate on the

     polish briefly with EG and keep the improvement if any.

3. **Time‑aware** – the whole routine monitors the wall‑clock and always

# ----------------------------------------------------------------------
# Imports and low‑level utilities

import time

# ----------------------------------------------------------------------
# Convolution utilities (linear convolution via real FFT)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector whose entries
    sum to 1.  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (all normalise to L1‑norm = 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope (β = 0.5)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β)‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape – the denominator is raised to p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Linear mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_gamma_shape(
    n: int, gamma: float, delta: float = 0.5, eps: float = 0.0
) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+δ)(n‑1‑i+δ)+ε)^{-γ}.
    The classic arcsine corresponds to γ=0.5, δ≈0.5, ε≈0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + delta) * (n - 1 - i + delta) + eps) ** gamma
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_sparse_shape(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random sparse shape: choose ``m`` positions uniformly at random and
    allocate Dirichlet(1) weights on them.  All other entries are zero.
    """
    idx = rng.choice(n, size=m, replace=False)
    # Dirichlet(1) = iid Exponential(1) normalised
    raw = rng.exponential(scale=1.0, size=m)
    raw /= raw.sum()
    seq = np.zeros(n, dtype=float)
    seq[idx] = raw
    return seq


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the exact sub‑gradient method
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (KL‑mirror) polish
# ----------------------------------------------------------------------
def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.015,
) -> np.ndarray:
    """
    Mirror‑descent on the original max‑convolution objective using KL
    divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # reverse mask for convolution with a
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # decaying stepsize (simple 1/√t schedule)
        eta = step0 / np.sqrt(it)
        a = a * np.exp(-eta * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 100_000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 2_000_000,
    step0: float = 0.07,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for “active” indices
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 200_000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42, budget_s: int = 1000, **kwargs
) -> List[float]:
    """
    Search for a step sequence with a low official ``evaluate_sequence``
    score.  The routine respects the 1000 s wall‑clock limit and always
    returns a non‑empty sequence of non‑negative numbers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load a fallback (if the environment already defines one)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Random parametric exploration (length 2048)
    # ------------------------------------------------------------------
    n = 2048
    print("\n=== Random parametric search (n = 2048) ===")
    # Give approx. 30 % of the total budget to this phase
    random_deadline = start + 0.30 * (deadline - start)

    iter_cnt = 0
    while time.time() < random_deadline:
        iter_cnt += 1
        shape = rng.choice(
            ["arcsine", "beta", "power", "gamma", "mixture", "sparse", "uniform"]
        )

        if shape == "arcsine":
            cand = arcsine_initial(n)

        elif shape == "beta":
            alpha = 10 ** rng.uniform(-1.0, 0.7)   # ≈[0.1,5]
            beta = 10 ** rng.uniform(-1.0, 0.7)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            cand = generate_beta_shape(n, alpha, beta, eps)

        elif shape == "power":
            p = rng.uniform(0.2, 0.9)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)

        elif shape == "gamma":
            gamma = rng.uniform(0.15, 0.85)   # broadened range
            delta = rng.uniform(0.05, 1.0)
            eps = rng.uniform(0.0, 0.5)
            cand = generate_gamma_shape(n, gamma, delta, eps)

        elif shape == "mixture":
            w = rng.uniform(0.0, 1.0)
            cand = generate_mixture_shape(n, w)

        elif shape == "sparse":
            # pick a number of spikes roughly sqrt(n) but vary a bit
            m = int(rng.integers(int(np.sqrt(n) // 2), int(np.sqrt(n) * 2) + 1))
            cand = generate_sparse_shape(n, m, rng)

        else:   # uniform
            cand = np.full(n, 1.0 / n)

        # Normalise and keep a tiny epsilon to avoid exact zeros in EG
        cand = np.clip(cand, 1e-15, None)
        cand = cand / cand.sum()

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.tolist()
            print(f"   New best it={iter_cnt:,d} score={val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Global polishing – aggressive optimisation on the current best
    # ------------------------------------------------------------------
    if not best_seq:
        # very unlikely, but guarantee a fallback
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))
        print("Fallback to uniform sequence.")

    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n=== Global polishing phase ===")
        a = np.array(best_seq, dtype=float)

        # 2a. Long KL‑mirror (exponentiated‑gradient) run
        eg_iters = int(min(800_000, remaining * 1500))
        print(f"   EG polish: up to {eg_iters:,d} iters")
        a = exponentiated_gradient_polish(
            a,
            deadline=deadline,
            max_iter=eg_iters,
            step0=0.012,
        )
        val = compute_score(a)
        if val < best_val:
            best_val, best_seq = val, a.tolist()
            print(f"   After EG polish new best score={val:.6f}")

        remaining = deadline - time.time()
        # 2b. Short stochastic local‑perturbation hill‑climb
        if remaining > 5.0:
            print("   Local stochastic hill‑climb")
            hill_start = time.time()
            sigma = 0.05  # initial perturbation scale
            while time.time() < deadline - 2.0:
                # gradually shrink perturbation radius
                sigma *= 0.9995
                a_candidate = a + sigma * rng.standard_normal(n)
                a_candidate = np.clip(a_candidate, 0.0, None)
                if a_candidate.sum() <= 0.0:
                    continue
                a_candidate /= a_candidate.sum()

                # brief EG polishing (5 k iterations) to settle the noisy point
                sub_deadline = time.time() + 3.0  # keep it cheap
                a_tmp = exponentiated_gradient_polish(
                    a_candidate,
                    deadline=sub_deadline,
                    max_iter=5_000,
                    step0=0.008,
                )
                cand_val = compute_score(a_tmp)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = a_tmp.tolist()
                    a = a_tmp  # move the centre of the search
                    print(
                        f"     Hill‑climb improvement: sigma={sigma:.5f} "
                        f"score={cand_val:.6f}"
                    )
            print(f"   Hill‑climb elapsed {time.time() - hill_start:.1f}s")

        remaining = deadline - time.time()
        # 2c. Final exact sub‑gradient descent (if time permits)
        if remaining > 3.0:
            print("   Final sub‑gradient descent")
            sub_iters = int(min(2_000_000, remaining * 2000))
            a = subgradient_descent_exact(
                a,
                deadline=deadline,
                max_iter=sub_iters,
                step0=0.07,
            )
            val = compute_score(a)
            if val < best_val:
                best_val, best_seq = val, a.tolist()
                print(f"   After sub‑gradient new best score={val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty valid return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # 4.  Final official evaluation (logging only)
    # ------------------------------------------------------------------
    try:
        final_official = evaluate_sequence(best_seq)
        print(f"\nSearch finished. Final official score = {final_official:.6f}")
    except Exception as exc:
        print(f"\nSearch finished. Official evaluation failed: {exc}")

    return best_seq
