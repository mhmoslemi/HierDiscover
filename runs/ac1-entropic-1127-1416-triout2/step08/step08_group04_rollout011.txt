"""
Propose a step‑function sequence with a low official score.

Algorithm
---------

1. **Coarse shape search** – generate a few families of candidate
   distributions (uniform, arcsine, power‑shape, arcsine/ uniform mixture)
   on several lengths.  The fast surrogate ``2·n·max(conv(a,a))`` is used
   to keep the best candidate.

2. **Conditional‑gradient (Frank–Wolfe) refinement** – starting from the
   best candidate, solve the linear programme  

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k,
                    g_i ≥ 0,

   where ``M = max(f * f)``.  The LP gives a feasible direction ``g`` that
   can reduce the maximal convolution.  A one‑dimensional line search
   finds the optimal mixing factor ``t`` for the convex combination
   ``f←(1‑t)·f + t·g``.  The process is repeated (≤ 20 iterations) while
   respecting the global time budget.

3. **Exponential‑gradient polish** – a few thousand mirror‑descent steps
   (KL‑projection) further lower the smooth‑max surrogate.

4. **Cheap stochastic hill climbing** – tiny mass transfers between random
   coordinates are accepted if they improve the surrogate.

5. **Official evaluation** – the final sequence is checked with the
   provided ``evaluate_sequence`` routine.  A non‑empty fallback (uniform
   of length 1024) guarantees a return value even if everything else fails.

The whole search respects the 1000 s wall‑clock limit.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate used throughout the search.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# Shape generators (used for the coarse grid search)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float = 1e-8) -> np.ndarray:
    """
    Generalised arcsine shape a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """
    Mixture weight_arc·arcsine + (1‑weight_arc)·uniform,
    normalised to sum = 1.
    """
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 150_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent) polish.
    Keeps the vector on the probability simplex.
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries attaining the max (within tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev_mask = mask[::-1]
        subgrad = 2.0 * convolution_fft(a, rev_mask)[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def local_random_search(a: np.ndarray,
                        deadline: float,
                        max_trials: int = 20_000,
                        step_scale: float = 0.001,
                        rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Very cheap stochastic hill climbing: move a tiny fraction of mass
    from one coordinate to another and keep improvements.
    """
    a_cur = a.copy()
    score_cur = compute_score(a_cur)
    n = a_cur.size

    while time.time() < deadline - 0.2 and max_trials > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step_scale * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue
        s_new = compute_score(a_new)
        if s_new < score_cur:
            a_cur = a_new
            score_cur = s_new
        max_trials -= 1
    return a_cur

def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n‑1, n)) such that A @ g
    equals the linear convolution f * g.
    """
    from scipy.linalg import toeplitz
    n = f.size
    c = np.concatenate([f, np.zeros(n - 1)], dtype=np.float64)
    r = np.concatenate([f[:1], np.zeros(n - 1)], dtype=np.float64)
    A = toeplitz(c, r)               # shape (2n‑1, n)
    return A


def frank_wolfe_lp_refine(init_f: np.ndarray,
                          deadline: float,
                          max_outer: int = 20,
                          max_line_search: int = 15) -> np.ndarray:
    """
    Conditional‑gradient refinement:
      * solve the LP maximise Σ g_i subject to (f*g) ≤ M,
      * mix f←(1‑t)f + t g with a line‑search for the optimal t.
    Returns the improved vector (still on the simplex).
    """
    import scipy.optimize as opt

    f = init_f.copy()
    n = f.size
    best = f.copy()
    best_score = compute_score(best)

    for outer in range(max_outer):
        if time.time() >= deadline:
            break

        # current bound
        conv_ff = convolution_fft(f, f)
        M = float(conv_ff.max())

        # linear programme
        A = build_conv_matrix(f)                     # (2n‑1, n)
        b = np.full(2 * n - 1, M * (1.0 + 1e-12))   # tiny slack for numerics
        c = -np.ones(n)                              # minimize –Σg_i  → maximize Σg_i
        bounds = [(0, None) for _ in range(n)]

        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
        if not res.success:
            print(f"    LP failed at outer iter {outer}: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1.0 + 1e-12:            # no useful direction
            print(f"    LP did not improve sum (outer {outer})")
            break

        # scale to keep the simplex constraint (Σg = 1)
        g = g0 / sum_g0

        # pre‑compute the three convolutions needed for the line search
        conv_fg = convolution_fft(f, g)
        conv_gg = convolution_fft(g, g)

        # ----- line search on t ∈ [0,1] to minimise max(conv_mix) -----
        def max_at_t(t: float) -> float:
            a = 1.0 - t
            mix = (a * a) * conv_ff + 2.0 * t * a * conv_fg + (t * t) * conv_gg
            return float(mix.max())

        # golden‑section search (few iterations → cheap)
        lo, hi = 0.0, 1.0
        phi = (np.sqrt(5.0) - 1.0) * 0.5      # ≈0.618
        c_pt = hi - phi * (hi - lo)
        d_pt = lo + phi * (hi - lo)
        fc = max_at_t(c_pt)
        fd = max_at_t(d_pt)
        for _ in range(max_line_search):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - phi * (hi - lo)
                fc = max_at_t(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + phi * (hi - lo)
                fd = max_at_t(d_pt)
        t_opt = (lo + hi) * 0.5

        # update f
        f = (1.0 - t_opt) * f + t_opt * g

        # enforce the simplex (numerical drift)
        total = f.sum()
        if total <= 0.0 or not np.isfinite(total):
            f = np.full_like(f, 1.0 / n)
        else:
            f = f / total

        # keep the best ever seen
        cur_score = compute_score(f)
        print(f"    FW outer {outer:2d}, t={t_opt:.4f}, surrogate={cur_score:.6f}")
        if cur_score < best_score:
            best_score, best = cur_score, f.copy()

    return best

            print(f"  New best (arcsine) surrogate={s:.6f}")

        # Power shapes
        for p in p_grid:
            shape = generate_power_shape(n, p, eps=1e-8)
            s = compute_score(shape)
            if s < best_surrogate:
                best_surrogate, best_seq = s, shape.tolist()
                print(f"  New best (p={p:.3f}) surrogate={s:.6f}")

        # Mixtures arcsine / uniform
        for w in w_grid:
            mix = generate_mixture_shape(n, w)
            s = compute_score(mix)
            if s < best_surrogate:
                best_surrogate, best_seq = s, mix.tolist()
                print(f"  New best (mix w={w:.2f}) surrogate={s:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Frank‑Wolfe LP refinement (conditional gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0:      # ensure we have enough time for further stages
        print("\n--- Stage 2: Frank‑Wolfe LP refinement ---")
        f_init = np.array(best_seq, dtype=float)
        f_refined = frank_wolfe_lp_refine(f_init, deadline,
                                          max_outer=20, max_line_search=12)
        s = compute_score(f_refined)
        print(f"  After FW refinement surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, f_refined.tolist()
    else:
        print("\nSkipping FW refinement (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print("\n--- Stage 3: exponentiated‑gradient polish ---")
        a = np.array(best_seq, dtype=float)
        a = eg_refine(a, deadline, max_iter=120_000, eta0=0.25)
        s = compute_score(a)
        print(f"  After EG polish surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping EG polish (insufficient time)")

    # ------------------------------------------------------------------
    # Stage 4 – cheap stochastic local search
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Stage 4: local random search ---")
        a = np.array(best_seq, dtype=float)
        a = local_random_search(a, deadline,
                                max_trials=30_000,
                                step_scale=0.001,
                                rng=rng)
        s = compute_score(a)
        print(f"  After local search surrogate={s:.6f}")
        if s < best_surrogate:
            best_surrogate, best_seq = s, a.tolist()
    else:
        print("\nSkipping local random search (insufficient time)")

    # ------------------------------------------------------------------
    # Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"\nOfficial final score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")
        final_val = float("inf")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        final_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform length {fallback_n}, score = {final_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_seq
