"""
propose_candidate
-----------------
A multi‑stage optimizer for the non‑negative step‑function problem.

Main ideas
~~~~~~~~~~
* Work first on a modest size (n≈4k) to explore many random starts quickly.
* Apply a short L2‑norm smoothing phase (gradient descent on Σ(conv²)).
* Upsample the best candidate to a larger dimension (n≈16k) where the
  theoretical optimum is approached.
* Refine the large vector with a long Adam minimisation of the
  smoothed max‑convolution (log‑sum‑exp) – the β‐schedule is pushed up
  to 5·10².
* Finish with a cheap projected sub‑gradient descent, a symmetric
  projection and a final simulated‑annealing hill‑climb.
* All iterates are kept on the probability simplex; after each step the
  vector is symmetrised (a ← (a+rev(a))/2) because high‑quality solutions
  are typically symmetric.

The routine respects the global time budget (default 1000 s) and prints
progress information.  It returns a plain Python list ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used during the search:
        2·n·max(a*a)
    a must lie on the simplex (sum=1, non‑negative).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def compute_l2_score(a: np.ndarray) -> float:
    """Auxiliary smooth objective Σ (a*a)²."""
    conv = convolution_fft(a, a)
    return float(np.sum(conv * conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize_and_project(a: np.ndarray) -> np.ndarray:
    """Force a to be symmetric and project back onto the simplex."""
    a = (a + a[::-1]) * 0.5
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Analytic families (used for a quick grid‑search on the small scale)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)        # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def gradient_l2(a: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ (a*a)² w.r.t. a.
    Formula: 4·cross‑correlation(conv, a_rev) on the central slice.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    cross = convolution_fft(conv, a[::-1])  # length 3n‑2
    grad = 4.0 * cross[n - 1:n - 1 + n]
    return grad


def gradient_smoothed_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max f_β(a) =
        (1/β)·log Σ_k exp(β·(a*a)_k)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    maxc = conv.max()
    exp_w = np.exp(beta * (conv - maxc))
    w = exp_w / exp_w.sum()               # softmax weights (length 2n‑1)

    a_rev = a[::-1]
    grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]
    return grad

# Optimisers
# ----------------------------------------------------------------------


def l2_adam(start_a: np.ndarray,
            deadline: float,
            max_iters: int = 60_000,
            lr: float = 0.02) -> tuple[np.ndarray, float]:
    """Adam on the smooth L2 objective Σ (a*a)²."""
    a = start_a.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_a = a.copy()
    best_val = compute_l2_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        grad = gradient_l2(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = symmetrize_and_project(a)

        if it % 2_000 == 0:
            val = compute_l2_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()

    # final check
    val = compute_l2_score(a)
    if val < best_val:
        best_val = val
        best_a = a.copy()
    return best_a, best_val


def smooth_max_adam(start_a: np.ndarray,
                    deadline: float,
                    max_iters: int = 300_000,
                    beta_start: float = 6.0,
                    beta_end: float = 500.0,
                    lr: float = 0.015) -> tuple[np.ndarray, float]:
    """Adam on the smoothed‐max objective (log‑sum‑exp)."""
    a = start_a.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        grad = gradient_smoothed_max(a, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = symmetrize_and_project(a)

        if it % 4_000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final evaluation
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    a = start_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = symmetrize_and_project(a)

        if t % 2_000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb.
    Moves: swap two entries, or transfer a fraction of mass,
    plus occasional Dirichlet‑reshuffle of a tiny block.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule – tiny hill‑climb temperature
    T0 = best_score * 1e-4
    T = T0

    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()

        rnd = np.random.rand()
        if rnd < 0.4:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif rnd < 0.8:                   # simple mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        else:                             # block reshuffle (size 3‑5)
            blk = np.random.choice(n, size=np.random.randint(3, 6), replace=False)
            total = cand[blk].sum()
            if total > 0:
                # draw a fresh Dirichlet distribution on the block
                new_weights = np.random.gamma(1.0, 1.0, size=blk.size)
                new_weights = new_weights / new_weights.sum() * total
                cand[blk] = new_weights

        cand = symmetrize_and_project(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            # accept
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def upsample_sequence(seq_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from a small to a large vector (keeps sum≈1)."""
    n_small = len(seq_small)
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    seq_large = np.interp(x_large, x_small, seq_small)
    # re‑normalise onto the simplex and enforce symmetry
    seq_large = symmetrize_and_project(seq_large)
    return seq_large

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback to the globally supplied construction, if it exists.
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Small‑scale exploration (n≈4k)
    # ------------------------------------------------------------------
    n_small = 4096
    print(f"[{time.time():.1f}] small‑scale search, n = {n_small}")

    # a) Analytic families (coarse grid)
    alphas = np.linspace(0.0, 4.0, 41)
    for a in alphas:
        if time.time() >= deadline - 150.0:
            break
        vec = symmetric_product_family(n_small, a)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={a:.2f}) = {best_score:.6f}")

    betas = np.linspace(0.0, 4.0, 41)
    for b in betas:
        if time.time() >= deadline - 150.0:
            break
        vec = left_heavy_power_family(n_small, b)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={b:.2f}) = {best_score:.6f}")

    # b) Random starts + a very short L2‑Adam warm‑up
    for _ in range(30):
        if time.time() >= deadline - 150.0:
            break
        vec = random_monotone_start(n_small)
        # a brief L2 fine‑tuning (≈10 k iters)
        vec, _ = l2_adam(vec, deadline - 100.0, max_iters=12_000, lr=0.025)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono+L2) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Upsample to a large dimension (n≈16k) and symmetrise.
    # ------------------------------------------------------------------
    n_large = 16384
    if best_seq is None:
        # fallback uniform vector if everything failed
        best_seq = [1.0 / n_small] * n_small
    a_small = np.array(best_seq, dtype=np.float64)
    a_large = upsample_sequence(a_small, n_large)
    best_seq = a_large.tolist()
    best_score = compute_score(a_large)
    print(f"[{time.time():.1f}] after up‑sampling (n={n_large}) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Long smoothed‑max Adam optimisation on the large vector.
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print(f"[{time.time():.1f}] starting long smoothed‑max Adam")
        a_opt, sc_opt = smooth_max_adam(
            np.array(best_seq, dtype=np.float64),
            deadline - 200.0,
            max_iters=300_000,
            beta_start=8.0,
            beta_end=500.0,
            lr=0.015,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after smoothed‑max Adam = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Classical projected sub‑gradient refinement (very cheap per iteration)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 60.0,
            max_iters=2_000_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation test (many optimal solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = symmetrize_and_project(a_arr)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
