def annealed_hill_climb(seq, time_deadline, max_steps=500_000, start_temp=0.01, end_temp=0.000001):
    a = np.array(seq, dtype=float)
    best_a = a.copy()
    best_score = compute_score(best_a)  # exact
    n = len(a)
    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        # linear cooling
        t_frac = steps / max_steps
        temp = start_temp * (1 - t_frac) + end_temp * t_frac
        # propose small random move
        cand = best_a.copy()
        i, j = np.random.randint(n, size=2)
        delta = (np.random.rand() * 0.02) * cand[i]  # up to 2% of mass
        cand[i] = max(cand[i] - delta, 0.0)
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
        else:
            # acceptance probability
            if np.random.rand() < np.exp(-(cur_score - best_score) / (temp + 1e-12)):
                best_a = cand
        steps += 1
    return best_a, best_score

"""
Hybrid optimizer for the step‑function evaluation problem.

The search proceeds in four stages, all respecting the 1000 s wall‑clock limit.

1. **Soft‑max (log‑sum‑exp) Adam optimisation** – a smooth approximation of the
   max‑convolution is minimised, starting from several random restarts.
   The annealing schedule for the soft‑max temperature β is extended to a
   larger final value (2000) to obtain a very tight surrogate.

2. **Projected sub‑gradient descent** – the exact ‑max‑objective is tackled
   directly; the sub‑gradient points to the reversed shift of the index where
   the convolution attains its maximum.

3. **Linear‑programming refinement** – for the current vector `a` we solve
   the linear program
        maximise   1ᵀg
        subject to  a∗g ≤ M,  g ≥ 0
   where `M = max(a∗a)`.  The solution `g` is normalised to the simplex and
   mixed with `a`.  This step tends to spread the mass away from the
   high‑overlap positions and often reduces the peak convolution further.

4. **Stochastic hill‑climb with simulated‑annealing acceptance** – tiny mass
   transfers between random indices are performed; moves that improve the
   exact score are always accepted, worse moves are accepted with a
   temperature‑controlled probability.  The schedule cools from 1e‑2 to 1e‑6.

Throughout we keep the vector on the probability simplex (non‑negative,
summing to 1) and evaluate the exact objective using an FFT‑based convolution.
The best vector found is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for the LP refinement
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large arrays)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_maxconv(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a∗a (a is assumed to sum to 1)."""
    conv = convolution_fft(a, a)
    return float(conv.max())


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used during the search.
    For a simplex vector this equals 2·n·max(a∗a).
    """
    n = a.shape[0]
    return 2.0 * n * compute_maxconv(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth max optimisation (log‑sum‑exp) with Adam momentum
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 0.5,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a soft‑max surrogate of  max(conv(a,a)) with Adam momentum.
    Returns the best exact score (2·n·maxconv) and the associated vector.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment estimate
    v = np.zeros_like(a)   # second moment estimate

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # -------- anneal β -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # -------- smoothed objective ---------------------------------------
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        # protect against under‑flow / overflow
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # -------- gradient of the soft‑max (cross‑correlation) -------------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)       # length 3n‑2, we need centre n
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # -------- Adam update ----------------------------------------------
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = a - update
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # small debug print
                print(f"[smooth] it={it}  score={best_score:.6f}")

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
        print(f"[smooth] final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Returns the best vector and its exact score.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the maximal entry (see analysis)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # debug
            if t % 5000 == 0:
                print(f"[subgrad] it={t}  score={best_score:.6f}")

        a = a_new

        # jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
        print(f"[subgrad] final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Linear‑programming refinement
# ----------------------------------------------------------------------


def lp_step(current_a: np.ndarray, max_val: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀg   subject to   current_a ∗ g ≤ max_val ,  g ≥ 0 .
    Returns the optimal g (not normalised) or None on failure.
    """
    if not SCIPY_AVAILABLE:
        return None

    n = current_a.shape[0]
    rows = 2 * n - 1

    # Build the constraint matrix A where A[k,i] = current_a[i] if i contributes to conv[k]
    A = np.zeros((rows, n), dtype=np.float64)
    for i, ai in enumerate(current_a):
        A[i:i + n, i] = ai

    b = np.full(rows, max_val, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                     # maximise sum(g)  →  minimise -sum(g)
    bounds = [(0, None) for _ in range(n)]

    # The HiGHS solver is fast for dense problems of this size.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={"presolve": True})
    if res.success:
        return res.x
    return None


def lp_refine(
    a: np.ndarray,
    time_deadline: float,
    max_outer_iters: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP refinement (mixing the current vector with the
    LP solution) while it yields an improvement.
    """
    cur_a = a.copy()
    cur_score = compute_score(cur_a)

    for outer in range(max_outer_iters):
        if time.time() >= time_deadline:
            break

        M = compute_maxconv(cur_a)
        g = lp_step(cur_a, M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g_norm = g / sum_g

        # mixing weight – start generous then shrink if needed
        t = 0.5
        mixed = (1.0 - t) * cur_a + t * g_norm
        mixed = project_to_simplex(mixed)

        mixed_score = compute_score(mixed)
        if mixed_score < cur_score:
            cur_score = mixed_score
            cur_a = mixed
            print(f"[LP] iteration {outer+1}: improved score = {cur_score:.6f}")
        else:
            # No improvement – stop early
            print(f"[LP] iteration {outer+1}: no improvement, terminating LP phase.")
            break

    return cur_a, cur_score


# ----------------------------------------------------------------------
# 4. Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def annealed_hill_climb(
    seq,
    time_deadline,
    max_steps: int = 500_000,
    start_temp: float = 1e-2,
    end_temp: float = 1e-6,
) -> tuple[list[float], float]:
    """
    Small random mass‑transfer moves with a temperature schedule.
    Moves that improve the exact score are always accepted, worse moves
    are accepted with probability exp(‑Δ/ T).  The temperature cools linearly.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        # linear cooling
        temp = start_temp + (end_temp - start_temp) * (step / max_steps)

        # propose a small transfer
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() * 0.02) * best_a[i]   # up to 2 % of the source mass
        cand = best_a.copy()
        cand[i] = max(cand[i] - delta, 0.0)
        cand[j] += delta
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
        else:
            # Metropolis acceptance
            if np.random.rand() < math.exp(-(cur_score - best_score) / (temp + 1e-12)):
                best_a = cand

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 5. Simple hill‑climb (always‑improving) – final polishing
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 200_000,
) -> tuple[list[float], float]:
    """
    Deterministic local improvement: swap or transfer a small fraction of mass.
    Only moves that strictly improve the exact score are kept; the step size
    adapts based on recent successes.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # initial step fraction
    step_scale = 0.07

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.96                       # become more conservative

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (the sequence).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness's post‑processing
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"[fallback] score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem dimension
    # ------------------------------------------------------------------
    # 2048 affords a good trade‑off between expressiveness and speed.
    n = 2048
    print(f"[setup] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2. Prepare a starting vector (simplex, sum = 1)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a0 = np.array(best_seq, dtype=float)
        if a0.sum() == 0.0:
            a0 = np.random.rand(n)
        a0 = a0 / a0.sum()
    else:
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum()

    # ------------------------------------------------------------------
    # 3. Smooth‑max Adam optimisation (many short restarts)
    # ------------------------------------------------------------------
    reserve = 120.0  # seconds reserved for later phases
    while time.time() < deadline - reserve:
        remaining = deadline - time.time() - reserve
        # each short run gets at most 8 s (or a fraction of remaining time)
        time_limit = time.time() + max(5.0, min(8.0, remaining * 0.5))

        start_vec = np.random.rand(n)
        start_vec = start_vec / start_vec.sum()

        a_opt, scr = smooth_gradient_optimize(
            start_vec,
            time_deadline=time_limit,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=2000.0,
            step0=0.5,
        )
        exact = compute_score(a_opt)
        if exact < best_val:
            best_val = exact
            best_seq = a_opt.tolist()
            print(f"[smooth‑restart] new best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Longer smooth optimisation starting from the best so far
    # ------------------------------------------------------------------
    if time.time() < deadline - reserve:
        time_limit = time.time() + reserve * 0.4
        a_start = np.array(best_seq, dtype=float)
        a_opt, scr = smooth_gradient_optimize(
            a_start,
            time_deadline=time_limit,
            max_iters=800_000,
            beta_start=5.0,
            beta_end=2000.0,
            step0=0.5,
        )
        exact = compute_score(a_opt)
        if exact < best_val:
            best_val = exact
            best_seq = a_opt.tolist()
            print(f"[smooth‑final] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[subgrad] starting refinement")
        refined_a, refined_score = subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 20.0,
            max_iter=600_000,
            step0=0.05,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"[subgrad] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Linear‑programming refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE and time.time() < deadline - 15.0:
        print("[LP] starting refinement")
        lp_a, lp_score = lp_refine(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 7.0,
            max_outer_iters=5,
        )
        if lp_score < best_val:
            best_val = lp_score
            best_seq = lp_a.tolist()
            print(f"[LP] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[annealed hill‑climb] polishing")
        annealed_seq, annealed_score = annealed_hill_climb(
            best_seq,
            time_deadline=deadline - 2.0,
            max_steps=400_000,
            start_temp=1e-2,
            end_temp=1e-6,
        )
        if annealed_score < best_val:
            best_val = annealed_score
            best_seq = annealed_seq
            print(f"[annealed] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final deterministic hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[final hill‑climb] polishing")
        final_seq, final_score = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=200_000,
        )
        if final_score < best_val:
            best_val = final_score
            best_seq = final_seq
            print(f"[final] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 10. Report the final official evaluation (the harness will call it again)
    # ------------------------------------------------------------------
    final_report = evaluate_sequence(best_seq)
    print(f"\n[finished] final reported score = {final_report:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
