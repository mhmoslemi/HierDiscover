               g_i ≥ 0

produces a partner `g` that puts mass where `f` is small.  A convex
combination `f ← (1‑t)·f + t·ĝ` (with `ĝ` the normalised solution)
strictly reduces the objective `2·n·max(f*f)` while preserving the
normalisation.  Several short LP‑refinements are performed for each
random start.  The best sequence seen across many restarts (different
lengths, families and random seeds) is returned.  If `cvxpy` is not
available a simple hill‑climbing fallback (the previous approach) is
used.

All sequences are kept non‑negative and are normalised to have sum = 1,
which makes the objective scale‑invariant and keeps the numbers safely
inside the range enforced by ``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------
def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:               # pragma: no cover
        return math.inf

def _objective(f):
    """
    Compute the objective 2·n·max(f*f) for a normalised sequence (sum=1).
    This is equivalent to evaluate_sequence() up to the final scaling
    checks, but is much faster.
    """
    n = len(f)
    max_conv = np.convolve(f, f).max()
    return 2.0 * n * max_conv

def _initial_sequence(n, family, rng):
    """Generate a random non‑negative sequence of length `n` and normalise its sum to 1."""
    if family == 'exp':
        # α sampled log‑uniformly in [1e‑5, 5]
        log_a = rng.uniform(-5, math.log10(5))
        alpha = 10.0 ** log_a
        seq = np.exp(-alpha * np.arange(n, dtype=float))
    elif family == 'pow':
        p = rng.uniform(0.0, 2.5)
        idx = np.arange(1, n + 1, dtype=float)
        seq = idx ** (-p)
    elif family == 'mix':
        log_a = rng.uniform(-5, math.log10(5))
        alpha = 10.0 ** log_a
        eps = rng.uniform(0.0, 0.5)
        seq = np.exp(-alpha * np.arange(n, dtype=float)) + eps
    elif family == 'lin':
        beta = rng.uniform(0.0, 5.0 / n)
        seq = np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float))
    else:  # 'rand' – descending random numbers
        seq = rng.rand(n)
        seq.sort()
        seq = seq[::-1]
    total = seq.sum()
    if total <= 0:
        seq = np.ones(n, dtype=float) / n
    else:
        seq = seq / total
    return seq

def _toeplitz_matrix(f):
    """
    Build the (2n‑1)×n matrix A such that (A @ g) = f * g  (convolution).
    For each column j the vector f is placed starting at row j.
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_optimize(f, M):
    """
    Solve the linear program:
        maximise   Σ g_i
        subject to (f * g)_k ≤ M   (all k)
                   g_i ≥ 0
    Returns the normalised solution ĝ (sum(ĝ)=1).  Returns None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:               # pragma: no cover
        return None

    n = len(f)
    A = _toeplitz_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is a fast interior‑point solver for small LPs
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:               # pragma: no cover
        return None

    if g.value is None:
        return None

    g_opt = np.asarray(g.value, dtype=np.float64)
    s = g_opt.sum()
    if s <= 0:
        return None
    return g_opt / s   # normalise to sum‑1

def _hill_climb(seq, rng, step=0.12, n_mut=5):
    """Simple multiplicative perturbation of a few entries."""
    cand = np.array(seq, copy=True)
    n = len(cand)
    m = rng.randint(1, min(n_mut, n))
    idxs = rng.choice(n, size=m, replace=False)
    for i in idxs:
        factor = 1.0 + rng.normal(0.0, step)
        cand[i] = max(0.0, cand[i] * factor)
    cand = cand / cand.sum()
    return cand.tolist()

# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds,
    mixing random parametric starts with an LP‑based refinement and occasional
    hill‑climbing.
    """
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # overall deadline (small safety margin)
    deadline = time.time() + budget_s - 0.3

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 0)  Start from the global reference sequence if it exists.
    # ------------------------------------------------------------------
    try:
        init = globals().get('height_sequence_1')
        if init is not None and isinstance(init, (list, tuple, np.ndarray)):
            arr = np.asarray(init, dtype=float)
            if arr.sum() > 0:
                arr = arr / arr.sum()
                sc = _score(arr.tolist())
                if sc < best_score:
                    best_score, best_seq = sc, arr.tolist()
                    print(f"Initial reference score: {best_score:.6f}")
    except Exception:               # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1)  Main search loop – many independent restarts.
    # ------------------------------------------------------------------
    lengths = [400, 600, 800, 1000, 1200]     # small enough for fast LP solves
    families = ['exp', 'pow', 'mix', 'lin', 'rand']

    # try to import cvxpy once – if it fails we fall back to pure hill‑climb
    try:
        import cvxpy as _cvxpy  # noqa: F401
        HAVE_CVX = True
    except Exception:               # pragma: no cover
        HAVE_CVX = False

    while time.time() < deadline:
        # pick a fresh start
        n = int(rng.choice(lengths))
        family = random.choice(families)
        f = _initial_sequence(n, family, rng)   # sum(f) == 1

        # quick check of current objective (no need to call evaluate_sequence yet)
        cur_obj = _objective(f)

        # ------------------------------------------------------------------
        # 2)  LP‑based refinement – few iterations only.
        # ------------------------------------------------------------------
        if HAVE_CVX:
            t = 0.2               # convex combination weight
            for _ in range(15):   # max LP iterations per start
                if time.time() >= deadline:
                    break
                M = np.convolve(f, f).max()
                g_hat = _lp_optimize(f, M)
                if g_hat is None:
                    break
                # convex update and re‑normalise
                f_new = (1.0 - t) * f + t * g_hat
                f_new = f_new / f_new.sum()
                new_obj = _objective(f_new)
                if new_obj < cur_obj - 1e-12:
                    f, cur_obj = f_new, new_obj
                    # Keep t modest; if no progress reduce it
                else:
                    t *= 0.5
                    if t < 1e-3:
                        break

        # ------------------------------------------------------------------
        # 3)  Evaluate the candidate with the true objective.
        # ------------------------------------------------------------------
        cand_seq = f.tolist()
        cand_score = _score(cand_seq)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand_seq
            print(f"New global best (n={n}, family={family}) → {best_score:.6f}")

        # ------------------------------------------------------------------
        # 4)  Occasionally apply a short hill‑climb to the current best.
        # ------------------------------------------------------------------
        if best_seq is not None and random.random() < 0.12:
            pert = _hill_climb(best_seq, rng, step=0.10, n_mut=4)
            pert_score = _score(pert)
            if pert_score < best_score:
                best_score, best_seq = pert_score, pert
                print(f"Hill‑climb improvement → {best_score:.6f}")

        # Early exit if we already beat the target (with a tiny safety margin)
        if best_score <= 1.503 + 1e-6:
            break

    # ------------------------------------------------------------------
    # 5)  Fallback if nothing was found (should never happen).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 500
        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
