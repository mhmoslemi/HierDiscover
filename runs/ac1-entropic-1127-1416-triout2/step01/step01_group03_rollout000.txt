#!/usr/bin/env python3
"""
Project‑subgradient optimisation for the max‑convolution objective.

The evaluation function is scale‑invariant, so we fix the sum of the
coefficients to 1 (probability simplex).  The problem reduces to
minimising  max_k (a*a)_k .  We perform a projected sub‑gradient descent:

* compute the self‑convolution b = a*a,
* locate the index (or indices) where b attains its maximum,
* a sub‑gradient of max(b) w.r.t. a is g_i = 2·a_{k-i} summed over all
  maximal indices k,
* take a step a ← a – η·g and project back onto the simplex.

The method is repeated from several random restarts (including a
restart from the best known sequence and from the classic
(1–z)^{-1/2} distribution).  An adaptive step‑size η_t = η₀/√t and
occasional jitter keep the search from stagnating.  The best
sequence found before the time budget expires is returned.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (top‑level – required by the platform)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to the uniform distribution
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def sqrt_distribution(n: int) -> np.ndarray:
    """
    Normalised coefficients of (1‑z)^(-1/2) up to length n.
    The raw coefficients satisfy c_0 = 1, c_{i+1}=c_i·(2i+1)/(2i+2).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def max_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution a*a."""
    return np.convolve(a, a)


def subgradient_max_conv(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max_k (a*a)_k at the current point.
    For each maximal index k we have ∂b_k/∂a_i = 2·a_{k-i} (zero if out of range).
    The returned gradient is the average over all maximal indices.
    """
    n = a.size
    grad = np.zeros_like(a)

    # Loop over each maximal index and add its contribution.
    for k in max_idxs:
        # i runs over indices where both i and k-i lie in [0, n‑1].
        # The valid range for i is:
        #   max(0, k-(n-1)) ≤ i ≤ min(k, n-1)
        i_start = 0 if k < n else k - (n - 1)
        i_end = n - 1 if k >= n - 1 else k
        # Slice of i's
        idx = np.arange(i_start, i_end + 1, dtype=int)
        # Corresponding a_{k-i}
        rev = a[k - idx]
        grad[idx] += 2.0 * rev

    if max_idxs.size > 1:
        grad /= float(max_idxs.size)
    return grad


def local_score(a: np.ndarray, n: int) -> float:
    """Fast evaluation of the objective: 2·n·max(a*a)."""
    b = np.convolve(a, a)
    return 2.0 * n * b.max()

    # (3) a random uniform point (used for the first restart)
    start_points.append(project_to_simplex(np.random.rand(n)))

    # Keep track of the best sequence seen so far
    best_seq = None
    best_score = float("inf")

    # Helper for updating the global best
    def maybe_update(a_vec: np.ndarray):
        nonlocal best_seq, best_score
        sc = local_score(a_vec, n)
        if sc < best_score:
            # Evaluate with the official function for safety (clip, etc.)
            try:
                true_sc = evaluate_sequence(a_vec.tolist())
            except Exception:
                true_sc = float("inf")
            if true_sc < best_score:
                best_score = true_sc
                best_seq = a_vec.copy()
                print(f"[update] new best score = {best_score:.6f}")
        return sc

    # --------------------------------------------------------------
    # Main optimisation loop: many short restarts, each running
    # a projected sub‑gradient descent.
    # --------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline:
        # -----------------------------------
        # Choose a start point for this restart
        # -----------------------------------
        if restart_counter < len(start_points):
            a = start_points[restart_counter].copy()
        else:
            # after the hand‑crafted starts, use jitter around the best
            if best_seq is not None and np.random.rand() < 0.8:
                jitter_scale = 0.005
                a = best_seq + jitter_scale * np.random.randn(n)
                a = project_to_simplex(a)
            else:
                a = np.random.rand(n)
                a = project_to_simplex(a)

        restart_counter += 1
        # --------------------------------------------------------------
        # Parameters for this descent run
        # --------------------------------------------------------------
        max_inner = 3000                # max inner iterations per restart
        eta0 = 0.6                      # initial step‑size

        # --------------------------------------------------------------
        # Inner projected sub‑gradient descent
        # --------------------------------------------------------------
        for it in range(max_inner):
            if time.time() >= deadline:
                break

            # self‑convolution and its maximum
            b = np.convolve(a, a)
            max_b = b.max()
            max_idxs = np.flatnonzero(b == max_b)

            # sub‑gradient of the max‑convolution
            g = subgradient_max_conv(a, max_idxs)

            # step‑size schedule (decreases like 1/√t)
            eta = eta0 / np.sqrt(it + 1.0)

            # gradient step + simplex projection
            a = project_to_simplex(a - eta * g)

            # occasional cheap score check (every 200 iterations)
            if (it + 1) % 200 == 0:
                cur_score = local_score(a, n)
                # Only call the expensive evaluate_sequence when we see an
                # improvement in the cheap bound.
                if cur_score < best_score:
                    maybe_update(a)

        # after the descent, evaluate the final point of this restart
        maybe_update(a)

    # --------------------------------------------------------------
    # Final sanity check – ensure we always return a valid list.
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (guaranteed finite score)
        best_seq = np.full(n, 1.0 / n, dtype=float)

    print(f"[done] best score achieved = {best_score:.6f} (length={n})")
    return best_seq.tolist()
