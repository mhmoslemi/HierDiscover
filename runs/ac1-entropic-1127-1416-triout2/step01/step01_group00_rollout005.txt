"""
Propose a low‑value step‑function sequence using a smooth max‑convolution
approximation and projected (mirrored) gradient descent.

The evaluation function is scale‑invariant, so we keep the sequence on the
probability simplex (non‑negative, sum = 1).  For a given vector `a` we
approximate
    max(a*a)  ≈  (1/p)·log Σ_k exp(p·(a*a)[k])
with a temperature parameter `p`.  For small `p` the objective is smooth and
easy to optimize; we gradually raise `p` (a deterministic‑annealing schedule)
so that the solution converges to the true L∞‑objective.
The gradient of the smooth approximation is a weighted sum of the convolution
gradients, which can be computed efficiently with FFTs and a sliding‑window
sum.  After each gradient step we project back onto the simplex (Euclidean
projection).  Several restarts (arcsine, uniform, random Dirichlet) and a
final small local search give a good chance of beating the previous bound
(~1.519) within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution (same as np.convolve) using FFT."""
    return convolution_fft(x, y)


def compute_score(a: np.ndarray) -> float:
    """Score 2·n·max(a*a) / (sum(a)²) – the exact quantity used by evaluate_sequence."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped initialization: a_i ∝ 1/√((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def softmax_weights(conv: np.ndarray, p: float) -> np.ndarray:
    """Soft‑max over the convolution values, temperature = 1/p."""
    scaled = p * conv
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)
    w = exp_vals / np.sum(exp_vals)
    return w


def gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max‑convolution objective:
        ∑_k w_k ∂(a*a)_k / ∂a
    where w is the soft‑max weight vector.
    The formula is
        grad_i = Σ_j w_{i+j} a_j + a_i· Σ_j w_{i+j},
    with indices taken modulo the valid range (0 ≤ i+j < 2n‑1).
    Both terms are computed in O(n log n) via FFTs.
    """
    n = len(a)
    # term1 = Σ_j w_{i+j} a_j  (cross‑correlation)
    w_rev = w[::-1]
    conv_full = full_convolution_fft(w_rev, a)          # length = 3n‑2
    idx = (2 * n - 2) - np.arange(n)                    # pick entries for each i
    term1 = conv_full[idx]

    # term2 = a_i * Σ_j w_{i+j}  (sliding window sum of w of length n)
    cumsum = np.cumsum(w)
    seg = cumsum[n - 1:] - np.concatenate(([0.0], cumsum[:-n]))
    term2 = a * seg

    grad = term1 + term2
    return grad


def local_refine(a: np.ndarray,
                 deadline: float,
                 max_steps: int = 5000,
                 step_sd: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic hill‑climbing: move a small amount of mass between two
    random positions and keep the change if it improves the exact score.
    """
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    for _ in range(max_steps):
        if time.time() >= deadline:
            break
        i, j = np.random.choice(n, size=2, replace=False)
        delta = np.random.randn() * step_sd
        # Clip delta so we do not create negative entries
        delta = np.clip(delta, -best_a[i], best_a[j])
        if abs(delta) < 1e-12:
            continue
        new_a = best_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # Re‑normalize (sum should stay 1, but floating‑point drifts)
        new_a = np.maximum(new_a, 0.0)
        new_a /= np.sum(new_a)
        sc = compute_score(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a
    return best_a, best_score


def improve_sequence_smooth(initial_a: np.ndarray,
                            time_deadline: float,
                            p_schedule: list[float] = [2, 4, 8, 16, 32, 64, 128, 256],
                            max_iter_total: int = 25000,
                            eta0: float = 0.8,
                            noise_every: int = 500,
                            noise_scale: float = 5e-5) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed max‑convolution objective.
    The temperature `p` is increased deterministically (annealing) as the
    optimisation proceeds.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    iter_cnt = 0
    p_idx = 0
    p = p_schedule[p_idx]

    while time.time() < time_deadline and iter_cnt < max_iter_total:
        # increase temperature every 2000 iterations (if possible)
        if iter_cnt // 2000 > p_idx and p_idx < len(p_schedule) - 1:
            p_idx += 1
            p = p_schedule[p_idx]
            print(f"  === temperature switched to p={p} at iter {iter_cnt} ===")

        # exact objective (for tracking)
        conv = convolution_fft(a, a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # smooth gradient
        w = softmax_weights(conv, p)
        grad = gradient_softmax(a, w)

        # step size (diminishing)
        eta = eta0 / np.sqrt(iter_cnt + 1.0)

        a_new = a - eta * grad

        if noise_every and (iter_cnt % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)
        a = a_new
        iter_cnt += 1

        if iter_cnt % 2000 == 0:
            print(f"    iter={iter_cnt:,d}  cur_score={cur_score:.6f}  best={best_score:.6f}  p={p}")

    # final local refinement if time permits
    if time.time() < time_deadline - 0.5:
        a_ref, scr_ref = local_refine(best_a, time_deadline, max_steps=8000, step_sd=5e-5)
        if scr_ref < best_score:
            best_score = scr_ref
            best_a = a_ref

    return best_a, best_score

            seq, val = improve_sequence_smooth(
                initial_a=init_a,
                time_deadline=deadline,
                p_schedule=p_sched,
                max_iter_total=25000,
                eta0=0.9,
                noise_every=400,
                noise_scale=1e-4,
            )
            print(f"    best smoothed score = {val:.6f}")

            # Exact evaluation (only once per run)
            exact_val = evaluate_sequence(seq.tolist())
            print(f"    exact evaluate_sequence score = {exact_val:.6f}")

            if exact_val < best_val:
                best_val = exact_val
                best_seq = seq.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # If nothing was found, fall back to a simple uniform sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats
    return best_seq
