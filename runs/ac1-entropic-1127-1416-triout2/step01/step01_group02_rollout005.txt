"""
Propose a candidate sequence for the convolution‐minimisation problem.

The algorithm starts from the analytically optimal “binomial’’ sequence
c_k = (2k choose k)/4**k (the discrete analogue of f(t)=t⁻¹ᐟ²).  This
sequence has a constant self‑convolution value of 1 on the interior,
giving the asymptotic bound π/2 ≈ 1.571.  To beat this bound we perform
a simple projected gradient descent on the quantity

    M = max_k (a * a)[k]

while keeping the ℓ₁‑norm ∑a_i fixed (mass is only transferred between
entries).  The gradient of M w.r.t. a_i is the sum of a_{k-i} over all
indices k where the convolution attains its current maximum; we compute
this efficiently by a cross‑correlation of the current sequence with the
mask of maximal entries.  At every step a small amount of mass is moved
from the index with the largest gradient contribution to the index with
the smallest contribution.  Moves are accepted only if they lower M;
occasionally a random perturbation is tried to escape plateaus.
The loop runs until the time budget (≈1000 s) is exhausted.
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Return c_k = (2k choose k) / 4**k for k = 0 .. length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        # recurrence: c_{k} = c_{k-1} * ((2k‑1)(2k)) / (4 k^2)
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def convolution_max(a: np.ndarray) -> float:
    """Return max of the linear convolution a*a."""
    b = np.convolve(a, a)
    return b.max()


# ----------------------------------------------------------------------
def gradient_weights(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of max(b) w.r.t. the entries of a.
    mask[i] == True  ⇔  (a*a)[i] is (one of) the maximal values.
    The gradient is w_i = Σ_{k∈max} a_{k-i}.
    Implemented via a cross‑correlation.
    """
    # cross‑correlation a ⋆ mask (mask is treated as float)
    corr = np.correlate(a, mask.astype(np.float64), mode='full')
    offset = len(mask) - 1               # shift that aligns with w_i
    return corr[offset: offset + a.size]   # length = a.size


# ----------------------------------------------------------------------

    Return a non‑negative step‑height sequence that (heuristically)
    minimises ``evaluate_sequence``.
    """

    deadline = time.time() + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # choose length – reuse a previous solution if available
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:

    # --------------------------------------------------------------
    # initialise with the canonical binomial sequence
    a = binomial_sequence(n)
    best_a = a.copy()
    best_score = float('inf')
    # initial scoring using the official evaluator (clipping, etc.)
    try:
        best_score = evaluate_sequence(best_a.tolist())
    except Exception:  # pragma: no cover – evaluator is guaranteed to exist
        pass
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # parameters for the gradient descent
    step_factor = 0.02          # fraction of the source entry to move
    decay = 0.9997              # geometric decay of the step size
    min_step = 1e-8
    temp_start = 1e-3          # simulated‑annealing temperature start
    temp_end = 1e-6

    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- current convolution and its maximum
        b = np.convolve(a, a)
        max_b = b.max()

        # ----- compute mask of indices attaining the maximum (allow tiny tolerance)
        atol = 1e-12 * max_b
        mask = np.isclose(b, max_b, atol=atol)

        # ----- gradient weights (how much each entry participates in the max)
        w = gradient_weights(a, mask)

        # ----- choose source (large contribution) and destination (small contribution)
        # we weight the source also by the amount of mass it holds
        src_candidates = np.where(a > 0)[0]
        if src_candidates.size == 0:
            break
        src_scores = w[src_candidates] * a[src_candidates]
        source = src_candidates[np.argmax(src_scores)]

        dest = np.argmin(w)            # smallest contribution
        if dest == source:             # pick the next best if they clash
            alt = np.argsort(w)
            for idx in alt:
                if idx != source:
                    dest = idx
                    break

        # ----- amount of mass to transfer
        delta = step_factor * a[source]
        if delta < min_step:
            # step has become too tiny – stop early
            break
        new_a = a.copy()
        new_a[source] -= delta
        new_a[dest] += delta

        new_max = np.convolve(new_a, new_a).max()

    except Exception:

    print(f"[final] best score (official)={final_score:.6f}")

    # return a plain Python list as required by the external framework
    return best_a.tolist()
