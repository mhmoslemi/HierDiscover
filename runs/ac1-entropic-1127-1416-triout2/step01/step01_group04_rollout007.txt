Implementation of projection onto simplex:

def project_onto_simplex(v):
    """Project onto the probability simplex {x>=0, sum=1}."""
    # algorithm from Duchi et al. 2008.
    u = np.sort(v)[::-1]
    sv = np.cumsum(u) - 1
    rho = np.where(u - sv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = sv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function sequence.

Algorithm
---------
1.  Choose the length `n`.  If a previously found sequence exists we keep
    its size, otherwise we fall back to `n = 2000`.

2.  Initialise the distribution with a discrete *arcsine* (Beta(½,½))
    shape, i.e.               a_i ∝ 1/√((i+½)*(n‑½‑i)) .
    This is the classic construction whose convolution is (almost)
    constant and gives a score ≈ π/2 ≈ 1.571.

3.  Perform a projected Adam optimisation on a *smooth* surrogate of the
    objective

        F(a) = 2·n·max_k (a∗a)_k  (with Σa_i = 1).

    The max is approximated by a soft‑max:
        w_k = exp(λ·c_k) / Σ_j exp(λ·c_j)   where c = a∗a .
    The gradient of the smoothed objective can be written in closed form
    and is computed using a single correlation (`np.convolve`) with the
    weight vector `w`.  After each Adam step the iterates are projected
    onto the probability simplex (non‑negative, sum = 1).

    * λ (soft‑max temperature) is increased slowly from 1 to ~30 so the
      optimisation first sees a smooth landscape and later focuses on the
      true max.

4.  To avoid stagnation a tiny random jitter is added every few thousand
    iterations and occasional full restarts (a fresh noisy arcsine) are
    performed if no improvement has been seen for a while.

5.  Throughout the run we keep the best sequence seen (according to the
    cheap surrogate score `2·n·max(c)`).  After the time budget we evaluate
    the best candidate with the provided `evaluate_sequence` function and
    return it.

All arithmetic uses `numpy` and the routine respects the 1000 s CPU time
budget.  The returned list contains only non‑negative floats.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# --------------------------------------------------------------

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i >= 0 , Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    # Find the largest rho where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Should not happen, but fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise exactly to 1
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= w_sum
    return w


def arcsine_initial(n: int, eps: float = 1e-14) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) distribution on {0,…,n‑1}.
    a_i ∝ 1/√((i+½)*(n‑½‑i) + eps) .
    The result is normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)          # just in case numerical negatives appear
    return vals / vals.sum()


def softmax_weights(c: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute soft‑max weights w_k = exp(lam*c_k) / Σ exp(lam*c_k)
    in a numerically stable way.
    """
    c_max = np.max(c)
    # shift for stability
    exps = np.exp(lam * (c - c_max))
    # guard against overflow/underflow
    total = np.sum(exps)
    if total == 0 or not np.isfinite(total):
        # fallback to uniform weights (very unlikely)
        return np.full_like(c, 1.0 / len(c))
    return exps / total


def gradient_smooth_obj(a: np.ndarray, lam: float) -> tuple:
    """
    Compute the gradient of the smoothed objective
        O(a) = 2*n * (c·w) / (Σ a)^2
    where c = a∗a, w = softmax(lam*c).

    Returns
    -------
    grad : np.ndarray
        Gradient w.r.t. a (same shape as a).
    c    : np.ndarray
        Full convolution a∗a (length 2n‑1).
    w    : np.ndarray
        Soft‑max weights (same length as c).
    t    : float
        Weighted average t = Σ_k w_k * c_k  (≈ max(c) for large λ).
    """
    n = a.shape[0]
    S = a.sum()
    # Full convolution a∗a
    c = np.convolve(a, a)                     # length 2n‑1
    # Soft‑max over the convolution entries
    w = softmax_weights(c, lam)
    t = np.dot(c, w)                          # smooth approximation of max(c)

    # Compute the term sum_j a_j * w_{i+j} for all i.
    # This is a correlation of w (reversed) with a.
    w_rev = w[::-1]
    corr = np.convolve(w_rev, a, mode='full')  # length 3n‑2
    # Slice to obtain exactly n values:
    # indices n‑1 … 2n‑2 (inclusive) → length n
    term = corr[n - 1: 2 * n - 1][::-1]       # term[i] = Σ_j a_j w_{i+j}

    # u_i = a_i + term_i
    u = a + term

    # Coefficient in front of the gradient
    C = 2.0 * n / (S * S)

    # Gradient (see derivation in the analysis)
    grad = C * (u - (2.0 * t / S) * np.ones_like(a))
    return grad, c, w, t

    """

    while time.time() < deadline:
        # Adaptive soft‑max temperature (linear schedule)
        lam = lam_start + (lam_end - lam_start) * (it / 200000.0)

        it += 1

            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = it

        final_score = best_score
