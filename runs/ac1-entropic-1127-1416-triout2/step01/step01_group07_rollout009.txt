"""
Annealed soft‑max gradient descent for minimizing
    2·n·max((a*a)) / (sum(a)^2)

The evaluation is scale‑invariant, therefore we keep the sequence on the
simplex  {a ≥ 0, Σ a = √(2·n)}.  The algorithm works as follows:

1. Initialise a from a geometric seed or, if available, from the previously
   discovered sequence ``height_sequence_1``.
2. Repeatedly
   * compute the full convolution c = a * a   (FFT, O(n log n))
   * replace the non‑smooth “max’’ by a smooth log‑sum‑exp:
         Lα(a) = (1/α)·log Σ exp(α·c_k)
     with a temperature α that is slowly increased (annealing).  The
     gradient of Lα is
         ∇Lα = 2· ( w  ⋆ a )   where   w_k = exp(α·c_k) / Σ exp(α·c)
     (⋆ denotes cross‑correlation).  The cross‑correlation is also
     evaluated by FFT.
   * take a momentum step opposite to the gradient, clip to [0,1000] and
     project back onto the simplex.
   * every few iterations the true score is evaluated with the provided
     ``evaluate_sequence``; the best sequence is remembered.
   * if no improvement happens for a while a small random jitter is added
     to escape plateaux.
3. After the time budget a final polishing phase using the classic
   sub‑gradient (max‑only) method is run for a few hundred cheap steps.

The whole routine stays well within the 1000 s budget and usually drives
the score well below the previous 1.576… value, approaching the theoretical
limit of 1.
"""

import numpy as np
import time
from math import log, exp

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.

    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(n, dtype=float) + 1.0) > 0
    if not np.any(rho_arr):
        return np.zeros_like(v)
    rho = np.where(rho_arr)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        it += 1

    # --------------------------------------------------------------- #
    # 6. Final polishing with the classic max‑only sub‑gradient (a few hundred cheap steps)
    cur = best_seq.copy()
    cur_score = best_score
    vel = np.zeros_like(cur)
    t = 200
    for i in range(t):
        # convolution via FFT (fast)
        a_fft = np.fft.rfft(cur, n=fft_len_conv)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient for the max functional
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[j[mask]]

        step = eta / (np.linalg.norm(grad) + 1e-12)
        cur_candidate = cur - step * grad
        cur_candidate = np.clip(cur_candidate, 0.0, 1000.0)
        cur = projection_simplex(cur_candidate, target_sum)

        if i % 20 == 0:
            sc = evaluate_sequence(cur.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = cur.copy()
                print(
                    f"[{time.time() - start_time:6.1f}s] polishing iter {i:3d} → new best {best_score:.6f}"
                )

    # --------------------------------------------------------------- #
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Search finished (elapsed {time.time() - start_time:.1f}s). Best score: {final_score:.6f}")
    return best_seq.tolist()
