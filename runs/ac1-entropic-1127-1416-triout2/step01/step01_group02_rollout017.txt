"""
Propose a sequence of non‑negative step heights that beats the simple
central‑binomial construction.

The idea follows the “alternating LP” method mentioned in the
statement:

1.  Start from the canonical binomial (central) sequence
    c_k = (2k choose k)/4^k – this sequence gives the classic value
    ≈ π/2.

2.  For the current sequence `f` compute its maximal auto‑convolution
    value `M = max(f*f)`.

3.  Solve a linear programme
       maximise   ∑ g_j
       subject to  (f * g)_i ≤ M   for all i
                   g_j ≥ 0
   The convolution (f*g)_i is linear in the unknown vector `g`, thus the
   LP is a standard linear programme.  The optimal `g` places as much
   mass as possible on positions where `f` is small, thereby increasing
   the ℓ¹–norm of the step function without raising the bound `M`.

4.  Rescale `g` to have the same ℓ¹–norm as `f` and mix the two
    sequences:
        f ← (1‑t)·f + t·g          (t ≈ 0.2)

    The mixing step reduces the maximal convolution value
    `max(f*f)` (the bound used in the evaluation function).

5.  Repeat steps 2‑4 a few times (while time permits).  Keep the best
    sequence seen.

All operations respect the non‑negativity requirement and the final
return value is a plain Python list that can be fed to
`evaluate_sequence`.  The routine adapts the number of iterations to
the provided time budget (default 1000 s) and falls back to the plain
binomial construction if anything goes wrong.
"""

import time
import numpy as np

# optional: cvxpy may be absent on some judges – we import lazily.
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:                       # pragma: no cover
    _CVXPY_AVAILABLE = False


def _binomial_sequence(length: int) -> np.ndarray:
    """Central binomial coefficients: c_k = (2k choose k) / 4**k."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise   1ᵀg   subject to   conv(f,g) ≤ M,   g ≥ 0.

    Parameters
    ----------
    f : np.ndarray
        Current step‑function (non‑negative, length n).
    M : float
        Upper bound for the convolution (the max of f*f).

    Returns
    -------
    np.ndarray or None
        Optimal `g` with the same length as `f`, or ``None`` on failure.
    """
    n = f.shape[0]
    # Build the convolution matrix A where (A @ g)_i = (f * g)_i.
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(rows):
        lo = max(0, i - (n - 1))
        hi = min(i, n - 1)
        # f[ i - j ] multiplies g[j] for j in [lo, hi]
        A[i, lo:hi + 1] = f[i - np.arange(lo, hi + 1)]

    # Linear programme
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-9,
                   reltol=1e-9, feastol=1e-9)
    except Exception:                     # pragma: no cover
        return None

    if g.value is None:
        return None
    # Numerical noise may produce tiny negative values – clip.
    g_opt = np.maximum(g.value, 0.0)
    return g_opt


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a step‑function height sequence that (heuristically) minimises
    ``evaluate_sequence``.  The algorithm starts from the central‑binomial
    construction and repeatedly solves a linear programme to move mass
    towards positions where the current sequence has small values.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # a tiny safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  A few hundred entries give a tractable LP;
    # we use the length of the previously discovered sequence if it
    # exists, otherwise a default of 500.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        n = max(200, min(n, 800))           # keep it moderate for the LP
    except Exception:
        n = 500

    # ------------------------------------------------------------------
    # Initial candidate – the binomial (central) sequence.
    f = _binomial_sequence(n)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop.  Each iteration:
    #   * compute current max convolution M,
    #   * solve the LP to obtain a mass‑rich g,
    #   * mix g into f,
    #   * keep the best sequence seen.
    # The loop stops when time runs out or after a fixed number of
    # iterations (whichever comes first).
    iteration = 0
    t_mix = 0.2               # mixing weight
    while time.time() < deadline and iteration < 30:
        iteration += 1

        # 1) current bound M
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        # 2) solve LP – skip if cvxpy unavailable
        if not _CVXPY_AVAILABLE:
            print("[skip] cvxpy not available – aborting LP loop.")
            break
        g = _solve_lp(f, M)
        if g is None:
            print("[lp] LP failed – exiting loop.")
            break

        # 3) bring g to the same ℓ¹‑norm as f (so that mixing does not
        #    alter the total mass too dramatically)
        sum_f = float(f.sum())
        sum_g = float(g.sum())
        if sum_g == 0:
            # nothing useful was found – exit early
            print("[lp] LP returned zero vector.")
            break
        g_scaled = g * (sum_f / sum_g)

        # 4) convex combination
        f_new = (1.0 - t_mix) * f + t_mix * g_scaled

        # Enforce non‑negativity and the clipping performed in
        # `evaluate_sequence` (they are cheap and keep the candidate
        # valid even under rounding).
        f_new = np.clip(f_new, 0.0, 1000.0)

        # 5) evaluate
        score = evaluate_sequence(f_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = f_new.copy()
            print(f"[iter {iteration}] new best score={best_score:.6f}")

        # Accept the new point for the next iteration – even if it does
        # not improve, because the LP tends to increase the ℓ¹‑norm and
        # hence lower the evaluation in the long run.
        f = f_new

        # Occasionally inject a tiny random perturbation to escape tiny
        # plateaus (cost is negligible).
        if np.random.rand() < 0.1:
            i, j = np.random.randint(0, n, size=2)
            if i != j and f[i] > 1e-9:
                amount = 0.01 * np.random.rand() * f[i]
                f[i] -= amount
                f[j] += amount
                f = np.clip(f, 0.0, 1000.0)

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
