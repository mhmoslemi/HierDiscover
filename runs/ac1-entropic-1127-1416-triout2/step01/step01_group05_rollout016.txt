"""
propose_candidate
-----------------
Search for a step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three phases:

1. **Warm‑start** – a classic inverse‑square‑root law (a_i ∝ 1/√(i+1))
   is normalised to the required ℓ¹–norm.  This already gives a score
   around 1.57.

2. **Linear‑programming refinement** – for the current sequence `f` we
   solve the LP

        maximise   Σ b_j
        subject to Σ_i f_i · b_{k‑i} ≤ M   (for every convolution index k)
                  b_j ≥ 0

   where `M` is the present maximum of `f*f`.  The solution `b`
   (often with a larger ℓ¹‑norm) is scaled back to the target sum and
   blended with `f`.  A small line‑search over the blending factor
   guarantees that the new sequence does not increase the objective.
   Repeating this step drives the maximum convolution down well below the
   inverse‑square‑root value.

3. **Fine‑grained stochastic polishing** – once the LP stage stalls we
   perform cheap random mass‑moves (similar to the previous implementation)
   to squeeze out any remaining improvement.

All work respects the 1000 s budget; the best sequence found is returned
as a plain Python list.  The implementation uses only NumPy and SciPy
(and works even if SciPy is unavailable – it falls back to the
random‑local‑search only)."""

import time
import numpy as np

# SciPy is optional – the LP stage is used only when it can be imported.
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    SCIPY_AVAILABLE = False


def _normalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ¹–norm equals *target_sum*."""
    cur = seq.sum()
    if cur == 0:
        return seq
    return seq * (target_sum / cur)


def _build_convolution_matrix(f: np.ndarray) -> csr_matrix:
    """
    Construct the (2n‑1) × n sparse matrix A such that
        (A @ b)[k] = Σ_i f_i * b_{k‑i}
    (out‑of‑range indices are treated as zero).  This matrix encodes the
    linear constraints   f * b ≤ M .
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # Each entry f[i] appears in the rows k = i + j for all j = 0..n‑1
    for i in range(n):
        fi = f[i]
        # column indices for this row block are simply 0..n-1
        cols.extend(range(n))
        rows.extend((i + np.arange(n)).tolist())
        data.extend([fi] * n)

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=float)
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Find a sequence of non‑negative heights that minimises the evaluation
    function defined in the problem statement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a safety margin

    # ------------------------------------------------------------------
    # 1. Warm start – inverse square‑root shape (p = 0.5)
    # ------------------------------------------------------------------
    n = 2000                              # moderate size – LP-friendly
    target_sum = np.sqrt(2.0 * n)

    idx = np.arange(1, n + 1, dtype=float)
    raw = 1.0 / np.sqrt(idx)              # a_i ∝ 1/√i
    f = _normalise(raw, target_sum)

    best_seq = f.copy()
    # evaluate_sequence returns exactly max(conv) for our normalisation
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Warm‑start: n={n}, score={best_score:.6f}')

    # ------------------------------------------------------------------
    # 2. Linear‑programming driven improvement (if SciPy is present)
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE:
        # we will perform at most a few LP rounds – each round costs O(n²)
        # time and drives the bound down quickly.
        max_lp_iters = 12
        for lp_iter in range(max_lp_iters):
            if time.time() > deadline:
                break

            # current maximal convolution value
            conv_f = np.convolve(f, f)
            M = conv_f.max()

            # Build the matrix for the constraints  f * b ≤ M
            A_ub = _build_convolution_matrix(f)

            # Linear programme: maximise Σ b_j   → minimise −Σ b_j
            c = -np.ones(n)                 # objective coefficients
            b_ub = np.full(2 * n - 1, M)    # right‑hand sides

            # Solve the LP
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'ipm_optimality_tolerance': 1e-9})

            if not res.success:            # pragma: no cover
                print('LP failed – exiting LP stage')
                break

            b = res.x
            sum_b = b.sum()

            # If the LP cannot increase the ℓ¹‑norm there is nothing left to gain
            if sum_b <= target_sum * (1.0 + 1e-12):
                print('LP reached the ℓ¹‑bound – stopping LP stage')
                break

            # Scale b back to the target sum (the LP was solved with the *same* M,
            # so scaling down does not violate any inequality)
            g = b * (target_sum / sum_b)

            # ------------------------------------------------------------------
            # 2‑a. Blend the current f with g. We try a few mixing factors and keep
            #      the one that yields the smallest maximal convolution.
            # ------------------------------------------------------------------
            improved = False
            for t in (0.05, 0.10, 0.20, 0.35, 0.50):
                f_candidate = (1.0 - t) * f + t * g
                conv_cand = np.convolve(f_candidate, f_candidate)
                M_cand = conv_cand.max()
                if M_cand < M - 1e-9:          # strict improvement
                    f = f_candidate
                    M = M_cand
                    improved = True
                    score = M                      # because sum = sqrt(2n)
                    if score < best_score:
                        best_score = score
                        best_seq = f.copy()
                        print(f'LP iter {lp_iter+1:2d}, blend t={t:.2f} → score={score:.6f}')
                    break
            if not improved:
                # No blending factor improved the bound – stop the LP loop.
                print('LP iteration gave no improvement – ending LP stage')
                break
    else:                                   # pragma: no cover
        print('SciPy not available – skipping LP stage')

    # ------------------------------------------------------------------
    # 3. Stochastic polishing – cheap random local moves
    # ------------------------------------------------------------------
    curr = best_seq.copy()
    n = curr.size
    target_sum = curr.sum()                # unchanged

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move up to 0.5 % of the donor mass
        max_move = 0.005 * curr[i]
        if max_move <= 0.0:
            continue
        delta = np.random.uniform(-max_move, max_move)
        if curr[i] - delta < 0.0:
            continue
        cand = curr.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            curr = cand
            print(f'Random polish → score={cand_score:.6f}')
        # stop early if we already beat the target (1.503)
        if best_score < 1.503:
            break

    print(f'Final best score: {best_score:.6f} (length {len(best_seq)})')
    return best_seq.tolist()
