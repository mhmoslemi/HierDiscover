def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that minimises

    F(a) = 2·n·max_k (a * a)[k]          (sum(a) is normalised to 1)

using a *smooth* approximation of the max‑operator and a simple
multiplicative (exponential) mirror‑descent update.  The algorithm works
as follows:

1.  Initialise the vector `a` with the classic central‑binomial sequence
    (the “constant‑convolution” construction) or, if a previous best
    sequence is present, with that one.  The vector is normalised so that
    `sum(a) = 1`.

2.  At each iteration:
    * compute the full convolution `c = a * a`;
    * form a soft‑max distribution `p_k ∝ exp( β·(c_k‑max(c)) )`.  
      For `β → ∞` this converges to a point mass on the maximal index,
      while for small `β` it yields an almost uniform weight.
    * the gradient of the smoothed objective `J = (1/β)·log Σ exp(β·c_k)` is  

          ∇J_i = 2·Σ_k p_k·a_{k‑i}  (indices out of range are ignored)

      which can be evaluated efficiently as a cross‑correlation
      `2·np.correlate(p, a, mode='valid')`.
    * take a multiplicative mirror‑descent step  

          a_i ← a_i·exp(‑η·∇J_i)

      and re‑normalise to keep `sum(a)=1`.  The step size η decays as
      `η = η₀ / sqrt(t+1)`.

3.  The smoothing parameter β is increased slowly (geometric schedule)
    so that the optimisation first moves in a smooth landscape and then
    focuses on the true max‑operator.

4.  The best vector encountered (according to the original
    `evaluate_sequence`) is stored.  Approximately half of the allotted
    time is spent from the initial point; if no improvement is seen
    afterwards, the algorithm restarts from a random uniform vector to
    escape possible local minima.

The method stays within the 1000 s budget, uses only NumPy, and returns
a Python list which can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _constant_conv_base(n: int) -> np.ndarray:
    """
    First n terms of a_k = binom(2k, k) / 4**k.
    Recurrence: a_0 = 1, a_k = a_{k-1} * (1 - 1/(2k)).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection only defined for 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) with a low
    evaluation_score.  Returns the best sequence (list of floats) found
    before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # 0.5 s safety margin
    half_deadline = start_time + budget_s * 0.5

    # --------------------------------------------------------------
    # Choose the length n (reuse previous best if available)

    if "height_sequence_1" in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
        a = np.asarray(height_sequence_1, dtype=np.float64)

    conv = np.convolve(a, a)
    best_score = float(2 * n * conv.max())

    # --------------------------------------------------------------
    eta0 = 5.0                  # initial step‑size for the mirror update
    beta = 1.0                  # smoothing for the soft‑max
    beta_max = 300.0
    beta_factor = 1.001         # geometric increase of beta
    beta_update_freq = 1000     # update beta every this many iterations
    restart_done = False        # whether we have performed the random restart
    iteration = 0

    while time.time() < deadline:

        # ----- evaluate current objective (quick, no evaluate_sequence call)
        conv = np.convolve(a, a)
        cur_score = 2.0 * n * conv.max()

        shift = conv - conv.max()

        grad = 2.0 * np.correlate(p, a, mode='valid')

        # ----- mirror‑descent (multiplicative) update
        eta = eta0 / np.sqrt(iteration)          # decaying step size
        # multiplicative step – guaranteed to stay non‑negative
        a = a * np.exp(-eta * grad)

        # ----- renormalise to the simplex
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            # very unlikely, but fall back to a uniform distribution
            a = np.full(n, 1.0 / n)
        else:

        if iteration % 5000 == 0:

            a = a / a.sum()

        # ----- increase the smoothing parameter β gradually
        if iteration % beta_update_freq == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # ----- random restart after half the time budget if we are stuck
        if (not restart_done) and (time.time() > half_deadline):
            # restart from a fresh uniform vector
            a = np.random.rand(n)
            a = a / a.sum()
            restart_done = True
            print(f"Iter {iteration:7d}: performed random restart")

    # --------------------------------------------------------------
    # Final sanity check with the official evaluator
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final score reported by evaluate_sequence: {final_score:.6f}")
        # use the official score for return if it is better (it should be identical)
        best_score = min(best_score, final_score)
    except Exception:
        # If evaluate_sequence is unavailable, we keep the internal score
        pass

    print(f"Search finished after {iteration} iterations, best score {best_score:.6f}")
    return best_seq.tolist()
