def fast_score(seq):
    n = len(seq)
    S = seq.sum()
    if S < 0.01:
        return np.inf
    # ensure within the allowed range
    # may clip but we will enforce during generation
    # Use FFT:
    m = 1 << ((2*n-1).bit_length())
    F = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(F*F, n=m)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (S**2)
```

def fast_score(seq):
    seq = np.asarray(seq, dtype=float)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(F * F, n=m)[:2 * n - 1]
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (s * s))
```

This implementation searches for a low‑score non‑negative step
function by (i) initializing from a geometric seed (or the previous
best if available), (ii) a fast simulated‑annealing on a low‑dimensional
block representation, (iii) a final projected‑gradient polishing stage.
All heavy work uses an FFT‑based convolution so that >10⁶ candidate
evaluations fit comfortably into the 1000 s budget.

import numpy as np
import time

def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(seq: np.ndarray) -> float:
    """Return the same value as evaluate_sequence but using FFT."""
    seq = np.asarray(seq, dtype=float)
    # enforce the same clipping performed by evaluate_sequence
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())   # next power of two
    F = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(F * F, n=m)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def blocks_to_sequence(blocks: np.ndarray, lens: np.ndarray) -> np.ndarray:

        pos += l

    # ------------------------------------------------------------------ #
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score {final_score:.6f}")
    return best.tolist()

            best_score = cur_score
            best = curr.copy()
            print(f"[{time.time() - start:6.1f}s] GD new best {best_score:.6f}")
            step_factor = max(step_factor * 0.9, min_step)

        # sub‑gradient of the max‑functional
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(curr)
        grad[mask] = 2.0 * curr[j[mask]]

        # step and projection onto the simplex Σx = target_sum
        step = step_factor * target_sum / n
        norm = np.linalg.norm(grad)
        if norm == 0:
            direction = np.random.randn(n)
        else:
            direction = grad / norm
        tentative = curr - step * direction
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # occasional jitter to avoid tiny local minima
        if np.random.rand() < 0.015:
            jitter = np.random.randn(n) * step * 0.4
            curr = projection_simplex(np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        it += 1
        if it % decay_interval == 0:
            step_factor = max(step_factor * decay, min_step)

    # --------------------------------------------------------------- #
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score {final_score:.6f}")
    return best.tolist()
