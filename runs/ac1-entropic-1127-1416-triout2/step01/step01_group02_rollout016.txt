def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import time, numpy as np, math, random

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    try:
        n = len(height_sequence_1)
    except Exception:
        n = 5000

    best_seq = None
    best_score = float('inf')
    best_p = None

    # coarse grid
    for p in np.linspace(0.55, 0.95, 9):
        seq = (np.arange(1, n+1, dtype=np.float64) ** (-p))
        seq /= seq.sum()
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score; best_seq = seq.copy(); best_p = p
            print(f"[grid] p={p:.3f} score={score:.6f}")

        if time.time() > deadline: break

    # fine grid around best
    if best_p is not None:
        low = max(0.55, best_p - 0.05)
        high = min(0.95, best_p + 0.05)
        for p in np.linspace(low, high, 21):
            if time.time() > deadline: break
            seq = (np.arange(1, n+1, dtype=np.float64) ** (-p))
            seq /= seq.sum()
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score; best_seq = seq.copy(); best_p = p
                print(f"[refine] p={p:.4f} score={score:.6f}")

    # optional local random perturbations
    cur_seq = best_seq.copy()
    cur_score = best_score
    step_frac = 0.02
    decay = 0.9995
    while time.time() < deadline:
        # propose random pair transfer
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        amount = step_frac * np.random.rand() * cur_seq[i]
        cand_seq = cur_seq.copy()
        cand_seq[i] -= amount
        cand_seq[j] += amount
        cand_seq = np.clip(cand_seq, 0, None)
        cand_seq /= cand_seq.sum()
        cand_score = evaluate_sequence(cand_seq.tolist())
        if cand_score < cur_score or np.random.rand() < 0.001:
            cur_seq = cand_seq
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand_seq
                best_score = cand_score
                print(f"[improve] new best score={best_score:.6f}")

        step_frac *= decay
        if np.random.rand() < 0.001:
            cur_seq = best_seq.copy()
            cur_score = best_score
            step_frac = 0.02

    print(f"[final] best score={best_score:.6f}, best exponent p={best_p:.6f}")
    return best_seq.tolist()

"""
propose_candidate
-----------------
Search for a step‑height sequence that beats the classic “binomial”
construction.  The method works in three stages:

1. **Start** from the analytically optimal binomial (central
   binomial) coefficients and normalise the sum to 1 (the evaluation
   function is scale‑invariant, a unit‑sum normalisation is convenient).

2. **Coarse / fine scan over a power‑law exponent** – a family
   a_i ∝ (i+1)^{-p} – to locate a better global shape.  For each p we
   normalise the vector and compute the exact evaluation score using
   an FFT based convolution (O(n log n)).  The best exponent found in
   this cheap grid search is kept as a warm start.

3. **Targeted mass‑transfer hill‑climbing.**  The current maximal
   convolution entry (or a randomly chosen one among the ties) is
   identified.  Its contribution is the element‑wise product
   a_i·a_{k‑i}.  We repeatedly move a tiny fraction of mass from an
   index with a large contribution to an index with a small contribution,
   re‑normalise, and accept the move only if the maximal convolution
   value (hence the evaluation score) does not increase.  Step size
   decays slowly; occasional restarts to the best ever sequence keep the
   search from getting stuck.

All operations respect the non‑negative, sum‑greater‑than‑0.01
requirements of ``evaluate_sequence``.  The routine runs for at most the
given time budget (default 1000 s) and finally returns the best list
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Central binomial coefficients c_k = (2k choose k) / 4**k."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def powerlaw_sequence(length: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, length + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT (efficient for the sizes we need).
    Returns the full convolution of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    # next power of two for speed
    L = 1 << (int(np.ceil(np.log2(n))))
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[:n]


# ----------------------------------------------------------------------
def score_of(seq: np.ndarray) -> float:
    """
    Exact evaluation used by the competition (same as evaluate_sequence
    but a little faster – no safety clipping because our sequences are
    well‑behaved).  The result equals
        2 * n * max(conv(seq, seq)) / (sum(seq) ** 2)
    """
    n = len(seq)
    b = convolve_fft(seq, seq)
    max_b = b.max()
    s = seq.sum()
    # the problem definition guarantees sum > 0.01 for any admissible seq
    return float(2.0 * n * max_b / (s * s))

    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    best_score = score_of(best_seq)
    print(f"[init] length={n}, binomial score={best_score:.6f}")

    # coarse scan

            best_score = cand_score

            print(f"[grid] p={p:.3f} → score={cand_score:.6f}")
    # fine scan around the best exponent found

    if best_seq is not None:

        # – we simply reuse the last p that gave an improvement
        #   (otherwise centre on 0.7 which is a typical good value)
        centre = p if 'p' in locals() else 0.70

    cur_seq = best_seq.copy()

    step_frac = 0.02          # initial fraction of source mass to move
    decay = 0.9995            # slowly shrink the step size
    improvement_counter = 0

        # compute full convolution once per iteration
        b = convolve_fft(cur_seq, cur_seq)
        max_b = b.max()
        # if the new max is already better than the best, record it
        cur_score = 2.0 * n * max_b / (cur_seq.sum() ** 2)

        # pick a shift achieving the maximum (random among ties)
        max_positions = np.flatnonzero(b == max_b)
        k_max = np.random.choice(max_positions)

        # contribution of each index to this particular entry
        # c_i = a_i * a_{k_max-i}   (zero when the partner index is out‑of‑range)
        partner = k_max - np.arange(n)          # partner indices for all i
        valid = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(cur_seq)
        contrib[valid] = cur_seq[valid] * cur_seq[partner[valid]]

        # source indices – bias towards large contribution
        if contrib.sum() == 0:
            source_idx = np.random.randint(0, n)
        else:
            prob_src = contrib / contrib.sum()
            source_idx = np.random.choice(n, p=prob_src)

        # target indices – bias towards small contribution (prefer edges)
        # we take the 10 % of indices with smallest contribution
        low_cut = np.percentile(contrib, 10)
        low_candidates = np.where(contrib <= low_cut)[0]
        if low_candidates.size == 0:
            target_idx = np.random.randint(0, n)
        else:
            target_idx = np.random.choice(low_candidates)

        if source_idx == target_idx:
            continue

        # amount to move – a modest fraction of the source entry
        amount = step_frac * np.random.rand() * cur_seq[source_idx]
        if amount == 0.0:
            continue

        cand_seq = cur_seq.copy()
        cand_seq[source_idx] -= amount
        cand_seq[target_idx] += amount

        # enforce non‑negativity and renormalise to keep the sum = 1
        np.maximum(cand_seq, 0, out=cand_seq)
        cand_seq /= cand_seq.sum()

        # evaluate candidate
        cand_score = score_of(cand_seq)

        if cand_score < cur_score or np.random.rand() < 1e-4:

            cur_score = cand_score
            if cand_score < best_score:

                best_seq = cand_seq.copy()
                improvement_counter += 1

                      f"score={best_score:.6f} time={time.time() - (deadline - budget_s):.2f}s")

        # linearly decay the step size
        step_frac *= decay

        # occasional restart to the best‑so‑far configuration
        if np.random.rand() < 0.001:
            cur_seq = best_seq.copy()
            cur_score = best_score
            step_frac = 0.02

    print(f"[final] best score={best_score:.6f}")
    # Return a plain Python list – the external framework expects a list.
