def solve_lp(a, M):
    n = len(a)

    a_const = cp.Constant(a)    # constant

"""
propose_candidate
-----------------
Search for a low‑max‑convolution step‑function.

The program starts from a simple inverse‑square‑root shape (or from the
previous best construction `height_sequence_1`).  It then repeatedly solves a
linear program

    maximise   Σ x_j
    subject to  (a * x)_k ≤ M    for all k
                x_j ≥ 0,

where * is the discrete convolution, a is the current sequence and
M = max(a*a).  This LP finds a feasible sequence x with a larger L1‑norm
while respecting the same convolution bound.  After rescaling x to the
required total mass √(2n) we mix it with a (convex combination) – the
mixed sequence has a strictly smaller max‑convolution.  The process is
repeated until no further improvement is observed.

When the time budget allows, a short random‑local‑move phase is added to
fine‑tune the result.  The whole routine runs inside the given time budget
(≤ 1000 s) and returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# cvxpy is used for the linear program.  If it cannot be imported we fall
# back to a simple random‑search (the previous algorithm).
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:                # pragma: no cover
    _CVXPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _powers_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Raw (un‑normalised) sequence a_i = (i+1+offset)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    return 1.0 / np.power(idx, p)


def _normalise_to_target(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative sequence so that its sum equals target_sum."""
    cur = seq.sum()
    if cur == 0.0:
        return seq
    return seq * (target_sum / cur)


def _max_convolution(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] (full convolution)."""
    return float(np.max(np.convolve(seq, seq)))


def _solve_lp(a: np.ndarray, M: float):
    """Solve the LP   max Σ x   s.t.   (a * x) ≤ M ,  x ≥ 0.

    Returns the optimiser x (numpy array) or None on failure.
    """
    n = a.shape[0]
    if not _CVXPY_AVAILABLE:          # pragma: no cover
        return None

    # Decision variable
    x = cp.Variable(n)

    # Constant version of a
    a_const = cp.Constant(a)

    # Linear constraints: the convolution of a with x must stay ≤ M
    constraints = [x >= 0, cp.conv(a_const, x) <= M]

    # Linear objective
    prob = cp.Problem(cp.Maximize(cp.sum(x)), constraints)

    # Try GLPK first (pure LP), fall back to ECOS if unavailable
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, glpk={})
    except Exception:                # pragma: no cover
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception as e:        # pragma: no cover
            print("LP solve failed:", e)
            return None

    if x.value is None:
        return None

    # Numerical noise may produce tiny negatives – clip them
    return np.maximum(0.0, np.asarray(x.value, dtype=np.float64))


def _mix_update(a: np.ndarray, g: np.ndarray, M_curr: float,
                t_initial: float = 0.5):
    """
    Return a new mixed sequence a_new = (1‑t)·a + t·g that does not increase
    the max‑convolution.  The function tries decreasing t (halving) up to
    five times; if no improvement is found the original a is returned.
    """
    t = t_initial
    for _ in range(5):
        a_new = (1.0 - t) * a + t * g
        M_new = _max_convolution(a_new)
        # accept non‑worsening mixtures (tiny tolerance for numerical noise)
        if M_new <= M_curr * (1.0 + 1e-12):
            return a_new, M_new, True
        t *= 0.5
    return a, M_curr, False


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop a few seconds earlier).

    Returns
    -------
    list[float]
        The best sequence found (list of non‑negative heights).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin for clean‑up

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1. Choose a few lengths; larger n gives more freedom but costs more
    #    time per LP.  We try them from large to small so that the best
    #    possible (most flexible) construction is attempted first.
    # ------------------------------------------------------------------
    lengths = [5000, 4000, 3000, 2500, 2000]

    for n in lengths:
        if time.time() > deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ----------------------------------------------------------------
        # 2. Initialise the sequence.
        #    Prefer the previously discovered construction (global variable)
        #    if it has the right length, otherwise fall back to the analytic
        #    inverse‑square‑root shape.
        # ----------------------------------------------------------------
        if 'height_sequence_1' in globals():
            cand = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if cand.shape[0] == n:
                a = _normalise_to_target(cand, target_sum)
            else:
                a = _normalise_to_target(_powers_sequence(n, p=0.5, offset=0.0),
                                         target_sum)
        else:
            a = _normalise_to_target(_powers_sequence(n, p=0.5, offset=0.0),
                                     target_sum)

        # initial score
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"[{n}] initial score = {score:.6f}")

        # ----------------------------------------------------------------
        # 3. Main LP‑based refinement loop.
        # ----------------------------------------------------------------
        M = _max_convolution(a)            # current max‑convolution
        iter_cnt = 0
        max_iters = 30                     # hard limit per length

        while time.time() < deadline and iter_cnt < max_iters:
            x = _solve_lp(a, M)
            if x is None:
                break

            sum_x = x.sum()
            if sum_x <= 0.0:
                break
            # Rescale to keep the total mass fixed
            g = x * (target_sum / sum_x)

            # Mix a and g, accepting only if the max‑convolution does not grow
            a_new, M_new, ok = _mix_update(a, g, M, t_initial=0.5)
            if not ok:
                # No improvement possible with this a – exit the LP loop
                break

            a = a_new
            M = M_new
            score = evaluate_sequence(a.tolist())
            iter_cnt += 1

            if score < best_score - 1e-9:
                best_score = score
                best_seq = a.tolist()
                print(f"[{n}] iter {iter_cnt:02d}  score = {score:.6f}  "
                      f"M = {M:.6f}")

            # stop early if improvement becomes negligible
            if abs(M - M_new) < 1e-8:
                break

        # ----------------------------------------------------------------
        # 4. Short random local‑move polishing phase (few seconds).
        # ----------------------------------------------------------------
        if time.time() < deadline:
            local_deadline = time.time() + min(5.0, deadline - time.time())
            n_moves = 0
            while time.time() < local_deadline:
                i, j = np.random.randint(n, size=2)
                if i == j:
                    continue
                max_move = 0.01 * a[i]
                if max_move <= 0.0:
                    continue
                delta = np.random.uniform(-max_move, max_move)
                if a[i] - delta < 0.0:
                    continue
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand_score = evaluate_sequence(cand.tolist())
                if cand_score < best_score - 1e-9:
                    best_score = cand_score
                    best_seq = cand.tolist()
                    a = cand
                    M = _max_convolution(a)
                    print(f"[{n}] random tweak #{n_moves}  score = {cand_score:.6f}")
                n_moves += 1

        # Early exit if we already beat the target bound
        if best_score < 1.503:
            print("Target bound reached – stopping early.")
            break

    # ----------------------------------------------------------------------
    # 5. Safety fallback (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        seq = [math.sqrt(2.0 * n) / n] * n
        best_seq = seq
        best_score = evaluate_sequence(seq)

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
