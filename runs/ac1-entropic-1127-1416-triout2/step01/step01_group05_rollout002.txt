def proj_simplex(v):
    v = v.copy()
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function by directly minimizing the
sup‑norm of the self‑convolution.

The algorithm works on a fixed length `n` (default 5000) and keeps the
coefficients on the probability simplex (non‑negative, sum = 1).  The
objective to minimise is

    2 * n * max(conv(a, a))

because the denominator (sum(a)^2) is fixed to 1.  This is achieved by a
projected sub‑gradient descent:

1. Start from the analytically good shape a_i ∝ 1/√(i+1) and normalise.
2. Repeatedly
   * compute the full convolution using an FFT,
   * locate the shift `k` where the convolution is maximal,
   * take the sub‑gradient of that maximal entry (which is simply the
     reversed vector a shifted by `k`),
   * move a a little in the opposite direction,
   * project back onto the simplex.
3. Occasionally perform a tiny random redistribution of mass to escape
   shallow plateaux.
4. Keep the best sequence seen during the run.

The whole search respects the given time budget (default 1000 s) and
returns the best list of heights found.  If something goes wrong the
previous construction stored in the global `height_sequence_1` is used
as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x >= 0, sum(x) = 1 }.
    Returns a new array (the input is not modified).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete convolution a * a using an FFT.
    Length of the result is 2*len(a) - 1.
    """
    n = a.shape[0]
    # length for FFT (next power of two for speed)
    size = 2 * n
    fft_len = 1 << (size - 1).bit_length()
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    # keep exactly the needed length
    return conv[: 2 * n - 1]


def gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (at shift k)
    with respect to the coefficient vector a.
    For each i that participates in the sum for that shift,
    the gradient entry is a[k-i]; otherwise it is zero.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, k - n + 1)
    high = min(k, n - 1)
    if low > high:
        return g
    idx = np.arange(low, high + 1, dtype=int)
    g[idx] = a[k - idx]
    return g


def objective_score(a: np.ndarray) -> float:
    """
    Compute 2 * n * max(conv(a, a)) for a vector that already satisfies
    sum(a) == 1.  This matches the value returned by evaluate_sequence.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * np.max(conv)

# Main search routine
# ----------------------------------------------------------------------

    """

    iteration = 0
