"""
Search for a very low‑score step function.

The construction consists of two parts:

* a prefix of length `m` that follows the optimal 1/√i decay
  (the unique shape for which the self‑convolution stays bounded while
  the ℓ₁‑norm grows like √n);
* a long tail of constant height `ε`.  The height is chosen so that the
  convolution term arising from mixing the prefix and the tail does not
  exceed the maximal convolution generated by the prefix alone.  This
  yields a total ℓ₁‑norm larger than the pure 1/√i sequence while keeping
  the maximal convolution unchanged, lowering the evaluation score to
  below 1.0.

We sweep a few problem sizes `n` and a range of prefix lengths `m`,
compute the corresponding `ε`, evaluate the candidate with the provided
`evaluate_sequence`, and keep the best result.  Finally a short stochastic
hill‑climbing phase tries to improve the best candidate slightly.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (identical to the previous baseline for safety)
# ----------------------------------------------------------------------
def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:               # pragma: no cover
        return math.inf

def _normalize(seq):
    """Scale so that max(seq) == 1 (no‑op if all zeros)."""
    arr = np.asarray(seq, dtype=float)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()

# ----------------------------------------------------------------------
# Core construction: prefix 1/sqrt(i+1) + constant tail
# ----------------------------------------------------------------------
def _make_candidate(n, m):
    """
    Build a candidate of total length `n` with a prefix of length `m`
    (m >= n/2) using a_i = 1/sqrt(i+1) and a constant tail.
    The tail height ε is chosen as π / sum(prefix[:t]) (t = n-m) so that
    the cross‑convolution term does not exceed the core's maximal value.
    """
    # Prefix (power‑law with exponent 0.5)
    core = 1.0 / np.sqrt(np.arange(1, m + 1, dtype=float))
    t = n - m                     # tail length, must be >0
    if t <= 0:
        return None
    # sum of the first `t` entries of the core (these are the ones that
    # meet the tail in the convolution)
    prefix_sum = core[:t].sum()
    if prefix_sum <= 0:
        return None
    # tail height ε ; ensures (core[:t] * ε) ≤ π
    epsilon = math.pi / prefix_sum
    if epsilon > 1.0:
        # cannot happen for the ranges we use, but guard against it
        epsilon = 1.0
    tail = np.full(t, epsilon, dtype=float)
    seq = np.concatenate([core, tail])
    # safety: clip to [0,1] and renormalise max to 1 (the shape already
    # satisfies this, but numerical noise may appear)
    seq = np.clip(seq, 0.0, 1.0)
    seq = _normalize(seq)        # keep max exactly 1
    return seq

# ----------------------------------------------------------------------
# Stochastic refinement (tiny multiplicative perturbations)
# ----------------------------------------------------------------------
def _perturb(seq, scale=0.05):
    """Apply a small random multiplicative perturbation to a few entries."""
    arr = np.array(seq, copy=True)
    n = len(arr)
    # modify up to 5 positions
    m = random.randint(1, min(5, n))
    idxs = random.sample(range(n), m)
    for i in idxs:
        factor = 1.0 + random.gauss(0.0, scale)
        arr[i] = max(0.0, min(1.0, arr[i] * factor))
    # renormalise maximum to 1
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()

# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The routine spends roughly ``budget_s`` seconds
    searching; it builds analytically‑motivated two‑part sequences,
    evaluates them, and finally refines the best one with a brief
    hill‑climbing phase.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.3

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 0) start from any externally supplied reference sequence
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = _normalize(list(height_sequence_1))
            sc = _score(init)
            if sc < best_score:
                best_seq, best_score = init, sc
                print(f"Initial reference score: {best_score:.6f}")
        except Exception:               # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1) systematic exploration of the two‑part construction
    # ------------------------------------------------------------------
    # lengths we will try (a few thousand entries give a good trade‑off)
    lengths = [1800, 2400, 3000, 3600, 4200, 4800]

    # fraction of the sequence used for the √‑decay prefix
    # optimal continuous value is α = 16/(π²+16) ≈ 0.6185,
    # we scan a neighbourhood around it.
    frac_candidates = [0.55, 0.58, 0.60, 0.618, 0.64, 0.66, 0.68, 0.70]

    for n in lengths:
        if time.time() >= deadline:
            break
        for α in frac_candidates:
            if time.time() >= deadline:
                break
            m = int(round(α * n))
            # ensure prefix at least as long as the tail (required for the
            # cross‑convolution analysis)
            if m < n // 2:
                m = n // 2
            cand = _make_candidate(n, m)
            if cand is None:
                continue
            sc = _score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand
                print(f"New best (n={n}, m={m}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) brief stochastic hill‑climbing from the best candidate
    # ------------------------------------------------------------------
    while time.time() < deadline:
        if best_seq is None:
            break
        cand = _perturb(best_seq, scale=0.07)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"Hill‑climb improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) final fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 2000
        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
