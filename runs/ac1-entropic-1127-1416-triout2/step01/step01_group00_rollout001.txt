def adam_optimize(initial_a, time_deadline, max_iter=200_000, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8, noise_every=500, noise_scale=1e-4):
    a = project_to_simplex(initial_a.copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    while time.time() < time_deadline and t < max_iter:
        t += 1
        max_val, k_max = max_convolution_and_index(a)
        g = subgradient_of_max_at_index(a, k_max)
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(len(a)) * noise_scale
        a_new = project_to_simplex(a_new)
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
        if t % 5000 == 0:
            print(f"    iter {t:,d} cur_score={cur_score:.6f} best_score={best_score:.6f}")
    return best_a, best_score
```

"""
Propose a low‑score step‑function.

Algorithm
---------
* Work on the simplex (non‑negative entries with total mass 1).  The
  evaluation function is scale‑invariant, so fixing the sum to 1 is safe.
* For each chosen length `n` we try a few simple parametric initial
  shapes (Beta‐densities with different exponents).  These give a good
  starting point close to the known arcsine‑type optimum.
* On each start we run an ADAM‑style projected sub‑gradient optimiser.
  The sub‑gradient is the gradient of the *maximal* entry of the
  self‑convolution, which is cheap to compute (one FFT + a linear scan).
* After the ADAM phase we perform a short projected sub‑gradient
  refinement with a diminishing step size.
* The best sequence seen across all runs is returned.  If a historic
  variable `height_sequence_1` exists it is used as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> tuple[float, int]:
    """Return the maximal value of a*a and the index where it occurs."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (a*a)[k] with respect to a.
    For each i, d/d a_i (a*a)[k] = a_{k-i} if the index is valid,
    and the diagonal term is doubled when k = 2*i.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def compute_score(a: np.ndarray) -> float:
    """Score equivalent to evaluate_sequence (sum is assumed = 1)."""
    n = len(a)
    max_val = float(np.max(convolution_fft(a, a)))
    return 2.0 * n * max_val  # sum(a) == 1 → denominator = 1


# ----------------------------------------------------------------------
# Parametric initial shapes (Beta(alpha,alpha) densities)
# ----------------------------------------------------------------------


def beta_initial(n: int, alpha: float) -> np.ndarray:
    """
    Discrete beta(alpha,alpha) density on n points.
    The raw density is proportional to (x(1-x))^{alpha-1}.
    """
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n  # centre of each bin
    # add a tiny epsilon to avoid exact zero when alpha<1
    eps = 1e-12
    vals = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, alpha - 1.0)
    # normalise to sum 1
    s = np.sum(vals)
    if s == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / s
    return vals


# ----------------------------------------------------------------------
# ADAM‑style optimiser on the max‑convolution objective
# ----------------------------------------------------------------------


def adam_optimize(initial_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 200_000,
                  lr: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  noise_every: int = 1000,
                  noise_scale: float = 5e-5) -> tuple[np.ndarray, float]:
    """
    Projected ADAM optimisation of max(a*a).  The iterates stay on the simplex.
    Returns the best vector seen and its (scale‑invariant) score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # maximal convolution entry and its index
        _, k_max = max_convolution_and_index(a)

        # sub‑gradient at that entry
        g = subgradient_of_max_at_index(a, k_max)

        # ADAM moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional tiny stochastic perturbation
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"      ADAM iter={t:,d} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Short projected sub‑gradient fine‑tuning
# ----------------------------------------------------------------------


def fine_tune(a_start: np.ndarray,
              time_deadline: float,
              max_iter: int = 30_000,
              eta0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent with step η_t = η0/√t.
    Used after the ADAM stage for a final polish.
    """
    a = project_to_simplex(a_start.copy())
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        _, k_max = max_convolution_and_index(a)
        g = subgradient_of_max_at_index(a, k_max)

        eta = eta0 / np.sqrt(t)
        a = project_to_simplex(a - eta * g)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if t % 5000 == 0:
            print(f"      fine‑tune iter={t:,d} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------

    """

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------

    except NameError:

    # ------------------------------------------------------------------
