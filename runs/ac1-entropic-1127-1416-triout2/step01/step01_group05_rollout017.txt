"""
propose_candidate
-----------------
Search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the discrete convolution used in ``evaluate_sequence``.
The algorithm works in three phases:

1. **Initialisation** – start from a theoretically good “inverse‑sqrt”
   (central‑binomial) sequence and optionally from the previously
   discovered ``height_sequence_1``.  The sequence is normalised so that
   ``sum(a) = sqrt(2 * n)``; for a normalised sequence the evaluation value
   is simply ``max(conv(a, a))``.

2. **Sub‑gradient descent** – repeatedly locate an index ``k`` where the
   convolution attains its maximum, compute the sub‑gradient
   ``g_i = 2 * a_{k-i}`` (zero outside the valid range), and move a small
   amount of mass away from the entries that most contribute to the peak.
   After each tentative step the vector is re‑normalised and the new
   peak value is evaluated.  The step size is chosen as a safe fraction of
   the largest admissible move (so that no entry becomes negative).  Any
   move that strictly reduces the peak is accepted.

3. **Fine‑grained random refinement** – after the deterministic phase a
   cheap random local‑move search (transfer a tiny amount of mass between
   two random positions) is performed until the time budget expires or
   the target bound (1.503) is reached.

All heavy work uses ``np.fft`` for an ``O(n log n)`` convolution, allowing
thousands of iterations well within the 1000 s limit.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ------- helper utilities -------------------------------------------------

def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """
    Fast real‑to‑real convolution of ``seq`` with itself using FFT.
    Returns the full convolution of length 2*len(seq)-1.
    Small negative rounding errors are clipped to zero.
    """
    n = seq.shape[0]
    L = 2 * n - 1
    # next power of two for efficient FFT
    nfft = 1 << ((L - 1).bit_length())
    fft = np.fft.rfft(seq, n=nfft)
    prod = fft * fft
    conv = np.fft.irfft(prod, n=nfft)[:L]
    np.maximum(conv, 0, out=conv)   # protect against -0.0… due to rounding
    return conv


def _binomial_coeffs(n: int) -> np.ndarray:
    """
    Return the first ``n`` coefficients of the series
    a_i = C(2i, i) / 4**i  (central binomial / 4^i).
    These coefficients satisfy a_i a_j summed over i+j = k == 1 for every k.
    Recurrence: a_{i+1} = a_i * (2*i+1) / (2*i+2)
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def _normalise_to_target(seq: np.ndarray, target: float) -> np.ndarray:
    """Scale ``seq`` so that its sum equals ``target`` (preserve sign)."""
    cur = seq.sum()
    if cur == 0.0:
        return seq
    return seq * (target / cur)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the current
    best upper bound (≈1.575).  The routine is limited to ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a few seconds buffer for tidy‑up

    best_seq = None
    best_score = float('inf')

    # -----------------------------------------------------------------
    # 1.  Choose a problem size – the previous best used n = 5000.
    #     We try a handful of sizes; the runs are cheap.
    # -----------------------------------------------------------------
    lengths = [3000, 5000, 8000]          # more points → richer search space
    for n in lengths:
        target_sum = np.sqrt(2.0 * n)    # the normalisation used in the definition

        # ---- 1a.  Start from the exact “inverse‑sqrt” (central‑binomial) family
        raw = _binomial_coeffs(n)        # a_i = C(2i,i)/4^i  → max(conv)=1
        a = _normalise_to_target(raw, target_sum)

        # ---- 1b.  If the global variable ``height_sequence_1`` exists,
        #          use it as an alternative start (it is already a good candidate)
        if 'height_sequence_1' in globals():
            hs1 = np.asarray(height_sequence_1, dtype=float)
            if hs1.size == n:
                a_alt = _normalise_to_target(hs1, target_sum)
                # evaluate both starts, keep the better one
                for cand in (a, a_alt):
                    cur_score = _full_convolution(cand).max()
                    if cur_score < best_score:
                        best_score, best_seq, a = float(cur_score), cand.tolist(), cand.copy()
                        print(f'Initial candidate (n={n}) score={best_score:.6f}')
                # continue with the better start already stored in ``a``

        # -----------------------------------------------------------------
        # 2.  Deterministic sub‑gradient descent on the peak of the convolution
        # -----------------------------------------------------------------
        while time.time() < deadline:
            conv = _full_convolution(a)
            peak = conv.max()
            if peak < best_score - 1e-12:
                best_score, best_seq = float(peak), a.tolist()
                print(f'*** New best (grad) n={n}  score={best_score:.6f}')
                if best_score < 1.503:          # target reached – stop early
                    break

            # pick one of the (possibly several) maximal indices at random
            max_idxs = np.flatnonzero(conv == peak)
            k = np.random.choice(max_idxs)

            # sub‑gradient of max(conv) w.r.t. a:

            g = np.zeros_like(a)

            # maximal admissible step that keeps all entries non‑negative
            positive = g > 0
            if not np.any(positive):

            eta_max = np.min(a[positive] / g[positive])

            # try a few decreasing step factors; accept the first improvement
            improved = False
            for factor in (0.9, 0.5, 0.2, 0.1, 0.05):
                eta = eta_max * factor
                cand = a - eta * g
                # clipping guards against tiny numerical negatives
                np.maximum(cand, 0.0, out=cand)

                # re‑normalise to keep the sum fixed (the evaluation uses the
                # normalised sequence, i.e. max(conv) directly)
                cand = _normalise_to_target(cand, target_sum)

                cand_peak = _full_convolution(cand).max()
                if cand_peak < best_score - 1e-12:
                    a = cand
                    best_score, best_seq = float(cand_peak), cand.tolist()
                    improved = True
                    print(f'Iter improve: n={n}  score={best_score:.6f}')
                    break
            if not improved:
                # fallback: tiny random redistribution to avoid stagnation
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = 0.001 * a[i]      # move 0.1 % of a[i]
                    if a[i] - delta >= 0.0:
                        cand = a.copy()
                        cand[i] -= delta
                        cand[j] += delta          # sum unchanged, no re‑normalise needed
                        cand_peak = _full_convolution(cand).max()
                        if cand_peak < best_score - 1e-12:
                            a = cand
                            best_score, best_seq = float(cand_peak), cand.tolist()
                            print(f'random improve: n={n}  score={best_score:.6f}')
                # if even the random move fails we simply continue looping

            # early exit if we already beat the target bound
            if best_score < 1.503:

        if best_score < 1.503:

    # -----------------------------------------------------------------
    if best_seq is None:

        n = 2000
        u = np.sqrt(2.0 * n) / n
        best_seq = [u] * n
        best_score = evaluate_sequence(best_seq)

    n = cur.size
    target_sum = cur.sum()                     # already normalised

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a tiny fraction (up to 0.5 % of the source entry)
        max_move = 0.005 * cur[i]
        if max_move <= 0.0:

    print(f'Final best score: {best_score:.6f} (length {len(best_seq)})')
    return best_seq
