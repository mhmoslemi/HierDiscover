import time, numpy as np, scipy.sparse as sp, scipy.optimize

def solve_lp_max_sum(a, M):
    n = len(a)
    # Build constraint matrix A (sparse)
    rows = []
    cols = []
    data = []
    for j in range(n):
        # for row k where k from j to j+n-1:
        start = j
        end = j + n  # exclusive
        # coefficients are a[0..n-1]
        # row indices = start ... end-1
        rows.extend(range(start, end))
        cols.extend([j]*n)
        data.extend(a)  # a[0..n-1]
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)  # maximize sum -> minimize negative sum
    bounds = [(0, 1000) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        # fall back
        return None
    return res.x
```

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    try:
        n = len(height_sequence_1)
    except NameError:

    f = binomial_sequence(n)

    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # precompute matrix template rows/cols
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    idx = 0
    for j in range(n):
        rows[idx:idx+n] = np.arange(j, j + n)
        cols[idx:idx+n] = j
        idx += n

    iteration = 0
    t_values = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    while time.time() < deadline:
        # compute current max
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)

        # solve LP
        data = np.tile(f, n)
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, 1000.0) for _ in range(n)]

        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print("[lp] failure, break")
            break
        g0 = np.maximum(0, np.minimum(1000, res.x))
        sum_g0 = np.sum(g0)
        if sum_g0 <= 1e-12:
            print("[lp] zero sum, break")
            break

        # scale g to have same sum_target
        scale = sum_target / sum_g0
        g = g0 * scale

        # try mixing with different t
        improved = False
        for t in t_values:
            f_candidate = (1 - t) * f + t * g
            score = evaluate_sequence(f_candidate.tolist())
            if score < best_score:
                best_score = score
                best_f = f_candidate.copy()
                improved = True
                f = f_candidate
                print(f"[iter {iteration}] t={t:.3f} new best score={best_score:.6f}")

        if not improved:

Improved step‑height search via a linear‑programming “max‑sum” iteration.

The algorithm starts from the theoretical central‑binomial (t⁻¹ᐟ²) sequence
scaled to the canonical sum √(2 n).  In each iteration:

1. Compute the current self‑convolution peak M = ‖f∗f‖∞.
2. Solve a linear program that maximises Σ g_j
   under the constraints  g≥0, g≤1000 and   (f∗g)_k ≤ M  for every
   convolution entry k.  This pushes mass into positions where f is
   small (the tails) without raising the peak.
3. Rescale the LP solution g so that Σ g = Σ f (the prescribed normalisation).
4. Form a convex combination f_new = (1‑t)·f + t·g for several step sizes t.
   The sum stays constant, the peak generally falls, and the evaluation
   score 2·n·max(f_new∗f_new)/(Σ f_new)² improves.
5. Accept the first improvement; otherwise perform a tiny random
   pair‑transfer to escape plateaus and occasionally restart from the
   pure binomial construction.

The loop runs until the time budget expires (or a modest iteration cap
is hit).  The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def binomial_sequence(length: int) -> np.ndarray:
    """Return the central binomial coefficients c_k = (2k choose k)/4**k."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


def random_pair_transfer(seq: np.ndarray, step: float = 0.01) -> np.ndarray:
    """Move a tiny random amount of mass from one index to another."""
    n = len(seq)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return seq.copy()
    amount = step * np.random.rand() * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= amount
    new_seq[j] += amount
    new_seq[i] = max(0.0, new_seq[i])
    return new_seq

    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # coefficients of the current sequence.
    rows = np.empty(n * n, dtype=np.int32)
