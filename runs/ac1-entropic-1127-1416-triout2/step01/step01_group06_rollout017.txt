"""
Propose a low‑score step function.

The algorithm mixes two phases:

1.  A quick random sampling of simple parametric families (exponential,
    power‑law, linear, mixed exponential+constant, logistic, random decreasing)
    to obtain a reasonable baseline candidate.

2.  A deterministic refinement based on a linear‑programming “mass‑filling”
    step.  Starting from a current sequence `f` we compute the current
    convolution bound `M = max(conv(f,f))`.  Keeping `M` fixed we solve a
    linear program that maximises the total mass `sum(g)` of a new
    non‑negative sequence `g` under the linear constraints
    `conv(f,g) ≤ M`.  Adding this extra mass lowers the ratio
    `M / (sum(g)**2)`.  We then replace `f ← (1‑t)·f + t·g` (with a small
    mixing factor `t`) and repeat the LP a few times.  The whole procedure
    is repeated with several random starts while respecting the overall
    time budget (≈1000 s).  The best sequence found is returned.
"""

import time
import math
import random
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)          # noqa: F821 (provided by the harness)
    except Exception:                          # pragma: no cover
        return math.inf

def _normalize(seq):
    """Scale so that max(seq) == 1 (no‑op if all zeros)."""
    arr = np.asarray(seq, dtype=float)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()

def _exp_seq(n, alpha):
    """Exponential decay a_i = exp(-alpha * i)."""
    return np.exp(-alpha * np.arange(n, dtype=float)).tolist()

def _pow_seq(n, p):
    """Power‑law decay a_i = (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=float)
    return (idx ** (-p)).tolist()

def _lin_seq(n, beta):
    """Linear decay a_i = max(0, 1 - beta*i)."""
    return np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float)).tolist()

def _mix_seq(n, alpha, eps):
    """Mixed exponential + constant a_i = exp(-alpha*i) + eps."""
    return (np.exp(-alpha * np.arange(n, dtype=float)) + eps).tolist()

def _logistic_seq(n, steep):
    """Logistic decay a_i = 1/(1+exp((i-mid)/steep))."""
    i = np.arange(n, dtype=float)
    mid = n / 2.0
    seq = 1.0 / (1.0 + np.exp((i - mid) / steep))
    return seq.tolist()

def _randdec_seq(n):
    """Random decreasing sequence (sorted random numbers)."""
    arr = np.random.rand(n)
    arr.sort()
    return arr[::-1].tolist()

def _exp_core_tail_seq(n, core_frac, c, beta):
    """
    Core exponential (length = core_len = int(core_frac * n)) with
    decay factor `c/core_len` and a constant tail of length n-core_len.
    The tail value equals the last core value multiplied by `beta`.
    """
    core_len = max(1, int(core_frac * n))
    alpha = c / core_len
    core = np.exp(-alpha * np.arange(core_len, dtype=float))
    tail_len = n - core_len
    if tail_len > 0:
        tail_val = core[-1] * beta
        tail = np.full(tail_len, tail_val, dtype=float)
        seq = np.concatenate([core, tail])
    else:
        seq = core
    # The first element is 1.0, so we only need to normalise for safety.
    return _normalize(seq.tolist())

# ----------------------------------------------------------------------
# Linear‑programming refinement (mass‑filling)
# ----------------------------------------------------------------------
def _lp_refine(f, t_mix=0.3, max_iters=4, max_entry=10.0):
    """
    Perform up to `max_iters` LP refinement steps on sequence `f`.
    Returns the best refined sequence (as a list) and its score.
    """
    n = len(f)
    f = np.asarray(f, dtype=float)
    if f.max() > 0:
        f = f / f.max()
    best_local = f.copy()
    best_score = _score(best_local.tolist())

    for it in range(max_iters):
        # current convolution bound
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())

        # Build the coefficient matrix for the linear constraints
        # coeffs[k, j] = f[k-j] when 0<=k-j<n else 0
        coeffs = np.zeros((2 * n - 1, n), dtype=float)
        for k in range(2 * n - 1):
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            i_vals = k - np.arange(j_min, j_max + 1)
            coeffs[k, j_min:j_max + 1] = f[i_vals]

        # Linear program: maximise sum(g)  s.t. coeffs @ g <= M   (componentwise)
        g = cp.Variable(n, nonneg=True)
        constraints = [coeffs @ g <= M,
                       g <= max_entry]          # keep numbers moderate
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
        except Exception:
            break

        if g.value is None:
            break

        g_val = np.asarray(g.value, dtype=float)

        # Mix the new mass into the current sequence
        f = (1.0 - t_mix) * f + t_mix * g_val
        if f.max() > 0:
            f = f / f.max()

        cur_score = _score(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_local = f.copy()
            # small progress report
            print(f"LP refinement iter {it+1}: improved score {best_score:.6f}")

    return best_local.tolist(), best_score

# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The routine spends roughly ``budget_s`` seconds
    searching; it mixes random sampling of parametric families with a
    linear‑programming mass‑filling refinement of the best candidate found.
    """
    random.seed(seed)
    np.random.seed(seed)

    # overall deadline (tiny margin)
    deadline = time.time() + budget_s - 0.3

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 0) Start from a global reference if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = _normalize(list(height_sequence_1))   # noqa: F821
            sc = _score(init)
            if sc < best_score:
                best_seq, best_score = init, sc
                print(f"Initial reference score: {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) Random sampling of simple parametric families
    # ------------------------------------------------------------------
    lengths = [400, 600, 800, 1200, 1800, 2600, 3400, 4200]
    families = ['exp', 'pow', 'lin', 'mix', 'log', 'randdec', 'coretail']

    while time.time() < deadline:
        n = int(random.choice(lengths))
        family = random.choice(families)

        if family == 'exp':
            # α ∈ [1e‑4, 2e‑2] (log‑uniform)
            log_alpha = random.uniform(-4, math.log10(0.02))
            alpha = 10 ** log_alpha
            cand = _exp_seq(n, alpha)

        elif family == 'pow':
            p = random.uniform(0.0, 3.0)
            cand = _pow_seq(n, p)

        elif family == 'lin':
            beta = random.uniform(0.0, 2.0 / n)
            cand = _lin_seq(n, beta)

        elif family == 'mix':
            log_alpha = random.uniform(-4, math.log10(0.02))
            alpha = 10 ** log_alpha
            eps = random.uniform(0.0, 0.5)
            cand = _mix_seq(n, alpha, eps)

        elif family == 'log':
            steep = random.uniform(0.01, n / 2.0)
            cand = _logistic_seq(n, steep)

        elif family == 'randdec':
            cand = _randdec_seq(n)

        else:  # coretail – exponential core + constant tail
            core_frac = random.uniform(0.4, 0.9)
            c = random.uniform(0.3, 2.5)          # core decay constant
            beta = random.uniform(0.0, 1.0)       # tail scaling
            cand = _exp_core_tail_seq(n, core_frac, c, beta)

        # discard pathological all‑zero vectors
        if sum(cand) < 1e-8:
            continue

        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"Random {family:8s} n={n:4d} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Linear‑programming refinement (mass‑filling)
    # ------------------------------------------------------------------
    # Try a few random starts for the LP step – each start uses a
    # different seed and a modest sequence length (≈400) to keep the LP
    # cheap.  The best refined sequence is kept.
    while time.time() < deadline:
        # choose a seed sequence: random exponential or the current best
        if best_seq is not None and random.random() < 0.5:
            # truncate / pad the best sequence to a manageable length
            n_lp = 400
            base = np.array(best_seq[:n_lp], dtype=float)
            if len(base) < n_lp:
                base = np.pad(base, (0, n_lp - len(base)), 'constant')
        else:
            # fresh random exponential start
            n_lp = 400
            alpha = random.uniform(0.0005, 0.01)
            base = np.exp(-alpha * np.arange(n_lp, dtype=float))

        base = _normalize(base.tolist())
        refined_seq, refined_score = _lp_refine(base, t_mix=0.3,
                                                max_iters=4, max_entry=10.0)

        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq
            print(f"LP‑refined candidate → {best_score:.6f}")

        # only a couple of LP attempts are needed under the time budget
        if time.time() + 3.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # 3) Final fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 500
        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
