"""
Propose a non‑negative step‑function sequence that minimizes
`evaluate_sequence`.  The search uses an alternating
linear‑programming / mixing scheme inspired by the state‑of‑the‑art
method described in the prompt, followed by a short
exponentiated‑gradient refinement.  It works for any
length (default 1000‑2000) and respects the 1000 s budget.
"""

import time
import numpy as np

# Helper: generate arcsine / sqrt distribution (coefficients of (1‑z)^(-1/2))
def sqrt_distribution(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()                 # normalise to sum = 1
    return a

def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Return matrix A such that (A @ g) = a * g (self‑convolution)."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # fill row by row
    for k in range(rows):
        i_start = max(0, k - n + 1)
        i_end = min(k + 1, n)
        # for j in [i_start, i_end) we have a_{k-j}
        A[k, i_start:i_end] = a[k - np.arange(i_start, i_end)]
    return A

# Exponentiated‑gradient refinement (optional)
def expgrad_refine(a: np.ndarray, steps: int, tau_start: float = 1.0,
                   eta_start: float = 0.6) -> np.ndarray:
    """Run a few steps of the smooth‑max EG update."""
    n = a.shape[0]
    a_cur = a.copy()
    tau = tau_start
    eta = eta_start
    for it in range(steps):
        # smooth max
        b = np.convolve(a_cur, a_cur)
        max_b = b.max()
        w = np.exp(tau * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        conv_rev = np.convolve(a_cur, w_rev)
        L = b.size
        grad = 2.0 * conv_rev[(L - 1) - np.arange(n)]
        a_cur = a_cur * np.exp(-eta * grad)
        # projection to probability simplex
        a_cur = np.maximum(a_cur, 0.0)
        s = a_cur.sum()
        if s > 0:
            a_cur /= s
        else:
            a_cur[:] = 1.0 / n
        # schedule
        if (it + 1) % 20 == 0:
            tau = min(30.0, tau * 1.05)
            eta = max(0.02, eta * 0.97)
    return a_cur

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    `evaluate_sequence` score.

      * solving a linear program that maximises the ℓ₁‑mass of a
        partner vector `g` while keeping the convolution `a*g`
        below the current worst value `M = max(a*a)`;
      * mixing the new partner into the current vector `a`
        with a small step size `t`;
    after a handful of LP steps a short exponentiated‑gradient
    refinement is performed.  The best sequence seen before the
    time limit expires is returned.
    """
    import cvxpy as cp

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Choose the length of the sequence
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 1000  # moderate size that keeps the LP fast

    # ------------------------------------------------------------------
    # Initialise with the optimal square‑root distribution (arcsine)
    # ------------------------------------------------------------------
    a = sqrt_distribution(n)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main alternating LP loop
    # ------------------------------------------------------------------
    mix_factor = 0.15          # step size (1‑t) in the paper
    iteration = 0
    stagnant = 0

    while time.time() < deadline and iteration < 200:
        # current convolution bound
        b = np.convolve(a, a)
        M = b.max()

        # Build the linear mapping matrix for a * g
        A = build_conv_matrix(a)

        # Linear programme:
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False,
                       max_iters=5000)
        except Exception as e:
            print(f"[{iteration}] LP solve failed: {e}")
            break

        if g.value is None:
            print(f"[{iteration}] LP returned no solution.")
            break

        g0 = np.array(g.value).flatten()
        sum_g0 = g0.sum()
        if sum_g0 == 0.0:
            print(f"[{iteration}] LP gave zero mass; abort.")
            break

        # Rescale g0 so that its ℓ₁‑norm equals sqrt(2·n) (the same as a)
        target_norm = np.sqrt(2.0 * n)
        g = g0 * (target_norm / sum_g0)

        if a_new_sum > 0:
            a_new /= a_new_sum
        else:
            a_new = np.full_like(a_new, 1.0 / n)

        # evaluate
        try:

        except Exception:
            cur_score = float("inf")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            stagnant = 0
            print(f"[{iteration:03d}] improved score = {best_score:.6f}")
        else:
            stagnant += 1

        a = a_new
        iteration += 1

        # If we have been stagnant for a while, add a small random jitter
        if stagnant >= 10:
            jitter = 0.005 * np.random.randn(n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            stagnant = 0
            print(f"[{iteration}] jitter applied")

    # ------------------------------------------------------------------
    # Final fine‑tuning using a few exponentiated‑gradient steps
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:

        try:

            if score_fine < best_score:
                best_score = score_fine
                best_seq = a_fine
                print(f"[fine] improved score = {best_score:.6f}")
        except Exception:
            pass

    print(f"[done] best score = {best_score:.6f}")
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative sequence) that minimises the
`evaluate_sequence` score.

The search follows the “alternating LP” idea described in the
statement:

1. Start from the optimal square‑root (arcsine) distribution, i.e.
   the coefficients of (1‑z)⁻¹ᐟ², normalised to an ℓ₁‑norm of 1.
2. Let M = max(a * a) (self‑convolution).  Solve the linear programme

       maximise   Σ g_i
       subject to g_i ≥ 0
                  (a * g)_k ≤ M   for every k

   where a is fixed and g is the variable.  The programme is linear
   because a is known, so the convolution constraints are linear in g.
3. Normalise the LP solution g to an ℓ₁‑norm of 1, then replace

       a ← (1‑t)·a + t·g               (0<t≤1)

   i.e. take a convex combination.  This decreases the worst
   convolution value M.
4. Repeat steps 2‑3 until the time budget expires or progress stalls.
   If progress stalls, add a small random jitter or restart from a fresh
   random point.
5. Finish with a short exponentiated‑gradient refinement of the best
   sequence found.

The algorithm keeps the best sequence seen so far and returns it
as a plain list before the 1000 s budget runs out.
"""

import time
import numpy as np
import cvxpy as cp


def sqrt_distribution(n: int) -> np.ndarray:
    """Coefficients of (1 - z)^(-1/2) of length n, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1) × n) such that
        (A @ g) = a * g   (convolution of a with g)
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - n + 1)
        i_end = min(k + 1, n)
        # A[k, i_start:i_end] = a[k - i]  for i = i_start … i_end-1
        A[k, i_start:i_end] = a[k - np.arange(i_start, i_end)]
    return A


def expgrad_refine(a: np.ndarray, steps: int,
                   tau_start: float = 1.0, eta_start: float = 0.6) -> np.ndarray:
    """
    A few steps of the smooth‑max exponentiated‑gradient update, used as a
    final polishing stage.  The sequence a is assumed to sum to 1.
    """
    n = a.shape[0]
    a_cur = a.copy()
    tau = tau_start
    eta = eta_start
    for it in range(steps):
        b = np.convolve(a_cur, a_cur)          # length 2n‑1
        max_b = b.max()
        w = np.exp(tau * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum
        # gradient = 2 * (a * w)  (convolution with reversed w)
        w_rev = w[::-1]
        conv_rev = np.convolve(a_cur, w_rev)
        L = b.size
        grad = 2.0 * conv_rev[(L - 1) - np.arange(n)]
        a_cur = a_cur * np.exp(-eta * grad)
        a_cur = np.maximum(a_cur, 0.0)
        s = a_cur.sum()
        if s > 0.0:
            a_cur /= s
        else:
            a_cur[:] = 1.0 / n
        # slowly increase temperature, slowly decay step size
        if (it + 1) % 20 == 0:
            tau = min(30.0, tau * 1.05)
            eta = max(0.02, eta * 0.97)
    return a_cur


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score sequence within the allotted time.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # a little safety margin

    # ------------------------------------------------------------------
    # Choose a reasonable length.
    # Prefer the length of a previously computed construction but cap it
    # so that the linear programmes stay fast.
    # ------------------------------------------------------------------
    try:
        n0 = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n0 = None
    if n0 is None or n0 > 1200:
        n = 1200
    else:
        n = max(200, n0)   # at least a couple of hundred entries

    # ------------------------------------------------------------------
    # Initialise with the arcsine (square‑root) distribution.
    # ------------------------------------------------------------------
    a = sqrt_distribution(n)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop (alternating linear programmes).
    # ------------------------------------------------------------------
    mix_factor = 0.15          # convex‑mixing weight t
    eps = 1e-12                # to avoid exact zeros in a
    iteration = 0
    stagnant = 0

    while time.time() < deadline and iteration < 250:
        # current worst convolution value
        b = np.convolve(a, a)
        M = b.max()

        # build the matrix for the convolution a * g
        A = build_conv_matrix(a)

        # --------------------------------------------------------------
        # Linear programme: maximise Σ g_i   s.t.   g≥0,  A g ≤ M
        # --------------------------------------------------------------
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       verbose=False, max_iters=5000)
        except Exception as e:
            print(f"[{iteration}] LP solve error: {e}")
            break

        if g.value is None:
            print(f"[{iteration}] LP returned no solution.")
            break

        g0 = np.array(g.value).flatten()
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print(f"[{iteration}] LP gave zero total mass.")
            break

        # Normalise the LP solution to ℓ₁‑norm = 1 (convex combination)
        g_norm = g0 / sum_g0

        # Convex combination and re‑normalise (the sum stays 1)
        a_new = (1.0 - mix_factor) * a + mix_factor * g_norm
        a_new = np.maximum(a_new, eps)      # keep all entries strictly positive
        a_new /= a_new.sum()

        # Evaluate the new candidate
        try:
            cur_score = evaluate_sequence(a_new.tolist())
        except Exception:
            cur_score = float("inf")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            stagnant = 0
            print(f"[{iteration:03d}] improved score = {best_score:.6f}")
        else:
            stagnant += 1

        a = a_new
        iteration += 1

        # ------------------------------------------------------------------
        # Adaptive actions when progress stalls
        # ------------------------------------------------------------------
        if stagnant >= 12:
            # small random perturbation to escape flat regions
            jitter = 0.004 * np.random.randn(n)
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            stagnant = 0
            print(f"[{iteration}] jitter applied")
        if stagnant >= 50:
            # full random restart (keep the best sequence found so far)
            a = np.random.rand(n)
            a = np.maximum(a, eps)
            a /= a.sum()
            stagnant = 0
            print(f"[{iteration}] random restart")

    # ------------------------------------------------------------------
    # Final polishing with a few exponentiated‑gradient steps
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        # use a fraction of the remaining time for refinement
        remaining = deadline - time.time()
        steps = min(3000, int(remaining * 80))   # ≈ 0.0125 s per step
        a_refined = expgrad_refine(best_seq, steps)
        try:
            refined_score = evaluate_sequence(a_refined.tolist())
            if refined_score < best_score:
                best_score = refined_score
                best_seq = a_refined
                print(f"[refine] improved score = {best_score:.6f}")
        except Exception:
            pass

    print(f"[done] best score = {best_score:.6f}")
    return best_seq.tolist()
