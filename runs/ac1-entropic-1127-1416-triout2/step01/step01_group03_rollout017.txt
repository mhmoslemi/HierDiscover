def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    try:
        n = len(height_sequence_1)
        init_seq = np.array(height_sequence_1, dtype=float)
        # Ensure sum=1
        init_seq = np.maximum(init_seq, 0)
        if init_seq.sum() == 0:
            init_seq = sqrt_distribution(n)
        else:
            init_seq = init_seq / init_seq.sum()
    except Exception:
        n = 2000
        init_seq = sqrt_distribution(n)
    # Convert to u (log) for softmax representation.
    u = np.log(init_seq + 1e-15)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    # Scheduled parameters
    tau = 1.0
    max_tau = 200.0
    lr = 0.1

    best_a = init_seq.copy()
    best_score = 2.0 * n * np.max(np.convolve(best_a, best_a))
    # Or compute via evaluate_sequence.
    # compute once more with evaluate function for sanity:
    try:
        best_score = evaluate_sequence(best_a.tolist())
    except Exception:
        best_score = float('inf')
    print(f"[init] n={n}, score={best_score:.6f}")

    stagnant = 0
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # softmax to get a
        u_shift = u - np.max(u)
        exp_u = np.exp(u_shift)
        a = exp_u / exp_u.sum()

        # convolution
        b = np.convolve(a, a)
        max_b = b.max()
        score = 2.0 * n * max_b

        # record best
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            print(f"[{iteration:05d}] improved score={best_score:.6f}")

        else:
            stagnant += 1

        # compute softmax weights
        w = np.exp(tau * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a
        grad_a = 2.0 * np.convolve(a, w[::-1])[(len(b)-1) - np.arange(n)]

        # gradient wrt u (softmax logits)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # schedule temperature
        if iteration % 10 == 0 and tau < max_tau:
            tau *= 1.01  # slowly increase

        # occasional jitter to avoid stagnation
        if stagnant > 2000:
            # small random noise
            u += 0.005 * np.random.randn(n)
            stagnant = 0
            print(f"[{iteration:05d}] jitter added")

        # occasional restart if long stagnation
        if stagnant > 20000:
            # restart around best solution with noise
            u = np.log(best_a + 1e-15) + 0.1 * np.random.randn(n)
            m = np.zeros_like(u)
            v = np.zeros_like(u)
            tau = 1.0
            stagnant = 0
            print(f"[{iteration:05d}] restart")

    # final best sequence to return
    # ensure non-negative and clipped to [0,1000] maybe not needed
    final_seq = best_a.tolist()
    print(f"[done] best score={best_score:.6f} after {iteration} iters")
    return final_seq

#!/usr/bin/env python3
"""
A fast stochastic mirror‑descent optimizer for the “minimise max‑convolution”
challenge.

The sequence `a` is represented as a soft‑max over free logits `u`:
    a_i = exp(u_i) / Σ_j exp(u_j)      (a_i ≥ 0, Σ a_i = 1)

For a given temperature `τ` we smooth the max‑operator
    max(b) ≈ Σ_k w_k b_k  ,   w_k = softmax_τ(b_k)
where b = a * a is the self‑convolution.
The gradient of the smoothed objective w.r.t. `a` is
    g = 2·a ⊛ w_rev                (⊛ is convolution, w_rev = w[::-1])

Using the chain‑rule we obtain the gradient w.r.t. the logits `u`:
    ∂f/∂u_i = a_i·(g_i – Σ_j a_j g_j)

The logits are updated with Adam (adaptive moments).  The temperature `τ`
starts modest and is increased slowly so that the optimisation first sees a
smooth landscape and later focuses on the true maximum.  If progress stalls
the algorithm injects a small random jitter; after a long stall it restarts
around the best solution found so far.

The routine runs for the allotted time budget (≤ 1000 s) and returns the
best sequence discovered.  All values are non‑negative, sum‑normalised and
compatible with `evaluate_sequence`.
"""

def sqrt_distribution(n: int) -> np.ndarray:

    the result is divided by its sum so that Σ a_i = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def softmax(u: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    u_shift = u - np.max(u)
    e = np.exp(u_shift)
    return e / e.sum()


def max_convolution(a: np.ndarray) -> float:
    """Return max(a*a) for a non‑negative vector with Σa = 1."""
    b = np.convolve(a, a)
    return float(b.max())

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The optimizer runs for at most ``budget_s``
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Length of the sequence – use the provided construction if present.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        # initialise from the previously discovered construction
        a0 = np.maximum(np.array(height_sequence_1, dtype=float), 0.0)
        if a0.sum() == 0.0:
            raise ValueError
        a0 /= a0.sum()
    except Exception:
        # fall back to the classic √‑distribution
        n = 2000
        a0 = sqrt_distribution(n)

    # ------------------------------------------------------------------
    # Represent the point on the probability simplex by logits u.
    # ------------------------------------------------------------------
    u = np.log(a0 + 1e-15)          # avoid log(0)

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # hyper‑parameters
    lr = 0.12                       # base learning‑rate on logits
    tau = 1.0                       # temperature of the soft‑max
    max_tau = 250.0                 # we never need a higher temperature

    best_a = a0.copy()
    best_score = float("inf")
    stagnant = 0
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- current point on the simplex -----
        a = softmax(u)                     # Σa = 1, a_i ≥ 0

        # ----- objective (exact, no call to evaluate_sequence) -----
        max_b = max_convolution(a)
        cur_score = 2.0 * n * max_b        # because Σa = 1

        # ----- keep the best discovered solution -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnant = 0
            print(f"[{iteration:05d}] improved score = {best_score:.6f}")
        else:
            stagnant += 1

        # ----- smooth‑max weights -----
        # w_k = exp( τ·(b_k – max_b) )  , normalised
        b = np.convolve(a, a)              # length 2n‑1
        w = np.exp(tau * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient wrt a (fixed w) -----
        # grad_a_i = 2· Σ_j a_j·w_{i+j}
        # compute via one convolution with the reversed weight vector
        w_rev = w[::-1]
        conv_rev = np.convolve(a, w_rev)                 # length 3n‑2
        L = b.size                                        # = 2n‑1
        grad_a = 2.0 * conv_rev[(L - 1) - np.arange(n)]

        # ----- gradient wrt logits u (softmax manifold) -----
        # ∂f/∂u_i = a_i·(grad_a_i – Σ_j a_j·grad_a_j)
        avg = np.dot(a, grad_a)
        grad_u = a * (grad_a - avg)

        # ----- Adam update on u -----
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule (focus more on the real max) -----
        if iteration % 10 == 0 and tau < max_tau:
            tau *= 1.015  # very gentle increase

        # ----- occasional random jitter to escape plateaus -----
        if stagnant > 1200:
            u += 0.004 * np.random.randn(n)
            stagnant = 0
            print(f"[{iteration:05d}] jitter injected (τ={tau:.2f})")

        # ----- long‑term stagnation → full restart around the best point -----
        if stagnant > 8000:
            u = np.log(best_a + 1e-15) + 0.1 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            tau = 1.0
            stagnant = 0
            print(f"[{iteration:05d}] full restart")

    # ------------------------------------------------------------------
    # Return the best sequence (plain Python list) before the time limit.
    # ------------------------------------------------------------------
    # Sanity check with the provided evaluator (optional but cheap)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[done] best score reported by evaluator = {final_score:.6f}")
    except Exception:
        print("[done] evaluation failed – returning internal best score")
    return best_a.tolist()
