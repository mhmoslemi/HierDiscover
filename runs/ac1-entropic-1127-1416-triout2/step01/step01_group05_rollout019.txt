def build_ub_matrix(f, active_idx):
    n = len(f)

    cols = []
    data = []
    for r, k in enumerate(active_idx):

propose_candidate
-----------------
Search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

with the same normalisation used by the benchmark (sum(a) = sqrt(2*n)).
The algorithm consists of three stages:

1. **Wide grid search** over length, exponent `p` and offset `α`.
   The analytic optimum for a continuous kernel is p = 1/2,
   α = 0.  By allowing p > 0.5 and a positive offset we can lower
   both the central self‑convolution term and the large edge term
   a₀², which together dominate the max.  The grid is deliberately
   generous (p ∈ [0.55, 0.85] and α up to 30) and searches a few
   thousand candidates – cheap compared to the 1000 s budget.

2. **Hill‑climbing on (`p`, `α`)** – small random nudges are accepted
   if they improve the score, allowing us to refine the discrete grid
   without rebuilding the whole search table.

3. **Fine‑grained random local moves** on the sequence itself,
   analogous to the previous solution but starting from the much
   better candidate found in steps 1‑2.

If `scipy` is available we additionally perform a few
*linear‑programming* refinements à la the classic “active‑set” method:
given the current `f`, we maximise the total mass of a non‑negative
vector `g` under the linear constraints `conv(f,g) ≤ M` (M is the
current maximal self‑convolution).  Mixing `f` with the rescaled `g`

(max‑k values) are kept as constraints, so each LP is tiny (≤ 30 rows).

def _make_power_sequence(n: int, p: float, offset: float) -> np.ndarray:
    """
    Return the raw (unnormalised) sequence a_i = (i + 1 + offset) ** (-p).
    """
    idx = np.arange(n, dtype=float)
    return 1.0 / np.power(idx + 1.0 + offset, p)


def _normalise_to_target(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale a non‑negative vector so that its sum equals target_sum.
    """
    cur = seq.sum()
    if cur == 0.0:
        return seq
    return seq * (target_sum / cur)


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq.tolist())


def _build_ub_matrix(f: np.ndarray, active_idx: np.ndarray) -> "scipy.sparse.csr_matrix":
    """
    Build the (len(active_idx) × n) matrix A such that
        (A @ g)[k] == conv(f, g)[active_idx[k]]
    for a given non‑negative vector f.
    """
    import scipy.sparse as sp
    n = f.size
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active_idx):
        # i runs over indices where both i and k-i are valid
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # loop is short (≈ min(k+1, 2n-1‑k))
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(row)
            cols.append(j)
            data.append(f[i])
    return sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))

            conv_f = np.convolve(f, f)
            M = conv_f.max()
            # pick the 20 largest convolution entries (or all if fewer)
            top_k = np.argsort(-conv_f)[:min(20, conv_f.size)]
            # build the sparse matrix for the active constraints
            A_ub = _build_ub_matrix(f, top_k)
            b_ub = np.full(A_ub.shape[0], M, dtype=float)
            c = -np.ones(f.size, dtype=float)   # maximise sum(g)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=(0, None), method='highs', options={'disp': False})
            if not res.success:
                break
            g = res.x
            if g.sum() == 0.0:
                break
            # rescale g to the required total mass
            g = _normalise_to_target(g, target_sum)
            # mix and evaluate
            f_new = (1 - t) * f + t * g
            new_score = _evaluate(f_new)
            if new_score < best_score:
                best_score = new_score
                best_seq = f_new.copy()
                f = f_new
                print(f'++ LP refine iter {iteration}: score={new_score:.6f}')
                if best_score < 1.503:
                    break
            else:
                # reduce step size if no improvement
                t = t * 0.5
                if t < 1e-3:
                    break
    # ------------------------------------------------------------------

    print(f'Final best score: {best_score:.6f} (length {len(best_seq)})')
    return best_seq.tolist()
