```python
import numpy as np
import time
import scipy.optimize as opt
import scipy.sparse as sp

def solve_lp_for_g(f):
    n = len(f)
    # compute M = max(conv(f,f))
    conv_ff = np.convolve(f, f)
    M = conv_ff.max()
    # Build A_ub: each row: conv(f, g) <= M
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        # j indices where g_j contributes: max(0, k-(n-1)) <= j <= min(k, n-1)
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        # g_j coefficient is f_{k-j}
        for j in range(j_min, j_max + 1):
            i = k - j # i index in f
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    # bounds: g_i >=0
    bounds = [(0, None) for _ in range(n)]
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None, M
    g = res.x
    return g, M
```

        return None

#!/usr/bin/env python3
"""
Hybrid optimizer for the “max‐convolution” evaluation problem.

The algorithm works in two stages inside the 1000 s budget:

1. **Linear‑programming (LP) step** – given the current sequence `a`
   (∑a = 1) we compute
        M = max(a * a) .
   We then solve the LP

        maximise   Σ g_i
        subject to  (a * g)_k ≤ M   for all k = 0 … 2n‑2
                    g_i ≥ 0

   The solution `g` points to a direction that can increase the total
   mass while respecting the current convolution bound `M`.  After
   normalising `g` to have unit sum we line‑search a mixing weight `t`
   and replace `a ← (1‑t)·a + t·ĝ`.  This step strictly reduces the
   maximal convolution (or leaves it unchanged) and quickly drives the
   sequence towards an “equi‑oscillating’’ optimal shape.

2. **Exponentiated‑gradient refinement** – once the LP phase stops
   improving, we fall back to a smooth‑max surrogate (soft‑max over the
   convolution) and perform exponentiated‑gradient updates with an
   annealing temperature `τ`.  This fine‑tunes the coefficients and helps
   escape tiny plateaus.

Both phases keep track of the best sequence ever seen (according to
`evaluate_sequence`).  The method works for any length; if a global
`height_sequence_1` is present its length is used, otherwise a default
size of 2000 is chosen.

The routine returns a plain Python list, ready for the external
`evaluate_sequence` call.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def sqrt_distribution(n: int) -> np.ndarray:
    """Return the normalised coefficients of (1‑z)^(-1/2) up to length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Create the sparse matrix A such that A @ g == np.convolve(f, g, mode='full')
    for any vector g (length = len(f)).
    A has shape (2n‑1, n) and each diagonal i (0 ≤ i < n) consists of the
    constant value f[i].
    """
    n = f.shape[0]
    # each diagonal is a constant vector of length n
    diags = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=np.int32)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def solve_lp_max_sum(f: np.ndarray) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_i
        subject to (f * g)_k ≤ M  ∀k
                    g_i ≥ 0
    where M = max(f * f).  Returns the optimal g (as a dense ndarray) or
    None if the solver fails.
    """
    n = f.shape[0]
    # current bound
    M = np.convolve(f, f).max()
    # build the constraint matrix once per call
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -Σg  → maximise Σg
    bounds = [(0.0, None) for _ in range(n)]

    # HiGHS is fast for large sparse LPs
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options={"presolve": True})
    if res.success:
        return res.x
    return None


def smooth_max_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the smooth surrogate   g(a)=∑_k w_k·(a*a)_k ,
    with soft‑max weights  w_k = exp(τ·( (a*a)_k – max )) / Σ .
    Returns (grad, max_b, full_convolution).
    """
    b = np.convolve(a, a)                 # length 2n‑1
    max_b = b.max()
    # soft‑max (numerically stable)
    w = np.exp(tau * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= w_sum

    # grad_i = 2·∑_j a_j·w_{i+j}
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)      # length 3n‑2
    L = b.size
    grad = 2.0 * conv_rev[(L - 1) - np.arange(a.size)]
    return grad, max_b, b
