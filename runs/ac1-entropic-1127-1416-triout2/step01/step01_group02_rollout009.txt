def gradient_step(a, lr):
    n = a.shape[0]
    conv = compute_conv(a)

"""step‑function optimizer
---------------------------------
Implements a simple projected gradient descent on the objective

    J(a) = 2·n·max(a∗a) / (∑a)²

where `∗` is the linear convolution.  The gradient of `J` is obtained
 analytically (the only non‑zero part of the gradient of max(a∗a) comes
 from the index where the maximum is attained).  Starting from the
 theoretical “binomial’’ sequence we repeatedly move mass away from the
 entries that participate in the current maximal convolution term and
 spread it to the other positions.  The update is projected onto the
 non‑negative orthant and the whole process is run until the time budget
 (default 1000 s) expires.  The best sequence found is returned as a
 plain Python list, ready to be passed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper: the (optimal) central‑binomial coefficients
def binomial_sequence(length: int) -> np.ndarray:
    """c_k = (2k choose k) / 4**k for k=0..length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def _conv_max_and_idx(a: np.ndarray) -> tuple[float, int, np.ndarray]:
    """
    Compute the linear convolution a*a using an FFT, return the maximal
    value, its index and the full convolution array.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_val = float(conv.max())
    max_idx = int(conv.argmax())
    return max_val, max_idx, conv
