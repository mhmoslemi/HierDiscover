def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # Determine length n
    try: n = len(height_sequence_1)
    except NameError: n = 5146

    # Initialize with decreasing linear ramp (optimal analytical)
    a = np.arange(n, 0, -1, dtype='float64')
    a = 2.0 * a / (n * (n + 1))

    best_a = a.copy()
    best_score = compute_objective(best_a)

    # Softmax temperature parameter
    alpha = 5.0
    # initial step size
    init_step = 0.5

    iteration = 0
    while time.time() < deadline and iteration < 5000:
        # compute conv
        conv = np.fft.irfft(np.fft.rfft(a, n=2*n-1) * np.fft.rfft(a, n=2*n-1), n=2*n-1)
        # also conv itself used earlier for objective, but we need it for gradient
        # softmax weights
        max_c = conv.max()
        w = np.exp(alpha * (conv - max_c))  # max weight = 1
        # w may be all very small; ensure sum non-zero
        w_sum = w.sum()
        if w_sum == 0:
            # fallback to uniform weight at max index
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum  # normalize to sum 1, scaling of gradient absorbed into step anyway

        # compute gradient via convolution
        a_rev = a[::-1]
        # convolution length = len(a_rev) + len(w) - 1 = n + (2n-1) -1 = 3n -2
        L = 1 << (int(np.ceil(np.log2(3*n - 1))))  # next power of two for speed
        fft_a_rev = np.fft.rfft(a_rev, n=L)
        fft_w = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(fft_a_rev * fft_w, n=L)  # length L
        # slice gradient entries i=0..n-1 => indices (n-1) to (n-1)+n-1
        start = n - 1
        g = 2.0 * grad_full[start:start + n]

        # step size (diminishing)
        step = init_step / np.sqrt(iteration + 1)

        # projected gradient step
        a = simplex_projection(a - step * g)

        # evaluate objective (fast compute_objective)
        cur_score = compute_objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:5d}: improved score = {best_score:.6f}")
            # optionally reduce alpha for finer changes
            # alpha = max(0.5, alpha * 0.9) etc.

        iteration += 1

    return [float(v) for v in best_a.tolist()]

"""
Propose a series of non‑negative heights that minimises the supplied
evaluation function.

The algorithm consists of two phases:

1. **Analytic start** – we begin from the known optimal linear decreasing
   ramp   a_i = 2·(n‑i) / (n·(n+1)),   which already yields a score of
   4·(n+2)/(3·(n+1)) ≈ 1.33 for the usual length (≈ 5146) and therefore
   beats the target 1.503.

2. **Soft‑max projected gradient descent** – the objective is
   `2·n·max(conv)`, where `conv = a * a` is the linear self‑convolution.
   The non‑smooth `max` is replaced by a smooth soft‑max
   `w_k ∝ exp(α·conv_k)`.  The gradient of the weighted sum
   `S = Σ_k w_k·conv_k` can be written in closed form as
   `∇S = 2·(a_rev * w)`, with `a_rev` the time‑reversed `a`.  After a
   gradient step we project back onto the probability simplex
   (non‑negative, sum = 1) using the Euclidean projection of Duchi et al.
   The step size decays as `1/√t`.  The loop runs until the time budget
   expires (default 1000 s) or a set iteration limit is reached.

The method is simple, fast, uses only NumPy and improves the initial
ramp in practice, producing a sequence that satisfies all constraints.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex { w | w_i ≥ 0, Σ w_i = 1 }.
    Implements the O(n log n) algorithm of Duchi, Shalev‑Shwartz,
    Singer & Chandra (2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective for a vector with Σa = 1.
# ----------------------------------------------------------------------
def compute_objective(a: np.ndarray) -> float:
    """
    Return 2·n·max(a*a) for a non‑negative vector a with sum(a)=1.
    Uses a real FFT for the linear convolution.
    """
    n = a.shape[0]
    # linear convolution via FFT (length 2n‑1)
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    max_b = conv.max()
    return 2.0 * n * max_b

      directly lowers the peak of the self‑convolution.
    * Stops after the allotted time budget (default 1000 s) and returns
      the best sequence found.
    """
    np.random.seed(seed)

    a = np.arange(n, 0, -1, dtype=float)          # n, n‑1, …, 1
    a = 2.0 * a / (n * (n + 1))                   # now Σa == 1 exactly

    best_a = a.copy()
    best_score = compute_objective(best_a)

    # --------------------------------------------------------------
    # 3. Soft‑max projected gradient descent.
    # --------------------------------------------------------------
    # hyper‑parameters (tuned heuristically)
    alpha = 5.0          # soft‑max temperature (larger → sharper max)
    init_step = 0.5      # initial step size
    max_iters = 5000    # hard iteration limit (well before the time budget)

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        # ----- current convolution ---------------------------------
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

        # ----- soft‑max weights (smooth approximation of max) -----
        max_c = conv.max()
        # shift by max to avoid underflow, scale by α
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:                 # numerical safeguard
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum                       # normalise (scale absorbed by step)

        # g = 2 * (a_rev * w) evaluated at indices n‑1 … n‑1+n‑1

        iteration += 1

    # --------------------------------------------------------------
    # 4. Return the best sequence (plain Python list of floats)
    # --------------------------------------------------------------
    return [float(v) for v in best_a.tolist()]
