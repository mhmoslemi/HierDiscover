def simplex_projection(v, S):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    # find rho: last index where u_i > cssv_i / (i)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def convolve_fft(a):
    n = len(a)

def project_simplex(v, S):
    v = v.copy()
    # Implementation from Duchi et al.
    # must handle negative values

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_array = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
    if len(rho_array) == 0:
        # all negative? then result is zeros.
        theta = cssv[-1] / len(v)
    else:
        rho = rho_array[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function with a very low evaluation value.

The algorithm works in two stages.
1) A fast random search over a 4‑parameter family
        a_i = (i+α)^(-p) * (1 + β * i/(n‑1))
   (α ≥ 0, 0.3 ≤ p ≤ 0.7, –0.9 ≤ β ≤ 2.5).  
   Each candidate is normalised so that
        Σ a_i = √(2·n) .
   The discrete self‑convolution is obtained with a real FFT, which makes
   the evaluation cheap even for several‑thousand‑element sequences.
   The best candidate from this coarse sweep is kept.

2) Starting from the best candidate a projected sub‑gradient descent
   minimises the quantity   max(conv(a,a)) .
   For a given sequence the indices attaining the current maximum are
   identified; the sub‑gradient of the max‑convolution is
        g_i = Σ_{k∈K} a_{k‑i}
   (out‑of‑range terms are ignored).  A step
        a ← a – η·g
   is taken, the vector is projected back onto the simplex
        {x ≥ 0, Σ x = √(2·n)} .
   The step size η decays like 1/√t.  Occasionally a tiny random
   perturbation is added to escape plateaus.  The loop runs until the
   time budget (≈ 1000 s) is exhausted or the target value (≈ 1.503) is
   reached.

The routine never calls ``evaluate_sequence`` inside the inner loops –
the score is simply the maximal convolution value because the sum is
kept exactly at √(2·n).  At the end the best sequence found is returned
as a plain list of floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { w | w_i ≥ 0, Σ w_i = target_sum } .
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    # Work on a copy
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    # Find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero – should not happen for our use case
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full (linear) convolution a*a using a real FFT.
    Returns an array of length 2·len(a)-1.
    """
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Return max_{k} (a*a)[k] via FFT."""
    return _full_convolution_fft(a).max()


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """
    Build an un‑normalised raw sequence
        a_i = (i + 1 + offset)^{-p} * (1 + beta * i/(n-1))
    and project it onto the simplex of sum √(2·n).
    """
    idx = np.arange(n, dtype=float)                # i = 0 … n‑1
    base = np.power(idx + 1.0 + offset, -p)
    # linear tilt (beta may be negative, but we keep the result non‑negative)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    # Clip any tiny negative values caused by a negative beta
    raw = np.maximum(raw, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


def _subgradient_step(a: np.ndarray, max_inds: np.ndarray, step: float,
                      target_sum: float) -> np.ndarray:
    """
    Perform one sub‑gradient step for the objective max(conv(a,a)).
    The gradient is computed only for the indices where the current
    maximum is attained (``max_inds``).
    """
    n = len(a)
    g = np.zeros_like(a)

    for k in max_inds:
        if k < n:
            # i runs from 0 … k
            g[: k + 1] += a[k :: -1]
        else:
            # i runs from k-(n-1) … n-1
            start = k - (n - 1)
            # vectorised addition
            idx = np.arange(start, n)
            g[start:] += a[k - idx]

    # Take a step opposite to the sub‑gradient and re‑project onto the simplex.
    cand = a - step * g
    cand = _project_simplex(cand, target_sum)
    return cand


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    Using a cheap parametric random search followed by a projected
    sub‑gradient descent.  The function respects the 1000 s time limit
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 0. Choose a sensible length.  Larger n gives a finer discretisation
    #    of the continuous ideal, but also costs more per iteration.
    # ------------------------------------------------------------------
    n = 7000                # comfortable size for the 1000 s budget
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Warm‑start: try to use a previously discovered sequence if
    #    available; otherwise fall back to a simple power law.
    # ------------------------------------------------------------------
    best_seq = None
    try:
        # ``height_sequence_1`` may be present globally.
        hs = globals().get("height_sequence_1")
        if hs is not None and len(hs) == n:
            best_seq = np.array(hs, dtype=float)
            # enforce exact sum (project onto simplex)
            best_seq = _project_simplex(best_seq, target_sum)
    except Exception:
        best_seq = None

    if best_seq is None:
        # Power‑law start (p = 0.5, zero offset)
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    # ------------------------------------------------------------------
    # 2. Coarse random exploration of the 4‑parameter family.
    # ------------------------------------------------------------------
    best_score = _max_conv(best_seq)          # because Σa = √(2n) ⇒ score = max conv
    print(f"[init] n={n:5d}  score={best_score:.6f}")

    # allocate roughly 30 % of the budget to the random sweep
    sweep_deadline = time.time() + 0.30 * (deadline - time.time())
    while time.time() < sweep_deadline:
        # Sample parameters
        p = np.random.uniform(0.35, 0.65)               # exponent around 0.5
        off = np.random.uniform(0.0, 10.0)              # offset
        beta = np.random.uniform(-0.8, 2.5)             # tilt of the tail
        cand = _make_candidate(n, p, off, beta)
        cand_score = _max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[sweep] new best {best_score:.6f}  p={p:.3f} α={off:.3f} β={beta:.3f}")

        # early exit if we are already below the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent starting from the best candidate.
    # ------------------------------------------------------------------
    curr = best_seq.copy()
    step0 = 1.0                     # initial step size (will decay)

    iteration = 0
    # protect against division by zero in the very unlikely case
    if target_sum == 0.0:
        target_sum = 1.0

    while time.time() < deadline:
        iteration += 1
        # Full convolution (FFT) – needed both for the value and for the set
        # of indices where the maximum is attained.
        conv_full = _full_convolution_fft(curr)
        M = conv_full.max()

        # Evaluate and possibly store the best solution.
        # With Σa = √(2n) the score equals M.
        if M < best_score:
            best_score = M
            best_seq = curr.copy()
            print(f"[sgd] iter {iteration:5d}  new best {best_score:.6f}")

            if best_score < 1.503:      # target reached – stop early
                break

        # Determine which convolution entries are (essentially) maximal.
        # A small tolerance avoids missing ties caused by round‑off.
        eps = 1e-12 * M if M != 0 else 1e-12
        max_inds = np.where(conv_full >= M - eps)[0]

        # Step size decays like 1/√t
        step = step0 / np.sqrt(iteration)

        # Perform the sub‑gradient move.
        cand = _subgradient_step(curr, max_inds, step, target_sum)

        # Occasionally add a tiny random jitter (1 % of the current values)
        # to avoid stagnation.
        if iteration % 50 == 0 and np.random.rand() < 0.3:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * cand
            cand = cand + jitter
            cand = _project_simplex(cand, target_sum)

        curr = cand

    # ------------------------------------------------------------------
    # 4. Final reporting
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    # Convert to plain Python list as required by the harness.
    return best_seq.tolist()
