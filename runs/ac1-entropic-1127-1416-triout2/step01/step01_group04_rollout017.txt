```
def proj_simplex(v):
    # v may be negative; we'll project onto {x >= 0, sum(x) = 1}
    # Algorithm: sort, find threshold
    if np.sum(v) == 1 and np.all(v >= 0): return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

```
def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, scipy.optimize...
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_score = np.inf
    best_seq = None
    # choose possible lengths
    lengths = [200, 300, 400, 500]   # small for speed
    # iterate over lengths while time left
    for n in lengths:
        # if time left insufficient break
        if time.time() >= deadline: break
        sum_target = np.sqrt(2.0 * n)
        # Build base
        base = _constant_conv_base(n)
        f = base / base.sum() * sum_target
        # Compute initial score
        conv = np.convolve(f, f)
        M = conv.max()
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"Init length {n} score {cur_score}")
        # iteration loop
        t = 0.2
        iter = 0
        max_iters = 50
        while time.time() < deadline and iter < max_iters:
            # Build matrix
            A_ub = build_conv_ub_matrix(f)
            b_ub = np.full(2*n-1, M)
            c = -np.ones(n)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n,
                          method='highs', options={'presolve': True})
            if not res.success:
                print(f"LP failed at iter {iter} length {n}")
                break
            g0 = res.x
            sum_g0 = g0.sum()
            if sum_g0 <= 0:
                break
            scaling = sum_target / sum_g0
            g = g0 * scaling
            # convex combo
            f_new = (1 - t) * f + t * g
            # renormalize exactly
            f_new *= sum_target / f_new.sum()
            conv_new = np.convolve(f_new, f_new)
            M_new = conv_new.max()
            # accept if no increase
            if M_new <= M * (1 + 1e-8):
                f = f_new
                M = M_new
                if M < best_score:
                    best_score = M
                    best_seq = f.copy()
                    print(f"Iter {iter} len {n} improved score {best_score}")
                    # optionally reduce t
            else:
                # reduce step size
                t *= 0.5
                if t < 1e-4:
                    break
            iter += 1
        # after iterations, maybe random perturb before next length
    # final check evaluate_sequence for best_seq
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluate_sequence score: {final_score}")
    except Exception:
        final_score = best_score
    print(f"Search completed. Best score {best_score}")
    return best_seq.tolist()
```

    Search for a step‑function giving a low value of ``evaluate_sequence``.
    The search combines a cheap random‑parameter sweep over simple parametric
    families (power‑law and beta‑distribution shapes) with a simple local
    refinement.  Sequences are always projected onto the simplex
    (non‑negative, fixed L1‑norm) before evaluation.  The routine stops
    after ``budget_s`` seconds and returns the best sequence found.
    """

    import numpy as np
    from scipy.optimize import linprog
    import scipy.sparse as sp

    deadline = time.time() + budget_s - 0.5

        n = len(height_sequence_1)

        n = 2000

    # ``evaluate_sequence`` equals the maximal auto‑convolution (the
    # quantity we are trying to minimise).
    sum_target = np.sqrt(2.0 * n)

    # --------------------------------------------------------------------
    # A simple helper: project onto the simplex Σx= sum_target , x≥0.
    # --------------------------------------------------------------------
    def project_to_simplex(v):
        """Scale v so that it lies in the simplex {x≥0, Σx = sum_target}."""
        # standard Euclidean projection onto the probability simplex,
        # then scale by sum_target.
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.where(u > cssv / np.arange(1, len(v) + 1))[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        # now Σw = 1, rescale
        if w.sum() > 0:
            w *= sum_target / w.sum()
        else:
            # fallback to uniform
            w = np.full_like(v, sum_target / len(v))
        return w

    # Baseline: the classic “constant‑convolution’’ sequence.
    # --------------------------------------------------------------------

        a = np.empty(n, dtype=np.float64)

"""
Search for a step‑function that minimises `evaluate_sequence`.

The algorithm mixes three ideas:
1.  A quick random sweep over two low‑dimensional families of shapes
    (power‑law and beta‑distribution) that are always projected onto the
    simplex {x≥0, Σx = √(2 n)} – this guarantees that the value returned by
    `evaluate_sequence` equals the maximal self‑convolution.
2.  Every few hundred candidates a cheap projected sub‑gradient step
    (the same update used in the reference solution) is applied to the
    current best sequence; it often squeezes a few extra digits.
3.  The whole search stops after the allotted wall‑clock time and returns
    the best sequence seen.

All arithmetic is performed with numpy; only the builtin `evaluate_sequence`
function is used for the final score.
"""

import time
import numpy as np


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = target_sum}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if target_sum <= 0:
        raise ValueError("target_sum must be positive")
    # First project onto the probability simplex (sum = 1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    rho = np.where(u > cssv / ind)[0]
    if rho.size == 0:               # all entries are zero – fall back to uniform
        w = np.full_like(v, 1.0 / len(v))
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        w_sum = w.sum()
        if w_sum == 0:               # numerical safety
            w = np.full_like(v, 1.0 / len(v))
    # Rescale to the requested sum
    w *= target_sum / w.sum()
    return w


def _constant_conv_sequence(n: int, target_sum: float) -> np.ndarray:
    """
    Classic “constant‑convolution’’ sequence.
    a_0 = 1, a_k = a_{k‑1}·(1‑1/(2k)).
    Afterwards the vector is scaled to have L1‑norm `target_sum`.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    a = a / a.sum() * target_sum
    return a


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for minimising max(conv(a,a)).
    Returns the updated vector (still non‑negative, not yet normalised).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    kmax = int(np.argmax(conv))

    g = np.zeros_like(a)

    if kmax < n:
        g[:kmax + 1] = 2.0 * a[kmax::-1]
    else:
        start = kmax - (n - 1)
        length = 2 * n - 1 - kmax
        g[start:] = 2.0 * a[:length][::-1]

    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    return a_new


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point.  Returns a list of non‑negative heights that gives a
    low value of `evaluate_sequence`.  The search respects the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose a length.  Re‑use a previous solution if it exists.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 2000

    # Target L1‑norm – with this normalisation the score returned by
    # `evaluate_sequence` is exactly the maximal self‑convolution.
    sum_target = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Start from the classic constant‑convolution construction.
    # --------------------------------------------------------------
    best_seq = _constant_conv_sequence(n, sum_target)
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        # should never happen – fallback to a large number
        best_score = float('inf')
    print(f"Initial constant‑conv score {best_score:.6f}")

    it = 0
    # ----------------------------------------------------------------
    # Randomised search over two simple parametric families.
    # ----------------------------------------------------------------
    while time.time() < deadline:
        # ---- choose a family -------------------------------------------------
        if np.random.rand() < 0.5:
            # Power‑law family: a_i ∝ (i + δ)^{-p}
            p = np.random.uniform(0.2, 0.9)               # exponent
            # δ on a log‑scale, avoiding zero
            log_delta = np.random.uniform(-6.0, np.log10(n))
            delta = 10.0 ** log_delta
            idx = np.arange(n, dtype=np.float64)
            a = (idx + delta) ** (-p)
        else:
            # Beta‑distribution family on the discrete grid.
            alpha = np.random.uniform(0.2, 3.0)
            beta = np.random.uniform(0.2, 3.0)
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

        # ---- project onto the simplex ----------------------------------------
        if a.sum() == 0.0:
            continue
        a = _project_to_simplex(a, sum_target)

        # ---- occasional jitter to escape plateaus ----------------------------
        if it % 1000 == 0 and it > 0:
            jitter = np.random.randn(n) * 1e-6
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, sum_target)

        # ---- evaluate ---------------------------------------------------------
        try:

        except Exception:

        if it % 500 == 0 and it > 0:

            except Exception:
                pass

        it += 1

    # Final sanity check using the provided evaluate_sequence().
    # ----------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Search finished after {it} iterations, final score {final_score:.6f}")
    except Exception:
        final_score = best_score
        print(f"Search finished after {it} iterations, best recorded score {best_score:.6f}")

    return best_seq.tolist()
