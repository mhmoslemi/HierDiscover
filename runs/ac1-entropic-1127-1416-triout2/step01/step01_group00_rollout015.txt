"""
Propose a low‑value step‑function sequence.

The algorithm proceeds in three stages:

1.  **Parametric initialization** – for a given length `n` we generate
    symmetric “Beta‑type’’ shapes a_i ∝ ((i+½)(n‑½‑i))**α and
    search a one‑dimensional exponent α (α≈‑0.5 reproduces the arcsine
    density).  The best α is selected by the fast scale‑invariant score
    2·n·max(a∗a).

2.  **Smoothed max‑convolution descent** – starting from the best
    parametric shape we minimise the smoothed objective
        Lβ(a)= (log Σ_i exp(β·(a∗a)_i))/β .
    The gradient of Lβ is a weighted correlation 2·a_rev ⋆ w
    (w is the soft‑max weight vector).  A projected gradient method with
    diminishing step size and a simple momentum term is used, all the
    while keeping `a` on the probability simplex.

3.  **Length sweep & final evaluation** – the procedure is repeated for
    several lengths (powers of two up to the time budget).  The best
    sequence found (by the official `evaluate_sequence`) is returned.

All helper routines are top‑level, use only NumPy, and respect the
1000‑second time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto {x | x_i≥0, Σx_i = 1}. """
    if np.all(v <= 0):
        # uniform vector is the projection of the non‑positive orthant
        return np.full_like(v, 1.0 / v.size)

    # Duchi et al. (2008) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """ smallest power of two ≥ x """
    return 1 << (x - 1).bit_length()


def convolve_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """ Full linear convolution a * b using real FFT (output real). """
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast (scale‑invariant) evaluation used during the search
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Assumes sum(a)=1.  Returns 2·n·max(a∗a)."""
    n = a.size
    conv = convolve_full_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Parametric family a_i ∝ ((i+0.5)*(n-0.5-i))**α
# ----------------------------------------------------------------------
def beta_shape(alpha: float, n: int) -> np.ndarray:
    """ Return a symmetric Beta‑type shape (α may be negative). """
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - 0.5 - i)) ** alpha
    # small epsilon prevents exact zeros for extreme α (not really needed)
    vals = np.maximum(vals, 0.0)
    return vals / np.sum(vals)


def best_alpha_for_n(n: int, alphas: np.ndarray) -> tuple[np.ndarray, float]:
    """Search the best α in the supplied list, return (a, score)."""
    best_a = None
    best_sc = np.inf
    for a in alphas:
        seq = beta_shape(a, n)
        sc = fast_score(seq)
        if sc < best_sc:
            best_sc = sc
            best_a = seq
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smoothed max‑convolution gradient descent
# ----------------------------------------------------------------------
def smoothed_descent(
    a0: np.ndarray,
    deadline: float,
    beta: float = 50.0,
    eta0: float = 0.7,
    momentum: float = 0.9,
    max_iter: int = 200_000,
    noise_every: int = 500,
    noise_scale: float = 4e-5,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth objective
        Lβ(a) = log_sum_exp(β·(a∗a)) / β
    with a simple momentum term.  Returns the best (a,score) seen.
    """
    a = a0.astype(np.float64).copy()
    a = project_to_simplex(a)
    n = a.size
    best_a = a.copy()
    best_sc = fast_score(a)

    # momentum buffer
    vel = np.zeros_like(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # ---- convolution a∗a and soft‑max weights ----
        conv = convolve_full_fft(a, a)               # length 2n-1
        max_val = np.max(conv)

        # stabilise exponentials
        shift = conv - max_val
        w = np.exp(beta * shift)
        w_sum = np.sum(w)
        w /= w_sum                                   # now w_i = soft‑max weight

        # ---- gradient: 2·(a_rev ⋆ w) sliced to length n ----
        a_rev = a[::-1]
        grad_full = convolve_full_fft(a_rev, w)     # length 3n‑2
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]

        # ---- step size schedule ----
        step = eta0 / np.sqrt(it)

        # ---- momentum update ----
        vel = momentum * vel - step * grad
        a_new = a + vel

        # occasional isotropic noise to escape flat plateaus
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        # ---- projection back onto the simplex ----
        a_new = project_to_simplex(a_new)

        # ---- keep the best sequence (according to the exact fast score) ----
        sc = fast_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # ---- optional progress report ----
        if it % 5000 == 0:
            print(f"   n={n}  iter={it:,d}  cur_score={sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc

    """

    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0      # small safety margin

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None

    # ------------------------------------------------------------------
    try:

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        seq_alpha, score_alpha = best_alpha_for_n(n, coarse_alphas)
        # --------------------------------------------------------------

        # --------------------------------------------------------------
        best_alpha = -0.5  # default arcsine

    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
