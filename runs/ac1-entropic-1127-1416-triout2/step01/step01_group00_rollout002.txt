def lp_optimize(initial_a, deadline, max_iter=30, t=0.5):
    a = initial_a.copy()
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    iter_cnt=0
    while time.time() < deadline and iter_cnt < max_iter:
        iter_cnt += 1
        # compute M
        conv = convolution_fft(a, a)
        M = float(np.max(conv))
        # Build convolution matrix
        # Build A_ub
        rows = []
        cols = []
        data = []
        for i in range(n):
            rows_i = i + np.arange(n)
            cols_i = np.arange(n)
            rows.append(rows_i)
            cols.append(cols_i)
            data.append(np.full(n, a[i]))
        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        data = np.concatenate(data)
        A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failed at iter {iter_cnt}, status: {res.message}")
            break
        b_opt = res.x
        sum_b = np.sum(b_opt)
        if sum_b <= 0:
            break
        g = b_opt / sum_b
        # update
        a = (1 - t) * a + t * g
        a = project_to_simplex(a)  # ensure sum=1 exactly
        # evaluate score
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # optionally early stopping if improvement < small epsilon over last few.
        if iter_cnt % 5 == 0:
            print(f"LP iter {iter_cnt} score {cur_score:.6f} best {best_score:.6f}")
    return best_a, best_score

"""
Algorithm
---------

We minimize the evaluation function

    E(a) = 2·n·max(a*a) / (sum(a)^2)

which is scale‑invariant.  We fix the sum to 1 (probability simplex) and
minimise `max(a*a)`.  Two complementary optimisation phases are used:

1. **Alternating linear‑programming phase** (inspired by the paper’s LP
   step).  With a current vector `f` (∑f=1) we compute
   `M = max(f*f)`.  The LP

       maximise   1ᵀ b
       subject to f*b ≤ M   (component‑wise)
                b ≥ 0

   finds a vector `b` that can be re‑scaled to a new candidate `g`
   (∑g=1) with a strictly smaller self‑convolution bound.
   The update `f ← (1−t)·f + t·g` (t≈0.5) is repeated a few dozen times,
   each iteration solving a modest‑size LP (≈2 n constraints, n variables).

2. **Projected sub‑gradient refinement** – a classic sub‑gradient descent
   on the convex objective `max(f*f)` keeping `f` on the simplex.
   This polishing step removes the remaining (tiny) sub‑optimality left
   by the LP phase.

Both phases are time‑controlled: they stop when the global budget
(≈1000 s) is exhausted.  The best sequence found is returned as a plain
Python list of floats.  If SciPy is unavailable the code gracefully falls
 back to the pure sub‑gradient optimiser.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑programming support (optional – SciPy may be missing)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
except Exception:                     # pragma: no cover
    linprog = None
    csr_matrix = None

# ----------------------------------------------------------------------
# Utility functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> tuple[float, int]:
    """Return (max value, index) of the full self‑convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the entry `c_k = (a*a)[k]` with respect to `a`.
    The derivative is a shifted copy of `a`, with the diagonal doubled.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def compute_score(a: np.ndarray) -> float:
    """
    Scale‑invariant score used during optimisation.
    The sequence is assumed to be normalised to sum = 1, so the formula
    reduces to 2·n·max(a*a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Continuous arcsine density discretised on n points.
    a_i ∝ 1/√((i+½)(n‑½‑i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Linear‑programming step (alternating optimisation)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> csr_matrix:
    """
    Construct the (2n‑1)×n Toeplitz matrix representing the convolution
    `c = f * b`.  Row k contains the coefficients of `b` that appear in
    the k‑th entry of `c`.
    """
    n = len(f)
    # Pre‑allocate arrays for the sparse matrix (n·n non‑zeros)
    data = np.empty(n * n, dtype=f.dtype)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)

    pos = 0
    for i, fi in enumerate(f):
        # For this i, the coefficient fi appears in rows i … i+n‑1,
        # once per column j = 0 … n‑1.
        rows[pos:pos + n] = i + np.arange(n, dtype=np.int32)
        cols[pos:pos + n] = np.arange(n, dtype=np.int32)
        data[pos:pos + n] = fi
        pos += n

    mat = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return mat


def lp_alternating_optimize(initial_a: np.ndarray,
                            deadline: float,
                            max_iter: int = 30,
                            t: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Alternating optimisation based on the LP described in the statement.
    The vector is always projected onto the simplex (∑=1).
    """
    if linprog is None:
        # SciPy missing – fall back to the current vector.
        return initial_a, compute_score(initial_a)

    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # 1. current bound M = max(f*f)
        conv_ff = convolution_fft(a, a)
        M = float(np.max(conv_ff))

        # 2. build Toeplitz matrix for convolution f * b
        A_ub = build_convolution_matrix(a)

        # 3. linear programme: maximise sum(b)  ⇔ minimise -1ᵀb
        c = -np.ones(n, dtype=float)
        b_ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0, None)] * n

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(1.0, (deadline - time.time()))})

        if not res.success:
            # Something went wrong – abort the LP loop.
            print(f"LP failed at iteration {it}: {res.message}")
            break

        b_opt = res.x
        sum_b = np.sum(b_opt)
        if sum_b <= 0:
            # Degenerate solution, stop.
            break

        # 4. scale to the simplex (∑=1)
        g = b_opt / sum_b

        # 5. convex combination
        a = (1.0 - t) * a + t * g
        a = project_to_simplex(a)  # re‑normalise exactly

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if it % 5 == 0:
            print(f"  LP iter {it:2d}  score={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (fine‑tuning)
# ----------------------------------------------------------------------


def improve_sequence(initial_a: np.ndarray,
                    deadline: float,
                    max_iter: int = 200_000,
                    eta0: float = 0.5,
                    noise_every: int = 500,
                    noise_scale: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the scale‑invariant objective.
    The vector stays on the probability simplex.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # maximal convolution entry and its index
        _, k_max = max_convolution_and_index(a)

        # sub‑gradient at the active index
        grad = subgradient_of_max_at_index(a, k_max)

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        a_new = a - eta * grad

        # tiny stochastic perturbation for escaping plateaus
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if it % 2000 == 0:
            print(f"  subgrad iter {it:,d}  score={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score
