    from scipy.optimize import linprog

    start_time = time.time()

    np.random.seed(seed)

    best_score = float('inf')
    best_seq = None

    # choose candidate lengths

    # iterate lengths approximate
    for n in lengths:

        t = 0.3   # step weight

            cols = []
            data = []

                fi = cur[i]

"""
propose_candidate
-----------------
Implementation of a deterministic “alternating linear‑program” search.
For a fixed length n the algorithm repeatedly

1. evaluates the current maximal self‑convolution value M = max(a*a),
2. solves a linear program that maximises the total mass Σ g_i under the
   linear constraints (a * g)_k ≤ M for all convolution positions k,
3. rescales the LP solution g to the required L1‑norm √(2n) and moves the
   current sequence a a fraction t towards g.

Each iteration never increases M, so the evaluated score
    2·n·max(a*a) / (Σ a)^2
converges downwards.  After a few LP steps a short random‑move refinement
is applied.  The routine tries several lengths (≤ 2000) and keeps the best
candidate found before the time budget expires.

The code works with only NumPy and SciPy (highs linear‑program solver).
If SciPy is unavailable it falls back to the simple power‑law + random walk
method used previously.
"""

import time
import numpy as np

# evaluate_sequence is provided by the execution environment.

# ----------------------------------------------------------------------
# Helper to build the sparse convolution‑matrix index pattern (independent of a)
# ----------------------------------------------------------------------
def _conv_matrix_indices(n: int):
    """Return row and column index arrays for the matrix A where
       (A @ g)[k] = (a * g)[k] for a of length n.
       The shape of A is (2*n-1, n) and each row k contains the
       coefficients a_i for all i with 0≤i<n and 0≤k-i<n.
    """
    i = np.arange(n, dtype=np.int64)[:, None]          # shape (n,1)
    j = np.arange(n, dtype=np.int64)[None, :]          # shape (1,n)
    rows = (i + j).ravel()                             # i+j for each pair
    cols = np.tile(np.arange(n, dtype=np.int64), n)    # repeat 0..n-1 for each i
    return rows, cols


# ----------------------------------------------------------------------
# Small fallback based on the power‑law + random walk that was used before
# ----------------------------------------------------------------------
def _fallback_propose(seed: int, deadline: float):
    """Simple random‑walk optimisation – used when SciPy is not available."""
    np.random.seed(seed)
    best_seq = None
    best_score = float("inf")
    # Very coarse grid to get a starting point
    n = 2000
    target = np.sqrt(2.0 * n)
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a = a * (target / a.sum())
    cur = a.copy()
    best_seq = cur.tolist()
    best_score = evaluate_sequence(best_seq)

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_move = 0.01 * cur[i]
        if max_move <= 0:
            continue
        delta = np.random.uniform(-max_move, max_move)
        if cur[i] - delta < 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score, best_seq, cur = sc, cand.tolist(), cand
            print(f"fallback improvement: {sc:.6f}")
        if best_score < 1.503:
            break
    return best_seq


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the evaluation function.  The algorithm runs for at most *budget_s* seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin for cleanup

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try to import SciPy – if unavailable, fall back to simple method.
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csc_matrix
        scipy_available = True
    except Exception:  # pragma: no cover
        print("SciPy not available – using fallback optimisation.")
        return _fallback_propose(seed, deadline)

    # ------------------------------------------------------------------
    # 1. Choose a few problem sizes (n).  Larger n gives more freedom but
    #    makes the LP larger.  1200–2000 is a good trade‑off for the time limit.
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 1800, 2000]

    for n in lengths:
        if time.time() > deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1a. Initialise with the classic a_i ∝ 1/√(i+1)
        # --------------------------------------------------------------
        a0 = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
        a0 = a0 * (target_sum / a0.sum())
        cur = a0.copy()

        # initialise best for this length
        cur_score = evaluate_sequence(cur.tolist())
        if cur_score < best_score:
            best_score, best_seq = cur_score, cur.tolist()
            print(f"[n={n}] initial score = {cur_score:.6f}")

        # --------------------------------------------------------------
        # 1b. Pre‑compute sparse matrix index pattern (independent of a)
        # --------------------------------------------------------------
        rows_idx, cols_idx = _conv_matrix_indices(n)

        # --------------------------------------------------------------
        # 1c. Alternating LP optimisation – run as many iterations as time permits
        # --------------------------------------------------------------
        t = 0.5                 # convex‑combination weight
        max_iter = 12          # hard cap per length

        for it in range(max_iter):
            if time.time() > deadline:
                break

            # ----- current maximal self‑convolution -----
            conv = np.convolve(cur, cur)
            M = conv.max()
            # ----- build constraints matrix A (a * g ≤ M) -----
            # data are repetitions of the current a‑vector
            data = np.repeat(cur, n)
            A = csc_matrix((data, (rows_idx, cols_idx)), shape=(2 * n - 1, n))

            # ----- linear programme: maximise Σ g_i  (i.e. minimise -Σ g_i) -----
            c = -np.ones(n, dtype=float)                 # minimise –sum(g)
            b_ub = np.full(2 * n - 1, M, dtype=float)   # right‑hand side
            bounds = [(0.0, None)] * n

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success:
                print(f"[n={n}, it={it}] LP failed: {res.message}")
                break

            g_raw = res.x
            sum_g = g_raw.sum()
            if sum_g == 0:
                print(f"[n={n}, it={it}] LP returned zero mass – aborting this n.")
                break

            # ----- rescale g to have the required L1‑norm -----
            g = g_raw * (target_sum / sum_g)

            # ----- take a convex step towards g -----
            cand = (1.0 - t) * cur + t * g

            # evaluate candidate
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.tolist()
                print(f"[n={n}, it={it}] new global best = {cand_score:.6f}")

            # accept the move only if it does not increase the score
            if cand_score <= cur_score:
                cur = cand
                cur_score = cand_score
                # optionally tighten step size as we get closer
                t = min(0.9, t * 1.05)
            else:
                # reject – shrink the step and try again
                t *= 0.5
                if t < 1e-4:
                    # step too tiny, stop iterating this length
                    break

            # early exit if we already beat the target
            if best_score < 1.503:
                print("Target 1.503 achieved – exiting early.")
                return best_seq

        # --------------------------------------------------------------
        # 2. Short random‑move refinement (as in the original script)
        # --------------------------------------------------------------
        n = len(cur)
        while time.time() < deadline:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            max_move = 0.01 * cur[i]
            if max_move <= 0:
                continue
            delta = np.random.uniform(-max_move, max_move)
            if cur[i] - delta < 0:
                continue
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best_score, best_seq = sc, cand.tolist()
                cur = cand
                print(f"random refinement improvement: {sc:.6f}")
                if best_score < 1.503:
                    print("Target reached during refinement.")
                    return best_seq
            # tiny random moves cheap enough – continue until deadline

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    if best_seq is None:
        # Safety net – a uniform sequence (score = 2)
        n = 1000
        uni = [np.sqrt(2.0 * n) / n] * n
        best_seq = uni
        best_score = evaluate_sequence(best_seq)

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
