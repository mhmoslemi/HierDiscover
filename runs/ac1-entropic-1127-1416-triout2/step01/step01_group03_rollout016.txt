def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, scipy.optimize

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose n
    try: n = len(height_sequence_1)
    except: n = 1000

    # start sequence
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        # project to simplex
        a = project_to_simplex(a)
    else:
        a = sqrt_distribution(n).copy()
        a = project_to_simplex(a)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={len(a)} score={best_score}")

    # algorithm phases
    phase = 'lp'
    iteration = 0
    stagnant = 0

    while time.time() < deadline:
        # compute convolution and max
        conv = np.convolve(a, a)
        M = conv.max()
        # build matrix A_ub
        rows = np.arange(2*len(a)-1)[:, None]
        cols = np.arange(len(a))[None, :]
        idx = rows - cols
        mask = (idx >= 0) & (idx < len(a))
        A = np.zeros((2*len(a)-1, len(a)), dtype=np.float64)
        A[mask] = a[idx[mask]]

        # solve LP
        c = -np.ones(len(a))
        bounds = [(0, 1000.0)] * len(a)
        b_ub = np.full(2*len(a)-1, M, dtype=np.float64)

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"[{iteration}] LP failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            print("lp produced zero sum")
            break
        g = g / sum_g  # normalized

        # step size adapt
        t = 0.5
        # ensure monotonic decrease in max
        while t > 1e-8:
            a_candidate = (1 - t) * a + t * g
            # optional re-project to simplex
            # but sum is already 1, ensure numeric
            # a_candidate = project_to_simplex(a_candidate)
            conv_new = np.convolve(a_candidate, a_candidate)
            M_new = conv_new.max()
            if M_new <= M + 1e-12:
                break
            t *= 0.5

        # if t tiny, we won't improve, break
        if t <= 1e-8:
            stagnant += 1
        else:
            a = a_candidate
            stagnant = 0

        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:05d}] improved score: {best_score:.6f}")

        if stagnant > 1000:
            # jitter or restart
            jitter = 0.001 * np.random.randn(len(a))
            a = project_to_simplex(a + jitter)
            stagnant = 0
            print(f"[{iteration}] jitter")

        iteration += 1

    return best_seq.tolist()

"""
Frank–Wolfe style optimizer for the step‑function problem.

The algorithm works on a probability simplex (∑a_i = 1) because the
evaluation function is homogeneous:  score = 2·n·max(a*a)/(∑a)^2.
A lower bound on the max convolution can be achieved by repeatedly
solving a linear programme

    maximise   ∑ b_i
    subject to  a * b ≤ M            (pointwise)
                b_i ≥ 0

where M = max(a*a).  The LP gives a direction b that raises the L1‑norm
as much as possible while keeping the maximum convolution unchanged.
A convex combination a←(1‑t)a + t·b (with the same L1‑norm) therefore
does not increase M and usually reduces the score.  The LP is solved
with SciPy’s high‑performance “highs” solver; a sparse constraint
matrix is built from the current a.  When progress stalls a small random
jitter (or a full random restart) is injected.

The routine runs for up to 1000 s and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, required by the platform)
# ----------------------------------------------------------------------


def sqrt_distribution(n: int) -> np.ndarray:
    """Normalised coefficients of (1‑z)^(-1/2) up to length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def project_to_simplex(v: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Scale a non‑negative vector to the probability simplex (sum=1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= eps:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def build_conv_matrix(a: np.ndarray):
    """
    Build the (2·n‑1)×n convolution matrix A such that A @ b = a * b.
    It is stored as a SciPy CSR sparse matrix to keep memory modest.
    """
    from scipy.sparse import diags
    n = a.size
    # each diagonal offset –l (l = 0…n‑1) contains the constant a[l]
    offsets = -np.arange(n, dtype=int)          # −0, −1, …, −(n‑1)
    data = [np.full(n, a[l], dtype=np.float64) for l in range(n)]
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def smooth_max_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the soft‑max surrogate:
        g_i = 2·∑_j a_j·w_{i+j}
    where w is a soft‑max over the self‑convolution a*a.
    Returns (gradient, current max of a*a, full convolution).
    """
    b = np.convolve(a, a)               # length 2n‑1
    max_b = b.max()
    w = np.exp(tau * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    conv = np.convolve(a, w_rev)        # length 3n‑2
    L = b.size
    grad = 2.0 * conv[(L - 1) - np.arange(a.size)]
    return grad, max_b, b


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluate_sequence()
    score.  The core loop solves a linear programme that maximises the L1‑norm
    while preserving the current maximum convolution, then takes a convex
    step towards the LP solution.  Occasional jitter / restart avoids stagnation.
    """
    import scipy.optimize

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Determine problem size.
    # --------------------------------------------------------------
    global height_sequence_1  # may be defined by the harness
    if 'height_sequence_1' in globals():
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        # keep the length moderate to stay within time / memory limits
        n = min(init_seq.size, 1024)
        a = init_seq[:n].copy()
    else:
        n = 1024
        a = sqrt_distribution(n)

    # Normalise to a probability simplex (∑a = 1)
    a = project_to_simplex(a)

    best_seq = a.copy()
    # score = 2·n·max(a*a) because ∑a = 1
    cur_conv = np.convolve(a, a)
    best_score = 2.0 * n * cur_conv.max()
    print(f"[init] n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Main Frank–Wolfe style optimisation loop
    # --------------------------------------------------------------
    iteration = 0
    stagnant = 0
    max_stagnant = 2000          # after this inject jitter
    max_restart = 8000           # after this perform full restart
    t_initial = 0.5              # starting convex‑combination weight
    while time.time() < deadline:
        # current max convolution
        conv = np.convolve(a, a)
        M = conv.max()

        # Build the constraint matrix A (sparse)
        A = build_conv_matrix(a)

        # Linear programme: maximise sum(b)  ⇔ minimise -∑b
        c = -np.ones(n, dtype=np.float64)               # objective: minimise -∑b
        b_ub = np.full(2 * n - 1, M, dtype=np.float64) # RHS = M for every row
        bounds = [(0, None)] * n                        # b_i ≥ 0, no upper cap

        # Solve the LP
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds,
                                    method='highs',
                                    options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            print(f"[{iteration}] LP failed ({res.message}) – terminating loop.")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            print(f"[{iteration}] LP returned all‑zero solution – terminating.")
            break

        # Normalise the LP solution to the simplex (same L1‑norm as a)
        g = g / sum_g

        # ----------------------------------------------------------
        # Take a convex step a ← (1‑t)a + t·g, decreasing t until M does not increase
        # ----------------------------------------------------------
        t = t_initial
        while t > 1e-8:
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = project_to_simplex(a_candidate)   # keep ∑=1
            conv_new = np.convolve(a_candidate, a_candidate)
            M_new = conv_new.max()
            if M_new <= M + 1e-12:         # non‑increasing max‑convolution
                break
            t *= 0.5

        if t <= 1e-8:
            # No improvement – treat as stagnation
            stagnant += 1
        else:
            a = a_candidate
            stagnant = 0
            # compute current score and keep best
            cur_score = 2.0 * n * M_new               # ∑a = 1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{iteration:06d}] improved score = {best_score:.6f}")

        # ----------------------------------------------------------
        # Simple stagnation handling
        # ----------------------------------------------------------
        if stagnant > max_stagnant:
            # tiny random jitter to escape flat region
            jitter = 0.002 * np.random.randn(n)
            a = project_to_simplex(a + jitter)
            stagnant = 0
            print(f"[{iteration}] jitter added (t={t:.3g})")

        if stagnant > max_restart:
            # full random restart (keep best found so far)
            a = np.random.rand(n)
            a = project_to_simplex(a)
            stagnant = 0
            print(f"[{iteration}] random restart")

        iteration += 1

        # ----------------------------------------------------------
        # Optional quick refinement with smooth‑max gradient descent
        # ----------------------------------------------------------
        if time.time() > deadline - 30.0:   # use the last ~30 s for fine‑tuning
            # a few steps of exponentiated‑gradient with a high temperature
            tau = 30.0
            eta = 0.5
            for _ in range(300):
                grad, max_b, _ = smooth_max_gradient(a, tau)
                a = a * np.exp(-eta * grad)
                a = project_to_simplex(a)
                # early exit if time runs out
                if time.time() >= deadline - 0.5:
                    break
            # final evaluation after refinement
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[final‑refine] new best score = {best_score:.6f}")
            break   # leave the main loop – budget almost exhausted

    print(f"[done] best score = {best_score:.6f} after {iteration} iterations")
    # Return as plain Python list
    return best_seq.tolist()
