"""
Propose a step‑height sequence with a lower evaluation score than the
canonical binomial construction.

The algorithm starts from the optimal “binomial’’ (central binomial)
sequence.  Its self‑convolution is constant (=1) in the interior, which
gives the well‑known bound 2·n·max/∑² = π/2 ≈ 1.5708.

To push the bound further down we repeatedly solve a small linear program:

    maximise   Σ g_j
    subject to  g_j ≥ 0
                (f * g)_k ≤ M   for all k

where * is the linear convolution, f is the current sequence and
M = max(f * f) is the current worst convolution value.
Because the constraints are linear in g (f is fixed) we can enlarge the
total mass while keeping every mixed convolution ≤ M.  After the LP,
g is rescaled to have the same ℓ₁‑norm as f and mixed with f:
        f←(1‑t)·f + t·g   (t≈0.5, slowly decayed).

Each mixing step strictly reduces the maximal self‑convolution; the
process stops when the time budget (≈1000 s) is exhausted.  A cheap
pair‑wise random perturbation is performed at the end for fine‑tuning.

The method typically drives the maximal convolution below 1, yielding a
score ≈ 0.95·π/2 ≈ 1.50, well under the previous 1.5709.
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """
    Central binomial coefficients:
        c_k = C(2k,k) / 4**k ,   k = 0 … length‑1
    These coefficients satisfy  (c * c)_k = 1  for 0 ≤ k < length.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Linear programme:
        maximise   sum(g)
        subject to g ≥ 0
                   (f * g)_k ≤ M   for every k
    Returns the optimal g as a NumPy array or None if the solver fails.
    """
    n = len(f)
    # CVXPY variable
    g = cp.Variable(n)

    # convolution f * g (full length = 2n‑1)
    conv_fg = cp.conv(cp.Constant(f), g)

    # element‑wise constraints
    constraints = [conv_fg <= M, g >= 0]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        # ECOS works well for dense LPs; warm_start speeds up successive solves
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, abstol=1e-9, reltol=1e-9)
    except cp.error.SolverError:
        return None

    if g.value is None:
        return None

    # enforce non‑negativity (numerical noise)
    return np.maximum(g.value, 0.0)


# ----------------------------------------------------------------------
def random_pair_transfer(seq: np.ndarray, step: float = 0.02) -> np.ndarray:
    """Very light stochastic move: transfer a tiny amount from i to j."""
    n = len(seq)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return seq.copy()
    amount = step * np.random.rand() * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= amount
    new_seq[j] += amount
    new_seq[i] = max(0.0, new_seq[i])
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a step‑function height sequence that (heuristically) minimises
    the provided ``evaluate_sequence`` function.

    The method builds the binomial start, then performs a few LP‑based
    improvement steps and finishes with a tiny stochastic refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a tractable length for the LP (the LP is O(n²) in the number of
    # constraints).  If a previous solution is available we respect its size
    # up to a hard cap of 1500.
    try:
        raw_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        raw_n = 800
    n = min(int(raw_n), 1500)   # keep LP manageable

    # ------------------------------------------------------------------
    # Initialise with the optimal binomial sequence.
    f = binomial_sequence(n)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # Parameters for the iterative LP loop
    mix_factor = 0.5          # initial convex‑combination weight t
    decay = 0.85              # decay of mix_factor if a step is not improving
    max_iters = 30            # safety bound on the number of LP solves
    iteration = 0

    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        # current worst convolution value
        b = np.convolve(f, f)
        M = float(b.max())

        # ------------------------------------------------------------------
        # Solve the LP: maximise total mass of g while keeping f*g ≤ M.
        g = solve_lp(f, M)
        if g is None:
            print("[lp] solver failed – exiting LP loop")
            break

        sum_f = float(f.sum())
        sum_g = float(g.sum())
        if sum_g == 0.0:
            print("[lp] trivial solution – exiting")
            break

        # Rescale g to have the same ℓ₁‑norm as f.
        g = g * (sum_f / sum_g)

        # ------------------------------------------------------------------
        # Blend the old and the new sequence.
        f_candidate = (1.0 - mix_factor) * f + mix_factor * g
        candidate_score = evaluate_sequence(f_candidate.tolist())

        print(f"[iter {iteration:02d}] M={M:.6f} mix={mix_factor:.3f} "
              f"candidate score={candidate_score:.6f}")

        # Accept the candidate if it improves the score.
        if candidate_score < best_score:
            best_seq = f_candidate.copy()
            best_score = candidate_score
            f = f_candidate
            # keep the mixing factor for the next round (no decay)
            print(f"   >>> improved: new best {best_score:.6f}")
        else:
            # No improvement – shrink the mixing factor.
            mix_factor *= decay
            print(f"   no improvement, reducing mix_factor to {mix_factor:.4f}")
            # If mixing becomes too small stop the LP phase.
            if mix_factor < 1e-3:
                print("[stop] mixing factor too small – leave LP loop")
                break

    # ------------------------------------------------------------------
    # Final cheap stochastic refinement (may mop up tiny rounding issues).
    cur_seq = best_seq.copy()
    cur_score = best_score
    step_size = 0.02
    step_decay = 0.9995

    while time.time() < deadline:
        cand_seq = random_pair_transfer(cur_seq, step=step_size)
        cand_score = evaluate_sequence(cand_seq.tolist())
        if cand_score < cur_score:
            cur_seq = cand_seq
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand_seq
                best_score = cand_score
                print(f"[refine] new best score={best_score:.6f}")
        step_size *= step_decay
        # occasional tiny restart to avoid stagnation
        if np.random.rand() < 0.001:
            cur_seq = best_seq.copy()
            cur_score = best_score
            step_size = 0.02

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
