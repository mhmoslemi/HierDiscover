    n = len(seq)
    size = 1 << ((2*n-1).bit_length())  # next power of two >= 2n-1

"""
A fast stochastic‑gradient search for the step‑function that minimises
``evaluate_sequence``.

The algorithm works in three stages:

1.  **Initialisation** – take the supplied reference sequence
    (``height_sequence_1``) if it exists, otherwise draw a few random
    candidates from several parametric families (exponential,
    power‑law, sinusoid, …).  The best among them becomes the current
    incumbent.

2.  **Gradient descent on the active convolution peak** – the full
    self‑convolution is obtained with an FFT (O(n log n)).  The index
    `k` where the convolution attains its maximum determines the
    sub‑gradient of the objective

            J(a) = log(max_b) - 2·log(∑a)

    (equivalent to the original evaluation function because of the
    scaling invariance).  The sub‑gradient with respect to a_i is

            g_i = 2·a_{k‑i} / max_b - 2 / Σa

    (terms outside the support are zero).  A small step
    ``seq ← seq – η·g`` is taken, the result is projected onto the
    non‑negative orthant and renormalised so that the maximal entry is
    1 (the objective does not depend on the global scale).  The step
    size η starts at 0.2 and decays slowly; occasional random
    perturbations are injected to escape shallow local minima.

3.  **Exploration** – with a modest probability a fresh candidate from
    the parametric families is sampled and, if it improves the best
    score, replaces the current incumbent.  This keeps the search
    diversified.

The loop runs until the time budget (default 1000 s) expires; the
best sequence seen is returned.  All heavy arithmetic uses NumPy, the
objective is evaluated via a cheap FFT‑based routine (identical to
``evaluate_sequence`` up to the final rounding) to stay well within the
time limit.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – fast evaluation (no security checks)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the objective value using an FFT based convolution.
    Identical to ``evaluate_sequence`` up to the scaling invariance.
    """
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 0.01:
        return math.inf
    # full linear convolution via FFT (length = 2n‑1)
    size = 2 * n - 1
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:size]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full (linear) convolution seq * seq using FFT."""
    n = seq.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(seq, n=size)
    return np.fft.irfft(f * f, n=size)[:size]


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Scale so that the maximal entry equals 1 (no‑op for all‑zero)."""
    mx = seq.max()
    if mx > 0:
        seq = seq / mx
    return seq


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _exp_seq(n: int, alpha: float) -> np.ndarray:
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)))

def _pow_seq(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    return _normalize(idx ** (-p))

def _lin_seq(n: int, beta: float) -> np.ndarray:
    return _normalize(np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float)))

def _mix_seq(n: int, alpha: float, eps: float) -> np.ndarray:
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)) + eps)

def _logistic_seq(n: int, steep: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mid = n / 2.0
    seq = 1.0 / (1.0 + np.exp((i - mid) / steep))
    return _normalize(seq)

def _randdec_seq(n: int) -> np.ndarray:
    arr = np.random.rand(n)
    arr.sort()
    return _normalize(arr[::-1])

def _sine_seq(n: int, power: float) -> np.ndarray:
    xs = np.arange(n, dtype=float) + 0.5
    seq = np.sin(np.pi * xs / n) ** power
    return _normalize(seq)

def _quad_seq(n: int, a: float) -> np.ndarray:
    t = np.arange(n, dtype=float) / (n - 1)
    seq = (1.0 - t ** 2) ** a
    return _normalize(seq)


_FAMILIES = [
    ('exp', _exp_seq, ('alpha', (1e-4, 5.0))),
    ('pow', _pow_seq, ('p', (0.0, 3.0))),
    ('lin', _lin_seq, ('beta', (0.0, 2.0))),
    ('mix', _mix_seq, ('alpha', (1e-4, 5.0), 'eps', (0.0, 0.5))),
    ('log', _logistic_seq, ('steep', (0.01, 2000.0))),
    ('randdec', _randdec_seq, ()),
    ('sine', _sine_seq, ('power', (0.2, 5.0))),
    ('quad', _quad_seq, ('a', (0.2, 5.0))),
]


def _sample_family(n: int) -> np.ndarray:
    """Pick a random family and random parameters, return a normalised sequence."""
    name, fn, *paraminfo = random.choice(_FAMILIES)
    if name == 'exp':
        alpha = 10 ** random.uniform(-4, math.log10(5))
        return fn(n, alpha)
    if name == 'pow':
        p = random.uniform(0.0, 3.0)
        return fn(n, p)
    if name == 'lin':
        beta = random.uniform(0.0, 2.0 / n)
        return fn(n, beta)
    if name == 'mix':
        alpha = 10 ** random.uniform(-4, math.log10(5))
        eps = random.uniform(0.0, 0.5)
        return fn(n, alpha, eps)
    if name == 'log':
        steep = random.uniform(0.01, n / 2.0)
        return fn(n, steep)
    if name == 'randdec':
        return fn(n)
    if name == 'sine':
        power = random.uniform(0.2, 5.0)
        return fn(n, power)
    if name == 'quad':
        a = random.uniform(0.2, 5.0)
        return fn(n, a)
    # fallback – uniform
    return np.ones(n, dtype=float)


# ----------------------------------------------------------------------
# Gradient step on the most active convolution offset
# ----------------------------------------------------------------------
def _gradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """One projected gradient descent step on J = log(max_b) - 2·log(sum)."""
    n = seq.shape[0]
    conv = _conv_fft(seq)
    max_b = conv.max()
    # pick a random maximising index (could be many)
    ks = np.flatnonzero(conv == max_b)
    k = int(random.choice(ks))

    sum_a = seq.sum()
    if sum_a == 0:
        return seq  # should not happen

    # target average contribution a_{k-i} that would make the gradient zero
    target = max_b / sum_a

    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    delta = np.zeros_like(seq)
    # delta_i = a_{k-i} - target  (sub‑gradient of J)
    delta[mask] = seq[idx[mask]] - target

    # gradient descent step (move opposite to the sub‑gradient)
    new_seq = seq - eta * delta
    # enforce non‑negativity
    new_seq = np.maximum(new_seq, 0.0)
    # avoid the all‑zero degenerate case
    if new_seq.max() == 0:
        new_seq = seq.copy()
    # normalise the scale (does not affect the score)
    new_seq = _normalize(new_seq)
    return new_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function within the allotted time.
    Returns a list of non‑negative heights (the best sequence discovered).
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.3   # tiny safety margin

    best_seq = None
    best_score = math.inf

    # --------------------------------------------------------------
    # 0) start from the globally supplied reference if it exists
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = _normalize(np.asarray(list(height_sequence_1), dtype=float))
            sc = _fast_score(init)
            best_seq, best_score = init, sc
            print(f"Reference sequence score: {best_score:.6f}")
        except Exception:       # pragma: no cover
            pass

    # --------------------------------------------------------------
    # 1) generate an initial candidate if we have nothing useful yet
    # --------------------------------------------------------------
    if best_seq is None:
        n0 = random.choice([800, 1200, 1600, 2000, 2600, 3200])
        cand = _sample_family(n0)
        best_seq = cand
        best_score = _fast_score(cand)
        print(f"Initial random candidate (n={n0}) → {best_score:.6f}")

    # current incumbent that we will improve locally
    cur_seq = best_seq.copy()
    cur_score = best_score

    # optimisation hyper‑parameters
    eta = 0.20                    # initial step size for gradient descent
    eta_decay = 0.9997            # slow decay
    prob_random = 0.12            # probability to replace by a fresh random draw
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1

        # ----------------------------------------------------------
        # Random re‑initialisation (exploration)
        # ----------------------------------------------------------
        if random.random() < prob_random:
            # occasionally try a fresh candidate from a parametric family
            n = int(random.choice([600, 800, 1000, 1400, 1800, 2200, 2600, 3000]))
            cand = _sample_family(n)
            sc = _fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc
                cur_seq, cur_score = cand.copy(), sc
                print(f"New random best ({n}) → {best_score:.6f}")
            continue

        # ----------------------------------------------------------
        # Gradient descent step on the current incumbent
        # ----------------------------------------------------------
        new_seq = _gradient_step(cur_seq, eta)
        new_score = _fast_score(new_seq)

        # Simple acceptance rule – keep improvements, otherwise
        # accept with a tiny probability to escape plateaus.
        if new_score < cur_score or random.random() < 0.005:
            cur_seq, cur_score = new_seq, new_score

        if new_score < best_score:
            best_seq, best_score = new_seq, new_score
            print(f"Improved (GD) iter {iter_count} → {best_score:.6f}")

        # occasional small stochastic perturbation (helps escape shallow minima)
        if iter_count % 250 == 0:
            pert = cur_seq.copy()
            # pick a few coordinates and jiggle them multiplicatively
            m = random.randint(1, min(4, pert.size))
            idxs = random.sample(range(pert.size), m)
            for i in idxs:
                factor = 1.0 + random.gauss(0.0, 0.06)
                pert[i] = max(0.0, pert[i] * factor)
            pert = _normalize(pert)
            pert_score = _fast_score(pert)
            if pert_score < cur_score:
                cur_seq, cur_score = pert, pert_score
            if pert_score < best_score:
                best_seq, best_score = pert, pert_score
                print(f"Improved (perturb) iter {iter_count} → {best_score:.6f}")

        # step‑size decay
        eta *= eta_decay

    # ------------------------------------------------------------------
    # Final verification with the official evaluate_sequence (the official
    # function also performs safety checks; it should give the same value
    # up to rounding).
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Finished search – best verified score {final_score:.6f} (len={len(final_seq)})")
    return final_seq
