"""
Optimise a non‑negative step function to minimise
evaluate_sequence.

The method combines:
  • a parametric family p_i ∝ (i+1)^{α-1}·(n-i)^{α-1}
    (Beta(α,α) lifted to a discrete simplex) – we scan a few α
    values to get diverse starting points.
  • a projected exponentiated‑gradient descent that uses the exact
    sub‑gradient of the max‑convolution (i.e. the derivative of the
    active convolution entry).  The update keeps the vector in the
    probability simplex, guaranteeing non‑negativity.
  • occasional random jitter and a fallback random restart to avoid
    stagnation.

The loop runs until the time budget is exhausted and returns the best
sequence found (as a plain Python list) – the external harness will
feed it to evaluate_sequence().
"""

import time
import numpy as np

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    # Using the simple renormalisation: clamp negatives then divide.
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def beta_family(n: int, alpha: float) -> np.ndarray:
    """Return a symmetric Beta(α,α)–like distribution on {0,…,n‑1}
    normalised to sum = 1.  The unnormalised weight of index i is
    (i+1)^{α‑1}·(n‑i)^{α‑1}.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 1.0, alpha - 1.0) * np.power(n - i, alpha - 1.0)
    w_sum = w.sum()
    if w_sum == 0:
        # fall back to uniform
        w[:] = 1.0
        w_sum = n
    return w / w_sum


def sqrt_distribution(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^(-1/2) up to length n, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def max_convolution_and_gradient(p: np.ndarray):
    """Return (conv, max_val, gradient) where conv = p∗p,
    max_val = max(conv) and gradient is a sub‑gradient of max(conv).
    The gradient follows d/dp_k max(p∗p) = 2·p_{m−k} for the active index m.
    """
    n = p.size
    conv = np.convolve(p, p)               # length 2n‑1
    m = conv.argmax()                      # active index
    max_val = conv[m]

    grad = np.zeros_like(p)
    lo = max(0, m - (n - 1))
    hi = min(n - 1, m)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * p[m - idx]
    return conv, max_val, grad

    The routine respects the time budget and returns the best sequence
    seen before the deadline.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------

    # normalise all
    candidates = [project_to_simplex(c) for c in candidates]

    best_score = float("inf")

        return 2.0 * n * conv.max()

"""
Optimise a non‑negative step sequence to minimise
`evaluate_sequence`.

The method combines three ideas:

1. **Diverse initialisations** – a small catalogue of
   Beta‑like families (including the classical square‑root
   distribution) that already give good scores.
2. **Projected exponentiated‑gradient descent** – a fast first‑order
   method that uses the *exact* sub‑gradient of the active
   convolution entry (the index where the max is attained).  The
   update `a ← a·exp(‑η·grad)` keeps the vector non‑negative and a
   final projection onto the probability simplex preserves
   normalisation (∑a = 1).  The step size η is slowly decayed.
3. **Escaping stagnation** – occasional tiny random jitter and a
   very rare full random restart keep the optimiser from getting
   stuck.

The loop stops when the allotted time budget (≈ 1000 s) is exhausted
and returns the best sequence seen.  The score is computed directly
as `2·n·max(conv)` because the evaluation function is homogeneous
and the sum of the coefficients is normalised to one.
"""

import time
import numpy as np


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def beta_family(n: int, alpha: float) -> np.ndarray:
    """Discrete symmetric Beta(α,α) style distribution on {0,…,n‑1}.
    The unnormalised weight of index i is (i+1)^{α‑1}·(n‑i)^{α‑1}.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 1.0, alpha - 1.0) * np.power(n - i, alpha - 1.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0
        w_sum = n
    return w / w_sum


def sqrt_distribution(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^{‑1/2} up to length n, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def max_convolution_and_gradient(p: np.ndarray):
    """Return (conv, max_val, grad) where
    conv = p∗p, max_val = max(conv) and grad is a sub‑gradient of max(conv).

    The active index m = argmax(conv) yields
    ∂/∂p_k max(conv) = 2·p_{m‑k} for all k where the term exists.
    """
    n = p.size
    conv = np.convolve(p, p)               # length 2n‑1
    m = conv.argmax()                      # active index
    max_val = conv[m]

    grad = np.zeros_like(p)
    lo = max(0, m - (n - 1))
    hi = min(n - 1, m)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * p[m - idx]
    return conv, max_val, grad
