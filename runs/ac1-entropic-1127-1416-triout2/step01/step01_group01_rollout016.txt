def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    # choose length
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 5000

    # ramp init
    a = np.arange(n,0,-1,dtype=float)
    a = 2*a/(n*(n+1))  # sum=1

    best_a = a.copy()
    best_val = compute_objective(best_a)
    print(f"Initial objective: {best_val:.6f}")

    # parameters
    step_size = 0.1
    iteration = 0
    while time.time() < deadline and iteration < 2000:
        # gradually reduce step_size
        step = step_size / np.sqrt(iteration+1)

        # propose new candidate by adding Gaussian noise
        cand = best_a + step * np.random.randn(n)

        # ensure non-negative and project onto simplex
        cand = simplex_projection(cand)

        # compute objective quickly via FFT
        val = compute_objective(cand)

        if val < best_val:
            best_val = val
            best_a = cand
            print(f"Iter {iteration:4d}: improved to {best_val:.6f}")

        iteration += 1

    # return as list of floats
    return [float(v) for v in best_a.tolist()]

Propose a step‑function (non‑negative vector) that minimizes

    score = 2·n·max(conv(a,a)) / (sum(a)²)

where `conv(a,a)` is the full self‑convolution.
The routine works on the probability simplex (∑a = 1) – scaling does not
change the score – and uses a smooth approximation of the ℓ∞‑norm.
A soft‑max (log‑sum‑exp) surrogate of the maximum is minimised by
projected gradient descent.  The parameter β of the soft‑max is
increased during the run so that the algorithm first moves in a smooth
direction and later focuses on the true peak.  All iterates are projected
onto the simplex by the Duchi‑et‑al. Euclidean projection.  The best
vector found within the time budget is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {w | w_i ≥ 0 , sum w_i = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical drift may break the sum‑to‑one constraint; renormalise.
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:   # fall back to uniform if something went wrong
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
