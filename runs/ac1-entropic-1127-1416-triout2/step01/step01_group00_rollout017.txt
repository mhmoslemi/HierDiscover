Propose a low‑score step function.

The evaluator uses the scale‑invariant quantity
    E(a) = 2·n·max(a∗a) / (∑a)² .
If we keep ∑a = 1 the problem is to minimise  max(a∗a).

The search consists of three stages

1. **Fast parametric sweep** – families that are known to give flat
   self‑convolutions (arcsine, powers of the arcsine, mixtures with a
   uniform vector, and a simple β‑parameterised 1/√‑law) are generated,
   evaluated with a cheap FFT‑based score and the best one is kept.

2. **Linear‑programming “mixing” step** – for a fixed f we solve
   ``max ∑b   s.t.   (f∗b)ₖ ≤ M  ∀k,  b ≥ 0``
   ( M = max(f∗f) ).  The LP is tiny (≤ 8000 × 2000) and therefore fast.
   The optimal b is mixed with the current distribution:
        f ← (1‑t)·f + t·(b / ∑b) .
   Repeating this step monotonically reduces M.

3. **Fine‑grained refinement** – the best vector from step 2 is polished
   with a projected sub‑gradient optimiser (the code that achieved
   1.519 in the previous run).  A few hundred thousand iterations on a
   vector of a few thousand entries are still cheap because each step
   needs only one FFT.

All stages respect the 1000 s time budget and the final answer is a
plain list of non‑negative floats ready for `evaluate_sequence`.

# Imports – all helpers are top‑level, no closures or lambdas.

import time
import numpy as np
import scipy.optimize  # linear programming (highs solver)

# --------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based convolution)
# --------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a∗b using real FFT (fast for our sizes)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> tuple[float, int]:
    """Return (max value, index) of a∗a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v <= 0):
        # Uniform vector is the projection of the zero vector.
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Score – the same quantity that `evaluate_sequence` computes
# (except without the final clipping).
# --------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Scale‑invariant score 2·n·max(a∗a) (sum(a) is assumed = 1)."""
    n = len(a)
    conv = convolution_fft(a, a)
    m = float(np.max(conv))
    return 2.0 * n * m


# --------------------------------------------------------------
# Known good initial shapes
# --------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Continuous arcsine density discretised on n points."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # avoid exact zeros at the ends
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def beta_shape(n: int, beta: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-0.5-i))^{-β}."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / ((i + 0.5) * (n - 0.5 - i) + 1e-12) ** beta
    return vals / np.sum(vals)


def power_transform(base: np.ndarray, p: float) -> np.ndarray:
    """Raise every entry of `base` to power `p` and renormalise."""
    a = base ** p
    return a / np.sum(a)


def mix_shapes(a1: np.ndarray, a2: np.ndarray, alpha: float) -> np.ndarray:
    """α·a1 + (1‑α)·a2, renormalised."""
    a = alpha * a1 + (1.0 - alpha) * a2
    return a / np.sum(a)


# --------------------------------------------------------------
# Sub‑gradient optimiser (the method that gave 1.519)
# --------------------------------------------------------------
def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of (a∗a)_k w.r.t. a."""
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        d = k // 2
        if d < n:
            grad[d] = 2.0 * a[d]
    return grad

    eta0: float = 0.5,
    noise_every: int = 500,
    noise_scale: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for `max(a∗a)`.
    Returns the best vector seen and its scale‑invariant score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1
        _, k = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k)

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new
        # occasional progress output
        if it % 2000 == 0:
            print(f"  sub‑grad iter={it:,d}  cur={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score


# --------------------------------------------------------------
# Linear‑programming mixing step (the “LP‑boost” described
# in the statement).
# --------------------------------------------------------------
def solve_lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_i
    s.t.    (f∗b)_k ≤ M   for all k,
            b_i ≥ 0.
    Returns the optimal b (or None on failure).
    """
    n = len(f)
    rows = 2 * n - 1

    # Build the (rows × n) constraint matrix.
    # Row k contains f_i for those i with 0 ≤ k‑i < n.
    A = np.zeros((rows, n), dtype=float)
    for i, fi in enumerate(f):
        A[i:i + n, i] = fi

    # Linear programme in “maximise Σ b_i” form → minimise −Σ b_i.
    c = -np.ones(n, dtype=float)
    b_ub = np.full(rows, M, dtype=float)

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        print("LP failed:", res.message)
        return None
    return res.x


def lp_mix_optimize(
    init_f: np.ndarray,
    time_deadline: float,
    max_lp_iters: int = 12,
    mix_coeff: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP‑mixing step.
    Returns the best vector seen (always normalised to sum = 1).
    """
    f = init_f.copy()
    best_f = f.copy()
    best_score = compute_score(f)

    it = 0
    while time.time() < time_deadline - 3 and it < max_lp_iters:
        it += 1
        M = np.max(convolution_fft(f, f))
        b = solve_lp_step(f, M)
        if b is None:
            break
        sum_b = np.sum(b)
        if sum_b < 1e-12:
            break
        g = b / sum_b               # normalise to sum = 1
        f = (1.0 - mix_coeff) * f + mix_coeff * g   # convex combine

        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score, best_f = cur_score, f.copy()
            print(f"  LP‑mix iter={it}  score={cur_score:.6f}")

    return best_f, best_score


# --------------------------------------------------------------
# Main entry point – orchestrates the three stages.
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low
    `evaluate_sequence` value.  The method combines a fast
    parametric sweep, a linear‑programming mixing phase and a
    final projected sub‑gradient polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # -----------------------------------------------------------------
    # Global best tracking (fallback to a previously known solution)
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # may be undefined
        val = evaluate_sequence(fallback)
        print(f"Fallback score = {val:.6f}")
        best_seq, best_val = fallback, val
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Candidate lengths – medium sized vectors give the best trade‑off.
    # -----------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline - 30:
            break
        print(f"\n=== Working on length n={n} ===")

        # -------------------------------------------------------------
        # Stage 1 – cheap parametric sweep.
        # -------------------------------------------------------------
        candidates = []

        # (i) arcsine and uniform
        candidates.append(arcsine_initial(n))
        candidates.append(np.full(n, 1.0 / n))

        # (ii) β‑parameter family  (0.35 … 0.65)
        for beta in np.linspace(0.35, 0.65, 13):
            candidates.append(beta_shape(n, beta))

        # (iii) power‑transform of the arcsine
        base = arcsine_initial(n)
        for p in np.linspace(0.8, 1.2, 9):
            candidates.append(power_transform(base, p))

        # (iv) mixture arcsine ↔ uniform
        uniform = np.full(n, 1.0 / n)
        for alpha in np.linspace(0.0, 1.0, 11):
            candidates.append(mix_shapes(base, uniform, alpha))

        # Evaluate all candidates (quick score, no expensive `evaluate_sequence`).
        for idx, a in enumerate(candidates):
            if time.time() > deadline - 10:
                break
            sc = compute_score(a)          # fast, scale‑invariant
            if sc < best_val:
                # verify with the official evaluator (once per improvement)
                val = evaluate_sequence(a.tolist())
                print(f"  New candidate (idx={idx}) → score {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a.tolist()

        # -------------------------------------------------------------
        # Stage 2 – LP‑mixing (only for the moderate sizes 1024/2048)
        # -------------------------------------------------------------
        if n <= 2048 and time.time() < deadline - 30:
            # start from the *best* vector we have of this length
            current = np.array(best_seq if (best_seq and len(best_seq) == n)
                               else arcsine_initial(n), dtype=float)

            lp_deadline = time.time() + min(0.3 * (deadline - time.time()), 120.0)
            f_lp, lp_score = lp_mix_optimize(
                current,
                time_deadline=lp_deadline,
                max_lp_iters=12,
                mix_coeff=0.25,
            )
            # verify improvement
            val = evaluate_sequence(f_lp.tolist())
            if val < best_val:
                best_val, best_seq = val, f_lp.tolist()
                print(f"  After LP‑mix: score {val:.6f}")

        # -------------------------------------------------------------
        # Stage 3 – fine sub‑gradient polishing (uses the fast routine)
        # -------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n and time.time() < deadline - 5:
            remaining = deadline - time.time() - 2.0
            max_iter = int(min(200_000, remaining * 150))   # heuristic
            print(f"  Polishing with sub‑gradient (max_iter≈{max_iter})")
            a0 = np.array(best_seq, dtype=float)
            polished, polished_score = improve_sequence(
                a0,
                time_deadline=deadline,
                max_iter=max_iter,
                eta0=0.4,
                noise_every=600,
                noise_scale=5e-5,
            )
            val = evaluate_sequence(polished.tolist())
            if val < best_val:
                best_val, best_seq = val, polished.tolist()
                print(f"  Polished improvement → score {val:.6f}")

    # -----------------------------------------------------------------
    # Final sanity check – ensure we have *some* answer.
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
