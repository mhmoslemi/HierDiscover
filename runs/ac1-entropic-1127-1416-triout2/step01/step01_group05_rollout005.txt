    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float('inf')
    # candidate lengths
    lengths = [2000, 2500, 3000, 3500, 4000, 4500]  # limit memory
    p_values = [0.48, 0.5, 0.52]
    offsets = [0.0, 0.5, 1.0, 2.0]

                    best_score = score
                    best_seq = seq.tolist()

                if time.time() > deadline:
                    break

"""
Optimise a non‑negative step‑function for the evaluation metric

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

The code works in three stages:

1. **Coarse grid search** over a small family a_i ∝ (i+1+α)^{-p} to obtain a
   decent starting sequence (normalised so that sum(a)=sqrt(2n)).
2. **Iterative refinement** using a linear programme.
   For a fixed sequence f we solve

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   for all k,
                    g_j ≥ 0,

   where M = max(conv(f,f)).  The LP finds a “mass‑rich” auxiliary sequence g.
   After scaling g to have the same sum as f we replace

        f ← (1‑t)·f + t·g     (t≈0.5)

   The convex combination preserves the normalisation and provably never
   increases the maximal self‑convolution.  Repeating this step drives the
   score down towards the theoretical optimum (~1.0 for large n).
3. **Final random local tweaks** for the few remaining seconds.

All heavy work is done with NumPy / SciPy; the final sequence is returned as a
plain list of floats.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalise_to_target(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector so that its sum equals target_sum."""
    s = seq.sum()
    if s == 0.0:
        return seq
    return seq * (target_sum / s)


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) == np.convolve(f, g).
    Entry A_{k,j} = f_{k‑j} when 0 ≤ k‑j < n, otherwise 0.
    """
    n = f.shape[0]
    # rows = i + j  (i runs over 0..n‑1, j over 0..n‑1)
    rows = np.concatenate([np.arange(n) + j for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_j   s.t.   (f * g)_k ≤ M,   g ≥ 0.
    Returns the optimal g (not yet scaled to match the target sum).
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(A.shape[0], M, dtype=float)
    c = -np.ones(n)                     # maximise Σ g  ↔ minimise –Σ g
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f'LP failed: {res.message}')
    return res.x


def _refine_via_lp(f: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based iterative improvement described in the docstring.
    Returns the best sequence found and its score (which equals max(conv)).
    """
    n = f.shape[0]
    target_sum = np.sqrt(2.0 * n)
    best_seq = f.copy()
    best_score = np.max(np.convolve(best_seq, best_seq))

    while time.time() < deadline:
        # Current maximal self‑convolution
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # Linear programme: maximise Σ g subject to f*g ≤ M
        try:
            g0 = _lp_max_sum(f, M)
        except RuntimeError:
            break

        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # Scale g so that it has the same sum as f (the target sum)
        g = g0 * (target_sum / sum_g0)

        # Try a few mixing factors; keep the first that improves the score
        improved = False
        for t in (0.5, 0.25, 0.75, 0.1):
            new_f = (1.0 - t) * f + t * g
            # Renormalise numerically (tiny drift)
            new_f = _normalise_to_target(new_f, target_sum)

            new_score = np.max(np.convolve(new_f, new_f))
            if new_score < M - 1e-12:
                f = new_f
                if new_score < best_score:
                    best_score = new_score
                    best_seq = f.copy()
                    print(f'LP iteration: n={n}  t={t:.2f}  score={new_score:.6f}')
                improved = True
                break

        if not improved:
            # No improvement – fall back to a tiny random mass move
            i, j = np.random.randint(n, size=2)
            if i != j:
                max_move = 0.01 * f[i]
                if max_move > 0:
                    delta = np.random.uniform(-max_move, max_move)
                    if f[i] - delta >= 0:
                        f[i] -= delta
                        f[j] += delta
                        f = _normalise_to_target(f, target_sum)

    return best_seq, best_score
