"""
Propose a low‑score non‑negative step function.

The algorithm works in three stages:

1. **Fast seeding** –  A short grid search over geometric sequences
   (a_i = C·r^i) together with a simple power‑law seed gives a decent
   starting point for a given length `n`.

2. **Smooth max‑convolution minimisation** –  The objective
   2·n·max_k (a*a)[k] / (Σa)^2 is approximated by the smooth
   “soft‑max” functional  

        f_γ(a) = (1/γ)·log Σ_k exp(γ·conv_k) ,

   where `conv = a*a` (full linear convolution).  Its gradient is a
   weighted sum of the individual gradients of the convolution terms.
   The weighted sum is obtained efficiently by a single FFT‑based
   cross‑correlation.  A projected gradient descent with Nesterov‑style
   momentum is performed, repeatedly projecting back onto the simplex
   `{x≥0, Σx = √(2·n)}`.

3. **Adaptive schedule & restarts** –  The soft‑max temperature γ is
   increased slowly so that the optimisation initially sees a smooth
   landscape and later focuses on the true maximum.  The step size is
   decayed, occasional random jitter is added, and a restart is triggered
   if no improvement is seen for many iterations.

The routine respects the 1000 s time budget, logs progress and finally
returns the best sequence found.  All numbers are non‑negative and the
sequence is projected onto the required simplex, guaranteeing that the
call to `evaluate_sequence` never returns `inf`.
"""

import numpy as np
import time
import math
from typing import List

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ∈ R^n | x_i ≥ 0, Σ x_i = s}.
    Uses the O(n log n) algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    # Choose a problem size.  Larger n gives more freedom but costs a bit
    # more per iteration.  We adapt n to the available budget.
    if budget_s > 800:
        n = 8000
    elif budget_s > 400:
        n = 6000
    else:
        n = 4000

    target_sum = math.sqrt(2.0 * n)          # Σ a_i must equal this value

    # ------------------------------------------------------------------- #
    # 1. Fast seeding – geometric families and a simple power‑law.
    best_seq = None
    best_val = float("inf")

    # helper to evaluate quickly (the supplied evaluate_sequence already does
    # all safety checks).
    def evaluate_fast(arr):
        return evaluate_sequence(arr.tolist())

    # ---- geometric scan --------------------------------------------------
    # λ = 1‑r ranges from 1e‑5 to ~0.3 (r between 0.99999 and 0.7)
    lambdas = np.logspace(-5, -0.3, 180)
    arange_n = np.arange(n, dtype=float)
    for lam in lambdas:
        r = 1.0 - lam
        # geometric sum: Σ r^i = (1‑r^n)/lam
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom
        a = C * (r ** arange_n)
        val = evaluate_fast(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    # ---- power‑law seed --------------------------------------------------
    idx = np.arange(1, n + 1, dtype=float)
    a_pow = 1.0 / np.sqrt(idx)
    C_pow = target_sum / a_pow.sum()
    a_pow = C_pow * a_pow
    val_pow = evaluate_fast(a_pow)
    if val_pow < best_val:
        best_val = val_pow
        best_seq = a_pow.copy()

    # ---- uniform fallback ------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # ---- incorporate any previously discovered construction -------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.size <= n:
            seq = np.concatenate([old, np.zeros(n - old.size)])
        else:
            seq = old[:n].copy()
        # renormalise to the required sum
        seq = projection_simplex(seq, target_sum)
        val_old = evaluate_fast(seq)
        if val_old < best_val:
            best_val = val_old
            best_seq = seq.copy()

    # ------------------------------------------------------------------- #
    # 2. Projected gradient descent on the smooth soft‑max of the convolution.
    curr = best_seq.copy()
    best = curr.copy()
    best_score = best_val

    # FFT length – need to accommodate linear convolution of length 2n‑1
    # and correlation of length 3n‑2.  Use the next power of two that covers
    # the larger of the two.
    fft_len = 1 << ((3 * n - 2).bit_length())   # >= 3n‑2

    # optimisation hyper‑parameters
    mu = 0.9                     # momentum factor
    momentum = np.zeros_like(curr)
    step_factor = 0.5            # controls absolute L2 step via step_factor*target_sum/n
    decay_factor = 0.85
    decay_every = max(2000, n)   # decay step size every few thousand iter.
    min_step = 1e-8
    gamma = 4.0                  # initial soft‑max temperature
    gamma_inc = 0.015            # increase per iteration
    gamma_max = 80.0

    it = 0
    last_improve = 0
    max_no_improve = max(5000, n // 2)

    while time.time() < deadline:
        # ---- FFT of current sequence ------------------------------------
        F_a = np.fft.rfft(curr, n=fft_len)

        # full linear convolution a*a (length 2n‑1)
        conv_full = np.fft.irfft(F_a * F_a, n=fft_len)[: 2 * n - 1]

        # current score (identical to evaluate_sequence's formula)
        sum_a = curr.sum()
        max_conv = conv_full.max()
        score = 2.0 * n * max_conv / (sum_a * sum_a)

        # record improvement
        if score < best_score:
            best_score = score
            best = curr.copy()
            last_improve = it
            # tighten step size when we improve
            step_factor = max(step_factor * 0.92, min_step)
            print(f"[{time.time() - start_time:6.2f}s] iter {it:7d} → new best {best_score:.6f}  γ={gamma:.2f}")

        # ---- soft‑max weights -------------------------------------------
        maxc = max_conv
        # stable soft‑max (the exponentials are ≤ 1)
        exp_vals = np.exp(gamma * (conv_full - maxc))
        weights = exp_vals / exp_vals.sum()

        # ---- gradient of the smooth objective ---------------------------
        # gradient = 2 * (weights ⋆ a)   (cross‑correlation)
        F_w = np.fft.rfft(weights, n=fft_len)
        corr = np.fft.irfft(np.conj(F_w) * F_a, n=fft_len)
        # linear correlation indices: lag -j is at position (n‑1‑j)
        idx = n - 1 - np.arange(n)
        grad = 2.0 * corr[idx]

        # ---- momentum step + projection ---------------------------------
        step_norm = step_factor * target_sum / n      # absolute L2 step size
        if np.linalg.norm(grad) > 0:
            direction = - grad / (np.linalg.norm(grad) + 1e-12)
        else:
            direction = np.zeros_like(grad)
        momentum = mu * momentum + step_norm * direction
        tentative = curr + momentum
        # clip to the evaluator's numeric limits before projection
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # ---- occasional random jitter ------------------------------------
        if np.random.rand() < 0.012:
            jitter = np.random.randn(n) * step_factor * target_sum / (40.0 * n)
            curr = projection_simplex(np.clip(curr + jitter, 0.0, 1000.0), target_sum)
            momentum = np.zeros_like(curr)   # reset momentum after a jitter

        # ---- schedule adjustments ----------------------------------------
        it += 1
        # increase soft‑max temperature slowly
        gamma = min(gamma + gamma_inc, gamma_max)

        # decay step size periodically
        if it % decay_every == 0:
            step_factor = max(step_factor * decay_factor, min_step)

        # restart if stagnated for a long time
        if it - last_improve > max_no_improve:
            # perturb the current best solution a little and continue
            perturb = np.random.randn(n) * step_factor * target_sum / (15.0 * n)
            curr = projection_simplex(np.clip(best + perturb, 0.0, 1000.0), target_sum)
            momentum = np.zeros_like(curr)
            last_improve = it
            gamma = 4.0
            print(f"[{time.time() - start_time:6.2f}s] iter {it:7d} – restart (stagnation)")

    # ------------------------------------------------------------------- #
    # Final evaluation (the evaluator does its own clipping)
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished after {it} iterations. Best score: {final_score:.6f}")
    return best.tolist()
