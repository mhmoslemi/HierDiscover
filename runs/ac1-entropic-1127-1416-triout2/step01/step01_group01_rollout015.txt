"""
Propose a candidate step‐function sequence by minimizing
    2·n·max(a*a) / (sum(a)^2)

The objective is convex in the simplex (a_i ≥ 0, Σa_i = 1) and its
sub‑gradient is easy to obtain: for the index k where the convolution
reaches its maximum the gradient equals 2·a_{k‑i} on the interval of
indices that contribute to that entry.

We start from the best sequence known to the environment (``height_sequence_1``)
or from a uniform distribution, normalise it to Σa = 1 and then run a
large number of projected (sub‑)gradient steps.  A classical
diminishing step‑size (0.5/√t) together with Euclidean projection onto the
probability simplex (Duchi‑et‑al. 2008) keeps the iterate feasible.
Occasionally a tiny random jitter is added to help escape shallow local
minima.  The best vector seen during the run is returned.

The routine respects the 1000 s time limit and prints progress
information, but otherwise contains no side effects.
"""

import time
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # condition for finding the threshold
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Compute the evaluation function for a vector with Σa = 1.
    Equivalent to evaluate_sequence after the same normalisation:
        2·n·max(conv(a, a))
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_b = conv.max()
    return 2.0 * n * max_b


def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step function.
    The function runs for up to ``budget_s`` seconds (default 1000) and
    returns the best sequence found.
    """
    np.random.seed(0)                     # reproducibility
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # --------------------------------------------------------------
    # 1. Determine problem size and initial point.
    # --------------------------------------------------------------
    try:
        # the environment may provide a previously found sequence
        init = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
    except Exception:
        # fallback: a moderate size (≈5 k) works well for the time budget
        n = 5146
        init = np.full(n, 1.0 / n)        # uniform start

    # Normalise to the probability simplex (Σa = 1)
    a = init.copy()
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total

    # --------------------------------------------------------------
    # 2. Main optimisation loop (projected sub‑gradient descent).
    # --------------------------------------------------------------
    best_a = a.copy()
    best_score = compute_objective(best_a)

    # momentum vector (simple Nesterov‑style)
    velocity = np.zeros_like(a)
    beta = 0.9                # momentum factor
    step0 = 0.5               # initial step‑size multiplier

    iteration = 0
    while time.time() < deadline:
        # ---- convolution and location of the current maximum ----
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # ---- compute a (sub‑)gradient of the max‑entry ----------
        # use all indices that are numerically tied for the max
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            # contribution 2·a[partner] for each participating component
            grad[idx] += 2.0 * a[partner]

        # average over the active set (any convex combination is a valid sub‑gradient)
        if active.size > 1:
            grad *= 1.0 / active.size

        # ---- step‑size (diminishing) and momentum update -------
        step = step0 / np.sqrt(iteration + 1)
        velocity = beta * velocity - step * grad
        a_new = simplex_projection(a + velocity)

        # ---- occasional stochastic perturbation to escape flat regions ----
        if iteration % 5000 == 0 and iteration > 0:
            eps = 0.01 / np.sqrt(iteration + 1)
            a_new = simplex_projection(a_new + eps * np.random.randn(n))

        a = a_new

        # ---- evaluate and keep the best solution seen so far ----
        cur_score = compute_objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:6d}: improved score = {best_score:.6f}")

        # ---- termination check (optional early stop) ----
        # if the improvement is already below the target we can break early
        if best_score < 1.503:
            print(f"Target reached at iteration {iteration}")
            break

        iteration += 1

    # --------------------------------------------------------------
    # 3. Return the best sequence as a plain Python list.
    # --------------------------------------------------------------
    return [float(v) for v in best_a.tolist()]
