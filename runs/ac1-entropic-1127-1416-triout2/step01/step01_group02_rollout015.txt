def compute_max_and_score(seq, nfft):
    # compute conv via FFT
    fft = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(fft * fft, n=nfft)[:2*len(seq)-1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf, np.inf, None, 0  # (score,max_b,conv,argmax)
    score = 2.0 * len(seq) * max_b / (sum_a * sum_a)
    return score, max_b, conv, int(conv.argmax())

"""
propose_candidate
-----------------
A stochastic hill‑climber that starts from the optimal “binomial”
sequence (c_k = C(2k,k)/4**k) and repeatedly moves a small amount of
mass from early (high‑impact) positions to late (low‑impact) positions.
The move is biased:
   * source indices are chosen from the first ~10 % of the array,
     with probability proportional to the current height (large entries
     have a larger effect on the convolution maximum);
   * target indices are taken from the last ~10 % of the array,
     preferring slots that are still close to zero (they contribute little
     to the convolution).    
The score *2·n·max(a∗a)/(sum(a)²)* is evaluated by a fast FFT‑based
convolution (O(n log n)).  A simulated‑annealing acceptance rule
allows occasional uphill steps, helping the search escape plateaus.
The temperature and the transfer step size are slowly cooled.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: central binomial (binomial) coefficients
def binomial_sequence(length: int) -> np.ndarray:
    """c_k = (2k choose k) / 4**k for k = 0 .. length-1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
# Fast evaluation of the target function (as in evaluate_sequence)
def fast_score(seq: np.ndarray, nfft: int) -> float:
    """Return 2 * n * max(conv(seq,seq)) / (sum(seq)**2)."""
    total = seq.sum()
    if total < 0.01:
        return np.inf
    # real FFT convolution
    ft = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(ft * ft, n=nfft)[:2 * len(seq) - 1]
    m = conv.max()
    return float(2.0 * len(seq) * m / (total * total))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Stochastic optimisation of the step heights.
    Starts from the binomial construction, then repeatedly transfers a
    tiny amount of mass from a “heavy” early position to a “light” late
    position while keeping the sequence non‑negative.  Accepts moves that
    improve the score, or (with a probability depending on a temperature)
    moves that are slightly worse – a classic simulated‑annealing scheme.
    The best sequence seen before the time budget expires is returned.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Length of the problem (prefer the length of a known solution)
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 5000

    # ------------------------------------------------------------------
    # Pad a small tail to give the algorithm room to move mass far away.
    # The tail is 10 % of the length (rounded up), filled with zeros.
    tail_len = max(1, n // 10)
    total_n = n + tail_len

    # initial sequence: binomial on the first n entries, zeros on the tail
    seq = np.zeros(total_n, dtype=np.float64)
    seq[:n] = binomial_sequence(n)

    # pre‑compute an FFT size that is a power of two and >= 2*total_n-1
    nfft = 1 << ((2 * total_n - 1).bit_length())

    best_seq = seq.copy()
    best_score = fast_score(best_seq, nfft)

    cur_seq = seq.copy()
    cur_score = best_score

    print(f"[init] total length={total_n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the annealing / hill‑climb
    start_temp = 0.02                     # initial temperature
    temperature = start_temp
    temp_decay = 0.9995                    # geometric cooling
    step = 0.08                            # initial transfer fraction of source entry
    step_decay = 0.9998

    # early part (source) and late part (target) indices
    early_len = max(1, total_n // 12)      # ~8 % of the array
    late_start = total_n - early_len

    early_idx = np.arange(early_len)
    late_idx = np.arange(late_start, total_n)

    # ------------------------------------------------------------------
    # Pre‑compute static probability vectors; they are refreshed when the
    # sequence changes substantially.
    def update_probs():
        src_weights = cur_seq[early_idx]
        if src_weights.sum() == 0:
            src_p = np.full_like(src_weights, 1.0 / src_weights.size)
        else:
            src_p = src_weights / src_weights.sum()
        # for the target we favour small values (larger 1/(value+eps) weight)
        tgt_weights = 1.0 / (cur_seq[late_idx] + 1e-12)
        tgt_p = tgt_weights / tgt_weights.sum()
        return src_p, tgt_p

    src_prob, tgt_prob = update_probs()
    prob_update_counter = 0

    deadline = time.time() + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Choose a source (early, large) and a target (late, small)
        i = np.random.choice(early_idx, p=src_prob)

        cand[j] += amount

            temperature = start_temp

    print(f"[final] best score={best_score:.6f}")

    return best_seq.tolist()
