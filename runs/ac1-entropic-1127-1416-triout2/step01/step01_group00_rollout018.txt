"""
Smoothed‑max gradient descent on the self‑convolution bound.
The objective  E(a)=2·n·max(a*a)/(sum a)^2 is scale‑invariant; we
keep a on the probability simplex (sum=1, a_i≥0) and minimise
the maximal convolution entry.  A smooth approximation
    f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
is used; its gradient can be computed efficiently by a single
FFT‑based convolution.  The algorithm runs several restarts
(arcsine‑shaped, uniform, random Dirichlet) for a few sequence
lengths, keeps the best candidate and finishes with a cheap
pairwise hill‑climbing refinement.
"""
import time
import numpy as np

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` for a sum‑normalised vector."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)

def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1/√((i+0.5)(n-0.5-i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)

def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed max function
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    """
    n = len(a)
    # self‑convolution
    conv = convolution_fft(a, a)               # length 2n‑1
    # soft‑max weights
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)            # length 2n‑1, Σ w = 1

    # reversed a
    a_rev = a[::-1]
    # convolution of reversed a with the weight vector
    conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]      # length n

    # diagonal part w_{2i}
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv

def improve_sequence_smoothed(initial_a: np.ndarray,
                              time_deadline: float,
                              max_iter: int = 200_000,
                              eta0: float = 0.5,
                              alpha_start: float = 1.0,
                              alpha_end: float = 200.0,
                              noise_every: int = 500,
                              noise_scale: float = 1e-4):
    """
    Projected gradient descent on the smoothed objective.
    Returns the best vector seen and its (unsmoothed) score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # homotopy on the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        eta = eta0 / np.sqrt(t)
        a_new = a - eta * grad

        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"    iter {t:,d}  score {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score

def hill_climb(seq, time_deadline: float, max_steps: int = 20000, init_step: float = 0.1):
    """
    Very cheap stochastic pair‑wise exchange local search.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step
    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0:
            continue
        candidate = best_a.copy()
        candidate[i] -= delta
        candidate[j] += delta
        if candidate[i] < 0:
            continue
        candidate = project_to_simplex(candidate)
        cur_score = compute_score(candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_a = candidate
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.98
        max_steps -= 1
    return best_a.tolist(), best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point.  Runs several multi‑start smoothed‑gradient searches,
    keeps the best sequence and applies a final hill‑climbing pass.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_val = float("inf")

    # fallback to a previously discovered construction if present
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    lengths = [1024, 2048, 4096]   # larger lengths give better asymptotic bound

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\nOptimising length n = {n}")

        # three initialisations per length
        inits = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            np.random.dirichlet(np.ones(n))
        ]
        for idx, init_a in enumerate(inits, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  start {idx}/{len(inits)}")
            seq, val = improve_sequence_smoothed(
                initial_a=init_a,
                time_deadline=deadline,
                max_iter=250_000,
                eta0=0.8,
                alpha_start=2.0,
                alpha_end=200.0,
                noise_every=800,
                noise_scale=5e-5,
            )
            print(f"    local best score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** New global best {best_val:.6f} (n={n}) ***")

    # final cheap refinement if we still have time
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nRefining best sequence with local hill‑climb")
        refined_seq, refined_val = hill_climb(best_seq, time_deadline=deadline, max_steps=50000, init_step=0.05)
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq

"""
Smoothed‑max gradient descent on the self‑convolution bound.

The quantity to minimise is  

    E(a) = 2·n·max(a∗a) / (sum(a)²)  

which is scale‑invariant, so we keep a on the probability simplex
(∑a_i = 1, a_i ≥ 0).  Direct sub‑gradient descent on the max is
prone to stagnation; we therefore minimise a smooth approximation  

    f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)  

which converges to the max as α → ∞.  The gradient of f_α can be
written in closed form and evaluated by two FFT‑based convolutions,
so each iteration is O(n log n).  The algorithm runs several
re‑starts (arcsine‑shape, uniform, random Dirichlet) for a few
different lengths, keeps the best candidate, and finally performs a
cheap stochastic hill‑climbing refinement.

The function `propose_candidate` is the entry point required by the
competition harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` when ∑a = 1."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient computation
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximum
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    Returns the gradient (same shape as a) and the raw convolution a∗a.
    """
    n = len(a)
    # self‑convolution
    conv = convolution_fft(a, a)                # length 2n‑1
    # soft‑max weights
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)             # ∑_k w_k = 1, length 2n‑1

    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]       # entries needed for each i

    # diagonal contribution w_{2i}
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def improve_sequence_smoothed(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.5,
    alpha_start: float = 1.0,
    alpha_end: float = 200.0,
    noise_every: int = 500,
    noise_scale: float = 1e-4,
):
    """
    Projected gradient descent on the smoothed objective.
    Returns the best vector encountered (as a NumPy array) and its
    unsmoothed score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # homotopy on the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        eta = eta0 / np.sqrt(t)
        a_new = a - eta * grad

        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"    iter {t:,d}  score {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Final cheap local refinement
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 20000, init_step: float = 0.1):
    """
    Stochastic pair‑wise exchange hill‑climbing.
    Small amounts of mass are moved from one index to another;
    moves that improve the true (max‑convolution) score are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.98
        max_steps -= 1
    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that yields a low
    `evaluate_sequence` value.  Several multi‑start smoothed‑gradient
    optimisations are carried out for a handful of problem sizes;
    the best result is polished by a cheap hill‑climbing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Fallback to a previously discovered construction, if it exists.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore.  Larger n gives a smaller asymptotic constant,
    # but also costs more per iteration.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\nOptimising length n = {n}")

        # Three different initialisations per length.
        inits = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            np.random.dirichlet(np.ones(n)),
        ]

        for idx, init_a in enumerate(inits, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  start {idx}/{len(inits)}")
            seq, val = improve_sequence_smoothed(
                initial_a=init_a,
                time_deadline=deadline,
                max_iter=250_000,
                eta0=0.8,
                alpha_start=2.0,
                alpha_end=200.0,
                noise_every=800,
                noise_scale=5e-5,
            )
            print(f"    local best score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** New global best {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final cheap stochastic refinement (if time permits).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nRefining best sequence with hill‑climb")
        refined_seq, refined_val = hill_climb(
            best_seq, time_deadline=deadline, max_steps=50000, init_step=0.05
        )
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that we never produced a candidate.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
