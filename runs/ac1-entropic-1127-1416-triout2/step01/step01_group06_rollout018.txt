def _sym_exp_seq(n, alpha):
    dist = np.minimum(np.arange(n), n-1 - np.arange(n))
    seq = np.exp(-alpha * dist)
    return _normalize(seq)

def _sym_lin_seq(n, beta):
    dist = np.minimum(np.arange(n), n-1 - np.arange(n))
    seq = np.maximum(0., 1. - beta * dist)
    return _normalize(seq)

def _edge_constant_seq(n, edge_len, const):
    seq = np.full(n, const, dtype=float)
    seq[:edge_len] = 1.0
    seq[-edge_len:] = 1.0
    return _normalize(seq)

def _greedy_seq(n):
    a = np.zeros(n)
    capacity = np.ones(2*n - 1)   # C_k residual capacities; initially 1 for all k except k=0 where a0=1 saturates
    # We'll store C_k as current sum of contributions.
    a[0] = 1.0
    capacity[0] = 1.0  # product a0*a0; but we need to track contributions sum (C). After setting a0, we update C0 = a0^2 = 1. So capacity_for_new = 1 - C_k.

    # Actually we keep track of current sum C_k. So allocate array C = np.zeros(2*n-1); after each update, set contributions.
    C = np.zeros(2*n-1)
    C[0] = 1.0

    for i in range(1, n):
        # compute max possible a_i
        bound = 1.0
        # self term: constraint k=2*i
        remaining_self = 1.0 - C[2*i]   # can be slightly negative due to rounding; if <=0, bound_self=0
        if remaining_self <= 0:
            bound_self = 0.0
        else:
            bound_self = math.sqrt(remaining_self)
            if bound_self < bound:
                bound = bound_self
        # pair constraints
        for j in range(i):
            if a[j] == 0.0:
                continue
            k = i + j
            remaining_pair = 1.0 - C[k]
            # pair contribution per each unordered pair is 2*a_i*a_j
            # So a_i <= remaining_pair/(2*a_j)
            if remaining_pair <= 0:
                bound_pair = 0.0
            else:
                bound_pair = remaining_pair / (2.0 * a[j])
            if bound_pair < bound:
                bound = bound_pair
                # early break if bound becomes zero?
                if bound <= 0:
                    break
        # ensure bound does not exceed 1
        if bound > 1.0:
            bound = 1.0
        if bound < 0:
            bound = 0.0
        a[i] = bound
        # update C contributions
        # pair with j < i
        for j in range(i):

def _greedy_seq(n):
    a = np.zeros(n, dtype=float)
    C = np.zeros(2 * n - 1, dtype=float)
    a[0] = 1.0
    C[0] = 1.0
    for i in range(1, n):
        bound = 1.0
        # self bound
        cap_self = max(0.0, 1.0 - C[2 * i])
        bound_self = math.sqrt(cap_self) if cap_self > 0 else 0.0
        if bound_self < bound:
            bound = bound_self
        # pair bounds
        for j in range(i):
            if a[j] == 0.0:
                continue
            k = i + j
            cap_pair = max(0.0, 1.0 - C[k])
            bound_pair = cap_pair / (2.0 * a[j]) if cap_pair > 0 else 0.0
            if bound_pair < bound:
                bound = bound_pair
                if bound <= 0:
                    break
        a[i] = bound
        # update C
        for j in range(i):
            if a[j] == 0.0:
                continue
            k = i + j
            C[k] += 2.0 * a[i] * a[j]
        C[2 * i] += a[i] * a[i]
    return a.tolist()

   a = np.array(seq, copy=True)

def _targeted(seq):
    a = np.array(seq, copy=True)
    n = len(a)
    b = np.convolve(a, a)
    max_idx = int(b.argmax())
    # compute contributions
    i_start = max(0, max_idx - (n - 1))
    i_end = min(max_idx, n - 1)
    i_range = np.arange(i_start, i_end + 1)
    prod = a[i_range] * a[max_idx - i_range]
    # get indices of top contributions
    top_inds = i_range[np.argsort(-prod)[:2]]

    for i in top_inds:

"""
Improved search for a low‑score step function.

The algorithm builds on two ideas:

1. **Greedy feasible construction** –  a deterministic pass that
   respects the exact convolution constraints b_k = Σ a_i a_{k‑i} ≤ 1.
   Starting from a_0 = 1 it repeatedly chooses the largest admissible
   value for a_i (the value that keeps all already saturated offsets
   ≤ 1).  This already beats the naïve families and reaches a score
   around 1.57 for n≈2600.

2. **Local re‑balancing** –  the greedy solution saturates the
   low‑index offsets, leaving slack only in the large‑k region.
   By moving a little mass from early (high‑impact) entries to later
   (low‑impact) entries we can increase the total sum while keeping
   the maximal convolution close to 1, which lowers the evaluation
   function.  A simple simulated‑annealing hill‑climber performs many
   random “transfer” or “early‑reduction’’ moves and keeps the best
   candidate found before the time budget expires.

The whole search runs within the given 1000 s budget, tries a few
different lengths, and finally returns the best sequence discovered.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities

def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:               # pragma: no cover
        return math.inf

def _normalize(seq):
    """Scale so that max(seq) == 1 (no‑op if all zeros)."""
    arr = np.asarray(seq, dtype=float)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()

# ----------------------------------------------------------------------
# Families used for the initial random sampling
# ----------------------------------------------------------------------
def _exp_seq(n, alpha):
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)))

def _pow_seq(n, p):
    idx = np.arange(1, n + 1, dtype=float)
    return _normalize(idx ** (-p))

def _lin_seq(n, beta):
    return _normalize(np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float)))

def _mix_seq(n, alpha, eps):
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)) + eps)

def _logistic_seq(n, steep):
    i = np.arange(n, dtype=float)
    mid = n / 2.0
    seq = 1.0 / (1.0 + np.exp((i - mid) / steep))
    return _normalize(seq)

def _randdec_seq(n):
    arr = np.random.rand(n)
    arr.sort()
    arr = arr[::-1]
    return _normalize(arr)

def _sym_exp_seq(n, alpha):
    """Symmetric exponential decay from both ends."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    seq = np.exp(-alpha * d.astype(float))
    return _normalize(seq)

def _sym_lin_seq(n, beta):
    """Symmetric linear decay from both ends."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    seq = np.maximum(0.0, 1.0 - beta * d.astype(float))
    return _normalize(seq)

def _dist_pow_seq(n, p):
    """Power‑law based on distance from the nearest edge."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    seq = (d + 1) ** (-p)
    return _normalize(seq)

# ----------------------------------------------------------------------
# Deterministic greedy construction (exactly respects b_k ≤ 1)
# ----------------------------------------------------------------------
def _greedy_seq(n):
    """
    Build a feasible sequence a_0 … a_{n‑1} with a_0 = 1 and
    Σ a_i a_{k-i} ≤ 1 for every convolution offset k.
    The construction is greedy: each a_i takes the largest value
    allowed by the constraints involving already fixed entries.
    """
    a = np.zeros(n, dtype=float)
    # current convolution sums C_k = Σ a_i a_{k-i}
    C = np.zeros(2 * n - 1, dtype=float)

    a[0] = 1.0
    C[0] = 1.0                                 # b_0 = a_0^2

    for i in range(1, n):
        # maximal possible value for a_i (≤ 1)
        bound = 1.0

        # self‑pair (k = 2*i)
        remain_self = max(0.0, 1.0 - C[2 * i])
        bound_self = math.sqrt(remain_self) if remain_self > 0 else 0.0
        bound = min(bound, bound_self)

        # pair‑wise constraints with earlier entries
        for j in range(i):
            if a[j] == 0.0:
                continue
            k = i + j
            remain = max(0.0, 1.0 - C[k])
            if remain <= 0.0:
                bound = 0.0
                break
            bound_pair = remain / (2.0 * a[j])
            bound = min(bound, bound_pair)
            if bound <= 0.0:
                break

        a[i] = bound

        # update the convolution sums with the newly fixed a_i
        for j in range(i):
            if a[j] == 0.0:
                continue
            k = i + j
            C[k] += 2.0 * a[i] * a[j]
        C[2 * i] += a[i] * a[i]

    # a already has max = 1, no further normalisation needed
    return a.tolist()

def _perturb(seq):
    """Small random multiplicative perturbations of a few entries."""
    a = np.array(seq, copy=True)
    n = len(a)
    m = random.randint(1, min(5, n))
    idxs = random.sample(range(n), m)
    for i in idxs:
        factor = 1.0 + random.gauss(0.0, 0.07)
        a[i] = max(0.0, a[i] * factor)
    mx = a.max()
    if mx > 0:
        a = a / mx
    return a.tolist()


def _transfer(seq):
    """
    Move a little mass from an early (high‑impact) index to a late
    (low‑impact) index.  This typically raises the total sum while
    keeping the maximal convolution close to 1.
    """
    a = np.array(seq, copy=True)
    n = len(a)
    if n < 4:
        return seq
    early = random.randint(0, max(0, n // 3 - 1))
    late = random.randint(max(early + n // 3, 2 * n // 3), n - 1)
    if a[early] <= 1e-12:
        return seq
    max_transfer = min(a[early] * 0.5, 1.0 - a[late])
    if max_transfer <= 0:
        return seq
    delta = random.uniform(0.0, max_transfer)
    a[early] -= delta
    a[late] += delta
    mx = a.max()
    if mx > 0:
        a = a / mx
    return a.tolist()


def _reduce_early(seq):
    """
    Slightly decrease a value at a low index (where it contributes to
    many convolution terms) and renormalise.  The renormalisation
    boosts the later entries, potentially improving the sum‑to‑max‑b
    ratio.
    """
    a = np.array(seq, copy=True)
    n = len(a)
    if n < 2:
        return seq
    i = random.randint(0, max(0, n // 5 - 1))
    factor = 1.0 - random.uniform(0.02, 0.15)   # shrink a_i
    a[i] *= factor
    mx = a.max()
    if mx > 0:
        a = a / mx
    return a.tolist()


def _random_move(seq):
    """Choose one of the specialised moves at random."""
    choice = random.random()
    if choice < 0.4:
        return _transfer(seq)
    elif choice < 0.7:
        return _reduce_early(seq)
    else:
        return _perturb(seq)

            cand = _sym_exp_seq(n, alpha)

        elif family == 'sym_lin':
            beta = random.uniform(0.0, 2.0 / n)
            cand = _sym_lin_seq(n, beta)

        else:   # 'dist_pow'
            p = random.uniform(0.0, 4.0)
            cand = _dist_pow_seq(n, p)

        sc = _score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"Random family ({family}) n={n} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) local stochastic refinement (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is None:      # should never happen
        best_seq = [1.0] * 500
        best_score = _score(best_seq)

    current_seq = best_seq[:]
    current_score = best_score

    # simple temperature schedule
    start_time = time.time()
    init_temp = 0.5
    min_temp = 0.001

    while time.time() < deadline:
        # annealing temperature decays linearly
        elapsed = time.time() - start_time
        T = max(min_temp, init_temp * (1.0 - elapsed / budget_s))

        cand = _random_move(current_seq)
        cand_score = _score(cand)

        # accept if better, otherwise accept with Metropolis probability
        if cand_score < current_score or random.random() < math.exp((current_score - cand_score) / T):
            current_seq, current_score = cand, cand_score

        if cand_score < best_score:
            best_seq, best_score = cand, cand_score
            print(f"Improved → {best_score:.6f}")

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
