def _make_power_sequence(n: int, p: float, offset: float) -> np.ndarray:
    """Un‑normalised a_i = (i+1+offset)^{-p}."""
    idx = np.arange(n, dtype=float)
    return 1.0 / np.power(idx + 1.0 + offset, p)


def _normalise_to_target_sum(seq: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative sequence so that its sum equals *target*."""
    cur = seq.sum()
    if cur == 0.0:
        return seq
    return seq * (target / cur)


def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray, n_fft: int = None) -> np.ndarray:
    """
    Compute the discrete self‑convolution a*a using an FFT.
    Returns the length‑2n‑1 real array.
    """
    n = a.shape[0]
    if n_fft is None:
        n_fft = _next_pow2(2 * n)
    fa = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(fa * fa, n=n_fft)
    # Only the first 2n‑1 entries are needed
    return conv[: 2 * n - 1]


def _smooth_max_and_weights(conv: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) and the associated soft‑max weights.
    Returns (soft_max, weight_array) where weight_array sums to 1.
    """
    # Stabilise the exponent by subtracting the maximum
    m = conv.max()
    shifted = beta * (conv - m)           # max becomes 0 → exp ≤ 1
    expb = np.exp(shifted)
    s = expb.sum()
    if s == 0.0:
        # numerically unlikely, but guard against division by zero
        w = np.full_like(expb, 1.0 / expb.size)
    else:
        w = expb / s
    # soft max = m + log(s)/beta
    soft = m + np.log(s) / beta
    return soft, w

        # ---- multiplicative (exponentiated‑gradient) update --------------------
        # a_i ← a_i * exp(−η·grad_i)
        factor = np.exp(-eta * grad)
        a_candidate = a * factor
        # re‑normalise to keep the exact sum constraint
        a_candidate = _normalise_to_target_sum(a_candidate, target_sum)

        # ---- evaluate the candidate --------------------------------------------
        cand_score = evaluate_sequence(a_candidate.tolist())
        if cand_score < best_score:
            # accept improvement, increase step size a little
            a = a_candidate
            best_score = cand_score
            best_seq = a.tolist()
            eta = min(max_eta, eta * 1.07)
            # accelerate smoothing slowly – we want a sharper approximation later
            beta = min(beta_max, beta * 1.03)
            print(f"[{iteration:04d}] Accept – score {cand_score:.6f} (η={eta:.4f}, β={beta:.2f})")
            continue
        else:
            # reject – shrink step size
            eta = max(min_eta, eta * 0.65)

        # ---- occasional random perturbation to escape plateaux ------------------
        if np.random.rand() < 0.03:
            i = np.random.randint(n)
            # add a tiny zero‑mean Gaussian bump, then renormalise
            bump = np.random.normal(scale=0.001 * target_sum / n)
            a[i] = max(0.0, a[i] + bump)
            a = _normalise_to_target_sum(a, target_sum)

    # ------------------------------------------------------------------
    # 3. Final cheap random pairwise swaps (like the original routine) –
    #    sometimes they find a last tiny improvement.
    # ------------------------------------------------------------------
    a = np.array(best_seq, dtype=float)
    n = a.size
    target_sum = a.sum()   # should already be correct

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a small fraction of the larger entry towards the smaller one
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = 0.02 * a[src]   # up to 2 % of the source
        if max_move <= 0.0:
            continue
        delta = np.random.uniform(0, max_move)
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # keep non‑negativity (already guaranteed) and sum (unchanged)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.tolist()
            a = cand
            print(f"[swap] improve → {best_score:.6f}")

        if best_score < 1.503:
            break

    print(f"Finished – best score = {best_score:.6f} (length {len(best_seq)})")
    return best_seq
