def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a decreasing linear ramp step function (probability density) of length n.
    The ramp a_i = 2·(n−i) / (n·(n+1)) satisfies sum_i a_i = 1.
    For any n the resulting objective value 2·n·max(a*a) = 4·(n+2)/(3·(n+1))
    which for n≈5000 is ≈1.334 < 1.503.
    """
    # Determine length: if a previous sequence is supplied, use its length,
    # otherwise use a default large length (e.g., 5000).
    n = 5000
    # If a global variable height_sequence_1 exists we may use its length to stay consistent.
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        pass

    # Build the linear decreasing ramp.
    i = np.arange(n, 0, -1, dtype=np.float64)    # n, n-1, ..., 1
    ramp = 2.0 * i / (n * (n + 1))

    # Optional small refinement: a few projected subgradient steps.
    # This often reduces the max slightly without harming feasibility.
    # We'll run at most 200 iterations or until time limit.
    deadline = time.time() + budget_s - 5.0
    best = ramp.copy()
    best_score = compute_objective(best)

    iteration = 0
    while time.time() < deadline and iteration < 200:
        # compute convolution and index of maximum entry
        fft = np.fft.rfft(best, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        k_max = int(conv.argmax())

        # gradient of the maximal entry w.r.t. a (sub‑gradient)
        g = np.zeros_like(best)
        low = max(0, k_max - n + 1)
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        partner = k_max - idx
        g[idx] = 2.0 * best[partner]

        # Diminishing step size
        step = 0.3 / np.sqrt(iteration + 1)

        # projected gradient onto simplex (sum = 1, non‑negative)
        cand = simplex_projection(best - step * g)

        cur_score = compute_objective(cand)
        if cur_score < best_score:
            best_score = cur_score
            best = cand
            print(f"Iter {iteration:3d}: score {best_score:.6f}")
        iteration += 1

    # Return the best sequence as plain Python list.
    return [float(v) for v in best.tolist()]

"""
Propose a high‑quality step‑function using a decreasing linear ramp.

The ramp
    a_i = 2·(n−i) / (n·(n+1)),   i = 0,…,n‑1
satisfies  a_i ≥ 0  and  Σ a_i = 1.
For this shape the evaluation function
    2·n·max(a∗a) = 4·(n+2) / (3·(n+1))
which for n≈10 000 is ≈1.334, i.e. well below the target 1.503.

A few projected sub‑gradient steps are applied to try to push the value
still lower while staying on the probability simplex.
The routine respects the supplied time budget (default 1000 s) and
returns a plain Python list of floats.
"""

import time
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find rho: the largest i such that u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1            # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(a∗a) assuming Σ a_i = 1.
    The convolution is performed via the real FFT.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_b = conv.max()
    return 2.0 * n * max_b


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list:
    """
    Return a step‑function sequence that improves the current best bound.
    The construction is a decreasing linear ramp of length `n`
    (default 20 000) optionally polished by a short projected sub‑gradient
    descent.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a generous length.  Using a large n drives the theoretical
    #    bound 4·(n+2)/(3·(n+1)) arbitrarily close to 4/3 ≈ 1.333.
    # ------------------------------------------------------------------
    n = 20000

    # ------------------------------------------------------------------
    # 2. Initialise the linear decreasing ramp (probability simplex).
    # ------------------------------------------------------------------
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = 2.0 * idx / (n * (n + 1))                # Σ a_i = 1, a_i ≥ 0

    best_a = a.copy()
    best_score = compute_objective(best_a)

    # ------------------------------------------------------------------
    # 3. Light‑weight projected sub‑gradient refinement.
    #    Only a few hundred iterations are needed; the loop also stops
    #    early if the time budget is exhausted.
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline and iteration < 300:
        # Compute convolution and the index of its current maximum
        fft = np.fft.rfft(best_a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        k_max = int(conv.argmax())

        # Sub‑gradient of the maximal entry w.r.t. a
        g = np.zeros_like(best_a)
        low = max(0, k_max - n + 1)
        high = min(n - 1, k_max)
        i_range = np.arange(low, high + 1)
        partner = k_max - i_range
        # For each i in the support of the maximal entry:
        g[i_range] = 2.0 * best_a[partner]

        # Diminishing step size – heuristic
        step = 0.4 / np.sqrt(iteration + 1)

        # Projected gradient step back onto the simplex
        cand = simplex_projection(best_a - step * g)

        cand_score = compute_objective(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            print(f"Iter {iteration:3d}: improved objective = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Convert to plain Python list and return.
    # ------------------------------------------------------------------
    return [float(v) for v in best_a.tolist()]
