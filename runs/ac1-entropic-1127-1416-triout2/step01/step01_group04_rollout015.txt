    a = a / a.sum()

        A = np.zeros((rows, n), dtype=np.float64)
        # Using slice assignment
        for j in range(n):
            A[j:j+n, j] = a

            break

        max_conv_new = conv_new.max()
        score_new = 2.0 * n * max_conv_new

"""
Optimisation of the step‑function bound.

The algorithm explores a variety of simple parametric families (constant‑convolution,
uniform, exponential and power‑law shapes) and then refines each candidate by a
smooth‑max gradient descent.  The objective is

    F(a) = 2·n·max_k (a*a)[k] / (∑a_i)^2

which, after normalising ∑a_i = 1, reduces to 2·n·max_k(a*a)[k].
The smooth‑max surrogate

    Sβ(a) = (1/β)·log(∑_k exp(β·(a*a)[k]))

is differentiable; its gradient can be evaluated with two FFT‑based convolutions.
A projected gradient step (onto the probability simplex) is taken with a
decaying step‑size.  Several random initialisations are tried within the 1000 s
budget and the best sequence found is returned.

No external files or network access are used; only numpy and the provided
`evaluate_sequence` function are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        { x | x_i >= 0, sum x_i = 1 }.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    # Fast O(n log n) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate Sβ(a) with respect to a.
    Returns a vector of the same length as a.
    """
    # a*a (full convolution)
    conv = np.convolve(a, a)                # length 2·n‑1
    m = conv.max()
    # weights for the soft‑max; all entries ≤ 1
    w = np.exp(beta * (conv - m))

    # gradient = 2 * (w ⋆ a)  (cross‑correlation)
    # cross‑correlation can be obtained via a convolution with the reversed a
    full = np.convolve(w, a[::-1])          # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1: n - 1 + n]     # slice corresponding to indices 0…n‑1
    return grad


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) / (sum(a)^2).  The sequence a is assumed to be
    non‑negative.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_conv = conv.max()
    s = a.sum()
    return float(2.0 * n * max_conv / (s * s))


def _constant_conv_base(n: int) -> np.ndarray:
    """
    First n terms of a_k = C(2k,k) / 4**k.
    This sequence satisfies (a*a)[k] = 1 for k < n (before normalisation).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """e^{-lam·i} normalised to L1 = 1."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    return a / a.sum()


def _make_power_law(n: int, p: float) -> np.ndarray:
    """(i+1)^{-p} normalised to L1 = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a / a.sum()


def _refine_candidate(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """
    Perform smooth‑max gradient descent starting from a0 (already normalised
    to sum = 1).  Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    iteration = 0
    step0 = 0.5                # initial step size (decays ~1/√it)
    beta = 1.0                 # start with a gentle smoothing
    beta_increment = 0.03      # slowly tighten the surrogate

    while time.time() < deadline - 0.1 and iteration < max_iter:
        step = step0 / np.sqrt(iteration + 1.0)

        # gradient of the smooth‑max surrogate
        grad = _smooth_gradient(a, beta)

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        # occasional tiny jitter to jump out of plateaus
        if iteration % 400 == 0 and iteration > 0:
            a_new += np.random.randn(n) * 5e-4
            a_new = _project_simplex(a_new)

        val = _objective(a_new)

        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            print(f"Iter {iteration:6d} – improved objective {best_val:.6f}")

        a = a_new
        beta = min(beta + beta_increment, 50.0)   # cap β to avoid overflow
        iteration += 1

    return best_seq, best_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Length selection (reuse a known construction if available)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 2000

    print(f"Search will use length n = {n}")

    # ------------------------------------------------------------------
    # Prepare a list of diverse starting points
    # ------------------------------------------------------------------
    starts = []

    # 1) classic constant‑convolution sequence (normalised)
    a_const = _constant_conv_base(n)
    a_const = a_const / a_const.sum()
    starts.append(a_const)

    # 2) uniform distribution
    starts.append(np.full(n, 1.0 / n))

    # 3) exponential families
    for lam in (0.0005, 0.0010, 0.0020):
        starts.append(_make_exponential(n, lam))

    # 4) power‑law families (including the exact ½‑exponent)
    for p in (0.40, 0.45, 0.55, 0.60):
        starts.append(_make_power_law(n, p))

    # 5) a few random Dirichlet samples (simple stochastic seeds)
    for _ in range(3):
        rand = np.random.dirichlet(alpha=np.ones(n))
        starts.append(rand)

    # ------------------------------------------------------------------
    # Run the refinement procedure on each start, keep the best overall
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float('inf')
    start_id = 0

    for a0 in starts:
        if time.time() >= deadline - 0.5:
            break
        start_id += 1
        print(f"\n--- Optimising start #{start_id} (initial objective = {_objective(a0):.6f}) ---")
        best_seq, best_val = _refine_candidate(a0, n, deadline)

        if best_val < global_best_val:
            global_best_val = best_val
            global_best_seq = best_seq.copy()
            print(f"*** New global best objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the supplied evaluate_sequence (optional)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"\nFinal reported objective by evaluate_sequence: {final_score:.6f}")
    except Exception:
        final_score = global_best_val
        print("\nCould not call evaluate_sequence – using internal objective value.")

    print(f"\nSearch finished; best objective = {global_best_val:.6f}")
    return global_best_seq.tolist()
