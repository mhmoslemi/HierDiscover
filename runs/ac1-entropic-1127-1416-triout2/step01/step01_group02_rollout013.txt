def lp_optimize(f, M, target_sum):
    N = len(f)
    rows = 2*N - 1
    # Build matrix A
    A = np.zeros((rows, N), dtype=np.float64)
    for i in range(N):
        A[i:i+N, i] = f[i]
    # Solve linear program: maximize sum(g) subject to A @ g <= M, g >=0
    c = -np.ones(N, dtype=np.float64)  # minimize -sum(g)
    b = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * N
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = np.maximum(res.x, 0.0)
    # Scale to target sum
    sum_g = np.sum(g)
    if sum_g <= 0:
        return None
    scale = target_sum / sum_g
    g = g * scale
    return g

"""
propose_candidate
-----------------
Search for a step‑height sequence that (heuristically) improves on the
theoretical “binomial’’ construction.  The algorithm alternates:

1.  Normalise a candidate `f` so that  Σ f = √(2 n)  (this makes the
    evaluation equal to the maximal auto‑convolution value).
2.  Solve a linear programme that, for the current `f`, finds a vector
    `g` with the largest possible ℓ₁‑norm under the linear constraints
    conv(f, g) ≤ M, where M = max(conv(f, f)).
3.  Rescale `g` to the same ℓ₁‑norm as `f` and mix:
        f ← (1‑t)·f + t·g .
    The mixing parameter `t` is gradually reduced.
4.  Repeat until the time budget expires.  The best sequence seen is
    returned.  If SciPy is unavailable the routine falls back to the
    simple random‑pair transfer heuristic from the previous baseline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper: generate the optimal binomial (central) coefficients
def binomial_sequence(length: int) -> np.ndarray:
    """Return the sequence c_k = (2k choose k) / 4**k for k=0..length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
# fallback: very light stochastic refinement (the original approach)
def fallback_propose_candidate(seed: int = 42,
                               budget_s: int = 1000,
                               **kwargs) -> list[float]:
    """Simple random‑pair transfer optimisation used when SciPy is missing."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 5000

    best_seq = binomial_sequence(n)
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[fallback‑init] n={n}, score={best_score:.6f}")

    cur_seq = best_seq.copy()
    cur_score = best_score
    step_size = 0.02
    decay = 0.9995

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        amount = step_size * np.random.rand() * cur_seq[i]
        cand = cur_seq.copy()
        cand[i] -= amount
        cand[j] += amount
        cand[i] = max(0.0, cand[i])

        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < cur_score or np.random.rand() < 1e-4:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                print(f"[fallback‑improve] score={best_score:.6f}")

        step_size *= decay
        if np.random.rand() < 0.001:
            cur_seq = best_seq.copy()
            cur_score = best_score
            step_size = 0.02

    print(f"[fallback‑final] best score={best_score:.6f}")
    return best_seq.tolist()


# ----------------------------------------------------------------------
# LP helper: maximise Σ g subject to conv(f,g) ≤ M
def _lp_maximise_sum(f: np.ndarray,
                     M: float,
                     target_sum: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j
    subject to  (f * g)[i] ≤ M   (i = 0..2n‑2)   and   g ≥ 0.
    The returned vector is rescaled to have ℓ₁‑norm = target_sum.
    Returns None if the optimisation fails.
    """
    from scipy.optimize import linprog

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix A where (A·g)[i] = Σ_j f_j g_{i‑j}
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f[j]

    c = -np.ones(n, dtype=np.float64)          # minimise –Σg   == maximise Σg
    b_ub = np.full(rows, M, dtype=np.float64)   # upper bound M for each convolution entry
    bounds = [(0.0, None)] * n                  # g_j ≥ 0

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None

    g = np.maximum(res.x, 0.0)                 # guard against tiny negatives
    s = g.sum()
    if s <= 0.0:
        return None
    scale = target_sum / s
    return g * scale


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a step‑function height sequence that aims to lower the
    evaluation function.  The routine prefers the fast linear‑program
    based optimiser (requiring SciPy); if SciPy cannot be imported it
    falls back to the random‑pair transfer heuristic.
    """
    import numpy as np
    import time

    # ------------------------------------------------------------------
    # Prefer a modest size (≤500) that keeps the linear programme cheap.
    # If a previous construction exists we cap its length; otherwise we
    # use 500 entries.
    try:
        n = min(500, len(height_sequence_1))  # type: ignore[name-defined]
    except Exception:
        n = 500

    # ------------------------------------------------------------------
    # Initialise with the canonical binomial sequence, then normalise so
    # that Σ a_i = √(2 n).  With this normalisation the evaluation equals
    # max(conv(a,a)), which is convenient for the LP scheme.
    f = binomial_sequence(n)
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / np.sum(f))

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Try the LP‑based improvement.  If SciPy is missing we use the
    # fallback random search.
    try:
        from scipy.optimize import linprog  # just to test availability
        have_scipy = True
    except Exception:  # pragma: no cover
        have_scipy = False

    deadline = time.time() + budget_s - 0.1
    iteration = 0
    mix = 0.5                     # initial mixing weight
    # Main optimisation loop (LP + mixing)
    while have_scipy and time.time() < deadline:
        # current maximal auto‑convolution
        M = float(np.max(np.convolve(f, f)))

        # linear programme – maximise the ℓ₁ norm of a new candidate g
        g = _lp_maximise_sum(f, M, target_sum)
        if g is None:
            print("[lp] solver failed – aborting LP loop")
            break

        # convex combination (mixing) and evaluation
        new_f = (1.0 - mix) * f + mix * g
        new_score = evaluate_sequence(new_f.tolist())

        # keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_seq = new_f.copy()
            print(f"[iter {iteration}] improved score={best_score:.6f}")
            # after a success we keep a relatively large mixing factor
            mix = 0.5
        else:
            # slowly decay the mixing factor; when it gets tiny we restart
            mix *= 0.98
            if mix < 0.05:
                # restart from the current best (helps escape plateaus)
                f = best_seq.copy()
                mix = 0.5
                continue

        f = new_f
        iteration += 1
        if iteration >= 2000:      # safety guard
            break

    # ------------------------------------------------------------------
    # If no linear‑programming steps were performed (SciPy missing) we
    # fall back to the simple stochastic refinement.
    if not have_scipy:
        return fallback_propose_candidate(seed=seed,
                                          budget_s=budget_s,
                                          **kwargs)

    # ------------------------------------------------------------------
    # A final tiny random fine‑tuning (pairwise mass transfer) often
    # scrapes a little additional improvement.
    cur_seq = best_seq.copy()
    cur_score = best_score
    step = 0.02
    decay = 0.9995

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        amount = step * np.random.rand() * cur_seq[i]
        cand = cur_seq.copy()
        cand[i] -= amount
        cand[j] += amount
        cand[i] = max(0.0, cand[i])

        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < cur_score:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                print(f"[refine] new best={best_score:.6f}")

        step *= decay

    print(f"[final] best score={best_score:.6f}")
    # Return a plain Python list as required by the external framework
    return best_seq.tolist()
