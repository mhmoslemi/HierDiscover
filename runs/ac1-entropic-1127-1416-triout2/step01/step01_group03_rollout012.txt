"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

Algorithm
---------

1. **Parameter sweep** –   Sample a two‑parameter family
       a_i(p,δ) ∝ (i + δ + 1)^(-p)   (p∈[0.5,0.95], δ≥0)
   Normalise each candidate so that its sum is 1 and evaluate the
   fast‑computed objective `score = 2·n·max(a*a) / (∑a)²`.
   Keep the best candidate.

2. **Refinement** – starting from the best candidate,
   perform a projected (exponentiated) gradient descent that directly
   minimises the maximal convolution entry.
   * Compute the self‑convolution by FFT (O(n log n)).
   * Locate the index `k` where the convolution attains its maximum.
   * The sub‑gradient of the maximum entry is
        g_i = 2·a_{k‑i}   (for i where k‑i is a valid index, 0 otherwise).
   * Update multiplicatively:   a ← a·exp(‑η·g) and project back onto the
     probability simplex (non‑negative, sum = 1).
   * Learning‑rate η decays slowly; after a long period without improvement
     a tiny random jitter is added to escape plateaus.

The whole procedure respects the 1000 s budget, constantly checking the
deadline.  The final sequence (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level as required by the execution platform
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the objective 2·n·max(conv) / (sum(a)²) using FFT.
    a  – non‑negative vector, sum(a) > 0.
    fft_len – power‑of‑two length >= 2·len(a)-1 (pre‑computed for speed).
    """
    n = a.shape[0]
    # real‑FFT of the sequence, element‑wise product gives the convolution
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    # small negative values can appear due to round‑off; clip them
    conv = np.maximum(conv, 0.0)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def max_conv_gradient(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of the maximal convolution entry.
    For the index `max_idx` where (a*a) is maximal, the gradient is
        g_i = 2·a_{max_idx-i}   for i in the overlap range,
        g_i = 0               otherwise.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    # Overlap range where both indices exist
    start = max(0, max_idx - (n - 1))
    end = min(max_idx, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        grad[idx] = 2.0 * a[max_idx - idx]
    return grad

        # if we are stuck for a while, add a tiny random jitter
        if stagnant > 2000:
            jitter = 1e-4 * np.random.randn(n)
            a = project_to_simplex(a + jitter)
            eta = max(eta * 0.7, eta_min)
            stagnant = 0
            print("[%06d] jitter injected, η=%.5f" % (iteration, eta))

        iteration += 1

    # ------------------------------------------------------------------
    # Final verification with the official evaluator (optional but cheap)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print("[final] best score reported by evaluate_sequence = %.6f"
              % final_score)
    except Exception:
        final_score = best_score
        print("[final] using internal score = %.6f" % final_score)

    # Return a plain Python list as required by the surrounding harness
    return best_seq.tolist()
