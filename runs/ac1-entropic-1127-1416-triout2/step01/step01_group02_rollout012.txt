def compute_objective(a):
    n = len(a)
    m = 1 << ((2*n - 1).bit_length())   # next power of 2
    A = np.fft.rfft(a, n=m)

    max_b = conv.max()

    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a * sum_a)

"""
propose_candidate
-----------------
Search for a step‑height sequence that beats the simple binomial
construction.  The idea is:

1.  Start from the optimal “central‑binomial’’ sequence f (the discrete
    analogue of t⁻¹ᐟ² whose auto‑convolution is constant = 1).

2.  Solve a linear program that maximises the ℓ₁‑norm of a non‑negative
    vector b while keeping the mixed convolution (f∗b) bounded by the
    current maximal auto‑convolution M = max(f∗f).  This adds as much
    mass as possible in the tail where f is tiny, without raising the
    bound M.

3.  Rescale b so that its ℓ₁‑norm equals √(2 n) (the normalising sum used
    in the original formulation) and mix it with f:
       f ← (1−t)·f + t·b .
    The mixing factor t is kept small at first; after a few iterations
    the bound M shrinks and the objective 2·n·M/‖f‖₁² drops below the
    pure binomial value (≈ π/2).

4.  Finally a cheap stochastic local‑search (pair‑wise mass transfer)
    is run for the remaining time to polish the solution.

The whole routine respects the 1000‑second budget, works for any length
available via the global ``height_sequence_1`` (capped at 3000 to keep
the linear programme tractable), and returns a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Central binomial coefficients c_k = (2k choose k) / 4**k for k=0..length-1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the linear convolution seq * seq (FFT based)."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    ft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def obj_value(seq: np.ndarray) -> float:
    """Exact objective 2 * n * max(conv) / (sum(seq)**2)."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return np.inf
    m = conv_max(seq)
    return 2.0 * n * m / (s * s)


# ----------------------------------------------------------------------
def random_pair_transfer(seq: np.ndarray, frac: float = 0.02) -> np.ndarray:
    """Move a tiny amount of mass from one index to another (keeps ℓ₁ unchanged)."""
    n = seq.size
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return seq.copy()
    amount = frac * np.random.rand() * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= amount
    new_seq[j] += amount
    # guard against negative numbers created by rounding
    if new_seq[i] < 0.0:
        new_seq[i] = 0.0
    return new_seq


# ----------------------------------------------------------------------
def build_lp_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the Toeplitz matrix A (shape (2n-1, n)) such that
        (A @ b)[k] = (f * b)[k]   for k = 0 .. 2n-2 .
    All entries are non‑negative; the matrix is dense but small enough for
    n ≤ 3000.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes f[0:n] shifted by +j
    for j in range(n):
        A[j : j + n, j] = f[: rows - j]
    return A


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a step‑function height sequence that (heuristically) minimises
    evaluate_sequence().  The algorithm mixes a binomial seed with a
    linear‑program‑derived tail and finishes with a fast local‑search.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # Length selection -------------------------------------------------
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]
    except Exception:

        n = 3000

    # 1) canonical binomial seed
    f = binomial_sequence(n)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, binomial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) linear‑program step (only if we have enough time)
    if time.time() + 30 < deadline:          # reserve a few seconds for LP
        M = conv_max(f)                       # current max of f*f, should be ≈1
        print(f"[lp] solving LP with bound M={M:.6f} ...")
        A = build_lp_matrix(f)
        ub = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)     # maximise sum(b)  → minimise -sum(b)
        bounds = [(0, None)] * n

        # use the efficient HiGHS interior simplex solver

            if mixed_score < best_score:

                best_score = mixed_score
                f = mixed.copy()               # continue from improved point
                print("[mix] accepted improvement")
        else:

    # ------------------------------------------------------------------
    # 3) cheap stochastic refinement (pairwise mass transfers)
    cur_seq = best_seq.copy()
    cur_score = best_score
    step_frac = 0.03

    while time.time() < deadline:

            cur_score = cand_score

        # occasional restart from the best known point

            cur_seq = best_seq.copy()
            cur_score = best_score
            step_frac = 0.03

    print(f"[final] best score={best_score:.6f}")
    # Return plain Python list as required by the external framework
    return best_seq.tolist()
