"""
propose_candidate
-----------------
Search for a step‑function (a list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the discrete convolution used in `evaluate_sequence`.

The strategy is three‑fold:

1. **Coarse grid** – try several lengths (n) together with a
   dense grid of the two most influential parameters:
   exponent `p` (≈0.5) and an additive offset `α`.  The raw
   sequence is `a_i = (i+1+α)^{-p}` and is scaled to any
   magnitude (scaling does not change the objective).

2. **Fine local grid** – zoom in on the best `(p,α)` found in step‑1
   with a finer step size, again for the same length that gave the
   best result.

3. **Targeted mass‑transfer refinement** – starting from the best
   sequence, repeatedly move a small amount of mass from a position
   that contributes most to the current maximal convolution entry to
   a position that contributes the least.  The convolution is updated
   incrementally (O(n) per move) and the move is kept only if the
   evaluation improves.  This deterministic hill‑climb quickly
   pushes the score towards the theoretical limit π/2 ≈ 1.570796.

All heavy work is done with NumPy and FFT‑based convolution
(`rfft`/`irfft`), which is fast enough to evaluate tens of thousands
of candidates within the 1000 s budget.  The final sequence is
returned as a plain Python list; its score is printed for reference.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – they are all top‑level functions (no closures)
# ----------------------------------------------------------------------


def _max_convolution(seq: np.ndarray) -> float:
    """
    Return max_k (seq * seq)[k] using an FFT of size a power of two.
    The convolution length is 2*len(seq)-1.
    """
    n = seq.size
    # smallest power of two >= 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """
    Compute the evaluation function exactly as `evaluate_sequence`
    would (but much faster).  The objective is scale‑invariant, so we
    do not bother with the caps present in `evaluate_sequence`.
    """
    n = seq.size
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    maxb = _max_convolution(seq)
    return float(2.0 * n * maxb / (s * s))


def _make_power_seq(n: int, p: float, offset: float) -> np.ndarray:
    """
    Unnormalised sequence a_i = (i+1+offset)^{-p}.
    """
    idx = np.arange(n, dtype=float) + 1.0
    return 1.0 / np.power(idx + offset, p)


def _incremental_update(conv: np.ndarray, a_old: np.ndarray,
                        i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a new convolution vector after moving `delta` from position i
    to position j (i ≠ j).  The update uses the exact quadratic formula
    (linear term + tiny quadratic correction) and runs in O(n).
    """
    n = a_old.size
    # linear part: -2*delta*shift_i + 2*delta*shift_j
    conv[i : i + n] -= 2.0 * delta * a_old
    conv[j : j + n] += 2.0 * delta * a_old

    # quadratic correction (usually tiny, but kept for exactness)
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    conv[i + j] -= 2.0 * delta * delta
    return conv


def _targeted_refine(seq: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Deterministic hill‑climb that moves a small amount of mass from the
    index contributing most to the current maximal convolution entry to
    the index contributing the least.  Stops when the time limit is hit.
    """
    a = seq.copy()
    n = a.size
    total = a.sum()
    # initial convolution (full length = 2n-1)
    conv = _max_convolution(a)  # we only need the max value here
    # however we also need the full vector for updates; compute once
    # with the fast FFT routine
    size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(a, n=size)
    full = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]

    best_val = _score(a)
    best_seq = a.copy()
    start = time.time()

    # main loop – each iteration costs O(n)
    while time.time() < start + time_limit:
        # 1) index where convolution is maximal
        k_max = int(np.argmax(full))

        # 2) product contributions for this k_max
        partner = k_max - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        prod = np.zeros(n, dtype=float)
        prod[mask] = a[mask] * a[partner[mask]]

        i = int(np.argmax(prod))               # worst contributor
        # pick a j that contributes the least (but still a valid index)
        # avoid the trivial zero‑product case by requiring a partner
        valid_j = np.where(mask & (prod > 0))[0]
        if valid_j.size == 0:
            # fall back to a random different index
            j = (i + 1) % n
        else:
            j = int(valid_j[np.argmin(prod[valid_j])])

        if i == j:
            continue

        # 3) propose a tiny move from i to j
        max_move = 0.02 * a[i]      # 2 % of the source mass
        if max_move <= 0.0:
            continue
        delta = max_move * np.random.rand()

        # keep a copy of the old state for a possible rollback
        a_old = a.copy()
        full_old = full.copy()

        # apply the move
        a[i] -= delta
        a[j] += delta

        # update the convolution incrementally (linear + quadratic terms)
        full = _incremental_update(full, a_old, i, j, delta)

        # evaluate the new sequence
        val = _score(a)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            # keep the updated convolution for the next iteration
        else:
            # rollback – the move was not beneficial
            a = a_old
            full = full_old

        # early exit if we already reach the theoretical bound
        if best_val < 1.571:
            break

    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises the
    expression used in `evaluate_sequence`.  The routine respects a total
    runtime of `budget_s` seconds and prints intermediate progress.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0     # safety margin

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 1. Coarse grid over length, exponent and offset
    # --------------------------------------------------------------
    lengths = [3000, 4000, 5000, 6000, 8000, 10000]
    p_grid = np.linspace(0.48, 0.54, 31)        # step 0.002
    off_grid = np.linspace(0.0, 2.0, 81)       # step 0.025

    for n in lengths:
        for p in p_grid:
            # pre‑allocate denominator part that does not depend on offset
            base = np.arange(n, dtype=float) + 1.0
            for off in off_grid:
                raw = 1.0 / np.power(base + off, p)
                # scaling is irrelevant – we keep the raw vector to avoid extra work
                val = _score(raw)
                if val < best_val:
                    best_val = val
                    best_seq = raw.copy()
                    print(
                        f"[grid] n={n:5d} p={p:.3f} α={off:.3f} → score={val:.6f}"
                    )
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 2. Fine local search around the best (p,α) found above
    # --------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.size
        # extract the parameters of the current best (by a cheap regression)
        # we simply keep using the same n and explore a tighter neighbourhood
        fine_ps = np.linspace(0.495, 0.505, 21)   # step 0.0005
        fine_offs = np.linspace(-0.2, 0.2, 41)   # step 0.01 around the previous offset
        # we need to know the offset that produced `best_seq`; we retrieve it
        # by solving for a single representative element (i=0)
        # (the raw sequence is (1+offset)^{-p})
        #   best_seq[0] = (1+offset)^{-p}
        # => offset = (best_seq[0]**(-1/p) - 1)
        # This is only an approximation but sufficient for a neighbourhood.
        try:
            approx_p = 0.5
            approx_off = max(
                0.0,
                (best_seq[0] ** (-1.0 / approx_p) - 1.0),
            )
        except Exception:
            approx_off = 0.0

        for p in fine_ps:
            base = np.arange(n, dtype=float) + 1.0
            for off in approx_off + fine_offs:
                raw = 1.0 / np.power(base + off, p)
                val = _score(raw)
                if val < best_val:
                    best_val = val
                    best_seq = raw.copy()
                    print(
                        f"[fine] n={n:5d} p={p:.4f} α={off:.3f} → score={val:.6f}"
                    )
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            pass

    # --------------------------------------------------------------
    # 3. Deterministic refinement via targeted mass moves
    # --------------------------------------------------------------
    if best_seq is None:
        # safety net – uniform sequence (score = 2.0)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_val = _score(best_seq)

    # allocate a modest part of the remaining budget for refinement
    remaining = deadline - time.time()
    if remaining > 1.0:
        # use at most half of the remaining time for refinement
        refine_time = min(remaining * 0.5, 300.0)
        refined = _targeted_refine(best_seq, refine_time)
        refined_val = _score(refined)
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined
            print(f"[refine] improved to {refined_val:.6f}")

    # final verification with the official evaluator (just in case)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final best score reported by evaluate_sequence: {final_score:.6f}")
    return best_seq.tolist()
