    a = initial_a.astype(float).copy()
    a = a / np.sum(a)   # ensure sum 1
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for t in range(1, n_iter_max+1):
        if time.time() >= deadline:
            break
        # compute conv
        conv = convolution_fft(a, a)  # length 2n-1
        max_c = np.max(conv)
        # smoothing weight
        lam = lam0 + lam_inc * t  # linear increase
        # compute softmax weights
        # to avoid overflow: shift by max_c
        w = np.exp(lam * (conv - max_c))
        Z = np.sum(w)
        if Z == 0:
            w = np.ones_like(w) / w.size
            Z = 1.0
        # compute gradient
        w_rev = w[::-1]
        conv_rev = convolution_fft(w_rev, a)  # length 3n-2
        # slice indices from offset = len(w)-1 = 2n-2
        offset = len(w) - 1
        grad_slice = conv_rev[offset:offset+n]  # length n
        grad = 2.0 * grad_slice / Z
        # step size
        eta = eta0 / np.sqrt(t)
        # exponentiated update
        a_new = a * np.exp(-eta * grad)
        # add noise
        if noise_every and t % noise_every == 0:
            a_new += np.random.randn(n) * noise_scale
        # renormalize to sum=1
        sum_a_new = np.sum(a_new)
        if sum_a_new <= 0:

            a_new = np.full(n, 1.0/n)
        else:

        # evaluate score

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
        if t % 2000 == 0:
            print(f"  iter={t:,d}  cur={cur_score:.6f}  best={best_score:.6f}  lam={lam:.2f}")

assistantfinal**Solution Overview**

The score returned by `evaluate_sequence` (after fixing the sum of the coefficients to 1) is  

\[
\text{score}=2\,n\;\max_{k}(a*a)_k .
\]

The problem is therefore equivalent to *minimising the maximum entry of the
self‑convolution* of a non‑negative vector that lives on the probability
simplex.  
The maximum is a non‑smooth function, but it can be approximated by the
smooth “log‑sum‑exp’’ (soft‑max)  

\[
F_\lambda(a)=\frac{1}{\lambda}\log\Bigl(\sum_k e^{\lambda\,c_k}\Bigr),
\qquad c_k=(a*a)_k .
\]

When \(\lambda\) is large, \(F_\lambda\) is almost the true maximum, while for
small \(\lambda\) it provides a gentle landscape that is easier to optimise.
Its gradient is

\[
\frac{\partial F_\lambda}{\partial a_i}
 = \frac{2}{Z}\sum_{k}p_k\,a_{k-i},
\qquad
p_k=\frac{e^{\lambda(c_k-\max c)}}{\sum_j e^{\lambda(c_j-\max c)}},
\; Z=\sum_k p_k=1.
\]

The weighted sum can be evaluated with **one FFT‑based convolution**.
We then perform a mirror‑descent step on the simplex
(the Kullback–Leibler projection is realised by the exponentiated‑gradient
update):

```
a_i ← a_i · exp(-η · ∂F/∂a_i) ,   a ← a / sum(a) .
```

A small random noise is added from time to time to escape flat regions.
The smoothing parameter λ is increased linearly during the run so that the
algorithm starts with a smooth objective and finishes by almost exactly
optimising the true max‑convolution.

Several independent runs are launched for different lengths (1024, 2048,
4096) and for a few different initialisations (arcsine‑shape, uniform,
random Dirichlet).  The best sequence found (according to the *exact*
`evaluate_sequence`) is returned.

The whole routine respects the 1000 s time budget, prints progress and never
uses file or network I/O.

```python
"""
Exponential mirror‑descent optimiser for the max‑convolution bound.

The algorithm minimises a smooth approximation of
    max_k (a * a)[k]
by an exponentiated‑gradient update on the probability simplex.
A soft‑max temperature λ is increased during the run, turning the smooth
objective gradually into the genuine max‑convolution.
Several restarts (different lengths, initial vectors) are performed;
the best sequence found before the deadline is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a * b using real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
    {x | x_i ≥ 0 , Σ_i x_i = 1}.
    Duchi et al. (2008) implementation.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring utilities (scale‑invariant version of evaluate_sequence)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact score for a (assumes sum(a) > 0)."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Continuous arcsine density discretised on n points.
    a_i ∝ 1 / sqrt((i+0.5)*(n-0.5-i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Smoothed gradient of the max‑convolution (soft‑max with temperature λ)
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Return ∇F_λ(a) where
        F_λ(a) = (1/λ) log Σ_k exp(λ·c_k),   c = a*a.
    The gradient is
        g_i = 2 / Z  Σ_k  exp(λ·c_k) a_{k-i},
    where Z = Σ_k exp(λ·c_k).
    Computed with a single FFT‑based convolution.
    """
    n = len(a)

    # full self‑convolution c = a * a
    c = convolution_fft(a, a)                # length 2n‑1
    c_max = np.max(c)

    # soft‑max weights (stability by subtraction of the max)
    w = np.exp(lam * (c - c_max))
    Z = np.sum(w)
    if Z == 0.0:
        w = np.full_like(w, 1.0 / w.size)
        Z = 1.0
    # w now contains the (unnormalised) probabilities p_k

    # gradient = 2 * (w_rev * a) / Z   (cross‑correlation)
    w_rev = w[::-1]
    conv_rev = convolution_fft(w_rev, a)          # length 3n‑2
    offset = len(w) - 1                           # = 2n‑2
    weighted_sum = conv_rev[offset:offset + n]    # slice of length n
    grad = 2.0 * weighted_sum / Z
    return grad


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (exponentiated‑gradient on the simplex)
# ----------------------------------------------------------------------


def improve_sequence_smooth(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lam0: float = 0.5,
    lam_inc: float = 5e-4,
    eta0: float = 0.8,
    noise_every: int = 400,
    noise_scale: float = 5e-5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimiser using a smoothed max‑convolution objective.

    Parameters
    ----------
    init_a : ndarray
        Starting vector (will be normalised to sum = 1).
    deadline : float
        Absolute Unix time after which the routine must stop.
    max_iter : int
        Upper bound on the number of iterations.
    lam0, lam_inc : float
        Starting temperature and linear increase per iteration.
    eta0 : float
        Initial step‑size (scaled by 1/√t later).
    noise_every, noise_scale : int, float
        Periodic additive Gaussian noise to prevent stagnation.

    Returns
    -------
    best_a, best_score
    """
    a = np.maximum(init_a.astype(float), 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / len(a))
    else:
        a /= a.sum()                     # enforce simplex

    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # --------- temperature (smoothness) ----------
        lam = lam0 + lam_inc * t

        # --------- gradient of the smooth objective ----------
        grad = softmax_gradient(a, lam)

        # --------- diminishing step size ----------
        eta = eta0 / math.sqrt(t)

        # --------- exponentiated‑gradient update ----------
        a_new = a * np.exp(-eta * grad)

        # ----- occasional tiny random perturbation -----
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        # ----- renormalise to the simplex -----
        s = a_new.sum()
        if s <= 0.0 or np.isnan(s):
            a_new = np.full(n, 1.0 / n)
        else:
            a_new = a_new / s

        # ----- keep the best (exact) score -----
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # ----- periodic progress report -----
        if t % 2000 == 0:
            print(
                f"    iter={t:,d}  cur_score={cur_score:.6f}  "
                f"best={best_score:.6f}  λ={lam:.3f}"
            )

    return best_a, best_score

        # Prepare a few initial vectors
        init_vectors = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            np.random.dirichlet(np.ones(n)),
        ]

        for idx, init in enumerate(init_vectors, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Run {idx}/{len(init_vectors)} (init = {'arcsine' if idx==1 else 'uniform' if idx==2 else 'random'})")

            seq, val = improve_sequence_smooth(
                init_a=init,
                deadline=deadline,
                max_iter=250_000,
                lam0=0.5,
                lam_inc=6e-4,          # slightly faster growth of λ
                eta0=0.9,
                noise_every=500,
                noise_scale=8e-5,
            )
            print(f"    best score this run = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** New global best = {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 3) Final sanity check – evaluate with the official function.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Very last resort: uniform vector of moderate size.
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
    else:
        final_val = evaluate_sequence(best_seq)
        print(f"\nFinished search. Final evaluated score = {final_val:.6f}")

    return best_seq
```
