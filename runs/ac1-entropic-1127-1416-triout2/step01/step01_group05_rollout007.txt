"""
Gradient-guided local search for step function coefficients.

The algorithm starts from the previous best sequence (if available) or from
the classic a_i ∝ 1/√(i+1) shape, normalises the sum to √(2n) and then
iteratively moves a small amount of mass from positions that most contribute
to the current maximal convolution value to positions that contribute the
least.  The max‑convolution is computed with an FFT for speed and the move
size decays over time.  Simulated‑annealing style uphill moves are
occasionally accepted to escape plateaus.  The search runs until the
time budget (≈ 1000 s) is exhausted and returns the best sequence found.
"""

import time
import numpy as np

def _max_convolution(seq):
    """Return the full linear convolution a*a using a real FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())   # next power of two
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)[: 2 * n - 1]
    return conv

    """Fast evaluation of the objective (identical to evaluate_sequence
    for non‑negative sequences that satisfy the sum constraint)."""
    n = seq.shape[0]
    total = seq.sum()
    if total < 0.01:
        return np.inf
    conv = _max_convolution(seq)
    max_conv = conv.max()
    return float(2.0 * n * max_conv / (total * total))

"""
Gradient‑guided stochastic optimisation of a non‑negative step function.

The programme starts from the best sequence discovered so far (the global
variable ``height_sequence_1``) or, if that is unavailable, from the classic
inverse‑square‑root shape a_i ∝ 1/√(i+1).  The sequence is normalised so that
∑a_i = √(2 n); with this normalisation the evaluation function
``evaluate_sequence`` reduces to the maximal value of the self‑convolution
a*a.

During the search we repeatedly
  1. compute the current convolution (via an FFT for speed) and locate the
     index k where it is maximal,
  2. form the sub‑gradient of the maximal term – for index i the gradient
     is 2 · a_{k‑i} if 0 ≤ k‑i < n,
  3. pick a source index i with probability proportional to this gradient
     (large contribution to the peak) and a destination index j with
     probability inversely proportional to the gradient,
  4. move a tiny fraction of mass from i to j (the fraction decays over the
     run) and evaluate the new maximal convolution.
If the new sequence improves the best value it is kept; otherwise it may
still be accepted with a small temperature‑controlled probability
(simulated‑annealing) to help escape plateaus.

All heavy work (convolution, score) is performed with a real FFT, which
makes each iteration cheap enough to run thousands of steps within the
≈ 1000 s budget.  At the end the candidate is verified with the official
``evaluate_sequence`` and returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions (top‑level, no closures)
# --------------------------------------------------------------

def _max_convolution(seq: np.ndarray) -> np.ndarray:
    """
    Return the full linear convolution a*a using a real FFT.
    The length of the result is 2·n‑1.
    """
    n = seq.shape[0]
    # smallest power of two ≥ 2·n‑1 (FFT works fastest with powers of two)
    m = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)[: 2 * n - 1]
    return conv


def _score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective.
    For non‑negative sequences with a realistic sum this is identical to
    ``evaluate_sequence`` because `evaluate_sequence` finally returns
    2·n·max(conv)/(sum**2).
    """
    n = seq.shape[0]
    total = seq.sum()
    if total < 0.01:
        return np.inf
    max_conv = _max_convolution(seq).max()
    return float(2.0 * n * max_conv / (total * total))
