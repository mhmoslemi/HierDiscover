"""
propose_candidate
=================

Search for a step‑function (non‑negative sequence with unit ℓ¹‑norm) that
minimises the objective used in ``evaluate_sequence`` :

    score = 2·n·max_k (a * a)[k]    (the ℓ¹‑norm is forced to 1, scaling‑invariant)

The algorithm works in three stages:

1. **Beta‑family sweep** –   For a family
        a_i(α) ∝ ((i+½)(n‑i‑½))^{α‑1} ,   α∈(0,½]
   the continuous analogue is a symmetric Beta(α,α) density.
   A bounded scalar optimisation (SciPy ``minimize_scalar``) finds the
   α that gives the smallest score for the chosen length ``n``.
   The resulting sequence is used as the starting point.

2. **Simulated‑annealing / hill‑climbing** –   Randomly transfer a small
   amount of mass between two positions while preserving the ℓ¹‑norm.
   Acceptance follows a temperature schedule (Metropolis criterion) which
   allows occasional uphill moves early on.  The step size shrinks
   linearly with time, giving fine‑grained refinements near the end.

3. **Final sanity check** –   The best sequence seen is evaluated with the
   provided ``evaluate_sequence`` function and returned.

The whole search respects the supplied time budget (default 1000 s) and
prints progress information (initial score, improvements, final score).

The method is fast: building the candidate sequence is O(n), each
convolution is O(n²) (≈4 · 10⁶ operations for n=2000) and tens of
thousands of iterations easily fit into the 1000 s window.
"""
import time
import math
import numpy as np
from scipy.optimize import minimize_scalar, linprog


def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Return a normalised non‑negative sequence of length ``n`` belonging to the
    symmetric Beta(α,α) family.
    a_i ∝ ((i+0.5)*(n-i-0.5))^{α‑1}.  The sum of the returned vector is 1.
    """
    # centre points of the intervals
    idx = np.arange(n, dtype=np.float64) + 0.5
    prod = idx * (n - idx)                 # (i+0.5)*(n-i-0.5) > 0
    # exponent α‑1 is negative for the relevant range (α≤0.5)
    a = np.power(prod, alpha - 1.0, out=np.empty_like(prod))
    s = a.sum()
    # Guard against pathological cases (should never happen)
    if s == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _score_of_sequence(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) – the quantity minimised by evaluate_sequence."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return 2.0 * n * conv.max()


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    Returns the best list of floats found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # small safety margin

    # ------------------------------------------------------------------
    # Choose a length.  Prefer the length of a previously found construction.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1,
                                                       (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 2000

    # ------------------------------------------------------------------
    # 1)  Sweep the Beta(α,α) family (α∈(0,½]) to obtain a good start.
    # ------------------------------------------------------------------
    def _objective_alpha(alpha):
        # minimise the score, not the raw objective of the optimizer
        seq = _beta_sequence(n, alpha)
        return _score_of_sequence(seq)

    # bounded scalar optimisation – “bounded” = Brent’s method on an interval
    res = minimize_scalar(_objective_alpha,
                          bounds=(0.05, 0.5),  # avoid extreme α that cause numerical issues
                          method='bounded',
                          options={'maxiter': 30})   # a few dozen evaluations are enough
    best_alpha = float(res.x)
    best_seq = _beta_sequence(n, best_alpha)
    best_score = _score_of_sequence(best_seq)
    print(f"Initial β‑sweep: α={best_alpha:.6f} → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Simulated‑annealing / hill‑climbing from the β‑optimal start.
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    it = 0
    # Linear schedules for temperature and maximal transfer step
    # Temperature controls acceptance of uphill moves; step size controls the
    # magnitude of the mass transfer between two coordinates.
    temp_start = 1e-2
    temp_end = 1e-7
    step_start = 0.05      # move up to 5 % of the total mass in early phase
    step_end = 0.001       # finish with very fine adjustments

    while time.time() < deadline:
        # fraction of elapsed time (0 → start, 1 → deadline)
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        # exponential decay can be used; linear works well for this scale
        temperature = temp_start * (1.0 - frac) + temp_end * frac
        max_step = step_start * (1.0 - frac) + step_end * frac

        # propose a random mass transfer between two distinct indices
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)
        # Choose direction uniformly
        if np.random.rand() < 0.5:
            # move from i → j
            available = cur_seq[i]
            delta = np.random.rand() * min(max_step, available)
            cand = cur_seq.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # move from j → p
            available = cur_seq[j]
            delta = np.random.rand() * min(max_step, available)
            cand = cur_seq.copy()
            cand[i] += delta
            cand[j] -= delta

        # No need to re‑normalise – transfer preserves ℓ¹‑norm.
        # Compute the new score.
        cand_score = _score_of_sequence(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / temperature):
            cur_seq = cand
            cur_score = cand_score

        # Record global best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Iter {it:6d}: new best score {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 3) Final sanity check using the provided evaluate_sequence (optional).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final score reported by evaluate_sequence: {final_score:.6f}")
    except Exception:
        final_score = best_score

    print(f"Search finished after {it} iterations. Best score found: {best_score:.6f}")
    return best_seq.tolist()
