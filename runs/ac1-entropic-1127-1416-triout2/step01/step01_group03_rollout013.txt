"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

Algorithm
---------
1.  Initialise the coefficients with the (1‑z)^(-1/2) distribution or,
    if the previous best sequence `height_sequence_1` is available, start
    from it (normalised to sum = 1).

2.  Main optimisation loop (gradient descent on the *exact* max‑convolution):
    *   Compute the self‑convolution `b = a*a` via a fast FFT.
    *   Determine the active set `S = {k | b_k = max(b)}` (all indices
        whose value is within a tiny tolerance of the maximum).
    *   The sub‑gradient of `max(b)` at `a` is
        `g_i = 2·∑_{k∈S} a_{k‑i}`.
        Using FFT we obtain `g` as a single convolution of `a` with the
        indicator of `S`.
    *   Exponentiated‑gradient update
            a ← a · exp(−step·g)   (element‑wise)
        and projection onto the probability simplex (sum = 1, non‑negative).
    *   Step size `step = step0 / sqrt(iter+1)` decays slowly.
    *   Keep the best sequence seen so far.  If no improvement is observed
        for a while, inject a small random multiplicative jitter to escape
        plateaus; after very long stagnation perform a full random restart
        while preserving the best solution.

3.  Fine‑grained local search: repeatedly transfer a tiny amount of mass
    from one index to another (keeping the sum = 1) and accept the move
    when it reduces the objective.

The objective after normalising the sum to 1 is simply
    score = 2·n·max(a*a) ,
so we evaluate it analytically without calling the external
`evaluate_sequence` during the search (the final returned list will be
checked by the harness).  FFT‑based convolutions make the inner loop
very fast, allowing tens of thousands of iterations within the 1000 s
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level)
# ----------------------------------------------------------------------


def sqrt_distribution(n: int) -> np.ndarray:
    """Normalised coefficients of (1‑z)^(-1/2) up to length *n*.

    The raw coefficients satisfy
        c₀ = 1,
        c_{i+1} = c_i·(2i+1)/(2i+2).
    They are then divided by their sum so that Σc = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return a non‑negative vector with the same shape as *v* whose entries sum to 1.
    Simple scaling after clipping is sufficient for our purpose.
    """
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def next_pow_two(x: int) -> int:
    """Small utility: next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def self_convolution_fft(a: np.ndarray, N: int) -> np.ndarray:
    """Return a*a using FFT of length *N* (full length 2·len(a)-1)."""
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[: 2 * a.shape[0] - 1]
    return c


def gradient_active_max(a: np.ndarray, b: np.ndarray, N: int) -> np.ndarray:
    """
    Compute the sub‑gradient of max(b) at the current *a*.

    Parameters
    ----------
    a : ndarray (n,)
        Current coefficient vector (sum = 1, non‑negative).
    b : ndarray (2n‑1,)
        Self‑convolution a*a (full length).
    N : int
        FFT size (≥ 3·n‑2).

    Returns
    -------
    g : ndarray (n,)
        Sub‑gradient (2·conv(a, 𝟙_S)), where S is the active set.
    """
    n = a.shape[0]
    max_b = b.max()
    # tolerance – use a relative value to catch floating‑point ties
    eps = max_b * 1e-12 + 1e-14
    indicator = (b >= max_b - eps).astype(np.float64)          # shape 2n‑1
    # reversed indicator for the convolution formula used below
    w_rev = indicator[::-1]
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(w_rev, N)
    conv_rev = np.fft.irfft(A * B, N)[: len(w_rev) + n - 1]    # length 3n‑2
    L = indicator.size
    # g_i = 2·conv_rev[(L−1)−i] for i = 0..n‑1
    g = 2.0 * conv_rev[(L - 1) - np.arange(n)]
    return g

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` score.  Returns the best sequence found
    before the time limit expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose length (use the existing construction when available)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be provided by the harness
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = init_seq.shape[0]
    except Exception:
        n = 2145                     # default used by the previous run
        init_seq = None

    # ------------------------------------------------------------------
    # Initialise a (normalised to sum = 1)
    # ------------------------------------------------------------------
    if init_seq is not None:
        a = project_to_simplex(init_seq.copy())
    else:
        a = sqrt_distribution(n)

    # Ensure exact normalisation
    a = project_to_simplex(a)

    # FFT size – must accommodate the longest convolution (a * indicator)
    N_fft = next_pow_two(3 * n - 2)

    # initial score (objective = 2·n·max(a*a) )
    b = self_convolution_fft(a, N_fft)
    cur_score = 2.0 * n * b.max()
    best_seq = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    step0 = 0.6                 # initial step for exponentiated update
    stagnant = 0                # iterations since last best improvement
    iteration = 0
    jitter_thresh = 5000        # trigger jitter after this many stagnations
    restart_thresh = 30000      # full restart after this many stagnations
    max_iter = 10 ** 7          # safety upper bound (will be limited by time)

    # ------------------------------------------------------------------
    # Main optimisation loop (gradient descent on exact max)
    # ------------------------------------------------------------------
    while time.time() < deadline and iteration < max_iter:
        # 1) self‑convolution and active set
        b = self_convolution_fft(a, N_fft)
        max_b = b.max()
        # 2) sub‑gradient of the max‑convolution
        g = gradient_active_max(a, b, N_fft)

        # 3) step size (decays like 1/√t)
        step = step0 / np.sqrt(iteration + 1.0)

        # 4) exponentiated‑gradient update (keeps positivity)
        exp_term = -step * g
        # avoid overflow / underflow in the exponential
        exp_term = np.clip(exp_term, -50.0, 50.0)
        a_new = a * np.exp(exp_term)
        a_new = project_to_simplex(a_new)

        # 5) evaluate the new point (fast analytic score)
        b_new = self_convolution_fft(a_new, N_fft)
        new_score = 2.0 * n * b_new.max()

        # 6) accept only non‑worsening moves (monotone descent)
        if new_score <= cur_score + 1e-15:
            a = a_new
            cur_score = new_score
        # else keep the old a (the step was too aggressive)

        # 7) keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            stagnant = 0
            print(f"[{iteration:06d}] improved best score = {best_score:.6f}")
        else:
            stagnant += 1

        # ------------------------------------------------------------------
        # Adaptive heuristics
        # ------------------------------------------------------------------
        # a) small jitter to shake out of plateaus
        if stagnant > jitter_thresh:
            sigma = 0.02
            noise = sigma * np.random.randn(n)
            a = a * np.exp(noise)          # multiplicative noise
            a = project_to_simplex(a)
            # recompute current score after jitter
            b = self_convolution_fft(a, N_fft)
            cur_score = 2.0 * n * b.max()
            stagnant = 0
            print(f"[{iteration:06d}] jitter applied, cur_score={cur_score:.6f}")

        # b) full random restart after very long stagnation (keep best)
        if stagnant > restart_thresh:
            a = np.random.rand(n)
            a = project_to_simplex(a)
            b = self_convolution_fft(a, N_fft)
            cur_score = 2.0 * n * b.max()
            stagnant = 0
            step0 = 0.6
            print(f"[{iteration:06d}] full restart performed")

        # c) slowly reduce the base step size (helps convergence)
        if iteration % 5000 == 0 and iteration > 0:
            step0 *= 0.96
            step0 = max(step0, 0.02)

        iteration += 1

    # ------------------------------------------------------------------
    # Fine‑grained local search (mass‑transfer moves) while time permits
    # ------------------------------------------------------------------
    local_iter = 0
    while time.time() < deadline:
        # propose a tiny transfer of mass from i -> j
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best_seq[i] <= 1e-16:
            continue
        # transfer a small fraction of the mass at i
        frac = 0.005 * np.random.rand()   # up to 0.5 % of a_i
        delta = best_seq[i] * frac
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        # evaluate candidate
        b_cand = self_convolution_fft(cand, N_fft)
        cand_score = 2.0 * n * b_cand.max()
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand
            print(f"[local {local_iter:06d}] improved score = {best_score:.6f}")
        local_iter += 1
        # safety break (in case of an extremely tight deadline)
        if local_iter % 5000 == 0 and time.time() >= deadline - 0.1:
            break

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f} after {iteration} main iters, "
          f"{local_iter} local iters")
    # Return a plain Python list as required by the evaluation harness
    return best_seq.tolist()
