    sum_a = arr.sum()

"""
propose_candidate
-----------------
Heuristic optimisation that starts from the optimal (central‑binomial)
sequence and then tries to *move mass from the large early coefficients
to the small tail coefficients*.  The interior coefficients of the
binomial sequence give a constant auto‑convolution equal to 1, which
makes the current best score ≈ π/2.  By shifting a little weight from the
front to the back we can lower the maximal convolution value while
preserving the total L¹‑mass; this reduces the evaluation metric.

The algorithm performs a simulated‑annealing walk:
* choose a source index in the first 10 % of the array (large values)
* choose a target index in the last 10 % (tiny values)
* transfer a random fraction of the source mass to the target
* accept the move if the score improves, otherwise accept with a small
  probability that decreases with time (temperature schedule).

The walk is repeated until the allocated time budget expires
(default 1000 s).  The best sequence found is returned as a plain
Python list, ready for the external framework.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Return c_k = (2k choose k) / 4**k for k = 0 … length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def compute_score(arr: np.ndarray, n: int) -> float:
    """
    Fast version of evaluate_sequence (no type‑checks).
    Returns 2 * n * max(a*a) / (sum(a)**2).
    """
    # linear convolution (full)
    conv = np.convolve(arr, arr)
    max_b = np.max(conv)
    s = np.sum(arr)
    # Guard against division by zero – not needed for our search
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function height sequence with a low evaluation
    score.  Starts from the central‑binomial construction and performs
    a biased simulated‑annealing mass‑transfer walk that preferentially
    moves weight from the large front coefficients to the small tail
    coefficients.  The best sequence found within the time budget is
    returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Length of the problem (use previously discovered sequence if
    # available, otherwise a reasonable default).
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 5000

    # --------------------------------------------------------------
    # Initialise with the theoretical “binomial’’ construction.
    cur_seq = binomial_sequence(n).astype(np.float64)
    cur_score = compute_score(cur_seq, n)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Parameters for the simulated‑annealing walk
    early_frac = 0.10        # first 10 % of indices are “source’’
    late_frac = 0.10         # last 10 % are “target’’
    early_end = max(1, int(n * early_frac))
    late_start = max(0, n - int(n * late_frac))

    step_frac = 0.30         # start by moving up to 30 % of source entry
    step_decay = 0.9998      # slowly shrink the transfer size
    min_step = 0.002

    temperature = 0.02       # initial temperature for Metropolis rule
    temp_decay = 0.9999      # geometric cooling
    min_temp = 1e-4

    # --------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # ---- choose a biased (source, target) pair ----
        if np.random.rand() < 0.70:      # 70 % of moves are biased
            i = np.random.randint(0, early_end)          # source – large coeff
            j = np.random.randint(late_start, n)         # target – small coeff
        else:                             # occasional completely random move
            i, j = np.random.randint(0, n, size=2)

        if i == j or cur_seq[i] <= 0.0:
            continue

        # ---- amount of mass to transfer ----
        delta = step_frac * np.random.rand() * cur_seq[i]
        if delta <= 0.0:
            continue

        # ---- propose a new sequence (preserves total L1‑mass) ----
        cand_seq = cur_seq.copy()
        cand_seq[i] -= delta
        cand_seq[j] += delta

        # ---- evaluate ----
        cand_score = compute_score(cand_seq, n)

        # ---- acceptance test (Metropolis) ----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability of accepting a worse move
            prob = math.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_seq = cand_seq
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand_seq.copy()
                best_score = cand_score
                elapsed = time.time() - (deadline - budget_s)
                print(f"[improve] score={best_score:.6f}  time={elapsed:.2f}s")

        # ---- slowly cool down / shrink step ----
        step_frac = max(min_step, step_frac * step_decay)
        temperature = max(min_temp, temperature * temp_decay)

        # ---- occasional restart from the pure binomial to escape dead‑ends ----
        if np.random.rand() < 0.001:
            cur_seq = binomial_sequence(n).astype(np.float64)
            cur_score = compute_score(cur_seq, n)
            step_frac = 0.30
            temperature = 0.02

    print(f"[final] best score={best_score:.6f}")
    # Return a plain Python list – the external framework expects this.
    return best_seq.tolist()
