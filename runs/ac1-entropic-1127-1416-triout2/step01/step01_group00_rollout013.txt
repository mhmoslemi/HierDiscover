def smoothed_max_and_gradient(a, alpha):
    n = len(a)
    c = convolution_fft(a, a)   # length 2n-1
    if alpha == 0:
        # just uniform weights
        w = np.full_like(c, 1.0 / len(c))
    else:
        # compute log-sum-exp in stable way
        max_c = np.max(alpha * c)
        exp_vals = np.exp(alpha * c - max_c)
        sum_exp = np.sum(exp_vals)
        w = exp_vals / sum_exp
    # compute gradient: grad_i = 2 * sum_k w_k a_{k-i}
    # use convolution with reversed w
    w_rev = w[::-1]
    conv = convolution_fft(w_rev, a)  # length = len(w_rev)+len(a)-1 = (2n-1)+n-1 = 3n-2
    # gradient index i = L-1 - i where L= len(w)
    L = len(w)

"""
Propose a low‑score step‑function by minimizing a smooth approximation
of the max‑convolution objective on the probability simplex.

Algorithm
---------
* Work on a fixed length `n` and keep the vector `a` on the simplex
  (non‑negative, sum = 1) using Euclidean projection.
* Instead of the non‑smooth max, minimise the log‑sum‑exp
  `f(a)= (1/α)·log Σ_k exp(α·(a*a)_k)` which converges to the true max as
  α → ∞.  Its gradient is

      ∇f(a)_i = 2· Σ_k w_k·a_{k‑i},
      w_k = exp(α·c_k) / Σ_j exp(α·c_j) ,

  where `c = a*a` is the full self‑convolution.
  The gradient is computed with two FFT‑based convolutions.
* Perform a projected (additive) gradient step with a diminishing step
  size η_t = η₀ / √t and increase the smoothing parameter α
  gradually (α ← α·γ every few thousand iterations).
* Run several restarts (arcsine‑shaped, uniform and noisy variants)
  on a few different lengths (1024 – 4096).  The best sequence seen
  (according to the official `evaluate_sequence`) is returned.
* All operations stay within the allowed time budget; the routine aborts
  gracefully when the deadline is reached.

The code is completely self‑contained, uses only NumPy and the provided
`evaluate_sequence` for the final check, and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v <= 0):
        # Uniform projection if everything is non‑positive.
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via real FFT (fast for the lengths we use)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Scale‑invariant score used in the optimiser:
        score = 2·n·max(a*a) / (sum(a)²)
    The optimiser works with sum(a)=1, hence the denominator is 1.
    """
    n = a.size
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """
    Initialise a distribution that mimics the continuous arcsine density.
    a_i ∝ 1 / sqrt((i+0.5)·(n‑0.5‑i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25               # avoid exact zeros at the boundaries
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def smoothed_max_and_grad(a: np.ndarray, alpha: float):
    """
    Return a smooth approximation of max(a*a) together with its gradient.
    The smooth objective is
        f(a) = (1/α)·log Σ_k exp(α·c_k)   with  c = a*a .
    Gradient:
        ∂f/∂a_i = 2·Σ_k w_k·a_{k-i},
    where w_k = exp(α·c_k) / Σ_j exp(α·c_j).

    Parameters
    ----------
    a     – probability vector (sum = 1, non‑negative)
    alpha – smoothing parameter (α>0, larger -> tighter to true max)

    Returns
    -------
    f_val – smooth objective value (useful for monitoring)
    grad  – gradient ∇f(a)   (numpy array of same shape as a)
    """
    n = a.size
    # full self‑convolution (length 2n‑1)
    c = convolution_fft(a, a)

    # ----- compute softmax weights w_k -----
    if alpha == 0.0:
        # degenerate case – uniform weights (equivalent to averaging)
        w = np.full_like(c, 1.0 / c.size)
        f_val = np.max(c)                     # not really used
    else:
        scaled = alpha * c
        max_scaled = np.max(scaled)            # for numerical stability
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = np.sum(exp_scaled)
        w = exp_scaled / sum_exp
        f_val = (max_scaled + np.log(sum_exp)) / alpha

    # ----- gradient : 2 * (w_rev * a)[L‑1‑i] -----
    # w_rev is w reversed; convolution length = (2n‑1) + n‑1 = 3n‑2
    w_rev = w[::-1]
    conv = convolution_fft(w_rev, a)          # length 3n‑2
    L = w.size                               # = 2n‑1
    idx = (L - 1) - np.arange(n)              # indices L‑1‑i
    grad = 2.0 * conv[idx]
    return f_val, grad


def optimise_length(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    eta0: float = 0.8,
    alpha_start: float = 0.5,
    alpha_max: float = 80.0,
    alpha_inc: float = 1.02,
    inc_every: int = 2000,
    noise_every: int = 0,
    noise_scale: float = 1e-5,
):
    """
    Perform projected gradient descent on the smooth max‑convolution
    objective for a fixed length `n`.

    Returns the best sequence (numpy array) and its (non‑scaled) score.
    """
    a = init_a.astype(float).copy()
    a = project_to_simplex(a)          # enforce simplex at start
    best_a = a.copy()
    best_score = compute_score(a)

    alpha = alpha_start
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- smooth objective & gradient -----
        _, grad = smoothed_max_and_grad(a, alpha)

        # ----- step size (diminishing) -----
        eta = eta0 / np.sqrt(it)

        # ----- optional stochastic perturbation -----
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale

        # ----- gradient step + projection -----
        a = a - eta * grad
        a = project_to_simplex(a)

        # ----- track best according to the true objective -----
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- progress reporting & α schedule -----
        if it % inc_every == 0:
            alpha = min(alpha * alpha_inc, alpha_max)
            print(
                f"      n={n} it={it:,d} α={alpha:.2f} cur_score={cur_score:.6f} best={best_score:.6f}"
            )
    return best_a, best_score
