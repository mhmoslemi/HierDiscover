        # Fill
        for k in range(K):
            i_start = max(0, k-(n-1))

        # Could vectorize using slicing?
        # Simple loops likely okay.

        b_ub = np.full(K, M)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        mx = new_seq.max()

            cur_score = new_score
            best_seq, best_score = cur_seq.tolist(), cur_score
            print(f"LP refinement (t={t}) improved to {cur_score:.6f}")

            # no improvement; break loop
            break

"""
Optimise a non‑negative step‑function for the provided evaluation metric.

The search proceeds in three stages:

1️⃣  Random sampling of a diverse set of parametric families
    (exponential, power‑law, linear, mixed, logistic, double‑exponential,
    gamma‑like and Gaussian shapes, plus random decreasing).  Each candidate
    is normalised so that its maximum entry equals 1 – scaling does not affect
    the score.  The best sequence found so far is kept.

2️⃣  A *linear‑programming refinement* inspired by the state‑of‑the‑art
    alternating scheme.  For the current best sequence `f` we compute
    `M = max(f*f)`.  We then solve

        maximise   Σ g_i
        subject to   f * g ≤ M   (element‑wise convolution)
                     g_i ≥ 0

    (implemented with `scipy.optimize.linprog`).  The optimal `g` is rescaled
    to have the same total mass as `f` and mixed into `f` (50 % blend).  This
    step reduces `max(f*f)` while keeping the ℓ₁‑norm large, driving the
    objective down.

3️⃣  A short stochastic hill‑climbing phase that perturbs a few randomly
    chosen coordinates (mass is transferred while preserving non‑negativity)
    and accepts any improvement.

All phases respect the global time budget (default 1000 s).  The function
returns the best sequence discovered.
"""
import time
import math
import random

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy for the LP refinement
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:  # pragma: no cover
        return math.inf


def _normalize(seq):
    """Scale so that max(seq) == 1 (no‑op if all zeros)."""
    arr = np.asarray(seq, dtype=float)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()


def _exp_seq(n, alpha):
    """Exponential decay a_i = exp(-alpha * i)."""
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)))


def _pow_seq(n, p):
    """Power‑law decay a_i = (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=float)
    return _normalize(idx ** (-p))


def _lin_seq(n, beta):
    """Linear decay a_i = max(0, 1 - beta*i)."""
    return _normalize(np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float)))


def _mix_seq(n, alpha, eps):
    """Mixed exponential + constant a_i = exp(-alpha*i) + eps."""
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)) + eps)


def _logistic_seq(n, steep):
    """Logistic decay a_i = 1/(1+exp((i-mid)/steep))."""
    i = np.arange(n, dtype=float)
    mid = n / 2.0
    seq = 1.0 / (1.0 + np.exp((i - mid) / steep))
    return _normalize(seq)


def _randdec_seq(n):
    """Random decreasing sequence (sorted random numbers)."""
    arr = np.random.rand(n)
    arr.sort()
    arr = arr[::-1]                     # descending
    return _normalize(arr)


def _double_exp_seq(n, alpha, beta, split):
    """Two‑stage exponential: fast then slow decay."""
    split = max(0, min(split, n - 1))
    i = np.arange(n, dtype=float)
    head = np.exp(-alpha * i)
    tail = np.exp(-alpha * split) * np.exp(-beta * (i - split))
    seq = np.where(i <= split, head, tail)
    return _normalize(seq)


def _gamma_seq(n, p, beta):
    """Gamma‑like shape a_i = (i+1)^p * exp(-beta*i)."""
    i = np.arange(1, n + 1, dtype=float)
    seq = (i ** p) * np.exp(-beta * (i - 1))
    return _normalize(seq)


def _gaussian_seq(n, mu, sigma):
    """Centered Gaussian bump (non‑negative)."""
    i = np.arange(n, dtype=float)
    seq = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return _normalize(seq)


# ----------------------------------------------------------------------
# Linear‑programming refinement (stage 2)
# ----------------------------------------------------------------------
def _lp_refine(seq, deadline):
    """Improve *seq* by the LP / mixing step described in the paper."""
    if not _SCIPY_AVAILABLE:
        return seq, _score(seq)

    a = np.array(seq, dtype=float)
    best_seq = seq
    best_score = _score(seq)

    while time.time() < deadline:
        n = len(a)
        conv = np.convolve(a, a)
        M = conv.max()
        K = 2 * n - 1

        # Build sparse constraint matrix A_ub where
        #   (A_ub @ g)_k = Σ_i a_i * g_{k-i}
        rows, cols, data = [], [], []
        for k in range(K):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                rows.append(k)
                cols.append(j)
                data.append(a[i])
        A_ub = csr_matrix((data, (rows, cols)), shape=(K, n))
        b_ub = np.full(K, M, dtype=float)
        bounds = [(0.0, 1000.0)] * n
        c = -np.ones(n, dtype=float)   # maximise Σ g_i  ⇢ minimise -Σ g_i

        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "max_iter": 5000},
            )
        except Exception:  # pragma: no cover
            break

        if not res.success:
            break

        g = np.array(res.x, dtype=float)
        sum_f = a.sum()
        sum_g = g.sum()
        if sum_g == 0.0:
            break

        # Rescale g to have the same ℓ₁‑mass as f
        g = g * (sum_f / sum_g)

        # 50 % convex combination (t=0.5) – enough to keep the process stable
        t = 0.5
        a_new = (1 - t) * a + t * g
        mx = a_new.max()
        if mx > 0:
            a_new = a_new / mx

        cand = a_new.tolist()
        sc = _score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            a = a_new
            print(f"LP mixing improvement → {best_score:.6f}")
        else:
            # No improvement – stop the refinement loop
            break

    return best_seq, best_score

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a low‑score step‑function within the time budget.
    The routine mixes random parametric families, an LP‑based refinement
    and a final stochastic hill‑climb.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.3  # tiny safety margin

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 0) start from any globally supplied reference (height_sequence_1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = _normalize(list(height_sequence_1))
            sc = _score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc
                print(f"Initial reference score: {best_score:.6f}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1) random sampling of many parametric families
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1800, 2400, 3000, 3600, 4200]  # favour a few thousand
    families = [
        "exp",
        "pow",
        "lin",
        "mix",
        "log",
        "randdec",
        "doubleexp",
        "gamma",
        "gaussian",
    ]

    # Phase 1 runs until half of the budget is exhausted
    phase1_deadline = time.time() + 0.5 * budget_s
    while time.time() < min(phase1_deadline, deadline):
        # Occasionally try a cheap hill‑climb on the current best
        if best_seq is not None and random.random() < 0.12:
            cand = np.array(best_seq, copy=True)
            n = len(cand)
            # move a small amount of mass from a random index to another
            i = random.randrange(n)
            j = random.randrange(n)
            if cand[i] > 1e-12:
                delta = 0.1 * cand[i] * random.random()
                cand[i] -= delta
                cand[j] += delta
                mx = cand.max()
                if mx > 0:
                    cand = cand / mx
                cand = cand.tolist()
                sc = _score(cand)
                if sc < best_score:
                    best_seq, best_score = cand, sc
                    print(f"Hill‑climb improvement → {best_score:.6f}")
            continue

        # --------- generate a fresh candidate -------------------------
        n = int(random.choice(lengths))
        family = random.choice(families)

        if family == "exp":
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            cand = _exp_seq(n, alpha)

        elif family == "pow":
            p = random.uniform(0.0, 3.0)
            cand = _pow_seq(n, p)

        elif family == "lin":
            beta = random.uniform(0.0, 2.0 / n)
            cand = _lin_seq(n, beta)

        elif family == "mix":
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            eps = random.uniform(0.0, 0.5)
            cand = _mix_seq(n, alpha, eps)

        elif family == "log":
            steep = random.uniform(0.01, n / 2.0)
            cand = _logistic_seq(n, steep)

        elif family == "randdec":
            cand = _randdec_seq(n)

        elif family == "doubleexp":
            # two decay rates and a split point
            log_a = random.uniform(-4, math.log10(5))
            log_b = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            beta = 10 ** log_b
            split = random.randint(1, n - 2)
            cand = _double_exp_seq(n, alpha, beta, split)

        elif family == "gamma":
            p = random.uniform(-1.0, 2.0)
            beta = random.uniform(0.0, 0.02)
            cand = _gamma_seq(n, p, beta)

        elif family == "gaussian":
            mu = random.uniform(0, n - 1)
            sigma = random.uniform(0.5, n / 4.0)
            cand = _gaussian_seq(n, mu, sigma)

        else:  # should never happen
            cand = _randdec_seq(n)

        if sum(cand) < 1e-8:
            continue

        sc = _score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"New best ({family}) n={n} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) LP‑based refinement (stage 2)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # give the LP step at most one‑third of the remaining budget
        lp_deadline = time.time() + (deadline - time.time()) * 0.33
        best_seq, best_score = _lp_refine(best_seq, lp_deadline)

    # ------------------------------------------------------------------
    # 3) final short hill‑climb (stage 3)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        if best_seq is None:
            # fallback to a simple uniform sequence
            n = 2000
            best_seq = _normalize([1.0] * n)
            best_score = _score(best_seq)
            continue

        cand = np.array(best_seq, copy=True)
        n = len(cand)

        # perturb a handful of entries (mass transfer)
        m = random.randint(1, min(6, n))
        idxs = random.sample(range(n), m)
        for i in idxs:
            # pick a partner index j ≠ i
            j = random.randrange(n)
            if i == j:
                continue
            # transfer a fraction of a_i to a_j
            amount = 0.15 * cand[i] * random.random()
            cand[i] = max(0.0, cand[i] - amount)
            cand[j] = cand[j] + amount

        mx = cand.max()
        if mx > 0:
            cand = cand / mx
        cand = cand.tolist()
        sc = _score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"Final hill‑climb improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 500
        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
