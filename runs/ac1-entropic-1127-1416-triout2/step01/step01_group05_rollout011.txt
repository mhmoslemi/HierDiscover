"""
Hybrid optimisation for the max‑convolution bound.

1. Analytic starter – a decreasing power‑law a_i ∝ (i+α)^‑p, normalised so
   that Σ a_i = √(2·n).  This family is known to give values close to the
   theoretical limit π/2 ≈ 1.571.

2. Global tuning – for a single length n (chosen large enough to approach the
   asymptotic limit) we run SciPy's differential‑evolution on the two
   continuous parameters (p, α).  The fitness is the exact objective
   2·n·max(conv(a,a))/Σ a_i², computed with an FFT‑based convolution that
   runs in O(n log n).

3. Local greedy refinement – after the DE step we repeatedly locate the shift
   k at which the convolution attains its maximum and move a small amount of
   mass from the index that contributes most to that peak to the one that
   contributes least.  The move is accepted only if the overall score
   improves.  This quickly eliminates residual peaks left by the coarse DE
   search.

The algorithm respects the 1000 s wall‑clock limit, prints progress and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# SciPy is optional – if unavailable we fall back to a simple random search.
try:
    from scipy.optimize import differential_evolution
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _make_sequence(n: int, p: float, offset: float) -> np.ndarray:
    """
    Return a non‑negative sequence a_i = (i+1+offset)^{-p} normalised to
    Σ a_i = √(2·n).  Values are clipped to the interval [0, 1000] to match
    the checks performed in ``evaluate_sequence``.
    """
    idx = np.arange(n, dtype=float) + 1.0 + offset          # i+1+α
    raw = np.power(idx, -p)                               # (i+1+α)^{-p}
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    seq = raw * scale
    # safeguard – the optimiser may wander close to the bounds
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def _max_convolution(seq: np.ndarray) -> float:
    """
    Compute max(conv(seq, seq)) using an FFT.  The full discrete convolution
    (length 2·n‑1) is required because the maximum can appear at any shift.
    """
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)
    conv = conv[: 2 * n - 1]               # discard the zero‑padding tail
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """
    Exact value of the objective used by ``evaluate_sequence``:
        2·n·max(conv) / (sum(seq)²)
    """
    n = seq.shape[0]
    max_c = _max_convolution(seq)
    s = seq.sum()
    return 2.0 * n * max_c / (s * s)


def _greedy_refine(seq: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Targeted local search: repeatedly locate the shift k where the convolution
    attains its maximum and transfer a tiny amount of mass from the index that
    contributes most to that peak to the index that contributes least.
    The move is kept only if the overall score improves.

    Returns the best sequence found and its score.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_val = _score(seq)

    while time.time() < deadline:
        # full convolution and the index of the current worst peak
        conv = _max_convolution(seq)                     # just the max value
        # we also need the *index* where it occurs – compute the full array
        size = _next_pow2(2 * n - 1)
        fft = np.fft.rfft(seq, n=size)
        full_conv = np.fft.irfft(fft * fft, n=size)[: 2 * n - 1]
        k = int(full_conv.argmax())                      # shift with max value

        # partner[j] = a_{k-j} when the index exists, else 0.
        partner = np.zeros_like(seq)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner[idx] = seq[k - idx]

        # choose donor (large partner) and receiver (small partner)
        donor = int(partner.argmax())
        receiver = int(partner.argmin())

        if donor == receiver or seq[donor] <= 0:
            # fall back to a tiny random reshuffle
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            max_move = 0.01 * seq[i]
            if max_move <= 0:
                continue
            delta = np.random.uniform(0.0, max_move)
            cand = seq.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            max_move = 0.01 * seq[donor]            # move at most 1 %
            if max_move <= 0:
                continue
            delta = np.random.uniform(0.0, max_move)
            cand = seq.copy()
            cand[donor] -= delta
            cand[receiver] += delta

        if np.any(cand < 0):
            continue          # safety – should never happen

        cand_val = _score(cand)
        if cand_val + 1e-12 < best_val:          # strict improvement
            best_val = cand_val
            best_seq = cand
            seq = cand                         # continue from the improved point
            print(f'Greedy step → score={best_val:.6f}  n={n}')
            if best_val < 1.503:
                break
    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine uses a three‑stage hybrid
    optimisation described in the module docstring.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # keep a few seconds for cleanup

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1. Choose a length large enough to be close to the asymptotic bound.
    #    Very large lengths are still cheap because we use FFTs.
    # ------------------------------------------------------------------
    lengths = [8000, 20000, 50000]   # a few candidates, increasing fidelity
    for n in lengths:
        if time.time() > deadline:
            break

        print(f'=== Optimising length n={n} ===')
        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # 2. Global optimiser on the two‑parameter family (p, offset).
        # ------------------------------------------------------------------
        def _de_obj(params):
            p, off = params
            # enforce cheap bounds inside the function – DE already respects them
            if p <= 0 or off < 0:
                return np.inf
            seq = _make_sequence(n, p, off)
            return _score(seq)

        if _SCIPY_AVAILABLE:
            bounds = [(0.35, 0.75), (0.0, 10.0)]   # (p, offset)
            # modest budget – each evaluation is O(n log n) and therefore cheap
            result = differential_evolution(
                _de_obj,
                bounds,
                maxiter=30,
                popsize=12,
                seed=seed,
                polish=False,
                workers=1,
                updating='deferred',
            )
            p_opt, off_opt = result.x
            seq_opt = _make_sequence(n, p_opt, off_opt)
            cur_score = _score(seq_opt)
            print(f'DE result: p={p_opt:.5f} offset={off_opt:.5f} score={cur_score:.6f}')
        else:
            # Simple random sampling fallback (still better than nothing)
            p_opt, off_opt = 0.5, 0.0
            cur_score = float('inf')
            for _ in range(300):
                p_try = np.random.uniform(0.4, 0.6)
                off_try = np.random.uniform(0.0, 5.0)
                seq_try = _make_sequence(n, p_try, off_try)
                val = _score(seq_try)
                if val < cur_score:
                    cur_score = val
                    p_opt, off_opt = p_try, off_try
            seq_opt = _make_sequence(n, p_opt, off_opt)
            print(f'Random sample best: p={p_opt:.5f} offset={off_opt:.5f} score={cur_score:.6f}')

        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq_opt.tolist()

        # ------------------------------------------------------------------
        # 3. Local greedy refinement starting from the best DE candidate.
        # ------------------------------------------------------------------
        refined_seq, refined_score = _greedy_refine(seq_opt, deadline)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.tolist()
            print(f'After greedy refinement: score={refined_score:.6f} (n={n})')

        # early exit if we already beat the aggressive target
        if best_score < 1.503:
            print('Target threshold achieved – stopping early.')
            break

    # ------------------------------------------------------------------
    # 4. Safety net – if nothing was found (should never happen) fall back
    #    to a uniform distribution which yields a score of exactly 2.0.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = [np.sqrt(2.0 * n) / n] * n
        best_score = _score(np.array(best_seq))

    print(f'Final best score: {best_score:.6f}  length={len(best_seq)}')
    return best_seq
