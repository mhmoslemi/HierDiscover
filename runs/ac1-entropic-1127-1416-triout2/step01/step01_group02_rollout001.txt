"""
propose_candidate
-----------------
A two‑stage optimizer for the step‑height problem.

Stage 1 – parameter sweep:
   Generate symmetric power‑law families
       a_i(p,α) = (i+α)^(-p)·(n‑i+α)^(-p) ,   i=0…n‑1 .
   Scan p∈[0,0.9] and a few α values.  Because the evaluation
   function is scale‑invariant, no normalisation is required.
   The best family member (lowest score) is kept.

Stage 2 – max‑convolution flattening:
   Starting from the best candidate, a simple projected sub‑gradient
   descent is applied to the quantity  C = max_k (a*a)_k .
   In each iteration the index k where the convolution attains its
   maximum is identified.  The gradient of C with respect to a is
   g_i = 2·a_{k‑i} (zero when the index is out of range).  Mass is
   transferred from the component with the largest gradient (the
   most “responsible” for the peak) to the component with the smallest
   gradient, preserving the total sum and non‑negativity.
   The step size decays slowly; occasional tiny random jumps are
   performed to escape plateaus.

Both stages respect the 1000 s budget and keep the best sequence
found.  The final list is returned as plain Python ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Central binomial coefficients c_k = (2k choose k) / 4**k."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


def powerlaw_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric power‑law family:
        a_i = (i+α)^(-p)·(n‑i+α)^(-p).
    """
    i = np.arange(n, dtype=np.float64)
    left = (i + alpha) ** (-p)
    right = (n - i + alpha) ** (-p)
    seq = left * right
    # protect against an all‑zero sequence (should not happen for the
    # parameter ranges we explore).
    if np.all(seq == 0):
        seq[0] = 1.0
    return seq


def max_convolution(seq: np.ndarray) -> tuple[float, np.ndarray]:
    """Return (max value, full convolution) of seq with itself."""
    conv = np.convolve(seq, seq)
    return float(conv.max()), conv


def gradient_of_max(seq: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of the convolution value at index k_max.
    For the quadratic form b_k = Σ a_i a_{k-i},
    ∂b_k/∂a_j = 2·a_{k-j}   (if 0≤k‑j<n, else 0).
    """
    n = len(seq)
    grad = np.zeros_like(seq)
    # valid range for i such that k_max‑i lies inside [0,n‑1]
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)
    grad[idx] = 2.0 * seq[k_max - idx]
    return grad


def flatten_max_convolution(
    seq: np.ndarray,
    deadline: float,
    *,
    init_step: float = 0.03,
    decay: float = 0.9998,
    min_step: float = 1e-7,
    patience: int = 300,
) -> np.ndarray:
    """
    Simple sub‑gradient procedure that tries to lower max(a*a).
    The routine stops either when the time budget expires or when
    no improvement has been observed for ``patience`` consecutive
    inner loops.
    """
    best_seq = seq.copy()
    best_max, _ = max_convolution(best_seq)

    cur_seq = seq.copy()
    cur_max = best_max
    step = init_step
    no_improve = 0

    while time.time() < deadline and step > min_step:
        # current convolution and its maximum
        k = np.argmax(np.convolve(cur_seq, cur_seq))
        grad = gradient_of_max(cur_seq, k)

        src = int(np.argmax(grad))          # source of mass (largest gradient)
        dst = int(np.argmin(grad))          # destination (smallest gradient)

        # if source and destination coincide, make a tiny random jump
        if src == dst:
            dst = (src + 1) % len(cur_seq)

        # amount to move – a fraction of the source entry,
        # capped to avoid making the source negative.
        amount = step * cur_seq[src]
        amount = min(amount, cur_seq[src])

        if amount == 0.0:
            # nothing to move – shrink step and continue
            step *= decay
            continue

        candidate = cur_seq.copy()
        candidate[src] -= amount
        candidate[dst] += amount

        cand_max, _ = max_convolution(candidate)

        # Accept if the peak got smaller (or with a vanishing acceptance
        # probability to allow occasional uphill moves).
        if cand_max < cur_max or np.random.rand() < 1e-5:
            cur_seq = candidate
            cur_max = cand_max
            no_improve = 0
            if cand_max < best_max:
                best_seq = candidate.copy()
                best_max = cand_max
        else:
            no_improve += 1

        # decay step size
        step *= decay

        # occasional tiny random perturbation to shake out of plateaus
        if no_improve >= patience:
            # restart from the current best with a small random bump
            cur_seq = best_seq.copy()
            bump = 0.001 * np.random.rand(len(cur_seq))
            cur_seq += bump
            cur_seq = np.maximum(cur_seq, 0.0)
            cur_max, _ = max_convolution(cur_seq)
            no_improve = 0
            step = init_step

    return best_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a step‑function height sequence that (heuristically) minimises
    the given ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total allowed running time (seconds).  The function returns
        as soon as the budget is exhausted.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # small safety margin

    # --------------------------------------------------------------
    # Choose a sequence length.  If the competition provides a
    # reference sequence (``height_sequence_1``) we reuse its length,
    # otherwise we fall back to a moderate default.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 4600

    # --------------------------------------------------------------
    # Stage 1 – quick scan of a low‑dimensional parametric family.
    # The family a_i = (i+α)^(-p)·(n‑i+α)^(-p) is symmetric, non‑negative,
    # and flexible enough to explore shapes between uniform (p≈0)
    # and strongly centre‑weighted (p≈0.8).
    param_alphas = [0.3, 0.5, 0.8, 1.0]
    param_ps = np.linspace(0.00, 0.90, 46)   # step ≈0.02

    best_seq = binomial_sequence(n)          # fallback if nothing better is found
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, binomial score={best_score:.6f}")

    # limit the time spent on the grid search – roughly one quarter
    grid_deadline = min(deadline, time.time() + budget_s * 0.25)

    for alpha in param_alphas:
        for p in param_ps:
            if time.time() >= grid_deadline:
                break
            seq = powerlaw_sequence(n, float(p), float(alpha))
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[grid] α={alpha:.2f}, p={p:.3f} → score={score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – sub‑gradient flattening of the auto‑convolution peak.
    # Start from the best candidate (binomial or one of the power‑law
    # scans) and iteratively move mass from the entries that most
    # contribute to the current peak towards the entries that contribute
    # the least.
    refinement_deadline = deadline - 0.5   # keep a tiny buffer for finalisation
    refined_seq = flatten_max_convolution(best_seq, refinement_deadline)

    # Final evaluation
    final_score = evaluate_sequence(refined_seq.tolist())
    print(f"[final] best score={final_score:.6f}")

    return refined_seq.tolist()
