def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # Determine length n
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000
    best_seq = None
    best_score = np.inf

    while time.time() < deadline:
        # Choose random init or arcsine base
        if np.random.rand() < 0.2 and 'height_sequence_1' in globals():
            a = np.array(height_sequence_1, dtype=float)
        else:
            # random Dirichlet
            a = np.random.rand(n)
        a /= a.sum()
        # apply a few iterations of gradient descent
        it = 0
        beta = 1.0
        lr0 = 0.5
        while time.time() < deadline and it < 5000:
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"Time {time.time():.1f}, new best {best_score:.6f}")
            # compute softmax weights
            maxv = conv.max()
            # temperature schedule: increase beta over iterations (exponential)
            beta = 1.0 + 0.02 * it   # gradually increase
            # compute wnorm
            exp_args = (conv - maxv) * beta
            w = np.exp(exp_args)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            # gradient via correlation
            g = 2.0 * np.correlate(w, a, mode='full')
            # slice appropriate slice
            start = n - 1
            g = g[start:start + n]
            # step size
            step = lr0 / np.sqrt(it + 1)
            a = a - step * g
            # projection onto simplex
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0:
                a = np.random.rand(n)
                a /= a.sum()
            else:
                a /= s
            # occasional jitter
            if it % 1000 == 0 and it > 0:
                a += np.random.randn(n) * 1e-5
                a = np.maximum(a, 0.0)
                a /= a.sum()
            it += 1
        # after each restart loop condition, continue next restart
    # final evaluation using evaluate_sequence
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluate_sequence score: {final_score:.6f}")
    except Exception:
        final_score = best_score
    return best_seq.tolist()

import scipy.optimize as opt
import numpy as np
import time

def lp_optimize(f, M):
    n = len(f)
    # Number of constraints = 2*n - 1
    K = 2*n-1
    # Build A_ub: shape (K, n)
    # For each k, coefficients are f_i for i = max(0, k-(n-1)) ... min(k, n-1)
    # Define rows as shift of f.
    data = []
    rows = []
    cols = []
    for k in range(K):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # i varies from i_start to i_end inclusive.
        # For each i, coefficient for variable b_{k-i}
        # b_index = k - i
        for i in range(i_start, i_end+1):
            b_idx = k - i
            rows.append(k)
            cols.append(b_idx)
            data.append(f[i])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)
    # bounds = [(0, None) for _ in range(n)]
    # objective = -np.ones(n) (since minimize -sum)
    c = -np.ones(n)
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if not res.success:
        raise RuntimeError('LP failed')
    b_opt = res.x
    return b_opt

"""
solve_convolution_opt.py

Search for a non‑negative step‑function that minimizes the evaluation
function

    2 * n * max(conv(a,a)) / (sum(a)**2)

where `conv` is the discrete convolution.

The algorithm consists of two phases:

1. **Linear‑programming phase** (inspired by the paper’s LP iteration).
   Starting from the classic “constant‑convolution” sequence,
   we solve the linear program

       maximise   Σ b_i
       subject to   a ⊛ b ≤ M   (component‑wise)
                     b_i ≥ 0

   where `M = max(conv(a,a))`.  The solution `b` has a larger ℓ₁‑norm
   but does not increase the convolution bound with the current `a`.
   After rescaling `b` back to the standard normalisation
   `Σ a = √(2n)` we move a small step towards it.
   Each iteration strictly decreases `M` (or leaves it unchanged) and
   therefore improves the objective.

2. **Projected sub‑gradient phase**.  When the LP phase finishes (or the
   time budget is almost exhausted) we fine‑tune the best sequence with
   a projected sub‑gradient method on the true max‑convolution objective.

Both phases respect the 1000 s budget, keep all entries non‑negative,
and re‑normalise the vector after every update.  The final sequence is
returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: classic “constant‑convolution” (central binomial) base
# --------------------------------------------------------------
def _constant_conv_base(n: int) -> np.ndarray:
    """First n terms of a_k = C(2k,k) / 4**k.
    Recurrence: a_0 = 1, a_k = a_{k‑1} * (1 - 1/(2k))."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


# --------------------------------------------------------------
# Sub‑gradient step (used for the final polishing)
# --------------------------------------------------------------
def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """One projected sub‑gradient step for minimising max(conv(a,a))."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    kmax = int(np.argmax(conv))

    g = np.zeros_like(a)

    # gradient of the active term (2 * a_{kmax-i})
    if kmax < n:
        g[: kmax + 1] = 2.0 * a[kmax::-1]
    else:
        start = kmax - (n - 1)
        length = 2 * n - 1 - kmax
        g[start:] = 2.0 * a[:length][::-1]

    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)

    tot = a_new.sum()
    if tot <= 0.0:
        a_new = np.random.rand(n)
        a_new /= a_new.sum()
        return a_new
    return a_new / tot

    step0 = 0.4
    while time.time() < deadline and sub_iter < 500_000:
        step = step0 / np.sqrt(sub_iter + 1.0)

        a = _subgradient_step(a, step)

        # occasional tiny random jitter to avoid stagnation
        if sub_iter % 3000 == 0 and sub_iter > 0:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()
            a *= target_sum / a.sum()   # keep exact normalisation

        # evaluate current max‑convolution
        cur_M = np.convolve(a, a).max()
        if cur_M < best_M:
            best_M = cur_M
            best_a = a.copy()
            print(f"Sub‑grad iter {sub_iter:6d}: improved M {best_M:.6f}")

        sub_iter += 1

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence (optional)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"Final evaluate_sequence score: {final_score:.6f}")
    except Exception:
        # evaluate_sequence not available – ignore; we already have best_M
        final_score = 2.0 * n * best_M / (target_sum ** 2)

    print(
        f"Search finished – length {n}, best max‑conv {best_M:.6f}, "
        f"objective ≈ {final_score:.6f}"
    )
    return best_a.tolist()
