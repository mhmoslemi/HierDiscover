    stages = [(4, 2000), (8, 2000), (16, 2000), (32, 2000), (64, 2000)]
    for p, stage_iters in stages:

                break
            t += 1

            c = convolution_fft(a, a)
            # compute smooth max S
            c_pow_p = c ** p
            S = np.sum(c_pow_p)

                continue
            factor = S ** (1.0/p - 1.0)
            w = c ** (p-1)  # shape same
            # compute B = convolution of w and a_rev (size n)
            a_rev = a[::-1]

            start = len(a) - 1

            # Adam update
            m = beta1 * m + (1 - beta1) * grad

    start = time.time()

    best_seq = None

        print(f"\n=== Optimizing length {n} ===")

    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)

    conv = np.fft.irfft(C, n=n_fft)[:m]

"""
Annealed Adam optimizer on the simplex for the max‑convolution objective.

The evaluation function (up to a scale factor) is

    E(a) = 2·n·max(a * a) / (sum(a)^2) ,

which is scale‑invariant.  By normalising ∑a_i = 1 the problem reduces to
minimising max(conv(a,a)).  The objective is convex; we minimise a smooth
approximation
    f_p(a) = (∑_k conv_k(a,a)^p)^{1/p}
and gradually increase p (deterministic annealing).  The gradient of f_p is
available analytically and we perform an Adam update followed by a Euclidean
projection onto the probability simplex.  Several sequence lengths and
initialisations are tried within the time budget; the best sequence found is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    Implements the algorithm of Duchi et al. (2008).  The function works for
    arbitrary 1‑d float arrays.
    """
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]                     # sorted in descending order
    cssv = np.cumsum(u) - 1.0                # cumulative sum minus 1
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                         # should not happen due to the test above
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors via the FFT.
    Returns a real‑valued ndarray of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Scale‑invariant score identical to `evaluate_sequence` for sum(a)=1."""
    n = a.size
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Continuous arcsine density discretised on {0,…,n‑1}.
    a_i ∝ 1/√((i+0.5)*(n-0.5-i)), with a tiny epsilon to avoid exact zeros.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Core optimiser
# ----------------------------------------------------------------------


def annealed_adam_optimizer(initial_a: np.ndarray,
                            n: int,
                            deadline: float,
                            stage_schedule: list[tuple[int, int]]) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a sequence a ∈ Δₙ (probability simplex).

    Parameters
    ----------
    initial_a : ndarray
        Starting point (not necessarily normalised).
    n : int
        Length of the sequence (used only for convenience).
    deadline : float
        Absolute Unix time after which the routine must stop.
    stage_schedule : list of (p, max_iter)
        Annealing schedule – for each stage we minimise the smooth‑max
        with exponent p for at most `max_iter` iterations.

    Returns
    -------
    best_a : ndarray
        Best sequence seen (still on the simplex).
    best_score : float
        Corresponding exact score (max‑convolution based).
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    it_global = 0

    for p, max_iter in stage_schedule:
        # learning‑rate heuristic that shrinks with growing p
        base_lr = 0.05 / np.sqrt(p)

        for it in range(max_iter):
            if time.time() >= deadline:
                return best_a, best_score

            it_global += 1
            # ----- smooth‑max and its gradient (p‑norm) -----
            conv = convolution_fft(a, a)                     # length 2n‑1
            # avoid negative zeros
            conv = np.maximum(conv, 0.0)

            # p‑norm based smooth max
            conv_p = conv ** p
            S = np.sum(conv_p)
            if S <= 0.0:
                # degenerate – fall back to exact max sub‑gradient
                k_max = int(np.argmax(conv))
                grad = subgradient_of_max_at_index(a, k_max)
            else:
                factor = S ** (1.0 / p - 1.0)                # = S^{1/p-1}
                w = conv ** (p - 1)                         # weights for the gradient
                # B_i = Σ_j a_j * w_{i+j}  (linear convolution of w with reverse(a))
                a_rev = a[::-1]
                w_conv = convolution_fft(w, a_rev)          # length 3n‑2
                start = n - 1
                B = w_conv[start:start + n]                 # exactly n entries
                grad = 2.0 * factor * B

            # ----- Adam update (with simplex projection) -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** it_global)
            v_hat = v / (1.0 - beta2 ** it_global)

            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            a = project_to_simplex(a)

            # ----- keep track of the best exact score -----
            if (it % 200) == 0:
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()

    return best_a, best_score


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal entry of conv(a,a) w.r.t. a.
    Exact derivative: ∂c_k/∂a_i = 2·a_{k‑i} for all valid indices.
    """
    n = a.size
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function (a list of floats) with a low
    `evaluate_sequence` value.  The routine respects the time budget, tries a
    handful of lengths and initialisations, runs an annealed Adam optimiser,
    and finally returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously found construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Optimisation schedule (p, max_iters) – gradually sharpen the max‑norm
    # ------------------------------------------------------------------
    schedule = [(4, 2500), (8, 2500), (12, 2500), (16, 2500),
                (24, 2500), (32, 2500)]

    # Lengths to explore – powers of two give fast FFTs.
    lengths = [512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
