"""
Propose a low‑value step‑function via smoothed max‑convolution descent.

The evaluation metric is
    E(a) = 2·n·max(a*a) / (sum(a)^2)                     (scale‑invariant)

We keep the vector on the probability simplex (sum = 1, a_i ≥ 0) and
minimise the scale‑invariant quantity
    f(a) = 2·n·max(a*a)      (since sum(a)=1).

Because the max is non‑smooth we replace it by a smooth log‑sum‑exp
approximation

    g_α(a) = (1/α)·log Σ_i exp(α·c_i)   ,   c = a*a ,

which converges to max(c) as α→∞.  The gradient of g_α can be written
in closed form using a single FFT‑based convolution; together with the
simple quotient rule we obtain an explicit gradient of f(a) that is
cheap to evaluate.

The optimisation uses the Adam variant of stochastic gradient descent.
At each iteration:

1. Compute the full self‑convolution `c = a*a` (FFT, O(n log n)).
2. Build the soft‑max weights  w_i ∝ exp(α·(c_i‑max(c))) .
3. Compute the vector  
       t_k = Σ_i w_i·a_{i‑k}   (a shifted‑copy of a, realised by a
       convolution of w with the reversed a).
4. The gradient of the original objective is  
       ∇f(a) = 4·n·(t – g·1)   (here g = max_α(a) and 1 is the all‑ones vector).
5. Apply one Adam step, then Euclidean‑project back onto the simplex.
6. Occasionally inject tiny random noise to escape flat regions.

The smoothing parameter α is increased linearly from a small value
(easy‑to‑optimise landscape) to a large value (close to the true max).
The routine runs several sequence lengths (powers of two up to 8192)
and keeps the best candidate seen.  All heavy work is performed with
NumPy/FFT and the inexpensive scale‑invariant score
    score = 2·n·max(a*a)               (sum(a)=1)

so the true `evaluate_sequence` call is needed only once at the very
end.  The total running time respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # All entries non‑positive → uniform vector.
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]               # Sorted descending.
    cssv = np.cumsum(u) - 1.0          # Cumulative sum minus 1.
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via real FFT (full mode)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Scale‑invariant score for a (assumes sum(a)≈1):
        2·n·max(a*a)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = np.max(conv)
    # sum is forced to 1 by the optimiser, but we keep the exact formula
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Initialise a vector that mimics the continuous arcsine density:
        a_i ∝ 1 / sqrt((i+0.5)*(n‑0.5‑i))
    This concentrates mass at the ends and makes the self‑convolution
    relatively flat.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25  # small offset to avoid exact zeros at the extremes.
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)

        # ---- time check -------------------------------------------------
        if time.time() >= time_deadline:
            break

        # ---- smoothing schedule -----------------------------------------
        if t <= alpha_ramp_iters:
            alpha = alpha_start + (alpha_end - alpha_start) * (t - 1) / (alpha_ramp_iters - 1)
        else:
            alpha = alpha_end

        # ---- self‑convolution and soft‑max weights ----------------------
        conv = convolution_fft(a, a)                 # length 2n‑1
        max_val = np.max(conv)

        # stable soft‑max
        shifted = conv - max_val                     # ≤ 0
        exp_shift = np.exp(alpha * shifted)         # max term = 1
        sum_exp = np.sum(exp_shift)
        w = exp_shift / sum_exp                      # shape (2n‑1,)

        # smooth max value (g_α)
        g = max_val + (1.0 / alpha) * np.log(sum_exp)

        # ---- gradient of the original objective -------------------------
        # term t_k = Σ_i w_i * a_{i‑k}
        a_rev = a[::-1]
        conv_w_arev = convolution_fft(w, a_rev)    # length 3n‑2
        # extract the slice that corresponds to k = 0…n‑1
        t_vec = conv_w_arev[n - 1 : n - 1 + n]      # shape (n,)

        grad = 4.0 * n * (t_vec - g)                # ∇f(a)

        # ---- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # ---- occasional random jitter ------------------------------------
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        # ---- projection onto simplex -------------------------------------
        a_new = project_to_simplex(a_new)

        # ---- keep best (cheap score) ------------------------------------
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"Iter {t:,d} (n={n}) – new best score: {best_score:.6f}")

        a = a_new

        # ---- occasional progress report ---------------------------------
        if t % 5000 == 0:
            print(f"Iter {t:,d} (n={n}) – current score: {cur_score:.6f} – best: {best_score:.6f}")

    return best_a, best_score

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low
    `evaluate_sequence` value.  The routine runs several
    Adam‑optimisation passes on different lengths (powers of two)
    and returns the best sequence discovered before the time budget
    expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small safety buffer
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Fallback: use the previously known construction if it exists.
    # ------------------------------------------------------------------
    try:
        height_sequence_1  # may raise NameError
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives a better theoretical bound,
    # but each iteration costs O(n log n).  We try a few powers of two.
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n=== Optimising length n = {n} ===")

        # Two distinct initialisations – arcsine (good start) and uniform.
        inits = [
            arcsine_initial(n),
            np.full(n, 1.0 / n)
        ]

        for idx, init_vec in enumerate(inits, start=1):
            if time.time() >= deadline - 5.0:
                break

            print(f"  Run {idx}/{len(inits)} ({'arcsine' if idx == 1 else 'uniform'})")
            seq, val = adam_optimize(
                init_a=init_vec,
                time_deadline=deadline,
                max_iter=200_000,
                lr=0.07,
                alpha_start=5.0,
                alpha_end=200.0,
                alpha_ramp_iters=20_000,
                noise_every=400,
                noise_scale=5e-5,
            )
            print(f"    Run best score (scale‑invariant) = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** Global best updated: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate with the official function (slow but
    # performed only once).
    # ------------------------------------------------------------------
    if best_seq is None:
        # As a last resort, a modest uniform sequence.
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
