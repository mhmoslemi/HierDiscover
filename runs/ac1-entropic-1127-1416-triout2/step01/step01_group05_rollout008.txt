"""
Proposes a step‑function (non‑negative sequence) that minimises the
`evaluate_sequence` metric.

Algorithm Overview
------------------
1. **Initialisation**
   * If a global variable ``height_sequence_1`` exists, use it as the
     starting point (after normalising its sum to √(2·n)).
   * Otherwise fall back to a uniform sequence.

2. **Randomised parameter sweep**
   * Sample sequences from two expressive families:
       – Power‑law: a_i ∝ (i+1+α)^{-p}
       – Symmetric product: a_i ∝ ((i+½+α)·(n‑i‑½+β))^{-p}
   * Lengths between 2000 and 10 000 and parameters (p, α, β) are drawn
     uniformly from broad intervals.
   * Each candidate is normalised so that Σa_i = √(2·n) and its maximal
     self‑convolution (the quantity returned by `evaluate_sequence`) is
     estimated with a fast FFT‑based convolution.
   * The best candidate found so far is kept.

3. **Targeted simulated‑annealing refinement**
   * Starting from the best candidate, repeatedly:
        – Compute the full convolution via FFT, locate the lag `ℓ`
          giving the maximal autocorrelation `M`.
        – Within that lag find the pair (i,i+ℓ) with the largest product
          a_i·a_{i+ℓ}.  Choose the larger element of the pair as the
          source index.
        – Move a small fraction of its mass to a random destination
          index, renormalise the sequence, and recompute `M`.
        – Accept the move if `M` decreases; otherwise accept with a
          probability `exp(-(ΔM)/T)` where the temperature `T` cools
          geometrically.
   * The fraction of mass moved and the temperature both decay,
     yielding fine‑grained local optimisation.
   * The current best sequence (lowest observed `M`) is stored.

4. **Final verification**
   * The best sequence obtained is evaluated once with the official
     `evaluate_sequence` (which also enforces the caps on the values) and
     returned.

The whole search runs for at most ``budget_s`` seconds (default 1000 s)
and prints progress messages.  All helper functions are top‑level and
no closures or lambdas are used.
"""

import time

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fast_max_conv(a: np.ndarray) -> float:
    """
    Return max_{k} (a * a)[k] using an FFT based convolution.
    ``a`` is assumed to be normalised already.
    """
    n = a.shape[0]
    m = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return float(conv.max())


def _normalise_to_target(seq: np.ndarray) -> np.ndarray:
    """
    Scale ``seq`` so that its sum equals √(2·len(seq)).
    The input may contain zeros – they stay zero.
    """
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    s = seq.sum()
    if s == 0.0:
        return seq
    return seq * (target / s)


def _make_power_sequence(n: int, p: float, off: float) -> np.ndarray:
    """Raw power‑law shape a_i = (i+1+off)^{-p}."""
    idx = np.arange(n, dtype=float) + 1.0
    return 1.0 / np.power(idx + off, p)


def _make_symmetric_product_sequence(
    n: int, p: float, off1: float, off2: float
) -> np.ndarray:
    """
    Symmetric shape a_i = ((i+½+off1)*(n‑i‑½+off2))^{-p}.
    The half‑offset avoids a zero denominator at the ends.
    """
    i = np.arange(n, dtype=float) + 0.5
    return 1.0 / np.power((i + off1) * (n - i + off2), p)

            idx_range = np.arange(l, n, dtype=int)
            products = a[idx_range] * a[idx_range - l]

        # index of the biggest product
        i_local = idx_range[np.argmax(products)]

        # decide which endpoint of the pair to use as the source of mass
        if lag >= 0:
            src = i_local if a[i_local] >= a[i_local + lag] else i_local + lag
        else:
            src = i_local if a[i_local] >= a[i_local - lag] else i_local - lag

        # choose a destination distinct from the source
        dest = np.random.randint(n - 1)
        if dest >= src:
            dest += 1

        # ----- propose a small mass transfer -----
        delta = move_frac * a[src]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[dest] += delta
        cand = np.maximum(cand, 0.0)
        cand = _normalise_to_target(cand)  # keep the prescribed L1 norm

        # ----- evaluate the candidate (fast FFT) -----
        fc = np.fft.rfft(cand, n=m_fft)
        conv_cand = np.fft.irfft(fc * fc, n=m_fft)[: 2 * n - 1]
        M_cand = float(conv_cand.max())

        # ----- acceptance test (simulated annealing) -----
        accept = False
        if M_cand < M:
            accept = True
        else:
            prob = math.exp(-(M_cand - M) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            if M_cand < best_score:
                best_score = M_cand
                best_a = cand.copy()
                print(f"[refine] iter {iteration:6d} – new best score={best_score:.6f}")

        # ----- schedule updates -----
        move_frac = max(min_frac, move_frac * frac_decay)
        temperature = max(1e-12, temperature * temp_decay)

    # --------------------------------------------------------------
    # 4. Final verification with the official evaluator.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] score={final_score:.6f} (length={len(final_seq)})")
    return final_seq
