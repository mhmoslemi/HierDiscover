    iteration = 0

    while time.time() < deadline:
        conv = convolve_full_fft(a)
        M = conv.max()
        k = int(conv.argmax())
        # construct rev vector
        idx = np.arange(n)

        a = np.maximum(a, 0.0)

        # renormalize sum to 1
        s = a.sum()
        if s == 0:
            a = binomial_sequence(n)
            a = a / a.sum()
        else:
            a = a / s

        iteration += 1

        # evaluate
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[iter {iteration}] new best {best_score:.6f} time={time.time()-(deadline-budget_s):.2f}s")
        # adjust step size

assistantfinal**Solution Overview**

The evaluation function  

\[
E(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is scale‑invariant – multiplying the whole sequence by a constant does not
change `E`.  
Consequently we can **keep the sequence normalised to unit sum** during the
search and only have to minimise the quantity  

\[
M(a)=\max_k (a*a)_k .
\]

For a given normalised vector `a` the gradient of `M` (taken at the index
`k*` where the maximum is attained) is  

\[
\frac{\partial M}{\partial a_j}=2\,a_{k* - j},
\]

so moving a little mass from an index whose contribution
`a_j·a_{k*-j}` is large to an index with a small contribution reduces `M`.
The algorithm therefore repeatedly

1. computes the convolution (by FFT – `O(n log n)`);
2. finds the active index `k*` and the *pairwise contributions*
   `c_j = a_j * a_{k*-j}`;
3. transfers a tiny fraction of the mass from the index with the largest
   contribution to the index with the smallest contribution;
4. projects back onto the probability simplex (clip to `≥0` and renormalise).

A tiny amount of random pair‑wise transfers is interleaved to avoid getting
stuck in plateaus, and the step size is slowly decayed.  The current best
sequence (the one with the smallest evaluation value) is always remembered.

The whole procedure runs until the time budget (≈ 1000 s) is exhausted and
returns the best sequence found.

propose_candidate

    2 * n * max(a * a) / (sum(a) ** 2)

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """
    Central binomial coefficients  c_k = (2k choose k) / 4**k  for k=0..length‑1.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq

def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << ((size - 1).bit_length())
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:size]
    return conv

    # ----------------------------------------------------------------
    # parameters for the guided mass transfer
    step_frac = 0.02          # start moving 2 % of the source mass
    decay = 0.9995            # slowly shrink the step size
    iteration = 0

    # ----------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- current convolution, maximal element and its index ----------
        conv = full_convolution_fft(a)
        M = conv.max()
        k_star = int(conv.argmax())               # index achieving the max

        # ----- contributions to the maximal coefficient --------------------
        # rev_j = a_{k* - j}  (zero when the index is out of range)
        idx = np.arange(n)
        rev_idx = k_star - idx
        rev = np.where((rev_idx >= 0) & (rev_idx < n), a[rev_idx], 0.0)

        contrib = a * rev                          # a_j * a_{k* - j}

        # ----- pick source (largest contribution) and target (smallest) ----
        src = int(np.argmax(contrib))
        dst = int(np.argmin(contrib))

        # avoid a no‑op when source and destination coincide
        if src == dst:
            # fall back to a tiny random move
            src, dst = np.random.choice(n, 2, replace=False)

        # ----- propose a small transfer of mass ---------------------------
        amount = step_frac * a[src]                # ≤ step_frac of the source
        cand = a.copy()
        cand[src] -= amount
        cand[dst] += amount

        # keep feasibility
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand /= cand_sum        # re‑normalise to unit sum
        else:
            cand = a.copy()         # degenerate case – revert

        # ----- evaluate the candidate ------------------------------------
        cand_score = evaluate_sequence(cand.tolist())

        # ----- accept if it improves (or with tiny probability to escape) -
        accept = False
        if cand_score < best_score:
            accept = True
        elif cand_score < evaluate_sequence(a.tolist()):   # improvement over current
            accept = True
        elif np.random.rand() < 1e-5:  # rare uphill move
            accept = True

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                elapsed = time.time() - (deadline - budget_s)
                print(f"[iter {iteration:5d}] new best {best_score:.6f}  "
                      f"time={elapsed:.2f}s")

        # ----- step size adaptation ---------------------------------------
        step_frac *= decay
        if step_frac < 1e-6:
            step_frac = 1e-6

        # ----- occasional random jitter to avoid stagnation ---------------
        if iteration % 3000 == 0:
            jitter = (np.random.rand(n) - 0.5) * step_frac * 0.5
            a += jitter
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum

        # ----- occasional restart from the best found sequence -----------
        if np.random.rand() < 0.001:
            a = best_seq.copy()
            step_frac = 0.02

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f}")
    # Convert to plain Python list as required by the external framework
    return best_seq.tolist()
