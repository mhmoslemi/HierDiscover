def generate_shape(n, p, eps):
    i = np.arange(n, dtype=float)
    # Use (i+eps)*(n-1-i+eps)
    vals = ( (i + eps) * (n - 1 - i + eps) ) ** (-p)
    # Avoid infinite if (i+eps)*(...)=0 => not possible because eps>0.
    # Normalize to sum=1.
    total = vals.sum()
    if total == 0:
        return np.full(n, 1.0/n)
    return vals / total

"""
A hybrid search for the optimal non‑negative step‑function.

The algorithm mixes three ideas:

1. **Parameterized shape search** –  a simple two‑parameter family
   a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ (p>0, ε>0) which contains the classic
   arcsine density (p=0.5, ε≈0.25).  A fast random scan over (p,ε)
   quickly finds very good candidates for each length n.

2. **Projected sub‑gradient refinement** –  Starting from the best
   shape discovered, a projected sub‑gradient optimiser (the same
   method that was used previously) is run for many iterations.
   This reduces the maximal self‑convolution while keeping the
   vector on the probability simplex.

3. **Global coordination** –  Several sequence lengths (256‑2048)
   are examined; the best sequence overall is kept and, if time
   permits, refined once more at the end.

The whole routine stops after the allotted budget (≈1000 s) and
returns the best sequence found as a plain ``list`` of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via real FFT (fast, zero‑padding to next power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, arg‑max index) of a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) with respect to a.

    For each i, ∂c_k/∂a_i = a_{k‑i} if 0 ≤ k‑i < n,
    and doubled diagonal term when k = 2·i.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    # double the diagonal term when it exists
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Scale‑invariant objective used by the official evaluator:
        2·n·max(a*a) / (sum(a)²)
    (the input is assumed to be a probability vector, i.e. sum≈1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine‑shaped start:  a_i ∝ 1/√((i+½)*(n‑½‑i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Produce a symmetric density a_i ∝ ((i+eps)*(n‑1‑i+eps))⁻ᵖ,
    then normalise to unit L¹ norm.
    """
    i = np.arange(n, dtype=float)
    # avoid division by zero – eps must be > 0
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        # degenerate case – fall back to uniform
        return np.full(n, 1.0 / n)
    return base / total


def improve_sequence(initial_a: np.ndarray,
                    time_deadline: float,
                    max_iter: int = 200_000,
                    eta0: float = 0.5,
                    noise_every: int = 500,
                    noise_scale: float = 1e-4) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution objective.
    Returns the best vector seen and its internal score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1
        # current worst convolution entry
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        a_new = a - eta * grad
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new

        if it % 5000 == 0:
            print(f"    iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low
    ``evaluate_sequence`` value.  The routine respects the time budget,
    tries several lengths, performs a random scan over a simple
    two‑parameter family, refines the best candidates with a
    projected sub‑gradient optimiser and finally returns the best vector
    found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 1.  Fallback – use any previously discovered construction if present.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be defined by the surrounding test harness
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 2.  Exploration over a few lengths.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ---- 2a. Random scan of the (p, eps) family --------------------
        local_best_a = None
        local_best_sc = float("inf")

        # a few deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for starter in starters:
            sc = compute_score(starter)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = starter.copy()

        # random samples (budget‑aware)
        max_samples = 500
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break
            # p ∈ (0.2, 0.8),    eps ∈ (1e‑3, max(2.0, 0.02·n))
            p = rng.uniform(0.2, 0.8)
            eps = rng.uniform(0.001, max(2.0, 0.02 * n))
            cand = generate_shape(n, p, eps)
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand

        # -----------------------------------------------------------------
        # 2b. Refine the locally best candidate with sub‑gradient descent.
        # -----------------------------------------------------------------
        if local_best_a is not None and time.time() < deadline - 5.0:
            # allocate a fraction of the remaining time to this length
            remaining = deadline - time.time()
            # aim for up to 150 k iterations but respect the time left
            max_iter = min(150_000, int(remaining * 30))
            print(f"  Refining (max_iter={max_iter}) …")
            refined_a, _ = improve_sequence(
                initial_a=local_best_a,
                time_deadline=deadline,
                max_iter=max_iter,
                eta0=0.7,
                noise_every=800,
                noise_scale=5e-5,
            )
            # evaluate with the official (slow) evaluator once
            val = evaluate_sequence(refined_a.tolist())
            print(f"  → refined score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
        else:
            # If we have no time for refinement, just test the raw candidate.
            if local_best_a is not None:
                val = evaluate_sequence(local_best_a.tolist())
                print(f"  Raw candidate score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = local_best_a.tolist()
                    print(f"  *** New global best (raw) ***")

    # ------------------------------------------------------------------
    # 3.  Final polishing on the current global best (time permitting)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = min(300_000, int(remaining * 30))
        print(f"\nFinal polishing (max_iter={max_iter}) …")
        a_refined, _ = improve_sequence(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter,
            eta0=0.6,
            noise_every=600,
            noise_scale=3e-5,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print(f"*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Fallback guard – ensure we always return *something*
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
