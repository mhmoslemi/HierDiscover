"""
Optimise a non‑negative step function by projected mirror descent on the
simplex.  We parameterise the coefficients with a softmax (so they are
always ≥0 and sum to 1) and minimise a smoothed version of the
objective `max_k (a*a)[k]`.  The smoothing uses a log‑sum‑exp with a
temperature β that is slowly increased.  Gradients are computed via
convolution (FFT for speed) and the parameters are updated with Adam.
The best sequence encountered (according to the true `evaluate_sequence`
function) is returned.
"""
import time
import numpy as np

def _constant_conv_base(n: int) -> np.ndarray:
    """First n terms of a_k = binom(2k,k)/4^k (≈1/√(πk))."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (returns length len(x)+len(y)-1)."""
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()      # next power‑of‑two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    res = np.fft.irfft(Z, N)[:n]
    # Numerical noise can produce tiny negative values
    return np.maximum(res, 0.0)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a coefficient vector that minimises `evaluate_sequence`.
    The algorithm runs for at most `budget_s` seconds and returns the
    best sequence encountered.  It works on the simplex by means of a
    soft‑max parametrisation and Adam optimisation of a
    log‑sum‑exp smoothed max‑convolution objective.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose the length of the step function.
    # We aim for a fairly large n (≈5000) – larger n gives the
    # optimisation more freedom without breaking the time budget.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, tuple)):
        base_len = len(height_sequence_1)
        n = max(base_len, 5000)
    else:
        n = 5000

    # --------------------------------------------------------------
    # Initialise the probability vector a (sum‑to‑one, non‑negative).
    # We start from the classic constant‑convolution sequence and,
    # if it is shorter than n, pad with zeros.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, tuple)):
        init = np.array(height_sequence_1, dtype=np.float64)
        if init.shape[0] < n:
            init = np.pad(init, (0, n - init.shape[0]), mode='constant')
        a0 = init[:n]
    else:
        a0 = _constant_conv_base(n)

    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0:
        a0 = np.random.rand(n)
    a0 = a0 / a0.sum()

    # Soft‑max parametrisation: a = softmax(u)
    eps = 1e-30
    u = np.log(a0 + eps)

    # Adam optimiser state
    lr = 0.2
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    t = 0

    # Temperature for the log‑sum‑exp smoothing of the max
    beta_s = 5.0
    beta_max = 150.0
    beta_factor = 1.001

    # Keep the best sequence seen so far
    best_seq = a0.copy()
    best_score = float('inf')
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass
    best_iter = 0

    print(f"Optimisation started: n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    while time.time() < deadline:
        t += 1

        # ----- soft‑max to obtain a -----
        u_shift = u - np.max(u)                 # for numerical stability
        exp_u = np.exp(u_shift)

  non‑negativity and Σa = 1.
* The objective `max_k (a*a)[k]` is smoothed with a log‑sum‑exp
  (temperature β).  β is increased gradually so the optimisation first
  sees a smooth landscape and then focuses on the true maximum.
* Gradients of the smoothed objective are obtained analytically:
      ∂/∂a_i  (log‑sum‑exp) = 2· Σ_k p_k a_{k-i}
  where p_k ∝ exp(β·conv_k).  The required convolution is evaluated
  efficiently with a real‑FFT.

def _constant_conv_base(n: int) -> np.ndarray:
    """First n terms of a_k = C(2k,k) / 4**k  (≈ 1/√(πk))."""

        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
