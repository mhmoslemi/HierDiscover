"""
Propose a low‑score step‑function using a simple 2‑exponential (valley)
family.  The family

    a_i = C·(r**i + r**(n‑1‑i)),   r = 1‑λ ,

has two mirrored exponential tails.  For a fixed length n the sum of the
coefficients is S = sqrt(2·n).  The scaling constant is

    C = S·λ / (2·(1‑r**n))

which guarantees the required normalisation.  By sweeping λ in a
log‑uniform grid we can locate the λ that minimises the evaluator.
A final short projected‑gradient polish (using the sub‑gradient of the
max‑convolution) is performed on the best candidate.

The whole procedure fits comfortably within the 1000 s budget.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = s}
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence with a small evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2          # safety margin

    # ------------------------------------------------------------------- #
    # problem size (3000 gives a good trade‑off between flexibility and speed)
    n = 3000
    target_sum = np.sqrt(2.0 * n)                 # Σ a_i = √(2n)

    # thin wrapper around the provided evaluator
    def eval_fast(seq: np.ndarray) -> float:
        return evaluate_sequence(seq.tolist())

    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------- #
    # 1) Baselines (uniform, pure geometric, power‑law) – cheap sanity checks
    uniform = np.full(n, target_sum / n)
    val = eval_fast(uniform)
    if val < best_val:
        best_val, best_seq = val, uniform.copy()
        print(f"[{time.time() - start:6.1f}s] uniform → {best_val:.6f}")

    # pure geometric a_i = C·r**i
    for lam in np.logspace(-6, -1.5, 150):
        r = 1.0 - lam
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom
        a = C * (r ** np.arange(n))
        a = np.clip(a, 0.0, 1000.0)
        a = a * (target_sum / a.sum())           # re‑normalise for safety
        val = eval_fast(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[{time.time() - start:6.1f}s] geo  λ={lam:.2e} → {best_val:.6f}")

    # power‑law a_i ∝ 1/(i+1)^α
    idx = np.arange(1, n + 1, dtype=float)
    for alpha in np.linspace(0.3, 2.0, 30):
        a = 1.0 / np.power(idx, alpha)
        a = a * (target_sum / a.sum())
        a = np.clip(a, 0.0, 1000.0)
        val = eval_fast(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[{time.time() - start:6.1f}s] pow α={alpha:.2f} → {best_val:.6f}")

    # ------------------------------------------------------------------- #
    # 2) The 2‑exponential (valley) family – main driver of improvement
    #    a_i = C·(r**i + r**(n‑1‑i)),  r = 1‑λ
    #    Scan λ on a fine log‑grid; keep the best.
    lam_grid = np.logspace(-5, -2, 2000)          # 1e‑5 … 1e‑2
    idx = np.arange(n)

    for lam in lam_grid:
        r = 1.0 - lam
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:

        C = target_sum * lam / (2.0 * denom)
        a = C * (r ** idx + r ** (n - 1 - idx))
        a = np.clip(a, 0.0, 1000.0)
        a = a * (target_sum / a.sum())
        val = eval_fast(a)
        if val < best_val:

            print(f"[{time.time() - start:6.1f}s] valley λ={lam:.2e} → {best_val:.6f}")

    # ------------------------------------------------------------------- #
    # 3) Short projected‑gradient polishing on the current best.
    #    Uses the sub‑gradient at the (few) largest convolution entries.
    curr = best_seq.copy()
    best = curr.copy()
    best_score = best_val

    # parameters for the gradient phase
    fft_len = 1 << ((2 * n - 1).bit_length())
    step = 0.5 * target_sum / n          # initial absolute step size

        top_k = np.argpartition(conv, -5)[-5:]   # 5 largest entries
        # sub‑gradient of max‑functional (aggregate over the active set)
        grad = np.zeros_like(curr)
        for k in top_k:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = k - idx_i
            grad[idx_i] += 2.0 * curr[idx_j]

        norm = np.linalg.norm(grad)
        if norm == 0:

        tentative = curr - step * grad / norm
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # evaluate
        sum_curr = curr.sum()
        max_conv = conv.max()
        curr_score = max_conv                     # because denom = Σa² = 2n
        if curr_score < best_score:
            best_score = curr_score
            best = curr.copy()
            print(f"[{time.time() - start:6.1f}s] iter {iteration:5d} → new best {best_score:.6f}")
            step = max(step * 0.9, min_step)     # tighten step when improving

        iteration += 1
        if iteration % decay_every == 0:
            step = max(step * decay, min_step)

    final_val = eval_fast(best)
    print(f"Search finished – best evaluation {final_val:.6f} (iterations {iteration})")
    return best.tolist()
