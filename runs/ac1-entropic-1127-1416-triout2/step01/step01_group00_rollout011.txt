def exp_gradient_descent(initial_a, time_deadline, max_iter=100_000, eta0=0.1, alpha0=1.0, alpha_factor=1.01, noise_every=1000, noise_scale=1e-5):

   a = initial_a.copy()
   a = a / np.sum(a)  # ensure sum=1
   best_a = a.copy()
   best_score = compute_score(a)

   n = len(a)
   alpha = alpha0

   for t in range(1, max_iter+1):
       if time.time() >= time_deadline:
           break
       # compute conv and score
       conv = convolution_fft(a, a)
       max_val = np.max(conv)
       # compute smoothing
       # softmax of alpha*conv
       max_alpha = np.max(alpha * conv)
       exp_vals = np.exp(alpha * conv - max_alpha)  # subtract max for stability
       p = exp_vals / np.sum(exp_vals)  # softmax weights sum to 1

       # compute gradient: grad = grad1 + grad2
       # grad1 via correlation
       a_rev = a[::-1]
       # compute correlation as described
       # p length = 2n-1
       # Compute convolution of a_rev and p via FFT
       grad1_full = convolution_fft(a_rev, p)   # length n + (2n-1) -1 = 3n-2
       grad1 = grad1_full[n-1:n-1+n]   # extract indices n-1..n-1+n-1, shape n

       # compute grad2: window sum of p length n
       # sum_window = np.convolve(p, np.ones(n), mode='valid')
       # or using np.cumsum for speed:
       cumsum = np.concatenate(([0.0], np.cumsum(p)))
       sum_window = cumsum[n:] - cumsum[:-n]  # shape n
       grad2 = a * sum_window

       grad = grad1 + grad2

       # optional noise
       if noise_every and (t % noise_every == 0):
           grad += np.random.randn(n) * noise_scale

       # step size
       eta = eta0 / np.sqrt(t)

       # exponential update
       a = a * np.exp(-eta * grad)
       # renormalize to sum to 1 (projection to simplex ensures positivity and sum)
       a = a / np.sum(a)

       # compute objective (actual score)
       cur_score = compute_score(a)
       if cur_score < best_score:
           best_score = cur_score
           best_a = a.copy()

   return best_a, best_score

def local_search(a, time_deadline, max_trials=100_000, step_scale=0.01):

    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    trial=0
    while time.time() < time_deadline - 1.0 and trial < max_trials:
        trial += 1
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)
        # choose epsilon proportional to a[i] maybe
        eps = step_scale * np.random.rand() * best_a[i]
        if eps <= 0 or eps > best_a[i]:
            continue
        a_new = best_a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # small numerical noise: ensure no negative due to rounding
        if a_new[i] < 0:
            continue
        # compute score
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new
            # optional: print improvement
            if trial % 1000 == 0:
                print(f"        Local search improvement: {best_score:.6f} at trial {trial}")
    return best_a, best_score

"""
search for a low‑value step‑function (non‑negative sequence) for the
max‑convolution bound.

The objective used by `evaluate_sequence` is scale‑invariant:
E(a) = 2·n·max(a*a) / (sum(a)^2).  We therefore keep the vector on the
probability simplex (sum = 1, all components ≥ 0).  Two complementary
gradient‑based heuristics are employed:

1. **Projected sub‑gradient descent** – the classic method that moves
   opposite the sub‑gradient of the currently maximal convolution entry
   and projects back onto the simplex.

2. **Exponentiated‑gradient descent with a soft‑max smoothing** – the
   non‑smooth “max” is approximated by a soft‑max (log‑sum‑exp).  The
   resulting smooth gradient is computed analytically via two FFT‑based
   convolutions.  An exponential (multiplicative) update keeps the iterate
   strictly positive; a renormalisation step restores the simplex constraint.

Both methods are run from several initialisations (arcsine‑shaped,
uniform, and a few random Dirichlet draws) for a variety of sequence
lengths.  The best sequence seen so far is improved further by a cheap
local‑search (small random mass transfers) before the time budget expires.
The function `propose_candidate` returns the best list of floats found.
"""

# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x | x_i≥0, Σx_i=1}."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the non‑positive orthant.
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Real‑to‑real convolution via FFT.  Returns the full convolution
    (len(a)+len(b)-1 entries).
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> tuple[float, int]:
    """Maximum value of the full self‑convolution a*a and its index."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry (index k) w.r.t. a.
    Handles the doubled diagonal term when k is even.
    """
    n = a.size
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


# ----------------------------------------------------------------------
# Scaled objective (identical to evaluate_sequence up to the clipping)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Scale‑invariant objective 2·n·max(a*a) / (sum(a)^2)."""
    n = a.size
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Initial distributions
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """
    Discretised arcsine density:
    a_i ∝ 1 / sqrt((i+0.5)*(n-0.5-i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def random_dirichlet_initial(n: int, conc: float) -> np.ndarray:
    """Draw a random point on the simplex with concentration `conc`."""
    return np.random.dirichlet(np.full(n, conc))


# ----------------------------------------------------------------------
# Optimiser 1 – projected sub‑gradient descent (original approach)
# ----------------------------------------------------------------------
def improve_subgradient(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 80_000,
    eta0: float = 0.7,
    noise_every: int = 600,
    noise_scale: float = 5e-5,
    top_k: int = 1,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution objective.
    `top_k` ≥ 1 replaces the pure max by the average of the top‑k worst
    entries, giving a mild smoothing.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = a.size
    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1

        # current convolution
        conv = convolution_fft(a, a)

        # indices of the top‑k entries (worst values)
        if top_k == 1:
            idxs = [int(np.argmax(conv))]
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        # combined sub‑gradient (average of the chosen entries)
        grad = np.zeros_like(a)
        for k in idxs:
            grad += subgradient_of_max_at_index(a, int(k))
        grad /= float(len(idxs))

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        a_new = a - eta * grad

        # occasional tiny stochastic perturbation
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional progress report
        if it % 4000 == 0:
            print(f"    subgrad iter={it:,d}  cur={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Optimiser 2 – exponentiated gradient with soft‑max smoothing
# ----------------------------------------------------------------------
def improve_exponential(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 30_000,
    eta0: float = 0.4,
    alpha0: float = 1.0,
    alpha_factor: float = 1.01,
    noise_every: int = 800,
    noise_scale: float = 5e-5,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponential) updates on the smoothed objective

        F_α(a) = (1/α)·log Σ_i exp(α·conv_i)

    where conv = a*a.  As α → ∞, F_α → max(conv).  The gradient of F_α
    is a convex combination of the individual max‑convolution gradients,
    which can be computed via two FFT‑based convolutions.
    """
    a = initial_a.astype(float).copy()
    a = a / np.sum(a)          # enforce sum=1 (stays on simplex)
    best_a = a.copy()
    best_score = compute_score(a)

    n = a.size
    it = 0
    alpha = alpha0
    while time.time() < time_deadline and it < max_iter:
        it += 1

        # 1. self‑convolution and softmax weights
        conv = convolution_fft(a, a)                     # length 2n‑1
        # softmax with numerical stability
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_vals = np.exp(scaled - max_scaled)
        p = exp_vals / np.sum(exp_vals)                  # probabilities, sum=1

        # 2. gradient of F_α:
        #    grad = Σ_i p_i · ∇ conv_i
        #    where ∇ conv_i has the same structure as subgradient_of_max_at_index
        #    (but all terms appear, not just the max).
        #    The first part (shifted copy of a) is a correlation,
        #    the second part is a windowed sum.

        # a_rev needed for the correlation term
        a_rev = a[::-1]

        # correlation a_rev ⋆ p  (size 3n‑2) – we need the slice n‑1 … n‑1+n‑1
        corr_full = convolution_fft(a_rev, p)
        grad1 = corr_full[n - 1 : n - 1 + n]             # length n

        # windowed sums of p (length n)
        cumsum = np.concatenate(([0.0], np.cumsum(p)))
        sum_window = cumsum[n:] - cumsum[:-n]            # Σ_{i=k}^{k+n-1} p_i
        grad2 = a * sum_window

        grad = grad1 + grad2

        # optional noise to avoid stagnation
        if noise_every and (it % noise_every == 0):
            grad += np.random.randn(n) * noise_scale

        # multiplicative update
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to stay on the simplex
        a = a / np.sum(a)

        # update smoothing parameter slowly
        alpha *= alpha_factor

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if it % 3000 == 0:
            print(f"    expgrad iter={it:,d}  cur={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight local search (random pairwise mass transfers)
# ----------------------------------------------------------------------
def local_search(
    a: np.ndarray,
    time_deadline: float,
    max_trials: int = 120_000,
    step_scale: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: repeatedly move a tiny amount of mass
    from one coordinate to another and keep the move if it lowers the
    objective.  The step size is a fraction of the donor mass.
    """
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = a.size
    trials = 0
    while time.time() < time_deadline - 0.5 and trials < max_trials:
        trials += 1
        i, j = np.random.choice(n, size=2, replace=False)
        if best_a[i] <= 0:
            continue
        eps = step_scale * np.random.rand() * best_a[i]
        a_new = best_a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # safeguard against round‑off negatives
        if a_new[i] < 0:
            continue
        cur = compute_score(a_new)
        if cur < best_score:
            best_score = cur
            best_a = a_new
            if trials % 2000 == 0:
                print(f"      local‑search improvement: {best_score:.6f} (trial {trials})")
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with minimal `evaluate_sequence` value.
    The routine mixes three optimisation ideas (sub‑gradient, exponential,
    and local search) across a few sequence lengths and returns the best
    sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Use any previously known construction as a safe fallback.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Candidate lengths – powers of two give a good FFT‑to‑size ratio.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]

    # ------------------------------------------------------------------
    # Main exploration loop – stop as soon as the global deadline is near.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # a few diverse initialisations
        inits = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),                     # uniform
        ]
        # add two random Dirichlet starts with low concentration (U‑shaped)
        for _ in range(2):
            conc = np.random.uniform(0.1, 0.4)
            inits.append(random_dirichlet_initial(n, conc))

        for init_id, init_a in enumerate(inits, start=1):
            if time.time() >= deadline - 5.0:
                break

            # --------------------------------------------------------------
            # 1) Projected sub‑gradient descent (with top‑k smoothing)
            # --------------------------------------------------------------
            print(f"  Run {init_id}/{len(inits)} – subgradient")
            seq_sg, val_sg = improve_subgradient(
                initial_a=init_a,
                time_deadline=deadline,
                max_iter=60_000,
                eta0=0.7,
                noise_every=700,
                noise_scale=8e-5,
                top_k=3,                     # slight smoothing
            )
            if val_sg < best_val:
                best_val = val_sg
                best_seq = seq_sg.tolist()
                print(f"    *** New global best (subgrad): {best_val:.6f} (n={n}) ***")

            # --------------------------------------------------------------
            # 2) Exponential‑gradient descent with soft‑max smoothing
            # --------------------------------------------------------------
            print(f"  Run {init_id}/{len(inits)} – exponential")
            seq_ex, val_ex = improve_exponential(
                initial_a=init_a,
                time_deadline=deadline,
                max_iter=30_000,
                eta0=0.45,
                alpha0=1.0,
                alpha_factor=1.015,
                noise_every=900,
                noise_scale=6e-5,
            )
            if val_ex < best_val:
                best_val = val_ex
                best_seq = seq_ex.tolist()
                print(f"    *** New global best (expgrad): {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final polishing with a cheap local search if time permits.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:
        print("\n--- Local search polishing ---")
        a_best = np.asarray(best_seq, dtype=float)
        a_best = a_best / np.sum(a_best)  # enforce exact simplex
        a_best, val_local = local_search(
            a_best,
            time_deadline=deadline,
            max_trials=150_000,
            step_scale=0.015,
        )
        if val_local < best_val:
            best_val = val_local
            best_seq = a_best.tolist()
            print(f"    *** New global best after local search: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # Final evaluation with the official (slow) scorer.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
