def propose_candidate(seed=42, budget_s=1000, **kwargs):

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose length: if height_sequence_1 exists, else default.
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 4000

    # Initialize with sqrt_distribution if not None
    a = sqrt_distribution(n).copy()
    # Optionally mix with random: a = a * 0.5 + 0.5*random
    a = project_to_simplex(a)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score}")

    # Hyperparameters
    eta = 0.5   # initial learning rate for exponentiated update (multiplicative)
    min_eta = 0.001
    decay_eta = 0.9995 # each iteration multiply by this
    stagnation = 0

    # tolerance for mask
    mask_tol = 1e-8

    iteration = 0

    while time.time() < deadline:
        # Compute convolution and max
        b = np.convolve(a, a)
        M = b.max()

        # compute mask of peaks (allow near)
        # Determine tolerance relative to M
        thr = M - max(mask_tol, 1e-6 * M)
        mask = (b >= thr).astype(np.float64)

        # compute subgradient
        # conv with reversed mask
        w_rev = mask[::-1]
        conv = np.convolve(a, w_rev)
        L = mask.size
        grad = 2.0 * conv[(L - 1) - np.arange(a.size)]

        # update multiplier
        a_new = a * np.exp(-eta * grad)

        # project onto simplex (sum=1)
        a_new = project_to_simplex(a_new)

        # evaluate
        cur_score = evaluate_sequence(a_new.tolist())

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[{iteration}] improved score = {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        a = a_new

        # adapt learning rate
        eta = max(min_eta, eta * decay_eta)

        # occasional random jitter
        if stagnation > 2000:
            jitter = 0.01 * np.random.randn(n)
            a = project_to_simplex(a + jitter)
            eta = max(min_eta, eta * 0.5)
            stagnation = 0
            print(f"[{iteration}] jitter applied")

        # occasional full restart if stuck for long
        if stagnation > 10000:
            a = np.random.rand(n)
            a = project_to_simplex(a)
            eta = 0.5
            stagnation = 0
            print(f"[{iteration}] random restart")

        iteration += 1

    # after loop, return best_seq as list
    print(f"[done] best score = {best_score}, iterations={iteration}")
    return best_seq.tolist()

"""
Propose a low‑score step‑function sequence.

Algorithm
---------
1. Initialise a non‑negative probability vector `a` (sum = 1):
   * If the global `height_sequence_1` is available, use it (normalised).
   * Otherwise start from the analytically good “square‑root’’ distribution
     (coefficients of (1‑z)^(-1/2)).
2. Iteratively reduce the maximum autocorrelation value.
   * At each iteration compute the self‑convolution `b = a * a`.
   * Determine the current maximal value `M = max(b)`.  All indices where
     `b` is within a tiny tolerance of `M` are regarded as “active’’.
   * The sub‑gradient of `max(b)` w.r.t. `a` is
        g_i = 2 * Σ_j a_j · 1_{i+j active}.
     This is efficiently obtained with one convolution using the binary mask
     of active indices.
   * Apply a multiplicative (exponentiated‑gradient) update
        a ← a * exp( – η * g )
     and project back onto the probability simplex.
   * The learning rate `η` decays slowly; when many iterations pass without
     improvement a small random jitter is added to escape plateaus.
3. Keep the best sequence (lowest score) seen so far.
4. Stop when the allotted wall‑clock time (≈1000 s) is exhausted and
   return the best sequence as a plain Python list.

The objective that is minimised is
        score = 2·n·max_k (a*a)_k          (because Σ a_i = 1)

which is exactly the value returned by `evaluate_sequence` after normalising.
The method works directly on the true max (no soft‑max smoothing) and
tends to flatten the autocorrelation, beating the previous π/2 ≈ 1.571 bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – required by the execution platform)
# ----------------------------------------------------------------------


def sqrt_distribution(n: int) -> np.ndarray:
    """
    Return the normalised coefficients of (1‑z)^(-1/2) up to length n.
    The recurrence is  c_0 = 1, c_{i} = c_{i‑1}·(2i‑1)/(2i).
    The vector is finally normalised to have unit sum.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a non‑negative vector onto the probability simplex (sum = 1).
    For a non‑negative vector this is simply scaling by the sum,
    with a fallback to the uniform distribution if the sum is zero.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def max_subgradient(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of `max(convolve(a,a))` given a binary mask
    of the active (maximal) convolution entries.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum = 1).
    mask : np.ndarray, shape (2*n‑1,)
        Binary (0/1) array: 1 at positions where the convolution attains its
        maximum (or is within a tiny tolerance).

    Returns
    -------
    grad : np.ndarray, shape (n,)
        Gradient g_i = 2 * Σ_j a_j * mask_{i+j}.
    """
    # reverse the mask and convolve
    w_rev = mask[::-1]
    conv = np.convolve(a, w_rev)                # length = n + (2n‑1) – 1 = 3n‑2
    L = mask.size                               # = 2n‑1
    grad = 2.0 * conv[(L - 1) - np.arange(a.size)]
    return grad

# Main entry point (required by the platform)
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low
    `evaluate_sequence` score.  Returns the best sequence found before the
    time limit expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a working length.
    # If a previous construction is available we keep its length,
    # otherwise we fall back to a moderately large size.
    # --------------------------------------------------------------
    try:

    except Exception:

        n = 3000

    # Ensure we are on the simplex.
    a = project_to_simplex(a)

    # Initial score (exact formula, because Σa = 1).
    b = np.convolve(a, a)
    best_score = 2.0 * n * b.max()
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    eta = 0.20            # initial multiplicative step size
    eta_min = 0.001
    decay_eta = 0.9997    # per‑iteration decay
    stagnation = 0        # iterations without improvement
    jitter_thresh = 2000 # when to inject random jitter
    restart_thresh = 8000
    rel_tol = 1e-8        # tolerance for defining the active set
    abs_tol = 1e-12

    iteration = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # 1) Self‑convolution and identification of the active set.
        b = np.convolve(a, a)
        M = b.max()
        # tolerance protects against numerical noise.
        tol = max(abs_tol, rel_tol * M)
        mask = (b >= M - tol).astype(np.float64)

        a_new = project_to_simplex(a_new)

        a = a_new

        # ----------------------------------------------------------

        # ----------------------------------------------------------
        eta = max(eta_min, eta * decay_eta)

        # ----------------------------------------------------------
        # Escape mechanisms when progress stalls.
        # ----------------------------------------------------------
        if stagnation >= jitter_thresh:
            # Small random perturbation keeps the search alive.
            jitter = 0.01 * np.random.randn(n)
            a = project_to_simplex(a + jitter)

            a = np.random.rand(n)
            a = project_to_simplex(a)
            eta = 0.20
            stagnation = 0
            print(f"[{iteration:06d}] full random restart")

        iteration += 1

    print(f"[done] best score = {best_score:.6f} after {iteration} iterations")
    # Return a plain Python list as required by the evaluation harness.
    return best_seq.tolist()
