"""
propose_candidate
-----------------
Iterative improvement based on the classic “active‑set” method described
in the problem statement.

1. Initialise the step heights with the optimal central‑binomial
   coefficients (the discrete analogue of f(t)=t⁻¹ᐟ²).  The sequence is
   normalised so that Σ a_i = √(2·n); with this normalisation the
   evaluation function reduces to the maximal entry of the
   auto‑convolution, i.e.  score = max(a*a).

2. For the current sequence f compute M = max(f*f).  Then solve the linear
   programme

       maximise   Σ b_j
       subject to  (f ⋆ b)[k] ≤ M   for all k = 0 … 2·n‑2
                   b_j ≥ 0

   The convolution (f⋆b) is linear in b, therefore the constraints are
   linear.  The LP is solved with SciPy’s HiGHS interior‑point solver.
   The solution b may have a larger ℓ₁‑norm than f while still respecting
   the same bound M.

3. Rescale b to the required ℓ₁‑norm ( √(2·n) ) obtaining g.

4. Update the current sequence by a convex combination
       f ← (1‑t)·f + t·g       with a step size t∈(0,1).
   This never increases the bound M, while the ℓ₁‑norm stays fixed;
   consequently the evaluation score (which equals M after normalisation)
   is non‑increasing.

5. Repeat steps 2‑4 until the time budget expires or no further
   improvement is observed.  The best sequence encountered is returned.

The algorithm uses a modest problem size (n≈800) so that the full
convolution matrix (≈2 n × n) comfortably fits in memory and each LP
solves in a few seconds.  Within the 1000 s budget several iterations
are performed, typically reducing the score from the theoretical
π/2≈1.5708 down below the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog


def binomial_sequence(length: int) -> np.ndarray:
    """Central binomial coefficients c_k = (2k choose k) / 4**k."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A such that A @ b = np.convolve(f, b)
    (full linear convolution).  A has shape (2*n-1, n).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Return a step‑height sequence that (heuristically) minimises
    the provided `evaluate_sequence` function.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # Choose a manageable length.  Use the supplied construction if it
    # exists, otherwise fall back to a default (≈800).
    MAX_N = 800
    try:
        n = min(len(height_sequence_1), MAX_N)   # type: ignore[name-defined]
    except Exception:
        n = MAX_N

    # ------------------------------------------------------------------
    # Initialise with the canonical binomial sequence and normalise it so
    # that Σ a_i = √(2·n).  With this normalisation the evaluation score is
    # exactly max(conv).
    f = binomial_sequence(n)
    target_norm = np.sqrt(2.0 * n)
    f = f * (target_norm / f.sum())

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop.
    iteration = 0
    t = 0.5                # initial convex‑combination step size
    while time.time() < deadline and iteration < 30:
        # current bound M = max(f*f)
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # build the linear constraints matrix A (full convolution)
        A = build_conv_matrix(f)

        # Linear programme: maximise Σ b_j  subject to A·b ≤ M , b ≥ 0
        c = -np.ones(n)                     # linprog minimises, so use –1
        b_ub = np.full(2 * n - 1, M + 1e-12)   # tiny tolerance
        bounds = [(0.0, None) for _ in range(n)]

        # Solve the LP (HiGHS is fast and robust for medium sized problems)
        lp_res = linprog(c,
                         A_ub=A,
                         b_ub=b_ub,
                         bounds=bounds,
                         method='highs',
                         options={'presolve': True})
        if not lp_res.success:
            print(f"[lp] iteration {iteration} – LP failed, terminating.")
            break
        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            print("[lp] obtained zero vector, terminating.")
            break

        # Rescale to keep the ℓ₁‑norm fixed ( √(2·n) )
        g = b * (target_norm / sum_b)

        # Convex combination – this never increases the bound M
        f_new = (1.0 - t) * f + t * g

        # Evaluate the new candidate
        new_score = evaluate_sequence(f_new.tolist())

        print(f"[it {iteration:02d}] M={M:.6f}  t={t:.3f}  score={new_score:.6f}")

        # Keep the best sequence seen so far
        if new_score < best_score:
            best_score = new_score
            best_seq = f_new.copy()
            print(f"[improve] new best score = {best_score:.6f}")

        # Accept the new point if it does not worsen the score
        if new_score <= best_score + 1e-9:
            f = f_new
        else:
            # If no improvement, reduce the step size
            t *= 0.5
            if t < 1e-3:
                print("[stop] step size too small, exiting.")
                break

        # Slowly decay the step size to aid convergence
        t = max(0.1, t * 0.95)

        iteration += 1

    print(f"[final] best score = {best_score:.6f}")
    # Return a plain Python list as required by the external framework
    return best_seq.tolist()
