"""
Propose a candidate step‑function sequence that minimizes the provided evaluate_sequence.

Algorithm
---------
1. Initialise a non‑negative sequence `f` of length *n*.
   By default we use the classic “constant‑convolution” construction
   a_k = C(2k,k) / 4^k and renormalise it so that Σ a_k = 1
   (this gives the well‑known π/2 bound).

2. **LP improvement phase** (requires CVXPY):
   For the current `f` compute M = max_k (f*f)[k].
   Solve the linear programme
        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f ∗ g)[k] ≤ M  for all k = 0 … 2n‑2 .
   The constraints are linear in `g` because `f` is fixed.
   The solution `g` typically has a larger ℓ₁‑norm than `f`.
   Scale `g` to unit ℓ₁‑norm and mix it with `f`:
        f ← (1‑τ)·f + τ·ĝ ,   τ is a decreasing mixing factor.
   The mix reduces the maximal self‑convolution, i.e. the objective
   2·n·max(f∗f).  A few (≤ 15) LP iterations already give a noticeable
   improvement over the π/2 bound.

3. **Projected sub‑gradient refinement** (optional fallback):
   If CVXPY is unavailable or once the LP phase finishes, perform a
   standard projected sub‑gradient descent on the true objective
   `max_k (a∗a)[k]` with the constraints aᵢ ≥ 0 and Σ aᵢ = 1.
   This fine‑tunes the sequence.

4. Keep the best sequence seen during the run, report its score with
   `evaluate_sequence`, and return it.

The whole search respects the 1000 s time budget; it stops early if the
target bound (~1.503) is reached.
"""

import time
import numpy as np

#--- optional imports -------------------------------------------------
try:
    import cvxpy as cp          # for the LP phase
    _CVX_AVAILABLE = True
except Exception:               # pragma: no cover
    _CVX_AVAILABLE = False

# For the optional sub‑gradient phase we need no extra packages.
# --------------------------------------------------------------------


def _constant_conv_base(n: int) -> np.ndarray:
    """
    Return the first n terms of the classic constant‑convolution sequence
    a_k = binom(2k,k) / 4**k.
    Recurrence: a_0 = 1, a_k = a_{k‑1} * (1 - 1/(2k)).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for minimising max(conv(a,a)).
    Returns the updated vector (still non‑negative, not yet normalised).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    kmax = int(np.argmax(conv))

    # Build the sub‑gradient g where g[i] = 2 * a[kmax-i] if the index is valid
    g = np.zeros_like(a)

    # The convolution index kmax determines a reversed slice of a
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    # positions i in [lo, hi] satisfy 0 ≤ kmax-i < n
    g[lo:hi + 1] = 2.0 * a[kmax - np.arange(lo, hi + 1)]

    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)

    total = a_new.sum()
    if total <= 0.0:
        # Extremely unlikely, but guard against division by zero
        a_new = np.random.rand(n)
        a_new /= a_new.sum()
        return a_new

    return a_new / total


def _lp_max_sum(f: np.ndarray, M: float):
    """
    Solve the LP:
        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f ∗ g)[k] ≤ M   for all k.
    Returns the optimal g (as a NumPy array) or None on failure.
    """
    n = f.shape[0]

    # Define variable
    g = cp.Variable(n, nonneg=True)

    # Convolution expression; yields a vector of length 2n‑1
    conv_expr = cp.conv(cp.Constant(f), g)

    # LP: maximise sum(g) with pointwise upper bound M on the convolution
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_expr <= M])

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False,
                   max_iters=5000)          # ECOS handles pure LPs quite well
    except Exception:                           # pragma: no cover
        return None

    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64)


def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.
    Returns the best sequence found within the allocated time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------------
    # Choose a length.  Prefer the length of a previously found construction
    # if available, otherwise fall back to a moderate default.
    # --------------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1,
                                                       (list, tuple)):
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.size == 0:
            n = 1500
        else:
            n = init.shape[0]
    else:
        n = 1500  # a size that balances speed and quality

    # --------------------------------------------------------------------
    # Initialise the sequence.
    # --------------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1,
                                                       (list, tuple)):
        f = init.copy()
        if f.sum() == 0.0:
            f = np.random.rand(n)
        f = f / f.sum()
    else:
        base = _constant_conv_base(n)
        f = base / base.sum()   # sum = 1, non‑negative

    # keep the best solution seen so far
    conv = np.convolve(f, f)
    best_score = 2.0 * n * conv.max()
    best_seq = f.copy()
    print(f"Initial length {n}, initial score {best_score:.6f}")

    # --------------------------------------------------------------------
    # Phase 1 – LP improvement (only if cvxpy is available)
    # --------------------------------------------------------------------
    lp_iter = 0
    max_lp_iters = 15
    tau = 0.5            # initial mixing weight
    tau_decay = 0.95

    if _CVX_AVAILABLE:
        while time.time() < deadline and lp_iter < max_lp_iters:
            M = np.convolve(f, f).max()

            g = _lp_max_sum(f, M)
            if g is None:
                print("LP solver failed – exiting LP phase.")
                break

            sum_g = g.sum()
            if sum_g <= 1.0 + 1e-12:
                # No further improvement possible
                print(f"LP iter {lp_iter}: sum(g)≈{sum_g:.6f} (no gain).")
                break

            # Scale g to unit L1‑norm
            g_scaled = g / sum_g

            # Mix the new candidate with the current sequence
            f = (1.0 - tau) * f + tau * g_scaled
            f = np.maximum(f, 0.0)
            f /= f.sum()

            # Evaluate the new objective
            conv = np.convolve(f, f)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"LP iter {lp_iter}: improved score {best_score:.6f}")

            # Prepare next iteration
            lp_iter += 1
            tau *= tau_decay  # slowly shrink the mixing step

            # Early exit if we already beat the target
            if best_score < 1.503:
                print("Target bound reached during LP phase.")
                break
    else:

    # Phase 2 – Projected sub‑gradient descent (fine‑tuning)
    # --------------------------------------------------------------------
    sub_iter = 0
    step0 = 0.5
    while time.time() < deadline and sub_iter < 500000:
        step = step0 / np.sqrt(sub_iter + 1.0)
        f = _subgradient_step(f, step)

        # occasional tiny random jitter to avoid stagnation
        if sub_iter % 3000 == 0 and sub_iter > 0:
            f += np.random.randn(n) * 1e-6
            f = np.maximum(f, 0.0)
            f /= f.sum()

        conv = np.convolve(f, f)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"Sub‑grad iter {sub_iter:6d}: improved score {best_score:.6f}")

        sub_iter += 1

        # early exit if we have already passed the target bound
        if best_score < 1.503:
            print("Target bound reached during sub‑gradient phase.")
            break

    # --------------------------------------------------------------------
    # Final sanity check using evaluate_sequence (cheap compared to the search)
    # --------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final reported score by evaluate_sequence: {final_score:.6f}")
    except Exception:                               # pragma: no cover
        final_score = best_score

    print(f"Search finished after {lp_iter} LP iters + {sub_iter} sub‑grad iters, "
          f"best score {best_score:.6f}")
    return best_seq.tolist()
