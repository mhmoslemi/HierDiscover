    import numpy as np, time, math

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    best_seq = None
    best_score = float('inf')
    # length options
    lengths = [2000, 3000, 4000, 5000, 8000]
    for n in lengths:
        target_sum = math.sqrt(2.0 * n)
        # seeding: geometric
        lambdas = np.logspace(-5, -1, 80)  # fewer
        for lam in lambdas:
            r = 1.0 - lam
            # compute C
            r_n = r**n
            den = 1 - r_n
            if den <= 0: continue
            C = target_sum * lam / den
            a = C * (r ** np.arange(n))
            a = np.clip(a, 0.0, 1000.0)
            score = eval_seq(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Init len {n} lam {lam:.4g} score {score:.6f}")

        # power law seed
        idx = np.arange(1, n+1)
        a_pow = 1.0 / np.sqrt(idx)
        C_pow = target_sum / a_pow.sum()
        a_pow = C_pow * a_pow
        a_pow = np.clip(a_pow, 0.0, 1000.0)
        score = eval_seq(a_pow)
        if score < best_score:

            best_seq = a_pow.copy()

        if best_seq is None:

        # Actually to keep per length, we have its own best (best_n_seq). Let's maintain per n.
        seq = best_seq.copy()
        # optimizer parameters:
        m = np.zeros_like(seq)
        v = np.zeros_like(seq)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        t = 0
        step_size = 0.5 * target_sum / n  # scale

        fft_len = 1 << ((2*n - 1).bit_length())

            F = np.fft.rfft(seq, n=fft_len)

            idx_arr = np.arange(n)

"""
Efficient search for a non‑negative step function minimizing the
evaluate_sequence() bound.

The algorithm proceeds in two stages.
1) A fast global scan over several problem sizes (n≈1000…8000) using
   one‑parameter geometric or power‑law families.  The best seed
   (length and parameters) is kept.
2) A long‐run projected Adam optimisation on the chosen seed.
   At each iteration the index where the self‑convolution attains its
   maximum is identified; the sub‑gradient of this max functional is
   used, an Adam step is taken, and the result is projected onto the
   simplex {x≥0, Σx = √(2 n)}.  The step size decays slowly and occasional
   random jitter helps escape plateaus.

The routine respects the 1000 s time budget, prints progress and finally
returns the best sequence found as a plain Python list.
"""
import numpy as np
import time
import math

def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def eval_seq(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence for a NumPy 1‑D array."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    # next power of two for linear convolution via FFT
    fft_len = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

                global_best_n = n

            global_best_n = n

    best_score = global_best_score

    beta2 = 0.999

    v = np.zeros_like(seq)

    step_size = 0.8 * target_sum / n  # typical move of a few % of the mean height

    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # ----- current convolution, max index and gradient ----------
        F = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
        k_max = int(np.argmax(conv))
        # sub‑gradient of the max‑functional
        idx_arr = np.arange(n)
        j = k_max - idx_arr
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(seq)
        grad[mask] = 2.0 * seq[j[mask]]

        # ----- Adam update -----------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = - step_size * m_hat / (np.sqrt(v_hat) + eps)

        trial = seq + step
        trial = np.clip(trial, 0.0, 1000.0)
        trial = projection_simplex(trial, target_sum)

        # ----- evaluate trial ----------------------------------------
        F2 = np.fft.rfft(trial, n=fft_len)
        conv2 = np.fft.irfft(F2 * F2, n=fft_len)[: 2 * n - 1]
        max_b2 = conv2.max()
        trial_score = float(2.0 * n * max_b2 / (trial.sum() * trial.sum()))

        # ----- record best -------------------------------------------
        if trial_score < best_score:
            best_score = trial_score
            best_seq = trial.copy()
            print(f"[{time.time()-start_time:6.1f}s] new global best {best_score:.6f}")

        # ----- accept if we made progress, else keep old ----------
        # Simple acceptance: keep trial if it does not increase the
        # current objective (helps stability).
        curr_max_b = conv.max()
        curr_score = float(2.0 * n * curr_max_b / (seq.sum() * seq.sum()))
        if trial_score <= curr_score:
            seq = trial
            # modestly increase step size when we are successful
            step_size = min(step_size * 1.02, 2.0 * target_sum / n)
        else:
            # otherwise shrink step size a bit
            step_size = max(step_size * 0.96, 1e-9)

            jitter = np.random.randn(n) * step_size * 0.15
            seq = projection_simplex(np.clip(seq + jitter, 0.0, 1000.0), target_sum)

        # early exit check
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3) Final sanity check using the official evaluate_sequence (optional)
    # ------------------------------------------------------------------
    try:
        # The external evaluator will clip internally; we just pass the list.
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Search finished. Final official score {final_score:.6f}")
    except Exception:
        # In case evaluate_sequence is not available in the environment.
        final_score = best_score
        print(f"Search finished. Best internal estimate {best_score:.6f}")

    return best_seq.tolist()

        return float("inf")
    fft_len = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function that minimises the given evaluation
    function.  Returns a list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # a tiny safety margin

    # --------------------------------------------------------------
    # 1️⃣  Global scan over several lengths
    # --------------------------------------------------------------
    length_candidates = [1200, 2000, 3000, 4000, 6000, 8000]
    global_best_seq = None
    global_best_score = float("inf")
    global_best_n = None

    for n in length_candidates:
        target_sum = math.sqrt(2.0 * n)
        idx = np.arange(n, dtype=float)

        # ----- geometric family ------------------------------------------------
        # λ = 1‑r  (we sweep λ from very small up to roughly 0.6)
        lambdas = np.linspace(1e-5, 0.6, 200)
        for lam in lambdas:
            r = 1.0 - lam
            if r <= 0.0:
                continue
            log_r = math.log(r)
            # r**n can under‑flow; treat very small values as zero
            log_rn = n * log_r
            r_n = math.exp(log_rn) if log_rn > -700 else 0.0
            denom = 1.0 - r_n
            if denom <= 0.0:
                continue
            C = target_sum * lam / denom
            a = C * np.exp(idx * log_r)          # C·r**i
            a = np.clip(a, 0.0, 1000.0)
            score = eval_seq(a)
            if score < global_best_score:
                global_best_score = score
                global_best_seq = a.copy()
                global_best_n = n
                print(f"[{time.time()-start_time:6.1f}s] geom n={n:4d} lam={lam:.4e} → {score:.6f}")

        # ----- power‑law family ------------------------------------------------
        ipow = np.arange(1, n + 1, dtype=float)
        a_pow = 1.0 / np.sqrt(ipow)
        C_pow = target_sum / a_pow.sum()
        a_pow = C_pow * a_pow
        a_pow = np.clip(a_pow, 0.0, 1000.0)
        score_pow = eval_seq(a_pow)
        if score_pow < global_best_score:
            global_best_score = score_pow
            global_best_seq = a_pow.copy()
            global_best_n = n
            print(f"[{time.time()-start_time:6.1f}s] power n={n:4d} → {score_pow:.6f}")

    # ----- optional user‑provided seed -----------------------------------------
    if "height_sequence_1" in globals():
        seq_user = np.array(height_sequence_1, dtype=float)
        n_user = seq_user.size
        if n_user > 0:
            target_sum_user = math.sqrt(2.0 * n_user)
            if seq_user.sum() > 0:
                scale = target_sum_user / seq_user.sum()
                seq_user = np.clip(seq_user * scale, 0.0, 1000.0)
                seq_user = projection_simplex(seq_user, target_sum_user)
                score_user = eval_seq(seq_user)
                if score_user < global_best_score:
                    global_best_score = score_user
                    global_best_seq = seq_user.copy()
                    global_best_n = n_user
                    print(f"[{time.time()-start_time:6.1f}s] user seed n={n_user} → {score_user:.6f}")

    # Fallback – uniform if nothing worked
    if global_best_seq is None:
        n = 3000
        target_sum = math.sqrt(2.0 * n)
        global_best_seq = np.full(n, target_sum / n)
        global_best_n = n
        global_best_score = eval_seq(global_best_seq)
        print("Fallback to uniform sequence.")

    # --------------------------------------------------------------
    # 2️⃣  Projected Adam optimisation on the chosen length
    # --------------------------------------------------------------
    n = global_best_n
    target_sum = math.sqrt(2.0 * n)
    seq = global_best_seq.copy()
    seq = projection_simplex(seq, target_sum)          # enforce exact sum

    best_seq = seq.copy()
    best_score = global_best_score

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    t = 0

    # initial absolute step size (a few percent of a typical entry)
    step_size = 0.8 * target_sum / n

    # FFT buffer size (fixed)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # timing helpers for restart logic
    last_improve = time.time()

    while time.time() < deadline:
        # ----- current convolution and max index ---------------------------------
        F = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # ----- sub‑gradient of the max‑functional ---------------------------------
        idx_arr = np.arange(n)
        j = k_max - idx_arr
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(seq)
        grad[mask] = 2.0 * seq[j[mask]]

        # ----- Adam update -------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = - step_size * m_hat / (np.sqrt(v_hat) + eps)

        trial = seq + step
        trial = np.clip(trial, 0.0, 1000.0)
        trial = projection_simplex(trial, target_sum)

        # ----- evaluate trial ----------------------------------------------------
        F2 = np.fft.rfft(trial, n=fft_len)
        conv2 = np.fft.irfft(F2 * F2, n=fft_len)[: 2 * n - 1]
        trial_score = float(2.0 * n * conv2.max() / (trial.sum() * trial.sum()))

        # ----- record global best ------------------------------------------------
        if trial_score < best_score:
            best_score = trial_score
            best_seq = trial.copy()
            last_improve = time.time()
            print(f"[{time.time()-start_time:6.1f}s] new global best {best_score:.6f}")

        # ----- simple acceptance (non‑increasing objective) ----------------------
        curr_score = float(2.0 * n * conv.max() / (seq.sum() * seq.sum()))
        if trial_score <= curr_score:
            seq = trial
            step_size = min(step_size * 1.02, 2.0 * target_sum / n)
        else:
            step_size = max(step_size * 0.96, 1e-9)

        # ----- random jitter every few iterations ---------------------------------
        if np.random.rand() < 0.03:
            jitter = np.random.randn(n) * step_size * 0.15
            seq = projection_simplex(np.clip(seq + jitter, 0.0, 1000.0), target_sum)

        # ----- restart from the best if we have been idle for a while ----------
        if time.time() - last_improve > 30.0:
            seq = projection_simplex(
                np.clip(best_seq + np.random.randn(n) * step_size * 0.1,
                        0.0, 1000.0), target_sum
            )
            last_improve = time.time()

        # ----- time check ---------------------------------------------------------
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 3️⃣  Final sanity check with the official evaluator (if available)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Search finished. Final official score {final_score:.6f}")
    except Exception:
        final_score = best_score
        print(f"Search finished. Best internal estimate {best_score:.6f}")

    return best_seq.tolist()
