"""
Propose a low‑score step function.

Algorithm
---------
1.  Choose a length `n` (4000) and the required ℓ¹‑norm
    `S = sqrt(2·n)`.  Initialise the sequence with a fast
    geometric scan and a power‑law seed.

2.  Iterate a *soft‑max* projected gradient descent.
    * The convolution `c = a * a` is obtained by an FFT.
    * A soft‑max weight `w_i ∝ exp(β·c_i)` (β starts small and is
      slowly increased) spreads the gradient over the *top*
      autocorrelation values instead of only the single maximum.
    * The gradient of the smoothed objective is
        `∇ = 2·(w ⊛ a_rev)`   (⊛ = linear convolution,
        `a_rev` = reversed `a`).  The convolution is again
        performed with an FFT.
    * After a step in the opposite direction we project the
      candidate back onto the simplex `{x≥0, Σx=S}` using the
      Duchi‑et‑al. algorithm.

3.  The step size decays gradually, a tiny amount of Gaussian
    jitter is added from time to time and a restart is performed
    if no improvement is seen for many iterations.

4.  The best sequence encountered is returned.  The final score is
    printed for reference.

The method keeps the per‑iteration cost low (a handful of FFTs) and
performs many thousands of iterations within the 1000 s budget,
typically beating the previous best value of 1.57657.
"""

import numpy as np
import time
from math import sqrt

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, Σx = s} (Duchi et al., 2008)."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """

    deadline = start_time + budget_s - 0.2    # safety margin

    # ------------------------------------------------------------------- #
    # 1. Problem size
    n = 4000                                 # a bit larger than before
    S = sqrt(2.0 * n)                        # required ℓ¹‑norm

    # ------------------------------------------------------------------- #
    # 2. Simple seeds (geometric sweep + power law)
    best_seq = None
    best_score = float("inf")

    # geometric families – λ = 1‑r
    lambdas = np.logspace(-5, -0.5, 80)      # fewer points, still enough
    for lam in lambdas:
        r = 1.0 - lam
        # avoid under/overflow for r**n
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = S * lam / denom                  # because Σ a_i = C·(1‑rⁿ)/λ = S
        a = C * (r ** np.arange(n))
        a = np.clip(a, 0.0, 1000.0)
        # quick score using the same formula as evaluate_sequence
        conv = np.convolve(a, a)
        cur = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()

    # power‑law seed a_i ∝ 1/√(i+1)
    idx = np.arange(1, n + 1, dtype=float)
    a_pow = 1.0 / np.sqrt(idx)
    C_pow = S / a_pow.sum()
    a_pow = C_pow * a_pow
    a_pow = np.clip(a_pow, 0.0, 1000.0)
    conv = np.convolve(a_pow, a_pow)
    cur = 2.0 * n * conv.max() / (a_pow.sum() ** 2)
    if cur < best_score:

    # fallback – uniform (should never happen)
    if best_seq is None:

    # ------------------------------------------------------------------- #
    # 3. Soft‑max projected gradient descent
    curr = best_seq.copy()
    best = curr.copy()
    best_score = best_score
    it = 0
    last_improve = 0

    # FFT lengths (next powers of two for speed)
    fft_len1 = 1 << ((2 * n - 1).bit_length())   # for a*a
    fft_len2 = 1 << ((3 * n - 1).bit_length())   # for w ⊛ a_rev

    # hyper‑parameters
    step_scale = 0.5          # absolute step magnitude (will be scaled)
    min_step = 1e-8
    decay_factor = 0.97
    decay_interval = 500
    beta = 1.0                # start with a smooth objective
    beta_max = 15.0

    # ------------------------------------------------------------------- #
    while time.time() < deadline:

        exps = np.exp(beta * (conv - mx))
        w = exps / exps.sum()                     # shape (2n‑1,)

            best_score = cur_score
            best = curr.copy()
            last_improve = it
            print(f"[{time.time() - start_time:6.1f}s] iter {it:7d} → best {best_score:.6f}")

            step_scale = max(step_scale * 0.9, min_step)

        # -----  schedule adaptation -----------------------------------
        # increase β very slowly so that the objective becomes sharper
        if beta < beta_max:

    # Final sanity check (the evaluator applies the same clipping)
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Best score = {final_score:.6f}")
    return best.tolist()
