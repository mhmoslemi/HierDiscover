def _conv_full_fft(seq):
    n = len(seq)
    m = 1 << ((2*n-1).bit_length())
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)[:2*n-1]
    # Might have small negative values due to numerical errors; ensure >=0
    # conv = np.maximum(conv, 0.0)
    return conv

def _conv_max(seq):
    return np.max(_conv_full_fft(seq))

import time
import math
import numpy as np

def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Return the full discrete convolution of seq with itself using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())   # next power of two
    fft = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft * fft, m)[:2 * n - 1]
    # numeric noise may produce tiny negative numbers
    np.maximum(conv, 0, out=conv)
    return conv

def _objective(seq: np.ndarray) -> float:
    """Compute the evaluate_sequence objective without the security checks."""
    n = seq.shape[0]
    total = seq.sum()
    if total == 0:
        return float('inf')
    max_c = _conv_full_fft(seq).max()
    return 2.0 * n * max_c / (total * total)

def _initial_sequence(n: int, target_sum: float) -> np.ndarray:
    """Create a deterministic start sequence (integral of 1/√x)."""
    idx = np.arange(n, dtype=float)
    raw = np.sqrt(idx + 1.0) - np.sqrt(idx)
    return raw * (target_sum / raw.sum())

    np.random.seed(seed)

"""step‑function optimizer

Searches for a non‑negative sequence a = (a0,…,a_{n‑1}) that minimises
the value returned by ``evaluate_sequence``:

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages

1.  Pick a length (we try a few values up to 8000) and initialise a
    deterministic “inverse‑square‑root’’ shape (the integral of x⁻¹ᐟ²).
    If a previously‑found construction is available through the global
    variable ``height_sequence_1`` we use it as a start point.

2.  Repeatedly improve the current sequence by *targeted* random moves.
    For the current maximum convolution entry k* we compute the
    contributions a_i·a_{k*‑i} of every index i that participates in that
    entry.  The move transfers a small amount of mass from the index that
    contributes the most to the entry to the index that contributes the
    least, then rescales the whole vector to keep the normalisation
    ``sum = sqrt(2·n)`` (the objective is scale‑invariant).  The move is
    kept only if it improves the objective; a tiny probability of
    accepting a non‑improving move is added to avoid shallow local
    minima.

3.  When the time budget (≈1000 s) is exhausted the best sequence found
    is evaluated with the official ``evaluate_sequence`` routine and
    returned.

All convolutions are performed with an FFT based routine, which makes
thousands of iterations feasible even for n≈8000.  The method stays well
within the one‑CPU limit and avoids any file or network I/O.  """

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures or lambdas)
# ----------------------------------------------------------------------
def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution of *seq* using an FFT."""
    n = seq.shape[0]
    # size of the FFT – the next power of two >= 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, m)
    conv = np.fft.irfft(ft * ft, m)[: 2 * n - 1]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(seq: np.ndarray) -> float:
    """Objective used during the search (exactly the value inside
    ``evaluate_sequence`` but calculated with the fast FFT convolution)."""
    n = seq.shape[0]
    s = seq.sum()
    if s == 0.0:
        return float("inf")
    max_c = _conv_full_fft(seq).max()
    return 2.0 * n * max_c / (s * s)


def _initial_sequence(n: int, target_sum: float) -> np.ndarray:
    """Deterministic start‑sequence – integral of x⁻¹ᐟ² on [0,n]."""
    idx = np.arange(n, dtype=float)
    raw = np.sqrt(idx + 1.0) - np.sqrt(idx)          # ∫_{i}^{i+1} x⁻¹ᐟ² dx
    return raw * (target_sum / raw.sum())


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the evaluation
# harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a step‑function with a low ``evaluate_sequence`` score.

    The search runs for at most *budget_s* seconds and returns a plain
    Python list of non‑negative heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a couple of seconds spare for cleanup

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # 1 – try several lengths (larger n gives a finer discretisation but
    #     costs a little more per iteration).
    # ------------------------------------------------------------------
    lengths = [3000, 4000, 5000, 6000, 8000]

    for n in lengths:
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1a – initialise a candidate.
        # --------------------------------------------------------------
        seq0 = None
        if "height_sequence_1" in globals():
            try:
                hs = np.asarray(height_sequence_1, dtype=float)
                if hs.shape[0] == n:
                    seq0 = hs.copy()
                else:
                    # linear interpolation to the required length
                    xs_old = np.linspace(0.0, 1.0, hs.shape[0])
                    xs_new = np.linspace(0.0, 1.0, n)
                    seq0 = np.interp(xs_new, xs_old, hs)
            except Exception:
                seq0 = None

        if seq0 is None:
            seq0 = _initial_sequence(n, target_sum)
        else:
            seq0 = seq0 * (target_sum / seq0.sum())

        cur_seq = seq0.copy()
        cur_obj = _objective(cur_seq)

        if cur_obj < best_obj:
            best_obj, best_seq = cur_obj, cur_seq.copy()
            print(f"[{n:4d}] initial objective = {cur_obj:.6f}")

        # --------------------------------------------------------------
        # 2 – optimisation loop for the chosen length.
        # --------------------------------------------------------------
        it = 0
        while time.time() < deadline:
            # compute the full convolution and locate the peak
            conv = _conv_full_fft(cur_seq)
            max_val = conv.max()
            max_idx = int(conv.argmax())

            # indices that actually take part in the peak entry
            low = max(0, max_idx - (n - 1))
            high = min(max_idx, n - 1)
            idxs = np.arange(low, high + 1, dtype=int)

            # contribution of each participating index to the peak
            prod = cur_seq[idxs] * cur_seq[max_idx - idxs]

            # pick a source (large contribution) and a destination (small)
            if prod.max() <= 0.0:
                i = np.random.randint(n)
                j = np.random.randint(n)
                while j == i:
                    j = np.random.randint(n)
            else:
                i = idxs[int(prod.argmax())]
                j = idxs[int(prod.argmin())]
                if i == j:
                    # choose a different destination uniformly
                    other = idxs[idxs != i]
                    if other.size == 0:
                        continue
                    j = np.random.choice(other)

            # propose moving a modest fraction of mass from i to j
            max_move = 0.015 * cur_seq[i]          # up‑to 1.5 % of the source amount
            if max_move <= 0.0:
                continue
            delta = np.random.uniform(0.0, max_move)

            cand = cur_seq.copy()
            cand[i] -= delta
            cand[j] += delta

            # re‑normalise to keep the sum fixed (scale‑invariance)
            cand = cand * (target_sum / cand.sum())

            cand_obj = _objective(cand)

            # accept only improving moves (tiny chance of uphill move)
            if cand_obj < best_obj - 1e-9:
                best_obj, best_seq, cur_seq = cand_obj, cand.copy(), cand
                print(f"[{n:4d}] it {it:6d}  improvement → {cand_obj:.6f}")
            else:
                if np.random.rand() < 0.001:   # occasional uphill jump
                    cur_seq = cand

            it += 1

            # early exit if we have already beaten the target
            if best_obj < 1.503:
                break

        if best_obj < 1.503:
            break   # no need to try longer sequences

    # ------------------------------------------------------------------
    # 3 – final verification with the official evaluator.
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Search finished – best objective ≈ {best_obj:.6f}, "
          f"official score = {final_score:.6f}, length = {len(final_seq)}")
    return final_seq
