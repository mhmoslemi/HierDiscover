```
def eval_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    # compute convolution via FFT
    # conv length 2n-1
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)
```

assistantfinal**Solution Overview**

The evaluation function is  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)[k]}{(\sum_i a_i)^2},
\]

where `a*a` is the (non‑reversed) convolution of the non‑negative vector `a`
with itself.  
The score is scale‑invariant; after fixing  

\[
\sum_i a_i = \sqrt{2n},
\]

the score is simply the maximal convolution value.  
The problem is therefore:

*find a non‑negative vector on the simplex*  

\[
\mathcal S=\{a\in\mathbb R_{\ge0}^n\mid \sum a_i =\sqrt{2n}\}
\]

*that minimises the maximal entry of its self‑convolution.*

The algorithm below is a two‑phase search:

1. **Global random search** – 4 families of simple analytic shapes
   (single exponential, two‑segment exponential, power‑law, exponential·power‑law)
   are sampled with random parameters.  Each candidate is projected onto the
   simplex and its score is evaluated with a fast FFT‑based convolution.
   The best candidate of this stage becomes the seed for the next phase.

2. **Projected gradient‑descent with momentum** –  
   Starting from the best seed we repeatedly
   * locate the convolution index `k*` that realises the current maximum,
   * build the sub‑gradient `g_i = 2·a_{k*‑i}` (zero outside the support),
   * take a step opposite to the gradient, add Nesterov‑style momentum,
   * project back onto the simplex.
   The step size decays slowly and occasional random jitter helps to escape
   plateaus.

A larger problem size (`n = 5000`) gives the optimizer more freedom while
still allowing > 10⁵ cheap FFT evaluations within the 1000 s budget.

The code follows the required interface (`propose_candidate`) and prints
useful progress information.

```python
"""
Propose a low‑score step‑function sequence.

Algorithm
---------
1. Choose a problem size  n = 5000  (≈ √2 n = 100 total mass).
2. Randomly generate candidates from several analytic families
   (single exponential, two‑segment exponential, power‑law,
   exponential·power‑law).  Each candidate is projected onto the simplex
   Σa_i = √(2n) and its score is evaluated with an FFT‑based convolution.
   The best candidate of this stage becomes the initial point.
3. Refine the initial point by projected gradient descent with momentum.
   At each iteration we move opposite to the sub‑gradient of the current
   maximal convolution entry, then project back onto the simplex.
   Step size decays slowly; occasional jitter allows the method to escape
   local minima.
4. Return the best sequence found within the time budget.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = s}
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x >= 0, Σx = s} (Duchi et al. 2008)."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the largest rho satisfying u[rho] > cssv[rho]/(rho+1)
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the
# security checks).  Returns the same score that the public evaluator
# would compute.
def fast_score(seq: np.ndarray) -> float:
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    # convolution via FFT (full length 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

# Families of parameterised sequences
def gen_geometric(n, lam, target_sum):
    """Single exponential a_i = C·r^i  with r = 1‑lam."""
    if lam <= 0.0 or lam >= 1.0:
        return None
    r = 1.0 - lam
    # r**n can under‑flow → use log
    rn = np.exp(n * np.log(r)) if r > 0 else 0.0
    denom = 1.0 - rn
    if denom <= 0.0:
        return None
    C = target_sum * lam / denom
    # C will be ≤ target_sum (≈100) → never exceeds the hard cap 1000
    a = C * (r ** np.arange(n))
    return projection_simplex(a, target_sum)


def gen_piecewise_geometric(n, lam1, lam2, split, target_sum):
    """Two‑segment exponential:
    a_i = C1·r1^i  for i<split,
    a_i = C1·r1^split·r2^{i‑split}  otherwise."""
    if not (0.0 < lam1 < 1.0 and 0.0 < lam2 < 1.0):
        return None
    r1 = 1.0 - lam1
    r2 = 1.0 - lam2
    log_r1 = np.log(r1) if r1 > 0 else -np.inf
    log_r2 = np.log(r2) if r2 > 0 else -np.inf

    # contributions of the two parts to the total sum
    r1_split = np.exp(split * log_r1)  # r1**split
    left_sum = (1.0 - r1_split) / lam1
    rest = n - split
    r2_rest = np.exp(rest * log_r2)   # r2**(n‑split)
    right_sum = r1_split * (1.0 - r2_rest) / lam2
    denom = left_sum + right_sum
    if denom <= 0.0:
        return None
    C1 = target_sum / denom

    left = C1 * np.exp(np.arange(split) * log_r1)
    right = C1 * r1_split * np.exp(np.arange(rest) * log_r2)
    a = np.concatenate([left, right])
    return projection_simplex(a, target_sum)


def gen_powerlaw(n, p, target_sum):
    """a_i ∝ (i+1)^{p}  (p < 0 gives a decreasing shape)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    raw = idx ** p
    if raw.sum() == 0.0:
        return None
    C = target_sum / raw.sum()
    a = C * raw
    return projection_simplex(a, target_sum)


def gen_exp_powerlaw(n, lam, p, target_sum):
    """a_i = C·r^i·(i+1)^{p} with r = 1‑lam."""
    if lam <= 0.0 or lam >= 1.0:
        return None
    r = 1.0 - lam
    log_r = np.log(r) if r > 0 else -np.inf
    idx = np.arange(n, dtype=np.float64)
    raw = (np.exp(idx * log_r)) * ((idx + 1) ** p)
    if raw.sum() == 0.0:
        return None
    C = target_sum / raw.sum()
    a = C * raw
    return projection_simplex(a, target_sum)

        step = step_scale * target_sum / n

        # Nesterov‑style momentum update
        momentum = beta * momentum - step * grad / gnorm
        tentative = curr + momentum
        tentative = np.clip(tentative, 0.0, 1000.0)

        # project back onto the simplex Σa_i = target_sum
        curr = projection_simplex(tentative, target_sum)

        # occasional jitter to jump out of tiny plateaus
        if rng.random() < 0.005:
            jitter = rng.normal(scale=step_scale * target_sum / (30.0 * n), size=n)
            curr = projection_simplex(np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        it += 1
        if it % decay_interval == 0:
            step_scale = max(step_scale * decay_factor, min_step)

    # -----------------------------------------------------------------
    # Final check with the official evaluator (sanity)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Best score reported by evaluate_sequence: {final_score:.6f}")
    return best.tolist()
```
