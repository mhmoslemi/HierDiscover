"""
Hybrid optimization: smooth max gradient descent followed by active‑set subgradient descent.

import time, numpy as np

def sqrt_distribution(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    a /= a.sum()
    return a
...

def project_to_simplex(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v

def smooth_max_gradient(a, tau):
    b = np.convolve(a, a)
    max_b = b.max()
    w = np.exp(tau * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)
    L = b.size
    grad = 2.0 * conv_rev[(L-1) - np.arange(a.size)]
    return grad, max_b, b

def max_convolution_and_active(a, eps_rel=1e-12):
    b = np.convolve(a, a)
    max_b = b.max()
    # Avoid zero tolerance on tiny max
    eps = max(eps_rel * max_b, 1e-15)
    active_idx = np.where(b >= max_b - eps)[0]
    return b, max_b, active_idx

def active_set_gradient(a, active_idx):
    n = a.shape[0]
    g = np.zeros_like(a)
    cnt = len(active_idx)
    if cnt == 0:
        return g
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(k, n - 1)
        i_range = np.arange(start_i, end_i + 1, dtype=int)
        # a[k - i] yields reversed order relative to i_range
        g[i_range] += a[k - i_range]
    # gradient of M is 2 * mean_{k in active} a_{k-i}
    g = (2.0 / cnt) * g
    return g

def propose_candidate(seed=42, budget_s=1000, **kwargs):

    except Exception:
        n = 1800

    # initial distribution
    a = sqrt_distribution(n)
    a = project_to_simplex(a)           # ensure sum 1
    best_seq = a.copy()
    # compute initial score manually (same as evaluate_sequence)
    b = np.convolve(a, a)
    best_score = 2.0 * n * b.max() / (a.sum()**2)
    print(f"[init] n={n}, score={best_score:.6f}")

    # hyper‑parameters
    tau = 0.8
    eta = 0.6            # for exponentiated gradient
    step = 0.2           # for active‑set descent
    phase = "smooth"
    it = 0
    last_improve = 0

    while time.time() < deadline:
        if phase == "smooth":
            grad, max_b, _ = smooth_max_gradient(a, tau)

            # exponentiated update
            a_new = a * np.exp(-eta * grad)
            a_new = project_to_simplex(a_new)

            # score
            b_new = np.convolve(a_new, a_new)
            cur_score = 2.0 * n * b_new.max() / (a_new.sum()**2)

            if cur_score < best_score:

                best_seq = a_new.copy()

"""
Hybrid optimiser for the max‑self‑convolution problem.

The objective to minimise is

    score(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

where *n* is the length of the sequence.  Scaling the whole vector does not
change the score, therefore we keep the coefficients on the probability
simplex (sum = 1) and directly minimise the quantity

    M(a) = max(conv(a,a))

which is equivalent up to the constant factor 2·n.

The algorithm works in two phases:

1. **Smooth‑max (exponentiated‑gradient) phase** – a soft‑max
   approximation of the maximum is optimised.  The temperature `tau`
   grows slowly, turning the smooth surrogate into the true max.
   Multiplicative (exponentiated) updates keep the vector non‑negative,
   and a projection onto the simplex restores the normalisation.

2. **Active‑set sub‑gradient phase** – once the soft‑max temperature is
   large enough the true active set (indices where the convolution attains
   its maximum) is identified.  For these indices the exact sub‑gradient
   of `M(a)` is `2·mean_k a_{k‑i}`.  A simple projected gradient step on the
   simplex is then performed.  When progress stalls a tiny random jitter
   is injected to escape flat regions.

Both phases run inside the allotted time budget (≈ 1000 s).  The best
sequence encountered is returned.  The implementation is completely
self‑contained, uses only NumPy and the provided `evaluate_sequence`,
and respects the platform restrictions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level as required by the platform
# ----------------------------------------------------------------------


def sqrt_distribution(n: int) -> np.ndarray:
    """
    Normalised coefficients of (1‑z)^(-1/2) up to length *n*.
    The raw coefficients satisfy
        c_0 = 1,   c_{i+1} = c_i·(2i+1)/(2i+2).
    They are finally scaled to sum to 1 (a probability simplex).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Orthogonal projection of a non‑negative vector onto the probability
    simplex {x ≥ 0, sum(x)=1}.  The implementation follows the classic
    O(n) algorithm based on sorting.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fall back to the centre of the simplex
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def smooth_max_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the smooth‑max surrogate
        g(a) = Σ_k w_k·(a*a)_k ,
    where w_k = softmax_τ((a*a)_k).
    Returns (gradient, current max of a*a, full convolution).
    """
    b = np.convolve(a, a)                     # length 2n‑1
    max_b = b.max()

    # soft‑max weights, numerically stable
    w = np.exp(tau * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2·∑_j a_j·w_{i+j}
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)                     # length 3n‑2
    L = b.size
    grad = 2.0 * conv_rev[(L - 1) - np.arange(a.size)]
    return grad, max_b, b


def max_convolution_and_active(a: np.ndarray, eps_rel: float = 1e-12):
    """
    Compute the self‑convolution, its maximum and the *active* set –
    indices where the convolution is within a tiny absolute tolerance
    of the maximum.
    """
    b = np.convolve(a, a)
    max_b = b.max()
    eps = max(eps_rel * max_b, 1e-15)
    active_idx = np.where(b >= max_b - eps)[0]
    return b, max_b, active_idx


def active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of M(a)=max(conv(a,a)) for a given active set.
    For every active index k, ∂(a*a)_k/∂a_i = 2·a_{k‑i}
    (out‑of‑range indices contribute zero).  The gradient of the max
    is the average over all active indices.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    cnt = len(active_idx)
    if cnt == 0:
        return g

    for k in active_idx:
        # i runs over indices that actually contribute to (a*a)_k
        start_i = max(0, k - (n - 1))
        end_i = min(k, n - 1)
        i_range = np.arange(start_i, end_i + 1, dtype=int)
        # a_{k‑i} for those i
        g[i_range] += a[k - i_range]

    # factor 2 from the derivative and division by the number of active ks
    g = (2.0 / cnt) * g
    return g
