"""
Iterative LP‑based refinement for the self‑convolution bound.

The algorithm follows the “alternating projection” idea described in the
statement:

1.  Start from a non‑negative sequence f (sum normalised to 1).  Its
    self‑convolution maximum M = max(f*f) is computed.

2.  Keeping f fixed, solve the linear programme
        maximise   1ᵀ g
        subject   f ∗ g  ≤  M   (for every entry of the convolution)
                    g ≥ 0
    The variables are the entries of g.  Because the constraints are linear
    in g, a standard LP solver (SciPy highs) can be used.

3.  Rescale the obtained g so that ∑g = ∑f (i.e. 1).  Move f a small step
    towards the rescaled g:
        f₊ = (1−t)·f + t·ĝ .
    The step size t is reduced automatically when the bound does not improve.

4.  Re‑evaluate M = max(f₊∗f₊) and repeat the LP/step cycle.  Each accepted
    step can only decrease the objective
        2·n·max(f*f) / (∑f)² .
    The best sequence encountered is memorised.

The outer driver occasionally restarts from a random parametric family
(exponential, power‑law or linear) and also tries to “grow” a good short
solution to a longer one by linear interpolation.  All operations respect
the non‑negativity requirement and the search runs until the time budget
(≈ 1000 s) is exhausted.
"""

import time
import math
import random

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:               # pragma: no cover
        return math.inf


def _build_conv_matrix(f):
    """
    Build the (2·n‑1)×n matrix A such that A @ g = np.convolve(f, g).

    Column j of A contains a copy of f starting at row j.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = sp.lil_matrix((rows, n), dtype=float)
    for j in range(n):
        start = j
        end = j + n
        A[start:end, j] = f
    return A.tocsr()


def _initial_sequence(n, rng):
    """Return a random decreasing sequence of length n (normalised sum=1)."""
    family = rng.choice(['exp', 'pow', 'lin'])
    if family == 'exp':
        # log‑uniform decay rate
        log_alpha = rng.uniform(-4.0, math.log10(5.0))
        alpha = 10.0 ** log_alpha
        seq = np.exp(-alpha * np.arange(n, dtype=float))
    elif family == 'pow':
        p = rng.uniform(0.0, 3.0)
        idx = np.arange(1, n + 1, dtype=float)
        seq = idx ** (-p)
    else:   # linear
        beta = rng.uniform(0.0, 2.0 / n)
        seq = np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float))
    if seq.sum() == 0.0:
        seq = np.ones(n, dtype=float)
    return seq / seq.sum()


def _interpolate_up(seq, new_n):
    """
    Linear interpolation of a sequence to a longer length.
    The result is renormalised to sum‑1.
    """
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    # guard against rounding to zero
    if new_seq.sum() == 0.0:
        new_seq = np.ones(new_n, dtype=float)
    return new_seq / new_seq.sum()

            n = int(rng.choice(lengths_small))
            f = _initial_sequence(n, rng)
        else:
            # continue from the current best, possibly after up‑scaling
            f = np.asarray(best_seq, dtype=float)
            # if we have plenty of time left, try to enlarge the vector
            if (time.time() < deadline - 150) and (len(f) < max(lengths_large)):
                # up‑scale once per outer iteration with ~30 % probability
                if random.random() < 0.3:
                    target_n = int(rng.choice(lengths_large))
                    f = _interpolate_up(f, target_n)

        n = len(f)

        # --------------------------------------------------------------
        # (b) inner LP‑refinement loop (a few iterations per outer step)
        # --------------------------------------------------------------
        # compute current self‑convolution bound M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # limit the number of inner steps so we stay inside the budget
        inner_max = int(rng.uniform(5, 15))
        for _ in range(inner_max):
            if time.time() >= deadline:
                break

            # build convolution matrix A (size (2n‑1)×n)
            A = _build_conv_matrix(f)

            # LP: maximise sum(g)  ⇔  minimise –sum(g)
            c = -np.ones(n, dtype=float)            # objective
            b_ub = np.full(2 * n - 1, M, dtype=float)  # upper bounds
            bounds = [(0.0, None)] * n               # g ≥ 0

            # solve – use the high‑performance HiGHS interior‑point method
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})

            if not res.success:
                # LP failed – abort inner loop and fall back to random restart
                break

            g = res.x
            sum_g = g.sum()
            if sum_g == 0.0:
                break

            # rescale g so that it has the same total mass as f (which is 1)
            g_hat = g / sum_g

            # -----------------------------------------------------------------
            # (c) convex combination step – adapt step size t dynamically
            # -----------------------------------------------------------------
            # start with a moderate step, shrink if no improvement
            t = 0.25
            while t > 1e-4:
                f_candidate = (1.0 - t) * f + t * g_hat
                f_candidate = f_candidate / f_candidate.sum()   # keep sum = 1

                # evaluate the candidate
                conv_cand = np.convolve(f_candidate, f_candidate)
                M_cand = conv_cand.max()
                score_cand = _score(f_candidate.tolist())

                # accept if the bound has decreased (or the score is better)
                if M_cand < M * (1 - 1e-7):
                    # good progress – keep the candidate
                    f = f_candidate
                    M = M_cand
                    break
                else:
                    # reduce step size and try again
                    t *= 0.5

            # -----------------------------------------------------------------
            # (d) keep the best overall sequence seen so far
            # -----------------------------------------------------------------
            if score_cand < best_score:
                best_score = score_cand
                best_seq = f_candidate.tolist()
                print(f"New best → {best_score:.6f} (len={len(best_seq)})")

        # --------------------------------------------------------------
        # (e) occasional pure hill‑climbing perturbation of the current best
        # --------------------------------------------------------------
        if best_seq is not None and random.random() < 0.12:
            cand = np.array(best_seq, copy=True)
            n_c = len(cand)
            # perturb a few coordinates multiplicatively
            m = rng.randint(1, min(5, n_c))
            idxs = rng.choice(n_c, size=m, replace=False)
            for i in idxs:
                factor = 1.0 + rng.normal(0.0, 0.07)
                cand[i] = max(0.0, cand[i] * factor)
            cand = cand / cand.sum()
            sc = _score(cand.tolist())
            if sc < best_score:
                best_score, best_seq = sc, cand.tolist()
                print(f"Hill‑climb improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) final fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 500
        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
