def gradient_smooth(a, w):
    # a shape n, w shape 2n-1
    n = a.shape[0]
    m = len(w)  # = 2n-1
    # compute correlation using FFT:
    # pad length L = next_pow2(n + m -1)
    L = next_pow2(n + m - 1)
    A = np.fft.rfft(a, n=L)
    W_rev = np.fft.rfft(w[::-1], n=L)
    R = A * np.conj(W_rev)  # elementwise multiplication
    corr = np.fft.irfft(R, n=L)[:n + m - 1]
    # gradient_i = corr[(m-1) - i] for i=0...n-1
    start = m - 1
    grad = corr[start - np.arange(n)]
    # add diagonal term:
    diag_idx = 2 * np.arange(n)
    within = diag_idx < m
    grad[within] += w[diag_idx[within]] * a[within]
    return grad
```

"""
Exploitation of the max‑convolution bound via a hierarchical
smooth‑gradient / sub‑gradient scheme.

* A smooth approximation of the max‑convolution (log‑sum‑exp) yields
  a well‑behaved gradient that spreads the update over many convolution
  entries.
* The gradient of the smooth objective is obtained efficiently by an
  FFT‑based correlation, avoiding O(n²) work.
* An exponentiated‑gradient (mirror descent w.r.t. KL) update keeps the
  vector on the probability simplex without an explicit projection.
* The algorithm proceeds in stages: start from a small arcsine‑shaped
  sequence, optimise it, then up‑sample (linear interpolation) to a
  larger length and repeat.  After the smooth phase a short projected
  sub‑gradient refinement (as in the previous state‑of‑the‑art) is run
  to polish the solution.
* The whole search respects the 1000 s budget and returns the best
  sequence found (as a plain list of floats) for the official evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Real‑to‑real convolution using FFT (full mode)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1} (Duchi et al. 2008)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring utilities (scale‑invariant version of evaluate_sequence)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2) – identical to evaluate_sequence up to clipping."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Objective & gradient (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute gradient of   L(a) = (1/α)·log Σ_k exp(α·c_k)   where c = a*a.
    Returns a vector of length len(a) on the probability simplex.
    """
    n = len(a)
    conv = convolution_fft(a, a)                     # length 2n‑1
    max_c = np.max(conv)

    # softmax weights (numerically stable)
    w = np.exp(alpha * (conv - max_c))
    w_sum = np.sum(w)
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / len(w))
    else:
        w /= w_sum

    # correlation a ⋆ w  (a[i]·w[i+shift]) → gradient before diagonal fix
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)                # length 3n‑2
    start = len(w) - 1                               # = 2n‑2
    grad = corr[start - np.arange(n)]

    # double contribution on the diagonal entries (k = 2·i)
    diag_idx = 2 * np.arange(n)
    mask = diag_idx < len(w)
    grad[mask] += w[diag_idx[mask]] * a[mask]

    return grad


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def smooth_optimize(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 20000,
    eta0: float = 0.5,
    alpha_start: float = 1.0,
    alpha_end: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smoothed objective.
    Returns the best vector (on the simplex) and its score.
    """
    a = a_init.astype(float).copy()
    a /= np.sum(a)                     # enforce Σa = 1
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ramp α linearly from start to end
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / max_iter)
        grad = smooth_gradient(a, alpha)

        # protect against extreme values in the exponential
        grad = np.clip(grad, -100.0, 100.0)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # tiny random perturbation occasionally (helps escape plateaus)
        if it % 500 == 0:
            a += np.random.randn(n) * 5e-5
            a = np.clip(a, 0.0, None)

        # renormalise to the simplex
        s = np.sum(a)
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional progress dump
        if it % 2000 == 0:
            print(f"    smooth iter={it:,d}  α={alpha:.2f}  score={cur_score:.6f}")

    return best_a, best_score


def max_convolution_and_index(a: np.ndarray) -> tuple[float, int]:
    """Return (max value, argmax) of full self‑convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the max convolution entry (index k) w.r.t. a.
    Implements the rule described in the problem statement.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def subgradient_optimize(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 50000,
    eta0: float = 0.8,
    noise_every: int = 800,
    noise_scale: float = 5e-5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Mirrors the earlier baseline implementation but is kept modular.
    """
    a = a_init.astype(float).copy()
    a = project_to_simplex(a)          # enforce simplex
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        max_val, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d}  score={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score

def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like density discretisation (ends are heavier)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def upscale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, preserving total mass."""
    old_n = len(seq)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    # re‑normalise to sum = 1 (the optimizer works on the simplex)
    new_seq = new_seq / np.sum(new_seq)
    return new_seq

        fallback = [float(x) for x in height_sequence_1]   # may be undefined
        val = evaluate_sequence(fallback)
        print(f"Fallback score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Start with a modest size and upscale progressively.
    # ------------------------------------------------------------------
    cur_seq = arcsine_initial(128)   # sum = 1 (probability simplex)

    # Desired lengths – powers of two give fast FFTs.
    target_lengths = [256, 512, 1024, 2048, 4096]

    for n in target_lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n=== Optimising length {n} (time left {deadline - time.time():.1f}s) ===")
        # Upsample the current best shape to the new size.
        cur_seq = upscale_sequence(cur_seq, n)

        # ------------------------------------------------------------------
        # Phase 1 – smooth log‑sum‑exp optimisation.
        # ------------------------------------------------------------------
        max_iter_smooth = 20000
        seq_smooth, score_smooth = smooth_optimize(
            a_init=cur_seq,
            deadline=deadline,
            max_iter=max_iter_smooth,
            eta0=0.5,
            alpha_start=1.0,
            alpha_end=30.0,
        )
        print(f"  after smooth phase: score = {score_smooth:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – projected sub‑gradient refinement (few thousand iters).
        # ------------------------------------------------------------------
        max_iter_sub = 20000
        seq_refined, score_refined = subgradient_optimize(
            a_init=seq_smooth,
            deadline=deadline,
            max_iter=max_iter_sub,
            eta0=0.7,
            noise_every=500,
            noise_scale=5e-5,
        )
        print(f"  after sub‑gradient phase: best score = {score_refined:.6f}")

        # Keep the globally best sequence.
        if score_refined < best_val:
            best_val = score_refined
            best_seq = seq_refined.tolist()
            print(f"  *** New global best: {best_val:.6f} at n={n} ***")

        # Use the refined vector as the seed for the next (larger) length.
        cur_seq = seq_refined.copy()

    # ------------------------------------------------------------------
    # Final safety net – guarantee a non‑empty answer.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
