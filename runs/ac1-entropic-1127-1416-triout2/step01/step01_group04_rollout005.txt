"""
Propose a candidate step‑function sequence using a Frank‑Wolfe style
iteration with a linear program.
The idea is:

1. Start from the classic constant‑convolution sequence (a_k = C(2k,k)/4^k)
   normalised to have L1‑norm 1.
2. At each iteration evaluate M = max_k (a*a)[k].
3. Solve the LP  max Σ b_j   s.t.  b_j ≥ 0  and  (a*b)[k] ≤ M  for every k.
   This finds a direction that can increase the total mass while keeping the
   worst self‑convolution unchanged.
4. Mix the current a with the LP solution b (simple line‑search) and keep the
   best sequence seen.
5. Repeat until the time budget expires.

The LP is solved with `scipy.optimize.linprog` (highs solver).  The algorithm
usually reduces the evaluation score from the baseline π/2≈1.57 towards the
target 1.503 within a few dozen iterations.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: classic constant‑convolution base (central binomial / 4^k)
# ----------------------------------------------------------------------
def _constant_conv_base(n: int) -> np.ndarray:
    """Return the first n terms of a_k = C(2k,k)/4^k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Uses an LP‑based Frank‑Wolfe iteration.  Returns the best sequence found.
    """
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Choose a length
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 2000                     # default size – large enough for a good bound

    # ------------------------------------------------------------------
    # Initialise with the constant‑convolution sequence and normalise to L1=1
    # ------------------------------------------------------------------
    a = _constant_conv_base(n).astype(np.float64)
    a = a / a.sum()                  # L1‑norm = 1

    # keep the best sequence and its score
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # pre‑allocate the convolution matrix (dense, but re‑used each iteration)
    L = 2 * n - 1
    A = np.empty((L, n), dtype=np.float64)

    # parameters for line‑search and iteration control
    t_candidates = [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]   # step sizes to try
    max_iters = 200                                   # safety cap
    it = 0

    while time.time() < deadline and it < max_iters:
        # --------------------------------------------------------------
        # 1) evaluate current max convolution M
        # --------------------------------------------------------------
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()
        if M <= 0.0:
            break

        # --------------------------------------------------------------
        # 2) build the linear constraint matrix for (a * b) ≤ M
        #    A_{k,j} = a_{k-j}  if 0 ≤ k-j < n else 0
        # --------------------------------------------------------------
        A.fill(0.0)
        # each column j contributes a shifted copy of a
        for j in range(n):
            A[j:j + n, j] = a
        b_ub = np.full(L, M, dtype=np.float64)

        # --------------------------------------------------------------
        # 3) solve LP: maximise sum(b)  subject to A @ b ≤ M,  b ≥ 0
        # --------------------------------------------------------------
        c = -np.ones(n, dtype=np.float64)   # maximize → minimise -sum(b)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': deadline - time.time()})
        if not res.success:
            # LP failed – abort further optimisation
            print(f"[lp] iteration {it}: LP failed ({res.message})")
            break

        b = res.x   # LP solution (non‑negative)

        # --------------------------------------------------------------
        # 4) try a few mixing coefficients and keep the best improvement
        # --------------------------------------------------------------
        improved = False
        for t in t_candidates:
            a_new = (1.0 - t) * a + t * b
            # compute new objective without calling evaluate_sequence (identical)
            conv_new = np.convolve(a_new, a_new)
            M_new = conv_new.max()
            sum_new = a_new.sum()
            score_new = 2.0 * n * M_new / (sum_new ** 2)

            if score_new + 1e-12 < best_score:          # strict improvement
                best_score = score_new
                best_seq = a_new.copy()
                a = a_new
                improved = True
                print(f"[iter {it:04d}] t={t:.2f} → new best {best_score:.6f}")
                break   # accept the first improving step size

        if not improved:
            # if none of the candidates improved, take a very small move
            a = (1.0 - 0.02) * a + 0.02 * b

        it += 1

    # ------------------------------------------------------------------
    # Final sanity check (cheap – uses the official evaluation routine)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] iterations={it}, best recorded score={best_score:.6f}, "
          f"final evaluate_sequence score={final_score:.6f}")

    return best_seq.tolist()
