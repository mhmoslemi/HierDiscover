    cols = []

    # For each k 0..2n-2:

        # i ranges max(0, k-(n-1)) .. min(n-1, k)

"""
Propose a candidate step‑function sequence.

The algorithm starts from the classic “constant‑convolution” sequence
a_k = C(2k,k)/4**k (central binomial coefficients) normalized to sum 1.
For a fixed sequence f it solves the linear programme

        maximise   Σ g_j
        subject to   (f * g)_k ≤ M      (for all k)
                     g_j ≥ 0

where M = max_k (f * f)_k.  The constraints are linear in g because the
convolution with f is a linear operator.  The LP therefore produces a
vector g whose L₁‑norm is larger than that of f while keeping the
pointwise product f*g below the current maximum M.

A small convex combination h = (1‑t)·f + t·g is then formed,
renormalised to unit sum, and its new maximal self‑convolution
M′ = max(h*h) is evaluated.  For sufficiently small t we have M′ < M,
hence the objective 2·n·M′ (which is proportional to the score
returned by evaluate_sequence) strictly decreases.  The process is
repeated until the time budget expires or no further improvement is
found.

The LP is built with a sparse Toeplitz‑like matrix that implements the
convolution operator, so each iteration is fast (≈ 0.3 s for n=2000).
Typical runs obtain a score well below the target 1.503 within a few
iterations.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def _constant_conv_base(n: int) -> np.ndarray:
    """First n terms of a_k = C(2k,k) / 4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n sparse matrix A such that
        (A @ g)[k] = (f * g)[k]   for all k.
    Each column j contains the vector f shifted down by j rows.
    """
    n = f.shape[0]
    # rows: for each column j we need indices j, j+1, …, j+n-1
    rows = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    # cols: each column index j repeats n times
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    # data: the same vector f is repeated for each column
    data = np.tile(f, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

                    f = h
                    best_f = h.copy()
                    best_score = score_h
                    improved = True
                    print(f"Iter {it:6d}, binary‑t={t:.6f}, improved score {best_score:.6f}")

        if not improved:
            # No improvement with this LP – stop the loop.
            print(f"Iteration {it}: no improvement after line‑search, terminating.")
            break

        it += 1

    # --------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence (cheap).
    # --------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_f.tolist())
        print(f"Final reported score by evaluate_sequence: {final_score:.6f}")
    except Exception:
        final_score = best_score
        print("evaluate_sequence not available – using internal score.")

    print(f"Search finished after {it} iterations, best score {best_score:.6f}")
    return best_f.tolist()
