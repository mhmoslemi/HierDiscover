def improve_sequence(seq, max_iters=5):
    n = len(seq)
    a = np.array(seq, dtype=float)
    # Normalize sum to 1 for convenience
    total = a.sum()
    if total <= 0:
        return seq
    a = a / total
    best = a.copy()
    best_score = _score(best.tolist())
    for iter_idx in range(max_iters):
        # compute convolution max
        conv = np.convolve(a, a)
        M = conv.max()
        # Build A matrix
        n = len(a)
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=float)
        for i in range(n):
            A[i:i+n, i] = a
        # Solve LP: max sum(g) s.t. A@g <= M
        # Use scipy.optimize.linprog
        from scipy.optimize import linprog
        c = -np.ones(n)  # minimize -sum(g) -> maximize sum(g)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        # Rescale g to have same total mass as a
        if sum_g > 0:
            g = g / sum_g * a.sum()
        # Convex combination
        t = 0.5  # mixing ratio
        a_new = (1 - t) * a + t * g
        # Optionally renormalize to sum 1 (since mixing kept sum constant)
        # but for safety:
        a_new = a_new / a_new.sum()
        # Compute new score
        sc = _score(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best = a_new.copy()
            print(f"Iter {iter_idx}: improved to {best_score:.6f}")
        a = a_new
    return best.tolist()

def conv_max(a):
    b = np.convolve(a, a)

def improve(a, max_iter=10, t=0.5):
    a = np.array(a, dtype=float)
    sum_a = a.sum()
    if sum_a <= 0:
        return a.tolist()
    # we don't need to scale but for stability we can let sum=1
    a = a / sum_a
    best_score = _score(a.tolist())
    best_a = a.copy()
    for it in range(max_iter):
        # compute M
        conv = np.convolve(a, a)
        M = conv.max()
        n = len(a)
        rows = 2*n - 1

        # Build matrix A
        # Using efficient approach:
        # A = np.zeros((rows, n))
        # for i in range(n):
        #    A[i:i+n, i] = a
        # This can be done with broadcasting perhaps.
        A = np.zeros((rows, n), dtype=float)
        for i in range(n):
            A[i:i+n, i] = a

        # Solve LP: maximize sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        if g.sum() == 0:
            break

        # Scale g to have same sum as a (as our sum is 1)
        g = g / g.sum() * a.sum()  # sum remains same (1)
        # Mix
        a = (1 - t) * a + t * g
        # Clip small negative
        a = np.maximum(a, 0.0)
        # Renormalize sum to 1 again
        a = a / a.sum()
        sc = _score(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        # Early exit if objective low enough
        if best_score < 1.2:
            break
    return best_a.tolist()

def propose_candidate(seed=0, budget_s=1000):
    import time
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = math.inf

    # Try the global reference if exists
    if 'height_sequence_1' in globals():
        init = _normalize(list(height_sequence_1))
        sc = _score(init)
        if sc < best_score:
            best_score, best_seq = sc, init

    # Try deterministic seeds initially
    length_options = [300, 400, 500, 600, 800]

    # initial simple sequences random Dirichlet
    while time.time() < deadline:
        # choose length randomly
        n = random.choice(length_options)

        # generate random positive Dirichlet
        raw = np.random.rand(n) + 1e-3  # ensure >0
        seq = raw / raw.sum()  # sum to 1
        # Scale max to 1 (optional)
        # We can keep as-is; max may be <1, scaling not needed.

        # Run local improvement
        seq_improved = improve(seq, max_iter=8, t=0.3)

        sc = _score(seq_improved)
        if sc < best_score:
            best_score, best_seq = sc, seq_improved
            print(f"New best after LP improve: n={n} score={best_score:.6f}")

        # Also try hill climbing small perturb of current best
        if best_seq is not None and random.random() < 0.2:
            cand = np.array(best_seq)
            n = len(cand)
            # randomly perturb few entries
            for _ in range(3):
                i = random.randint(0, n - 1)
                delta = random.uniform(-0.1, 0.1)
                cand[i] = max(0.0, cand[i] + delta)
            # renorm sum to 1
            cand = np.maximum(cand, 0.0)
            if cand.sum() > 0:
                cand = cand / cand.sum()
                sc = _score(cand.tolist())
                if sc < best_score:
                    best_score, best_seq = sc, cand.tolist()
                    print(f"Hill perturb gave improvement: {best_score:.6f}")

    # final fallback: uniform
    if best_seq is None:
        n = 500
        best_seq = [1.0] * n
        best_score = _score(best_seq)

    print(f"Finished with best score {best_score:.6f} length={len(best_seq)}")
    return best_seq

"""
LP‑based refinement of a step‑function.

The algorithm starts from a random (or the supplied) non‑negative sequence,
normalises it to unit ℓ₁‐norm and then repeatedly solves a linear program

    max  Σ g_i
    s.t. (f * g)[k] ≤ M  for all k = 0 … 2n‑2
         g_i ≥ 0

where `f` is the current sequence and `M = max (f * f)`.  
Because the constraints are linear in `g` the LP can be solved with
SciPy’s high‑performance “highs” backend.  The solution `g` is scaled to
have the same ℓ₁‑norm as `f` and mixed with `f` (convex combination).  This
procedure never increases the maximal self‑convolution and quickly drives
the sequence towards a shape whose convolution is almost flat, i.e. a
small value of the evaluation function

    2·n·max(f*f) / (Σ f)² .

Several random restarts with different lengths are performed until the
time budget (≈ 1000 s) expires.  The best sequence found is returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (same as in the reference solution)
# ----------------------------------------------------------------------
def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)   # evaluate_sequence is provided by the harness
    except Exception:               # pragma: no cover
        return math.inf


def _normalize(seq):
    """Scale so that max(seq) == 1 (no‑op if all zeros)."""
    arr = np.asarray(seq, dtype=float)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def _improve_seq(seq,
                 max_iters: int = 12,
                 mix_factor: float = 0.35,
                 time_deadline: float = None) -> list[float]:
    """
    Refine ``seq`` by repeatedly solving the LP described in the docstring.
    The ℓ₁‑norm of the sequence is kept equal to 1 (scale‑invariant for the
    evaluation function).  The loop stops after ``max_iters`` iterations,
    or earlier if the time budget runs out.
    """
    a = np.asarray(seq, dtype=float)

    # keep a unit ℓ₁‑norm – this does not change the objective value
    s = a.sum()
    if s <= 0:
        return seq
    a = a / s

    n = a.shape[0]
    best = a.copy()
    best_score = _score(best.tolist())

    # pre‑allocate the right‑hand side of the LP (M will be updated each loop)
    rows = 2 * n - 1
    b_ub_template = np.empty(rows, dtype=float)

    # build the constraint matrix A once – it only depends on the current a
    # A[k, j] = a[k‑j]  (zero when the index is out of range)
    # we rebuild it each iteration because a changes.
    while max_iters > 0:
        max_iters -= 1
        if time_deadline is not None and time.time() > time_deadline:
            break

        # ---- current maximal self‑convolution ----
        conv = np.convolve(a, a)
        M = conv.max()
        b_ub_template[:] = M

        # ---- build matrix A (dense, but small enough for n ≤ 800) ----
        A = np.zeros((rows, n), dtype=float)
        for j in range(n):
            A[j:j + n, j] = a   # each column is a shifted copy of a

        # ---- linear programme: maximise Σ g_i  (i.e. minimise –Σ g_i) ----
        from scipy.optimize import linprog
        c = -np.ones(n)               # minimise –sum(g)
        bounds = [(0.0, None)] * n    # g_i ≥ 0
        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub_template,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})

        if not res.success:
            # LP failed – abort further refinement for this seed
            break

        g = res.x
        g_sum = g.sum()
        if g_sum <= 0:
            break

        # ---- keep ℓ₁‑norm fixed (scale g to the same total mass as a) ----
        g = g * (a.sum() / g_sum)

        # ---- convex mixture (t ∈ (0,1) ) ----
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = np.maximum(a, 0.0)

        # re‑normalise the ℓ₁‑norm (numerical drift)
        a = a / a.sum()

        # ---- evaluate ----
        sc = _score(a.tolist())
        if sc < best_score:
            best_score = sc
            best = a.copy()
            # a small improvement is printed for monitoring
            print(f"  LP‑refine iteration improvement → {best_score:.6f}")

        # Early exit when we are already close to the theoretical lower bound
        # The lower bound is ≈ 1 (2·n·(1/(2n‑1))).
        if best_score <= 1.10:
            break

    return best.tolist()


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The routine spends roughly ``budget_s`` seconds
    searching; it mixes random Dirichlet seeds with the LP‑based refinement
    described above.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.35   # tiny safety margin

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 0) start from the globally supplied construction if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = _normalize(list(height_sequence_1))
            sc = _score(init)
            if sc < best_score:
                best_seq, best_score = init, sc
                print(f"Initial reference score: {best_score:.6f}")
        except Exception:               # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1) random Dirichlet seeds + LP refinement
    # ------------------------------------------------------------------
    # a handful of lengths – too large makes the LP expensive,
    # too small limits the attainable bound.
    lengths = [260, 340, 420, 520, 640, 800]

    while time.time() < deadline:
        n = random.choice(lengths)

        # ---- generate a random positive vector (Dirichlet) ----
        raw = np.random.rand(n) + 1e-8           # ensure strictly positive
        seq = (raw / raw.sum()).tolist()        # unit ℓ₁‑norm

        # ---- optional small random perturbation before LP ----
        if random.random() < 0.15:
            # change a few entries multiplicatively
            for _ in range(3):
                idx = random.randrange(n)
                factor = 1.0 + random.gauss(0.0, 0.06)
                seq[idx] = max(0.0, seq[idx] * factor)
            # renormalise
            s = sum(seq)
            if s > 0:
                seq = [x / s for x in seq]

        # ---- improve via the LP‑based scheme --------------------
        improve_deadline = deadline - 0.05   # a little head‑room
        seq_improved = _improve_seq(seq,
                                    max_iters=14,
                                    mix_factor=0.33,
                                    time_deadline=improve_deadline)

        sc = _score(seq_improved)
        if sc < best_score:
            best_score, best_seq = sc, seq_improved
            print(f"New best (n={n}) → {best_score:.6f}")

        # ---- occasional hill‑climb step on the best solution so far ----
        if best_seq is not None and random.random() < 0.12:
            cand = np.array(best_seq, copy=True)
            m = random.randint(1, min(5, len(cand)))
            idxs = random.sample(range(len(cand)), m)
            for i in idxs:
                cand[i] = max(0.0, cand[i] * (1.0 + random.gauss(0.0, 0.07)))
            cand = cand / cand.sum()            # keep ℓ₁‑norm = 1
            sc = _score(cand.tolist())
            if sc < best_score:
                best_score, best_seq = sc, cand.tolist()
                print(f"Post‑hill‑climb improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        best_seq = [1.0] * n
        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    # The evaluation function is scale‑invariant, but we keep the maximum entry ≤1
    # for safety with the built‑in checks.
    return _normalize(best_seq)
