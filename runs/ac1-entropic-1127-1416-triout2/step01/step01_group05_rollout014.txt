def simplex_projection(v):
    # v: numpy array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix(f):
    n = len(f)
    # Number of constraints = 2*n - 1
    rows = []
    cols = []
    data = []
    for j in range(n):
        # For each column j, define rows from j to j + n - 1 (inclusive) where k row index.
        # The coefficient for row k = f[k - j]
        start = j
        end = j + n  # exclusive
        # fill rows start to end-1.
        # We'll generate arrays of length n for row indices.
        rows.extend(range(start, end))
        cols.extend([j] * n)
        # corresponding f values: f[0..n-1]
        data.extend(f.tolist())
    # create sparse matrix
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

"""
Propose a step‑function with a lower evaluation score than the previous best (~1.5754).

Algorithm
---------
1. Initialise a probability distribution `f` on `n` points using the
   classic inverse–square‑root shape (a_i ∝ (i+1)^{-0.5}) and normalise it
   to sum = 1.  With this normalisation the evaluation reduces to
   `E = 2·n·max(conv(f,f))`.

2. Repeatedly improve `f` by an *alternating linear‑program* step:
   * For the current `f` compute `M = max(conv(f,f))`.
   * Solve the LP  

         maximise    Σ b_j
         subject to  b_j ≥ 0
                     (f * b)_k ≤ M   for all k = 0 … 2n‑2

     (`f * b` is a linear function of `b`).  
     The LP gives a vector `b` with a larger ℓ₁‑norm than `f` while still
     respecting the convolution bound.

   * Rescale `b` to unit sum (the feasible region is a cone, scaling keeps the
     bound satisfied) → `b̂ = b / Σb`.
   * Mix the old and the new distribution:
         f ← (1‑t)·f + t·b̂      (t = 0.5 by default)
     Renormalise to unit sum and keep the best‑scoring `f` seen so far.

   The mixing guarantees that the maximal self‑convolution never increases,
   while the LP tends to shift mass towards indices where `f` is small,
   lowering the peak.

3. After a number of LP iterations (or when the time budget is almost exhausted)
   perform a short random‑local‑move refinement exactly as in the previous
   script – this helps escape tiny local plateaus.

The whole procedure runs within the 1000 s budget, prints progress, and finally
returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# evaluate_sequence is assumed to be available in the execution environment.


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i ≥ 0, Σ x_i = 1 }.
    """
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find the rho index
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that
        (A @ b)[k] = (f * b)[k]
    for all k.  The matrix entries are simply the coefficients of the
    discrete convolution, i.e. each column i contains the vector f shifted by i.
    """
    n = f.size
    rows = []
    cols = []
    data = []

    for i in range(n):
        # rows i … i+n‑1 receive f[0 … n‑1] in column i
        rows.extend(range(i, i + n))
        cols.extend([i] * n)
        data.extend(f.tolist())

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=np.float64)
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the evaluation function defined in ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a moderate length (fast LP) – 1500 works well in practice.
    #    Larger n gives a slightly better asymptotic constant but the LP
    #    becomes much slower.
    # ------------------------------------------------------------------
    n = 1500
    # initialise with the classic inverse‑square‑root shape (sum = 1)
    raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)   # a_i ∝ (i+1)^{-0.5}
    f = raw / raw.sum()                                   # probability simplex

    # keep the best sequence found so far
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f'Initial score (n={n}): {best_score:.6f}')

    # parameters for the alternating optimisation
    t = 0.5                      # mixing weight
    iteration = 0
    max_iter = 30               # safety bound; we stop earlier if time runs out

    while time.time() < deadline and iteration < max_iter:
        # ----- current statistics -------------------------------------------------
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()
        cur_score = 2.0 * n * M                 # because Σf = 1
        # store improvement if any
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f'Iter {iteration:02d}: new best score = {best_score:.6f}')

        # ----- build linear constraints (f * b ≤ M) -------------------------------
        A = _build_conv_matrix(f)                # shape (2n‑1, n), sparse
        b_ub = np.full(2 * n - 1, M)             # RHS of all ≤ constraints

        # ----- linear programme: maximise Σb_j  (=> minimise -Σb_j) ---------------
        c = -np.ones(n)                          # linprog minimises cᵀx
        bounds = [(0, None)] * n                 # b_j ≥ 0

        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f'LP failed at iteration {iteration}: {res.message}')
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            print('LP produced zero vector, aborting.')
            break

        # ----- rescale to keep sum = 1 -------------------------------------------
        b_hat = b / sum_b

        # ----- convex combination -------------------------------------------------
        f_new = (1.0 - t) * f + t * b_hat
        # enforce the simplex constraints (numerical drift)
        f_new = _simplex_projection(f_new)

        # ----- evaluate the new candidate -----------------------------------------
        cand_score = evaluate_sequence(f_new.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_f = f_new.copy()
            print(f'Iter {iteration:02d}: improvement after mixing = {cand_score:.6f}')
        else:
            # if the mixing did not improve, slightly reduce the step size
            t = max(0.1, t * 0.9)

        f = f_new
        iteration += 1

        # early exit if we already beat the target
        if best_score < 1.503:
            print('Target reached – exiting early.')
            break

    # ------------------------------------------------------------------
    # 2. Fine‑grained random local optimisation (identical to the previous script)
    # ------------------------------------------------------------------
    curr = best_f.copy()
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_move = 0.01 * curr[i]               # move at most 1 % of source mass
        if max_move <= 0.0:
            continue
        delta = np.random.uniform(-max_move, max_move)
        if curr[i] - delta < 0.0:
            continue

        cand = curr.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_f = cand.tolist()
            curr = cand
            print(f'Random tweak: new best score = {cand_score:.6f}')

        if best_score < 1.503:
            break

    print(f'Final best score: {best_score:.6f} (length {len(best_f)})')
    return best_f
