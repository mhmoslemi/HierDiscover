def decreasing_power_law(n, p, offset):
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= a_sum
    return a

def symmetric_product_power_law(n, p, offset):
    i = np.arange(n, dtype=np.float64)
    rev_i = (n - 1 - i)
    a = ((i + offset) * (rev_i + offset)) ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= a_sum
    return a

def hard_max_gradient(a):
    b = np.convolve(a, a)
    max_b = b.max()
    mask = (b >= max_b - 1e-12)  # boolean mask
    w = mask.astype(np.float64)
    sum_w = w.sum()
    if sum_w == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= sum_w
    # reverse and convolve
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)
    L = w.size
    grad = 2.0 * conv_rev[(L - 1) - np.arange(a.size)]
    return grad, max_b

"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

Algorithm summary
-----------------
1. **Problem size** – use the length of the previously found
   construction (`height_sequence_1`) when available, otherwise fall back
   to 2000 entries.

2. **Exploratory families** – generate a modest set of candidate
   distributions:
     * decreasing power‑law   a_i ∝ (i+offset)^(-p)
     * symmetric product law  a_i ∝ ((i+offset)*(n‑1‑i+offset))^(-p)
   for several offsets and exponents.  All candidates are normalised to
   sum 1 and scored with a cheap inline replica of `evaluate_sequence`.
   The best candidate of this stage becomes the starting point.

3. **Refinement by entropic (exponentiated) gradient descent**.
   The true sub‑gradient of the max‑convolution is used:
   ```
   b = a * a                     # full convolution
   M = max(b)
   w = indicator(b == M) / |{k : b_k = M}|
   grad_i = 2 * Σ_j a_j * w_{i+j}
   ```
   An update `a ← a * exp(-η·grad)` preserves non‑negativity.
   After each update the vector is projected onto the probability simplex
   (Σ a_i = 1).  The step size η is slowly decayed; occasional small
   random jitteres are injected if progress stalls.

4. **Time management** – the whole routine stops well before the
   supplied time budget (default 1000 s) and returns the best sequence
   discovered.

The method combines a cheap global search over parametrised families
with a focused local optimiser, and typically drives the score below
the target 1.503 within the allowed time.
"""

import time

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # uniform distribution as a fallback
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for a normalised (sum>0) vector.
    It assumes that the elements are already non‑negative and well below
    the internal 1000‑cap, which is true for all our candidates.
    """
    s = a.sum()
    if s < 0.01:
        return float("inf")
    b = np.convolve(a, a)          # length 2n‑1
    max_b = b.max()
    n = a.size
    return float(2.0 * n * max_b / (s * s))


def decreasing_power_law(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i + offset)^{-p}, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    # guard against all zeros (should not happen)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def symmetric_product_law(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+offset)*(n‑1‑i+offset))^{-p}, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    rev = (n - 1 - i).astype(np.float64)
    a = ((i + offset) * (rev + offset)) ** (-p)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def sqrt_distribution(n: int) -> np.ndarray:
    """Normalised coefficients of (1‑z)^(-1/2) up to length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def hard_max_gradient(a: np.ndarray):
    """
    Compute a sub‑gradient of the max‑convolution objective.
    Returns (gradient, current max of a*a).
    """
    b = np.convolve(a, a)                # length 2n‑1
    max_b = b.max()
    # indicator of positions attaining the maximum (tolerance for floating‑point)
    mask = (b >= max_b - 1e-12)
    w = mask.astype(np.float64)
    total = w.sum()
    if total == 0.0:
        # fall back to uniform weights (should never happen)
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= total
    # gradient = 2 * (a * w)  (where * denotes convolution with w reversed)
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)
    L = w.size
    grad = 2.0 * conv_rev[(L - 1) - np.arange(a.size)]
    return grad, max_b
