def evaluate_fast(arr):
    n = len(arr)
    # ensure arr is np.ndarray float
    a = np.asarray(arr, dtype=float)
    # clip to [0, 1000] before evaluation to match evaluate_sequence
    a = np.clip(a, 0.0, 1000.0)
    target_sum = np.sqrt(2.0*n)
    # we may need to keep sum fixed already; but keep for safety
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    # For convolution we need length at least 2*n-1
    L = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:2*n - 1]
    max_conv = conv.max()
    score = 2.0 * n * max_conv / (sum_a * sum_a)
    return float(score)

def evaluate_fast(seq):
    # expect np.ndarray

    s = seq.sum()
    if s < 0.01:

    L = 1 << ((2 * n - 1).bit_length())

    max_conv = conv.max()
    return float(2.0 * n * max_conv / (s * s))

"""
Hybrid simulated‐annealing + gradient search for a low‑autocorrelation
step function.

The algorithm works in three stages:

1. **Initialisation** – If a previous construction is available as the
   global variable ``height_sequence_1`` it is used (projected onto the
   simplex Σa_i = √(2n)).  Otherwise a cheap geometric grid search is
   performed to obtain a reasonable seed.

2. **Search loop** – While the allotted time remains, one of three
   neighbourhood moves is selected at random:

   * **pairwise transfer** – move a small amount of mass from index i
     to index j (keeps the ℓ₁‑norm exactly constant).
   * **gradient step** – a sub‑gradient of the peak autocorrelation
     (the index where the convolution attains its maximum) is taken and
     a step opposite to it is performed.
   * **global jitter** – add a Gaussian perturbation to all coordinates
     and re‑project onto the simplex.

   The candidate is evaluated with a fast FFT‑based routine that mimics
   ``evaluate_sequence``.  It is accepted if it improves the score or
   with probability ``exp(-(Δ)/T)`` where the temperature ``T`` decays
   exponentially during the run (simulated‑annealing style).

3. **Result** – The best sequence seen so far is returned as a plain
   Python list.  A final call to the official ``evaluate_sequence`` is
   performed for logging only.

All numbers are clipped to the interval [0, 1000] before evaluation,
the sum is kept at the required value ``√(2 n)`` by projection onto the
simplex, and the implementation uses only NumPy (no external files or
network access).
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector `v` onto the simplex {x ≥ 0, Σx = s}.

    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _fast_score(seq: np.ndarray, conv_len: int) -> float:
    """
    Compute the evaluation metric using FFT.
    The sequence is assumed to be already clipped to [0, 1000].
    """
    n = seq.size
    total = seq.sum()
    if total < 0.01:
        return float("inf")
    # Linear convolution via FFT (length `conv_len` is a power of two ≥ 2*n‑1)
    A = np.fft.rfft(seq, n=conv_len)
    conv = np.fft.irfft(A * A, n=conv_len)[: 2 * n - 1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv / (total * total))


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided evaluation function.  The routine runs for at most
    ``budget_s`` seconds and returns the best list of floats found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------- #
    # 1. Determine problem size and target ℓ₁‑norm.
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=float)
        n = init.size
    else:
        # fallback: a moderate length that still fits comfortably in the budget
        n = 3000
        init = np.full(n, 1.0, dtype=float)   # dummy, will be replaced below

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------- #
    # 2. Build an initial feasible point.
    if init.size != n or np.any(init < 0):
        # perform a cheap geometric search (identical to the previous code)
        best_val = float("inf")
        best_seq = None
        lambdas = np.logspace(-5, -1, 120)
        for lam in lambdas:
            r = 1.0 - lam
            r_n = r ** n
            denom = 1.0 - r_n
            if denom <= 0.0:
                continue
            C = target_sum * lam / denom
            seq = C * (r ** np.arange(n))
            seq = np.clip(seq, 0.0, 1000.0)
            val = _fast_score(seq, 1 << ((2 * n - 1).bit_length()))
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
        # power‑law seed as a backup
        idx = np.arange(1, n + 1, dtype=float)
        a_pow = 1.0 / np.sqrt(idx)
        C_pow = target_sum / a_pow.sum()
        a_pow = C_pow * a_pow
        a_pow = np.clip(a_pow, 0.0, 1000.0)
        val_pow = _fast_score(a_pow, 1 << ((2 * n - 1).bit_length()))
        if val_pow < best_val:
            best_seq = a_pow.copy()
        # final fallback: uniform
        if best_seq is None:
            best_seq = np.full(n, target_sum / n)
        init = best_seq.copy()

    # ensure feasibility (clip, then project)
    init = np.clip(init, 0.0, 1000.0)
    init = projection_simplex(init, target_sum)

    # ------------------------------------------------------------------- #
    # 3. Pre‑compute constants.
    conv_len = 1 << ((2 * n - 1).bit_length())   # power‑of‑two ≥ 2n‑1
    best_seq = init.copy()
    best_score = _fast_score(best_seq, conv_len)

    # ------------------------------------------------------------------- #
    # 4. Hyper‑parameters for the hybrid search.
    #    These have been tuned empirically for the 1000 s budget.
    pair_frac = 0.05            # typical fraction of a[i] moved in a pairwise step
    grad_frac = 0.5             # step size factor for gradient moves
    jitter_frac = 0.3           # scale of global jitter (relative to mean value)
    temp_start = best_score * 0.3
    temp_end = 1e-6
    it = 0
    last_improve = 0

    # ------------------------------------------------------------------- #
    # 5. Main simulated‑annealing / gradient loop.
    while time.time() < deadline:
        # ----- evaluate current point (conv, score, argmax) -----------------
        A = np.fft.rfft(best_seq, n=conv_len)
        conv_full = np.fft.irfft(A * A, n=conv_len)[: 2 * n - 1]
        cur_score = 2.0 * n * conv_full.max() / (best_seq.sum() ** 2)
        k_max = int(conv_full.argmax())

        # ----- temperature schedule (exponential decay) -------------------
        elapsed = (time.time() - start_time) / budget_s
        temperature = max(
            temp_start * ((temp_end / temp_start) ** elapsed), temp_end
        )

        # ----- decide which move to perform --------------------------------
        move_type = np.random.rand()
        if move_type < 0.80:                     # pairwise transfer (most common)
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = pair_frac * np.random.rand() * best_seq[i]
            # ensure we stay in the feasible box
            if delta > best_seq[i]:
                delta = best_seq[i]
            new_i = best_seq[i] - delta
            new_j = best_seq[j] + delta
            if new_j > 1000.0:
                excess = new_j - 1000.0
                new_j = 1000.0
                new_i += excess   # give excess back to i (still keeps sum)
                if new_i < 0.0:
                    new_i = 0.0
            cand = best_seq.copy()
            cand[i] = new_i
            cand[j] = new_j
        elif move_type < 0.95:                   # gradient step
            # sub‑gradient of the max‑convolution at k_max:
            idx = np.arange(n)
            src = k_max - idx
            mask = (src >= 0) & (src < n)
            grad = np.zeros_like(best_seq)
            grad[mask] = 2.0 * best_seq[src[mask]]
            gnorm = np.linalg.norm(grad)
            if gnorm == 0:
                cand = best_seq.copy()
            else:
                step = grad_frac * target_sum / n
                cand = best_seq - step * (grad / gnorm)
                cand = np.clip(cand, 0.0, 1000.0)
                cand = projection_simplex(cand, target_sum)
        else:                                    # global jitter (exploratory)
            jitter = jitter_frac * target_sum / n * np.random.randn(n)
            cand = best_seq + jitter
            cand = np.clip(cand, 0.0, 1000.0)
            cand = projection_simplex(cand, target_sum)

        # ----- evaluate candidate -------------------------------------------
        cand_score = _fast_score(cand, conv_len)

        # ----- Metropolis acceptance ----------------------------------------
        delta = cand_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # avoid overflow in exp()
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            best_seq = cand
            cur_score = cand_score
            # update best ever seen
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(
                    f"[{time.time() - start_time:6.1f}s] iter {it:7d} → new best {best_score:.6f}"
                )
                last_improve = it

        # occasional aggressive restart if we have been stagnant for a while
        if it - last_improve > 200000:
            # build a fresh random point on the simplex (still respects bounds)
            rand_dir = np.random.rand(n)
            rand_dir = np.clip(rand_dir, 0.0, 1000.0)
            rand_dir = projection_simplex(rand_dir, target_sum)
            cand_score = _fast_score(rand_dir, conv_len)
            if cand_score < best_score:
                best_seq = rand_dir.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:6.1f}s] iter {it:7d} → restart improvement {best_score:.6f}"
                )
                last_improve = it
            else:
                # adopt the random point anyway (allows escape)
                best_seq = rand_dir.copy()
                last_improve = it

        it += 1

    # ------------------------------------------------------------------- #
    # 6. Final sanity check using the official evaluator (optional)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Search finished after {it} iterations. Final score: {final_score:.6f}")

    return best_seq.tolist()
