def _shift_mass(seq, eps_scale=0.05):
    arr = np.array(seq)
    n = len(arr)
    i,j = random.sample(range(n), 2)
    if arr[i] > 0:
        delta = arr[i] * eps_scale * random.random()
        arr[i] = max(0.0, arr[i] - delta)
        arr[j] += delta
    # renorm
    mx = arr.max()
    if mx>0:
        arr /= mx
    return arr.tolist()
```

def propose_candidate(seed=0, budget_s=1000):
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.1
    best_seq = None
    best_score = math.inf

    # optionally init from height_sequence_1 if present
    if 'height_sequence_1' in globals():
        try:
            init = _normalize(list(height_sequence_1))
            sc = _score(init)
            best_seq, best_score = init, sc
        except Exception:
            pass

    # length options
    lengths = [800, 1200, 1600, 2000, 2600, 3200, 3800, 4400, 5000]
    families = ['exp', 'pow', 'lin', 'mix', 'log', 'randdec', 'symexp', 'symtri', 'symquad', 'sine', 'piecewise']

    while time.time() < deadline:
        # With certain probability attempt hill climbing on best_seq
        if best_seq is not None and random.random() < 0.2:
            # clone and perturb
            cand = np.array(best_seq, copy=True)
            n = len(cand)
            # random perturbation type
            if random.random() < 0.7:
                # multiplicative perturbation
                m = random.randint(1, min(10, n))
                idxs = random.sample(range(n), m)
                for i in idxs:
                    factor = math.exp(random.gauss(0.0, 0.05))
                    cand[i] = max(0.0, cand[i] * factor)
            else:
                # shift mass
                cand = np.array(_shift_mass(cand.tolist(), eps_scale=0.1))
            # renorm to keep max=1
            mx = cand.max()
            if mx > 0:
                cand = cand / mx
            cand = cand.tolist()
            sc = _score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand
                print(f"Hill‑climb improvement → {best_score:.6f}")
            continue

        # generate fresh candidate
        n = random.choice(lengths)
        family = random.choice(families)

        if family == 'exp':

            alpha = 10 ** log_a
            cand = _exp_seq(n, alpha)
        elif family == 'pow':
            p = random.uniform(0.1, 10.0)
            cand = _pow_seq(n, p)
        elif family == 'lin':

            cand = _lin_seq(n, beta)
        elif family == 'mix':
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            eps = random.uniform(0.0, 1.0)
            cand = _mix_seq(n, alpha, eps)
        elif family == 'log':
            steep = random.uniform(0.01, n / 2.0)
            cand = _logistic_seq(n, steep)
        elif family == 'randdec':
            cand = _randdec_seq(n)
        elif family == 'symexp':
            # symmetric exponential around centre
            alpha = random.uniform(0.001, 0.5)
            i = np.arange(n, dtype=float)
            mid = (n - 1) / 2.0
            seq = np.exp(-alpha * np.abs(i - mid))
            cand = _normalize(seq)
        elif family == 'symtri':

            i = np.arange(n, dtype=float)
            mid = (n - 1) / 2.0
            seq = np.maximum(0.0, 1.0 - np.abs(i - mid) / width)
            cand = _normalize(seq)
        elif family == 'symquad':
            width = random.uniform(0.1 * n, 0.9 * n)

        elif family == 'piecewise':

        # evaluate

        if sc < best_score:

        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq

"""Optimization of a non‑negative step‑function for the evaluation metric.

The search combines:
  * a wide collection of parametric families (exponential, power‑law,
    linear, mixtures, logistic, symmetric shapes, sine‑based and
    piecewise‑constant).
  * a simple stochastic hill‑climber / simulated‑annealing loop that
    repeatedly perturbs the current best sequence (multiplicative noise,
    mass‑shifts, or global Gaussian noise) and accepts improvements.
  * occasional random restarts from fresh parametric samples.
The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ----------------------------------------------------------------------
def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)  # evaluate_sequence is provided by the harness
    except Exception:               # pragma: no cover
        return math.inf


def _normalize(seq):
    """Scale so that max(seq) == 1 (no‑op if all zeros)."""
    arr = np.asarray(seq, dtype=float)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _exp_seq(n, alpha):
    """Exponential decay a_i = exp(-alpha * i)."""
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)))


def _pow_seq(n, p):
    """Power‑law decay a_i = (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=float)
    return _normalize(idx ** (-p))


def _lin_seq(n, beta):
    """Linear decay a_i = max(0, 1 - beta*i)."""
    return _normalize(np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float)))


def _mix_seq(n, alpha, eps):
    """Mixed exponential + constant a_i = exp(-alpha*i) + eps."""
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)) + eps)


def _logistic_seq(n, steep):
    """Logistic decay a_i = 1/(1+exp((i-mid)/steep))."""
    i = np.arange(n, dtype=float)
    mid = n / 2.0
    seq = 1.0 / (1.0 + np.exp((i - mid) / steep))
    return _normalize(seq)


def _randdec_seq(n):
    """Random decreasing sequence (sorted random numbers)."""
    arr = np.random.rand(n)
    arr.sort()
    arr = arr[::-1]                     # descending
    return _normalize(arr)


def _sym_exp_seq(n, alpha):
    """Symmetric exponential around the centre."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    seq = np.exp(-alpha * np.abs(i - mid))
    return _normalize(seq)


def _sym_tri_seq(n, width):
    """Symmetric triangular shape."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    seq = np.maximum(0.0, 1.0 - np.abs(i - mid) / width)
    return _normalize(seq)


def _sym_quad_seq(n, width):
    """Symmetric quadratic (parabolic) shape."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    seq = np.maximum(0.0, 1.0 - (np.abs(i - mid) / width) ** 2)
    return _normalize(seq)


def _sine_seq(n, power):
    """Sine‑based bump: sin(pi*(i+0.5)/n) ^ power."""
    i = np.arange(n, dtype=float)
    seq = np.sin(np.pi * (i + 0.5) / n) ** power
    return _normalize(seq)


def _piecewise_const_seq(n, segs):
    """Random piecewise‑constant non‑negative sequence."""
    cuts = sorted(random.sample(range(1, n), segs - 1))
    cuts = [0] + cuts + [n]
    seq = np.empty(n, dtype=float)
    for k in range(segs):
        start = cuts[k]
        end = cuts[k + 1]
        height = random.random()
        seq[start:end] = height
    return _normalize(seq)


def _dual_exp_seq(n, alpha, beta, w):
    """Mixture of two exponentials."""
    i = np.arange(n, dtype=float)
    seq = w * np.exp(-alpha * i) + (1.0 - w) * np.exp(-beta * i)
    return _normalize(seq)


def _pow_exp_seq(n, p, alpha):
    """Power‑law multiplied by an exponential."""
    i = np.arange(n, dtype=float)
    idx = np.arange(1, n + 1, dtype=float)
    seq = (idx ** (-p)) * np.exp(-alpha * i)
    return _normalize(seq)


# ----------------------------------------------------------------------
# Local perturbations used by the hill‑climber
# ----------------------------------------------------------------------
def _perturb_multiplicative(seq, sigma=0.08, max_change=5):
    """Multiplicative Gaussian perturbation on a few random indices."""
    arr = np.array(seq, copy=True)
    n = len(arr)
    m = random.randint(1, min(max_change, n))
    idxs = random.sample(range(n), m)
    for i in idxs:
        factor = math.exp(random.gauss(0.0, sigma))
        arr[i] = max(0.0, arr[i] * factor)
    # renormalise (scale does not affect the score)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()


def _perturb_shift_mass(seq, scale=0.1):
    """Shift a fraction of mass from one position to another."""
    arr = np.array(seq, copy=True)
    n = len(arr)
    i, j = random.sample(range(n), 2)
    if arr[i] > 0.0:
        delta = arr[i] * scale * random.random()
        arr[i] = max(0.0, arr[i] - delta)
        arr[j] += delta
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()


def _perturb_global_noise(seq, sigma=0.05):
    """Add Gaussian noise to every entry (keep non‑negative)."""
    arr = np.array(seq, copy=True)
    noise = np.random.randn(len(arr)) * sigma
    arr = arr * (1.0 + noise)
    arr = np.maximum(0.0, arr)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The routine spends roughly ``budget_s`` seconds
    searching; it mixes random sampling of a rich set of parametric families
    with a stochastic hill‑climbing/annealing refinement of the best
    candidate discovered so far.
    """
    random.seed(seed)
    np.random.seed(seed)

    # overall deadline – a small safety margin
    deadline = time.time() + budget_s - 0.1

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 0) initialise from a reference construction if the harness provides one
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = _normalize(list(height_sequence_1))
            sc = _score(init)
            if sc < best_score:
                best_seq, best_score = init, sc
                print(f"Initial reference score: {best_score:.6f}")
        except Exception:               # pragma: no cover
            pass

    # candidate lengths (we favour a few thousand entries, but also try smaller ones)
    lengths = [800, 1200, 1600, 2000, 2600, 3200, 3800, 4400, 5000]

    # all families we will draw from
    families = [
        'exp', 'pow', 'lin', 'mix', 'log', 'randdec',
        'symexp', 'symtri', 'symquad', 'sine',
        'piecewise', 'dualexp', 'powexp'
    ]

    # ------------------------------------------------------------------
    # 1) main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        #   a) occasional hill‑climb from the current best
        # ------------------------------------------------------------------
        if best_seq is not None and random.random() < 0.20:
            # pick a perturbation type
            r = random.random()
            if r < 0.5:
                cand = _perturb_multiplicative(best_seq, sigma=0.08, max_change=8)
            elif r < 0.75:
                cand = _perturb_shift_mass(best_seq, scale=0.15)
            else:
                cand = _perturb_global_noise(best_seq, sigma=0.07)

            sc = _score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand
                print(f"Hill‑climb improvement → {best_score:.6f}")
            continue

        # ------------------------------------------------------------------
        #   b) fresh candidate from a random family
        # ------------------------------------------------------------------
        n = random.choice(lengths)
        family = random.choice(families)

        if family == 'exp':
            # log‑uniform α ∈ [1e‑4, 5]
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            cand = _exp_seq(n, alpha)

        elif family == 'pow':
            p = random.uniform(0.0, 10.0)
            cand = _pow_seq(n, p)

        elif family == 'lin':
            beta = random.uniform(0.0, 4.0 / n)
            cand = _lin_seq(n, beta)

        elif family == 'mix':
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            eps = random.uniform(0.0, 1.0)
            cand = _mix_seq(n, alpha, eps)

        elif family == 'log':
            steep = random.uniform(0.01, n / 2.0)
            cand = _logistic_seq(n, steep)

        elif family == 'randdec':
            cand = _randdec_seq(n)

        elif family == 'symexp':
            alpha = random.uniform(0.001, 1.0)
            cand = _sym_exp_seq(n, alpha)

        elif family == 'symtri':
            width = random.uniform(0.1 * n, 0.9 * n)
            cand = _sym_tri_seq(n, width)

        elif family == 'symquad':
            width = random.uniform(0.1 * n, 0.9 * n)
            cand = _sym_quad_seq(n, width)

        elif family == 'sine':
            power = random.uniform(0.1, 10.0)
            cand = _sine_seq(n, power)

        elif family == 'piecewise':
            segs = random.randint(2, 8)
            cand = _piecewise_const_seq(n, segs)

        elif family == 'dualexp':
            log_a = random.uniform(-4, math.log10(5))
            log_b = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            beta = 10 ** log_b
            w = random.random()
            cand = _dual_exp_seq(n, alpha, beta, w)

        elif family == 'powexp':
            p = random.uniform(0.0, 5.0)
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            cand = _pow_exp_seq(n, p, alpha)

        else:   # fallback (should never be reached)
            cand = _randdec_seq(n)

        # skip pathological all‑zero vectors (should not happen)
        if sum(cand) < 1e-8:
            continue

        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"New best ({family}) n={n} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) final fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 1000
        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
