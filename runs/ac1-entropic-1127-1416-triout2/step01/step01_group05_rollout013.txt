"""
propose_candidate
-----------------
Search for a low‑value step function by:
1. Building a few simple analytic families (power law, sqrt‑difference,
   and the previous best sequence if available) and picking the best
   among them.
2. Refining the chosen sequence with a projected sub‑gradient descent
   on the convex objective  max(conv(a,a))  (the evaluation function is
   2·n·max/∑a², i.e. proportional to the maximal convolution value).
   The sub‑gradient of max(conv) at a point is easy to compute:
   for the index k where the maximum is attained,
   ∂/∂a_i  (a*a)_k = 2·a_{k‑i} (if the index exists).
   The direction d = –(g – avg(g)) has zero sum, thus preserves the
   sum‑constraint; a step size is chosen so that no entry becomes
   negative.  The step size is adapted (increase after improvement,
   decrease after rejection) and the process stops when time runs out
   or the improvement becomes negligible.
3. A light random‑move polishing phase finishes the search.

The algorithm stays within the 1000 s budget, prints progress and
returns the best sequence found (as a plain list of floats).
"""

import time
import numpy as np

# ---------- helper functions -------------------------------------------------

def _power_sequence(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-p} (unnormalised)."""
    idx = np.arange(n, dtype=float) + 1.0 + offset
    return np.power(idx, -p)


def _sqrtdiff_sequence(n: int) -> np.ndarray:
    """
    a_i = √2·(√(i+1) – √i).  The sum of this sequence is exactly √(2n).
    """
    sqrt2 = np.sqrt(2.0)
    i = np.arange(n, dtype=float)
    return sqrt2 * (np.sqrt(i + 1.0) - np.sqrt(i))


def _normalise_to_sum(seq: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have sum == target."""
    cur = seq.sum()
    if cur == 0.0:
        return seq
    return seq * (target / cur)


def _score_from_seq(seq: np.ndarray) -> float:
    """Direct evaluation of the objective (no safety checks)."""
    n = len(seq)
    # convolution via NumPy (fast enough for the intended sizes)
    conv = np.convolve(seq, seq)
    maxb = conv.max()
    s = seq.sum()
    if s == 0.0:
        return float('inf')
    return 2.0 * n * maxb / (s * s)


def _subgradient_descent(a: np.ndarray,
                         deadline: float,
                         target_score: float = 1.503,
                         verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Perform a projected sub‑gradient descent on max(conv(a,a)).
    The sum of `a` is kept constant (the initial sum) and the entries stay
    non‑negative.
    Returns the best vector seen and its score.
    """
    n = len(a)
    S = a.sum()                     # target sum (kept constant)
    # initialise step size as a modest fraction of the mean entry
    step = 0.1 * np.mean(a)
    min_step = 1e-12
    best_a = a.copy()
    best_score = _score_from_seq(best_a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and score
        conv = np.convolve(a, a)
        maxb = conv.max()
        score = 2.0 * n * maxb / (S * S)

        # early exit if we have already beaten the target
        if score < target_score:
            best_a = a.copy()
            best_score = score
            if verbose:
                print(f'  [subgrad] target reached at it={it}, score={score:.6f}')
            break

        # index (or one of the indices) where the maximum is attained
        k = int(np.argmax(conv))

        # compute sub‑gradient g = 2 * a_{k‑i} on the overlapping part
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[idx] = 2.0 * a[k - idx]

        # direction that preserves the sum (projected sub‑gradient)
        avg_g = g.mean()
        d = -(g - avg_g)               # sum(d) == 0

        # maximal feasible step that keeps a >= 0
        neg = d < 0
        if np.any(neg):
            max_feas = np.min(a[neg] / (-d[neg]))
        else:
            max_feas = np.inf
        # take a safe fraction of the feasible step
        alpha = min(step, max_feas * 0.99)

        if alpha < min_step:
            # step became too tiny – stop the descent
            if verbose:
                print(f'  [subgrad] step too small at it={it}, stop')
            break

        a_new = a + alpha * d

        # numerical drift may change the sum a bit; re‑normalise exactly
        sum_new = a_new.sum()
        if abs(sum_new - S) > 1e-12:
            a_new *= (S / sum_new)

        # evaluate the candidate
        new_score = _score_from_seq(a_new)

        if new_score < score - 1e-12:
            # improvement – accept and try a slightly larger step
            a = a_new
            step *= 1.1
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                if verbose:
                    print(f'  [subgrad] it={it}  improved score={new_score:.6f}')
        else:
            # no improvement – shrink step size
            step *= 0.5

    return best_a, best_score


def _random_polish(a: np.ndarray,
                   current_best_score: float,
                   deadline: float,
                   target_score: float = 1.503,
                   verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Perform cheap random moves (transfer a tiny amount of mass between two
    positions).  Keep a move only if it improves the score.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = current_best_score

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move up to 2 % of a[i]; this gives enough exploration while staying safe
        max_move = 0.02 * a[i]
        if max_move <= 0.0:
            continue
        delta = np.random.uniform(-max_move, max_move)
        if a[i] - delta < 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum is unchanged, still ≈ S
        cand_score = _score_from_seq(cand)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = cand
            a = cand
            if verbose:
                print(f'  [random] improved score={cand_score:.6f}')
            if best_score < target_score:
                break
    return best_a, best_score


# ---------- main entry point -------------------------------------------------

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The function respects the time budget, prints
    intermediate progress and finally returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a few seconds margin for cleanup

    # ----------------------------------------------------------------------
    # 1. generate a few analytically motivated candidates
    # ----------------------------------------------------------------------
    candidates = []

    # a) use the previously discovered sequence if it exists
    prev = globals().get('height_sequence_1')
    if isinstance(prev, list) and prev:
        seq = np.array(prev, dtype=float)
        if seq.size > 0:
            # ensure we are inside the allowed range of values
            seq = np.clip(seq, 0.0, 1000.0)
            # normalise to the exact required sum √(2n) (helps later steps)
            nprev = len(seq)
            target_sum = np.sqrt(2.0 * nprev)
            seq = _normalise_to_sum(seq, target_sum)
            candidates.append(seq)

    # b) sqrt‑difference family (known to be close to optimal)
    for n in (3000, 4000, 5000, 8000):
        seq = _sqrtdiff_sequence(n)
        candidates.append(seq)

    # c) power‑law family (the one used in the previous baseline)
    lengths = (2000, 3000, 4000, 5000)
    p_vals = (0.48, 0.49, 0.5, 0.51, 0.52)
    offsets = (0.0, 0.2, 0.5, 1.0, 2.0, 5.0)
    for n in lengths:
        target_sum = np.sqrt(2.0 * n)
        for p in p_vals:
            for off in offsets:
                raw = _power_sequence(n, p, off)
                seq = _normalise_to_sum(raw, target_sum)
                candidates.append(seq)

    # ----------------------------------------------------------------------
    # 2. pick the best initial candidate (cheap evaluation)
    # ----------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    for seq in candidates:
        sc = _score_from_seq(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq
    if best_seq is None:
        # fallback – uniform distribution (guaranteed to give score 2)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_score = _score_from_seq(best_seq)

    # 3. convex sub‑gradient descent on the chosen sequence
    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------

                              deadline,
                              target_score=1.503,
                              verbose=True)

    # ----------------------------------------------------------------------
    # 5. finish
    # ----------------------------------------------------------------------
    print(f'Final best score: {score:.6f}  length={len(a)}')
    return a.tolist()
