"""
Propose a candidate sequence for the step‑function evaluation.

The algorithm starts from the classic “constant‑convolution” sequence
(a_k = C(2k,k)/4^k) whose self‑convolution is flat (=1).  After normalising
the L‑1 norm to 1 this yields an evaluation ≈π/2≈1.57.  To push the
bound lower we perform a Frank‑Wolfe‑style iteration:

* For the current vector `a` (sum = 1) compute M = max(conv(a,a)).
* Solve a linear program that maximises Σ g_i under the linear constraints
  conv(a,g) ≤ M (each convolution entry is a linear functional of g).
  The LP spreads as much mass as possible into positions where a_i is
  small, while never increasing the convolution above M.
* Scale the obtained `g` so that it has the same L‑1 norm as `a` and mix
  the two vectors (simple averaging).  The mixed vector still has sum = 1
  and, in practice, its maximal convolution strictly decreases.
* Occasionally apply a tiny projected sub‑gradient step and a random
  jitter to avoid stagnation.

The linear program is solved with SciPy’s high‑performance “highs”
solver using a sparse Toeplitz matrix that represents the convolution
operator.  The whole routine runs until the time budget (≈1000 s) is
exhausted, keeping the best sequence found.  Finally the sequence is
checked with the provided `evaluate_sequence` function before being
returned.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def _constant_conv_base(n: int) -> np.ndarray:
    """First n terms of a_k = binom(2k,k) / 4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


def _lp_max_sum_under_bound(a: np.ndarray, M: float,
                            row_ind: np.ndarray, col_ind: np.ndarray,
                            n: int) -> np.ndarray:
    """
    Solve
        max   Σ g_i
        s.t.  conv(a,g) ≤ M      (one inequality per convolution entry)
              g_i ≥ 0
    The convolution operator is represented by a sparse matrix built from
    the pre‑computed (row_ind, col_ind) index arrays.
    """
    # data for the sparse matrix: each column i contributes a[i] to n rows
    data = a[col_ind]                       # shape (n*n,)
    A = sp.csr_matrix((data, (row_ind, col_ind)), shape=(2 * n - 1, n))

    # bounds for each row: conv(a,g) ≤ M
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_i  -> minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)

    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                      method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """One projected sub‑gradient step for min max(conv(a,a))."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    kmax = int(np.argmax(conv))

    g = np.zeros_like(a)

    if kmax < n:
        # indices 0 … kmax are valid
        g[:kmax + 1] = 2.0 * a[kmax::-1]
    else:
        start = kmax - (n - 1)
        length = 2 * n - 1 - kmax
        g[start:] = 2.0 * a[:length][::-1]

    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)

    tot = a_new.sum()
    if tot <= 0.0:
        a_new = np.random.rand(n)
        a_new /= a_new.sum()
        return a_new
    return a_new / tot


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------------
    # Length choice
    # --------------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1,
                                                       (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 2000

    # --------------------------------------------------------------------
    # Initial sequence: constant‑convolution, normalised to L1 = 1
    # --------------------------------------------------------------------
    a = _constant_conv_base(n)
    a = a / a.sum()                     # sum = 1, all entries ≥ 0

    # --------------------------------------------------------------------
    # Pre‑compute index arrays for the convolution LP matrix
    #   rows = i + j , cols = i   for all i,j∈[0,n)
    # --------------------------------------------------------------------
    idx = np.arange(n, dtype=np.int32)
    col_ind = np.repeat(idx, n)                     # shape (n*n,)
    row_ind = col_ind + np.tile(idx, n)             # i + j, also shape (n*n,)

    # --------------------------------------------------------------------
    # Keep the best solution seen so far
    # --------------------------------------------------------------------
    conv = np.convolve(a, a)
    best_M = conv.max()
    best_seq = a.copy()
    best_score = 2.0 * n * best_M               # because sum(a)=1
    print(f"Initial max‑conv {best_M:.6e}, eval {best_score:.6f}")

    # --------------------------------------------------------------------
    # Main optimisation loop (Frank‑Wolfe LP + occasional sub‑grad steps)
    # --------------------------------------------------------------------
    it = 0
    no_improve = 0
    while time.time() < deadline:
        # current max‑convolution
        conv = np.convolve(a, a)
        M = conv.max()

        # ----- solve the LP to push mass into low‑a positions -----
        try:
            g = _lp_max_sum_under_bound(a, M, row_ind, col_ind, n)
        except Exception as e:
            print(f"[Iteration {it}] LP failure: {e}")
            break

        sum_g = g.sum()
        if sum_g <= 0:
            # fallback: tiny random perturbation
            a = np.random.rand(n)
            a = a / a.sum()
            continue

        # Scale g so that it has the same L1 norm as a (which is 1)
        g = g * (a.sum() / sum_g)

        # Simple averaging (t = 0.5) – preserves the L1 norm = 1
        a_new = 0.5 * (a + g)

        # Small random jitter every few iterations to avoid cycling
        if it % 7 == 0 and it > 0:
            a_new += np.random.randn(n) * 1e-6
            a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()          # re‑normalise to sum = 1

        # Occasionally enrich with a cheap sub‑gradient step (helps fine‑tune)
        if it % 15 == 0:
            step_sz = 0.4 / np.sqrt(it + 1.0)
            a_new = _subgradient_step(a_new, step_sz)

        a = a_new

        # evaluate improvement
        conv = np.convolve(a, a)
        cur_M = conv.max()
        cur_score = 2.0 * n * cur_M

        if cur_M < best_M - 1e-9:
            best_M = cur_M
            best_seq = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"Iter {it:6d}: improved max‑conv {best_M:.6e}, eval {best_score:.6f}")
        else:
            no_improve += 1
            # if we have been stuck for many iterations, re‑inject jitter
            if no_improve >= 40:
                a += np.random.randn(n) * 1e-5
                a = np.maximum(a, 0.0)
                a /= a.sum()
                no_improve = 0

        it += 1

    # --------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # --------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final reported evaluation: {final_score:.6f}")
    except Exception:
        final_score = best_score
        print(f"Final evaluation (fallback) : {final_score:.6f}")

    print(f"Search finished after {it} iterations. Best eval = {final_score:.6f}")
    return best_seq.tolist()
