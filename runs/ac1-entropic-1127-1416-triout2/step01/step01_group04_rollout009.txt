def propose_candidate(seed=...):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    # Determine N: try 3000 for start
    n = 3000

    Lpad = 1 << int(np.ceil(np.log2(2*n - 1)))

    def conv_fft(a):
        A = np.fft.rfft(a, n=Lpad)
        conv = np.fft.irfft(A * A, n=Lpad)[:2*n-1]
        return conv

    def gradient(a, conv, alpha):
        max_c = np.max(conv)
        # to avoid overflow/underflow:
        w = np.exp(alpha * (conv - max_c))
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum
        A = np.fft.rfft(a, n=Lpad)
        W = np.fft.rfft(w, n=Lpad)
        grad = 2 * np.fft.irfft(W * np.conj(A), n=Lpad)[:n]
        return grad

    best_seq = None
    best_score = np.inf

    # Define list of initial generators functions to produce a start vector
    def init_constant():
        base = np.empty(n, dtype=np.float64)
        base[0] = 1.0
        for k in range(1, n):
            base[k] = base[k-1] * (1.0 - 1.0/(2.0*k))
        a = base / base.sum()
        return a

    def init_random():
        a = np.random.rand(n)
        return a / a.sum()

    def init_triangular():
        i = np.arange(n)
        a = 1 - np.abs(i - (n-1)/2) / ((n-1)/2)  # linear decreasing to zero at edges
        a = np.maximum(a, 0)
        return a / a.sum()

        a = np.sin(np.pi * (i+0.5) / n)  # half-sine positive
        return a / a.sum()

    init_fns = [init_constant, init_random, init_triangular, init_sine]

    for init_fn in init_fns:
        if time.time() >= deadline:
            break
        a = init_fn()
        # Adam parameters
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        t = 0

        max_alpha = 10 * n  # maybe large
        while time.time() < deadline:

            cur_max = np.max(conv)

"""
Optimisation of a non‑negative step function for the objective

    F(a) = 2·n·max_k (a*a)[k]   (the sum of a is forced to 1)

The algorithm works with an arbitrary length **n** (taken from the
previous best construction if it exists, otherwise a default of 8000).
A smooth approximation of the “max’’ is used:

    w_k = exp( α·(c_k – max(c)) ) / Σ_j exp( α·(c_j – max(c)) )
    ∇F ≈ 2·corr(a, w)

where *c* = a*a (linear convolution) and *corr* denotes the
cross‑correlation (implemented via FFT).  The weight exponent α is
ramped from a modest value up to a large one, so the gradient initially
sees a softened maximum and later focuses on the true peak.  An Adam
style step‑size schedule together with occasional random jitter helps
escape the degenerate “constant‑convolution’’ plateau.

Multiple initialisations (constant‑convolution, random, triangular and
half‑sine) are tried; the best sequence found before the time budget
expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _next_power_of_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def compute_convolution(a: np.ndarray, pad: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``pad``.
    Returns the first (2·len(a)‑1) entries.
    """
    A = np.fft.rfft(a, n=pad)
    conv = np.fft.irfft(A * A, n=pad)[: 2 * a.shape[0] - 1]
    # tiny negative values can appear because of round‑off; force non‑neg.
    np.maximum(conv, 0.0, out=conv)
    return conv


def compute_weighted_gradient(a: np.ndarray, conv: np.ndarray,
                              alpha: float, pad: int) -> np.ndarray:
    """
    Gradient of the smooth max approximation.
    ``conv`` is a*a (length 2·n‑1).  ``alpha`` controls the softness.
    The gradient equals 2·corr(a, w) where
        w_k ∝ exp(alpha·(conv_k – max(conv))).
    """
    # soften the max – shift by its value to avoid overflow
    cmax = np.max(conv)
    # exponent is non‑positive, so no overflow
    w = np.exp(alpha * (conv - cmax))
    w_sum = np.sum(w)
    if w_sum == 0.0:                # should never happen, but guard
        w = np.ones_like(conv) / conv.size
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w  (same as corr(a,w))
    A = np.fft.rfft(a, n=pad)
    W = np.fft.rfft(w, n=pad)
    grad = 2.0 * np.fft.irfft(W * np.conj(A), n=pad)[: a.shape[0]]
    # numerical noise may give tiny negative entries – clip later
    return grad.real


def _init_constant_convolution(n: int) -> np.ndarray:
    """Truncated central‑binomial sequence, normalised to sum‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    a /= a.sum()
    return a


def _init_random_distribution(n: int) -> np.ndarray:
    """Uniform‑random positive vector, normalised to sum‑1."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def _init_triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle, peak in the centre."""
    i = np.arange(n, dtype=np.float64)
    peak = (n - 1) / 2.0
    a = 1.0 - np.abs(i - peak) / peak
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _init_half_sine_distribution(n: int) -> np.ndarray:
    """Half‑sine shape (positive on the whole interval)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_ignored) -> list:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the allotted time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a length.  Use the length of the previously found sequence
    # if it exists; otherwise fall back to a reasonably large default.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1,
                                                       (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 8000                     # a size that still fits comfortably
    print(f"Optimising length n = {n}")

    # Pad length for all FFTs (power of two for speed)
    pad_len = _next_power_of_two(2 * n - 1)

    # ------------------------------------------------------------------
    # Prepare a list of initialisers.
    # ------------------------------------------------------------------
    init_fns = [
        _init_constant_convolution,
        _init_random_distribution,
        _init_triangular_distribution,
        _init_half_sine_distribution,
    ]

    best_seq = None
    best_score = np.inf
    total_iters = 0

    # ------------------------------------------------------------------
    # Optimisation loop for each initialiser.
    # ------------------------------------------------------------------
    for init_fn in init_fns:
        if time.time() >= deadline:
            break

        a = init_fn(n)                     # start with sum(a) == 1
        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-12

        it = 0
        # soft‑max exponent ramp parameters
        alpha_start = 5.0
        alpha_end = 12.0 * n                # aggressive final smoothing
        ramp_iters = 5000                   # number of iterations over which to ramp

        while time.time() < deadline:
            it += 1
            total_iters += 1

            # --------------------------------------------------------------
            # Compute convolution, current score, and possibly update best.
            # --------------------------------------------------------------
            conv = compute_convolution(a, pad_len)
            cur_max = np.max(conv)
            cur_score = 2.0 * n * cur_max        # identical to evaluate_sequence
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{init_fn.__name__}] it {it:6d} → score {best_score:.6f}")

            # --------------------------------------------------------------
            # Adaptive smoothing parameter α – linear ramp.
            # --------------------------------------------------------------
            alpha = alpha_start + (alpha_end - alpha_start) * min(1.0,
                                                                  it / ramp_iters)

            # --------------------------------------------------------------
            # Gradient (soft‑max of the max) and Adam update.
            # --------------------------------------------------------------
            grad = compute_weighted_gradient(a, conv, alpha, pad_len)

            # Adam moments
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # Step size – slowly decaying
            step = 0.5 / (np.sqrt(it) + 1.0)

            # Parameter update
            a -= step * m_hat / (np.sqrt(v_hat) + eps)

            # --------------------------------------------------------------
            # Projection onto the feasible set: a_i ≥ 0, Σ a_i = 1.
            # --------------------------------------------------------------
            np.maximum(a, 0.0, out=a)
            s = a.sum()
            if s == 0.0:               # extremely unlikely, re‑initialise
                a = np.random.rand(n)
                s = a.sum()
            a /= s

            # --------------------------------------------------------------
            # Occasional random jitter to break symmetry.
            # --------------------------------------------------------------
            if it % 2500 == 0:
                a += np.random.randn(n) * 1e-5
                np.maximum(a, 0.0, out=a)
                a /= a.sum()

            # --------------------------------------------------------------
            # Stop the inner loop after a reasonable number of steps.
            # --------------------------------------------------------------
            if it >= 120000:           # ~1 ‑ 2 seconds per init on a modern CPU
                break

    # ------------------------------------------------------------------
    # Final sanity check – call the supplied ``evaluate_sequence``.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a single unit entry (guaranteed valid)
        best_seq = [1.0]
        best_score = evaluate_sequence(best_seq)
        print("No improvement found; using trivial sequence.")
    else:
        try:
            final_score = evaluate_sequence(best_seq.tolist())
            print(f"Final reported score from evaluate_sequence: {final_score:.6f}")
        except Exception:  # pragma: no cover
            print("evaluate_sequence call failed (should not happen).")
            final_score = best_score

    print(f"Search finished after {total_iters} total iterations, best score {best_score:.6f}")
    return best_seq.tolist()
