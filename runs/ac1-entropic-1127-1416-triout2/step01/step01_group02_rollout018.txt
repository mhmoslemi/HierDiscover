def build_conv_matrix(f):
    n = len(f)
    rows = []
    for k in range(2 * n - 1):
        # create a row of length n
        row = np.zeros(n, dtype=np.float64)
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        # i runs from start to end inclusive
        # j = k - i

        # vectorized: row[start:end+1] = f[k - np.arange(start, end+1)]

    seq = np.empty(length, dtype=np.float64)
    seq[0]=1.0

propose_candidate
-----------------
A directed stochastic search that starts from the classic
central‑binomial (≈ t⁻¹ᐟ²) step heights and repeatedly nudges mass
away from the coefficients that are most responsible for the current
maximum of the auto‑convolution.  The move is biased by the actual
contribution a_i·a_{k‑i} to the peak index k and accepts only changes
that lower the evaluation score (with a tiny probability of uphill
steps to escape plateaus).  For very small problems (n ≤ 200) an
additional linear‑programming refinement – the “g‑step’’ described in
the problem statement – is performed occasionally.  The routine runs
until the supplied time budget expires and returns the best sequence
found.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Central binomial coefficients c_k = (2k choose k) / 4**k for k=0..L‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2.0 * k - 1.0) * (2.0 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def conv_fft(a: np.ndarray) -> np.ndarray:
    """Fast linear convolution a*a using an FFT of length power‑of‑2."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length() + 1)   # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv

    cur_score = evaluate_sequence(cur_seq.tolist())

    best_seq = cur_seq.copy()
    best_score = cur_score

    # Parameters for the guided random walk.
    step_frac = 0.02          # start with moving 2 % of a source entry
    decay = 0.9995           # shrink step size slowly
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ----- current convolution & its maximum ---------------------------------
