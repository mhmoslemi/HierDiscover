"""
Propose a candidate sequence that beats the classic constant‑convolution construction.

The algorithm combines several ideas:

* Several diverse initialisations (the optimal constant‑convolution sequence,
  a few power‑law sequences and a few random Dirichlet draws).
* A projected sub‑gradient descent step on the true objective
  (the max of the autocorrelation) with a slowly decaying step size.
* Small Gaussian jitter every few hundred iterations to escape the large
  plateau where many lag values are tied.
* A cheap simulated‑annealing style “mass‑transfer’’ move:
  pick two indices i≠j, move a small amount of probability mass from i to j,
  re‑project onto the probability simplex and accept the move if it improves
  (or with a tiny probability otherwise).
* The whole search runs under the 1000 s budget, constantly keeping the best
  sequence seen.  The final score is verified with the provided
  ``evaluate_sequence`` routine.

The objective is scale‑invariant, therefore all sequences are kept normalised
to unit ℓ₁‑norm (a simplex).  Convolutions are computed with an FFT for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the objective for a simplex‑normalised vector a.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Compute 2 * n * max(conv(a,a)) / (sum(a)**2)  (the exact value used by
    evaluate_sequence).  The vector a does NOT need to be normalised, but the
    function works for any non‑negative a.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # FFT‑based convolution (real‑valued)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Subgradient step for minimising max_k (a*a)[k] under the simplex.
# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    n = a.shape[0]
    # convolution via FFT (fast enough for the subgradient)
    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    kmax = int(np.argmax(conv))

    # build sub‑gradient g where g[i] = 2 * a[kmax‑i] if the index is valid
    g = np.zeros_like(a)
    if kmax < n:
        g[: kmax + 1] = 2.0 * a[kmax :: -1]
    else:
        start = kmax - (n - 1)
        length = 2 * n - 1 - kmax
        g[start:] = 2.0 * a[:length][::-1]

    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    # re‑project onto the simplex (preserves sum = 1)
    a_new = _project_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Base initialisations
# ----------------------------------------------------------------------
def _constant_conv_base(n: int) -> np.ndarray:
    """First n terms of a_k = C(2k,k)/4^k (the classical constant‑convolution)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


def _power_law_base(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-alpha}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-alpha)
    return a


def _random_dirichlet_base(n: int) -> np.ndarray:
    """Random point on the simplex (exponential/Dirichlet(1) sampling)."""
    a = np.random.rand(n)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    evaluate_sequence.  Returns the best sequence found before the time budget
    expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose a length.  Prefer the length of the previously found
    # construction if it exists, otherwise fall back to 2000.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(
        height_sequence_1, (list, tuple)

        n = len(height_sequence_1)
    else:
        n = 2000

    fft_len = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    init_pool = []

    # 1) classic constant‑convolution shape
    base = _constant_conv_base(n)
    a0 = base / base.sum()
    init_pool.append(a0)

    # 2) a few power‑law shapes (α around the optimal 0.5)
    for alpha in (0.50, 0.55, 0.60, 0.65):
        a = _power_law_base(n, alpha)
        init_pool.append(a / a.sum())

    # 3) random points on the simplex
    for _ in range(2):

    # --------------------------------------------------------------

    # --------------------------------------------------------------

        if sc < best_score:

    print(f"Initial best score {best_score:.6f}")

    # --------------------------------------------------------------

    it = 0
    base_step = 0.8                 # initial sub‑gradient step size
    jitter_every = 3000            # how often we add a small random jitter
    jitter_scale = 1e-4            # amplitude of the jitter
    pair_move_prob = 0.12          # probability of performing a random mass‑transfer move
    temperature_start = 0.02       # initial temperature for the SA acceptance rule

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- adaptive step size for the sub‑gradient -----
        step = base_step / np.sqrt(it + 1.0)

        # ----- one sub‑gradient step ----- (keeps us on the simplex)
        a = _subgradient_step(best_seq, step)

        # ----- occasional tiny jitter to break symmetry -----
        if it % jitter_every == 0 and it > 0:

            a = np.maximum(a, 0.0)

        # ----- keep the global best -----
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        # ----- random pair‑mass move (simulated‑annealing style) -----
        if np.random.rand() < pair_move_prob:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

            # amount to move: a small fraction of the source entry
            frac = 0.05 * np.random.rand()          # up to 5 % of a[i]
            delta = min(frac * a[i], a[i])          # keep a[i] non‑negative

            if delta > 0.0:
                a2 = a.copy()
                a2[i] -= delta
                a2[j] += delta
                # projection is cheap because sum is unchanged, we only need to
                # enforce non‑negativity (which is already true)
                a2 = _project_simplex(a2)

                new_score = _objective(a2, n, fft_len)

                # simple Metropolis acceptance

                accept = new_score < cur_score or np.random.rand() < np.exp(
                    -(new_score - cur_score) / max(temp, 1e-12)
