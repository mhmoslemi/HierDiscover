import numpy as np
import scipy.sparse as sp

def solve_lp(f, M):
    n = len(f)
    # Build sparse matrix A shape (2*n-1, n)
    # For each row k, entries at columns i where 0 <= i < n and 0 <= k-i < n => i ∈ [max(0, k-n+1), min(k, n-1)]
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_min = max(0, k - (n-1))
        i_max = min(k, n-1)
        # the indices i in [i_min, i_max]
        # for each i, coefficient = f[i]
        cols_i = np.arange(i_min, i_max+1)
        rows_i = np.full_like(cols_i, k)
        rows.append(rows_i)
        cols.append(cols_i)
        data.append(np.full_like(cols_i, f[i_min:i_max+1]))
    # concatenate
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x  # solution vector g

"""Propose a low‑score step‑function sequence.

The algorithm proceeds in three stages:

1. **Broad seeding** – a collection of simple families (geometric,
   power‑law, uniform) is scanned.  For each candidate we keep the
   exact ℓ₁‑norm (the “target sum” √(2 n)) by projecting onto the
   simplex and we evaluate the score by a fast FFT‑based convolution.

2. **Active‑set projected sub‑gradient descent** – starting from the
   best seed we repeatedly locate all convolution indices whose
   value is within a small tolerance of the current maximum.
   The sub‑gradient of the max‑functional is the (normalised) sum of
   the individual gradients belonging to the active set.  After a
   step we project back onto the simplex {x≥0, Σx = √(2 n)}.
   Momentum (Nesterov‑type) and occasional random jitter help to
   escape plateaus.

3. **Linear‑programming refinement** – using the current vector `f`,
   we solve the LP  

        max Σ g_i               (g_i ≥ 0)
        s.t.  (f * g)_k ≤ M    for all k

   where `M` is the present maximal convolution value.
   The solution `g` is rescaled to the target ℓ₁‑norm and mixed
   with `f` (small mixing weight).  A few LP‑updates are performed
   until no further improvement is observed.

All operations respect the security checks performed by
``evaluate_sequence`` (non‑negative, ≤ 1000).  The routine stops
when the allotted time (≈ 1000 s) expires and returns the best
sequence discovered.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Duchi et al. 2008)
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _fast_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq)) / (sum(seq)²) using FFT."""
    n = seq.size
    # length of FFT – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    # the denominator is never too small (we keep a proper ℓ₁‑norm)
    return float(2.0 * n * max_conv / (s * s))


# --------------------------------------------------------------------------- #
def _build_toeplitz_matrix(f: np.ndarray):
    """Return a sparse matrix A (shape (2n‑1, n)) such that A·g = f* g."""
    # Each column i of A contains the vector f shifted by i positions.
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    ptr = 0
    for i in range(n):
        rows[ptr: ptr + n] = np.arange(i, i + n)          # rows i … i+n‑1
        cols[ptr: ptr + n] = i                           # column i
        data[ptr: ptr + n] = f
        ptr += n

    from scipy.sparse import csr_matrix
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The function respects the 1000 s time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2      # safety margin for the return

    # ------------------------------------------------------------------- #
    # 1. Problem size – a few thousand entries give a good trade‑off.
    n = 2500
    target_sum = np.sqrt(2.0 * n)          # we keep the ℓ₁‑norm fixed

    # ------------------------------------------------------------------- #
    # 2. Helper: fast scoring, FFT plan (shared length)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------- #
    # 3. Generate a coarse set of seeds.
    seeds = []

    # geometric families (λ = 1‑r)
    lambdas = np.logspace(-5, -1, 40)
    for lam in lambdas:
        r = 1.0 - lam
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom
        a = C * (r ** np.arange(n))
        a = np.clip(a, 0.0, 1000.0)
        seeds.append(a)

    # power‑law families a_i ∝ (i+1)^‑p
    for p in np.linspace(0.3, 0.95, 30):
        base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        C = target_sum / base.sum()
        a = C * base
        a = np.clip(a, 0.0, 1000.0)
        seeds.append(a)

    # a simple uniform seed (fallback)
    seeds.append(np.full(n, target_sum / n))

    # If the global reference sequence has the same length, use it as a seed.
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.shape[0] == n:
                hs = np.clip(hs, 0.0, 1000.0)
                hs = hs * (target_sum / hs.sum())
                seeds.append(hs)
    except Exception:
        pass

    # ------------------------------------------------------------------- #
    # 4. Scan the seeds, keep the best (projected onto the simplex).
    best_seq = None
    best_score = float("inf")

    for idx, cand in enumerate(seeds):
        if time.time() >= deadline:
            break
        # ensure exact sum via projection (otherwise the score formula differs)
        cand = projection_simplex(cand, target_sum)
        sc = _fast_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time() - start:6.1f}s] seed {idx:02d} → score {sc:.6f}")

    # ------------------------------------------------------------------- #
    # 5. Active‑set projected sub‑gradient descent (with momentum).
    curr = best_seq.copy()
    best_score = _fast_score(curr)
    best = curr.copy()

    # momentum buffer
    velocity = np.zeros_like(curr)

    step = 0.7 * target_sum / n      # initial absolute step
    min_step = 1e-9
    momentum = 0.9
    it = 0
    # tolerance for the active set (relative to current max)
    rel_tol = 1e-4

    while time.time() < deadline:
        # ----- convolution via FFT ----------------------------------------
        F = np.fft.rfft(curr, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]

        max_val = conv.max()
        score = float(2.0 * n * max_val / (curr.sum() ** 2))

        if score < best_score - 1e-9:
            best_score, best = score, curr.copy()
            print(f"[{time.time() - start:6.1f}s] iter {it:7d} → new best {score:.6f}")
            # tighten step when we make progress
            step = max(step * 0.9, min_step)

        # ----- active set -------------------------------------------------
        tol_abs = max_val * rel_tol
        active_idx = np.where(conv >= max_val - tol_abs)[0]

        # combined sub‑gradient (sum over active constraints)
        grad = np.zeros_like(curr)
        idx_arr = np.arange(n)
        for k in active_idx:
            j = k - idx_arr
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * curr[j[mask]]

        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0:
            break

        # ----- Nesterov‑type momentum step ---------------------------------
        velocity = momentum * velocity - step * (grad / gnorm)
        tentative = curr + velocity
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # ----- occasional jitter (helps to escape flat regions) ----------
        if np.random.rand() < 0.015:
            jitter = (np.random.randn(n) *
                      step * target_sum / (30.0 * n))
            curr = projection_simplex(
                np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        it += 1
        # slowly decay the step size
        if it % 400 == 0:
            step = max(step * 0.85, min_step)

    # ------------------------------------------------------------------- #
    # 6. Linear‑programming refinement (if SciPy is available).
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:
        linprog = None

    if linprog is not None and time.time() < deadline - 5.0:
        f = best.copy()
        for lp_iter in range(12):            # a handful of LP‑updates
            if time.time() >= deadline - 5.0:
                break
            # current maximal convolution value
            F = np.fft.rfft(f, n=fft_len)
            conv_f = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
            M = conv_f.max()

            # Build the Toeplitz matrix for the constraints  (f * g)_k ≤ M
            A = _build_toeplitz_matrix(f)      # shape (2n‑1, n)
            b = np.full(2 * n - 1, M)

            # Maximise Σ g_i  →  minimise -Σ g_i
            res = linprog(
                c=-np.ones(n),
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method='highs',
                options={'presolve': True, 'time_limit': 30}
            )
            if not res.success or res.x is None:
                break

            g = res.x
            sum_g = g.sum()
            if sum_g < 1e-12:
                break

            # Rescale g to the target ℓ₁‑norm and mix.
            g = g * (target_sum / sum_g)
            t = 0.15   # modest mixing weight
            f_new = (1.0 - t) * f + t * g
            f_new = projection_simplex(f_new, target_sum)

            # compute new score
            Fnew = np.fft.rfft(f_new, n=fft_len)
            conv_new = np.fft.irfft(Fnew * Fnew, n=fft_len)[: 2 * n - 1]
            Mnew = conv_new.max()
            score_new = float(2.0 * n * Mnew / (f_new.sum() ** 2))

            if score_new < best_score - 1e-8:
                best_score, best = score_new, f_new.copy()
                print(f"[{time.time() - start:6.1f}s] LP‑iter {lp_iter} → best {score_new:.6f}")

            # accept the update if it does not worsen the bound
            if Mnew <= M * 0.999:
                f = f_new
            else:
                # no improvement – stop LP phase
                break

    # ------------------------------------------------------------------- #
    # 7. Final sanity check with the official evaluator.
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best.tolist()
