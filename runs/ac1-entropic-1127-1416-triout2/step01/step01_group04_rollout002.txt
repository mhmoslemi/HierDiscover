"""
Improved search for a step‑function sequence that minimizes the
`evaluate_sequence` objective.

The algorithm mixes two complementary ideas:

1. **LP‑based mass redistribution**  
   For the current sequence `f` we compute the maximal self‑convolution
   value `M = max_k (f*f)[k]`.  Keeping this bound fixed we solve a linear
   program

        maximise   Σ b_j
        subject to (f * b)[k] ≤ M   for all k
                   b_j ≥ 0

   The optimal `b` can carry a larger ℓ¹‑mass than `f` while respecting the
   same supremum bound.  After normalising `b` to a probability vector we
   blend it with `f`.  This step provably never increases `M` and typically
   flattens the convolution.

2. **Projected sub‑gradient refinement**  
   After each mixing step we run a short schedule of projected sub‑gradient
   updates that directly target the index attaining the current maximal
   convolution.  This quickly removes any remaining spikes.

The outer loop repeats the LP‑mix‑subgradient cycle while the time budget
remains.  The best sequence observed (lowest `evaluate_sequence` value) is
returned.  A small random jitter is occasionally injected to avoid getting
stuck on plateaus.  The routine works for any length `n`; by default it
uses the length of an already‑available `height_sequence_1` or falls back to
`n = 2000`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: classic “constant‑convolution” base sequence (binomial coefficients)
# ----------------------------------------------------------------------
def _constant_conv_base(n: int) -> np.ndarray:
    """Return the first n terms of a_k = C(2k,k)/4^k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


# ----------------------------------------------------------------------
# Helper: one projected sub‑gradient step for max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Perform a single sub‑gradient step for the objective
    max_k (a*a)[k] with a ≥ 0 and Σ a = 1 (normalisation performed later).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    kmax = int(np.argmax(conv))

    # gradient g_i = 2 * a_{kmax-i} when the index is valid, else 0
    g = np.zeros_like(a)
    if kmax < n:
        g[:kmax + 1] = 2.0 * a[kmax::-1]
    else:
        start = kmax - (n - 1)            # first i with kmax‑i < n
        length = 2 * n - 1 - kmax        # number of valid positions
        g[start:] = 2.0 * a[:length][::-1]

    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    tot = a_new.sum()
    if tot <= 0.0:                       # extremely unlikely
        a_new = np.random.rand(n)
        a_new /= a_new.sum()
    else:
        a_new /= tot
    return a_new


# ----------------------------------------------------------------------
# Helper: build the (2n‑1) × n matrix for the LP constraints
#   (f * b)[k] = Σ_i f_i * b_{k-i}  ≤  M .
# ----------------------------------------------------------------------
def _build_ub_matrix(f: np.ndarray) -> np.ndarray:
    """Construct the inequality matrix A with rows (f * b)[k] ≤ M."""
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # Each column i contributes f[i] to rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function that minimises the provided
    `evaluate_sequence`.  The function respects the given time budget and
    returns the best sequence found as a plain Python list.
    """

        from scipy.optimize import linprog
        SCIPY_AVAILABLE = True
    except Exception:
        SCIPY_AVAILABLE = False

    np.random.seed(seed)

    # --------------------------------------------------------------

    else:

        a = _constant_conv_base(n)
        print(f"Initialising with constant‑convolution base, length {n}.")

    # Ensure feasibility (non‑negative, normalised)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum

    # --------------------------------------------------------------
    # Keep track of the best solution seen so far
    # --------------------------------------------------------------
    best_seq = a.copy()

    # Parameters for the outer LP‑mixing loop
    outer_iter = 0

    mix_min = 0.05            # never go below this
    subgrad_steps = 80       # number of sub‑gradient refinements per outer step

        A = _build_ub_matrix(a)

        b_ub = np.full(m, M, dtype=np.float64)

            # If LP fails we fall back to a few sub‑gradient steps and continue
            step = 0.5 / np.sqrt(outer_iter + 1.0)
            a = _subgradient_step(a, step)
            outer_iter += 1
            continue

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0.0:

            break

        # ----- 3) blend current a with the LP solution g -----
        mix = max(mix_min, mix_initial / np.sqrt(outer_iter + 1.0))
        a_new = (1.0 - mix) * a + mix * g
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        # ----- 4) short sub‑gradient refinement to clean residual spikes -----
        for i in range(subgrad_steps):

            a_new = _subgradient_step(a_new, step)

        a = a_new

        # ----- 5) occasional random jitter to escape plateaus -----
        if outer_iter % jitter_interval == 0 and outer_iter > 0:
            a += np.random.randn(n) * 1e-6
            a = np.maximum(a, 0.0)
            a /= a.sum()

        # ----- 6) evaluate and store the best sequence -----
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{outer_iter:04d}] improved score {best_score:.6f}")

        # optional early stop when target is reached
        if best_score <= 1.503:
            print(f"Target threshold reached at outer iteration {outer_iter}.")
            break

        outer_iter += 1

    # --------------------------------------------------------------
    # Final sanity check (cheap) and return the best sequence found
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
