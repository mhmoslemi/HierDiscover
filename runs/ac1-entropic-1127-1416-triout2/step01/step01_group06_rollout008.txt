def improve_sequence(f, M, iterations, t):
   for iter in range(iterations):
        # Solve LP:
        g = cvxpy.Variable(n)
        Mvar = M (constant)
        constraints = [g >= 0]
        # Build convolution constraints:
        for k in range(2*n - 1):
            # compute coeff vector
            # Equivalent: sum_i f_i * g_{k-i} <= M
            # use indices boundaries
            idx_start = max(0, k - (n-1))
            idx_end = min(k, n-1)
            # Build sum: sum_{i=idx_start}^{idx_end} f_i * g_{k-i}
            # Introduce expression: sum(f_i * g_{k-i})
            # Must map index j = k - i
        constraints.append( sum( f[i] * g[k - i] for i = idx_start.. idx_end) <= Mvar )
        # For linear constraints each term is variable g with coefficient f_i.
        # Use cvxpy sum.
        objective = cvxpy.Maximize( cvxpy.sum(g) )
        prob = cvxpy.Problem(objective, constraints)
        prob.solve(solver='ECOS', warm_start=True) # might need.

        # Get g0 as np.array( g.value )
        if g.value is None:
            break
        # Rescale to have same sum as f (or any target):
        sum_g = np.sum(g0)
        if sum_g <= 0: break
        g = g0 * (np.sum(f) / sum_g)
        # Mix:
        f = (1 - t) * f + t * g
        # Compute M = max(conv(f,f))
   return f

    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix, csr_matrix

    random.seed(seed); np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5

    # initialize f as exponentials with small alpha to get low integral? Let's start with exp decay
    n = 1200 # we can adjust based on time
    alpha = 0.0015 # moderate decay giving sum ~ 1/alpha ~ 666
    f = np.exp(-alpha * np.arange(n))
    f = f / f.max()  # max=1
    # Normalize sum to 1
    f = f / f.sum()
    # Evaluate current score
    best_seq = f.tolist()
    best_score = evaluate_sequence(best_seq)
    print(f"Init score: {best_score}")

    # iterative LP improvement
    M = np.max(np.convolve(f, f))
    t = 0.3
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # Build convolution constraint matrix A: shape (2n-1, n)
        # Use LIL for incremental fill
        n = len(f)
        rows = 2*n - 1
        A = lil_matrix((rows, n), dtype=float)
        for k in range(rows):
            i_start = max(0, k-(n-1))
            i_end = min(k, n-1)
            # iterate i and set coefficient f[i] at column j = k - i
            # Document: for each i from i_start to i_end:
            #   j = k - i
            #   A[k, j] = f[i]
            # Use vectorized for speed?
            for i in range(i_start, i_end+1):
                j = k - i
                A[k, j] = f[i]
        A = csr_matrix(A)

        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        b = np.full(rows, M)
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'presolve': True})
        if not res.success:
            print(f"LP failed at iteration {iteration}, status {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        # Scale g to same sum as f
        if sum_g > 0:
            g = g * (f.sum() / sum_g)
        # Mix
        f = (1 - t) * f + t * g
        # recompute M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        score = evaluate_sequence(f.tolist())
        if score < best_score:
            best_score = score
            best_seq = f.tolist()
            print(f"Iter {iteration}: improved score {best_score:.6f}")

        # Optionally adapt t: reduce t if no improvement after some steps
        if iteration % 10 == 0:
            t = max(0.05, t * 0.9)

    print(f"Finished best score {best_score:.6f}")
    return best_seq

The score of a step‑function `a = (a₀,…,aₙ₋₁)` is  

```
score(a) = 2·n·max(a*a) / ( Σ aᵢ )²                 (1)
```

`max(a*a)` is the largest entry of the auto‑convolution `a*a`.  
The problem is invariant under a global scaling of the vector, therefore we
keep the maximum entry equal to 1 – this prevents overflow and makes the LP
constraints well‑conditioned.

The algorithm is an implementation of the *alternating linear programme*
described in the statement:

* start from a feasible sequence `f`;
* let `M = max(f*f)`;
* **LP step** – maximise `Σ gᵢ` under the linear constraints  
  `f * g ≤ M` (element‑wise) and `gᵢ ≥ 0`.  
  The solution `g` has a larger total mass than `f` while keeping the
  convolution never larger than `M`;
* **mix** `f` and `g` (`f ← (1‑t)·f + t·g` with a small mixing factor `t`);
* repeat – each iteration can only decrease the objective (1).

A tiny amount of stochastic perturbation is added occasionally to escape
possible plateaus.  All operations respect the non‑negativity requirement and
the helper `evaluate_sequence` is used only for reporting the current best
value.

The whole search runs for the allotted time (≈ 1000 s) and returns the best
sequence found.

```python
"""search for a low‑score step function

The algorithm implements the LP‑based improvement described in the statement.
Starting from a (possibly supplied) sequence it repeatedly
  – builds the convolution matrix A of the current sequence f,
  – solves   max Σ g   s.t.  A·g ≤ M,  g ≥ 0,
  – rescales g to have the same total mass as f,
  – mixes f ← (1‑t)·f + t·g  and renormalises the maximum to 1,
  – keeps the best sequence according to evaluate_sequence().
A few random “kick’’ steps are inserted to avoid stagnation.
The loop stops when the time budget expires and the best sequence is
returned.
"""

    mx = arr.max()

def _build_conv_matrix(f):
    """
    Build the (2·n‑1) × n sparse matrix A of the linear map
        g ↦ f * g   (full convolution)
    using scipy.sparse.diags.  Row k, column j contains f[k‑j] when
    0 ≤ k‑j < n, otherwise 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    diags_data = []
    offsets = []
    # each offset d (0 ≤ d < n) corresponds to the diagonal where row‑col = d
    for d in range(n):
        val = f[d]
        # diagonal length = rows‑d
        diags_data.append(np.full(rows - d, val, dtype=float))
        offsets.append(d)
    return sp.diags(diags_data, offsets, shape=(rows, n), format='csr')


def _solve_lp(f, M):
    """
    Solve the linear programme
        maximise Σ gᵢ
        subject to   f * g ≤ M   (element‑wise)
                     g ≥ 0
    Returns the optimal vector g (numpy array) or None on failure.
    """
    if not _HAS_SCIPY:          # pragma: no cover
        return None

    A = _build_conv_matrix(f)           # shape (2n‑1, n)
    n = f.shape[0]
    rows = A.shape[0]

    # maximise Σ g  ⇔ minimise −Σ g
    c = -np.ones(n, dtype=float)

    # a tiny slack avoids numerical infeasibility
    b = np.full(rows, M * (1.0 + 1e-12), dtype=float)

    bounds = [(0.0, None)] * n

    try:
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
    except Exception:      # pragma: no cover
        return None

    if res.success and res.x is not None:
        return res.x
    return None


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided `evaluate_sequence`.  The routine runs for roughly `budget_s`
    seconds, mixing LP‑based improvements with occasional stochastic
    perturbations, and finally returns the best sequence discovered.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # time limit (small safety margin)
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.3

    # ------------------------------------------------------------------
    # initialise f
    # ------------------------------------------------------------------
    f = _initial_sequence()
    best_seq = f.tolist()
    best_score = _score(best_seq)
    print(f"[Init] length={len(f)}  score={best_score:.6f}")

    # mixing factor – starts relatively large, shrinks when progress stalls
    t = 0.30
    stagnation = 0      # number of iterations without improvement

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----------------------------------------------------------------
        # 1) normalise (max = 1) – keeps numbers inside the safe range of the
        #    evaluation routine
        # ----------------------------------------------------------------
        f = _normalise_max_one(f)

        # ----------------------------------------------------------------
        # 2) current convolution bound M
        # ----------------------------------------------------------------
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # ----------------------------------------------------------------
        # 3) LP step – find a vector g with larger total mass that respects
        #    the same convolution bound
        # ----------------------------------------------------------------
        g = _solve_lp(f, M)
        if g is None:
            # LP failed – fall back to a cheap random hill‑climb step
            # (prevents the whole search from stalling)
            # ----- random perturbation ------------------------------------
            n = len(f)
            cand = f.copy()
            idxs = random.sample(range(n), k=min(5, n))
            for i in idxs:
                factor = 1.0 + random.gauss(0.0, 0.07)
                cand[i] = max(0.0, cand[i] * factor)
            f = _normalise_max_one(cand)
            continue

        # ----------------------------------------------------------------
        # 4) rescale g so that its total sum equals the sum of f
        # ----------------------------------------------------------------
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g == 0:
            # degenerate solution – abort the LP loop
            break
        g = g * (sum_f / sum_g)

        # ----------------------------------------------------------------
        # 5) mixing step
        # ----------------------------------------------------------------
        f_new = (1.0 - t) * f + t * g
        f = _normalise_max_one(f_new)

        # ----------------------------------------------------------------
        # 6) evaluate current candidate
        # ----------------------------------------------------------------
        sc = _score(f.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = f.tolist()
            print(f"[Iter {iteration:04d}] improvement → {best_score:.6f}  len={len(f)}")
            stagnation = 0
        else:
            stagnation += 1

        # ----------------------------------------------------------------
        # 7) adapt mixing factor – reduce it if we have been stagnant
        # ----------------------------------------------------------------
        if stagnation >= 12:
            t = max(0.05, t * 0.7)
            stagnation = 0

        # ----------------------------------------------------------------
        # 8) occasional random kick to escape flat regions (≈5 % of steps)
        # ----------------------------------------------------------------
        if random.random() < 0.05:
            n = len(f)
            m = random.randint(1, min(5, n))
            idxs = random.sample(range(n), m)
            for i in idxs:
                f[i] = max(0.0, f[i] * (1.0 + random.gauss(0.0, 0.12)))
            f = _normalise_max_one(f)

        # ----------------------------------------------------------------
        # 9) safety break – if the LP cost is already very close to the lower
        #    bound 1.0 we stop early.
        # ----------------------------------------------------------------
        if best_score < 1.52:                     # target is 1.503
            # continue a little longer to see if we can push further
            if time.time() > deadline - 30:
                break

    print(f"[Done] best score={best_score:.6f} (length={len(best_seq)})")
    return best_seq
```
