def simplex_projection(v):
    # v: numpy array, any real. Returns projection onto simplex sum=1, v_i >=0.
    # Standard algorithm:
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, n+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑score step‑function.

The algorithm proceeds in two phases.
1)  Random sampling of several parametric families (exponential, power‑law,
    linear, mixed, logistic, symmetric‑exponential, valley‑triangle,
    cosine‑valley, random decreasing).  The best candidate found is kept.
2)  A smooth‑max gradient‑descent refinement of the current best sequence.
    The objective  f(a)=2·n·max(a*a) is approximated by a soft‑max over the
    full convolution.  The gradient is obtained analytically via another
    convolution, and each step is projected onto the probability simplex
    (non‑negative entries summing to 1).  The learning rate decays and the
    soft‑max temperature (beta) is increased, so the optimisation eventually
    targets the true max.  The routine respects the 1000‑second budget and
    always returns the best sequence discovered.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:               # pragma: no cover
        return math.inf

def _project_simplex(v):
    """Project a vector onto the probability simplex (sum=1, v>=0)."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        # All entries are zero or negative; return uniform distribution.
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _normalize_sum(arr):
    """Make the vector non‑negative and sum to one (uniform fallback)."""
    arr = np.maximum(0.0, np.asarray(arr, dtype=float))
    s = arr.sum()
    if s > 0:
        arr = arr / s
    else:
        arr = np.full_like(arr, 1.0 / arr.size)
    return arr

def _exp_seq(n, alpha):
    """Exponential decay a[i] = exp(-alpha*i)."""
    return _normalize_sum(np.exp(-alpha * np.arange(n, dtype=float)))

def _pow_seq(n, p):
    """Power‑law decay a[i] = (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=float)
    return _normalize_sum(idx ** (-p))

def _lin_seq(n, beta):
    """Linear decay a[i] = max(0, 1 - beta*i)."""
    seq = np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float))
    return _normalize_sum(seq)

def _mix_seq(n, alpha, eps):
    """Mixed exponential + constant."""
    seq = np.exp(-alpha * np.arange(n, dtype=float)) + eps
    return _normalize_sum(seq)

def _logistic_seq(n, steep):
    """Logistic decay."""
    i = np.arange(n, dtype=float)
    mid = n / 2.0
    seq = 1.0 / (1.0 + np.exp((i - mid) / steep))
    return _normalize_sum(seq)

def _randdec_seq(n):
    """Random decreasing (sorted random numbers)."""
    arr = np.random.rand(n)
    arr.sort()
    arr = arr[::-1]
    return _normalize_sum(arr)

def _symexp_seq(n, alpha):
    """Symmetric exponential (high at both ends, low in middle)."""
    i = np.arange(n, dtype=float)
    seq = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    return _normalize_sum(seq)

def _valley_tri_seq(n):
    """Valley‑triangle: zero in centre, linear rise to the ends."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    seq = np.abs(i - centre) * 2.0 / (n - 1)   # goes from 1 at ends to 0 at centre
    return _normalize_sum(seq)

def _cosine_valley_seq(n):
    """Cosine valley: (1+cos(pi * i/(n-1)))/2 shifted to have minimum at centre."""
    i = np.arange(n, dtype=float)
    seq = (1.0 + np.cos(np.pi * i / (n - 1))) / 2.0   # 1 at ends, 0 at centre
    return _normalize_sum(seq)

# ----------------------------------------------------------------------
# Convolution helpers (FFT based – fast for n≈few‑thousand)
# ----------------------------------------------------------------------
def _full_convolve_fft(a, b):
    """Full convolution of two 1‑D real arrays using FFT."""
    size = a.size + b.size - 1
    nfft = 1 << ((size - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    conv = np.fft.irfft(A * B, nfft)[:size]
    return conv

def _smooth_max_weights(conv, beta):
    """Soft‑max weights for a convolution array."""
    m = conv.max()
    # subtract the max to avoid overflow
    exp_vals = np.exp(beta * (conv - m))
    w = exp_vals / exp_vals.sum()
    return w

def _grad_softmax_max(seq, weights):
    """
    Gradient of the smooth max w.r.t. seq.
    weights shape = (2*n-1,), seq shape = (n,).
    Returns array of shape (n,).
    """
    n = seq.size
    # reverse weights
    w_rev = weights[::-1]
    # convolution seq * w_rev (full)
    conv_rev = _full_convolve_fft(seq, w_rev)   # length = n + (2n-1) -1 = 3n-2
    # indices: len(weights)-1 - i  (i=0..n-1)
    idx = (len(weights) - 1) - np.arange(n)
    g = 2.0 * conv_rev[idx]
    return g

            print(f"[New] ({family}) n={n} → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – smooth‑max gradient descent refinement
    # --------------------------------------------------------------
    if best_seq is None:                     # fallback safety
        n = 2000
        best_seq = _randdec_seq(n).tolist()
        best_score = _score(best_seq)

    # Ensure we work with a NumPy array that sums to 1
    a = _project_simplex(np.asarray(best_seq, dtype=float))
    n = a.size
    beta_start = 5.0
    beta_end = 120.0
    lr0 = 0.04                               # initial learning rate

    max_iter = 8000                          # generous upper bound
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute full convolution a*a
        conv = _full_convolve_fft(a, a)      # length 2n-1
        M = conv.max()
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"[Grad] Iter {it:4d} → improved score {best_score:.6f}")

        # schedule for beta (smoothness) and learning rate
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / max_iter)
        lr = lr0 / math.sqrt(it + 1.0)

        # soft‑max weights over convolution entries
        weights = _smooth_max_weights(conv, beta)

        # gradient of the smooth max
        gM = _grad_softmax_max(a, weights)   # shape (n,)

        # gradient of the objective (2·n·M)
        grad = 2.0 * n * gM

        # gradient step & projection onto simplex
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)

        a = a_new
        it += 1

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
