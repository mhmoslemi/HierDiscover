"""
Search for a low‑score step‑function.

The routine first performs a broad random search over a richer set of
parametric families (exponential, power‑law, polynomial decay,
mixed exponential+constant, and a simple two‑level “block’’ shape).  The
search space is explored for as many candidates as the time budget
allows.  The best candidate found is then refined by a simulated‑annealing
hill‑climbing phase that repeatedly perturbs several coefficients,
accepting improving moves and, early on, occasional worsening moves.
All sequences are kept non‑negative; they are only rescaled so that the
maximum entry equals 1 (the objective is scaling‑invariant).  The function
returns the best sequence discovered before the deadline.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:               # pragma: no cover
        return math.inf

def _normalize_max(seq):
    """Scale so that max(seq) == 1 (no‑op if all zeros)."""
    arr = np.asarray(seq, dtype=float)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _exp_seq(n, alpha):
    """Exponential decay a_i = exp(-alpha * i)."""
    return _normalize_max(np.exp(-alpha * np.arange(n, dtype=float)))


def _pow_seq(n, p):
    """Power‑law decay a_i = (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=float)
    return _normalize_max(idx ** (-p))


def _poly_seq(n, p):
    """Polynomial decay a_i = (1 - i/(n-1))**p, 0 <= i < n."""
    # when p is large the tail becomes very small
    i = np.arange(n, dtype=float) / (n - 1)
    seq = (1.0 - i) ** p
    return _normalize_max(seq)


def _mix_seq(n, alpha, eps):
    """Mixed exponential + constant a_i = exp(-alpha*i) + eps."""
    seq = np.exp(-alpha * np.arange(n, dtype=float)) + eps
    return _normalize_max(seq)


def _block_seq(n, frac, beta):
    """
    Two‑level block shape.
    First ``frac * n`` entries are 1, the remainder are ``beta`` (0 ≤ beta ≤ 1).
    """
    m = max(1, int(round(frac * n)))
    tail = np.full(n - m, beta, dtype=float)
    head = np.ones(m, dtype=float)
    seq = np.concatenate([head, tail])
    return _normalize_max(seq)


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber
# ----------------------------------------------------------------------
def _perturb(seq, sigma=0.12, max_idx_change=8):
    """
    Produce a neighbour by mutating up to ``max_idx_change`` random indices.
    Multiplicative log‑normal noise with std ``sigma`` is applied.
    """
    cand = np.array(seq, copy=True)
    n = len(cand)
    m = random.randint(1, max_idx_change)
    idxs = random.sample(range(n), m)
    for i in idxs:
        factor = math.exp(random.gauss(0.0, sigma))
        cand[i] = max(0.0, cand[i] * factor)

    # occasional small mass‑transfer between two positions
    if random.random() < 0.2:
        i, j = random.sample(range(n), 2)
        transfer = cand[i] * random.random() * 0.3
        cand[i] -= transfer
        cand[j] += transfer

    # optional random reset of a single entry
    if random.random() < 0.1:
        k = random.randrange(n)
        cand[k] = random.random()

    # keep scale for numerical safety
    mx = cand.max()
    if mx > 0:
        cand = cand / mx
    return cand.tolist()


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The routine spends roughly ``budget_s`` seconds
    searching; it mixes an extensive random sampling of several families
    with a simulated‑annealing hill‑climb refinement of the best candidate
    found so far.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.3   # tiny safety margin

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 0) initialise from globally supplied construction if present
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = _normalize_max(list(height_sequence_1))
            sc = _score(init)
            if sc < best_score:
                best_seq, best_score = init, sc
                print(f"Initial reference score (global): {best_score:.6f}")
        except Exception:               # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1) random sampling of enriched parametric families
    # ------------------------------------------------------------------
    # we favour a few thousand entries – long enough to beat the 2‑bound
    lengths = [1800, 2200, 2600, 3000, 3400, 3800, 4200, 4600]

    families = ['exp', 'pow', 'poly', 'mix', 'block']

    # simple counter for progress logging
    it = 0
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # occasional aggressive hill‑climb on the current best
        # ------------------------------------------------------------------
        if best_seq is not None and random.random() < 0.12:
            cand = _perturb(best_seq, sigma=0.15, max_idx_change=10)
            sc = _score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc
                print(f"Hill‑climb improvement → {best_score:.6f}")
            continue

        # ------------------------------------------------------------------
        # generate a fresh candidate from one of the families
        # ------------------------------------------------------------------
        n = int(random.choice(lengths))
        family = random.choice(families)

        if family == 'exp':
            # log‑uniform α ∈ [1e‑4, 5]
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            cand = _exp_seq(n, alpha)

        elif family == 'pow':
            p = random.uniform(0.0, 3.0)          # exponent
            cand = _pow_seq(n, p)

        elif family == 'poly':
            # polynomial decay exponent p ∈ [0.5, 5]
            p = random.uniform(0.5, 5.0)
            cand = _poly_seq(n, p)

        elif family == 'mix':
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            # eps now explored on a log‑scale up to 5 (instead of 0.5)
            eps = 10 ** random.uniform(-4, math.log10(5))
            cand = _mix_seq(n, alpha, eps)

        else:  # 'block'
            # fraction of the block with value 1, beta ∈ [0,1]
            frac = random.uniform(0.05, 0.95)
            beta = random.uniform(0.0, 1.0)
            cand = _block_seq(n, frac, beta)

        # sanity guard (should never happen)
        if sum(cand) < 1e-12:
            continue

        sc = _score(cand)
        it += 1
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"New best ({family}) n={n} → {best_score:.6f}")

        # simple progress report every few thousand iterations
        if it % 3000 == 0:
            print(f"Iter {it}: best score so far {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) simulated‑annealing refinement on the current best
    # ------------------------------------------------------------------
    if best_seq is not None:
        cur_seq = best_seq[:]
        cur_score = best_score
        start = time.time()
        while time.time() < deadline:
            # cooling schedule – linearly decreasing temperature
            elapsed = time.time() - start
            progress = elapsed / budget_s
            temperature = max(0.001, 0.1 * (1.0 - progress))

            cand = _perturb(cur_seq, sigma=0.12, max_idx_change=8)
            cand_score = _score(cand)

            if cand_score < cur_score:
                cur_seq, cur_score = cand, cand_score
                if cand_score < best_score:
                    best_seq, best_score = cand, cand_score
                    print(f"Annealing improvement → {best_score:.6f}")
            else:
                # Metropolis acceptance
                prob = math.exp(-(cand_score - cur_score) / temperature)
                if random.random() < prob:
                    cur_seq, cur_score = cand, cand_score

    # ------------------------------------------------------------------
    # 3) final fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 800
        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
