"""
Propose a non‑negative step‑function sequence with a low evaluation score.

Algorithm
---------
1. Small‑scale linear‑programming refinement (n≈800)
   * Start from a cheap geometric / power‑law seed.
   * Repeatedly:
        – compute M = max_k (f*f)[k];
        – solve the LP  max  Σ g_j   s.t.   (f*g)_k ≤ M  ∀k,   g ≥ 0;
        – rescale g to the required ℓ₁‑norm (√(2n));
        – update f ← (1‑t)·f + t·g with a decreasing step size t.
   * Keep the best f seen.

2. Upscale to a larger problem (n≈3000) by linear interpolation,
   renormalise the ℓ₁‑norm and polish with projected‑gradient descent
   (the same scheme that achieved 1.5765 in the previous run).

Both phases respect the time budget (≈1000 s).  The LP phase reduces
the convolution peak dramatically, after which the gradient phase
fine‑tunes the shape on a longer sequence.  The final sequence is
returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s} (Duchi et al. 2008)."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")

    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1) × n) such that (f * g)[k] = (A·g)[k].
    For each column j of A the vector f is placed starting at row j.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    A.fill(0.0)
    # each column j contributes f to rows j … j+n‑1
    for j in range(n):
        A[j:j + n, j] = f
    return A

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list:
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – we need a little time for the interpreter to finish.
    overall_deadline = start_time + budget_s - 0.2

    ###########################################################################
    # Phase 1 – small‑scale LP refinement
    ###########################################################################
    n_small = 800
    target_small = np.sqrt(2.0 * n_small)      # required ℓ₁‑norm
    # ---- cheap geometric / power‑law seeds ---------------------------------
    best_score = float("inf")
    best_seq = None

    # geometric family a_i = C·r^i
    lambdas = np.logspace(-5, -0.5, 80)
    for lam in lambdas:
        r = 1.0 - lam
        r_n = r ** n_small
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = target_small * lam / denom
        seq = C * (r ** np.arange(n_small))
        val = evaluate_sequence(seq.tolist())
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # power‑law family a_i ∝ (i+1)^{-p}
    idx = np.arange(1, n_small + 1, dtype=float)
    for p in np.linspace(0.1, 1.5, 30):
        seq = 1.0 / (idx ** p)
        C = target_small / seq.sum()
        seq = C * seq
        val = evaluate_sequence(seq.tolist())
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    if best_seq is None:
        best_seq = np.full(n_small, target_small / n_small)

    # enforce exact ℓ₁‑norm (the evaluator already does it, but we keep it)
    best_seq = best_seq * (target_small / best_seq.sum())
    f = best_seq.copy()
    best_f = f.copy()
    print(f"[{time.time() - start_time:6.2f}s] phase‑1 start score = {best_score:.6f}")

    # LP loop parameters
    t = 0.25            # convex‑combination weight (will decay)
    it = 0
    # keep about 200 s for the final gradient phase
    phase1_deadline = overall_deadline - 200.0

    while time.time() < phase1_deadline:
        # 1) current maximal convolution value M
        conv = np.convolve(f, f)
        M = conv.max()
        # 2) build linear constraints (f * g) ≤ M
        A = _build_constraint_matrix(f)            # shape (2n‑1, n)
        rows = A.shape[0]
        # 3) linear program: maximise Σ g_j   ⇔ minimise −Σ g_j
        c = -np.ones(n_small, dtype=np.float64)
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0, None)] * n_small
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        if not res.success:
            print(f"  LP failed at iteration {it}, exiting LP phase.")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print(f"  LP returned zero sum at iteration {it}.")
            break

        # 4) scale to required ℓ₁‑norm
        g = g0 * (target_small / sum_g0)

        # 5) convex update
        f = (1.0 - t) * f + t * g

        # 6) current score (no extra call to evaluate_sequence)
        cur_score = 2.0 * n_small * M / (f.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP iter {it:5d} → new best {best_score:.6f}")

        # decay the step size slowly
        it += 1
        if it % 200 == 0:
            t = max(t * 0.85, 0.02)

    print(f"[{time.time() - start_time:6.2f}s] phase‑1 finished best = {best_score:.6f}")

    ###########################################################################
    # Phase 2 – upscale + projected gradient descent
    ###########################################################################
    remaining = overall_deadline - time.time()
    if remaining < 5.0:
        # not enough time – just return the small‑scale solution
        final_score = evaluate_sequence(best_f.tolist())
        print(f"Insufficient time for phase‑2, final score = {final_score:.6f}")
        return best_f.tolist()

    n_large = 3000
    target_large = np.sqrt(2.0 * n_large)

    # interpolate the best small sequence to the larger size
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    seq_large = np.interp(x_large, x_small, best_f)
    # normalise the ℓ₁‑norm
    seq_large = seq_large * (target_large / seq_large.sum())

    # projected‑gradient descent (same flavour as the previous code)
    curr = seq_large.copy()
    best_score_large = evaluate_sequence(curr.tolist())
    best_seq_large = curr.copy()

    it2 = 0
    step_scale = 0.5
    decay_interval = 200
    decay_factor = 0.85
    min_step = 1e-9
    fft_len = 1 << ((2 * n_large - 1).bit_length())

    while time.time() < overall_deadline:
        # convolution via FFT (linear, not circular)
        F = np.fft.rfft(curr, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n_large - 1]

        k_max = int(np.argmax(conv))
        max_val = conv[k_max]
        sum_curr = curr.sum()
        cur_score = 2.0 * n_large * max_val / (sum_curr ** 2)

        if cur_score < best_score_large:
            best_score_large = cur_score
            best_seq_large = curr.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD iter {it2:6d} → new best {best_score_large:.6f}")
            # tighten step size when we improve
            step_scale = max(step_scale * 0.9, min_step)

        # gradient of the max‑functional (sub‑gradient)
        idx = np.arange(n_large)
        j = k_max - idx
        mask = (j >= 0) & (j < n_large)
        grad = np.zeros_like(curr)
        grad[mask] = 2.0 * curr[j[mask]]

        # a small step in the opposite direction
        step = step_scale * target_large / n_large
        tentative = curr - step * grad / (np.linalg.norm(grad) + 1e-12)
        tentative = np.clip(tentative, 0.0, 1000.0)

        # project back onto the simplex {x ≥ 0, Σx = √(2n)}
        curr = projection_simplex(tentative, target_large)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.02:
            jitter = np.random.randn(n_large) * step_scale * target_large / (20.0 * n_large)
            curr = projection_simplex(np.clip(curr + jitter, 0.0, 1000.0), target_large)

        it2 += 1
        if it2 % decay_interval == 0:
            step_scale = max(step_scale * decay_factor, min_step)

    final_score = evaluate_sequence(best_seq_large.tolist())
    print(f"Search finished. Best score = {final_score:.6f}")
    return best_seq_large.tolist()
