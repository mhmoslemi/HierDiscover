"""
Improved search for a low‑value step‑function.

The routine combines three optimisation ideas tuned for the
scale‑invariant objective  

    E(a) = 2·n·max(a*a) / (sum(a)^2)   (sum(a)=1 ⇒ E = 2·n·max(a*a))

* **Projected sub‑gradient descent** – the same method as the reference
  but with a larger iteration budget, a slightly larger step‑size and
  occasional noise to escape flat regions.

* **Exponentiated gradient (mirror descent on the simplex)** – a
  multiplicative update that stays automatically on the probability
  simplex.  The gradient used is the sub‑gradient at the maximising
  convolution index.

* **Simple stochastic hill‑climbing** – after the continuous methods a
  short random local search moves a tiny amount of mass between two
  coordinates; if the move lowers the objective it is kept.

All three phases respect the 1000 s wall‑clock budget, switch to the
next phase when time is short and keep the best sequence found so far.
The code is fully self‑contained, uses only NumPy and the provided
`evaluate_sequence` function and prints modest progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.  Implementation follows Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using real FFT (fast for all lengths)."""
    m = len(a) + len(a) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> tuple[float, int]:
    """Return (max value, arg‑max index) of the full self‑convolution."""
    conv = convolution_fft(a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry (index k) w.r.t. a.
    For each i, ∂c_k/∂a_i = a_{k-i} (or 0 if out of range);
    the diagonal entry i=k/2 (when k even) gets doubled.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def compute_score(a: np.ndarray) -> float:
    """Fast proxy for `evaluate_sequence` when sum(a)=1."""
    n = len(a)
    conv = convolution_fft(a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val  # identical to evaluate_sequence for sum=1


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25                     # avoid exact zeros at the ends
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def improve_subgradient(initial_a: np.ndarray,
                        deadline: float,
                        max_iter: int = 300_000,
                        eta0: float = 1.0,
                        noise_every: int = 800,
                        noise_scale: float = 5e-5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on max(a*a).  Returns the best vector seen.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        max_val, k_max = max_convolution_and_index(a)

        grad = subgradient_of_max_at_index(a, k_max)
        eta = eta0 / np.sqrt(it)

        a_new = a - eta * grad
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)
        cur_score = compute_score(a_new)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"  Sub‑grad iter={it:,d}  cur={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score


def improve_exponentiated(initial_a: np.ndarray,
                          deadline: float,
                          max_iter: int = 300_000,
                          eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using the exponentiated‑gradient update.
    Uses the same sub‑gradient (the one at the maximising convolution index).
    """
    # keep strictly positive entries for the exponential map
    a = np.maximum(initial_a, 1e-12).astype(float)
    a = a / np.sum(a)

    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)      # multiplicative step
        a = a / np.sum(a)                # simplex normalisation

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"  EG iter={it:,d}  cur={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score


def hill_climb(start_a: np.ndarray,
               deadline: float,
               steps: int = 8000,
               move_factor: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Very lightweight stochastic local search.
    Repeatedly moves a small random amount of mass from a random index i
    to another index j.  Accepts the move only if it improves the score.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for s in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # amount to transfer: a fraction of a[i] (never exceeds a[i])
        eps = move_factor * a[i] * np.random.rand()
        if eps <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            a = a_new
        # otherwise keep the old a (no simulated‑annealing acceptance)

        if (s + 1) % 2000 == 0:
            print(f"  Hill‑climb step={s+1:,d}  cur={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score

    """
    Search for a sequence with a low `evaluate_sequence` value.

    The algorithm proceeds in three stages (sub‑gradient → exponentiated
    gradient → hill‑climbing) on a handful of problem sizes, always
    respecting the global time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback (use an already known construction if present)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # may be undefined
        fb_val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val = fb_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Problem sizes.  2048 is the sweet spot for a good trade‑off
    # between expressiveness and runtime; we also try 4096 if time permits.
    # ------------------------------------------------------------------
    lengths = [2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n=== Optimising length n={n} ===")

        # Two simple starting points:
        inits = [arcsine_initial(n), np.full(n, 1.0 / n)]

        for init_id, a0 in enumerate(inits, start=1):
            if time.time() >= deadline - 5.0:
                break

            kind = "arcsine" if init_id == 1 else "uniform"
            print(f"\n-- init {init_id} ({kind}) --")

            # -------- Sub‑gradient phase ----------
            seq_sg, val_sg = improve_subgradient(
                initial_a=a0,
                deadline=deadline,
                max_iter=300_000,
                eta0=1.0,
                noise_every=800,
                noise_scale=5e-5,
            )
            print(f"   Sub‑grad best: {val_sg:.6f}")
            if val_sg < best_val:
                best_val = val_sg
                best_seq = seq_sg.tolist()
                print(f"   *** Global best improved to {best_val:.6f} ***")

            # -------- Exponentiated‑gradient phase ----------
            seq_eg, val_eg = improve_exponentiated(
                initial_a=a0,
                deadline=deadline,
                max_iter=300_000,
                eta0=0.5,
            )
            print(f"   EG best: {val_eg:.6f}")
            if val_eg < best_val:
                best_val = val_eg
                best_seq = seq_eg.tolist()
                print(f"   *** Global best improved to {best_val:.6f} ***")

            # Choose the better of the two continuous runs as a seed for hill‑climb
            seed_vec = seq_sg if val_sg <= val_eg else seq_eg

            # -------- Hill‑climbing refinement ----------
            seq_hc, val_hc = hill_climb(
                start_a=seed_vec,
                deadline=deadline,
                steps=8000,
                move_factor=0.03,
            )
            print(f"   Hill‑climb best: {val_hc:.6f}")
            if val_hc < best_val:
                best_val = val_hc
                best_seq = seq_hc.tolist()
                print(f"   *** Global best improved to {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # In case nothing succeeded (should not happen), return a trivial list.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    # Final official evaluation (slow but definitive)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The competition expects a plain Python list of floats.
    return best_seq
