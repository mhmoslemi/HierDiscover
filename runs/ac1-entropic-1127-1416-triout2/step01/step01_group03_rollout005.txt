#!/usr/bin/env python3
"""
A fast Adam‑based optimiser for the self‑convolution L∞‑norm.

The code works with any length `n`.  If a global variable
`height_sequence_1` (the best construction discovered so far) exists,
its normalised version is used as a warm start.  Otherwise the classic
square‑root distribution (coefficients of (1‑z)^‑½) is taken.

The optimisation variable is `p = log(a)` where `a` lives on the
probability simplex (`a_i ≥ 0`, Σa_i = 1).  Using the soft‑max
approximation of the maximum,
    g(a) = Σ_k w_k (a*a)_k,
with   w_k = softmax_τ((a*a)_k) ,
the gradient w.r.t. `p` is obtained analytically and an Adam update
is applied.  The temperature `τ` is slowly increased so that the
surrogate tightens to the true max.  When the search stagnates a small
Gaussian jitter is added and the Adam moments are reset; after a long
stall a full random restart is performed.  The best sequence found
before the time budget expires is returned.

The implementation uses FFT‑based convolution (∼O(n log n)) and
keeps the evaluation of the objective cheap (no call to
`evaluate_sequence` inside the loop – the score is computed directly).

The function `propose_candidate` is the entry point required by the
harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level (required by the platform)
# ----------------------------------------------------------------------


def sqrt_distribution(n: int) -> np.ndarray:
    """
    Normalised coefficients of (1‑z)^(-1/2) up to length n.
    The raw coefficients are c_0=1, c_{i}=c_{i‑1}·(2i‑1)/(2i).
    After the loop they are divided by their sum so that Σc_i = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a power‑of‑two FFT.
    Returns an array of length len(x) + len(y) - 1.
    """
    n = len(x) + len(y) - 1
    # next power of two for efficient FFT
    N = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    c = np.fft.irfft(Z, N)
    return c[:n]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds,
    prints a few progress lines and finally returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5               # tiny safety margin

    # ------------------------------------------------------------------
    # Choose the length of the sequence; prefer an existing construction.
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` may be supplied by the surrounding harness.
        init = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if init.size == 0:
            raise ValueError
        # enforce non‑negativity and normalise to a probability vector
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s == 0.0:
            raise ValueError
        a = init / s
        n = a.size
    except Exception:
        # fall back to a reasonable default length
        n = 2000
        a = sqrt_distribution(n)

    # ------------------------------------------------------------------
    # Parameterisation: p = log(a)  (softmax ensures simplex)
    # ------------------------------------------------------------------
    eps = 1e-16
    p = np.log(a + eps)                       # start point in the log‑space

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    alpha = 0.5                               # base step size for Adam

    # temperature for the soft‑max (approximates the true max)
    tau = 0.5
    tau_max = 30.0
    tau_inc = 1.001                           # multiplicative increase per iteration

    # bookkeeping
    best_a = a.copy()
    # direct computation of the current objective (no extra call to evaluate_sequence)
    cur_b = conv_fft(best_a, best_a)
    best_score = 2.0 * n * cur_b.max() / (best_a.sum() ** 2)
    stagnant = 0
    it = 0

    print(f"[init] n={n}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- build a from p (softmax) ---------------------------------
        p_shift = p - np.max(p)            # improve numerical stability
        a = np.exp(p_shift)
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full(n, 1.0 / n)
            sum_a = 1.0
        else:
            a = a / sum_a

        # ----- convolution a * a ---------------------------------------
        b = conv_fft(a, a)                 # length 2n‑1
        max_b = b.max()

        # ----- soft‑max weights -----------------------------------------
        # w_k ∝ exp( τ·(b_k – max_b) )   (all ≤ 0, safe from overflow)
        w = np.exp(tau * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate wrt a -------------------
        # ∇_a g = 2 * (a *conv* w_rev)
        w_rev = w[::-1]
        conv_rev = conv_fft(a, w_rev)                     # length 3n‑2
        L = b.size                                         # = 2n‑1
        idx = (L - 1) - np.arange(n)                      # positions that give ∑_j a_j w_{i+j}
        grad_a = 2.0 * conv_rev[idx]

        # ----- map gradient onto p (mirror‑descent on the simplex) -----
        avg = np.dot(a, grad_a)            # scalar = Σ a_j ∂g/∂a_j
        grad_p = a * (grad_a - avg)        # ∂g/∂p_i

        # ----- Adam update on p -----------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

        # keep p in a numerically comfortable range
        np.clip(p, -30.0, 30.0, out=p)

        # ----- current objective (exact, no extra convolution) ----------
        cur_score = 2.0 * n * max_b / (sum_a * sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnant = 0
            print(f"[{it:07d}] improved score = {best_score:.6f}  (tau={tau:.3f})")
        else:
            stagnant += 1

        # ----- temperature schedule --------------------------------------
        tau = min(tau * tau_inc, tau_max)

        # ----- occasional jitter to escape plateaus ----------------------
        if stagnant > 2000:
            sigma = 0.05 * (1.0 + math.log1p(stagnant))
            p += sigma * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            stagnant = 0
            print(f"[{it:07d}] jitter applied (sigma={sigma:.4f})")

        # ----- full random restart after a very long stall ---------------
        if stagnant > 15000:
            p = np.log(np.random.rand(n) + eps)
            m.fill(0.0)
            v.fill(0.0)
            tau = 0.5
            stagnant = 0
            print(f"[{it:07d}] full random restart")

    # ------------------------------------------------------------------
    # Return the best sequence obtained.
    # ------------------------------------------------------------------
    print(f"[done] iterations={it}, best score={best_score:.6f}")
    # Ensure the list respects the evaluator's sanity checks
    best_a = np.clip(best_a, 0.0, 1000.0)
    # (evaluate_sequence would also clip negatives and huge values,
    #  but our construction stays well inside the bounds.)
    return best_a.tolist()
