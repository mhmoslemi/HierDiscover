"""
Propose a low‑score step‑function.

The search combines three stages:
1.  Random sampling of a richer set of parametric families
    (exponential, exponential with a floor, a two‑exponential mixture,
    power‑law, linear, logistic, mixed exponential+constant and random decreasing).
    Hyper‑parameters (length, decay rates, floor, mixture weight, etc.) are drawn
    from wide log‑uniform ranges.  The best candidate is kept.

2.  Targeted stochastic refinement: the current best sequence is repeatedly
    perturbed on indices that contribute most to the current maximal convolution
    value.  This focuses the hill‑climb on the “bottleneck’’ lag and typically
    reduces the objective further.

3.  Generic hill‑climbing (random multiplicative tweaks) as a fallback.

All operations respect the non‑negative requirement; each candidate is normalised
so that its maximum entry is 1 – the evaluation function is scale‑invariant, this
avoids overflow and keeps the numbers in the safe range of ``evaluate_sequence``.
The routine stops when the allotted time budget expires and returns the best
sequence discovered.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------
def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:               # pragma: no cover
        return math.inf


def _normalize(seq):
    """Scale so that max(seq) == 1 (no‑op if all zeros)."""
    arr = np.asarray(seq, dtype=float)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()


def _exp_seq(n, alpha):
    """Simple exponential decay a_i = exp(-alpha * i)."""
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)))


def _pow_seq(n, p):
    """Power‑law decay a_i = (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=float)
    return _normalize(idx ** (-p))


def _lin_seq(n, beta):
    """Linear decay a_i = max(0, 1 - beta*i)."""
    return _normalize(np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float)))


def _mix_seq(n, alpha, eps):
    """Mixed exponential + constant a_i = exp(-alpha*i) + eps."""
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)) + eps)


def _logistic_seq(n, steep):
    """Logistic decay a_i = 1/(1+exp((i-mid)/steep))."""
    i = np.arange(n, dtype=float)
    mid = n / 2.0
    seq = 1.0 / (1.0 + np.exp((i - mid) / steep))
    return _normalize(seq)


def _randdec_seq(n):
    """Random decreasing sequence (sorted random numbers)."""
    arr = np.random.rand(n)
    arr.sort()
    arr = arr[::-1]                     # descending
    return _normalize(arr)


def _floor_exp_seq(n, alpha, eps):
    """Exponential with a hard floor: a_i = max(exp(-alpha*i), eps)."""
    arr = np.exp(-alpha * np.arange(n, dtype=float))
    arr = np.maximum(arr, eps)
    return _normalize(arr)


def _double_exp_seq(n, alpha1, alpha2, w):
    """Mixture of two exponentials: a_i = exp(-alpha1*i) + w*exp(-alpha2*i)."""
    arr = np.exp(-alpha1 * np.arange(n, dtype=float)) + \
          w * np.exp(-alpha2 * np.arange(n, dtype=float))
    return _normalize(arr)


def _targeted_perturb(seq):
    """
    Produce a small perturbation that tries to lower the current maximal
    convolution entry.  The indices participating in that entry are identified
    and a few of them are scaled multiplicatively.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())                     # lag with maximal value
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idxs = np.arange(i_start, i_end + 1, dtype=int)

    # contribution of each pair (i, k-i) to conv[k]
    contrib = np.asarray(seq)[idxs] * np.asarray(seq)[k - idxs]

    # pick up to 5 indices with the largest contribution
    m = random.randint(1, min(5, len(idxs)))
    top = idxs[np.argpartition(contrib, -m)[-m:]]

    cand = np.array(seq, copy=True)
    for i in top:
        factor = 1.0 + random.gauss(0.0, 0.09)   # modest random change
        cand[i] = max(0.0, cand[i] * factor)

    # renormalise max to 1 (scale does not affect the score)
    mx = cand.max()
    if mx > 0:
        cand = cand / mx
    return cand.tolist()


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The routine spends roughly ``budget_s`` seconds
    searching; it mixes random sampling of an extended set of parametric families
    with a targeted stochastic hill‑climbing phase that focuses on the current
    maximal convolution lag.
    """
    np.random.seed(seed)
    random.seed(seed)

    # deadline a little before the official timeout to avoid overshoot
    deadline = time.time() + budget_s - 0.3

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 0) start from the globally supplied construction if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = _normalize(list(height_sequence_1))
            sc = _score(init)
            if sc < best_score:
                best_seq, best_score = init, sc
                print(f"Initial reference score: {best_score:.6f}")
        except Exception:               # pragma: no cover
            pass

    # lengths to explore – a few thousand entries give a good trade‑off
    lengths = [1200, 1600, 2000, 2400, 2800, 3200, 3600, 4000]

    # ------------------------------------------------------------------
    # 1) random sampling of parametric families
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        #   a) occasional targeted hill‑climb on the current best
        # ------------------------------------------------------------------
        if best_seq is not None and random.random() < 0.18:
            cand = _targeted_perturb(best_seq)
            sc = _score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand
                print(f"Targeted improvement → {best_score:.6f}")
                continue
            # fall through to generic hill‑climb with small probability
            if random.random() < 0.25:
                # generic multiplicative tweak (few positions)
                cand2 = np.array(best_seq, copy=True)
                n = len(cand2)
                m = random.randint(1, min(5, n))
                idxs = random.sample(range(n), m)
                for i in idxs:
                    factor = 1.0 + random.gauss(0.0, 0.07)
                    cand2[i] = max(0.0, cand2[i] * factor)
                cand2 = cand2 / cand2.max()
                cand2 = cand2.tolist()
                sc2 = _score(cand2)
                if sc2 < best_score:
                    best_score, best_seq = sc2, cand2
                    print(f"Generic hill‑climb → {best_score:.6f}")
                continue

        # ------------------------------------------------------------------
        #   b) generate a fresh candidate from a parametric family
        # ------------------------------------------------------------------
        n = int(random.choice(lengths))
        family = random.choice(['exp', 'pow', 'lin', 'mix',
                                'floor', 'double', 'log', 'randdec'])

        if family == 'exp':
            # log‑uniform α ∈ [1e‑4, 5]
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            cand = _exp_seq(n, alpha)

        elif family == 'pow':
            p = random.uniform(0.0, 3.0)          # exponent
            cand = _pow_seq(n, p)

        elif family == 'lin':
            beta = random.uniform(0.0, 2.0 / n)   # slope – keep a few positives
            cand = _lin_seq(n, beta)

        elif family == 'mix':
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            eps = random.uniform(0.0, 0.5)
            cand = _mix_seq(n, alpha, eps)

        elif family == 'floor':
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            eps = random.uniform(0.0, 0.5)        # floor value
            cand = _floor_exp_seq(n, alpha, eps)

        elif family == 'double':
            # two exponentials with independent decay rates
            log_a1 = random.uniform(-4, math.log10(3))
            log_a2 = random.uniform(-4, math.log10(3))
            alpha1 = 10 ** log_a1
            alpha2 = 10 ** log_a2
            w = random.uniform(0.0, 2.0)          # relative weight of the second term
            cand = _double_exp_seq(n, alpha1, alpha2, w)

        elif family == 'log':
            steep = random.uniform(0.01, n / 2.0)
            cand = _logistic_seq(n, steep)

        else:  # 'randdec'
            cand = _randdec_seq(n)

        # Guard against pathological all‑zero vectors (should not happen)
        if sum(cand) < 1e-8:
            continue

        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"New best ({family}) n={n} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) final fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 500
        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
