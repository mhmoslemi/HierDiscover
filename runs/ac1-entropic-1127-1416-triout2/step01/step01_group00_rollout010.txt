def optimize_subgradient(initial_a, n, deadline, max_iter=300000, eta0=0.5, noise_every=500, noise_scale=1e-4):

def hill_climb(seq, n, deadline, steps=20000, delta_frac=0.01):
    a = seq.copy()
    a = a / np.sum(a)  # ensure sum=1
    best_a = a.copy()
    best_max, _ = max_convolution_and_index(a)
    for step in range(steps):
        if time.time() >= deadline - 1:
            break
        # pick random indices i,j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # compute amount delta to move
        delta = delta_frac * np.random.rand() * a[i]
        # propose move mass from i to j
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non-neg
        if a_new[i] < 0:
            continue
        a_new = a_new / np.sum(a_new)  # maintain sum=1 (should be same)
        cur_max, _ = max_convolution_and_index(a_new)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a_new.copy()
            a = a_new
        else:
            # maybe accept with small probability? skip
            pass
        if step % 2000 == 0:
            print(f"    hillclimb step={step} best_max={best_max:.6f}")
    return best_a, best_max

"""
Improved search for a non‑negative step‑function that minimises
evaluate_sequence.  The algorithm combines two simplex‑constrained
optimisers:

* Projected Euclidean sub‑gradient descent on the max‑self‑convolution.
* Exponentiated‑gradient (mirror descent with KL‑projection).

Both methods use a fast FFT convolution to obtain the maximal
convolution entry and its index, which supplies a (sub)‑gradient.
The search is performed for several sequence lengths (up to 8192);
the best candidate found before the time budget expires is returned.
A final short local hill‑climb refines the best sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def max_convolution_and_index(a: np.ndarray) -> tuple[float, int]:
    """
    Returns (max value, index) of the full self‑convolution a*a.
    Uses a single forward FFT (A) and one inverse FFT.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k)
    with respect to a.  For the self‑convolution the gradient is
    2·a_{k‑i} for all i where the index is in range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Discrete analogue of the arcsine density:
        a_i ∝ 1 / sqrt((i+0.5)*(n-0.5-i) + eps)
    The eps term prevents division by zero at the extremes.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def uniform_initial(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet_initial(n: int) -> np.ndarray:
    """Random point on the simplex (Dirichlet(1,…,1))."""
    return np.random.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def optimize_subgradient(
    init_a: np.ndarray,
    n: int,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.8,
    noise_every: int = 500,
    noise_scale: float = 5e-5,
) -> tuple[np.ndarray, float]:
    """
    Projected Euclidean sub‑gradient descent.
    Returns the best probability vector (sum = 1) and its maximal
    convolution value.
    """
    a = project_to_simplex(init_a.astype(float))
    best_a = a.copy()
    best_max, _ = max_convolution_and_index(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        cur_max, k = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k)

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale

        a = project_to_simplex(a)

        cur_max, _ = max_convolution_and_index(a)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        if it % 20000 == 0:
            print(f"    subgrad n={n} iter={it:,d} cur_max={cur_max:.6f} best_max={best_max:.6f}")

    return best_a, best_max


def optimize_exponential(
    init_a: np.ndarray,
    n: int,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.6,
    noise_every: int = 500,
    noise_scale: float = 5e-5,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex with KL‑divergence (Exponentiated
    Gradient).  Returns the best probability vector and its maximal
    convolution value.
    """
    a = project_to_simplex(init_a.astype(float))
    best_a = a.copy()
    best_max, _ = max_convolution_and_index(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        cur_max, k = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k)

        eta = eta0 / np.sqrt(it)
        # Clip gradients to avoid numerical overflow in exp()
        grad_clip = np.clip(grad, -10.0, 10.0)
        a = a * np.exp(-eta * grad_clip)

        if noise_every and (it % noise_every == 0):
            a += np.random.rand(n) * noise_scale

        s = np.sum(a)
        if s == 0.0:
            a = uniform_initial(n)
        else:
            a = a / s  # KL‑projection (normalisation)

        cur_max, _ = max_convolution_and_index(a)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        if it % 20000 == 0:
            print(f"    expgrad n={n} iter={it:,d} cur_max={cur_max:.6f} best_max={best_max:.6f}")

    return best_a, best_max


def hill_climb(
    init_a: np.ndarray,
    n: int,
    deadline: float,
    steps: int = 20_000,
    delta_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Very simple stochastic hill‑climb that moves a small amount of mass
    from one random coordinate to another, accepting only improving
    moves.
    """
    a = init_a.copy()
    a = a / np.sum(a)  # ensure normalised
    best_a = a.copy()
    best_max, _ = max_convolution_and_index(a)

    for step in range(steps):
        if time.time() >= deadline - 1.0:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # propose moving a small fraction of mass from i to j
        delta = delta_frac * np.random.rand() * a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0.0:
            continue

        a_new = a_new / np.sum(a_new)
        cur_max, _ = max_convolution_and_index(a_new)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a_new.copy()
            a = a_new  # keep the improvement

        if step % 2000 == 0 and step > 0:
            print(f"    hillclimb step={step:,d} best_max={best_max:.6f}")

    return best_a, best_max
