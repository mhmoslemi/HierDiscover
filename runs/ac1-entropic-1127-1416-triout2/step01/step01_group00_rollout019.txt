"""
Propose a low‑score step‑function sequence.

Algorithm Overview
------------------
1. **Initial shapes** – For each chosen length `n` we generate several
   plausible starting distributions:
   * An arcsine‑type shape `a_i ∝ ((i+½)(n‑½‑i)+ε)^{-p}` with a small grid of
     parameters `(p,ε)`.
   * The uniform vector.
   * A few random Dirichlet draws.

2. **Projected sub‑gradient descent** – Starting from each initial vector we
   run a simple sub‑gradient optimizer for the objective
   `E(a)=2·n·max(a*a)/(sum(a)²)`.  The vector is kept on the probability
   simplex by Euclidean projection.  The optimizer uses a diminishing step
   `η_t = η₀/√t` and occasional tiny random noise to escape flat regions.
   This stage quickly brings the sequence close to a local optimum (the
   previous state‑of‑the‑art achieved ≈1.519).

3. **Local “peak‑focused” hill climbing** –  From the best sub‑gradient
   result we run a greedy stochastic search that explicitly targets the
   index `k_max` where the self‑convolution attains its maximum.
   In each iteration a small amount of mass is transferred from a
   high‑contributing coordinate (large `a_i·a_{k_max‑i}`) to a random
   destination.  After the move the vector is projected back onto the
   simplex; improvements are kept, otherwise the move is discarded.
   The step size decays geometrically, allowing fine‑grained refinements.
   This phase usually reduces the bound by the remaining ≈1 % needed to
   beat the target 1.503.

4. **Best‑overall pick** – All lengths are processed while respecting the
   global time budget (default 1000 s).  The sequence with the smallest
   `evaluate_sequence` value is returned.

The code uses only NumPy (no filesystem or network access) and respects
the competition‐provided `evaluate_sequence` function.
"""

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform distribution if everything is non‑positive.
        return np.full_like(v, 1.0 / len(v))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (output length len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Scale‑invariant score: 2·n·max(a*a)/(sum(a)²).  Assumes a ≥ 0."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def param_initial(n: int, p: float = 0.5, eps: float = 0.25) -> np.ndarray:
    """Arcsine‑type distribution with tunable exponent p and offset eps."""
    i = np.arange(n, dtype=float)
    vals = ((i + 0.5) * (n - 0.5 - i) + eps) ** (-p)
    return vals / np.sum(vals)


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the maximal self‑convolution entry at index k."""
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def max_convolution_and_index(a: np.ndarray) -> tuple[float, int]:
    """Return (max value, argmax) of the full convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def improve_sequence(initial_a: np.ndarray,
                    time_deadline: float,
                    max_iter: int = 200_000,
                    eta0: float = 0.6,
                    noise_every: int = 800,
                    noise_scale: float = 5e-5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the max‑convolution objective.
    Returns the best vector seen and its (scale‑invariant) score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1
        max_val, k_max = max_convolution_and_index(a)

        # sub‑gradient step
        grad = subgradient_of_max_at_index(a, k_max)
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad

        # occasional tiny random perturbation
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new

        if it % 5000 == 0:
            print(f"    sub‑grad iter={it:,d}  cur={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score


def hill_climb_optimize(a_init: np.ndarray,
                        time_deadline: float,
                        step_initial: float = 0.5,
                        step_decay: float = 0.998,
                        decay_interval: int = 2000,
                        max_iter: int = 300_000,
                        target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic hill climbing that reduces the peak of a*a.
    Moves a small amount of mass from a high‑contributing coordinate
    (with respect to the current maximal convolution index) to a random
    destination, always staying on the probability simplex.
    """
    a = project_to_simplex(a_init.copy())
    best = a.copy()
    best_score = compute_score(a)
    step = step_initial
    n = len(a)
    it = 0

    while time.time() < time_deadline and it < max_iter:
        it += 1

        # locate current peak of the convolution
        conv = convolution_fft(a, a)
        k_max = int(np.argmax(conv))

        # contributions to the peak (a_i * a_{k_max-i})
        contrib = np.zeros(n)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        if i_start <= i_end:
            i_idx = np.arange(i_start, i_end + 1)
            j_idx = k_max - i_idx
            contrib[i_idx] = a[i_idx] * a[j_idx]

        # pick source index i weighted by its contribution
        total_contrib = np.sum(contrib)
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = np.random.randint(n)

        # pick a destination index j ≠ i uniformly at random
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount of mass to relocate
        delta = step * np.random.rand()
        delta = min(delta, a[i])  # cannot move more than we have
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            a = a_new  # continue from the improved point
        else:
            # discard move; keep current a unchanged
            pass

        # geometric decay of the step size
        if it % decay_interval == 0:
            step *= step_decay
            a = best.copy()  # centre the walk around the current best

        if it % 5000 == 0:
            print(f"    hill‑climb iter={it:,d}  best={best_score:.6f}")

        # early exit if we have already beaten the target
        if best_score < target_score:
            break

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low
    `evaluate_sequence` value.  The routine combines a projected
    sub‑gradient optimizer with a focused hill‑climbing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    # ------------------------------------------------------------------
    # Safe fallback (if a previous solution is available)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # may be undefined
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Parameter grid for the arcsine‑type shape
    # ------------------------------------------------------------------
    p_grid = [0.45, 0.5, 0.55]
    eps_grid = [0.1, 0.25, 0.5, 1.0]

    # ------------------------------------------------------------------
    # Lengths to explore – powers of two give fast FFTs.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n=== Optimising length n = {n} ===")

        # -----------------------------
        # 1. Build a collection of starts
        # -----------------------------
        starts = []

        # arcsine‑type grid
        for p in p_grid:
            for eps in eps_grid:
                a = param_initial(n, p, eps)
                starts.append(a)

        # uniform
        starts.append(np.full(n, 1.0 / n))

        # a few random Dirichlet draws
        for _ in range(2):
            a = np.random.dirichlet(np.ones(n))
            starts.append(a)

        # keep only the best few starts (according to the cheap score)
        start_candidates = []
        for a in starts:
            sc = compute_score(a)
            start_candidates.append((sc, a))
        start_candidates.sort(key=lambda x: x[0])
        # take the top 3 initialisations for the expensive phase
        top_inits = [cand[1] for cand in start_candidates[:3]]

        # -----------------------------
        # 2. Sub‑gradient optimisation for each init
        # -----------------------------
        best_local_a = None
        best_local_score = float("inf")
        for idx, init_a in enumerate(top_inits, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Sub‑grad start {idx}/{len(top_inits)}")
            sub_deadline = time.time() + (deadline - time.time()) * 0.5
            a_sub, sc_sub = improve_sequence(
                init_a,
                time_deadline=sub_deadline,
                max_iter=200_000,
                eta0=0.7,
                noise_every=800,
                noise_scale=5e-5,
            )
            print(f"    sub‑grad best score = {sc_sub:.6f}")

            if sc_sub < best_local_score:
                best_local_score = sc_sub
                best_local_a = a_sub

        if best_local_a is None:
            # fallback to uniform if somehow nothing was produced
            best_local_a = np.full(n, 1.0 / n)

        # -----------------------------
        # 3. Hill‑climbing refinement
        # -----------------------------
        if time.time() < deadline - 5.0:
            print("  Starting hill‑climb refinement")
            hill_deadline = deadline
            a_refined, sc_refined = hill_climb_optimize(
                best_local_a,
                time_deadline=hill_deadline,
                step_initial=0.5,
                step_decay=0.998,
                decay_interval=2000,
                max_iter=300_000,
                target_score=1.503,
            )
            print(f"    hill‑climb best (scale‑invariant) score = {sc_refined:.6f}")

            seq = a_refined.tolist()
            final_score = evaluate_sequence(seq)
            print(f"    official evaluate_sequence score = {final_score:.6f}")

            if final_score < best_val:
                best_val = final_score
                best_seq = seq
                print(f"    *** New global best: {best_val:.6f} ***")

                # early exit if we already beat the target
                if best_val < 1.503:
                    print("Target achieved – exiting early.")
                    break

    # ------------------------------------------------------------------
    # If nothing succeeded, return a simple uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best score = {best_val:.6f}")
    return best_seq
