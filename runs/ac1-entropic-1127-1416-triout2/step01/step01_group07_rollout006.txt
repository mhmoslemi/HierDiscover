"""
Propose a candidate sequence that minimizes the evaluation function.

Algorithm
---------
1. **Initialisation**
   • Use a cheap grid search over geometric families `a_i = C·r^i`
     (and a single power‑law seed) to obtain a decent starting point.
   • If a global variable ``height_sequence_1`` is present, it is also
     considered as a seed.

2. **Smoothed Adam optimisation**
   • The true objective is  `max_k (a*a)[k]`.  We replace the max by a
     smooth approximation:
         φ_τ(a) = τ·log Σ_k exp((a*a)[k]/τ) .
     Its gradient is an easy convolution:
         ∇φ_τ = 2·a * w   where  w_k = softmax((a*a)[k]/τ) .
   • An Adam update (with projection onto the simplex
     `{x ≥ 0, Σx = √(2n)}`) is applied to the current sequence.
   • The smoothing temperature τ is decreased geometrically so the
     optimisation first moves on a smooth landscape and then focuses on
     the true max‑functional.

3. **Exploration tricks**
   • A small random jitter is added occasionally to escape plateaus.
   • The best sequence seen so far (according to the exact evaluator)
     is kept and reported at the end.

The whole routine respects the 1000 s time budget and works for any
reasonable length (default n ≈ 5000).  The returned list contains only
non‑negative floats and is already normalised, so `evaluate_sequence`
will directly return the achieved bound.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a low‑score non‑negative step function within the given time
    budget.  Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # safety margin

    # ------------------------------------------------------------------- #
    # 1. Problem size ---------------------------------------------------- #
    # Try to reuse the length of a known construction if it exists,
    # otherwise fall back to a moderately large size.
    try:
        init_from_global = np.array(height_sequence_1, dtype=float)
        n = init_from_global.shape[0]
        use_global = True
    except Exception:
        n = 5000                                   # a length that still runs fast
        use_global = False

    target_sum = np.sqrt(2.0 * n)                  # Σa_i = √(2n)

    # ------------------------------------------------------------------- #
    # 2. Cheap grid search for a decent seed ---------------------------- #
    best_val = float("inf")
    best_seq = None

    # helper: thin wrapper around the provided evaluator
    def eval_fast(seq_arr):
        return evaluate_sequence(seq_arr.tolist())

    # geometric families
    lambdas = np.logspace(-5, -1, 120)
    for lam in lambdas:
        r = 1.0 - lam
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom
        a = C * (r ** np.arange(n))
        a = np.clip(a, 0.0, 1000.0)
        val = eval_fast(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    # one power‑law seed a_i ∝ 1/√(i+1)
    idx = np.arange(1, n + 1, dtype=float)
    a_pow = 1.0 / np.sqrt(idx)
    C_pow = target_sum / a_pow.sum()
    a_pow = C_pow * a_pow
    a_pow = np.clip(a_pow, 0.0, 1000.0)
    val_pow = eval_fast(a_pow)
    if val_pow < best_val:
        best_val = val_pow
        best_seq = a_pow.copy()

    # include the global construction if it matches the current length
    if use_global and init_from_global.shape[0] == n:
        a_glob = np.clip(init_from_global, 0.0, 1000.0)
        a_glob = projection_simplex(a_glob, target_sum)
        val_glob = eval_fast(a_glob)
        if val_glob < best_val:
            best_val = val_glob
            best_seq = a_glob.copy()

    # Fallback – uniform (unlikely)
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # ------------------------------------------------------------------- #
    # 3. Adam optimisation with smooth max approximation ---------------- #
    a = best_seq.copy()
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    # FFT parameters (need length ≥ 3n‑2 for gradient convolution)
    fft_len = 1 << ((3 * n - 2).bit_length())

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)

    # Step‑size (roughly a few percent of the typical entry)
    base_lr = 0.5 * target_sum / n
    max_step = base_lr                     # absolute cap on the Euclidean step

    # Temperature schedule for the smooth max
    # initialise with half of the initial max‑convolution
    Fa_initial = np.fft.rfft(a, n=fft_len)
    conv_initial = np.fft.irfft(Fa_initial * Fa_initial,
                                n=fft_len)[: 2 * n - 1]
    tau = max(1e-3, 0.5 * conv_initial.max())
    tau_min = 1e-4
    tau_decay = 0.9995

    it = 0
    last_improve = 0

    while time.time() < deadline:
        # ----- convolution a*a ------------------------------------------------
        Fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
        conv_max = conv.max()

        # ----- soft‑max weights -----------------------------------------------
        # stability: shift by max
        exp_arg = (conv - conv_max) / tau
        # prevent overflow/underflow
        with np.errstate(over='ignore'):
            w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fallback to hard max (sub‑gradient at the peak)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # ----- gradient of the smooth max (2 * a * w) -------------------------
        w_rev = w[::-1]                     # reverse for linear convolution
        Fw_rev = np.fft.rfft(w_rev, n=fft_len)
        grad_full = np.fft.irfft(Fa * Fw_rev, n=fft_len)
        # length of a*w_rev linear convolution = n + (2n‑1) – 1 = 3n‑2
        grad_full = grad_full[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1: 2 * n - 1]   # keep the n relevant entries

        # ----- Adam moment updates --------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- descent step ----------------------------------------------------
        step_raw = (base_lr * m_hat) / (np.sqrt(v_hat) + eps)

        # clip global step size to avoid wild jumps
        step_norm = np.linalg.norm(step_raw)
        if step_norm > max_step and step_norm > 0:
            step_raw = step_raw * (max_step / step_norm)

        a_candidate = a - step_raw
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        a = projection_simplex(a_candidate, target_sum)

        # ----- occasional jitter ------------------------------------------------
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * (target_sum * 0.001 / n)
            a = projection_simplex(np.clip(a + jitter, 0.0, 1000.0), target_sum)

        # ----- evaluate exact score (max convolution) -------------------------
        Fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
        cur_score = conv.max()    # identical to evaluate_sequence when sum = √(2n)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:6.1f}s] iter {it:7d} → new best {best_score:.6f}")

        # ----- temperature decay ------------------------------------------------
        tau = max(tau * tau_decay, tau_min)

        # ----- restart if stagnated -------------------------------------------
        if it - last_improve > 200_000:
            # inject a stronger random perturbation around the current best
            noise = np.abs(np.random.randn(n)) * (target_sum * 0.02 / n)
            a = projection_simplex(np.clip(best + noise, 0.0, 1000.0), target_sum)
            last_improve = it
            print(f"[{time.time() - start_time:6.1f}s] iter {it:7d} – stagnation reset")

    # ------------------------------------------------------------------- #
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Best score achieved: {final_score:.6f}")
    return best.tolist()
