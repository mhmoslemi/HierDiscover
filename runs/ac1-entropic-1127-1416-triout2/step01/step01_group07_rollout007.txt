"""
Propose a candidate step‑function that (empirically) beats the previous best
score (~1.576).  

Algorithm
---------
1. Choose a moderate length n (power‑of‑2 for fast FFT).  All candidates are
   normalised to the same ℓ₁‑norm √(2n); the evaluation is scale‑invariant.
2. **Rapid parametric search** – generate candidates from a few simple families
   (geometric, double‑exponential, power‑law, logistic, two‑level constant).
   Each candidate is evaluated with a fast FFT‑based convolution; the best
   candidate becomes the starting point for the refinement.
3. **Adam‑style projected optimisation** – minimise the quantity
   `max_k (a*a)[k]` on the simplex `{a≥0, Σa = √(2n)}`.
   The sub‑gradient of the max‑functional at the active index k* is
   `g_i = 2·a_{k*‑i}` (zero when the index is out of range).  An Adam update
   is performed, after which the vector is projected back onto the simplex.
   Occasionally a random jitter is added to escape plateaus.
4. The best sequence seen during the whole run is returned.

The whole routine respects the 1000 s time budget, prints progress and
uses only NumPy (no file or network I/O).
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------------------- #
def generate_geometric(r, n, target_sum):
    """a_i = C·r**i, 0<r<1."""
    if r <= 0.0 or r >= 1.0:
        a = np.zeros(n)
        a[0] = target_sum
        return a
    denom = 1.0 - r**n
    C = target_sum * (1.0 - r) / denom
    return C * (r ** np.arange(n))


def generate_double_exponential(r1, r2, frac, n, target_sum):
    """Two‑stage geometric decay; continuity at the split point."""
    if r1 <= 0.0:
        r1 = 1e-12
    if r2 <= 0.0:
        r2 = 1e-12
    m = int(round(frac * n))
    m = max(1, min(n - 1, m))
    sum1 = (1.0 - r1**m) / (1.0 - r1) if r1 != 1.0 else m
    sum2 = (1.0 - r2**(n - m)) / (1.0 - r2) if r2 != 1.0 else (n - m)
    C = target_sum / (sum1 + (r1**m) * sum2)
    a = np.empty(n, dtype=float)
    a[:m] = C * (r1 ** np.arange(m))
    a[m:] = C * (r1 ** m) * (r2 ** np.arange(n - m))
    return a


def generate_power_law(p, n, target_sum):
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    vals = idx ** (-p)
    C = target_sum / vals.sum()
    return C * vals


def generate_logistic(alpha, beta, n, target_sum):
    """Monotonically decreasing logistic shape."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / (1.0 + np.exp(alpha * (i - beta)))
    C = target_sum / vals.sum()
    return C * vals


def generate_piecewise_constant(frac, ratio, n, target_sum):
    """Two constant levels: first frac·n entries taller than the rest."""
    m = int(round(frac * n))
    m = max(1, min(n - 1, m))
    denom = m + ratio * (n - m)
    h1 = target_sum / denom
    h2 = h1 * ratio
    a = np.empty(n, dtype=float)
    a[:m] = h1
    a[m:] = h2
    return a


# --------------------------------------------------------------------------- #
def compute_score_fast(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact copy of evaluate_sequence but using FFT (much faster)."""
    a = np.asarray(a, dtype=float)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    F = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()
    return float(2.0 * n * max_val / (s * s))
