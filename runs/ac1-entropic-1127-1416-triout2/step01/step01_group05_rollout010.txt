        cand[j] += delta
        cand_score = score_of(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f'Random move improved to {best_score:.6f}')
        # break if close to target maybe.
        if best_score < 1.55:
            break

    # Phase 2: LP based improvement
    # limit number of LP iterations
    step_t = 0.5
    while time.time() < deadline - 5.0:
        # compute max conv for current a
        conv_a = np.convolve(a, a)
        M_a = conv_a.max()

        # Build convolution matrix A (2n-1 x n) with each upper diagonal equal to a
        # Use scipy.sparse.diags
        diag_vals = [np.full(n, a[d]) for d in range(n)]
        offsets = np.arange(n)
        A = sp.diags(diag_vals, offsets, shape=(2*n-1, n), format='csr')

        # Linear program: maximize sum(g) subject to A @ g <= M_a
        c = -np.ones(n)          # minimize -sum(g)
        b_ub = np.full(2*n-1, M_a)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(1, int(deadline - time.time()))})
        if not res.success:
            print('LP failed, exiting LP loop')
            break

        g = res.x
        sum_g = g.sum()
        # if sum_g is too close to target, no improvement possible
        if sum_g <= target_sum * (1 + 1e-9):
            print('No slack found in LP (sum_g≈target). Stopping LP phase.')
            break

        scale = target_sum / sum_g
        g_scaled = g * scale

        # convex combine
        a_new = (1 - step_t) * a + step_t * g_scaled

        # compute new score
        new_score = score_of(a_new)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            print(f'LP step improved to {best_score:.6f}')
        else:
            # if not improved, reduce step size
            step_t *= 0.5
            if step_t < 1e-3:
                print('Step size too small, terminating LP loop')
                break
            continue

        # optionally increase step size slightly if improvements good
        step_t = min(0.5, step_t * 1.1)

    print(f'Final best score: {best_score:.6f} (n={len(best_seq)})')
    return best_seq.tolist()

"""
Propose a step‑function sequence using a two‑phase optimisation:

1. **Random‐local search** – start from the classic inverse‑square‑root shape,
   perform tiny mass‑transfer moves that keep the ℓ¹‑norm fixed and keep the best
   candidate found.

2. **Linear‑programming refinement** – given a current sequence `f`,
   solve the LP
       maximise   ∑ g_i
       subject to  (f * g)_k ≤ M_f   for all k   (where M_f = max(f*f)),
                   g_i ≥ 0.
   The LP finds a vector `g` that can carry *more* total mass while still
   respecting the maximal convolution bound `M_f`.  Scaling `g` back to the
   required ℓ¹‑norm reduces the bound, and a convex combination between the
   old and the new sequence monotonically lowers the objective.
   The LP is built with SciPy’s sparse matrix facilities and solved with the
   HiGHS interior‑point solver.

The algorithm respects the 1000 s time budget, prints progress, and returns
the best sequence found (as a plain Python list).  All numbers are non‑negative
and the sequence is normalised so that Σ a_i = √(2 n), which makes the
evaluation score equal to `max(conv(a, a))`.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def _score(seq: np.ndarray) -> float:
    """Return the exact value of evaluate_sequence without the overhead."""
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    return float(2.0 * len(seq) * conv.max() / (s * s))


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑value step function within the allotted time."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # ----------------------------------------------------------------------
    # 0. Choose a problem size.
    #    1500 gives a reasonably sized LP (≈ 4.5 M entries) and a decent
    #    approximation of the continuous optimum.
    # ----------------------------------------------------------------------
    n = 1500
    target_sum = np.sqrt(2.0 * n)

    # ----------------------------------------------------------------------
    # 1. Initialise from a known good construction (inverse‑sqrt) or from the
    #    global `height_sequence_1` if it exists.
    # ----------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, list) and len(global_seq) > 0:
        a = np.array(global_seq, dtype=float)
        if len(a) > n:
            a = a[:n].copy()
        else:
            a = np.pad(a, (0, n - len(a)), constant_values=0.0)
    else:
        # classic a_i ∝ 1/√(i+1)
        a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))

    # Normalise to the required ℓ¹‑norm.
    a = a * (target_sum / a.sum())
    a = np.maximum(a, 0.0)

    best_seq = a.copy()
    best_score = _score(best_seq)
    print(f"[init] n={n}, score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 2. Quick stochastic refinement (tiny random transfers).
    # ----------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move at most 0.5 % of the source mass
        max_move = 0.005 * a[i]
        if max_move <= 0.0:
            continue
        delta = np.random.uniform(-max_move, max_move)
        if a[i] - delta < 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[rand] improved to {best_score:.6f}")
        # early exit if we already dip under an interesting threshold
        if best_score < 1.55:
            break

    # ----------------------------------------------------------------------
    # 3. LP‑based monotone improvement.
    #    For a fixed f we maximise the total mass of another sequence g under
    #    the same max‑convolution bound.  If the optimum carries more mass than
    #    the target, a convex combination reduces the bound.
    # ----------------------------------------------------------------------
    step = 0.5          # convex‑combination weight
    failed_lp = 0
    while time.time() < deadline - 5.0:
        # current maximal convolution value
        conv_f = np.convolve(a, a)
        M_f = conv_f.max()

        # Build the (2n‑1)×n convolution matrix A where
        #   (A @ g)[k] = (f * g)[k] .
        # Every upper diagonal d (0 ≤ d < n) contains the constant f[d].
        diagonals = [np.full(n, a[d]) for d in range(n)]
        offsets = np.arange(n, dtype=int)
        A = sp.diags(diagonals, offsets, shape=(2 * n - 1, n), format="csr")

        # Linear program: maximise Σ g_i  ⇔ minimise −Σ g_i .
        c = -np.ones(n, dtype=float)                     # objective
        b_ub = np.full(2 * n - 1, M_f, dtype=float)     # right‑hand side
        bounds = [(0, None)] * n                         # g_i ≥ 0

        # Solve the LP; HiGHS handles sparse matrices efficiently.
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print("[LP] solver failed – terminating LP phase")
            break

        g = res.x
        sum_g = g.sum()

        # If the LP cannot carry more mass than the target, we are at a fixed‑point.
        if sum_g <= target_sum * (1.0 + 1e-9):
            failed_lp += 1
            if failed_lp >= 2:
                print("[LP] no slack left – stopping LP iterations")
                break
            # try a smaller step size before giving up
            step *= 0.5
            if step < 1e-3:
                break
            continue
        failed_lp = 0

        # Scale g back to the required ℓ¹‑norm.
        scale = target_sum / sum_g
        g_scaled = g * scale

        # Convex combination, guarantees monotone decrease of the bound.
        a_new = (1.0 - step) * a + step * g_scaled
        # Normalisation is preserved by construction (both operands have sum = target_sum).

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            print(f"[LP] improved to {best_score:.6f}  step={step:.3f}")
            # gently increase step size when things go well
            step = min(0.5, step * 1.1)
        else:
            # No improvement – shrink the update magnitude.
            step *= 0.5
            if step < 1e-3:
                print("[LP] step too small – terminating")
                break

    # ----------------------------------------------------------------------
    # 4. Final tiny random walk for polishing (very cheap).
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_move = 0.001 * a[i]
        if max_move <= 0.0:
            continue
        delta = np.random.uniform(-max_move, max_move)
        if a[i] - delta < 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[polish] improved to {best_score:.6f}")

    print(f"[final] best score={best_score:.6f}  length={len(best_seq)}")
    # Return a plain Python list as required.
    return best_seq.tolist()
