#!/usr/bin/env python3
"""
Propose a step‑function that improves the “max‑auto‑convolution’’ bound.

Algorithm outline
-----------------
1. **Length selection** – use the length of the previously found
   construction (`height_sequence_1`) if it exists, otherwise a default
   of 2500 (a good trade‑off between quality and speed).

2. **Exploratory Beta sweep** – the continuous Beta(α,α) density on
   [0,1] yields the classic “square‑root’’ (α=0.5) shape.  For α<0.5 the
   mass concentrates near the ends and the resulting auto‑convolution
   becomes flatter.  We evaluate a small grid of α values, keep the best
   candidate and use it as a warm start.

3. **Projected (exponentiated) gradient descent on the true max‑convolution**
   * At each iteration the convolution `b = a*a` is computed, the index
     `k* = argmax(b)` is identified and the sub‑gradient of `max(b)`
     w.r.t. `a` is `g_i = 2*a_{k*-i}` (zero outside the support).
   * An exponentiated‑gradient update `a←a·exp(-η g)` is performed and
     the vector is projected back onto the probability simplex
     (`sum(a)=1, a_i≥0`).
   * A simple momentum term (exponential moving average of the gradient)
     smooths the updates.
   * The learning rate η decays slowly, but is increased again when
     stagnation is detected.  Small random jitter is injected periodically
     to escape plateaus.

4. **Local random moves** – when the descent stalls for a long time we
   perform a cheap “mass‑transfer’’ perturbation: move a tiny amount of
   probability from a randomly chosen index to another.  This can find
   improvements that the gradient step cannot see.

5. **Best‑ever tracking** – every evaluated candidate is compared with the
   best score seen so far; the corresponding sequence is retained.

The whole routine stops just before the allotted time budget (default
1000 s) and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (required to be top‑level)
# ----------------------------------------------------------------------


def sqrt_distribution(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^(-1/2) up to length *n*, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) discretised on the mid‑points of n bins."""
    xs = (np.arange(n, dtype=np.float64) + 0.5) / n
    # pdf ∝ x^{α-1} (1‑x)^{β-1}
    a = np.power(xs, alpha - 1.0) * np.power(1.0 - xs, beta - 1.0)
    # Numerical safety – values can become extremely large for very small α/β
    a = np.where(np.isfinite(a), a, 0.0)
    if a.sum() == 0.0:
        a = np.ones_like(a)
    a /= a.sum()
    return a


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def max_convolution_gradient(a: np.ndarray) -> tuple[np.ndarray, int, float]:
    """
    Return (gradient, index_of_max, max_value) where the gradient is the
    sub‑gradient of max(a*a) w.r.t. a (using the current argmax of the
    convolution).
    """
    n = a.size
    b = np.convolve(a, a)                     # length 2n‑1
    k_star = int(np.argmax(b))
    max_val = float(b[k_star])

    # Build gradient: g_i = 2 * a_{k* - i} for valid indices, else 0
    grad = np.zeros_like(a)
    # i runs over indices where 0 ≤ k*‑i < n  →  i ∈ [k*‑(n‑1), k*]
    i_low = max(0, k_star - (n - 1))
    i_high = min(n - 1, k_star)
    i_range = np.arange(i_low, i_high + 1, dtype=np.int64)
    j_range = k_star - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad, k_star, max_val


def score_from_a(a: np.ndarray) -> float:
    """
    Compute the evaluation score for a (sum(a) is assumed to be 1).
    The formula reduces to 2·n·max(a*a) because the denominator is 1.
    """
    n = a.size
    b = np.convolve(a, a)
    return 2.0 * n * float(b.max())

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low `evaluate_sequence`
    score.  The routine combines a cheap Beta‑parameter sweep with a

    true max‑convolution.  It returns the best sequence found before the
    time limit expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    # ------------------------------------------------------------------
    # Choose sequence length
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2500

    # ------------------------------------------------------------------
    # Initial candidate (fallback to the known “square‑root’’ sequence)
    # ------------------------------------------------------------------
    a = sqrt_distribution(n).copy()
    best_seq = a.copy()
    best_score = score_from_a(best_seq)
    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – cheap sweep over symmetric Beta(α,α) distributions
    # ------------------------------------------------------------------
    # Allocate roughly 8 % of the total budget for this phase.
    phase1_deadline = start_time + min(30.0, budget_s * 0.08)
    alphas = np.concatenate((
        np.linspace(0.05, 0.45, 17, endpoint=False),  # <0.5 region
        np.linspace(0.5, 0.9, 9)                     # around the classic value
    ))
    for alpha in alphas:
        if time.time() >= phase1_deadline:
            break
        a_beta = beta_distribution(n, alpha, alpha)
        sc = score_from_a(a_beta)
        if sc < best_score:
            best_score = sc
            best_seq = a_beta.copy()
            print(f"[beta] α={alpha:.3f} → score={sc:.6f}")
    a = best_seq.copy()    # start descent from the best β‑candidate found

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    eta = 0.6               # initial learning rate
    eta_min = 0.001
    eta_max = 2.0
    decay = 0.9997          # slow decay per iteration
    momentum = 0.9
    grad_avg = np.zeros_like(a)

    stagnation = 0
    iter_cnt = 0

    while time.time() < deadline:
        # compute gradient of the current max‑convolution
        grad, k_star, max_val = max_convolution_gradient(a)

        # exponential moving average of the gradient (simple momentum)
        grad_avg = momentum * grad_avg + (1.0 - momentum) * grad

        # exponentiated‑gradient (multiplicative) update

        a_new = project_to_simplex(a_new)

        cur_score = score_from_a(a_new)

        # keep the best ever seen

            best_score = cur_score
            best_seq = a_new.copy()
            stagnation = 0
            print(f"[{iter_cnt:06d}] new best = {cur_score:.6f} (max@{k_star})")
        else:
            stagnation += 1

        # replace current point (accept everything – we are descending)
        a = a_new

        # ------------------------------------------------------------------
        # Adaptive schedule
        # ------------------------------------------------------------------
        # decay learning rate gradually, but never fall below eta_min
        eta = max(eta_min, eta * decay)

        # if we have been stuck for a while, either raise η a little or jitter
        if stagnation >= 2000:
            # increase step‑size a bit to escape a flat region
            eta = min(eta_max, eta * 1.5)
            # tiny random jitter (preserves simplex)
            jitter = 5e-5 * np.random.randn(n)
            a = project_to_simplex(a + jitter)
            stagnation = 0
            print(f"[{iter_cnt:06d}] jitter injected, eta={eta:.6f}")

        if stagnation >= 12000:
            # keep the best sequence, but re‑initialise a fresh search point
            a = np.random.rand(n)
            a = project_to_simplex(a)
            eta = 0.6
            grad_avg.fill(0.0)
            stagnation = 0
            print(f"[{iter_cnt:06d}] random restart")

            delta = min(a[i_from], 1e-4)  # move at most 1e‑4 mass
            a[i_from] -= delta
            a[i_to]   += delta
            a = project_to_simplex(a)  # re‑normalise (cheap)
            # no need to recompute the score here – next loop will do it

        iter_cnt += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f} after {iter_cnt} iterations")
    return best_seq.tolist()
