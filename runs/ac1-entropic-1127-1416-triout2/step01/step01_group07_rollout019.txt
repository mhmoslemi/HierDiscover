"""
Improved step‑function search

The algorithm works in three stages:

1. **Coarse geometric seed (small problem).**
   We search a one‑parameter geometric family a_i = C·r^i for a modest
   length (≈1200).  This cheap grid search already yields a bound far
   below 2.0.

2. **Linear‑programming “inflation”.**
   With the current sequence f we compute M = max (f*f).  A LP
   (SciPy ``linprog``) maximises the total ℓ₁‑mass of a new sequence g
   under the linear constraints (f*g)[k] ≤ M.  After rescaling g to the
   required ℓ₁‑norm we move a little towards it:
   f ← (1‑t)·f + t·g  (t is reduced if no improvement).  Repeating a
   few times drastically lowers the peak of the convolution.

3. **Fine‑grained smooth gradient descent (large problem).**
   The best small‑scale sequence is interpolated to the target length
   (3000).  We minimise a smooth approximation of the max‑functional
   using a soft‑max (temperature 1/p) and Adam‑style updates, always
   projecting back onto the simplex {x ≥ 0, Σx = √(2n)}.  Random jitter
   and a simple restart strategy avoid stagnation.

All operations use NumPy (FFT‑based convolutions) and SciPy (LP).  The
search stops when the provided time budget is exhausted and returns the
best sequence discovered.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Duchi et al. 2008)
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# --------------------------------------------------------------------------- #
def max_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)[k] (full linear convolution)."""
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    F = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(F * F, n=L)[: 2 * n - 1]
    return float(conv.max())


# --------------------------------------------------------------------------- #
def geometric_seed(n: int, target_sum: float):
    """Return the best geometric sequence a_i = C·r^i for length n."""
    best_seq = None
    best_val = float("inf")
    # λ = 1‑r scanned on a log‑grid (λ ∈ [1e‑5, 1e‑0])
    lambdas = np.logspace(-5, -0.5, 120)
    for lam in lambdas:
        r = 1.0 - lam
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom
        a = C * (r ** np.arange(n))
        val = max_convolution(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    return best_seq, best_val


# --------------------------------------------------------------------------- #
def powerlaw_seed(n: int, target_sum: float):
    """Power‑law a_i ∝ 1/√(i+1) and normalise."""
    idx = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(idx)
    a = a * (target_sum / a.sum())
    return a, max_convolution(a)


# --------------------------------------------------------------------------- #
def build_convolution_matrix(f: np.ndarray):
    """Create the (2n‑1)×n sparse matrix A where (A·b)[k]=(f*b)[k]."""
    import scipy.sparse as sp

    n = f.shape[0]
    # each column i contributes f[i] on the diagonal with offset i
    diags = [np.full(n, fi) for fi in f]
    offsets = np.arange(n)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csc")
    return A


# --------------------------------------------------------------------------- #
def lp_inflate(f: np.ndarray, M: float, target_sum: float, t: float):
    """Solve the LP (max Σb  s.t. 0≤b, (f*b)≤M) and return the convex blend."""
    import scipy.optimize as opt

    n = f.shape[0]
    A = build_convolution_matrix(f)                # (2n‑1)×n
    c = -np.ones(n)                                # maximise Σb ⇒ minimise –Σb
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M)

    res = opt.linprog(
        c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options={"presolve": True}
    )
    if not res.success:
        return None, M

    b_opt = res.x
    if b_opt.sum() == 0:
        return None, M

    scale = target_sum / b_opt.sum()
    g = b_opt * scale
    f_new = (1 - t) * f + t * g
    f_new = projection_simplex(f_new, target_sum)
    M_new = max_convolution(f_new)
    return f_new, M_new


# --------------------------------------------------------------------------- #
def smooth_gradient(seq: np.ndarray, p: float):
    """
    Compute the gradient of the smoothed max functional
    using a soft‑max with temperature 1/p.
    Returns 2·∇_i (∑_k softmax_k·(seq*seq)_k).
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)

    # full convolution (seq*seq)
    F = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(F * F, n=L)[: 2 * n - 1]

    # soft‑max weights (avoid overflow)
    max_c = conv.max()
    w = np.exp(p * (conv - max_c))
    w /= w.sum()

    # correlation grad = 2·(rev(seq) * w)
    rev = seq[::-1]
    rev_fft = np.fft.rfft(rev, n=L)
    w_fft = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(rev_fft * w_fft, n=L)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]  # length n
    return grad, max_c

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step function within ``budget_s`` seconds.
    Returns the best sequence (as a list of floats).
    """
    import scipy.optimize  # ensure SciPy is present; otherwise LP stage is skipped

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2  # small safety margin

    # ------------------------------------------------------------------- #
    # 1. Small‑scale problem (LP stage)
    n_small = 1200                     # small enough for fast LP solves
    target_small = np.sqrt(2.0 * n_small)

    # a) geometric seed
    seq_geo, val_geo = geometric_seed(n_small, target_small)

    # b) power‑law seed (just in case it is better)
    seq_pow, val_pow = powerlaw_seed(n_small, target_small)

    if val_pow < val_geo:
        f = seq_pow.copy()
        best_val = val_pow
    else:
        f = seq_geo.copy()
        best_val = val_geo

    # make sure the ℓ₁‑constraint is satisfied
    f = projection_simplex(f, target_small)
    M = max_convolution(f)

    # LP “inflation” – a few iterations are cheap and effective
    t = 0.5
    for _ in range(6):
        if time.time() >= deadline:
            break
        new_f, new_M = lp_inflate(f, M, target_small, t)
        if new_f is None:               # LP failed – give up the LP stage
            break
        # accept if the peak decreased
        if new_M < M:
            f, M = new_f, new_M
            t = min(t * 1.05, 0.8)      # a bit larger step when we improve
        else:
            t = max(t * 0.7, 0.1)       # shrink step otherwise

    # ------------------------------------------------------------------- #
    # 2. Upsample to the final problem size
    n_final = 3000
    target_final = np.sqrt(2.0 * n_final)

    x_small = np.linspace(0.0, 1.0, n_small)
    x_final = np.linspace(0.0, 1.0, n_final)
    f_large = np.interp(x_final, x_small, f)          # linear interpolation
    f_large = projection_simplex(f_large, target_final)

    # ------------------------------------------------------------------- #
    # 3. Fine‑grained smooth gradient descent (Adam + momentum)
    curr = f_large.copy()
    best_seq = curr.copy()
    best_score = max_convolution(best_seq)

    # Adam state
    m = np.zeros_like(curr)
    v = np.zeros_like(curr)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # FFT settings
    L = _next_pow2(2 * n_final - 1)

    # hyper‑parameters
    step_scale = 0.5 * target_final / n_final   # typical absolute move per iteration
    p = 8.0                                     # temperature for the soft‑max
    iter_no_imp = 0
    iteration = 0

    while time.time() < deadline:
        # ---- compute current convolution and objective -----------------
        F = np.fft.rfft(curr, n=L)
        conv = np.fft.irfft(F * F, n=L)[: 2 * n_final - 1]
        cur_max = float(conv.max())

        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = curr.copy()
            iter_no_imp = 0
            print(f"[{time.time() - start:6.1f}s] iter {iteration:6d} → new best {best_score:.6f}")
        else:
            iter_no_imp += 1

        # ---- smooth gradient (soft‑max) ---------------------------------
        grad, _ = smooth_gradient(curr, p)

        # ---- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        step = -step_scale * m_hat / (np.sqrt(v_hat) + eps)
        cand = projection_simplex(curr + step, target_final)

        # ---- evaluate candidate -----------------------------------------
        F_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(F_c * F_c, n=L)[: 2 * n_final - 1]
        cand_max = float(conv_c.max())

        if cand_max < cur_max:
            curr = cand
            step_scale *= 1.02                     # grow a bit if we improve
        else:
            step_scale *= 0.90                     # shrink step on rejection
            # occasional random jitter to escape flat plateaus
            if np.random.rand() < 0.02:
                jitter = np.random.randn(n_final) * step_scale * 0.5
                curr = projection_simplex(curr + jitter, target_final)

        # ---- temperature schedule ---------------------------------------
        if iteration % 2000 == 0 and p < 40.0:
            p = min(p * 1.5, 40.0)

        # ---- restart if stuck -------------------------------------------
        if iter_no_imp > 5000:
            jitter = np.random.randn(n_final) * step_scale * 5.0
            curr = projection_simplex(best_seq + jitter, target_final)
            m.fill(0.0)
            v.fill(0.0)
            iter_no_imp = 0

        iteration += 1

    # ------------------------------------------------------------------- #
    # Final sanity check (the evaluator does its own clipping)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Search finished. Final best score: {final_score:.6f}")
    return best_seq.tolist()
