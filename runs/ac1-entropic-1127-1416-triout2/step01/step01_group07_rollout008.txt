def geometric_seed(n, S):
    best_val = np.inf; best_seq = None
    for lam in np.logspace(-5, -1, 200):
        r = 1.0 - lam
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0: continue
        C = S * lam / denom
        a = C * (r ** np.arange(n))

""" LP‑based iterative refinement for minimizing max self‑convolution.
...
"""

import numpy as np
import time
import math

# projection onto simplex
def projection_simplex(v, s):
    """Project onto simplex {x >=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# build sparse convolution matrix for constraints a*b <= M
def build_conv_matrix(a):
    """Return a scipy.sparse matrix A such that (A @ b) = a * b (full convolution)."""
    try:
        import scipy.sparse as sp
    except Exception as e:
        raise RuntimeError("scipy is required for LP‑based search") from e
    n = a.shape[0]
    offsets = np.arange(n, dtype=int)
    # each diagonal is constant = a_i
    data = [np.full(n, float(ai)) for ai in a]  # list length n, each length n
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A

def evaluate_fast(seq):
    """Thin wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq.tolist())

def geometric_seed(n, target_sum):
    """Search a one‑parameter geometric family and a simple power‑law."""
    best_val = float('inf')
    best_seq = None

    # geometric
    lambdas = np.logspace(-5, -1, 150)  # try a finer grid
    idx = np.arange(n)
    for lam in lambdas:
        r = 1.0 - lam
        # avoid underflow
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom
        a = C * (r ** idx)
        a = np.clip(a, 0.0, 1000.0)
        val = evaluate_fast(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    # power‑law a_i ∝ 1/√(i+1)
    i = np.arange(1, n + 1, dtype=float)
    a_pow = 1.0 / np.sqrt(i)
    C_pow = target_sum / a_pow.sum()
    a_pow = C_pow * a_pow
    a_pow = np.clip(a_pow, 0.0, 1000.0)
    val_pow = evaluate_fast(a_pow)
    if val_pow < best_val:
        best_val = val_pow
        best_seq = a_pow.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Search for a non‑negative step function with a low evaluation score.

    The search starts from a geometric / power‑law seed, then repeatedly
    solves a linear program that maximises the total mass under the current
    peak‑convolution constraint.  The LP solution is scaled back to the
    required ℓ₁–norm and mixed with the current sequence.  A small amount
    of random jitter and a simple step‑size schedule are used to avoid
    stagnation.
    """
    import numpy as np
    import time
    import math

    # optional scipy imports
    try:
        import scipy.optimize as opt
        import scipy.sparse as sp
    except Exception:
        opt = None
        sp = None

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------ #
    # 0. Determine problem size
    if 'height_sequence_1' in globals():
        seq0 = np.array(globals()['height_sequence_1'], dtype=float)
        n = len(seq0)
        # make sure it respects the clipping limits
        seq0 = np.clip(seq0, 0.0, 1000.0)
    else:
        n = 2500                     # a compromise between quality and speed
        seq0 = None

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------ #
    # 1. Obtain a good deterministic seed (geometric / power‑law)
    best_seq, best_val = geometric_seed(n, target_sum)

    # 2. Try to use the provided construction if it is better
    if seq0 is not None:
        val0 = evaluate_fast(seq0)
        if val0 < best_val:
            best_seq = seq0.copy()
            best_val = val0

    # ------------------------------------------------------------------ #
    # 3. Main LP‑based refinement loop
    curr = best_seq.copy()
    best = curr.copy()
    best_score = best_val

    # hyper‑parameters
    t = 0.5          # initial mixing weight
    t_min = 0.02
    t_decay = 0.97
    it = 0
    no_improve = 0   # counter for stagnation

    # pre‑allocate FFT length
    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # ---- compute current peak of a*a ----
        F = np.fft.rfft(curr, n=fft_len)
        conv_full = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
        M = float(conv_full.max())

        # ---- solve LP: maximise Σ b  s.t. conv(curr, b) ≤ M,  b ≥ 0 ----
        if opt is None:
            # fall back to simple gradient step if scipy unavailable
            break

        # Build the convolution matrix only once per iteration
        # A b gives full convolution curr * b
        A = build_conv_matrix(curr)               # shape (2n‑1, n)
        ub = np.full(2 * n - 1, M, dtype=float)

        c = -np.ones(n, dtype=float)   # maximise sum(b) => minimise -sum(b)
        bounds = [(0, None)] * n

        # Solve with HiGHS – it's very fast for this size
        res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True})
        if not res.success:
            # In the unlikely case the LP fails we simply abort the LP loop
            print(f"LP failed at iteration {it}: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            # nothing to gain – abort
            print(f"LP returned zero mass at iter {it}")
            break

        # Scale back to the required ℓ₁ norm (≤ target_sum, because sum_b ≥ target_sum)
        scale = target_sum / sum_b
        g = scale * b

        # Convex combination
        curr = (1.0 - t) * curr + t * g
        # Explicit clipping – the evaluation routine also does it, but it keeps numbers bounded
        np.clip(curr, 0.0, 1000.0, out=curr)

        # ---- evaluate current candidate ----
        cur_val = evaluate_fast(curr)
        if cur_val < best_score:
            best_score = cur_val
            best = curr.copy()
            print(f"[{time.time() - start_time:.2f}s] iter {it:5d} → new best {best_score:.6f}")
            no_improve = 0
            # tighten mixing weight a little when we make progress
            t = max(t * 0.9, t_min)
        else:
            no_improve += 1

        # ---- simple stagnation handling ----
        if no_improve >= 40:
            # random jitter around the best known sequence
            jitter = np.random.randn(n) * target_sum * 0.005 / n
            tmp = best + jitter
            tmp = np.clip(tmp, 0.0, 1000.0)
            # re‑project onto the simplex to keep the ℓ₁ norm fixed
            tmp = projection_simplex(tmp, target_sum)
            curr = tmp
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] iter {it:5d} – random restart")

        # ---- schedule updates ----
        it += 1
        t = max(t * t_decay, t_min)

    # ------------------------------------------------------------------ #
    # 4. Final polishing with a few gradient‑descent steps (optional)
    # If time permits we run a very cheap projected‑gradient sweep.
    while time.time() < deadline:
        # sub‑gradient at the current maximum index
        F = np.fft.rfft(curr, n=fft_len)
        conv_full = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
        k_max = int(np.argmax(conv_full))
        grad = np.zeros_like(curr)
        # g_i = 2 * a_{k_max-i} when index inside range
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * curr[j[mask]]

        # step size proportional to current norm
        step = 0.3 * target_sum / n
        cand = curr - step * grad / (np.linalg.norm(grad) + 1e-12)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = projection_simplex(cand, target_sum)

        cur_val = evaluate_fast(cand)
        if cur_val < best_score:
            best_score = cur_val
            best = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] gradient tweak → new best {best_score:.6f}")
        curr = cand

    # ------------------------------------------------------------------ #
    final_score = evaluate_fast(best.tolist())
    print(f"Search finished. Best score {final_score:.6f}")
    return best.tolist()

"""
LP‑based iterative refinement to minimise the max self‑convolution metric.

The algorithm works in three stages:

1. **Seeding** – a cheap grid search over geometric families (a_i = C·r^i)
   together with a simple power‑law seed provides a reasonable starting
   sequence with a respectable score.

2. **LP refinement** – for the current sequence `f` we compute the peak
   `M = max_k (f*f)[k]`.  We then solve the linear program

        max   Σ b_j
        s.t.  (f * b)[k] ≤ M   for all k,
              b_j ≥ 0.

   The LP is built with a sparse Toeplitz matrix that encodes the
   convolution constraints.  The optimal `b` is scaled back to the target
   ℓ₁‑norm `S = √(2 n)` and mixed with the current sequence:
        f ← (1‑t)·f + t·g   (g = scaled b).

   The mixing weight `t` is gradually decreased; occasional random
   jitteres prevent stagnation.

3. **Optional polishing** – if time permits a few projected‑gradient
   steps are performed to fine‑tune the best candidate.

All numbers are kept non‑negative and clipped to the range required by
`evaluate_sequence`.  The routine stops after the supplied time budget
(≈ 1000 s) and returns the best sequence discovered.
"""

            print(f"[{time.time() - start_time:.2f}s] iter {it:5d} → new best {best_score:.6f}")
            no_improve = 0
            t = max(t * 0.9, t_min)     # tighten the mixing weight after improvement
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve >= 40:
            # inject a small random perturbation around the current best
            jitter = np.random.randn(n) * target_sum * 0.005 / n
            tmp = best + jitter
            tmp = np.clip(tmp, 0.0, 1000.0)
            tmp = projection_simplex(tmp, target_sum)
            curr = tmp
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] iter {it:5d} – random restart")

        # ----- schedule updates -----
        it += 1
        t = max(t * t_decay, t_min)

    # ------------------------------------------------------------------
    # 4. Optional cheap projected‑gradient polishing (if time left)
    while time.time() < deadline:
        # gradient of the max‑functional (sub‑gradient at the peak)
        F = np.fft.rfft(curr, n=fft_len)
        conv_full = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
        k_max = int(np.argmax(conv_full))

        grad = np.zeros_like(curr)
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * curr[j[mask]]

        # step size proportional to the typical scale of the variables
        step = 0.3 * target_sum / n
        cand = curr - step * grad / (np.linalg.norm(grad) + 1e-12)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = projection_simplex(cand, target_sum)

        cur_val = evaluate_fast(cand)
        if cur_val < best_score:
            best_score = cur_val
            best = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] gradient tweak → new best {best_score:.6f}")

        curr = cand

    # ------------------------------------------------------------------
    final_score = evaluate_fast(best.tolist())
    print(f"Search finished. Best score found: {final_score:.6f}")
    return best.tolist()
