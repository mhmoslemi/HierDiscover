def build_toeplitz(f):
    n = len(f)
    rows = 2*n - 1
    # T[i, j] = f[i - j] if 0 <= i-j < n else 0
    # we can build via loops (efficient enough).
    T = np.zeros((rows, n), dtype=np.float64)
    for i in range(rows):
        # j from max(0, i-(n-1)) to min(i, n-1)
        j_start = max(0, i - (n-1))
        j_end = min(i, n-1)
        # for j in range(j_start, j_end+1):
        #    T[i,j] = f[i-j]
        # vectorized: assign slice
        T[i, j_start:j_end+1] = f[i - np.arange(j_start, j_end+1)]
    return T

def propose_candidate(seed=0, budget_s=1000):
    import numpy as np
    import time
    from scipy.optimize import linprog

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.05

    try: n = len(height_sequence_1)
    except Exception: n = 500   # moderate size for LP

    # generate binomial
    # ... same binomial code as before
    f = binomial_sequence(n)
    target_sum = np.sqrt(2 * n)
    f = f * (target_sum / f.sum())
    # compute initial M
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, init score={best_score:.6f}")

    # mix parameters
    mix_steps = [0.5, 1.0]   # try half step and full step
    # iterations
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1
        # Build convolution matrix
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = f
        # Bound vector
        b = np.full(rows, M, dtype=np.float64)
        # LP
        c = -np.ones(n, dtype=np.float64)   # maximize sum(g)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            # fallback: break or random perturb
            # attempt small random move
            idx_src, idx_dst = np.random.randint(0, n, size=2)
            if idx_src != idx_dst:
                amount = 0.01 * np.random.rand() * f[idx_src]
                f[idx_src] -= amount
                f[idx_dst] += amount
                f = np.maximum(f, 0)
                conv_f = np.convolve(f, f)
                M = conv_f.max()
            continue
        g_raw = res.x
        sum_g_raw = g_raw.sum()
        if sum_g_raw <= 0:
            continue
        # scale to same sum
        scale = target_sum / sum_g_raw
        g = g_raw * scale

        improved = False
        for mix in mix_steps:
            new_f = (1 - mix) * f + mix * g
            # ensure non-negative (still)
            new_conv = np.convolve(new_f, new_f)
            new_M = new_conv.max()
            new_score = float(2 * n * new_M / (new_f.sum()**2))
            if new_score < best_score - 1e-6:  # improvement
                best_f = new_f.copy()
                best_score = new_score
                f = new_f
                M = new_M
                improved = True
                print(f"[iter {iter_count}] improve to {best_score:.6f} mix={mix:.2f}")
                break
        if not improved:
            # Occasionally perturb to escape plateau

                    conv_f = np.convolve(f, f)
                    M = conv_f.max()
    print(f"[final] best score={best_score:.6f}")
    return best_f.tolist()

import numpy as np
import time
from scipy.optimize import linprog

def binomial_sequence(length):
    """Return central binomial coefficients c_k = (2k choose k)/4**k."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq

def build_conv_matrix(f):
    """Construct matrix A such that (A @ g) == np.convolve(f, g) for zero‑padded convolution."""
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i corresponds to f shifted by i
    for i in range(n):

    try:

    conv_f = np.convolve(f, f)
    M = conv_f.max()

    print(f"[init] n={n}, initial score={best_score:.6f}")

    mix_factors = (0.5, 1.0)

    iteration = 0

        iteration += 1

        A = build_conv_matrix(f)                 # shape (2n‑1, n)
        b = np.full(A.shape[0], M, dtype=np.float64)

                         bounds=[(0, None)] * n,
                         method='highs')

            i, j = np.random.randint(0, n, size=2)

                f[i] -= delta
                f[j] += delta
                f = np.maximum(f, 0.0)
                conv_f = np.convolve(f, f)
                M = conv_f.max()
            continue

        sum_g_raw = g_raw.sum()
        if sum_g_raw == 0.0:

            break

            conv_cand = np.convolve(candidate, candidate)
            M_cand = conv_cand.max()

                    M = conv_f.max()

    print(f"[final] best score={best_score:.6f}")

"""
Propose a step‑function sequence using an alternating linear‑programming scheme.

The algorithm starts from the (almost) optimal analytic binomial
coefficients c_k = (2k choose k)/4**k.  The sequence is normalised so that
∑a_i = sqrt(2 n); with this normalisation the evaluation function returned
by ``evaluate_sequence`` equals the maximum value of the auto‑convolution
M = max(a*a).  Keeping the sum fixed, we repeatedly solve the linear program

    maximise   ∑ g_i
    subject to  g_i ≥ 0
                (f * g)_k ≤ M   for every convolution index k

where f is the current sequence.  The solution g has the largest possible
mass while still respecting the current bound M.  After rescaling g to have
the same ℓ₁‑norm as f we blend the two sequences (convex combination).  The
new sequence usually has a smaller convolution peak, thus a lower score.
The loop runs until the time budget expires and the best sequence found is
returned.
"""

import time
import numpy as np
from scipy.optimize import linprog


def binomial_sequence(length: int) -> np.ndarray:
    """Central binomial coefficients c_k = (2k choose k)/4**k for k=0..length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the matrix A such that (A @ g) == np.convolve(f, g, mode='full').
    A has shape (2*len(f)-1, len(f)).
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A
