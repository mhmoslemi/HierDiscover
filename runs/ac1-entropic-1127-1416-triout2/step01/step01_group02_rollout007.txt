    cur_score = eval_seq(seq)
    best_seq = seq.copy()
    best_score = cur_score

    # temperature schedule: start high e.g., temp = 1e-2? Actually evaluation values around 1.57. So we can accept worse by up to maybe 5% early.

    temp = 0.01   # might start with high number
    # step size for transfer fraction
    step = 0.2    # start with bigger transfers (percentage of source)

    # Additional constants
    decay_temp = 0.9995
    decay_step = 0.9999

    while time.time() < deadline:
        # compute convolution and max
        conv = np.convolve(seq, seq)
        max_b = conv.max()
        idx = conv.argmax()
        # compute contributions
        # i runs from max(0, idx-(n-1)) to min(idx, n-1)
        i_start = max(0, idx - (n-1))
        i_end = min(idx, n-1)
        # contributions array length = i_end - i_start + 1
        i_vals = np.arange(i_start, i_end+1)
        j_vals = idx - i_vals
        contrib = seq[i_vals] * seq[j_vals]

        # pick source index weighted by contributions
        total_contrib = contrib.sum()
        if total_contrib == 0:
            # fallback random
            i = np.random.randint(0, n)
        else:
            probs = contrib / total_contrib
            i = np.random.choice(i_vals, p=probs)

        # ensure seq[i] > small epsilon
        if seq[i] < 1e-12:
            continue
        # pick target index j different
        # we can pick j far from i, maybe random in tail segment
        # Choose j from uniform distribution over [0, n-1] excluding i
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1

        # amount to move
        amount = step * np.random.rand() * seq[i]
        # ensure not exceed
        amount = min(amount, seq[i])
        new_seq = seq.copy()
        new_seq[i] -= amount
        new_seq[j] += amount

        # evaluate
        new_score = evaluate_sequence(new_seq.tolist())
        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # simulated annealing prob
            # temperature scaling: use relative diff
            delta = new_score - cur_score
            if np.random.rand() < np.exp(-delta / temp):
                accept = True
        if accept:
            seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[improve] best={best_score:.6f} time={time.time()- (deadline - budget_s):.2f}s")
        # update parameters
        temp *= decay_temp
        step *= decay_step

        # optionally restart after long stagnation
        # maybe random reset to binomial occasionally

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()

"""
propose_candidate
-----------------
Search for a non‑negative step‑height sequence that beats the
canonical “central‑binomial’’ construction (value ≈ π/2 ≈ 1.5708).
The algorithm has two stages:

1. **Coarse parametrised search** – the classic binomial sequence is
   locally distorted by scaling a prefix of the vector and (optionally)
   redistributing the removed mass over the remainder.  A tiny grid
   over the prefix length and the scaling factor yields a better
   starting point in a few milliseconds.

2. **Guided stochastic refinement** – repeatedly identify the Cartesian
   product a_i·a_j that dominates the current maximum of the
   auto‑convolution, shrink those two entries and spread the released
   mass over a few far‑away indices.  An annealing schedule allows
   occasional uphill moves, which helps escape plateaus.

The whole procedure respects the 1000 s time budget, works with any
pre‑existing `height_sequence_1` (fallback length 5000) and returns a
plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# central‑binomial (catalan) coefficients  c_k = (2k choose k) / 4**k
def binomial_sequence(length: int) -> np.ndarray:
    """Return c_k for k = 0 .. length‑1 (double precision)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        # recurrence  c_{k} = c_{k‑1} * ((2k‑1)(2k)) / (4 k²)
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
# cheap evaluation wrapper (avoids converting the list many times)
def fast_evaluate(arr: np.ndarray) -> float:
    """Exact same result as ``evaluate_sequence`` but works on ndarrays."""
    # the external function already clamps values, we mimic that behaviour
    a = np.clip(arr, 0.0, 1000.0)
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:          # same guard as the original evaluator
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def parametric_distortion(base: np.ndarray,
                          prefix_len: int,
                          scale: float,
                          compensate: bool = True) -> np.ndarray:
    """
    Scale the first ``prefix_len`` entries of *base* by ``scale`` (0<scale≤1).
    If *compensate* is True the mass removed from the prefix is spread
    uniformly over the remaining entries so that the total ℓ₁‑norm does not
    shrink dramatically.
    """
    seq = base.copy()
    if prefix_len > 0:
        seq[:prefix_len] *= scale

    if compensate and prefix_len < seq.size:
        # amount taken away from the prefix
        removed = (base[:prefix_len] - seq[:prefix_len]).sum()
        # spread uniformly over the suffix
        tail_len = seq.size - prefix_len
        if tail_len > 0 and removed != 0.0:
            seq[prefix_len:] += removed / tail_len
    return seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a step‑function height vector that (heuristically) minimises
    ``evaluate_sequence``.  The routine respects the 1000 s runtime limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.05   # safety margin

    # ------------------------------------------------------------------
    # use the length of the previously supplied sequence if it exists
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 5000

    # ------------------------------------------------------------------
    # canonical construction – a solid baseline
    base_seq = binomial_sequence(n)

    # ------------------------------------------------------------------
    # stage 1 – a tiny grid over (prefix length, scaling factor)
    best_seq = base_seq.copy()
    best_score = fast_evaluate(best_seq)

    # grid parameters (chosen for speed – a few dozen trials)
    prefix_candidates = np.geomspace(10, n // 2, num=12, dtype=int)
    scale_candidates = np.linspace(0.25, 1.0, num=13)   # 0.25 … 1.0

    for pref in prefix_candidates:
        for sc in scale_candidates:
            cand = parametric_distortion(base_seq, pref, sc, compensate=True)
            sc_score = fast_evaluate(cand)
            if sc_score < best_score:
                best_score = sc_score
                best_seq = cand.copy()
                print(f"[grid] pref={pref:<5d} scale={sc:.3f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # stage 2 – guided stochastic refinement (annealed pair‑reduction)
    cur_seq = best_seq.copy()
    cur_score = best_score

    # annealing / step‑size schedules
    temperature = 1e-2
    temp_decay = 0.9995
    step_frac = 0.15          # fraction of a chosen entry to move
    step_decay = 0.9998

    # number of indices that receive the released mass (kept small)
    num_targets = max(3, n // 300)

    while time.time() < deadline:
        # full auto‑convolution and the index attaining the maximum
        conv = np.convolve(cur_seq, cur_seq)
        max_idx = int(conv.argmax())

        # indices that actually take part in that maximal sum
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = max_idx - i_vals

        # contribution of each pair (i, j) to the maximal convolution entry
        contrib = cur_seq[i_vals] * cur_seq[j_vals]

        # pick the pair with the largest contribution
        pair_pos = int(np.argmax(contrib))
        i = int(i_vals[pair_pos])
        j = int(j_vals[pair_pos])

        # Both entries must be positive to take away mass
        if cur_seq[i] < 1e-12 or cur_seq[j] < 1e-12:
            # fallback – random pair
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue

        # amount to take from each side (kept small enough not to break non‑negativity)
        take_i = step_frac * np.random.rand() * cur_seq[i]
        take_j = step_frac * np.random.rand() * cur_seq[j]
        delta = min(take_i, take_j)          # we remove the same amount from both

        if delta <= 0.0:
            continue

        cand_seq = cur_seq.copy()
        cand_seq[i] -= delta
        cand_seq[j] -= delta

        # distribute the released mass (2·δ) over a few far indices
        # we deliberately avoid the two source positions
        forbidden = {i, j}
        possible = np.array([k for k in range(n) if k not in forbidden])
        if possible.size == 0:
            continue
        targets = np.random.choice(possible,
                                   size=min(num_targets, possible.size),
                                   replace=False)
        cand_seq[targets] += (2.0 * delta) / targets.size

        # evaluate – use the reference evaluator (it does the required clipping)
        cand_score = fast_evaluate(cand_seq)

        # accept if improvement, otherwise with Metropolis probability
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # temperature may become tiny; safeguard against division by zero
            prob = math.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_seq = cand_seq
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq.copy()
                elapsed = time.time() - (deadline - budget_s)
                print(f"[improve] score={best_score:.6f}  time={elapsed:.1f}s")

        # update annealing parameters
        temperature *= temp_decay
        step_frac *= step_decay

        # occasional reset to the current best to avoid drift
        if np.random.rand() < 0.001:
            cur_seq = best_seq.copy()
            cur_score = best_score
            temperature = 1e-2
            step_frac = 0.15

    print(f"[final] best score={best_score:.6f}")
    # ``evaluate_sequence`` expects a plain Python list
    return best_seq.tolist()
