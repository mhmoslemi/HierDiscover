"""
Propose a step‑function (list of non‑negative heights) with a low
``evaluate_sequence`` score.

The search consists of three phases:

1. **Broad random sampling** – many parametric families (exponential,
   power‑law, stretched‑exponential, piecewise power‑law, logistic,
   random decreasing) are sampled with random lengths up to a few
   thousand.  The best sequence found is kept.

2. **Linear‑programming refinement** – starting from the best sequence,
   we solve a linear program
        maximise   Σ b_j
        subject to a * b ≤ M                (convolution)
                    b_j ≥ 0
   where ``a`` is the current sequence and ``M`` = max(a*a).
   The returned vector ``b`` has a larger ℓ₁‑norm while respecting the
   same convolution bound.  We mix ``b`` into ``a`` (convex combination)
   and repeat while time permits.  This is exactly the “LP‑step’’ of the
   state‑of‑the‑art method but implemented with cvxpy.

3. **Fine‑grained hill‑climbing** – finally we perturb a few random
   entries of the current best candidate (multiplicative Gaussian
   noise) and keep any improvement.  This phase cleans up the result.

All sequences are normalised so that the maximal entry equals 1
(scaling does not change the objective).  The routine respects the
1000 s budget and always returns a valid list before the deadline.
"""

import time
import math
import random
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:               # pragma: no cover
        return math.inf


def _normalize(seq):
    """Scale a sequence so that its maximum value equals 1 (no‑op if all zero)."""
    arr = np.asarray(seq, dtype=float)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _exp_seq(n, alpha):
    """Exponential decay a_i = exp(-alpha * i)."""
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)))


def _pow_seq(n, p):
    """Power‑law decay a_i = (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=float)
    return _normalize(idx ** (-p))


def _lin_seq(n, beta):
    """Linear decay a_i = max(0, 1 - beta*i)."""
    return _normalize(np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float)))


def _mix_seq(n, alpha, eps):
    """Mixed exponential + constant a_i = exp(-alpha*i) + eps."""
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)) + eps)


def _logistic_seq(n, steep):
    """Logistic decay a_i = 1/(1+exp((i-mid)/steep))."""
    i = np.arange(n, dtype=float)
    mid = n / 2.0
    seq = 1.0 / (1.0 + np.exp((i - mid) / steep))
    return _normalize(seq)


def _randdec_seq(n):
    """Random decreasing sequence (sorted uniform numbers)."""
    arr = np.random.rand(n)
    arr.sort()
    arr = arr[::-1]                     # descending
    return _normalize(arr)


def _stretched_exp_seq(n, alpha, beta):
    """Stretched exponential a_i = exp(-alpha * i**beta)."""
    i = np.arange(n, dtype=float)
    return _normalize(np.exp(-alpha * (i ** beta)))


def _piecewise_pow_seq(n, p1, p2, split, scale):
    """Two‑segment power law:
       a_i = (i+1)^(-p1) for i < split,
       a_i = scale * (i+1)^(-p2) for i >= split."""
    i = np.arange(n, dtype=float) + 1.0
    part1 = (i ** (-p1))
    part2 = scale * (i ** (-p2))
    seq = np.where(np.arange(n) < split, part1, part2)
    return _normalize(seq)


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_refine(seq, cur_score, time_budget):
    """
    Perform a few LP‑based improvement steps starting from ``seq``.
    Returns (new_seq, new_score).  The routine stops when the allocated
    time_budget expires or when no further progress is observed.
    """
    start = time.time()
    a = np.asarray(seq, dtype=float)
    n = len(a)

    # current bound on the convolution
    M = float(np.max(np.convolve(a, a)))
    best_a = a.copy()
    best_sc = cur_score
    t = 0.25                     # initial mixing weight

    while time.time() - start < time_budget:
        # --- build and solve the LP -------------------------------------------------
        b = cp.Variable(n, nonneg=True)
        conv = cp.conv(cp.Constant(a), b)                 # length 2n‑1
        constraints = [conv <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception as e:                  # pragma: no cover
            print("LP solver error:", e)
            break

        if b.value is None:                     # infeasible / timeout
            print("LP did not return a solution.")
            break

        g = np.maximum(0.0, np.asarray(b.value).flatten())

        # --- convex combination + renormalisation ----------------------------------
        a_new = (1.0 - t) * a + t * g
        mx = a_new.max()
        if mx > 0:
            a_new = a_new / mx

        new_sc = _score(a_new.tolist())
        if new_sc < best_sc - 1e-9:
            best_sc = new_sc
            best_a = a_new
            a = a_new
            M = float(np.max(np.convolve(a, a)))   # tighter bound for the next step
            print(f"LP improvement → score {best_sc:.6f}")
            # keep the same mixing weight
        else:
            # no improvement – decrease the mixing weight to be more conservative
            t = max(t * 0.7, 0.02)
            if t <= 0.02:
                break

    return best_a.tolist(), best_sc

# ----------------------------------------------------------------------
def _hill_climb(seq, cur_score, time_budget):
    """Simple stochastic hill‑climbing on the current best sequence."""
    start = time.time()
    best = np.array(seq, copy=True)
    best_sc = cur_score
    n = len(best)

    while time.time() - start < time_budget:
        # modify a small random subset
        cand = best.copy()
        m = random.randint(1, min(8, n))
        idxs = random.sample(range(n), m)
        for i in idxs:
            factor = 1.0 + random.gauss(0.0, 0.10)
            cand[i] = max(0.0, cand[i] * factor)

        # renormalise (max = 1)
        mx = cand.max()
        if mx > 0:
            cand = cand / mx

        sc = _score(cand.tolist())
        if sc < best_sc:
            best_sc, best = sc, cand
            print(f"Hill‑climb improvement → {best_sc:.6f}")

    return best.tolist(), best_sc

            init = _normalize(list(height_sequence_1))
            sc = _score(init)
            if sc < best_score:
                best_seq, best_score = init, sc
                print(f"Initial reference score: {best_score:.6f}")
        except Exception:               # pragma: no cover
            pass

    # lengths we will try – favour a few thousand entries
    lengths = [800, 1200, 1600, 2000, 2500, 3000, 3500, 4000]

    # ------------------------------------------------------------------
    # 1) Broad random sampling of parametric families
    # ------------------------------------------------------------------
    while time.time() < deadline * 0.55:   # leave ~45 % of the time for refinement
        n = int(random.choice(lengths))
        family = random.choice([
            'exp', 'pow', 'lin', 'mix', 'log', 'randdec',
            'strexp', 'piecepow'
        ])

        if family == 'exp':
            # α ∈ [1e‑4, 5] (log‑uniform)
            log_alpha = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_alpha
            cand = _exp_seq(n, alpha)

        elif family == 'pow':
            p = random.uniform(0.1, 1.5)    # concentrate around the promising 0.5‑1 range
            cand = _pow_seq(n, p)

        elif family == 'lin':
            beta = random.uniform(0.0, 2.0 / n)
            cand = _lin_seq(n, beta)

        elif family == 'mix':
            log_alpha = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_alpha
            eps = random.uniform(0.0, 0.5)
            cand = _mix_seq(n, alpha, eps)

        elif family == 'log':
            steep = random.uniform(0.01, n / 2.0)
            cand = _logistic_seq(n, steep)

        elif family == 'randdec':
            cand = _randdec_seq(n)

        elif family == 'strexp':
            # stretched exponential – α ∈ [1e‑4,5], β ∈ [0.2,1.0]
            log_alpha = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_alpha
            beta = random.uniform(0.2, 1.0)
            cand = _stretched_exp_seq(n, alpha, beta)

        else:  # 'piecepow'
            p1 = random.uniform(0.1, 0.9)
            p2 = random.uniform(0.6, 1.4)
            split = random.randint(int(0.2 * n), int(0.8 * n))
            scale = random.uniform(0.3, 1.0)
            cand = _piecewise_pow_seq(n, p1, p2, split, scale)

        # discard degenerate candidates
        if sum(cand) < 1e-8:
            continue

        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"New best ({family}) n={n} → {best_score:.6f}")

        # occasional tiny hill‑climb during the random phase
        if best_seq is not None and random.random() < 0.05:
            cand_hc, sc_hc = _hill_climb(best_seq, best_score, 0.0)  # zero‑time call – just a quick try
            if sc_hc < best_score:
                best_score, best_seq = sc_hc, cand_hc

    # ------------------------------------------------------------------
    # 2) Linear‑programming refinement (if we have a decent start)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 60.0:
        # allocate roughly half of the remaining budget to LP refinement
        lp_budget = time_left * 0.5
        best_seq, best_score = _lp_refine(best_seq, best_score, lp_budget)

    # ------------------------------------------------------------------
    # 3) Final hill‑climbing phase
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 5.0:
        best_seq, best_score = _hill_climb(best_seq, best_score, time_left)

    # ------------------------------------------------------------------
    # 4) Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 500
        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
