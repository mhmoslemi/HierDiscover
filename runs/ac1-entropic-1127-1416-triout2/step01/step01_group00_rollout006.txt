"""
Propose a step‑function with a low evaluate_sequence value.

Main ideas
---------
1.  Work on the probability simplex (non‑negative, sum = 1).  The objective
    `E(a) = 2·n·max(a*a)/(sum(a)**2)` is scale‑invariant, therefore we keep
    a normalised to unit sum and only care about minimising `max(a*a)`.
2.  Sub‑gradient descent on the convex, non‑smooth function
    `max(a*a)`.  The sub‑gradient is computed as the average of the
    gradients of *all* convolution entries that are within a small relative
    tolerance of the current maximum – this encourages the algorithm to move
    towards points where several constraints are active.
3.  Periodically improve the current vector by solving a linear program:
    maximise  Σ b_i   subject to   b_i ≥ 0  and  (a * b)_k ≤ M  for all k,
    where `M = max(a*a)`.  The LP gives a direction `b` that can increase the
    ℓ₁‑mass without raising the worst convolution entry.  Mixing the current
    vector with the normalised LP solution reduces the maximum value.
4.  Alternate a few rounds of LP improvement with short sub‑gradient
    refinements.  The whole search runs under the 1000 s budget and keeps the
    best sequence seen (across several lengths and random restarts).

The algorithm stays completely self‑contained, uses only NumPy and (if
available) SciPy for the LP, and returns a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (no closures)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors using an FFT of size a power of two."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_indices(a: np.ndarray,
                                rel_tol: float = 1e-4) -> tuple[float, np.ndarray]:
    """
    Return the maximal convolution value and the indices whose value is
    within ``rel_tol`` of that maximum.
    """
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    # indices that are "almost maximal"
    if rel_tol <= 0:
        idxs = np.array([int(np.argmax(conv))])
    else:
        cutoff = max_val * (1.0 - rel_tol)
        idxs = np.where(conv >= cutoff)[0]
    return max_val, idxs


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry (index k) with respect to a.
    Implements the rule described in the problem statement.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def compute_score_simplex(a: np.ndarray) -> float:
    """
    Scale‑invariant score: 2·n·max(a*a) (since sum(a)=1).
    Returned as a plain float.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    return 2.0 * n * max_val  # denominator is 1² because sum(a)=1


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (averaged near‑max gradients)
# ----------------------------------------------------------------------


def subgradient_optimize(initial_a: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 100_000,
                         eta0: float = 0.9,
                         rel_tol: float = 1e-4,
                         noise_every: int = 800,
                         noise_scale: float = 5e-5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the objective ``max(a*a)``.
    The vector stays on the simplex.  Returns the best vector and its score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score_simplex(a)

    n = len(a)
    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1

        max_val, idxs = max_convolution_and_indices(a, rel_tol=rel_tol)

        # average sub‑gradient over all almost‑max entries
        grad = np.zeros_like(a)
        for k in idxs:
            grad += subgradient_of_max_at_index(a, int(k))
        grad /= float(len(idxs))

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        a_new = a - eta * grad

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score_simplex(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional progress report (does not affect the harness)
        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d}  cur={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# LP improvement (requires SciPy)
# ----------------------------------------------------------------------


def lp_improve(a: np.ndarray, M: float) -> tuple[np.ndarray, bool]:
    """
    Solve  max Σ b_i   s.t.  (a * b)_k ≤ M  (for all k)  and  b ≥ 0.
    Returns the optimised vector ``b`` (already normalised to sum=1) and a flag
    indicating success.  If SciPy is unavailable or the LP fails, ``(None,
    False)`` is returned.
    """
    try:
        from scipy.linalg import toeplitz
        from scipy.optimize import linprog
    except Exception:  # pragma: no cover
        return None, False

    n = len(a)
    # first column of the Toeplitz matrix
    col = np.concatenate([a, np.zeros(n - 1)])
    # first row (only the very first element is non‑zero)
    row = np.concatenate([a[:1], np.zeros(n - 1)])
    A_ub = toeplitz(col, row)               # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M)

    c = -np.ones(n)                         # minimise –Σ b_i
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None, False

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None, False

    # Normalise to unit sum (so that mixing keeps the simplex property)
    b_norm = b / sum_b
    return b_norm, True


def lp_and_mix(a: np.ndarray,
               time_deadline: float,
               mix_strengths: tuple[float, ...] = (0.3, 0.5, 0.7),
               max_mix_iter: int = 3) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvements followed by short sub‑gradient
    refinements.  The function returns the best vector obtained.
    """
    best_a = a.copy()
    best_score = compute_score_simplex(best_a)

    for phase in range(max_mix_iter):
        if time.time() >= time_deadline:
            break

        M = convolution_fft(best_a, best_a).max()
        b, ok = lp_improve(best_a, float(M))
        if not ok or b is None:
            break

        # try a few mixing weights and keep the best
        for t in mix_strengths:
            cand = (1.0 - t) * best_a + t * b
            # cand already sums to 1 (convex combination of simplex vectors)
            cand_score = compute_score_simplex(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"    LP mix phase={phase+1} t={t:.2f} new best={best_score:.6f}")

        # after mixing, polish a little with sub‑gradient steps
        deadline_sub = min(time_deadline, time.time() + 2.0)  # small budget
        best_a, _ = subgradient_optimize(best_a,
                                         time_deadline=deadline_sub,
                                         max_iter=5_000,
                                         eta0=0.6,
                                         rel_tol=1e-4,
                                         noise_every=0,
                                         noise_scale=0.0)

    return best_a, best_score

def arcsine_initial(n: int) -> np.ndarray:
    """
    Continuous arcsine density discretised on n points.
    a_i ∝ 1 / sqrt((i+0.5)*(n‑0.5‑i) + ε)  with a small epsilon.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def uniform_initial(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n)
