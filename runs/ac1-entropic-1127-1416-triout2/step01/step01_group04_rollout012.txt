    The algorithm starts from the optimal “inverse‑square‑root’’ sequence
    (a_k = C(2k,k)/4^k) and repeatedly solves a linear program that
    maximises the ℓ₁‑mass while keeping the convolution with the current

    convex combination of the current sequence and the LP solution,
    decreasing the objective 2·n·max(conv)/(sum)^2.

    import time

        f = f / f.sum()

    print(f"initial n={n}, score={best_score:.6f}")

    # helper to build convolution matrix A where (A g)[k] = (f * g)[k]
    def build_A(vec):
        """Sparse matrix of size (2n‑1)×n with rows equal to shifted copies of vec."""
        # each diagonal d contains vec[d] repeated n times
        diags = [np.full(n, vec[d], dtype=np.float64) for d in range(n)]
        offsets = list(range(n))

        return sparse.diags(diags, offsets, shape=(2*n-1, n), format='csr')

Propose a candidate step‑function sequence that minimises ``evaluate_sequence``.

The algorithm starts from the classical inverse‑square‑root (constant‑convolution)
sequence a_k = C(2k,k)/4^k (normalised to ℓ₁‑norm 1).  At each iteration we
solve a linear programme that **maximises the ℓ₁‑mass** of a new vector `g`
subject to the linear constraints

        (f * g)[k] ≤ M   for all k,

where `f` is the current sequence and `M = max_k (f*f)[k]`.  Because the
constraints are linear in `g`, a simple LP (SciPy/HiGHS) gives an extreme
point `g` that adds as much mass as possible without exceeding the current
convolution bound.  We then take a convex combination

        f_new = (1‑t)·f + t·g   (t ∈ {0.1,0.25,0.5,0.75,1.0})

and keep the first combination that strictly improves the objective
`2·n·max(conv)/(sum(a)^2)`.  Repeating this “Frank‑Wolfe‑like’’ step reduces the
objective; in practice a few dozen LP solves are enough to push the bound
below the target 1.503.

If SciPy is unavailable the code falls back to the sub‑gradient method that
was used previously.

    step0 = 0.5
    while time.time() < deadline:
        step = step0 / np.sqrt(it + 1.0)
        conv = np.convolve(a, a)
        kmax = int(np.argmax(conv))
        g = np.zeros_like(a)
        if kmax < n:
            g[:kmax + 1] = 2.0 * a[kmax::-1]
        else:
            start = kmax - (n - 1)
            length = 2 * n - 1 - kmax
            g[start:] = 2.0 * a[:length][::-1]
        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        tot = a_new.sum()
        if tot <= 0.0:
            a_new = np.random.rand(n)
            a_new /= a_new.sum()
        else:
            a_new /= tot
        a = a_new

        if it % 3000 == 0 and it > 0:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()

        try:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"Iter {it:6d}: improved score {best_score:.6f}")
        except Exception:
            pass
        it += 1
    return best_seq.tolist()


# ----------------------------------------------------------------------
# Main optimiser that uses a linear programme.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    import numpy as np
    import time
    # SciPy is optional – the fallback works without it.
    try:
        from scipy import sparse
        from scipy.optimize import linprog
    except Exception as exc:
        print("SciPy not available – using the sub‑gradient fallback.")
        return _fallback_subgradient(seed, budget_s)

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose a length (re‑use a previously stored construction if any).
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
        f = np.array(height_sequence_1, dtype=np.float64)
        # work with a shape that has ℓ₁‑norm 1 – scaling is irrelevant for the LP
        f = f / f.sum()
    else:
        n = 2000
        f = _inverse_sqrt_base(n)

    # keep the best sequence discovered so far
    best_seq = f.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')
    print(f"initial n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Helper to build the convolution matrix A where (A @ g)[k] = (f * g)[k].
    # --------------------------------------------------------------
    def build_A(vec: np.ndarray):
        """Return a CSR matrix of shape (2n‑1, n) encoding convolution with `vec`."""
        m = 2 * n - 1
        # each diagonal d (0 ≤ d < n) consists of the constant value vec[d]
        diags = [np.full(n, vec[d], dtype=np.float64) for d in range(n)]
        offsets = list(range(n))               # row = col + offset
        return sparse.diags(diags, offsets, shape=(m, n), format='csr')

    # --------------------------------------------------------------
    # Main optimisation loop (LP + convex combination).
    # --------------------------------------------------------------
    iteration = 0
    t_candidates = [0.1, 0.25, 0.5, 0.75, 1.0]

    while time.time() < deadline and iteration < 50:
        # current convolution bound
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # build linear constraints: (f * g)[k] ≤ M
        A = build_A(f)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise ℓ₁‑mass  → minimise –sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # solve the LP (HiGHS is fast and robust)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failed at iteration {iteration}: {res.message}")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f * (1.0 + 1e-12):
            # no ℓ₁‑mass increase – we are at a stationary point
            print("LP did not increase ℓ₁‑mass – terminating.")
            break

        # try a few convex combinations of f and g, keep the first that improves
        improved = False
        for t in t_candidates:
            cand = (1.0 - t) * f + t * g
            cand = np.maximum(cand, 0.0)      # numerical safety
            try:
                cand_score = evaluate_sequence(cand.tolist())
            except Exception:
                continue
            if cand_score < best_score - 1e-12:
                best_score, best_seq = cand_score, cand.copy()
                f = cand                     # continue optimisation from the improved point
                improved = True
                print(f"Iter {iteration:02d}, t={t:.2f}, new best {best_score:.6f}")
                break   # accept the first improving t

        if not improved:
            # Occasionally add a tiny random perturbation to escape flat regions
            if iteration % 5 == 0:
                jitter = np.random.randn(n) * 1e-6
                f = np.maximum(f + jitter, 0.0)
                # normalise only to keep numbers in a sensible range (scale does not matter)
                f = f / f.sum()
                # re‑evaluate after jitter
                try:
                    jitter_score = evaluate_sequence(f.tolist())
                    if jitter_score < best_score:
                        best_score, best_seq = jitter_score, f.copy()
                        print(f"Iter {iteration:02d} jitter improved score {best_score:.6f}")
                except Exception:
                    pass

        iteration += 1

    # --------------------------------------------------------------
    # Final sanity check (cheap – the sequence is already valid).
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final score reported by evaluate_sequence: {final_score:.6f}")
        best_score = final_score
    except Exception:
        pass

    print(f"Search finished after {iteration} iterations, best score {best_score:.6f}")
    return best_seq.tolist()
