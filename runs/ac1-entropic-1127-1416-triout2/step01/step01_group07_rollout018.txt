"""
Hybrid optimizer for the step‑function evaluation problem.

The algorithm works in two phases:

1. **Projected sub‑gradient descent** –  
   A fast gradient loop (the exact same sub‑gradient as the
   reference code) is run for roughly half of the time budget.
   The iterate is always projected onto the simplex
   {x ≥ 0, Σx = sqrt(2·n)} so that the objective
   ``2·n·max(conv(x,x))/sum(x)^2`` is scale‑invariant.
   Random jitter is added occasionally to escape plateaus.

2. **Simulated annealing on permutations / mass transfers** –  
   Using the best sequence from phase 1 as a start, a cheap
   SA loop explores discrete moves:
        * swap two positions (preserves the ℓ₁‑norm),
        * move a random fraction of mass from one index to another.
   The objective is evaluated with a single FFT per trial
   (O(n log n)).  A decreasing temperature gradually makes the
   search greedy.  Whenever a new best value is found it is
   printed.

Both phases respect the security checks of ``evaluate_sequence``:
all entries stay in the interval [0, 1000] and the final sequence
is returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")

    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _max_convolution(seq: np.ndarray, fft_len: int) -> float:
    """Return max(convolution(seq, seq)) using an FFT of length fft_len."""
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * len(seq) - 1]
    return float(conv.max())


def _metric(seq: np.ndarray) -> float:
    """Exact metric 2·n·max(conv)/sum(seq)^2 (no security checks)."""
    n = len(seq)
    if n == 0:
        return float("inf")
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    fft_len = 1 << ((2 * n - 1).bit_length())
    max_conv = _max_convolution(seq, fft_len)
    return float(2.0 * n * max_conv / (sum_a ** 2))


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑score non‑negative step function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2               # safety margin

    # ------------------------------------------------------------------- #
    # 1. Problem size.  3000 gives a good trade‑off between flexibility
    #    and per‑iteration cost.
    n = 3000
    target_sum = np.sqrt(2.0 * n)                        # desired ℓ₁‑norm

    # ------------------------------------------------------------------- #
    # 2. Initial sequence.
    #    Prefer the previously discovered construction if it exists.
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a = np.asarray(init_seq, dtype=float).copy()
        if a.shape[0] > n:
            a = a[:n]
        elif a.shape[0] < n:
            a = np.pad(a, (0, n - a.shape[0]), "constant")
    else:
        # fallback – uniform start (will be improved quickly)
        a = np.full(n, target_sum / n, dtype=float)

    # ensure feasibility
    a = np.clip(a, 0.0, 1000.0)
    cur_sum = a.sum()
    if cur_sum > 0:
        a = a * (target_sum / cur_sum)
    else:
        a = np.full(n, target_sum / n, dtype=float)

    # ------------------------------------------------------------------- #
    # 3. Projected sub‑gradient descent (phase 1)
    #    Use roughly half of the allocated time.
    grad_deadline = start_time + budget_s * 0.55
    fft_len = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_score = _metric(best)
    curr = a.copy()
    step_scale = 0.6               # starting absolute step magnitude
    min_step = 1e-8
    it = 0

    while time.time() < grad_deadline:
        # current convolution and its peak index
        F = np.fft.rfft(curr, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
        k_max = int(conv.argmax())
        max_val = float(conv[k_max])

        # current objective value (identical to evaluate_sequence)
        sum_curr = curr.sum()
        curr_score = 2.0 * n * max_val / (sum_curr ** 2)

        if curr_score < best_score:
            best_score = curr_score
            best = curr.copy()
            print(f"[{time.time() - start_time:6.1f}s] GD iter {it:6d} → new best {best_score:.6f}")
            step_scale = max(step_scale * 0.9, min_step)

        # sub‑gradient of the max‑functional at k_max
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(curr)
        grad[mask] = 2.0 * curr[j[mask]]

        # take a step and project back onto the simplex
        step = step_scale * target_sum / n
        tentative = curr - step * grad / (np.linalg.norm(grad) + 1e-12)
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.02:
            jitter = np.random.randn(n) * step_scale * target_sum / (30.0 * n)
            curr = projection_simplex(np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        # simple step‑size decay
        if it % 250 == 0 and it > 0:
            step_scale = max(step_scale * 0.85, min_step)

        it += 1

    # ------------------------------------------------------------------- #
    # 4. Simulated annealing (phase 2) – explore permutations / transfers
    #    The ℓ₁‑norm is already fixed (≈ target_sum) so we keep it constant.
    sa_deadline = deadline
    curr = best.copy()
    curr_score = best_score
    T0 = max(best_score, 1.0)          # initial temperature
    T = T0
    decay = 0.9997                     # exponential cooling

    sa_iter = 0
    while time.time() < sa_deadline:
        # choose a move type
        if np.random.rand() < 0.55:   # swap two positions
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            trial = curr.copy()
            trial[i], trial[j] = trial[j], trial[i]
        else:                         # transfer a random fraction of mass
            i = np.random.randint(0, n)
            if curr[i] < 1e-12:
                continue
            j = np.random.randint(0, n)
            if i == j:
                continue
            frac = np.random.rand() * 0.3          # up to 30 % of source
            delta = curr[i] * frac
            # respect the upper bound 1000
            delta = min(delta, 1000.0 - curr[j])
            trial = curr.copy()
            trial[i] -= delta
            trial[j] += delta

        # evaluate trial (full convolution via FFT)
        trial_score = _metric(trial)

        # Metropolis acceptance
        if trial_score < curr_score or np.random.rand() < np.exp(-(trial_score - curr_score) / max(T, 1e-12)):
            curr = trial
            curr_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best = trial.copy()
                print(f"[{time.time() - start_time:6.1f}s] SA iter {sa_iter:6d} → new best {best_score:.6f}")

        # temperature schedule
        T *= decay
        sa_iter += 1

    # ------------------------------------------------------------------- #
    # Final sanity check (the evaluator does its own clipping)
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Best score reported by evaluator: {final_score:.6f}")
    return best.tolist()
