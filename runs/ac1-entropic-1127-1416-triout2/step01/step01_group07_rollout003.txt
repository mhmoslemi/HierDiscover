```
def max_conv_fft(a):
    n = len(a)
    fft_len = 1 << ((2*n -1).bit_length())
    F = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[:2*n -1]
    return conv.max()
```

```
def projected_gradient(seq, target_sum, max_iter, step_start, deadline):
    n = len(seq)
    best_local = seq.copy()
    best_val = max_conv_fft(seq)
    step = step_start
    for iter in range(max_iter):
        if time.time() >= deadline:
            break
        # compute conv via FFT
        fft_len = 1 << ((2*n -1).bit_length())
        F = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[:2*n -1]
        k_max = np.argmax(conv)
        # compute gradient
        grad = np.zeros_like(seq)
        idx_arr = np.arange(n)
        j = k_max - idx_arr
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * seq[j[mask]]
        # step direction opposite gradient
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        step_vec = -step * grad / norm
        tentative = seq + step_vec
        # clip
        tentative = np.clip(tentative, 0.0, 1000.0)
        # project onto simplex (sum = target_sum)
        seq = projection_simplex(tentative, target_sum)

        cur_val = max_conv_fft(seq)
        if cur_val < best_val:
            best_val = cur_val
            best_local = seq.copy()
        # adapt step
        if (iter+1) % 1000 == 0:
            step *= 0.9
        # occasionally random jitter
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * step * 0.1
            seq = projection_simplex(np.clip(seq + jitter, 0.0, 1000.0), target_sum)
    return best_local, best_val
```

assistantfinal**Algorithm summary**  
The search proceeds in three stages:

1. **Exploration of several problem sizes** – lengths `n` from 2000 up to 20000 are tried.  
2. **Wide‑range seeding** – for each `n` a collection of simple families (random Dirichlet, Gaussian, power‑law, geometric and the previously found `height_sequence_1`) is generated and the fast FFT‑based convolution‐max is evaluated. The best seed per length is kept.  
3. **Local polishing** – starting from the best seed a projected gradient descent (sub‑gradient of the maximal convolution entry) with occasional jitter is run until the time budget is exhausted.  The iterate with the smallest maximal convolution value is recorded.  

All constraints (non‑negativity, sum = √(2 n), clipping to [0, 1000]) are enforced by a simplex projection.  The expensive `evaluate_sequence` function is called **only once**, for the final candidate.

```python
"""
Propose a candidate step‑function that minimizes the evaluation metric.

The algorithm:
  1. Tries several lengths n (2000 … 20000).  For each n a set of
     inexpensive seeds (random Dirichlet, Gaussian, power‑law,
     geometric, and the optional previous best) is generated.
  2. The seed with the smallest maximal self‑convolution (computed via
     FFT, i.e. max(a*a)) is taken as the starting point.
  3. A projected gradient descent with adaptive step size and occasional
     random jitter refines the candidate while keeping the vector on the
     simplex {x ≥ 0, Σx = √(2 n)}.
  4. The best sequence found over all lengths is finally checked with the
     provided evaluate_sequence() and returned.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, Σx = s}.
    Implementation of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def max_convolution_fft(a: np.ndarray) -> float:
    """
    Return max_k (a * a)[k] using an FFT of length power‑of‑two.
    The vector a is assumed non‑negative; clipping is handled elsewhere.
    """
    n = a.size
    # length for linear convolution (next power of two ≥ 2n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


# --------------------------------------------------------------------------- #

                   target_sum: float,

                   global_start: float,

    Projected (sub‑)gradient descent on the maximal self‑convolution.
    Returns the best sequence seen and its maximal convolution value.
    """
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())

    best_seq = seq.copy()
    best_val = max_convolution_fft(seq)

    step_scale = 0.5          # initial absolute step magnitude (relative to avg height)
    min_step = 1e-9
    decay_interval = 800
    decay_factor = 0.85

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current convolution and the indices with largest values
        F = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]

        # take the three largest positions (helps when the maximum is not unique)
        top_k = np.argpartition(conv, -3)[-3:]

        # sub‑gradient of max‑functional (sum of gradients of the active constraints)
        grad = np.zeros_like(seq)
        idx = np.arange(n)
        for k in top_k:
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * seq[j[mask]]

        norm = np.linalg.norm(grad)
        if norm == 0.0:          # degenerate – nothing to move
            break

        # step size proportional to average entry size
        step = step_scale * target_sum / n
        tentative = seq - step * grad / norm
        tentative = np.clip(tentative, 0.0, 1000.0)
        seq = projection_simplex(tentative, target_sum)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.02:
            jitter = np.random.randn(n) * step * 0.1
            seq = projection_simplex(np.clip(seq + jitter, 0.0, 1000.0), target_sum)

        cur_val = max_convolution_fft(seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = seq.copy()
            print(f"[{time.time() - global_start:6.1f}s] iter {it:6d} → new local best {best_val:.6f}")
            step_scale = max(step_scale * 0.9, min_step)

        # step‑size schedule
        if (it + 1) % decay_interval == 0:
            step_scale = max(step_scale * decay_factor, min_step)

    return best_seq, best_val


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimizes the provided evaluation
    function.  The whole search stops after `budget_s` seconds.
    """
    np.random.seed(seed)
    global_start = time.time()
    deadline = global_start + budget_s - 0.2   # safety margin

    best_overall_seq = None
    best_overall_val = float("inf")

    # ------------------------------------------------------------------- #
    # 0. If a previous construction exists, keep it as a possible seed.
    previous_seed = None
    if "height_sequence_1" in globals():
        try:
            prev = np.array(globals()["height_sequence_1"], dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.clip(prev, 0.0, 1000.0)
                target = np.sqrt(2.0 * prev.size)
                if prev.sum() > 0:
                    prev = prev / prev.sum() * target
                previous_seed = prev
                print(f"[{time.time() - global_start:6.1f}s] Loaded previous seed (n={prev.size})")
        except Exception:   # pragma: no cover
            previous_seed = None

    # ------------------------------------------------------------------- #
    # 1. Choose a collection of problem sizes.
    # Larger n give a smaller theoretical lower bound (≈1) but cost more.
    n_list = [2000, 4000, 8000, 12000, 20000]

    for n in n_list:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        if remaining < 5.0:
            # not enough time for another length
            break

        target_sum = np.sqrt(2.0 * n)
        print(f"[{time.time() - global_start:6.1f}s] === Trying n = {n} (target sum = {target_sum:.4f})")

        # ---------------------------------------------------------------- #
        # 2. Generate a variety of seeds.
        seeds = []

        # a) previous construction (if same length, else resize with simple interpolation)
        if previous_seed is not None:
            if previous_seed.size == n:
                seeds.append(previous_seed.copy())
            else:
                # simple linear interpolation / truncation to the new size
                x_old = np.linspace(0, 1, previous_seed.size)
                x_new = np.linspace(0, 1, n)
                interp = np.interp(x_new, x_old, previous_seed)
                interp = np.clip(interp, 0.0, 1000.0)
                interp = interp / interp.sum() * target_sum
                seeds.append(interp)

        # b) Random Dirichlet samples (uniform over the simplex)
        num_dirichlet = 30 if n <= 8000 else 15
        for _ in range(num_dirichlet):
            if time.time() >= deadline:
                break
            a = np.random.gamma(1.0, 1.0, size=n)
            a = a / a.sum() * target_sum
            a = np.clip(a, 0.0, 1000.0)
            seeds.append(a)

        # c) Discrete Gaussian shapes
        center = (n - 1) / 2.0
        sigma_vals = np.linspace(0.05 * n, 0.45 * n, 12)
        for sigma in sigma_vals:
            if time.time() >= deadline:
                break
            i = np.arange(n, dtype=float)
            a = np.exp(- ((i - center) ** 2) / (2.0 * sigma ** 2))
            a = a / a.sum() * target_sum
            a = np.clip(a, 0.0, 1000.0)
            seeds.append(a)

        # d) Power‑law (a_i ∝ (i+1)^{-p})
        idx = np.arange(1, n + 1, dtype=float)
        p_vals = np.linspace(0.2, 2.0, 12)
        for p in p_vals:
            if time.time() >= deadline:
                break
            a = idx ** (-p)
            a = a / a.sum() * target_sum
            a = np.clip(a, 0.0, 1000.0)
            seeds.append(a)

        # e) Geometric families (a_i = C·r^i)
        lambdas = np.logspace(-5, -1, 20)
        for lam in lambdas:
            if time.time() >= deadline:
                break
            r = 1.0 - lam
            if r <= 0.0:
                continue
            # sum of geometric series = C * (1 - r**n) / (1 - r)
            denom = 1.0 - r ** n
            if denom <= 0.0:
                continue
            C = target_sum * lam / denom
            a = C * (r ** np.arange(n))
            a = np.clip(a, 0.0, 1000.0)
            seeds.append(a)

        # ---------------------------------------------------------------- #
        # 3. Pick the best seed using the cheap FFT‑based objective.
        best_local_seq = None
        best_local_val = float("inf")
        for a in seeds:
            if time.time() >= deadline:
                break
            val = max_convolution_fft(a)
            if val < best_local_val:
                best_local_val = val
                best_local_seq = a.copy()
        print(f"[{time.time() - global_start:6.1f}s] n={n} best seed value = {best_local_val:.6f}")

        # ---------------------------------------------------------------- #
        # 4. Local polishing with projected gradient descent.
        if best_local_seq is not None:
            # allocate a fraction of the remaining time for this size
            size_deadline = time.time() + min(remaining * 0.5, 30.0)   # at most ~30 s per n
            polished_seq, polished_val = local_optimize(
                best_local_seq,
                target_sum,
                size_deadline,
                global_start,
                max_iter=4000,
            )
            # keep the better of the raw seed and the polished one
            if polished_val < best_local_val:
                best_local_val = polished_val
                best_local_seq = polished_seq

        # ---------------------------------------------------------------- #
        # 5. Record the best sequence across all sizes.
        if best_local_seq is not None and best_local_val < best_overall_val:
            best_overall_val = best_local_val
            best_overall_seq = best_local_seq.copy()
            print(f"[{time.time() - global_start:6.1f}s] ** New global best (n={n}) = {best_overall_val:.6f}")

    # ------------------------------------------------------------------- #
    # Final safety check with the official evaluator (slow but accurate).
    if best_overall_seq is None:
        # fall back to a uniform construction
        n = 3000
        target_sum = np.sqrt(2.0 * n)
        best_overall_seq = np.full(n, target_sum / n)

    final_score = evaluate_sequence(best_overall_seq.tolist())
    print(f"[{time.time() - global_start:6.1f}s] Search finished. Final score = {final_score:.6f}")

    return best_overall_seq.tolist()
```
