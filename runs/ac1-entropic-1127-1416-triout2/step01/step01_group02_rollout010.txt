def project_to_simplex(v):
    # v may not sum to 1 or be non-negative
    # algorithm from https://arxiv.org/pdf/1101.6081.pdf
    n = len(v)
    u = np.sort(v)[::-1]

    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def binomial_sequence(length):

    for k in range(1,length):

"""Propose a step‑function heights sequence.

The algorithm starts from the (theoretically) optimal central‑binomial
sequence, normalises it to a unit L¹‑norm and then performs a
projected gradient‑descent on a smooth approximation of the
``max(convolution)`` term in the evaluation function.  The smooth
approximation is a soft‑max (log‑sum‑exp) with an annealed temperature.
Each gradient step is projected onto the probability simplex
(non‑negative entries that sum to one) so the iterate stays feasible.
Occasional random perturbations and a simple pair‑transfer local move
help to escape plateaus.  The best sequence seen within the time budget
is returned.

The optimisation directly minimises the quantity
    max(a*a) / (sum(a)**2)
which is scale‑invariant; fixing sum(a)=1 lets us work only with the
max‑convolution term.
"""

import time
import numpy as np

# -------------------------------------------------------------
# helpers
def binomial_sequence(length: int) -> np.ndarray:
    """Central binomial coefficients c_k = (2k choose k) / 4**k."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(seq: np.ndarray, step: float = 0.02) -> np.ndarray:
    """Move a tiny amount of mass from one coordinate to another."""
    n = seq.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j or seq[i] == 0.0:
        return seq.copy()
    amount = step * np.random.rand() * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= amount
    new_seq[j] += amount
    # safety against rounding negatives
    new_seq[i] = max(0.0, new_seq[i])
    return new_seq


# -------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (heuristically) minimises
    the provided ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int
        Random seed.
    budget_s : int
        Time budget in seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # ---------------------------------------------------------
    # sequence length – use the length of the previously known
    # construction if it exists, otherwise a reasonable default.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 5000

    # ---------------------------------------------------------
    # start from the classical central‑binomial construction,
    # normalised to sum = 1 (scale does not affect the evaluation).
    a = binomial_sequence(n).astype(np.float64)
    a = a / a.sum()
    # tiny random jitter to break perfect symmetry caused by floating
    a += 1e-12 * np.random.rand(n)
    a = project_to_simplex(a)

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ---------------------------------------------------------
    # optimisation hyper‑parameters
    lr = 0.5               # learning‑rate for the projected gradient step
    min_lr = 1e-5
    lr_decay = 0.9995

    beta = 1.0             # soft‑max temperature (larger → closer to true max)
    beta_target = 30.0
    beta_inc = 1.001

    iter_count = 0
    # main optimisation loop
    while time.time() < deadline:
        # -----------------------------------------------------
        # 1) evaluate convolution and smooth weights
        b = np.convolve(a, a)               # linear convolution, length 2n‑1
        max_b = b.max()
        # soft‑max weights (stable formulation)
        shift = max_b
        weights = np.exp(beta * (b - shift))
        weights /= weights.sum()            # now sum(weights)=1

        # -----------------------------------------------------
        # 2) gradient of the smooth max (w.r.t. a)
        #    g_i = 2 * sum_k weights[k] * a_{k-i}
        #    which is exactly 2 * (weights * a[::-1]) sliced appropriately
        conv_full = np.convolve(weights, a[::-1], mode='full')
        g = 2.0 * conv_full[n - 1:n - 1 + n]   # length‑n gradient

        # -----------------------------------------------------
        # 3) gradient step with projection onto the simplex
        a_new = a - lr * g
        a_new = project_to_simplex(a_new)

        # -----------------------------------------------------
        # 4) evaluate candidate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"[iter {iter_count}] new best {best_score:.6f}"
                  f"  t={time.time() - (deadline - budget_s):.2f}s")

        # -----------------------------------------------------
        # 5) occasional random perturbation (helps to leave plateaus)
        if np.random.rand() < 0.001:
            a_new = random_pair_transfer(a_new, step=0.02)
            a_new = project_to_simplex(a_new)

        # -----------------------------------------------------
        # 6) update state and hyper‑parameters
        a = a_new
        lr = max(lr * lr_decay, min_lr)
        beta = min(beta * beta_inc, beta_target)
        iter_count += 1

    print(f"[final] best score={best_score:.6f}")
    return best_a.tolist()
