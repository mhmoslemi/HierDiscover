    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the largest rho satisfying u[rho] > cssv[rho] / (rho + 1)
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]

        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
