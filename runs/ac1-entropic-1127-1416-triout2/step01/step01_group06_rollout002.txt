"""
Search for a low‑score step function.

The evaluation metric is 2·n·max(conv(f,f)) / (sum(f)²) and is invariant under
global scaling.  Theory shows that a monotone power‑law
    a_i ∝ (i+1)^{-p}   (p>½)
gives a much smaller constant than the uniform sequence (p=0).  By scanning the
exponent p and the length n we can quickly locate a shape with score well below
the target 1.503.  After a short deterministic grid search we perform a light
hill‑climbing phase that perturbs the best candidate and accepts improvements.
All candidates are normalised so that max(a)=1 (the objective does not depend on
the scale, this also guarantees the values stay inside the safe range of
``evaluate_sequence``).

The routine runs for roughly ``budget_s`` seconds and returns the best sequence
found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
#  Helper utilities (must be top‑level – no closures)
# ----------------------------------------------------------------------
def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:               # pragma: no cover
        return math.inf


def _normalize(seq):
    """Scale so that max(seq) == 1 (no‑op if all zeros)."""
    arr = np.asarray(seq, dtype=float)
    mx = arr.max()
    if mx > 0.0:
        arr = arr / mx
    return arr.tolist()


def _pow_seq(n, p):
    """Power‑law decay a_i = (i+1)^(-p)  (i starts at 0)."""
    # i+1 to avoid division by zero; the first term is 1 before normalisation.
    idx = np.arange(1, n + 1, dtype=float)
    seq = idx ** (-p)
    return _normalize(seq)


def _exp_seq(n, alpha):
    """Exponential decay a_i = exp(-alpha·i)."""
    seq = np.exp(-alpha * np.arange(n, dtype=float))
    return _normalize(seq)


def _randdec_seq(n):
    """Random decreasing sequence (sorted random numbers)."""
    arr = np.random.rand(n)
    arr.sort()
    arr = arr[::-1]                     # descending
    return _normalize(arr)


# ----------------------------------------------------------------------
#  Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The routine spends roughly ``budget_s`` seconds
    searching; it first scans a grid of power‑law exponents (the theoretically
    promising family) and then runs a lightweight stochastic hill‑climbing
    refinement.
    """
    np.random.seed(seed)
    random.seed(seed)

    # tiny safety margin so we do not exceed the budget
    deadline = time.time() + budget_s - 0.3

    best_seq = None
    best_score = math.inf

    def _try(seq, description):
        """Evaluate *seq*, update global best if it improves, and print a message."""
        nonlocal best_seq, best_score
        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq
            print(f"New best ({description}) → {best_score:.6f} (len={len(seq)})")
        return sc

    # ------------------------------------------------------------------
    # 0) start from the globally supplied construction if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = _normalize(list(height_sequence_1))
            _try(init, "initial reference")
        except Exception:               # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1) Deterministic grid search over power‑law families
    # ------------------------------------------------------------------
    # Lengths we are willing to try – they are modest so that many
    # evaluations fit easily in the time budget.
    lengths = [800, 1200, 1600, 2000, 2400, 2800, 3200, 3600, 4000]

    # Exponents p > 0.5 are the promising region; we sweep it with a
    # moderate resolution.  The step size is deliberately coarse – the
    # hill‑climbing phase later refines the best point.
    exponents = [0.55 + 0.02 * i for i in range(int((0.95 - 0.55) / 0.02) + 1)]

    for n in lengths:
        if time.time() > deadline:
            break
        for p in exponents:
            if time.time() > deadline:
                break
            seq = _pow_seq(n, p)
            # a guard against pathological all‑zero vectors (should not happen)
            if sum(seq) < 1e-8:
                continue
            _try(seq, f"pow n={n} p={p:.2f}")

    # ------------------------------------------------------------------
    # 2) Light stochastic hill‑climbing around the current best
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Occasionally perform a more global random draw (different family)
        if best_seq is not None and random.random() < 0.15:
            # ----- hill‑climbing step -----------------------------
            cand = np.array(best_seq, copy=True)
            n = len(cand)

            # modify a few random positions multiplicatively (log‑normal perturbation)
            m = random.randint(1, min(5, n))
            for _ in range(m):
                i = random.randrange(n)
                factor = 10 ** random.uniform(-0.12, 0.12)   # ≈[0.75,1.33]
                cand[i] = max(0.0, cand[i] * factor)

            # re‑normalise (scaling does not affect the score)
            mx = cand.max()
            if mx > 0.0:
                cand = cand / mx
            _try(cand.tolist(), "hill‑climb")

        else:
            # ----- random parametric candidate --------------------
            # Randomly pick a family and parameters; this keeps the search
            # diverse and may discover an unexpected shape.
            n = int(random.choice(lengths))
            family = random.choice(['pow', 'exp', 'randdec'])

            if family == 'pow':
                p = random.uniform(0.55, 0.98)
                cand = _pow_seq(n, p)
                _try(cand, f"rand pow n={n} p={p:.3f}")

            elif family == 'exp':
                # exponential decay, α in a log‑uniform range
                log_a = random.uniform(-4.0, math.log10(5.0))
                alpha = 10 ** log_a
                cand = _exp_seq(n, alpha)
                _try(cand, f"rand exp n={n} α={alpha:.3e}")

            else:  # 'randdec'
                cand = _randdec_seq(n)
                _try(cand, f"randdec n={n}")

    # ------------------------------------------------------------------
    # 3) final fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        # a simple uniform sequence – its score is exactly 2.
        best_seq = [1.0] * 1000
        best_score = _score(best_seq)
        print("Fallback – uniform sequence score:", best_score)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
