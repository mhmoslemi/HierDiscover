def lp_step(a, M):
    n = len(a)
    # A shape (2n-1, n)
    rows = []
    data = []
    col_ind = []
    row_ind = []
    for k in range(2*n-1):
        start = max(0, k - (n-1))
        end = min(k, n-1)
        # entries for j from start..end where a_{k-j} used.
        # For each j, coefficient = a_{k-j}
        for j in range(start, end+1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    # Build sparse matrix
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # b: right-hand side M * 1
    b = np.full(2*n-1, M)
    # objective minimize -sum(b)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        b0 = res.x
        # Normalize to sum=1
        sumb = b0.sum()
        if sumb > 0:
            g = b0 / sumb
            # combine
            t = 0.2  # mixing factor
            new_a = (1 - t) * a + t * g
            new_a /= new_a.sum()
            return new_a, res.fun  # negative optimum? Not needed.
        else:
            return a, None
    else:
        return a, None

"""
Propose a candidate step‑function sequence.

The algorithm combines three ideas:

1. **Constant‑convolution base** – the classic sequence a_k=C(2k,k)/4^k,
   whose auto‑convolution is (almost) constant and gives the
   π/2≈1.5708 bound.

2. **Mixture search** – we build a few simple families (uniform,
   reversed, shifted copies of the base) and scan convex combinations
   of the base with each family.  Because the bound depends on the
   maximum of the convolution, moving mass from the head of the base
   to later positions can lower that maximum without raising any
   entry above it.

3. **Local refinement** – starting from the best mixture we run a
   projected sub‑gradient descent that reduces the peak of the
   convolution.  Occasionally we also try a smoothed‑max step
   (log‑sum‑exp) and random Dirichlet perturbations to escape plateaus.

All steps respect the 1000 s time budget, the sequence is kept
non‑negative and normalised to L1‑norm 1, and the final result is
checked with the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _constant_conv_base(n: int) -> np.ndarray:
    """
    First n terms of a_k = C(2k,k)/4^k = binom(2k,k)/4**k.
    Recurrence: a_0 = 1, a_k = a_{k‑1} * (1 - 1/(2k)).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Fast convolution of a with itself using real FFT.
    Returns the length 2·len(a)-1 array.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(Af * Af, n=size)[: 2 * n - 1]
    return conv_full


def _score_from_seq(a: np.ndarray) -> float:
    """
    Return the evaluation value for a normalised sequence
    (i.e. L1‑norm equal to 1).  This is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


def _project_and_normalise(v: np.ndarray) -> np.ndarray:
    """
    Project onto the non‑negative orthant and re‑normalise to L1‑norm 1.
    If the vector becomes all zeros we replace it by a random probability
    vector.
    """
    v = np.maximum(v, 0.0)
    tot = v.sum()
    if tot <= 0.0:
        v = np.random.rand(v.shape[0])
        v /= v.sum()
    else:
        v /= tot
    return v


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for minimising  max(conv(a,a)).
    Uses the *average* gradient over all active (maximal) constraints.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()
    # active set – all positions whose value is within tiny relative tolerance
    eps = max_val * 1e-12
    active = np.where(conv >= max_val - eps)[0]

    # accumulate gradient contributions from each active index
    g = np.zeros_like(a)
    for k in active:
        if k < n:
            # indices 0 … k are valid
            g[: k + 1] += a[k::-1]
        else:
            # k ≥ n
            start = k - (n - 1)                 # first i with k‑i < n
            length = 2 * n - 1 - k              # number of valid positions
            g[start:] += a[:length][::-1]

    # average and the factor 2 from the derivative of a*a
    g = 2.0 * g / max(1, active.size)

    a_new = a - step * g
    return _project_and_normalise(a_new)


def _smooth_max_step(a: np.ndarray, step: float, temperature: float) -> np.ndarray:
    """
    Smoothed max via log‑sum‑exp.  Returns a projected step.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()

    # stabilised softmax weights
    exp_vals = np.exp((conv - max_val) / temperature)
    w = exp_vals / exp_vals.sum()

    # gradient = 2 * (w * a) reversed convolution
    rev_a = a[::-1]
    tmp = np.convolve(w, rev_a)  # length = (2n‑1)+n‑1
    g = 2.0 * tmp[n - 1 : n - 1 + n]

    a_new = a - step * g
    return _project_and_normalise(a_new)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a sequence that minimises `evaluate_sequence`.
    Returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a length (reuse a previous solution if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
        a = np.array(height_sequence_1, dtype=np.float64)
        a = _project_and_normalise(a)            # ensure proper normalisation
    else:
        n = 2000
        base = _constant_conv_base(n)
        a = base / base.sum()                     # normalised base sequence

    best_seq = a.copy()
    best_score = _score_from_seq(best_seq)
    print(f"Initial length {n}, initial score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Build a small library of simple shapes to mix with the base
    # ------------------------------------------------------------------
    base = _constant_conv_base(n)
    base_norm = base / base.sum()

    uniform = np.full(n, 1.0 / n)                # uniform distribution
    rev_base = base_norm[::-1]                   # reversed base

    # shifted copies (step size ≈ 5% of n)
    shift_step = max(1, n // 40)

    # ------------------------------------------------------------------

    for shape, name in shape_library:
        if name == "base":

        for w in np.linspace(0.0, 1.0, 21):      # 0, 0.05, … ,1
            cand = (1.0 - w) * base_norm + w * shape

            cand = _project_and_normalise(cand)

        sc = _score_from_seq(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

        # ----- every few thousand iterations do a deterministic step -----
        if it % 3000 == 0 and it > 0:
            # choose between plain sub‑gradient and smoothed step
            if np.random.rand() < 0.5:
                step = 0.3 / np.sqrt(it + 1.0)
                best_seq = _subgradient_step(best_seq, step)
            else:
