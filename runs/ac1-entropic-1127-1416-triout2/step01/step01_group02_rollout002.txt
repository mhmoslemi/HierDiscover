"""
Optimising a step‑function by reducing the peak of its autoconvolution.

The evaluation function is
    score = 2 * n * max(a*a) / (sum(a)**2)

where a*a is the linear convolution of the non‑negative list a.
The classic “central binomial’’ construction yields max(a*a)=1 and
score≈π/2≈1.571.  The optimum of the problem can be written as

    maximise   S = sum(a_i)
    subject to   (a*a)[k] ≤ 1   for all k

(the scaling invariance makes the bound M=1 without loss of generality).

The binomial sequence saturates every constraint, including the
k=0 constraint a0²≤1, which forces a0=1.  By moving a small amount of
mass away from the indices that most influence the current maximal
convolution entry we can lower the peak while keeping the total sum
essentially unchanged, thus improving the score.

The algorithm:

* start from the binomial (central) sequence;
* while time remains
    – compute the convolution, locate the index k* of the current peak,
      and the gradient g_i = a_{k*‑i} (zero when the index is out of range);
    – choose a source index i_src where g_i is maximal (largest contribution
      to the peak) and a destination index i_dst where g_i is minimal
      (ideally zero);
    – transfer a fraction of a[i_src] to a[i_dst] (the “step’’ size decays
      slowly);
    – accept the move if the score improves (or with a tiny probability);
    – keep the best sequence seen.

The move is deterministic and directly targets the constraints that are
active, so it reduces the maximum of the convolution much faster than a
pure random pair‑transfer.  A small amount of stochastic restarts and a
tiny chance to accept worse moves help escape plateaus.

The function returns the best sequence found within the allotted time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Central binomial coefficients c_k = C(2k,k)/4**k, k=0..length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def fast_score(arr: np.ndarray) -> float:
    """Direct evaluation of the objective (no clipping needed for our range)."""
    s = np.sum(arr)
    if s < 0.01:
        return np.inf
    conv = np.convolve(arr, arr)
    m = np.max(conv)
    n = arr.shape[0]
    return float(2.0 * n * m / (s * s))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑height sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The function returns as soon as the budget
        is exhausted.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # small safety margin

    # ------------------------------------------------------------------
    # Determine the length of the candidate.  Prefer the length of the
    # previously known construction if it exists.
    try:
        n = len(globals()["height_sequence_1"])   # type: ignore[name-defined]
    except Exception:
        n = 5000

    # initialise with the optimal “binomial’’ sequence
    a = binomial_sequence(n)
    cur_score = fast_score(a)
    best_score = cur_score
    best_a = a.copy()
    conv = np.convolve(a, a)

    print(f"[init] n={n}, score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # parameters for the guided mass‑transfer search
    step_frac = 0.20          # initial fraction of source mass to move
    decay = 0.9998           # shrink the step slowly
    min_step = 1e-6
    iters_without_improve = 0
    max_no_improve = 2000    # after many stalled iterations we re‑boost step

    while time.time() < deadline:
        # locate the current maximum of the convolution
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # gradient of the maximal convolution entry w.r.t. a:
        #   g_i = a_{k_max-i}   (0 when the index is out of bounds)
        idx = np.arange(n)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)
        grad = np.zeros_like(a)
        grad[valid] = a[partner[valid]]

        # source = index with largest gradient (most responsible for the peak)
        src = int(np.argmax(grad))
        # destination = index with smallest gradient (ideally zero)
        zero_mask = grad == 0.0
        if np.any(zero_mask):
            # pick a random zero‑gradient index that is not the source
            candidates = np.where(zero_mask)[0]
            candidates = candidates[candidates != src]
            if candidates.size == 0:
                dst = int(np.argmin(grad))
            else:
                dst = int(np.random.choice(candidates))
        else:
            # fall back to the minimal gradient index
            dst = int(np.argmin(grad))
            if dst == src:
                # pick a random different index
                dst = (src + 1) % n

        # amount to move – a fraction of the source entry
        amount = step_frac * a[src]
        if amount <= 0.0:
            # nothing to move – shrink the step and continue
            step_frac = max(step_frac * decay, min_step)
            continue

        # propose the new sequence
        cand = a.copy()
        cand[src] -= amount
        cand[dst] += amount

        # (numerical safety)
        if cand[src] < 0.0:
            cand[src] = 0.0

        cand_score = fast_score(cand)

        # accept if it improves, or with a tiny probability to escape flat regions
        if cand_score < cur_score or np.random.rand() < 1e-4:
            a = cand
            conv = np.convolve(a, a)
            cur_score = cand_score
            iters_without_improve = 0

            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                elapsed = time.time() - (deadline - budget_s)
                print(f"[improve] score={best_score:.6f}  time={elapsed:.2f}s")
        else:
            iters_without_improve += 1

        # slowly decay the step size
        step_frac = max(step_frac * decay, min_step)

        # occasional random “shake‑up’’ to avoid getting stuck
        if np.random.rand() < 0.005:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                eps = step_frac * a[i]
                tmp = a.copy()
                tmp[i] -= eps
                tmp[j] += eps
                tmp_score = fast_score(tmp)
                if tmp_score < best_score:
                    a, conv, cur_score = tmp, np.convolve(tmp, tmp), tmp_score
                    best_score, best_a = tmp_score, tmp.copy()
                    elapsed = time.time() - (deadline - budget_s)
                    print(f"[rand‑improve] score={best_score:.6f} time={elapsed:.2f}s")

        # if we have stalled for a while, give the step a quick boost
        if iters_without_improve > max_no_improve:
            step_frac = 0.20
            iters_without_improve = 0

        # early exit if we have already reached the target region
        if best_score < 1.52:       # well below π/2, comfortably under the known bound
            # keep searching a little longer – the loop will finish when time expires
            pass

    print(f"[final] best score={best_score:.6f}")
    return best_a.tolist()
