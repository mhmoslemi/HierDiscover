    n = len(seq)

"""
Hybrid LP + Projected Gradient Search for Low‑Peak Autocorrelation

The evaluation function (under the natural normalization Σa = √(2·n))
reduces to the peak of the auto‑convolution:
    score = max_k (a * a)[k] .
Hence we minimise the convolution peak while keeping the L1‑norm fixed.

Algorithm
---------
1. **Initial geometric seed** – Scan a one‑parameter geometric family
   a_i = C·r^i (r≈1) for a range of decay rates and keep the best
   sequence (evaluated with the provided ``evaluate_sequence``).

2. **Projected gradient descent** – Repeatedly locate all indices where the
   convolution attains its current maximum, build the sub‑gradient
   (sum of 2·a_{k−i} over those indices) and take a step opposite to it.
   After each step the vector is projected onto the simplex
   {x ≥ 0, Σx = √(2·n)}.

3. **Linear‑programming boost** – Once (or if SciPy is available) we solve
   the linear program
        maximise   Σ g_j
        s.t.       (f * g)_k ≤ M   ,   ∀k
                    g_j ≥ 0
   where M = max_k (f*f)[k] of the current sequence f.
   The solution g_0 is rescaled to the required sum and mixed with f
   (convex combination t = 0.5).  This step can strictly lower the peak.

4. **Loop** – Alternate gradient steps and (at most one) LP step until the
   time budget expires, keeping the best sequence seen.

All numbers are clipped to the range required by ``evaluate_sequence``.
The routine prints progress messages and finally returns the best list
found within the allotted 1000 s.
"""

import time
import math
import numpy as np

# optional SciPy import – if unavailable we simply skip the LP boost
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        { x | x_i ≥ 0 , Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_max(seq: np.ndarray) -> float:
    """
    Compute the maximum value of the full auto‑convolution of ``seq``.
    Uses an FFT of length next power‑of‑two for speed.
    """
    n = seq.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


def solve_lp_for_g(f: np.ndarray, M: float, target_sum: float,
                   time_limit: float = 30.0) -> np.ndarray:
    """
    Solve the linear program
        maximise Σ g_j
        s.t.   (f * g)_k ≤ M  for all k
               g_j ≥ 0
    Returns a feasible vector g with Σ g = target_sum (scaled after solving).
    If SciPy is not available or the solver fails, the current f is returned.
    """
    if not _SCIPY_AVAILABLE:
        raise RuntimeError("SciPy not available")

    n = f.shape[0]
    # Build the constraint matrix A (shape (2n‑1, n)).
    # Row k, column j contains f_{k‑j} when 0 ≤ k‑j < n, else 0.
    i = np.arange(n)[:, None]                # shape (n,1)
    j = np.arange(n)[None, :]                # shape (1,n)
    rows = (i + j).ravel()                   # size n*n
    cols = np.tile(np.arange(n), n)          # size n*n
    data = np.tile(f, n)                     # repeat f for each column
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Upper‑bound vector (M for every convolution index)
    b = np.full(2 * n - 1, M, dtype=float)

    # Objective: minimise -Σ g  (i.e. maximise Σ g)
    c = -np.ones(n, dtype=float)

    bounds = [(0, None)] * n

    # ``linprog`` does not have a direct time‑limit option, but we can
    # abort manually if we exceed the allotted budget.
    # The solver is fast for the modest sizes we use (n ≤ 3000).
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    g0 = np.asarray(res.x, dtype=float)

    # Rescale to the required L1‑norm
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        raise RuntimeError("LP returned zero vector")
    g = g0 * (target_sum / sum_g0)
    # Clip to the numerical limits that evaluate_sequence enforces
    g = np.clip(g, 0.0, 1000.0)
    return g


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the provided evaluation function.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------------- #
    # 1. Problem size ------------------------------------------------------ #
    # Try to reuse a previously found construction if it exists.
    init_seq = globals().get('height_sequence_1')
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        init_seq = np.asarray(init_seq, dtype=float)
        if init_seq.ndim != 1:
            init_seq = None
    else:
        init_seq = None

    # Default length; larger n offers more freedom but costs more per iteration.
    n = 3000 if init_seq is None else int(len(init_seq))
    target_sum = math.sqrt(2.0 * n)          # the “natural” normalisation

    # --------------------------------------------------------------------- #
    # 2. Initialise with a geometric family -------------------------------- #
    best_val = float('inf')
    best_seq = None

    def eval_fast(seq_arr):
        return evaluate_sequence(seq_arr.tolist())

    # Scan a modest grid of decay rates λ = 1‑r  (log‑spaced)
    lambdas = np.logspace(-5, -0.5, 80)   # λ ∈ [1e‑5, 0.3]
    idx = np.arange(n, dtype=float)

    for lam in lambdas:
        r = 1.0 - lam
        # Avoid under‑flow when rⁿ is essentially zero.
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom        # ensures Σ a = target_sum
        seq = C * (r ** idx)
        seq = np.clip(seq, 0.0, 1000.0)
        val = eval_fast(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # Power‑law fallback (a_i ∝ 1/√(i+1))
    a_pow = 1.0 / np.sqrt(idx + 1.0)
    C_pow = target_sum / a_pow.sum()
    seq_pow = np.clip(C_pow * a_pow, 0.0, 1000.0)
    val_pow = eval_fast(seq_pow)
    if val_pow < best_val:
        best_val = val_pow
        best_seq = seq_pow.copy()

    # If a previous construction is available it may already be superior.
    if init_seq is not None:
        # Project it onto the required simplex (just in case)
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        init_seq = projection_simplex(init_seq, target_sum)
        val_init = eval_fast(init_seq)
        if val_init < best_val:
            best_val = val_init
            best_seq = init_seq.copy()

    # Fallback safety
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # Current working sequence ------------------------------------------------
    f = best_seq.copy()
    best_score = best_val
    best = f.copy()

    # --------------------------------------------------------------------- #
    # 3. Helper utilities --------------------------------------------------- #
    idx_arr = np.arange(n)

    # 4. Main optimisation loop -------------------------------------------- #
    it = 0
    lp_done = False          # we perform at most one LP boost
    last_eval = time.time()
    while time.time() < deadline:
        remaining = deadline - time.time()

        # --------------------------------------------------------------- #

            try:

                                   time_limit=remaining - 4.0)

                t = 0.5

                f = projection_simplex(f_new, target_sum)

                cur_score = evaluate_sequence(f.tolist())
                if cur_score < best_score:
                    best_score = cur_score
                    best = f.copy()
                    print(f"[{time.time() - start_time:6.1f}s] LP boost → new best {best_score:.6f}")
                lp_done = True
                continue
            except Exception as exc:                # pragma: no cover
                print(f"[{time.time() - start_time:6.1f}s] LP failed ({exc}), skipping")
                lp_done = True    # avoid repeated attempts

        # --------------------------------------------------------------- #
        # 4.2 Projected gradient descent step ----------------------------- #
        fft_len = 1 << ((2 * n - 1).bit_length())
        F = np.fft.rfft(f, n=fft_len)

        for k in max_idxs:
            j = k - idx_arr

        else:
            direction = np.zeros_like(f)

        tentative = f + step * direction
        tentative = np.clip(tentative, 0.0, 1000.0)
        f = projection_simplex(tentative, target_sum)

        if np.random.rand() < 0.015:

            f = projection_simplex(np.clip(f + jitter, 0.0, 1000.0), target_sum)

        # --------------------------------------------------------------- #
        # 4.3 Periodic evaluation (every few thousand iterations or when time is short)
        if it % 2000 == 0 or (remaining < 5.0 and time.time() - last_eval > 0.4):
            cur_score = evaluate_sequence(f.tolist())

    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished after {it} iterations. Best score = {final_score:.6f}")
    return best.tolist()
