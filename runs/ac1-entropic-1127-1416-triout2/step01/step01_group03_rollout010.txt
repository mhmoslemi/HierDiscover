#!/usr/bin/env python3
"""
Propose a non‑negative step‑function sequence that minimises
`evaluate_sequence`.

Algorithm (single–CPU, 1000 s budget)

1. Choose a relatively large length (default 5000).  A larger n reduces
   discretisation error of the continuous optimisation problem.
2. Initialise the vector as a mixture of two analytic candidates:
      * the discrete (1‑z)^(-1/2) coefficients (the “square‑root’’ shape);
      * the arcsine distribution on [0,1] (symmetric, singular at the
        ends).  Add a tiny random jitter and project onto the probability
        simplex (sum = 1).
3. Optimise a smooth surrogate of the max‑convolution:
      * let `b = a * a` (self‑convolution).  The objective is
        `max(b)`.  We replace the hard max by a soft‑max with temperature
        τ that is slowly increased.
      * The gradient of the surrogate w.r.t. `a` is
        `grad = 2·a * w_rev` where `w` are the soft‑max weights.
        All convolutions are performed with FFTs (O(n log n)).
      * Mirror‑descent (exponentiated updates) is performed on the log‑
        variables `x = log a`.  Adam‑style moments (β₁,β₂) give an
        adaptive step size.
      * After every 20 iterations the true score (the original
        `evaluate_sequence` function) is evaluated; the best sequence
        seen is retained.
      * τ starts at 0.5 and is multiplied by 1.07 every 200 iterations
        until it reaches ~30, at which point the soft‑max collapses to a
        hard max (the true objective).
4. If progress stalls (no improvement for >2000 iterations) the
   current run is aborted and a fresh random restart is launched
   (keeping the global best).  Up to ten restarts are performed or the
   time budget expires.
5. The best sequence found before the deadline is returned as a plain
   Python list.

The whole search uses only NumPy, no external solvers, and respects the
time limit by frequent `time.time()` checks.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – required by the platform)
# ----------------------------------------------------------------------


def sqrt_distribution(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^(-1/2) truncated to length n and normalised."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law on [0,1]: a_i ∝ 1/√(x_i·(1‑x_i))."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0)
    a /= a.sum()
    return a


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def self_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Compute a*a (full self‑convolution) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    ft = np.fft.rfft(a, n=size)
    b_full = np.fft.irfft(ft * ft, n=size)
    n = a.shape[0]
    return b_full[: 2 * n - 1]


def gradient_from_weights_fft(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Given a weight vector w (length 2·n‑1) for a soft‑max over the
    convolution entries, compute the gradient ∂/∂a Σ_k w_k·(a*a)_k.
    Returned gradient has length n.
    """
    n = a.shape[0]
    L = w.size                     # = 2·n‑1
    w_rev = w[::-1]                # reversed weight vector
    ft_a = np.fft.rfft(a, n=size)
    ft_w = np.fft.rfft(w_rev, n=size)
    conv_full = np.fft.irfft(ft_a * ft_w, n=size)
    # length of conv_full is ≥ n+L‑1 = 3·n‑2
    idx_start = L - 1              # = 2·n‑2
    grad = 2.0 * conv_full[idx_start - np.arange(n)]
    return grad

    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine follows the multi‑restart Adam‑mirrored
    gradient descent described in the module docstring.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------

    try:
        base_len = len(height_sequence_1)   # type: ignore[name-defined]

    n = max(base_len, 5000)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    def initialise():

        a2 = arcsine_distribution(n)
        w = np.random.rand()

    # ------------------------------------------------------------------

        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999

        tau = 0.5
        eta = 0.25

        while time.time() < deadline:

            b = self_convolution_fft(a, size)          # length 2·n‑1
            max_b = b.max()

            # occasional true evaluation

                if cur_score < best_score:
