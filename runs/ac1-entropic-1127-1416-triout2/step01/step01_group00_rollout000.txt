    return a

"""
search_algorithm.py

Algorithm Overview
-----------------
1. **Parametric shape sweep** –  generate candidate step‑functions of the
   form  

        a_i = (1‑w)·U_i + w·S_i(beta)

   where `U` is the uniform distribution, `S(beta)` is a beta‑shaped
   density `S_i ∝ ((i+0.5)*(n‑0.5‑i))^beta` (beta < 0 puts more mass at the
   ends), and `w∈[0,1]` interpolates between them.  
   A modest grid on `beta` and `w` is explored for several lengths
   (`n = 1024, 2048, 4096`).  For each candidate we compute the
   max‑self‑convolution via an FFT (O(n log n)) and keep the best
   sequence.

2. **Local refinement** – starting from the best shape discovered we run a
   projected sub‑gradient descent on the scale‑invariant objective  

        f(a) = 2·n·max(a*a)   (sum(a) = 1)

   The sub‑gradient of the maximal convolution entry is a shifted copy of
   `a` (with a doubled diagonal term).  After each step we project back
   onto the probability simplex using the Euclidean projection of Duchi
   et al.  A diminishing step size `η_t = η0 / sqrt(t)` and occasional
   tiny random noise help escape flat regions.  The best vector seen
   during the refinement is retained.

3. **Result** – the final candidate (as a plain Python list of floats) is
   returned.  All operations respect the 1000 s time budget; intermediate
   progress is printed for diagnostics.

The code is self‑contained, uses only NumPy and the provided
`evaluate_sequence` function, and follows the competition’s interface.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via real FFT (fast for large vectors)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> tuple[float, int]:
    """Return the maximal value of a*a and its index."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) w.r.t. a.
    For each i, ∂c_k/∂a_i = a_{k‑i} if the index exists; the diagonal term
    (i = k/2) is doubled.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def compute_score(a: np.ndarray) -> float:
    """
    Scale‑invariant objective used internally:
        score = 2·n·max(a*a)   (assuming sum(a)=1)
    """
    n = len(a)
    max_val, _ = max_convolution_and_index(a)
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Parametric shape generation
# ----------------------------------------------------------------------


def beta_shape(n: int, beta: float) -> np.ndarray:
    """
    Generate a non‑negative shape proportional to
        ((i+0.5)*(n-0.5-i))**beta
    for i = 0..n-1.
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 0.5 - i)
    # prod > 0 for all i, so we can safely raise to a negative power
    shape = np.power(prod, beta)
    # Numerical safety: replace possible infinities/NaNs
    shape = np.nan_to_num(shape, nan=0.0, posinf=0.0, neginf=0.0)
    return shape


def mixed_shape(n: int, beta: float, w: float) -> np.ndarray:
    """
    Return a simplex‑normalised vector:
        a = (1‑w)·uniform + w·shape(beta)
    where `shape(beta)` itself is normalised to sum 1.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    shape = beta_shape(n, beta)
    s = shape.sum()
    if s > 0:
        shape = shape / s
    else:
        shape = uniform.copy()
    a = (1.0 - w) * uniform + w * shape
    # By construction the sum is 1, but numerical drift may occur:
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Grid search over (beta, w) for a given length
# ----------------------------------------------------------------------


def search_grid_for_length(
    n: int,
    beta_vals: np.ndarray,
    w_vals: np.ndarray,
    time_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exhaustively evaluate the (beta, w) grid for length n.
    Returns the best vector (as ndarray) and its `compute_score`.
    """
    best_a = None
    best_score = float("inf")
    # Pre‑allocate a buffer to avoid repeated allocations (optional)
    for beta in beta_vals:
        # Early exit if time is up
        if time.time() >= time_deadline - 0.5:
            break
        for w in w_vals:
            a = mixed_shape(n, beta, w)
            # Evaluate quickly via FFT
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Local refinement via projected sub‑gradient descent
# ----------------------------------------------------------------------


def refine_sequence(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    eta0: float = 0.8,
    noise_every: int = 500,
    noise_scale: float = 5e-5,
) -> tuple[np.ndarray, float]:
    """
    Starting from a0 (already on the simplex), run projected sub‑gradient
    descent to lower max(a*a).  Returns the best sequence seen and its
    corresponding score.
    """
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline - 0.2:
            break

        # find maximal convolution entry and its index
        max_val, k_max = max_convolution_and_index(a)

        # sub‑gradient at that entry
        grad = subgradient_of_max_at_index(a, k_max)

        # diminishing step size
        eta = eta0 / math.sqrt(t)

        a_new = a - eta * grad

        # occasional isotropic noise
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        # project back onto simplex
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional progress report
        if t % 5000 == 0:
            print(f"    refinement iter={t:,d}  cur_score={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low
    `evaluate_sequence` value.  The routine combines a parametric shape
    sweep with a short projected sub‑gradient refinement and respects the
    given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # safety margin

    # ------------------------------------------------------------------
    # Global best holder
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback: a previously known construction, if present.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # may raise NameError
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Parameters for the shape grid
    # ------------------------------------------------------------------
    beta_vals = np.linspace(-0.80, -0.20, 31)   # step ≈ 0.02
    w_vals = np.linspace(0.0, 1.0, 21)        # step = 0.05

    # Lengths to explore – powers of two are FFT‑friendly.
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            print("Time almost exhausted – stopping length loop.")
            break

        print(f"\n--- Grid search for length n = {n} ---")
        cand_a, cand_score = search_grid_for_length(
            n=n,
            beta_vals=beta_vals,
            w_vals=w_vals,
            time_deadline=deadline,
        )

        if cand_a is None:
            continue

        # Convert the score from our scale‑invariant form to the official one
        # (sum is already 1, so they are identical up to the factor 2·n).
        official_score = evaluate_sequence(cand_a.tolist())
        print(f"  Grid best for n={n}: score={official_score:.6f}")

        if official_score < best_val:
            best_val = official_score
            best_seq = cand_a.tolist()
            print(f"  *** New global best after grid: {best_val:.6f} (n={n}) ***")

        # ------------------------------------------------------------------
        # Local refinement (only on the current best of this length)
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            print("  Starting local refinement...")
            refined_a, refined_score = refine_sequence(
                a0=cand_a,
                time_deadline=deadline,
                max_iter=150_000,
                eta0=0.7,
                noise_every=400,
                noise_scale=3e-5,
            )
            refined_official = evaluate_sequence(refined_a.tolist())
            print(f"  Refinement result: score={refined_official:.6f}")

            if refined_official < best_val:
                best_val = refined_official
                best_seq = refined_a.tolist()
                print(f"  *** New global best after refinement: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        print("No candidate found – returning a simple uniform sequence.")
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The framework expects a plain Python list of floats.
    return best_seq
