def project_onto_simplex(v, S):
    if np.sum(v) == S and np.all(v >= 0):
        return v
    # shift to ensure sum S.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

```
def project_onto_simplex(v, S):
    """Project vector v onto simplex {x >= 0, sum(x)=S}."""
    if S < 0:
        raise ValueError
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all zero, return uniform?
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **Initial families** – generate a few candidates from simple families
   (power‑law, exponential, the previously found best sequence).  All
   candidates are normalised so that `sum(a)=sqrt(2*n)`, which makes the
   evaluation equal to `max(conv(a,a))`.

2. **Smooth‑max gradient descent** – minimise a differentiable surrogate
   of the `max` using a soft‑max with a temperature that is slowly
   increased (beta from 2 to 30).  Gradients are computed by FFT‑based
   convolutions, the step is projected onto the simplex (non‑negative,
   fixed sum) and the learning‑rate is adapted on‑the‑fly.

3. **Fine‑grained random refinement** – after the gradient phase a cheap
   random “move mass between two positions” local search is applied for
   a few seconds.

The whole search respects a 1000 s wall‑time budget.  The best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, sum x_i = S}.
    """
    if S < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_optimize(a0: np.ndarray,
                         S: float,
                         deadline: float,
                         max_iters: int = 600,
                         beta_start: float = 2.0,
                         beta_end: float = 30.0,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using projected
    gradient descent.  Returns the best vector found and the corresponding
    *max* convolution value.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    # pre‑compute FFT lengths (powers of two)
    size_conv = _next_pow2(2 * n - 1)          # for a * a
    size_grad = _next_pow2(3 * n - 2)          # for weight * a_rev

    # the evaluation for the normalised sum is simply the max of the
    # self‑convolution (because S**2 = 2*n)
    best_val = np.max(np.fft.irfft(
        np.fft.rfft(a, n=size_conv) ** 2, n=size_conv)[:2 * n - 1])

    eta = eta0
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # ----- temperature schedule -------
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)

        # ----- convolution a*a -------------
        a_fft = np.fft.rfft(a, n=size_conv)
        conv = np.fft.irfft(a_fft * a_fft, n=size_conv)[:2 * n - 1]

        # ----- soft‑max weights ------------
        # w_i ∝ exp(beta * conv_i)  (numerically stable)
        z = beta * conv
        zmax = np.max(z)
        w = np.exp(z - zmax)
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (w * a_rev) ---------
        w_fft = np.fft.rfft(w, n=size_grad)
        rev_fft = np.fft.rfft(a[::-1], n=size_grad)
        grad_full = np.fft.ifft(w_fft * rev_fft).real  # length 3n‑2
        # the part we need is indices n‑1 … n‑1+n‑1
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # ----- projected step ---------------
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        step = eta * S / norm
        a_new = a - step * grad
        a_new = _project_onto_simplex(a_new, S)

        # ----- evaluate new candidate -------
        a_fft_new = np.fft.rfft(a_new, n=size_conv)
        conv_new = np.fft.irfft(a_fft_new * a_fft_new, n=size_conv)[:2 * n - 1]
        max_new = np.max(conv_new)

        # ----- accept if improvement ----------
        if max_new < best_val - 1e-12:
            best_val = max_new
            best_a = a_new.copy()
            # slightly increase step size after a success
            eta *= 1.02
        else:
            # otherwise shrink the step
            eta *= 0.90

        a = a_new

        # early exit if we already beat the target bound
        if best_val < 1.503:          # because evaluation == max_val for the normalisation
            break

    return best_a, best_val


def _random_refinement(a: np.ndarray,
                       S: float,
                       best_val: float,
                       deadline: float,
                       max_trials: int = 5000) -> tuple[np.ndarray, float]:
    """
    Very cheap random local moves: move a small fraction of the mass from
    one index to another, keep the change if the max‑convolution improves.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_val = best_val
    # pre‑compute FFT length for fast evaluation inside the loop
    size_conv = _next_pow2(2 * n - 1)

    for _ in range(max_trials):
        if time.time() > deadline:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_move = 0.01 * cur[i]          # move at most 1 % of the source mass
        if max_move <= 0.0:
            continue
        delta = np.random.uniform(-max_move, max_move)
        if cur[i] - delta < 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluation (fast via FFT, then max)
        cand_fft = np.fft.rfft(cand, n=size_conv)
        conv = np.fft.irfft(cand_fft * cand_fft, n=size_conv)[:2 * n - 1]
        cand_val = np.max(conv)

        if cand_val < cur_val - 1e-12:
            cur = cand
            cur_val = cand_val

    return cur, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects a wall‑time of ``budget_s``
    seconds and returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1.  Choose a handful of lengths – larger n gives a lower theoretical
    #     bound but also more work per iteration.
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000]   # adjust if time runs out

    for n in lengths:
        if time.time() > deadline:
            break
        target_sum = np.sqrt(2.0 * n)   # normalisation used by the original code

        # ------------------------------------------------------------------
        # 2.  Create a few different initialisations for this length.
        # ------------------------------------------------------------------
        init_vectors = []

        # (a)  power‑law family – the classic 1/(i+α)^p
        for _ in range(3):

            offset = np.random.uniform(0.0, 4.0)
            raw = 1.0 / np.power(np.arange(n, dtype=float) + 1.0 + offset, p)
            a = raw * (target_sum / raw.sum())
            a = _project_onto_simplex(a, target_sum)
            init_vectors.append(a)

        # (b)  exponential decay – a_i ∝ exp(-λ i)
        for _ in range(2):

            raw = np.exp(-lam * np.arange(n, dtype=float))
            a = raw * (target_sum / raw.sum())
            a = _project_onto_simplex(a, target_sum)
            init_vectors.append(a)

        # (c)  reuse the previously best construction if it matches the length
        if "height_sequence_1" in globals():
            hs = globals()["height_sequence_1"]
            if isinstance(hs, list) and len(hs) == n:
                a = np.array(hs, dtype=float)
                a = _project_onto_simplex(a, target_sum)
                init_vectors.append(a)

        # ------------------------------------------------------------------
        # 3.  Optimise each initial vector.
        # ------------------------------------------------------------------
        for a0 in init_vectors:

            a_opt, max_conv = _smooth_max_optimize(
                a0,
                target_sum,
                deadline,
                max_iters=600,
                beta_start=2.0,
                beta_end=30.0,
                eta0=0.5,
            )
            score = max_conv                     # because sum = √(2n) → evaluation == max_conv

            if score < best_score:
                best_score = score
                best_seq = a_opt.tolist()
                print(f"** New best (gradient)  n={n:4d}  score={score:.6f}")

                a_opt,
                target_sum,
                max_conv,
                deadline,

                best_score = score_refined
                best_seq = a_refined.tolist()
                print(f"** New best (refined)  n={n:4d}  score={score_refined:.6f}")

            # Early stopping if we already reached the target bound
            if best_score < 1.503:

    # ------------------------------------------------------------------
    # 4.  Fallback – uniform distribution (guaranteed value 2.0)
