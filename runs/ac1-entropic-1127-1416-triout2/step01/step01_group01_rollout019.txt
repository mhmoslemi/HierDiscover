"""
propose_candidate – LP‑based iterative refinement

The algorithm starts from a simple analytically‑derived linear ramp
(a_i = 2·(n‑i)/(n·(n+1))) which already beats the naïve uniform bound.
At each outer iteration we

1. compute the current self‑convolution peak M,
2. solve the linear programme

       maximise   Σ g_j
       subject to  (f * g)_k ≤ M    for all k = 0 … 2n‑2
                   g_j ≥ 0

   where f is the current step function and the convolution (f * g) is
   linear in the unknown g.  The feasible point g = f guarantees that the
   programme is never infeasible.

3. normalise the LP solution to a probability simplex (Σ g = 1) and
   update the current function by a small convex combination
   f ← (1‑t)·f + t·ĝ .  The convex combination preserves non‑negativity
   and the unit‑sum constraint.

4. keep the best vector (lowest value of the original evaluation
   function 2·n·max(a*a) ) seen so far.

A few hundred outer iterations are cheap for n≈1000 (the constraint
matrix has only 2·n‑1 rows and n columns, i.e. ~10⁶ non‑zeros).  The
routine stops when the time budget (≈1000 s) is exhausted or when a
pre‑set iteration limit is reached.  The returned sequence works
directly with the supplied evaluate_sequence().
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: objective for a vector with sum=1  (the same as evaluate_sequence
# after fixing the normalisation)
# ----------------------------------------------------------------------
def compute_objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a non‑negative vector a with sum(a)=1."""
    n = a.shape[0]
    # convolution via FFT (real version, length 2n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Helper: build the (2n‑1)×n convolution matrix for a fixed left vector f:
#   (f * g)[k] = Σ_j f_{k‑j} g_j          with the convention f_i = 0
#   when i∉[0,n‑1].
# The matrix has a very simple Toeplitz‑like structure:
#   column j is just the vector f placed starting at row j.
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    for j in range(n):
        A[j : j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list:
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best upper bound.  The method uses a linear‑programming
    improvement step followed by a short projected‑gradient polishing.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0    # safety margin

    # ------------------------------------------------------------------
    # 1. Determine problem size.
    #    Use the previously discovered construction if it exists,
    #    otherwise fall back to a modest length (≈1000) that keeps the
    #    LP tractable.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)                     # type: ignore[name-defined]
        use_existing = True
        init = np.array(height_sequence_1, dtype=float)
        # normalise to sum‑one (the optimisation works on the simplex)
        if init.sum() == 0:
            init = np.ones_like(init)                 # fallback to uniform
        init = np.maximum(init, 0.0)
        init /= init.sum()
    except Exception:
        n = 1000
        use_existing = False
        # analytically optimal decreasing linear ramp (sum = 1)
        init = np.arange(n, 0, -1, dtype=float)
        init = 2.0 * init / (n * (n + 1))

    a = init.copy()
    best_a = a.copy()
    best_score = compute_objective(a)

    # LP hyper‑parameters
    t = 0.15                # convex‑combination step size
    max_outer_iters = 30   # number of LP‑updates (adjustable)
    outer_iter = 0

    while time.time() < deadline and outer_iter < max_outer_iters:
        n = a.shape[0]

        # --------------------------------------------------------------
        # 2. Current self‑convolution peak M
        # --------------------------------------------------------------
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv_self = np.fft.irfft(fft * fft, n=2 * n - 1)
        M = conv_self.max()

        # --------------------------------------------------------------
        # 3. Build the linear programme: maximise Σ g_j  s.t.  (a * g) ≤ M
        # --------------------------------------------------------------
        A = build_conv_matrix(a)                     # (2n‑1) × n
        c = -np.ones(n)                              # linprog minimises, –Σg → maximise Σg
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12)) # slight slack for numeric safety
        bounds = [(0.0, None)] * n

        # Solve with HiGHS (fast interior‑point).  If it fails, break.
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            print(f"LP failed at outer iteration {outer_iter}: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print("LP returned zero mass – stopping.")
            break

        # --------------------------------------------------------------
        # 4. Normalise the LP solution to the simplex and update a
        # --------------------------------------------------------------
        g_norm = g / sum_g
        a = (1.0 - t) * a + t * g_norm                # convex combination (still sums to 1)

        # occasional re‑projection onto the simplex (numerical drift)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if abs(a_sum - 1.0) > 1e-12:
            a = a / a_sum

        # --------------------------------------------------------------
        # 5. Evaluate and keep the best vector seen so far
        # --------------------------------------------------------------
        cur_score = compute_objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Outer {outer_iter:2d}: improved bound = {best_score:.6f}")

        outer_iter += 1

    # ------------------------------------------------------------------
    # 6. Optional fine‑grained projected sub‑gradient polishing
    #    (a few hundred cheap steps can shave a tiny bit more off the bound)
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline and it < 500:
        n = a.shape[0]
        # current convolution and index of its maximum
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        k_max = int(conv.argmax())

        # sub‑gradient of the active convolution entry
        g_sub = np.zeros_like(a)
        low = max(0, k_max - n + 1)
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        partner = k_max - idx
        g_sub[idx] = 2.0 * a[partner]

        # diminishing step size
        step = 0.4 / np.sqrt(it + 1)

        # projected gradient step onto the simplex (Euclidean projection)
        # Simple Euclidean projection onto the probability simplex:
        v = a - step * g_sub
        # --------------------------------------------------------------
        # Projection (see Duchi et al. 2008) – small, self‑contained impl.
        # --------------------------------------------------------------
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        a = np.maximum(v - theta, 0.0)

        cur_score = compute_objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"   Sub‑grad {it:3d}: improved bound = {best_score:.6f}")
        it += 1

    print(f"Finished. Best bound found = {best_score:.6f} with length {best_a.size}")
    return best_a.tolist()
