def project_onto_simplex(v):
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    # v: numpy array
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all <=0
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
Smooth (soft‑max) gradient descent on the max‑convolution objective.

We minimise
        F(a) = max_k (a * a)[k]
subject to a_i ≥ 0 and Σ a_i = 1.
The function is invariant under scaling, therefore we keep the L¹‑norm equal
to one and work directly with the unconstrained (except for the simplex) vector.

A soft‑max approximation of the max is used:
    w_k ∝ exp( ( (a*a)[k] - max(a*a) ) / ε )
so that the gradient is an average over all near‑maximal convolution entries.
The gradient of (a*a)[k] w.r.t. a_i is 2·a_{k‑i}, thus
    ∇F_ε(a) = 2· (a_rev  ⋆  w)   (restricted to the original length)

where a_rev is a reversed copy of a and “⋆’’ denotes ordinary convolution.
After a gradient step we project back onto the probability simplex
(Σ a_i = 1, a_i ≥ 0).  The step size decays like 1/√t.
Occasional random jitter and a simple hill‑climbing move are added to escape
plateaus.  The whole routine respects the given time budget (default 1000 s)
and returns the best sequence found.

The implementation is pure NumPy, uses an FFT based convolution for speed,
and contains no closures or lambdas, as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _constant_conv_base(n: int) -> np.ndarray:
    """
    First n terms of a_k = binom(2k, k) / 4**k.
    Recurrence: a_0 = 1, a_k = a_{k‑1} * (1 - 1/(2k)).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σ x_i = 1}.
    """
    # shift to make the projection well‑defined
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Guard against numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def _convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution using a real FFT (fast for moderate sizes).
    Result length = len(a) + len(b) - 1.
    """
    size = a.shape[0] + b.shape[0] - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)[:size]
    return conv


def _softmax_gradient(a: np.ndarray, eps: float) -> tuple[np.ndarray, float]:
    """
    Compute the soft‑max weighted gradient of the max‑convolution.
    Returns (grad, current_max), where grad has the same shape as a.
    """
    conv = _convolve_fft(a, a)               # length 2n‑1
    cur_max = conv.max()

    # soft‑max weights (numerically stable)
    w_exp = np.exp((conv - cur_max) / eps)   # max entry becomes 1.0
    w = w_exp / w_exp.sum()                  # sum = 1

    # gradient = 2 * (reverse(a) * w) restricted to n entries
    a_rev = a[::-1]
    grad_full = _convolve_fft(a_rev, w)      # length 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * grad_full[start:start + a.shape[0]]
    return grad, cur_max


def _random_hill_climb(a: np.ndarray, sigma: float) -> np.ndarray:
    """
    Propose a random nearby point on the simplex.
    """
    cand = a + sigma * np.random.randn(*a.shape)
    return _proj_simplex(cand)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  A smooth (soft‑max) gradient descent is used,
    together with occasional random perturbations and a simple line‑search
    on the step size.  The best sequence found before the time budget expires
    is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Choose a length; reuse an existing construction if available.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 2000

    # --------------------------------------------------------------
    # Initialise with the classic constant‑convolution shape.
    # --------------------------------------------------------------
    base = _constant_conv_base(n)
    a = base / base.sum()                     # L¹‑norm = 1, all ≥0

    # compute initial score
    conv = _convolve_fft(a, a)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max
    best_seq = a.copy()
    print(f"Initial length {n}, initial score {best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    it = 0
    step0 = 0.5               # initial step size
    eps0 = 0.05               # base soft‑max temperature
    jitter_interval = 4000    # iterations between random jitter
    stall_limit = 5000        # reset after no improvement for this many steps
    last_improve = it
    sigma_noise = 1e-4        # magnitude of stochastic moves

    while time.time() < deadline:
        # ----- compute soft‑max gradient -----
        # Temperature adapts with iteration count.
        eps = max(eps0 / np.sqrt(it + 1.0), 1e-12)

        grad, cur_max = _softmax_gradient(a, eps)

        # ----- decide on a move (gradient step or random walk) -----
        if np.random.rand() < 0.7:          # mostly gradient descent
            # line‑search on the step size (up to three halvings)
            step = step0 / np.sqrt(it + 1.0)
            accepted = False
            for _ in range(3):
                a_candidate = _proj_simplex(a - step * grad)
                conv_cand = _convolve_fft(a_candidate, a_candidate)
                max_cand = conv_cand.max()
                if max_cand < cur_max:       # improvement
                    a = a_candidate
                    cur_max = max_cand
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                # fallback: small random perturbation
                a = _random_hill_climb(a, sigma_noise)
                conv = _convolve_fft(a, a)
                cur_max = conv.max()
        else:
            # pure stochastic hill‑climb move
            a = _random_hill_climb(a, sigma_noise)
            conv = _convolve_fft(a, a)
            cur_max = conv.max()

        # ----------------------------------------------------------
        # Keep the best ever seen
        # ----------------------------------------------------------
        score = 2.0 * n * cur_max
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            last_improve = it
            print(f"Iter {it:6d}: new best score {best_score:.6f}")

        # ----------------------------------------------------------
        # Periodic jitter to escape possible plateaus
        # ----------------------------------------------------------
        if it % jitter_interval == 0 and it > 0:
            a = _random_hill_climb(a, sigma_noise * 5)

        # ----------------------------------------------------------
        # If we have stalled for a long time, re‑initialise from the best
        # ----------------------------------------------------------
        if it - last_improve > stall_limit:
            a = best_seq + 1e-3 * np.random.randn(n)
            a = _proj_simplex(a)
            last_improve = it
            print(f"Iter {it:6d}: stall detected – re‑initialising from best.")

        it += 1

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluate_sequence
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final reported score by evaluate_sequence: {final_score:.6f}")
    except Exception:
        final_score = best_score

    print(f"Search finished after {it} iterations, best score {best_score:.6f}")
    return best_seq.tolist()
