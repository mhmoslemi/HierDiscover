"""
Propose a low‑score step‑function.

Algorithm combines several optimisation stages:
1. Initialise many candidates (uniform, Dirichlet, random sparse).
2. For each, run a Adam‑based minimisation of a smooth log‑sum‑exp
   approximation of `max(conv(a,a))`.  The smoothing parameter β is
   annealed from a small to a large value, turning the problem into
   the true max‑objective.
3. The best candidate is refined by projected sub‑gradient descent.
4. A stochastic hill‑climb with mass‑transfer moves finishes the
   polishing.

All operations respect the simplex constraint (non‑negative,
sum = 1).  The routine adapts its time‑budget to stay within the
1000 s wall‑clock limit.
"""

import time
import numpy as np

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a ⊛ b using real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact score used by `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    n = v.shape[0]
    # if all entries non‑positive, return uniform
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def adam_optimize(start_a: np.ndarray,
                  deadline: float,
                  max_iters: int = 300_000,
                  beta_start: float = 5.0,
                  beta_end: float = 150.0,
                  lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth log‑sum‑exp surrogate of max(conv).
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal smoothing parameter
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed max via log‑sum‑exp
        conv = convolution_fft(a, a)                       # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()                # probability vector

        # gradient = 2 * (a_rev ⊛ w) sliced to length n
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)               # length 3n‑2
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update (minimisation => move against gradient)
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * (grad * grad)
        m1_hat = m1 / (1 - beta1 ** it)
        m2_hat = m2 / (1 - beta2 ** it)
        step = lr0 / (np.sqrt(m2_hat) + eps)
        a = a - step * m1_hat
        a = project_to_simplex(a)

        # occasional exact evaluation
        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    # final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    `2·n·max(conv(a,a))`.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k:
        # ∂ conv_k / ∂ a_j = a_{k-j} + a_{k-j} (two contributions) = 2 * a_{k-j}
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:

"""
Propose a low‑score step‑function.

The routine mixes three optimisation phases that respect the simplex
constraint (non‑negative entries that sum to 1).

1. **Adam‑based smoothed optimisation** – minimise a log‑sum‑exp
   surrogate of `max(conv(a,a))`.  The smoothing parameter β is
   annealed from a modest to a large value; Adam’s adaptive steps
   accelerate convergence.

2. **Projected sub‑gradient refinement** – after the smooth phase a few
   thousand exact sub‑gradient steps on the true max‑objective sharpen
   the result.

3. **Stochastic hill‑climb** – mass‑transfer swaps and small perturbations
   explore the neighbourhood of the best candidate.

All phases keep track of the best exact score (the value returned by
`evaluate_sequence`) and stop well before the 1000 s wall‑clock limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 150.0,
    lr0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth log‑sum‑exp surrogate of max(conv).
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m1 = np.zeros_like(a)   # first moment
    m2 = np.zeros_like(a)   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # -------- anneal smoothing parameter β -------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # -------- smooth max via log‑sum‑exp -------------------------
        conv = convolution_fft(a, a)            # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()     # probabilities, length 2n‑1

        # -------- gradient of the surrogate -------------------------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)    # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]  # slice of length n

        # -------- Adam update (we minimise) -------------------------
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * (grad * grad)
        m1_hat = m1 / (1 - beta1 ** it)
        m2_hat = m2 / (1 - beta2 ** it)
        step = lr0 / (np.sqrt(m2_hat) + eps)

        a = a - step * m1_hat
        a = project_to_simplex(a)

        # -------- occasional exact evaluation -------------------------
        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final exact evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


def subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    `2·n·max(conv(a,a))`.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv_k w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # small jitter to help escape flat plateaus
        if t % 500 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


def hill_climb(
    seq: np.ndarray,
    deadline: float,
    max_steps: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a fraction of mass between two
    positions (or swaps two entries).  Each move is projected back onto
    the simplex.
    """
    a = seq.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12  # initial fraction of mass moved

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a, best_score
