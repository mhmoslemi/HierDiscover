        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (conv >= mx - tol).astype(float)
        cnt = mask.sum()
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)

        # multiplicative update
        # ensure subgrad finite
        # update:
        a = a * np.exp(-step * subgrad)
        # re-normalize to sum 1
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / sum_a
        # update step size schedule: step = init_step / sqrt(it)
        if it % 1000 == 0:
            step = init_step / np.sqrt(it)
    return a

"""
Propose a low‑value step sequence.

The search proceeds in four stages:

1. **Diverse seeds** – arcsine, uniform, mixtures and many random
   beta / power shapes are generated for a given length.
2. **Smooth‑max Adam** – the non‑smooth objective
   max(conv(a,a)) is approximated by a soft‑max (log‑sum‑exp) and
   minimised with Adam on the simplex using a soft‑max
   parametrisation.  The temperature β is ramped from 5 to 300.
3. **Exact mirror‑descent** – a projected sub‑gradient (exponentiated‑gradient)
   descent on the true max‑convolution is performed for many
   iterations, keeping the iterate on the probability simplex.
4. **Hierarchical up‑scaling** – the best sequence found for a
   smaller length is linearly interpolated to the next larger length,
   providing a powerful warm‑start.

All heavy work respects the wall‑clock budget (default 1000 s) and
the routine always returns a valid non‑negative sequence.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (linear convolution via real FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for the sizes we need)."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast exact evaluation of the official metric when the vector is
    normalised to sum = 1: 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple families of candidate shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on {0,…,n‑1} (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type shape  f(i) ∝ (i+eps)^{α‑1}·(n‑1‑i+eps)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape  f(i) ∝ 1/(((i+0.5)*(n‑0.5‑i)+eps)^p)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Simple convex mixture of arcsine and the uniform distribution."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def upscale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a larger grid."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up /= up.sum()               # re‑normalise (tiny numerical drift)
    return up


# ----------------------------------------------------------------------
# Soft‑max utilities (used by the Adam optimiser)
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 300.0,
    ramp_iters: int = 50000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimisation of the smooth surrogate  log‑sum‑exp_{β}(conv(a,a) – max).
    The iterate a = softmax(z) always lies on the simplex.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for the temperature β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # soft‑max weights w ∝ exp(β·(conv – mx))
        shifted = conv - mx                 # ≤ 0
        w = np.exp(beta * shifted)          # never overflows
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of the smooth max w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)    # length = n + (2n‑1) – 1 = 3n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # central slice, length n

        # chain‑rule for the soft‑max parametrisation
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    [Adam] it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final a
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) on the true max‑conv
# ----------------------------------------------------------------------


def mirror_descent_exponential(
    a0: np.ndarray,
    deadline: float,
    init_step: float = 0.15,
    max_iter: int = 300_000,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)),
    performed as an exponentiated‑gradient update (mirror descent w.r.t.
    the KL‑divergence).  Operates directly on the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    # work in the log‑domain for numerical stability
    log_a = np.log(a + 1e-12)
    step = init_step
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # recover a from log_a (softmax)
        a = softmax(log_a)

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # indices where the convolution is (almost) maximal
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)
        cnt = mask.sum()
        if cnt == 0:
            break

        # sub‑gradient = average of the 2·shift(a) over active shifts
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[n - 1 - np.arange(n)] / cnt

        # exponentiated‑gradient (mirror) step in log‑space
        log_a = log_a - step * subgrad

        # simple decay of the step‑size
        if it % 5000 == 0:
            step = init_step / np.sqrt(it)
            # optional progress report
            print(
                f"    [Mirror] it={it:,d} step={step:.6f} score={2.0*n*mx:.6f}"
            )

    a = softmax(log_a)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:

        except Exception:

    # ------------------------------------------------------------------
    # Lengths (increasing – allows cheap warm‑starts for larger n)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]

    for n in lengths:
        time_now = time.time()
        if time_now >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------

        seeds = [

        # up‑scale the best sequence from the previous (smaller) length

        # --------------------------------------------------------------
        # 2️⃣  Choose the most promising seed (fast surrogate ranking)
        # --------------------------------------------------------------
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        best_seed = scored[0][1]

        # --------------------------------------------------------------
