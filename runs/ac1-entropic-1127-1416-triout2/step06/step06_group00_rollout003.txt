    # ensure sum softmax(u) = 1; but we use softmax each iteration.
    # Initialize Adam moments
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1=0.9
    beta2=0.999
    eps_adam=1e-8
    best_a = compute_softmax(u) # compute a at start
    best_score = compute_score(best_a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute current a
        a = compute_softmax(u) # softmax to ensure simplex
        # schedule beta
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        # conv

        maxc = conv.max()

        a_rev = a[::-1]

    final_score = compute_score(final_a)
    if final_score < best_score:

        best_a = final_a.copy()

FFT‑based convolutions, and tries several random restarts (uniform,
sparse and Dirichlet‑distributed starts).  A fallback solution stored in
the global `height_sequence_1` is also considered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a ⊛ b using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex helpers (soft‑max version)
# ----------------------------------------------------------------------


def softmax(u: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a vector on the simplex."""
    m = np.max(u)
    e = np.exp(u - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimisation of the smoothed max‑objective
# ----------------------------------------------------------------------


def adam_optimize(start_a: np.ndarray,
                  time_deadline: float,
                  max_iters: int = 200_000,
                  beta_start: float = 5.0,
                  beta_end: float = 200.0,
                  lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on
        f_β(a) = (1/β)·log Σ_k exp(β·(a⊛a)_k)
    keeping a on the probability simplex via a soft‑max parametrisation.
    Returns the best vector (in exact score) seen.
    """
    n = start_a.shape[0]

    # initialise logits so that softmax(logits) = start_a
    eps = 1e-20
    u = np.log(start_a + eps)

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # keep the best exact solution found
    best_a = softmax(u)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # current probability vector
        a = softmax(u)

        # --- anneal β -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # --- smooth surrogate -----------------------------------------
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()                                # soft‑max weights

        # gradient of the smoothed objective w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # --- chain rule for the soft‑max parametrisation -------------
        # ∂f/∂u_j = a_j ( ∂f/∂a_j - Σ_i a_i ∂f/∂a_i )
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # --- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u * grad_u)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 200 == 0:
            cur_score = compute_score(softmax(u))
            if cur_score < best_score:
                best_score = cur_score
                best_a = softmax(u).copy()

    # final check
    final_score = compute_score(softmax(u))
    if final_score < best_score:
        best_score = final_score
        best_a = softmax(u).copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 300_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient (2·a shifted)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g

        # project onto the simplex (Euclidean)
        # ––––––––––––––––––––––––––––––––––––––––––––––––––––––––
        u = a_new - a_new.mean()                     # cheap approx. projection
        u = np.maximum(u, 0.0)
        if u.sum() == 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = u / u.sum()

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape tiny plateaus
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            # re‑project
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq: np.ndarray,
               time_deadline: float,
               max_steps: int = 400_000) -> tuple[np.ndarray, float]:
    """Stochastic local search that swaps or transfers a fraction of mass."""
    a = seq.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10

    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        # re‑project onto simplex (simple Euclidean projection)
        s = cand.sum()
        if s == 0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand /= s

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The algorithm follows a 3‑stage schedule (Adam smoothing → sub‑grad → hill‑climb)
    and respects the supplied wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a short safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # -------------------------------------------------------------
    # 0. optional fallback from a previously discovered sequence
    # -------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = np.array(fallback, dtype=float)
    except Exception:
        pass

    # -------------------------------------------------------------
    # Choose length (larger gives more flexibility but costs more time)
    # -------------------------------------------------------------
    # We aim for 4096 if the budget permits a decent amount of time
    if deadline - time.time() > 400:
        n = 4096
    else:
        n = 2048
    print(f"Chosen sequence length = {n}")

    # -------------------------------------------------------------
    # 1. Random restarts with Adam smoothed optimisation
    # -------------------------------------------------------------
    def random_start():
        """Return a random probability vector of size n."""
        method = np.random.choice(3)
        if method == 0:                      # uniform
            a = np.full(n, 1.0 / n)
        elif method == 1:                    # sparse equal‑mass support
            p = np.random.uniform(0.05, 0.6)
            k = max(2, int(round(p * n)))
            idx = np.random.choice(n, size=k, replace=False)
            a = np.zeros(n, dtype=float)
            a[idx] = 1.0 / k
        else:                                # Dirichlet / Gamma draw
            alpha = np.random.uniform(0.2, 5.0)
            a = np.random.gamma(alpha, 1.0, size=n)
            a /= a.sum()
        return a

    # allocate at most 60 % of the total budget to this stage
    stage1_deadline = start_time + budget_s * 0.60
    while time.time() < stage1_deadline - 30.0:   # keep ~30 s for later stages
        start_a = random_start()
        cand_a, cand_score = adam_optimize(
            start_a,
            time_deadline=stage1_deadline,
            max_iters=180_000,
            beta_start=6.0,
            beta_end=200.0,
            lr=0.05,
        )
        if cand_score < best_val:
            best_val = cand_score
            best_seq = cand_a.copy()
            print(f"[Adam] New best score = {best_val:.6f}")

    # -------------------------------------------------------------
    # 2. Sub‑gradient refinement (exact objective)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print("\nSub‑gradient refinement")
        refined_a, refined_score = subgrad_optimize(
            best_seq,
            time_deadline=deadline - 40.0,
            max_iter=250_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.copy()
            print(f"[Sub‑grad] Improved best score = {best_val:.6f}")

    # -------------------------------------------------------------
    # 3. Final hill‑climb polishing
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nHill‑climb polishing")
        polished_seq, polished_score = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=600_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq.copy()
            print(f"[Hill‑climb] Improved best score = {best_val:.6f}")

    # -------------------------------------------------------------
    # Safeguard: if nothing was found, fall back to a modest uniform vector
    # -------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = np.full(n, 1.0 / n)
        best_val = evaluate_sequence(best_seq.tolist())
        print("Fallback to small uniform vector")

    # Final official evaluation (uses the provided `evaluate_sequence`)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness
    return best_seq.tolist()
