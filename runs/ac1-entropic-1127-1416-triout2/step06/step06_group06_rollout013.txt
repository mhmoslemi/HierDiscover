    v = np.zeros_like(x)

    beta2 = 0.999

    while time.time() < deadline:
        t += 1
        # compute a from logits (stable softmax)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / np.sum(exp_x)
        # convolution
        conv = convolution_fft(a, a)
        M = np.max(conv)
        # lambda schedule: increase to lambda_max gradually
        lam = lambda_max * (1.0 - np.exp(-t / 20000.0))  # slower early

        exp_vals = np.exp(lam * (conv - M))

"""
Hybrid optimizer with Adam‑mirrored updates and simulated‑annealing hill climbing.

The objective is   score(a) = 2·n·max(conv(a,a)) / (∑a)²,
which is scale‑invariant.  We therefore work on the probability simplex
(∑a = 1, a ≥ 0) and minimise 2·n·max(conv(a,a)).

Key ideas
---------
* parameters are kept in the log‑softmax space  a_i = exp(x_i)/∑exp(x)
  – this automatically enforces positivity and the simplex, so no
  costly Euclidean projection is needed.
* a smooth upper‑bound  log‑sum‑exp_λ(conv)  is used; the temperature
  λ is increased from a small value to a large value so that the
  surrogate gradually turns into the true max‑function.
* Adam (first‑ and second‑moment adaptation) is applied to the mirror
  gradient   ∇_x  = a ⊙ (g – ⟨a,g⟩)   where g = 2·(a_rev∗w) is the
  gradient of the surrogate w.r.t. a.
* After a number of gradient runs we polish the best vector with a
  simulated‑annealing hill‑climber that moves a random fraction of
  mass between two positions; occasional uphill moves are accepted
  with a decaying temperature.
* The routine performs several random restarts (Dirichlet draws with
  a random concentration) and keeps the overall best sequence.

The whole search respects the 1000 s budget, prints occasional progress
messages and finally returns a list of floats that can be fed to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * b using a real FFT.
    Result length = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective on the simplex (fast version; denominator = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors already on the simplex.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Adam‑mirrored optimisation of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def adv_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    lambda_max: float = 200.0,
    eval_interval: int = 800,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑Adam optimisation.
    Parameters are stored as logits `x` so that a = softmax(x).
    A smoothed max (log‑sum‑exp with temperature λ) is used as surrogate.
    Returns the best vector seen and its (true) score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    # initialise logits from the starting probabilities
    x = np.log(a + 1e-14)          # tiny offset avoids log(0)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- recompute probabilities from logits (stable softmax) ----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / np.sum(exp_x)

        # ---- convolution of a with itself (full length) ----
        conv = convolution_fft(a, a)          # length 2n‑1
        M = np.max(conv)

        # ---- temperature schedule for the log‑sum‑exp surrogate ----
        # slowly raise λ from a small value to λ_max
        lam = lambda_max * (1.0 - np.exp(-it / 15000.0))

        # ---- softmax weights over all shifts (numerically stable) ----
        # w_k ∝ exp( λ·(conv_k – M) ), so that the maximal entry gets weight 1
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)       # length 2n‑1, sum = 1

        # ---- gradient of the surrogate w.r.t. a ----
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)   # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]    # central n entries

        # ---- mirror‑gradient w.r.t. logits (x) ----
        # grad_x_i = a_i * (g_i - <a,g>)
        a_dot_g = np.dot(a, g)
        grad_x = a * (g - a_dot_g)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size decays like η / sqrt(it)
        step = eta0 / (np.sqrt(it) + 1e-12) * m_hat / (np.sqrt(v_hat) + eps)

        x -= step

        # ---- occasional evaluation of the true (non‑smoothed) score ----
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] iter {it:,}   new best {best_score:.6f}")

    # final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill climber (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps: int = 600_000):
    """
    Stochastic hill‑climber with a tiny temperature schedule.
    Moves a random fraction of the mass from one index to another.
    Accepts worsening moves with a decaying probability.
    Returns the best sequence (as list) and its score.
    """
    a = np.array(seq, dtype=float)   # already sums to 1
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    # start with a moderate transfer fraction
    step_scale = 0.12

    # annealing temperature (linear decay)
    t0 = time.time()
    total = deadline - t0
    temp_start, temp_end = 0.015, 0.001

    steps = 0
    while steps < max_steps and time.time() < deadline:
        # linear temperature based on elapsed fraction
        elapsed = time.time() - t0
        progress = min(elapsed / total, 1.0)
        temp = temp_start * (1.0 - progress) + temp_end * progress

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a fraction of the mass at i to j
        delta = step_scale * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # (sum is preserved, negativity cannot appear because delta ≤ a[i])
        cur_score = compute_score(cand)

        # accept if better or with Metropolis probability
        if cur_score < best_score or np.random.rand() < np.exp(-(cur_score - best_score) / (temp + 1e-12)):
            best_score = cur_score
            best_a = cand
            # slightly increase the step size after a success
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            # shrink step on failure
            step_scale *= 0.98

        steps += 1

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine mixes random Dirichlet restarts, Adam‑mirrored
    optimisation of a smooth max‑surrogate and a final annealed hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------------
    # Determine the length of the vector (use the fallback if present)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2048

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # If the harness already provides a fallback sequence, evaluate it
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # Main optimisation loop – several random restarts with Adam‑mirror
    # --------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline - 15.0:          # leave ~15 s for polishing
        restart_counter += 1
        # Random Dirichlet start (concentration α controls sparsity)
        alpha = np.random.uniform(0.1, 5.0)
        a0 = np.random.dirichlet(np.full(n, alpha))

        # split remaining time: ~70 % for gradient phase, the rest for the
        # final hill‑climb (which will be performed only once at the end)
        remaining = deadline - time.time()
        grad_deadline = time.time() + max(6.0, remaining * 0.65)

        print(f"[{time.time():.1f}] Restart {restart_counter}, α={alpha:.2f}")

        refined_a, refined_score = adv_optimize(
            a0,
            deadline=grad_deadline,
            eta0=0.55,
            lambda_max=220.0,
            eval_interval=700,
        )

        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"   → new global best (score {best_val:.6f}) after gradient")

    # --------------------------------------------------------------
    # Final polishing hill‑climb (uses the exact objective)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print("\nFinal hill‑climb polishing")
        polished_seq, polished_val = hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=800_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback in case nothing was produced (unlikely)
    # --------------------------------------------------------------
    if best_seq is None:
        uniform = np.full(n, 1.0 / n)
        best_seq = uniform.tolist()
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq
