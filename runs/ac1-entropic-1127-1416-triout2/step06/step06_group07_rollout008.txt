"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1.  Choose a sequence length (2048 is a good compromise between
    expressive power and speed).  A parametric “U‑shaped” family
    a_i ∝ ((i+½)(n‑½‑i))^p (p<0) is evaluated – these distributions
    concentrate mass near the ends and tend to flatten the
    self‑convolution.
2.  Add a few random Dirichlet starts.  The best few candidates are
    refined by a **sub‑gradient descent** on the exact
    ‑∞‑norm of the convolution.
        * In each iteration the current convolution is computed,
          the index k* where it attains its maximum is found, and the
          sub‑gradient ∂‖a∗a‖∞/∂a_i = 2·a_{k*‑i} (zero out‑of‑range
          entries) is used.
        * An exponentiated (mirror) update a_i ← a_i·exp(‑η·g_i) keeps
          the vector non‑negative; a Euclidean projection onto the
          probability simplex restores Σa_i=1.
        * The step‑size decays as η = η₀/√t.
3.  After the sub‑gradient phase a short stochastic hill‑climber
    (swap / mass‑transfer moves) is applied to the best vector found.
4.  If everything fails a tiny uniform fallback is returned.

All work respects the 1000 s time budget.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` when Σa = 1:
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Parametric initialisations (U‑shaped families)
# ----------------------------------------------------------------------


def u_shaped_start(p: float, n: int) -> np.ndarray:
    """
    Return a normalized U‑shaped vector a_i ∝ ((i+½)(n‑½‑i))^p.
    For p = 0 the vector is uniform; p < 0 pushes mass to the ends.
    """
    i = np.arange(n, dtype=float)
    base = (i + 0.5) * (n - 0.5 - i)          # strictly positive for all i
    a = np.power(base, p, where=base > 0, out=np.ones_like(base))
    # Guard against possible overflow/underflow
    a = np.clip(a, 1e-30, 1e30)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient optimisation on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.5,
    report_every: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise ‖a∗a‖∞ using a projected exponentiated‑gradient scheme.
    Returns the best vector discovered and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline - 0.5:
            break

        # --- current convolution and its maximum ----------------------
        conv = convolution_fft(a, a)
        max_c = conv.max()
        cur_score = 2.0 * n * max_c

        if cur_score < best_val:
            best_val = cur_score
            best_a = a.copy()

        # index attaining the maximum (any one of them is fine)
        k_star = int(conv.argmax())

        # --- sub‑gradient of the max (2·a_{k*‑i}) --------------------
        j = k_star - np.arange(n)                     # potential partner indices
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # --- exponentiated (mirror) update ---------------------------
        eta = eta0 / np.sqrt(t)
        # Clamp exponent to avoid overflow/underflow
        exp_factor = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_factor
        a = project_to_simplex(a)

        if t % report_every == 0:
            print(f"    subgrad iter {t:7d} – best score {best_val:.6f}")

    # final check (in case the loop terminated early)
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climber for polishing
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline,
    max_steps: int = 400_000,
    step_scale_start: float = 0.1,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climber that swaps two entries or transfers a
    fraction of mass between two positions.  Uses the exact score.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = fast_score(best_a)

    step_scale = step_scale_start

    for _ in range(max_steps):
        if time.time() >= time_deadline - 0.1:
            break

        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 1e-12:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cur_score = fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # rewarding moves enlarge step
        else:
            step_scale *= 0.96                         # non‑rewarding moves shrink step

    return best_a.tolist(), best_score
