def random_dirichlet_vector(n, alpha=0.5):
    x = np.random.gamma(alpha, 1.0, size=n)
    x_sum = x.sum()
    if x_sum == 0:
        x[:] = 1.0/n
    else:
        x = x / x_sum
    return x

def annealing_hill_climb(a, deadline, max_steps):
    best = a.copy()
    best_score = compute_score(best)
    cur = a.copy()
    cur_score = best_score
    T0 = 0.1
    for step in range(max_steps):
        if time.time() >= deadline:
            break
        # propose move
        i, j = np.random.randint(len(a), size=2)
        delta = np.random.rand() * cur[i] * 0.5   # up to half mass
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / (T0 * (0.999**step))):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
    return best, best_score
```

"""
Improved search for a low‑score step‑function.

The optimiser combines several ideas that proved useful in the baseline,
but with more aggressive scheduling and extra exploration:

  * many random starts (uniform and Dirichlet with tiny concentration);
  * a smoother soft‑max surrogate with a fast FFT‑based gradient;
    Nesterov‑style momentum is added and the temperature ε is driven
    to zero much faster than in the reference code;
  * long sub‑gradient polishing (exact max‑convolution sub‑gradient);
  * a simulated‑annealing hill‑climber that can also move mass between
    several coordinates at once;
  * periodic symmetry averaging (a ↔ reverse(a)) and a final
    “local jitter’’ step to shake out flat regions.

All vectors are kept on the probability simplex (non‑negative, sum = 1)
so the fast `compute_score` (2·n·max(conv)) exactly matches the
official `evaluate_sequence`.  The routine respects a 1000 s budget
and returns the best list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities – FFT convolution, simplex projection, scoring
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` for vectors that sum to 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # implementation from Duchi et al., 2008
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------


def random_sparse_vector(n: int) -> np.ndarray:
    """Create a random probability vector with a random support size."""
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dirichlet_vector(n: int, alpha: float = 0.3) -> np.ndarray:
    """Dirichlet draw – small α yields a sparse distribution."""
    x = np.random.gamma(alpha, 1.0, size=n)
    total = x.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return x / total


# ----------------------------------------------------------------------
# 1. Soft‑max surrogate gradient descent (with momentum)
# ----------------------------------------------------------------------


def smooth_grad_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 10_000,
    step0: float = 0.8,
    eps_factor: float = 0.15,
) -> np.ndarray:
    """Minimise ε·log‑sum‑exp(conv/ε) with a Nesterov‑style momentum."""
    a = a0.astype(float).copy()
    v = np.zeros_like(a)                 # momentum term
    beta = 0.9                            # momentum coefficient
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature ε – shrinks geometrically, never below 1 % of maxc
        eps = maxc * max(eps_factor * (0.98 ** (t - 1)), 0.01)

        # soft‑max weights
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)

        # gradient of the smooth surrogate
        w_rev = w[::-1]
        grad_rev = convolution_fft(a, w_rev)          # length 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * grad_rev[idx_start - np.arange(n)]

        # Nesterov momentum + projection
        step = step0 / np.sqrt(t)
        v = beta * v + step * grad
        a_new = a - v
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny random jitter to help escape plateaus
        if t % 800 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 250_000,
    step0: float = 0.06,
) -> np.ndarray:
    """Classic sub‑gradient descent using the shift where the convolution is maximal."""
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the max‑shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # gentle random shake
        if t % 1500 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# 3. Simulated‑annealing hill‑climber
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a0: np.ndarray,
    deadline: float,
    max_steps: int = 300_000,
) -> np.ndarray:
    """Stochastic mass‑transfer hill‑climber with an annealing acceptance rule."""
    n = a0.size
    cur = a0.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # exponential temperature schedule
    T0 = 0.12
    decay = 0.9999

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = T0 * (decay ** step)

        # propose a move: choose a random number of participating indices (2–4)
        m = np.random.choice([2, 3, 4])
        idx = np.random.choice(n, size=m, replace=False)
        src, *dst = idx

        # move a random fraction of src’s mass to the destinations
        delta = np.random.rand() * cur[src] * 0.6
        cand = cur.copy()
        cand[src] -= delta
        share = delta / len(dst)
        for d in dst:
            cand[d] += share
        cand = project_to_simplex(cand)

        cand_sc = compute_score(cand)

        # accept better moves outright, otherwise accept with Metropolis probability
        if cand_sc < cur_sc or np.random.rand() < np.exp(-(cand_sc - cur_sc) / max(T, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc

        # occasional symmetry enforcement (helps flatten the autocorrelation)
        if step % 70_000 == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur = project_to_simplex(cur)
            cur_sc = compute_score(cur)
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

    return best


# ----------------------------------------------------------------------
# 4. Symmetry averaging (cheap final polish)
# ----------------------------------------------------------------------


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Replace a by (a+reverse(a))/2 and re‑project onto the simplex."""
    a2 = (a + a[::-1]) * 0.5
    return project_to_simplex(a2)
