"""
Propose a low‑score step‑function (non‑negative sequence).

Algorithm
---------
1. Choose a reasonable length (8192 if time permits, otherwise 4096).
2. Perform many random “sparse‑support” restarts.
   – Each restart creates a Dirichlet‑distributed probability vector on a random
     support size and projects it onto the simplex (∑a=1).
   – First stage: smooth “log‑sum‑exp” gradient descent on the soft‑max of the
     convolution a⊛a.  The smoothing parameter β is annealed from 0.1 to ≈120,
     the update is exponentiated‑gradient (multiplicative) and the vector is
     re‑normalised onto the simplex.
   – Second stage: projected sub‑gradient descent on the true max‑objective
     (2·n·max(conv)).  The sub‑gradient is obtained from the index where the
     convolution attains its maximum.  A simple 1/√t step‑size schedule is
     used.
3. Keep the best vector found over all restarts (exact evaluation using the
   provided ``evaluate_sequence``).
4. Final polishing: a stochastic hill‑climb with simulated‑annealing style
   acceptance.  Random mass‑transfer moves are proposed; they are accepted if
   they improve the exact score or with a temperature‑dependent probability.
5. Return the best sequence as a plain list of floats.

The whole search respects the 1000 s wall‑clock limit; intermediate results
(and timings) are printed for diagnostics.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b using a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x≥0, Σx=1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_exact_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the simplex (∑a = 1):
        score = 2 * n * max(a ⊛ a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

# Random initialisation
# ----------------------------------------------------------------------


def random_init_vector(n: int) -> np.ndarray:
    """
    Generate a random probability vector on a random support.
    The support size is between 2 % and 70 % of n.
    """
    p = np.random.uniform(0.02, 0.70)
    k = max(2, int(round(p * n)))
    idx = np.random.choice(n, size=k, replace=False)
    vals = np.random.rand(k) + 1e-9          # avoid exact zeros
    vals /= vals.sum()
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    return a


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp gradient descent (multiplicative update)
# ----------------------------------------------------------------------


def smooth_optimize(start_a: np.ndarray,
                    deadline: float,
                    max_iters: int = 300_000,
                    beta_start: float = 0.1,
                    beta_end: float = 120.0,
                    step0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of ``max(conv(a,a))`` using exponentiated
    gradient (multiplicative) updates.  The smoothing parameter β is
    annealed exponentially from ``beta_start`` to ``beta_end``.
    """
    a = start_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_exact_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- anneal β -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- smoothed objective ---------------------------------------
        conv = convolution_fft(a, a)           # length 2n‑1
        maxc = conv.max()
        # stabilized soft‑max weights
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()    # probabilities, sum to 1

        # ----- gradient (cross‑correlation) -----------------------------
        a_rev = a[::-1]
        # cross‑correlation of a_rev with weight vector w
        tmp = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * tmp[n - 1: n - 1 + n]      # slice of length n

        # ----- exponentiated‑gradient step -------------------------------
        step = step0 / np.sqrt(it)
        # multiplicative update, staying positive
        a_new = a * np.exp(-step * grad)
        # re‑normalise onto the simplex
        s = a_new.sum()
        if s <= 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new = a_new / s

        a = a_new

        # ----- occasional exact evaluation -------------------------------
        if it % 500 == 0:
            cur_score = compute_exact_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final exact check
    cur_score = compute_exact_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()

    return best_a, best_score

# Projected sub‑gradient descent on the true max‑objective

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 400_000,
                     step0: float = 0.09) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent on
        f(a) = 2·n·max(conv(a,a)).
    The sub‑gradient is derived from the index where the convolution attains
    its maximum.
    """
    a = start_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_exact_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k-i} whenever the index exists
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_exact_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    # final exact check
    cur_score = compute_exact_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()

    return best_a, best_score

# Simulated‑annealing hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb_sa(start_seq,
                  deadline,
                  max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a simple simulated‑annealing acceptance rule.
    At each step a small mass transfer (or a swap) is performed, the vector
    is projected back onto the simplex, and the move is accepted if it
    improves the exact objective or with probability exp(–Δ/Temp).
    """
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_exact_score(best_a)
    cur_a = a.copy()
    cur_score = best_score

    # temperature schedule (linear decay)
    temp_start = 0.3 * best_score
    temp_end = 0.001 * best_score
    temp_decay = (temp_end / temp_start) ** (1.0 / max_steps)
    Temp = temp_start

    step_scale = 0.12  # initial fraction of mass moved

    step = 0
    while step < max_steps and time.time() < deadline:
        cand = cur_a.copy()

        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a small fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        new_score = compute_exact_score(cand)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probabilistic acceptance
            if Temp > 0.0:
                prob = np.exp(-(new_score - cur_score) / Temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            cur_a = cand
            cur_score = new_score
            # keep best ever seen
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
                # slowly increase the step scale when improving
                step_scale = min(step_scale * 1.04, 0.5)

        else:
            # shrink step scale when move rejected
            step_scale *= 0.97

        # update temperature
        Temp *= temp_decay
        step += 1

    return best_a, best_score
