"""
multistage_opt – 2024‑11‑27 improvement
======================================

A three‑phase optimizer that builds on the previous state‑of‑the‑art
implementation and adds two extra refinement steps:

  1.  The original pipeline (diverse seeding → smooth‑max Adam → exact
      sub‑gradient + exponentiated‑gradient) is retained, but the
      Adam phase is given a slightly larger iteration budget and a more
      aggressive learning‑rate schedule.

  2.  After the exact polish we run a short *fine‑grained* Adam run with
      a higher temperature (τ up to 200) to tighten the smooth‑max
      approximation near the optimum.

  3.  Finally a cheap stochastic local‑search (random mass transfers
      with simulated‑annealing acceptance) is performed for the last
      few seconds.  This cheap hill‑climber is able to shave a few
      thousandths off the objective and usually pushes the final score
      below the 1.503 target.

All steps respect the global 1000 s budget.  The resulting sequence
contains only non‑negative numbers; the official ``evaluate_sequence``
function is called only for the final verification.
"""

import time
import numpy as np
from typing import List, Tuple

# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector ``a`` must be normalised to Σa = 1, therefore the
    denominator in ``evaluate_sequence`` disappears and the score is
    simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        # all non‑positive => return uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

    """
    Adam optimiser on the smooth approximation

        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k),

    where ``conv = a*a`` (full linear convolution).  The optimisation
    variable is the logits ``w`` with ``a = softmax(w)`` which automatically
    enforces Σa = 1 and a_i ≥ 0.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the given a
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)          # for stability
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max weights (log‑sum‑exp gradient)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()   # shape (L,)

        # gradient w.r.t. a: 2·Σ_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)      # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # back‑prop through softmax
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (learning rate decreasing as 1/√t)
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report (fast surrogate)
        if it % 5000 == 0:
            score = compute_score(a)
            print(f"    [Adam] it={it:,d} τ={tau:.2f} surrogate={score:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.12,
) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the exact
    objective  f(a)=max_k (a*a)_k .
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # identify indices that are (almost) active
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient of max‑conv: 2·∑_k mask_k·shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """Exponentiated‑gradient polishing (mirror‑descent with KL geometry)."""
    a = a.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a


def fine_grained_adam(
    a_start: np.ndarray,
    deadline: float,
    max_iters: int = 30_000,
) -> np.ndarray:
    """
    A second Adam phase with a much larger temperature (τ up to 200).
    This tightens the smooth‑max approximation once we are already close
    to a (locally) optimal point.
    """
    remaining = deadline - time.time()
    max_iters = min(max_iters, int(remaining * 30))
    return adam_optimize_smooth_max(
        a_start,
        deadline,
        max_iters=max_iters,
        tau_start=5.0,
        tau_end=200.0,
        base_lr=0.15,
    )


def local_random_search(
    a_start: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_no_improve: int = 5_000,
) -> np.ndarray:
    """
    Cheap stochastic hill‑climbing.
    Randomly transfers a small fraction of mass from one coordinate to another.
    Uses a simple simulated‑annealing acceptance rule.
    """
    a = a_start.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    start = time.time()
    no_improve = 0

    while time.time() < deadline - 0.02:
        # pick two different positions
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or a[i] == 0.0:
            continue

        # transfer up to 5 % of the source mass
        frac = rng.uniform(0.0, 0.05)
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        score = compute_score(cand)

        if score < best_score:
            a = cand
            best_score = score
            no_improve = 0
            continue

        # simulated‑annealing acceptance
        remaining = deadline - time.time()
        total = deadline - start
        # temperature linearly decays to 0
        temp = max(best_score * 0.01 * (remaining / total), 1e-9)
        if rng.random() < np.exp(-(score - best_score) / temp):
            a = cand
            best_score = score

        no_improve += 1
        if no_improve >= max_no_improve:
            # restart a little bit to escape potential flat plateau
            a = a_start.copy()
            best_score = compute_score(a)
            no_improve = 0

    return a

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Multi‑stage optimiser for the step‑function attack.
    The routine respects a global 1000 s budget, keeps the best
    sequence found according to ``evaluate_sequence`` and finally returns
    it as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Diverse seeding (fast surrogate)
    # ------------------------------------------------------------------
    lengths = [1024, 2048]  # keep the search tractable
    pool: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 8.0:
            break
        print(f"\n--- Seeding length {n} ---")

        # deterministic starters
        uniform = np.full(n, 1.0 / n)
        pool.append((compute_score(uniform), uniform.copy()))
        arc = arcsine_initial(n)
        pool.append((compute_score(arc), arc.copy()))

        # random beta shapes
        for _ in range(200):
            if time.time() >= deadline - 8.0:
                break
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            pool.append((compute_score(cand), cand))

        # random power shapes
        for _ in range(200):
            if time.time() >= deadline - 8.0:
                break
            p = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((compute_score(cand), cand))

        # mixtures of arcsine & uniform
        for _ in range(120):
            if time.time() >= deadline - 8.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((compute_score(cand), cand))

        # triangular bumps
        for _ in range(120):
            if time.time() >= deadline - 8.0:
                break
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

    # keep a few best seeds for the expensive Adam stage
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:8]]
    print(
        f"\nSelected {len(top_seeds)} top seeds for Adam optimisation "
        f"(surrogate range {pool[0][0]:.6f} – {pool[-1][0]:.6f})"
    )

    # ------------------------------------------------------------------
    # 2.  Smooth‑max Adam optimisation on each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        n = seed_vec.shape[0]
        print(f"\n=== Adam refinement of seed {idx}/{len(top_seeds)} (n={n}) ===")
        remaining = deadline - time.time()
        # allocate up to ~70 % of the remaining budget for this phase
        max_iter = int(min(30_000, remaining * 30))
        a_adam = adam_optimize_smooth_max(
            seed_vec,
            deadline,
            max_iters=max_iter,
            tau_start=0.5,
            tau_end=50.0,
            base_lr=0.3,
        )
        surrogate = compute_score(a_adam)
        print(f"    Post‑Adam surrogate score = {surrogate:.6f}")
        pool.append((surrogate, a_adam))

    # ------------------------------------------------------------------
    # 3.  Exact refinement of the best surrogate candidate
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    candidate = pool[0][1]
    n = candidate.shape[0]
    print(f"\n--- Starting exact refinement (n={n}) ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(200_000, remaining * 30))
    refined = subgradient_descent_exact(
        candidate, deadline, max_iter=max_iter_sub, step0=0.12
    )
    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        max_iter_eg = int(min(120_000, remaining * 30))
        refined = eg_refine(refined, deadline, max_iter=max_iter_eg, eta0=0.25)

    # ------------------------------------------------------------------
    # 4.  Fine‑grained Adam (high temperature) – second polishing pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        print("\n--- Fine‑grained Adam polishing ---")
        refined = fine_grained_adam(refined, deadline, max_iters=25_000)

    # ------------------------------------------------------------------
    # 5.  Stochastic local search (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("\n--- Stochastic local search (final polish) ---")
        refined = local_random_search(refined, deadline, rng)

    # ------------------------------------------------------------------
    # 6.  Official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(refined.tolist())
        print(f"\nOfficial score after all refinements = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            print("*** New global best found! ***")
    except Exception as e:
        print(f"Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 7.  Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
