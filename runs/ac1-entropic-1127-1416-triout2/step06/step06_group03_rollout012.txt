def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0  # 1 second margin

    # Set length
    n = 8192
    # Ensure time enough
    if deadline - time.time() < 200:
        n = 4096

    # Helper functions: next_pow2, convolution_fft, compute_score, arcsine_initial, generate_beta_shape, softmax, etc.

    # Step 1: generate candidate shapes
    shapes = []
    # arcsine
    shapes.append(arcsine_initial(n))
    # uniform
    shapes.append(np.full(n, 1.0/n))
    # beta symmetric grid
    alphas = np.linspace(0.25, 2.5, 12)  # maybe 12 values
    for a in alphas:
        shapes.append(generate_beta_shape(n, a, a, eps=1e-6))
    # random beta and power to add diversity
    for _ in range(4):
        a = rng.uniform(0.1, 3.0)
        b = rng.uniform(0.1, 3.0)
        eps = rng.uniform(0.0, 0.5)
        shapes.append(generate_beta_shape(n, a, b, eps))
    for _ in range(4):
        p = rng.uniform(0.1, 3.0)
        eps = rng.uniform(0.0, 2.0)
        shapes.append(generate_power_shape(n, p, eps))

    # evaluate surrogate scores
    scored = [(compute_score(s), s) for s in shapes]
    scored.sort(key=lambda x: x[0])
    best_seed = scored[0][1]
    best_score = scored[0][0]
    print(f"Initial best surrogate score {best_score:.6f}")

    # Step 2: Adam optimization
    # Initialize z
    eps = 1e-12
    z0 = np.log(best_seed + eps)

    # define per-iteration termination time for main optimization:
    adam_deadline = deadline - 30.0  # keep 30 seconds for polishing
    if adam_deadline <= time.time():
        adam_deadline = time.time() + 1.0

    a_opt = smooth_adam_optimize(z0, adam_deadline, n=n,
                                 beta0=2.0, beta_max=250.0,
                                 ramp_iters=60000, lr=0.07)

    # Step 3: evaluate official
    val = evaluate_sequence(a_opt.tolist())
    print(f"Official score after Adam: {val:.6f}")
    best_seq = a_opt.tolist()
    best_val = val

    # Step 4: Polishing with subgradient descent if time remains
    if time.time() < deadline - 5:
        polish_deadline = deadline
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(a_arr, polish_deadline,
                                          max_iter=200000, step0=0.1)
        try:
            val2 = evaluate_sequence(a_arr.tolist())
            print(f"Polished score: {val2:.6f}")
            if val2 < best_val:
                best_val, best_seq = val2, a_arr.tolist()
                print("Polish improved best.")
        except Exception as e:
            print("Polish eval failed:", e)

    # Guard:
    if not best_seq:
        n0 = 512
        best_seq = [1.0/n0]*n0
        best_val = evaluate_sequence(best_seq)

    print(f"Search finished. Best official score {best_val:.6f}")
    return best_seq

"""
propose_candidate
-----------------

Search for a non‑negative step sequence that gives a low
``evaluate_sequence`` value.

Strategy
~~~~~~~~
1. **Large problem size** – work with a long sequence (default 8192
   elements).  The bound decreases when the discretisation gets finer.

2. **Parametric seeding** – a small grid of symmetric Beta‑shapes
   (including the classic arcsine density) is evaluated with a fast
   surrogate ``2·n·max(conv)``.  The best shape is taken as a start
   point.

3. **Smooth‑max Adam optimisation** – the non‑smooth objective
   ``max(conv(a,a))`` is replaced by a log‑sum‑exp surrogate with a
   temperature β that is raised linearly from a modest value to a large
   one.  The sequence is represented by a soft‑max of a free vector
   ``z`` (automatically enforces positivity and Σa=1).  Adam with
   momentum adapts the steps and runs until the time budget is almost
   exhausted.

4. **Exact sub‑gradient polishing** – a projected sub‑gradient descent
   on the true objective (the max‑convolution) is performed for the
   remaining time.  This finishes on the simplex and usually nudges the
   score a little further down.

The whole routine respects the 1000 s wall‑clock limit and always
returns a valid non‑negative list (fallback to a uniform sequence if
something goes wrong).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using the real FFT."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score (the denominator is 1 because
    the vector is normalised to sum to 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple parametric families for diverse initial seeds
# ----------------------------------------------------------------------


def _arcsine_initial(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    # the epsilon avoids division by zero at the boundaries
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def _generate_beta_shape(n: int, a: float, b: float, eps: float) -> np.ndarray:
    """
    Symmetric (or asymmetric) Beta‑density on the discrete grid.
    a, b > 0, eps > 0 is a small regulariser.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (a - 1.0)
    right = (n - 1 - i + eps) ** (b - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def _generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Shape proportional to ((i+0.5)*(n-0.5-i)+eps)^{-p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Optimisation utilities
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 2.0,
    beta_max: float = 250.0,
    ramp_iters: int = 60000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Minimise the smooth log‑sum‑exp surrogate of ``max(conv(a,a))``.
    The optimisation variable is the unconstrained vector ``z`` with
    ``a = softmax(z)``.  The routine runs until ``deadline`` is reached.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                # simplex point (Σa = 1)
        conv = _convolution_fft(a, a)  # length 2n‑1
        mx = float(np.max(conv))

        # linearly increase the temperature β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # weights w ∝ exp(β·(conv - mx))  (they sum to 1)
        shifted = conv - mx
        # shifted ≤ 0, no overflow risk
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = _convolution_fft(a, w_rev)               # length 3n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]           # slice of length n

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate={mx + np.log(w_sum)/beta:.6f}")

    return _softmax(z)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.09,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv(a,a))  (sum(a) = 1 is already satisfied).
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance to identify the (approximately) active indices
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2 * (a ⋆ mask_rev) restricted to original length
        mask_rev = mask[::-1]
        rev = _convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 10_000 == 0:
            print(f"    SubGrad it={it:,d} score={_fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) gives a low
    ``evaluate_sequence`` value.  The routine follows the plan described
    in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 0.  Optional fallback to a previously discovered sequence
    # ------------------------------------------------------------------
    best_seq = []
    best_val = float("inf")
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose a problem size – longer sequences give a better bound
    # ------------------------------------------------------------------
    # Use 8192 unless we are already pressed for time.
    n = 8192
    if deadline - time.time() < 300.0:
        n = 4096
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 2.  Build a modest pool of parametric seeds and pick the best one
    # ------------------------------------------------------------------
    seeds = []

    # classical arcsine
    seeds.append(_arcsine_initial(n))

    # uniform
    seeds.append(np.full(n, 1.0 / n))

    # symmetric beta grid (including the arcsine case α=0.5)
    alphas = np.linspace(0.25, 2.5, 12)
    for a in alphas:
        seeds.append(_generate_beta_shape(n, a, a, eps=1e-6))

    # a few random beta families (asymmetric)
    for _ in range(4):
        a = rng.uniform(0.1, 3.0)
        b = rng.uniform(0.1, 3.0)
        eps = rng.uniform(0.0, 0.5)
        seeds.append(_generate_beta_shape(n, a, b, eps))

    # random power‑law shapes
    for _ in range(4):
        p = rng.uniform(0.1, 3.0)
        eps = rng.uniform(0.0, 2.0)
        seeds.append(_generate_power_shape(n, p, eps))

    # quick surrogate ranking
    scored = [( _fast_score(s), s) for s in seeds]
    scored.sort(key=lambda x: x[0])
    best_seed = scored[0][1]
    print(f"Best surrogate after seeding: {_fast_score(best_seed):.6f}")

    # ------------------------------------------------------------------
    # 3.  Main smooth‑max Adam optimisation
    # ------------------------------------------------------------------
    eps = 1e-12
    z0 = np.log(best_seed + eps)

    # reserve a few seconds for the final polishing phase
    adam_deadline = deadline - 40.0
    if adam_deadline <= time.time():
        adam_deadline = time.time() + 5.0

    a_opt = _smooth_adam_optimize(
        z0,
        adam_deadline,
        n=n,
        beta0=2.0,
        beta_max=250.0,
        ramp_iters=60000,
        lr=0.07,
    )

    # ------------------------------------------------------------------
    # 4.  Official evaluation of the Adam result
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(a_opt.tolist())
        print(f"Official score after Adam: {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()
    except Exception as e:
        print(f"Evaluation after Adam failed: {e}")

    # ------------------------------------------------------------------
    # 5.  Optional light random restarts (tiny perturbations) if time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 25.0:
        # create a perturbed version of the current best sequence
        arr = np.array(best_seq, dtype=float)
        # multiplicative Gaussian noise, clipped to non‑negative
        noise = rng.normal(scale=0.02, size=n)
        pert = arr * (1.0 + noise)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()
        z_start = np.log(pert + eps)

        # give each restart a short slice of the remaining time
        restart_deadline = min(time.time() + 12.0, deadline - 23.0)
        a_new = _smooth_adam_optimize(
            z_start,
            restart_deadline,
            n=n,
            beta0=2.0,
            beta_max=250.0,
            ramp_iters=40000,
            lr=0.07,
        )
        try:
            val_new = evaluate_sequence(a_new.tolist())
            print(f"  Restart candidate score = {val_new:.6f}")
            if val_new < best_val:
                best_val, best_seq = val_new, a_new.tolist()
                print("  *** New best found by restart! ***")
        except Exception:
            pass
        # continue as long as we have budget left

    # ------------------------------------------------------------------
    # 6.  Final exact sub‑gradient polishing (uses the true max‑conv objective)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal exact polishing …")
        a_arr = np.array(best_seq, dtype=float)
        polish_deadline = deadline
        a_arr = _subgradient_descent_exact(
            a_arr,
            polish_deadline,
            max_iter=200_000,
            step0=0.09,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Polished official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"Polish evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 7.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
