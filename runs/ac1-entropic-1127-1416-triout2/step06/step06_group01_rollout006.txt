"""
Improved step‑function optimizer
================================

This implementation builds on the three‑stage scheme described earlier.
The key new ideas are

* **Larger resolution** – we work mainly with 4096 points (≈ 4 × the
  previous best length).  The theoretical optimum is a constant
  independent of the discretisation, and a finer grid lets the optimizer
  approach that constant more closely.

* **Hierarchical warm‑start** – an arcsine shape is first optimised at
  n=1024, then up‑sampled to 4096 and continued.  The coarse optimisation
  provides a good “global” shape, while the fine phase refines the
  discretisation errors.

* **More aggressive smooth‑max Adam** – a larger temperature schedule
  (τ from 0.5 to 200) and a larger learning‑rate give a tighter
  approximation of the true max‑convolution.  The Adam loop runs for a
  very large number of iterations (as many as time permits).

* **Extensive exact polishing** – after the smooth phase a long
  Euclidean projected sub‑gradient descent followed by exponentiated‑gradient
  (mirror‑descent) is performed.  Both loops use a schedule that
  decreases the step‑size with √iteration, which is known to be effective
  for nonsmooth convex problems.

* **Time‑aware budgeting** – the algorithm continuously checks the
  remaining wall‑clock time and scales the number of iterations accordingly,
  guaranteeing termination well before the 1000 s limit.

The returned list is a non‑negative sequence whose sum is 1 (the
objective is scale‑invariant).  The code is self‑contained, uses only
NumPy and the supplied ``evaluate_sequence`` function, and respects all
competition rules.
"""

import time
import numpy as np
from typing import List, Tuple

# Low‑level primitives

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    The vector a is assumed to be on the simplex (Σa = 1);
    the surrogate is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform, normalised."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_n points.
    The result is re‑normalised to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    # positions of the original points (mid‑point of each bin)
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    interp = np.interp(x_new, x_old, seq)
    return interp / interp.sum()

# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 200_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the smooth approximation of max(conv).
    The optimisation variable is the logits w (softmax → a) which
    automatically satisfies a ≥ 0, Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probabilities
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)                 # for numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)            # length L

        # smooth max via log‑sum‑exp (weights)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)    # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress output
        if it % 25000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a

def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.06) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of entries that are (almost) maximal
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror descent with KL geometry).
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low
    ``evaluate_sequence`` value.  The routine works in three phases:

    1.  Coarse optimisation at n=1024 (arcsine + Adam).
    2.  Upsample to the target resolution (default n=4096) and continue
        Adam with a much larger temperature schedule.
    3.  Exact polishing (projected sub‑gradient + exponentiated‑gradient).

    All phases respect a global 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # 1 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1 – coarse optimisation at n=1024
    # ------------------------------------------------------------------
    n_coarse = 1024
    print(f"\n--- Phase 1: coarse optimisation (n={n_coarse}) ---")

    # deterministic start: arcsine
    a0 = arcsine_initial(n_coarse)

    # Adam on the coarse grid (moderate budget)
    remaining = deadline - time.time()
    max_iter_coarse = int(min(120_000, remaining * 30))
    a_coarse = adam_optimize_smooth_max(
        a0,
        deadline,
        max_iters=max_iter_coarse,
        tau_start=0.5,
        tau_end=50.0,
        base_lr=0.4,
    )
    surrogate_coarse = compute_score(a_coarse)
    print(f"Coarse Adam surrogate score = {surrogate_coarse:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upsample to high resolution and continue Adam
    # ------------------------------------------------------------------
    target_n = 4096
    # If we are already very close to the deadline, fall back to 2048
    if deadline - time.time() < 150.0:
        target_n = 2048
    print(f"\n--- Phase 2: upsample to n={target_n} and refine ---")
    a_high = upsample(a_coarse, target_n)

    # Continue Adam with a long schedule (τ up to 200)
    remaining = deadline - time.time()
    max_iter_high = int(min(250_000, remaining * 30))
    a_high = adam_optimize_smooth_max(
        a_high,
        deadline,
        max_iters=max_iter_high,
        tau_start=0.5,
        tau_end=200.0,
        base_lr=0.5,
    )
    surrogate_high = compute_score(a_high)
    print(f"High‑res Adam surrogate score = {surrogate_high:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact polishing (sub‑gradient + exponentiated‑gradient)
    # ------------------------------------------------------------------
    print("\n--- Phase 3: exact polishing ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(400_000, remaining * 30))
    a_refined = subgradient_descent_exact(a_high, deadline,
                                          max_iter=max_iter_sub,
                                          step0=0.07)

    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        max_iter_eg = int(min(250_000, remaining * 30))
        a_refined = eg_refine(a_refined, deadline,
                              max_iter=max_iter_eg,
                              eta0=0.25)

    # Official evaluation
    try:
        val = evaluate_sequence(a_refined.tolist())
        print(f"Official score after polishing = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_refined.tolist()
            print("*** New global best found! ***")
    except Exception as e:
        print(f"Evaluation failed after polishing: {e}")

    # ------------------------------------------------------------------
    # Optional final global polish (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Final global polish (extra time) ---")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.05)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(200_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.20)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
