"""
Hybrid optimiser for the “step‑function attack”.

The problem is to minimise

    score(a) = 2·n·max(conv(a,a)) / (sum(a))**2 ,

which is scale invariant.  By projecting all candidates onto the
probability simplex (sum = 1) the denominator disappears and we may
simply minimise

    F(a) = 2·n·max(conv(a,a)) .

The optimiser has three stages:

1️⃣ **Random restarts** – a Dirichlet draw (concentration α drawn
   uniformly in [0.1, 5]) gives a diverse start on the simplex.

2️⃣ **Worst‑shift exponentiated gradient descent** – at each iteration
   only the *worst* convolution shift (the index attaining the maximum)
   is used.  The sub‑gradient of this worst shift is computed in O(n)
   time, and a multiplicative update

        a ← a · exp(‑η·g)

   is performed, followed by Euclidean projection onto the simplex.
   The step‑size η decays as η₀/√t and a modest sharpening of the
   gradient (increase of the soft‑max temperature λ) is applied.

3️⃣ **Annealed hill‑climber** – after the gradient phase a large number
   of stochastic mass‑transfer moves are tried.  Moves that improve the
   true objective are always accepted; otherwise they are accepted
   with probability exp(‑Δ/T) where the temperature T slowly cools.
   The size of the transferred mass adapts to the recent success
   rate.

All helper functions are defined at top level (no closures) and the
search respects the 1000 s budget.  The best sequence found is
returned as a plain ``list[float]``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# FFT utilities (linear convolution via real FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT.
    Returned length is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    {x | x_i ≥ 0, ∑_i x_i = 1}.
    """
    if np.all(v <= 0):
        # uniform point if everything is non‑positive
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective (fast surrogate for a simplex point)
# ----------------------------------------------------------------------


def objective(a: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(a,a))  (the quantity we really want to minimise
    when Σa = 1).  Uses the fast FFT‑based convolution.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Gradient of the worst convolution shift
# ----------------------------------------------------------------------


def worst_shift_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of max(conv(a,a)) w.r.t. *a*.
    *conv* must be the linear convolution a*a (full length 2n‑1).
    The gradient is:
        g_i = a_{i+Δ} + a_{i‑Δ}   (where Δ = shift of the worst entry)
    with out‑of‑range terms omitted.  Returned as a dense vector.
    """
    n = a.shape[0]
    k0 = int(np.argmax(conv))                # index of the worst shift
    shift = k0 - (n - 1)                     # Δ ∈ [‑(n‑1), n‑1]

    g = np.zeros_like(a)

    if shift >= 0:
        # a_{i+Δ} part
        g[: n - shift] += a[shift:]
        # a_{i‑Δ} part
        g[shift:] += a[: n - shift]
    else:
        s = -shift
        # a_{i+Δ} part (Δ negative)
        g[s:] += a[: n - s]
        # a_{i‑Δ} part
        g[: n - s] += a[s:]

    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (worst shift)
# ----------------------------------------------------------------------


def worst_shift_exp_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.8,
    eval_interval: int = 250,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent that follows only the
    worst convolution shift.  The iteration stops when *deadline* is
    reached and returns the best vector encountered (together with its
    objective value).
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # linear convolution (FFT – cheap) and worst‑shift gradient
        conv = convolution_fft(a, a)
        g = worst_shift_gradient(a, conv)

        # step‑size schedule (decreasing)
        eta = eta0 / np.sqrt(it)

        # multiplicative update, with safe clipping before the exponent
        step = -eta * g
        step = np.clip(step, -50, 50)        # avoid overflow in exp
        a = a * np.exp(step)

        # back onto the simplex
        a = project_to_simplex(a)

        # periodic exact evaluation (full np.convolve)
        if it % eval_interval == 0:
            cur_val = objective(a)           # exact because objective() uses FFT as well – fast enough
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]  iter {it:,}  new best {best_val:.6f}")

    # final check
    final_val = objective(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Annealed hill‑climber
# ----------------------------------------------------------------------


def annealed_hill_climb(seq, deadline, max_steps: int = 1_500_000):
    """
    Stochastic hill‑climber with a simple temperature schedule.
    Moves transfer a fraction of the mass from one index to another.
    Accepted either if they improve the objective or with probability
    exp(‑Δ/T) where the temperature T slowly cools.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_val = objective(best_a)

    # temperature schedule
    T0 = 0.5
    step_scale = 0.15
    steps = 0
    while steps < max_steps and time.time() < deadline:
        # linear cooling
        T = max(T0 * (1.0 - steps / max_steps), 1e-6)

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * best_a[i]
        if delta <= 0:
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # enforce feasibility before projection
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cur_val = objective(cand)
        diff = cur_val - best_val

        if diff < 0 or np.random.rand() < np.exp(-diff / T):
            best_a = cand
            best_val = cur_val
            # successful move → slightly increase step size
            step_scale = min(step_scale * 1.03, 0.5)
        else:
            # unsuccessful → shrink step a bit
            step_scale *= 0.99

        steps += 1

    return best_a.tolist(), best_val

def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine performs many random restarts, a worst‑shift exponentiated
    gradient phase and finishes with a long annealed hill‑climber.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the harness
    deadline = start + budget_s - 1.0

    # ------------------------------------------------------------------
    # Sequence length – try to use the length of the provided fallback,
    # otherwise fall back to 2048 (a good trade‑off between quality and speed)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:
        n = 2048

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # If a fallback sequence is present, evaluate it first
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    restart = 0
    # ------------------------------------------------------------------
    # Main optimisation loop – keep trying new starts until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline - 12.0:      # reserve a bit for final polishing
        restart += 1
        # draw a Dirichlet start (α controls sparsity)
        alpha = np.random.uniform(0.1, 5.0)
        a0 = np.random.dirichlet(np.full(n, alpha))

        # time slice for this restart – 60 % for gradient, rest for hill‑climb
        remaining = deadline - time.time()
        grad_deadline = time.time() + max(6.0, remaining * 0.58)

        print(f"[{time.time():.1f}] Restart {restart}, α={alpha:.2f}")

        # --------------------------------------------------------------
        # 1️⃣ worst‑shift exponentiated gradient descent
        # --------------------------------------------------------------
        refined_a, refined_val = worst_shift_exp_optimize(
            a0,
            deadline=grad_deadline,
            eta0=0.9,
            eval_interval=400,
        )

        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_a.tolist()
            print(f"   → new global best after EG  {best_val:.6f}")

        # --------------------------------------------------------------
        # 2️⃣ annealed hill‑climb polishing (use the remaining time)
        # --------------------------------------------------------------
        if time.time() < deadline - 3.0:
            polished_a, polished_val = annealed_hill_climb(
                refined_a,
                deadline=deadline,
                max_steps=800_000,
            )
            if polished_val < best_val:
                best_val = polished_val
                best_seq = polished_a
                print(f"   → new global best after HC  {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback: if nothing succeeded, output a uniform distribution
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq
