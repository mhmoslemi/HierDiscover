"""
Propose a non‑negative step sequence with a small value of

    2·n·max(conv(a,a)) / (Σ a)² .

The search works in three phases:

* Phase 0 – a quick fallback if a known good sequence exists.
* Phase 1 – systematic sampling of a small family of symmetric shapes
  (uniform, arcsine, Beta‑family, simple “high‑ends / low‑middle” plateaus
  and random mixtures).  All candidates are projected onto the probability
  simplex and the  2 n·max‑convolution (fast_score) is used as a cheap
  surrogate.  The best few candidates are kept.
* Phase 2 – each survivor is polished with three consecutive optimisation
  stages that respect the time budget.
    – A long Adam run on a smooth log‑sum‑exp surrogate.  The temperature
      λ is annealed from 1 up to a very large value (≈ 2000) so the
      surrogate becomes almost the true max.  After each Adam step the
      vector is projected onto the simplex and forced to stay symmetric.
    – Projected sub‑gradient descent on the true max‑convolution (non‑smooth
      but guarantees a non‑increase of the official objective).
    – A cheap stochastic Gaussian walk on the simplex that occasionally
      accepts slight uphill moves to escape plateaus.

All stages are stopped as soon as the global deadline (≈ 1000 s) is reached.
If nothing better than the fallback is found a uniform 256‑vector is
returned (which is always valid).  The final sequence is returned as a
list of Python floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for any length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x≥0, Σx = 1}.  Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    # If everything is ≤0 return uniform point
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry (v[i] = v[-i-1]) while preserving the sum."""
    return (v + v[::-1]) / 2.0


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family density a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} (symmetrised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density (the shape used in the original baseline)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def plateau_shape(n: int, k: int, ratio: float) -> np.ndarray:
    """
    Simple “high‑ends / low‑middle” plateau:
    first/last k entries are multiplied by `ratio` (≥1), the rest are 1.
    The vector is finally normalised to sum to 1.
    """
    if k <= 0:
        return np.full(n, 1.0 / n)
    a = np.full(n, 1.0, dtype=float)
    a[:k] = ratio
    a[-k:] = ratio
    return a / a.sum()


# ----------------------------------------------------------------------
# Fast surrogate for the official objective
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate: the vector is assumed to lie on the simplex (Σa = 1),
    therefore the denominator of the true objective is 1 and the score
    equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value) where the surrogate approximates
    max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                       # soft‑max weights
    B = max_c + np.log(sum_exp) / lam            # smooth approximation of the max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Optimisation primitives (all keep the vector symmetric)
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 400.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    The vector is kept on the simplex and forced to stay symmetric.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # λ grows from lam_start to lam_end (sqrt schedule for smoother start)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)
        a_new = symmetrize(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress report (optional; harmless)
        if it % 100_000 == 0:
            print(f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    The vector stays symmetric.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # sub‑gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step size decays like 1/√t
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            a_new = symmetrize(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40_000 == 0:
            print(f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Cheap Gaussian walk on the simplex (with symmetry).  Occasionally
    accepts a tiny uphill move to escape plateaus.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)
        a_prop = symmetrize(a_prop)

        cur_sc = fast_score(a_prop)

        # accept if strictly better, otherwise with a tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50_000 == 0:
            print(f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

            if time.time() < sub_deadline - 2.0:
                max_iter_loc = int(min(300_000, (sub_deadline - time.time()) * 45))
                a_final, _ = random_local_search(
                    a_sub,
                    deadline=sub_deadline,
                    init_step=0.012,
                    max_iter=max_iter_loc,
                )
            else:
                a_final = a_sub

            # Official evaluation for this length
            val = official_score(a_final)
            print(f"  → official score for length {n} (survivor {idx + 1}) = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_final.tolist()
                print(f"  *** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Global polishing of the overall best sequence (if any)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the overall best sequence ===")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_adam = int(min(800_000, remaining * 45))
        a, _ = adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.09,
            lam_start=1.0,
            lam_end=2500.0,
            noise_every=1300,
            noise_scale=1.5e-5,
        )

        if time.time() < deadline - 4.0:
            max_iter_sub = int(min(250_000, (deadline - time.time()) * 35))
            a, _ = subgradient_optimize(
                a,
                deadline=deadline,
                step0=0.20,
                max_iter=max_iter_sub,
            )

        if time.time() < deadline - 2.0:
            max_iter_loc = int(min(350_000, (deadline - time.time()) * 50))
            a, _ = random_local_search(
                a,
                deadline=deadline,
                init_step=0.011,
                max_iter=max_iter_loc,
            )

        final_val = official_score(a)
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # safe uniform fallback (length 256)
        best_seq = [1.0 / 256.0] * 256
        best_val = official_score(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
