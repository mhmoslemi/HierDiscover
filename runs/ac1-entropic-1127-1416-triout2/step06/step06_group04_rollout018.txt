def project_simplex(v, sum_target=1.0):

    u = np.sort(v)[::-1]

    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)

"""
Improved "max‑convolution / Σ²" search.

Main idea
---------
* Start from the best known sequence (the global variable ``height_sequence_1``)
  or from an arcsine‑shaped vector.
* Repeatedly solve a tiny linear programme:
        maximise   Σ g_j
        subject to  f * g ≤ M   (elementwise)

* Rescale the LP solution to keep the total sum unchanged and either
  replace the current vector or mix it with the old one.
* A few cheap sub‑gradient steps are applied after each LP to polish
  the result.
* The whole routine is bounded by the 1000 s budget and always returns
  a non‑empty sequence.
"""

import time
import numpy as np
from typing import List
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Basic convolution / score utilities
# ----------------------------------------------------------------------


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of ``a``."""
    # ``np.convolve`` is fast enough for the modest lengths used here.
    return float(np.max(np.convolve(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official score (no clipping).
    Works for vectors already normalised to Σa = 1.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a) / (np.sum(a) ** 2)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – works for any target sum
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray, sum_target: float = 1.0) -> np.ndarray:
    """
    Project ``v`` onto the set {x ≥ 0, Σx = sum_target}.
    Duchi et al. (2008) – O(n log n) algorithm.
    """
    n = v.shape[0]
    # work with a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - sum_target
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generate simple starter shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic U‑shaped (arcsine) density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Build the dense convolution matrix A such that   A @ g = f * g
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1) × n matrix that maps ``g`` to ``f * g``."""
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=f.dtype)
    # each column i of A is the vector f placed starting at row i
    for i, val in enumerate(f):
        A[i : i + n, i] = val
    return A


# ----------------------------------------------------------------------
# One LP improvement step
# ----------------------------------------------------------------------


def lp_improve(
    f: np.ndarray,
    M: float,
    sum_target: float,
    time_deadline: float,
) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   f * g ≤ M,   g ≥ 0.
    Return a new vector (scaled to ``sum_target``) or ``None`` if the LP
    cannot be solved in the remaining time.
    """
    n = f.size
    # Build matrix once – it is dense but still modest (≈ 8 Mi entries for n=2048)
    A = build_conv_matrix(f)

    # Linear programme: minimise -Σ g_j  (i.e. maximise Σ g_j)
    c = -np.ones(n, dtype=float)
    b = np.full(2 * n - 1, M, dtype=float)

    # bounds for each variable: g_j ≥ 0
    bounds = [(0.0, None)] * n

    # solve – HiGHS is fast for medium‑size LPs
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": max(0.0, time_deadline - time.time())},
    )
    if not res.success:
        return None

    g = res.x
    if np.sum(g) <= 1e-12:
        return None

    # Rescale to keep the total sum unchanged (the denominator of the score)
    scaling = sum_target / np.sum(g)
    g_scaled = g * scaling
    return g_scaled


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine blends a tiny linear‑programming improvement step with
    a few sub‑gradient updates.  It always returns a valid (non‑empty) list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # tiny safety margin

    # ------------------------------------------------------------------
    # 0.  Initialise from the best known construction, otherwise arcsine
    # ------------------------------------------------------------------
    try:
        init = np.array([float(x) for x in height_sequence_1], dtype=float)  # type: ignore
        print("Loaded height_sequence_1 (length = %d)." % len(init))
    except Exception:
        # fallback to a 2048‑point arcsine shape
        n = 2048
        init = arcsine_initial(n)
        print("No external sequence – using arcsine of length %d." % n)

    # Normalise to Σ = 1 (the evaluation is scale‑invariant)
    init = init / init.sum()
    f = init.copy()

    best_seq = f.tolist()
    best_score = evaluate_sequence(best_seq)  # official evaluation
    print(f"Initial official score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1.  Repeated LP‑based improvement (as many times as time permits)
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline - 5.0:
        iteration += 1
        M = max_convolution(f)
        sum_f = float(np.sum(f))               # should be 1.0 after normalisation

        # ---- linear programme -------------------------------------------------
        g_new = lp_improve(f, M, sum_f, deadline)
        if g_new is None:
            print("LP failed or timed out – stopping LP loop.")
            break

        # ---- try a direct replacement -----------------------------------------
        cand = g_new.copy()
        cand = project_to_simplex(cand, sum_target=sum_f)   # safety projection
        cand_score = compute_score(cand)

        improve = False
        if cand_score < best_score - 1e-7:
            improve = True
            print(f"Iter {iteration:02d}: Replacement after LP improves score to {cand_score:.6f}")

        # ---- otherwise try a convex blend --------------------------------------
        if not improve:
            # blending factor (how far we move towards the LP solution)
            t = 0.6
            blended = (1.0 - t) * f + t * cand
            blended = project_to_simplex(blended, sum_target=sum_f)
            blended_score = compute_score(blended)
            if blended_score < best_score - 1e-7:
                cand = blended
                cand_score = blended_score
                improve = True
                print(f"Iter {iteration:02d}: Blended vector improves score to {cand_score:.6f}")

        # ---- optional cheap sub‑gradient polishing (few iterations) ----------
        if improve and time.time() < deadline - 2.0:
            # a very short projected sub‑gradient descent – cheap but often helpful
            sub_f = cand.copy()
            sub_best = sub_f.copy()
            sub_best_sc = cand_score
            n = sub_f.size
            for it in range(1, 15_001):
                if time.time() >= deadline - 2.0:
                    break
                conv = np.convolve(sub_f, sub_f)
                max_val = float(np.max(conv))
                idx = int(np.argmax(conv))
                # sub‑gradient of the active quadratic term
                grad = np.zeros_like(sub_f)
                j = idx - np.arange(n)
                mask = (j >= 0) & (j < n)
                grad[mask] = 2.0 * sub_f[j[mask]]
                step = 0.3 / np.sqrt(it)
                sub_f = sub_f - step * grad
                sub_f = project_to_simplex(sub_f, sum_target=sum_f)
                sc = compute_score(sub_f)
                if sc < sub_best_sc:
                    sub_best_sc = sc
                    sub_best = sub_f.copy()
                # early exit if no further improvement is likely
                if it % 5000 == 0:
                    pass
            cand = sub_best
            cand_score = sub_best_sc

        # ---- accept if we have made progress ----------------------------------
        if cand_score < best_score - 1e-8:
            f = cand
            best_seq = f.tolist()
            best_score = cand_score
            print(f"Iter {iteration:02d}: New best official‑equivalent score = {best_score:.6f}")
        else:
            print(f"Iter {iteration:02d}: No improvement – terminating LP loop.")
            break

    # ------------------------------------------------------------------
    # 2.  Final official evaluation and safeguard return
    # ------------------------------------------------------------------
    final_official = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Best official score = {final_official:.6f}")

    if not best_seq:
        # guarantee a non‑empty return
        best_seq = [1.0] * 256
        final_official = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector, score =", final_official)

    return best_seq
