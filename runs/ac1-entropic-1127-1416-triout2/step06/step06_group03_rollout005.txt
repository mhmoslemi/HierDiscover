"""
Optimiser for the step‑function bound.

The algorithm consists of three stages:

1. **Coarse parametric sweep** –  for several lengths (up to 8192)
   a handful of simple families (arcsine, uniform, symmetric Beta and
   power‑law shapes) are generated and evaluated with a cheap surrogate
   `2·n·max(conv(a,a))`.  The best few candidates of each length are kept.

2. **Smooth‑max Adam refinement** – each retained candidate is refined by
   optimiser `smooth_adam_optimize`.  The non‑negativity and the simplex
   constraint Σa=1 are enforced by a soft‑max parametrisation.  A log‑sum‑exp
   surrogate for the max is used with an annealing temperature β that grows
   from ~1 to a large value (≈300).  Adam with bias‑correction and an
   adaptive learning‑rate drives the search until the per‑candidate time budget
   expires.

3. **Projected sub‑gradient polishing** – the best sequence found so far is
   projected onto the probability simplex and improved with a classic
   sub‑gradient descent on the true objective `2·n·max(conv(a,a))`.  This step
   removes the smoothing bias and can further lower the official score.

The whole routine respects the 1000 s wall‑clock limit, always returns a
valid non‑negative list, and attempts a final global polish if time permits.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The input *a* is assumed to be normalised (Σa = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple families of candidate shapes (all are normalised)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, Σ = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta‑type shape, normalised to sum 1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape a_i ∝ ((i+0.5)(n‑0.5‑i)+eps)^{-p}, normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    step0: float = 0.10,
) -> np.ndarray:
    """
    Projected sub‑gradient descent for f(a)=2·n·max(conv(a,a)).
    Operates on the simplex; stops early when *deadline* is reached.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1          # length of the full linear convolution

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # mask of indices attaining the (near) maximum
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # Sub‑gradient = 2 * (a ⋆ mask_rev) evaluated on the original length
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)            # length = n + L - 1
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional progress report (quiet on very long runs)
        if it % 20000 == 0:
            print(f"    Sub‑grad it={it:,d}, score≈{2*n*mx:.6f}")

    return a


# ----------------------------------------------------------------------
# Soft‑max + smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 1.0,
    beta_max: float = 300.0,
    ramp_iters: int = 50000,
    lr: float = 0.03,
) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Parameters *beta0* → *beta_max* control the annealing of the smooth
    approximation; a larger final β yields a tighter approximation.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linearly increase β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # log‑sum‑exp weights (smooth max)
        shifted = conv - mx                # ≤ 0 component‑wise
        w = np.exp(beta * shifted)        # more weight on near‑max entries
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev) restricted to original length
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)           # length = n + (2n‑2) – 1 = 3n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]      # slice of length n

        # chain rule for the soft‑max map
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update with bias correction
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # light reporting
        if t % 5000 == 0:
            print(
                f"    Adam it={t:,d} β={beta:.1f} surrogate_score≈{2*n*mx:.6f}"
            )

    # final a (soft‑max ensures feasibility)
    return softmax(z)

            candidates.append((f"pow_{p:.2f}", shape))

        # a few random draws in the same families (adds diversity)
        for _ in range(8):
            a = rng.uniform(0.15, 0.95)
            b = rng.uniform(0.15, 0.95)
            eps = rng.uniform(0.0, 1e-6)
            shape = generate_beta_shape(n, a, b, eps)
            candidates.append((f"rbeta_{a:.2f}_{b:.2f}", shape))

        for _ in range(8):
            p = rng.uniform(0.30, 1.60)
            eps = rng.uniform(0.0, 1e-6)
            shape = generate_power_shape(n, p, eps)
            candidates.append((f"rpow_{p:.2f}", shape))

        # --------------------------------------------------------------
        # 1b.  Quick surrogate evaluation, keep a few best seeds
        # --------------------------------------------------------------
        scored: List[Tuple[float, str, np.ndarray]] = []
        for tag, vec in candidates:
            if time.time() >= deadline - 5.0:
                break
            sc = compute_score(vec)
            scored.append((sc, tag, vec))
        scored.sort(key=lambda x: x[0])
        top_k = 5
        top_candidates = [(tag, vec) for _, tag, vec in scored[:top_k]]
        print(f"   Top {top_k} surrogate candidates (best ≈{scored[0][0]:.6f})")

        # --------------------------------------------------------------
        # 1c.  Allocate a slice of time for each retained seed
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        per_seed_time = max(3.0, remaining / (len(top_candidates) * 2))

        for idx, (tag, seed_vec) in enumerate(top_candidates, start=1):
            if time.time() >= deadline - 5.0:
                break

            print(f"\n   Seed {idx}/{len(top_candidates)} ({tag}) – budget ≈{per_seed_time:.1f}s")
            seed_deadline = time.time() + per_seed_time

            # initialise Adam on the log‑softmax parametrisation
            z0 = np.log(seed_vec + 1e-12)

            # ----------------------------------------------------------
            # 2.  Smooth‑max Adam optimisation
            # ----------------------------------------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=1.0,
                beta_max=300.0,
                ramp_iters=50000,
                lr=0.03,
            )

            # ----------------------------------------------------------
            # 3.  Official evaluation
            # ----------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"      Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"      *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"      Evaluation failed: {e}")

            # ----------------------------------------------------------
            # 4.  Light exact polishing (if time permits)
            # ----------------------------------------------------------
            if time.time() < deadline - 5.0:
                # Use the current global best for polishing – it is already a
                # high‑quality candidate.
                base_arr = np.array(best_seq if best_seq else a_opt, dtype=float)
                base_arr = subgradient_descent_exact(
                    base_arr,
                    deadline - 2.0,
                    max_iter=60_000,
                    step0=0.10,
                )
                try:
                    val2 = evaluate_sequence(base_arr.tolist())
                    print(f"      Polished score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, base_arr.tolist()
                        print(f"      *** Updated best after polish: {best_val:.6f} ***")
                except Exception as e:
                    print(f"      Polish failed: {e}")

    # ------------------------------------------------------------------
    # 5.  Final global polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=120_000,
            step0=0.08,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print(f"   Final official score = {final_val:.6f}")
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 2048
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
