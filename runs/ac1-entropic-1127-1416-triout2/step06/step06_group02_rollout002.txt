"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence.  The algorithm combines several complementary
heuristics:

*  Random dense/sparse initial vectors (uniform, Dirichlet,
   sparse equal‑mass patterns).
*  A smooth “soft‑max” surrogate of the max‑convolution minimised with
   Adam (adaptive moments) and projection onto the probability simplex.
*  Classic sub‑gradient descent on the exact objective.
*  Simulated‑annealing hill‑climbing that accepts occasional worsening
   moves (swap / mass‑transfer / jitter) with a geometric temperature
   schedule.
*  A final symmetry averaging step.

The search runs for the allotted wall‑time (default 1000 s) and always
returns the best sequence discovered.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Basic linear‑algebra helpers
# ------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1
    (the simplex).  Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Random initialisation utilities
# ------------------------------------------------------------

def random_sparse_vector(n: int) -> np.ndarray:
    """
    Random probability vector of length n.
    A random subset (density 10‑% … 90‑%) receives equal mass.
    """
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dirichlet_vector(n: int, alpha: float) -> np.ndarray:
    """
    Dirichlet distribution with concentration parameter `alpha`
    (smaller alpha → more spiky vectors).
    """
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec.astype(float)


# ------------------------------------------------------------
# Optimisation phases
# ------------------------------------------------------------

def adam_smooth_gd(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    alpha: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eps_factor: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of the max
    convolution:

        f_ε(a) = ε·log Σ_k exp( (a⊛a)_k / ε ).

    The gradient is computed exactly with one FFT per iteration.
    After each update the vector is projected onto the simplex.
    Returns the best vector (lowest exact score) seen.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε shrinks geometrically, but never below 1 % of maxc
        eps_s = maxc * max(eps_factor * (0.98 ** (t - 1)), 0.01)

        # soft‑max weights (stable log‑sum‑exp trick)
        exp_vals = np.exp((conv - maxc) / eps_s)
        w = exp_vals / np.sum(exp_vals)   # Σ w = 1

        # gradient = 2· (w_rev ⊛ a)  restricted to the original n entries
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)         # length 2n‑1
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny random jitter to escape flat plateaus
        if t % 600 == 0:
            jitter = np.random.randn(n) * (alpha * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the true objective
        2·n·max(a⊛a) .
    The sub‑gradient is taken at the shift where the convolution attains
    its maximum.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient: g_i = 2·a_{k‑i} on the overlapping indices
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # small random shake every 2000 iterations
        if t % 2000 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


def annealing_hill_climb(
    seq,
    time_deadline: float,
    max_steps: int = 300_000,
    init_temp: float = 0.04,
    cooling: float = 0.9997,
    min_temp: float = 1e-5,
    step_scale_initial: float = 0.12,
) -> tuple[list, float]:
    """
    Simulated‑annealing hill‑climber.
    Moves: swap two coordinates, transfer a fraction of mass,
    or add a small Gaussian jitter.
    Accepts worsening moves with probability exp(-Δ/temperature).
    Returns the best sequence encountered.
    """
    a_cur = np.array(seq, dtype=float)
    n = a_cur.shape[0]

    cur_sc = compute_score(a_cur)
    best_a = a_cur.copy()
    best_sc = cur_sc

    temp = init_temp
    step_scale = step_scale_initial

    for step in range(max_steps):
        if time.time() >= time_deadline:
            break

        cand = a_cur.copy()
        move_type = np.random.rand()

        if move_type < 0.45:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        elif move_type < 0.90:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                if delta > 0:
                    cand[i] -= delta
                    cand[j] += delta

        else:
            # tiny Gaussian jitter on a random coordinate
            i = np.random.randint(n)
            delta = step_scale * np.random.randn()
            cand[i] += delta

        cand = project_to_simplex(cand)
        new_sc = compute_score(cand)
        delta_sc = new_sc - cur_sc

        # Metropolis acceptance
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(temp, 1e-12)):
            a_cur = cand
            cur_sc = new_sc
            # Successful move → become a little bolder
            step_scale = min(step_scale * 1.03, 0.5)
        else:
            # Unsuccessful → become more cautious
            step_scale *= 0.97

        # Update global best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_cur.copy()

        # Geometric cooling
        temp = max(temp * cooling, min_temp)

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Symmetrise the vector (a + reverse(a))/2 and re‑project to the simplex.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 9.0:
            start_vec = best_seq if best_seq is not None else a_sub.tolist()
            polished, pol_sc = annealing_hill_climb(
                start_vec,
                time_deadline=deadline - 1.5,
                max_steps=250_000,
            )
            if pol_sc < best_score:
                best_score = pol_sc
                best_seq = polished
                print(f"[{outer_iter}] annealing improved  score={best_score:.6f}")

            # symmetry averaging (cheap and often beneficial)
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[{outer_iter}] symmetry avg improved  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If nothing succeeded, fall back to a uniform vector of moderate size
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
