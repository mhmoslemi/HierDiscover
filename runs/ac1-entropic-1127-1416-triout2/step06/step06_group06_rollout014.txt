def deterministic_hillclimb(a, deadline, max_iter=200000):
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.1
    it = 0
    while time.time() < deadline and it < max_iter:
        g, M, conv = subgradient(best_a)
        i_max = np.argmax(g)
        i_min = np.argmin(g)
        # if g[i_max] <= g[i_min]: break (no descent)

"""
Hybrid optimizer for the step‑function attack.

The goal is to minimise
    score(a) = 2·n·max(conv(a,a)) / (sum(a))**2

Because the objective is scale‑invariant we work on the probability
simplex (sum == 1, a_i ≥ 0).  The algorithm consists of three phases
executed sequentially until the allotted 1000 s budget runs out.

1️⃣ **Smoothed exponentiated gradient descent** – a softmax
    surrogate of the max‑convolution is minimised.  The update is
    multiplicative (mirror‑descent in the simplex) and the step size
    decays like 1/√t while the softmax temperature λ is raised from a
    low value to ≈30.  This quickly lowers the peak of the convolution.

2️⃣ **Deterministic sub‑gradient hill‑climbing** – after the smooth
    stage we compute the exact sub‑gradient of the true objective
    (gradient of the max‑convolution).  Mass is transferred from the
    index with the largest gradient to the one with the smallest,
    i.e. from the “worst” to the “best’’ coordinate.  The transferred
    fraction is adapted on the fly; every accepted move reduces the
    true score.  This phase usually yields the largest remaining
    improvement.

3️⃣ **Random polishing** – a short stochastic mass‑transfer hill‑climb
    (the same routine that was used in the previous baseline) is run
    as a final finishing touch.

All helper routines are written at top level, no closures or I/O are
used.  The function `propose_candidate` is the entry point expected by
the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real FFT, length padded to next power‑of‑2)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT.
    Result length = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast evaluation of the scaled objective (sum == 1 → score = 2·n·max(conv))
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a on the simplex."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Smoothed exponentiated gradient descent (same idea as the baseline)
# ----------------------------------------------------------------------


def smooth_exp_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.55,
    lam_start: float = 0.8,
    lam_end: float = 30.0,
    eval_interval: int = 1000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a softmax surrogate of max(conv(a,a)).
    Returns the best vector seen and its (scaled) score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # convolution of a with itself
        conv = convolution_fft(a, a)          # length 2n‑1
        M = np.max(conv)

        # temperature schedule
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / 2_000_000)

        # softmax over all shifts (stable)
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)      # length 2n‑1

        # gradient of the smooth surrogate
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)   # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # step size
        eta = eta0 / np.sqrt(it)

        # multiplicative update (mirror descent)
        step = -eta * g
        step = np.clip(step, -50, 50)        # avoid overflow
        a = a * np.exp(step)

        # reuse feasibility
        a = project_to_simplex(a)

        # occasional exact evaluation
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] smooth_opt iter {it:,}  new best {best_score:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                break

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic sub‑gradient hill‑climbing
# ----------------------------------------------------------------------


def deterministic_hillclimb(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Guided hill‑climb that always moves mass from the index with the
    largest sub‑gradient to the one with the smallest.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    # starting step‑size (fraction of the source mass)
    step_scale = 0.2
    eps = 1e-14
    it = 0

    while it < max_iter and time.time() < deadline:
        # compute sub‑gradient of max(conv)
        conv = convolution_fft(best_a, best_a)
        M = np.max(conv)
        mask = (conv == M).astype(float)
        if mask.sum() == 0:
            break
        w = mask / mask.sum()                       # normalized weights
        rev = best_a[::-1]
        g_full = convolution_fft(rev, w)            # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]         # central part

        # choose source (high gradient, positive mass) and sink (low gradient)
        src_candidates = np.argsort(-g)             # descending
        dst_candidates = np.argsort(g)              # ascending

        i_src = None
        for idx in src_candidates:
            if best_a[idx] > eps:
                i_src = idx
                break
        if i_src is None:
            break                               # nothing to move

        i_dst = dst_candidates[0]                # smallest gradient

        # amount to transfer
        delta = step_scale * best_a[i_src]
        if delta < eps:
            break

        cand = best_a.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta

        # re‑project to avoid tiny negativities caused by rounding
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # reward success
            # early success check
            if best_score < 1.503:
                break
        else:
            step_scale *= 0.94                        # punish failure

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Random mass‑transfer hill‑climber (kept for final polishing)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps: int = 500_000):
    """
    Stochastic hill‑climber that moves a fraction of mass from a random
    source index to a random destination index, accepting only improving
    moves (step‑size adapts automatically).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.15
    steps = 0
    while steps < max_steps and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        cand = best_a.copy()
        delta = step_scale * best_a[i]
        if delta <= 0:
            continue
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.98

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The search consists of (i) a smoothed exponential‑gradient phase,
    (ii) a deterministic sub‑gradient hill‑climb, and finally
    (iii) a short random polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Choose vector length (use fallback length if available)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 2048

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # If a fallback is provided, start from it
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        fallback = None

    # ------------------------------------------------------------------
    # Main optimisation loop – we perform a few independent restarts.
    # Each restart:
    #   1) random Dirichlet start,
    #   2) smoothed exponential‑gradient descent,
    #   3) deterministic sub‑gradient hill‑climb.
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 15.0:      # reserve ~15 s for final polishing
        restart += 1
        # Initialise from fallback for the first restart,
        # otherwise sample a fresh Dirichlet point.
        if restart == 1 and fallback is not None:
            a0 = np.array(fallback, dtype=float)
            a0 = a0 / np.sum(a0)           # ensure exact simplex
        else:
            alpha = np.random.uniform(0.1, 5.0)
            a0 = np.random.dirichlet(np.full(n, alpha))

        # ---- 1) smoothed exponential‑gradient descent -----------------
        sub_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.5)
        a_opt, score_opt = smooth_exp_optimize(
            a0,
            deadline=sub_deadline,
            eta0=0.6,
            lam_start=0.7,
            lam_end=32.0,
            eval_interval=800,
        )
        cur_score = compute_score(a_opt)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] Restart {restart} → new best after SGD {best_val:.6f}")

        # ---- 2) deterministic sub‑gradient hill‑climb -----------------
        # give the hill‑climb the remaining time of this restart
        hill_deadline = time.time() + max(6.0, (deadline - time.time()) * 0.3)
        a_hc, score_hc = deterministic_hillclimb(a_opt, hill_deadline, max_iter=300_000)
        if score_hc < best_val:
            best_val = score_hc
            best_seq = a_hc.tolist()
            print(f"[{time.time():.1f}] Restart {restart} → new best after deterministic HC {best_val:.6f}")

        # early exit if target already reached
        if best_val < 1.503:
            print("Target threshold reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # Final random polishing (uses the exact evaluate_sequence)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print("\nFinal random polishing")
        polished_seq, polished_val = hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=500_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Polish improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq
