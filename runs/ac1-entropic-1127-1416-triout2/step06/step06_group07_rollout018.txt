def subgradient_max(conv, a):
    k = int(np.argmax(conv))
    n = len(a)
    g = np.zeros(n)
    # left part
    g[:k+1] += a[:k+1][::-1]
    # right part
    g[k:] += a[:n - k]
    return g

assistantfinal**Solution Overview**

The score  

\[
\text{score}(a)=\frac{2n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

is invariant under scaling, therefore we may (and will) keep the
coefficients on the probability simplex  
\(\sum_i a_i = 1\).  
For a given vector `a` the only expensive operation is the (linear)
convolution `a*a`.  
The maximum of the convolution is a piece‑wise linear convex function,
its sub‑gradient at the active index `k* = argmax(a*a)` can be written
in closed form:

```
g_i = a_{k*‑i}          (i ≤ k*)
g_i = a_{i‑k*}          (i ≥ k*)
```

(terms out of range are taken as 0).  
Moving `a` opposite to this sub‑gradient and projecting back onto the
simplex reduces the maximal entry of the convolution.

The algorithm consists of three stages

1. **random restarts** – Dirichlet draws give diverse starting points,
2. **sub‑gradient descent** – many cheap iterations that directly
   minimise the ∞‑norm of the self‑convolution,
3. **guided hill‑climbing** – a short deterministic local search that
   swaps mass from indices with large gradient to indices with a small
   one.

All work respects the 1000 s time limit; the best vector found is
returned.

```python
"""Sub‑gradient optimiser for the max‑convolution score.

The programme searches for a non‑negative step function that minimises

    score = 2 * n * max( a * a ) / (sum(a) ** 2)

where * denotes the linear convolution.  The optimisation proceeds in
three phases: random Dirichlet restarts, a large number of sub‑gradient
steps (the true max‑norm is used, no soft‑max approximation), and a
short guided hill‑climber that moves mass from the most “responsible”
indices to the least responsible ones.  All vectors are projected onto
the probability simplex after every update, guaranteeing non‑negativity
and a unit sum (the scale does not influence the final score).

The function `propose_candidate` is the entry point required by the
evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # all non‑positive → uniform vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact (fast) score – works when Σa = 1
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` for a vector on the simplex.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient optimisation on the true max‑norm
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that directly minimises the
    max‑convolution value.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # initial score
    conv = convolution_fft(a, a)
    best_a = a.copy()
    best_val = 2.0 * n * float(np.max(conv))

    for t in range(1, max_iters + 1):
        if time.time() >= deadline - 0.5:
            break

        # --- current max index and gradient -------------------------
        k = int(np.argmax(conv))          # index where the max occurs
        left_len = k + 1                  # elements that contribute from the left side
        right_len = n - k                 # elements that contribute from the right side

        g = np.zeros_like(a)
        # left contribution (mirrored)
        g[:left_len] += a[:left_len][::-1]
        # right contribution
        g[k:] += a[:right_len]

        # --- step (sub‑gradient descent) ---------------------------
        eta = eta0 / np.sqrt(t)          # diminishing step size
        a = a - eta * g
        a = project_to_simplex(a)

        # --- evaluate new point ------------------------------------
        conv = convolution_fft(a, a)
        cur_val = 2.0 * n * float(np.max(conv))

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional tiny random perturbation to escape plateaus
        if t % 10_000 == 0:
            a += np.random.normal(scale=0.001, size=n)
            a = project_to_simplex(a)

    # final check (in case the loop stopped before the last update)
    final_val = 2.0 * n * float(np.max(convolution_fft(best_a, best_a)))
    if final_val < best_val:
        best_val = final_val
        best_a = best_a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Focused hill‑climber (gradient‑guided mass transfers)
# ----------------------------------------------------------------------


def focused_hill_climb(
    seq,
    deadline: float,

    transfer_frac: float = 0.1,
) -> tuple[list[float], float]:
    """
    Deterministic hill‑climber that moves a fraction of mass from the
    index with the largest sub‑gradient to the one with the smallest.
    Uses the exact score for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_val = fast_score(best_a)

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        # current max index
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient for the active index
        left_len = k + 1
        right_len = n - k
        g = np.zeros_like(a)
        g[:left_len] += a[:left_len][::-1]
        g[k:] += a[:right_len]

        i = int(np.argmax(g))
        j = int(np.argmin(g))

        cand[i] -= delta
```
