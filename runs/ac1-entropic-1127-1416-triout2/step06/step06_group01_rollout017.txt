Propose a low‑dimensional step sequence that minimises the official
evaluate_sequence() objective.

Algorithm outline
-----------------
1. **Diverse seeding** – generate a few thousand candidate vectors of
   several parametric families (uniform, arcsine, beta, power, mixtures)
   for a handful of lengths (512, 1024, 2048).  Every candidate is
   normalised to Σa = 1 and scored with a *fast* surrogate:
        S(a) = 2·n·max(conv(a,a))          (n = len(a))
   The best few seeds are kept.

2. **Smooth‑max Adam refinement** – minimise a smooth approximation of the
   max‑convolution,
        Φτ(a) = (1/τ)·log Σk exp(τ·convk(a)),
   using Adam on the logits w (softmax(w) = a).  The temperature τ is
   increased from 0.5 to a large value (≈150) so that the surrogate
   converges to the true max.

3. **Exact sub‑gradient descent + exponentiated‑gradient polishing** –
   improve the result with a projected sub‑gradient method on the exact
   max‑convolution and a KL‑mirror‑descent (EG) finish.

4. **Local stochastic hill‑climbing** – starting from the best vector,
   repeatedly perturb a small random subset of entries, renormalise and
   keep the change if the surrogate improves (simulated‑annealing style).

5. **Final polish** – a last run of sub‑gradient + EG if time permits,
   followed by the official evaluate_sequence() call.

All stages respect the global 1000 s budget, constantly checking the
deadline.  The routine always returns a non‑empty list of non‑negative
floats.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """β‑distribution‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Simple surrogate (fast version of evaluate_sequence)
# ----------------------------------------------------------------------
def surrogate_score(a: np.ndarray) -> float:
    """Fast 2·n·max(conv) surrogate; a is assumed non‑negative."""
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    a_norm = a / s
    n = a_norm.shape[0]
    conv = convolution_fft(a_norm, a_norm)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Project onto the probability simplex (Σa = 1, a_i ≥ 0)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1} (Duchi et al., 2008)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------
def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """Euclidean projected sub‑gradient descent on max(conv(a,a))."""
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·∑ mask·shift(a)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a -= eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.3) -> np.ndarray:
    """
    Exponentiated‑gradient polishing (mirror‑descent w.r.t. KL).
    """
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------
def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 30_000,
                             tau_start: float = 0.5,
                             tau_end: float = 50.0,
                             base_lr: float = 0.3) -> np.ndarray:
    """
    Minimise the smooth log‑sum‑exp surrogate of max(conv) using Adam
    on the softmax logits.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # logits → a via softmax (guarantees Σa = 1, a ≥ 0)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # log‑sum‑exp weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()   # shape (2n‑1,)

        # gradient w.r.t. a (2·∑_k w_k·shift(a,k))
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine mixes diverse random seeding, Adam on a smooth max‑conv
    surrogate, exact sub‑gradient + EG polishing, and a final stochastic
    hill‑climbing phase.  All stages respect the global 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # a small safety margin

    best_seq: np.ndarray = None
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = np.array([float(x) for x in globals()["height_sequence_1"]],
                                dtype=float)
            val = evaluate_sequence(fallback.tolist())
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Build a diverse candidate pool (fast surrogate)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048]          # keep the runtime affordable
    pool: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Seeding length {n} ---")

        # deterministic starters
        uniform = np.full(n, 1.0 / n)
        pool.append((surrogate_score(uniform), uniform.copy()))
        arc = arcsine_initial(n)
        pool.append((surrogate_score(arc), arc.copy()))

        # random beta family
        for _ in range(200):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            pool.append((surrogate_score(cand), cand))

        # random power‑shape
        for _ in range(200):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.2, 2.5)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((surrogate_score(cand), cand))

        # mixtures of arcsine & uniform
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((surrogate_score(cand), cand))

    # keep the very best few seeds for the expensive Adam stage
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:8]]
    print(f"\nSelected {len(top_seeds)} top seeds for Adam optimisation "
          f"(surrogate range {pool[0][0]:.6f} – {pool[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2.  Smooth‑max Adam optimisation on each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 4.0:
            break
        n = seed_vec.shape[0]
        print(f"\n=== Adam refinement of seed {idx}/{len(top_seeds)} (n={n}) ===")
        a0 = seed_vec / seed_vec.sum()

        # allocate most remaining time to Adam (≈60 %)
        remaining = deadline - time.time()
        max_iter = int(min(100_000, remaining * 30))
        a_adam = adam_optimize_smooth_max(a0,
                                          deadline,
                                          max_iters=max_iter,
                                          tau_start=0.5,
                                          tau_end=150.0,
                                          base_lr=0.4)

        # quick surrogate score
        surrogate = surrogate_score(a_adam)
        print(f"    Post‑Adam surrogate = {surrogate:.6f}")

        # store candidate for later exact polishing
        pool.append((surrogate, a_adam))

    # ------------------------------------------------------------------
    # 3.  Exact refinement of the best surrogate candidate
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    candidate = pool[0][1]               # best surrogate found so far
    n = candidate.shape[0]
    print(f"\n--- Starting exact refinement (n={n}) ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(300_000, remaining * 30))
    refined = subgradient_descent_exact(candidate,
                                        deadline,
                                        max_iter=max_iter_sub,
                                        step0=0.09)

    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        max_iter_eg = int(min(200_000, remaining * 30))
        refined = eg_refine(refined, deadline,
                            max_iter=max_iter_eg,
                            eta0=0.25)

    # fast surrogate after polishing
    proxy = surrogate_score(refined)
    print(f"    Refined surrogate = {proxy:.6f}")
    if proxy < best_val:
        best_val = proxy
        best_seq = refined.copy()
        print("*** New global best (surrogate) ***")

    # ------------------------------------------------------------------
    # 4.  Stochastic local improvement (hill‑climbing)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        n = a.shape[0]
        print(f"\n--- Local stochastic improvement (n={n}) ---")
        while time.time() < deadline - 2.0:
            # modify a small random subset (≈0.5 % of entries)
            k = max(1, n // 200)
            idx = rng.choice(n, size=k, replace=False)
            # Gaussian perturbation, magnitude relative to typical coefficient
            scale = max(1e-4, a.mean())
            delta = rng.normal(0.0, scale * 0.05, size=k)
            a_new = a.copy()
            a_new[idx] += delta
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()
            new_score = surrogate_score(a_new)
            if new_score < best_val:
                best_val = new_score
                best_seq = a_new.copy()
                a = a_new
                print(f"    Improved surrogate → {new_score:.6f}")
            else:
                # occasional uphill move (simulated annealing)
                temp = max(1e-6, (deadline - time.time()) / budget_s)
                if rng.random() < np.exp(-(new_score - best_val) / temp):
                    a = a_new

    # ------------------------------------------------------------------
    # 5.  Final polish (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n--- Final polish ---")
        a = best_seq.copy()
        remaining = deadline - time.time()
        max_iter_sub = int(min(200_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.07)
        if time.time() < deadline - 1.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(150_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.22)
        # evaluate again with the fast surrogate
        final_surrogate = surrogate_score(a)
        print(f"    Final surrogate = {final_surrogate:.6f}")
        # keep if better
        if final_surrogate < best_val:
            best_val = final_surrogate
            best_seq = a.copy()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 6.  Official evaluation & return
    # ------------------------------------------------------------------
    if best_seq is None:
        # very simple fallback – uniform vector of length 1024
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    official_val = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Official final score = {official_val:.6f}")
    return best_seq.tolist()
