def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    '''
    Search for a probability vector that minimizes max self-convolution.
    ...
    '''
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5 # safety

    best_seq = None
    best_score = float('inf')

    lengths = [1024, 2048, 4096]
    for n in lengths:
        if time.time() >= deadline - 5:
            break
        print(f"=== Length {n} ===")
        # quick sampling
        # allocate small budget for sampling
        samp_budget = min(8.0, (deadline - time.time()) * 0.1) # 10% or 8 secs
        cand, cand_score = quick_sampling(n, samp_budget, rng)
        if cand is None:
            continue
        # Full optimization
        remaining = deadline - time.time()
        if remaining <= 0: break
        # flatten stage
        a_flat, _ = adam_flatten(cand, deadline, max_iter=500000, lr0=0.45)
        a_flat = symmetrize(a_flat)
        # smooth max stage
        if time.time() < deadline - 2:
            a_max, _ = adam_max(a_flat, deadline, max_iter=800000, lr0=0.08,
                                lam_start=1.0, lam_end=2000.0, noise_every=1200,
                                noise_scale=5e-5)
            a_max = symmetrize(a_max)
        else:
            a_max = a_flat
        # subgradient
        if time.time() < deadline - 2:
            a_sub, _ = subgradient_optimize(a_max, deadline, step0=0.2, max_iter=300000)
            a_sub = symmetrize(a_sub)
        else:
            a_sub = a_max
        # random walk
        if time.time() < deadline - 1:
            a_final, _ = random_local_search(a_sub, deadline, init_step=0.02, max_iter=400000)
            a_final = symmetrize(a_final)
        else:
            a_final = a_sub
        # evaluate official
        val = evaluate_sequence(a_final.tolist())
        print(f"Length {n} official score: {val:.6f}")
        if val < best_score:
            best_score = val
            best_seq = a_final.tolist()
            print(f"*** New global best: {best_score:.6f} ***")

    # Global polishing if time left
    if best_seq is not None and time.time() < deadline - 5:
        print("\n=== Global polishing ===")
        a = np.array(best_seq, dtype=float)
        # flatten
        a, _ = adam_flatten(a, deadline, max_iter=1000000, lr0=0.5)
        a = symmetrize(a)

                        noise_scale=2e-5)
        a = symmetrize(a)
        # subgradient
        a, _ = subgradient_optimize(a, deadline, step0=0.15, max_iter=500000)
        a = symmetrize(a)
        # random walk
        a, _ = random_local_search(a, deadline, init_step=0.008, max_iter=600000)
        a = symmetrize(a)
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished score: {final_val:.6f}")
        if final_val < best_score:
            best_score = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")
    # fallback
    if best_seq is None:
        best_seq = [1.0] * 256
        best_score = evaluate_sequence(best_seq)
        print("Fallback uniform 256-vector: score", best_score)
    print(f"Search finished. Best score {best_score:.6f}")

"""
search_step_function.py

Heuristic search for a non‑negative step sequence that minimizes

    2·n·max(conv(a, a)) / (sum(a) ** 2)

(the official evaluation is scale‑invariant, so we keep sum(a)=1).

The algorithm proceeds in four phases:

1️⃣  **Exploratory sampling** – a quick, time‑driven sweep of
    several families of probability vectors (uniform, arcsine,
    Beta‑shapes, mixtures, Hann / Hamming windows, Kaiser windows,
    Dirichlet draws).  The best‑scoring candidate for each chosen
    length is retained.

2️⃣  **Local polishing** – the retained vector is refined by a
    cascade of gradient‑based optimisers:
        • Adam on an L₂ flattening loss (conv ≈ constant)
        • Adam on a smooth log‑sum‑exp surrogate of the max‑convolution
          (λ is annealed up to 2 000)
        • Projected sub‑gradient descent on the true max value
        • A cheap stochastic Gaussian walk on the simplex.

    After each stage the vector is symmetrised (a ← (a+a[::-1])/2)
    and re‑projected onto the probability simplex.

3️⃣  **Official evaluation** – the current best vector is scored with
    the official `evaluate_sequence`.  The globally best sequence
    across all lengths is kept.

4️⃣  **Global polishing** – if time permits the overall best vector
    undergoes a final, longer pass through the same four stages.

All time limits are respected; if nothing is found a uniform 256‑vector
is returned as a safe fallback.
"""

import time
import numpy as np
from typing import List, Tuple
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (linear) self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score; works for vectors already
    normalised to Σa = 1.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}
    (Duchi et al. 2008).  O(n log n).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize_and_project(a: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry and project back onto the simplex."""
    a_sym = (a + a[::-1]) / 2.0
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return uniform_shape(n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return uniform_shape(n)
    return mix / mix_sum


def generate_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return w / w.sum()


def generate_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def generate_raised_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = np.sin(np.pi * i / (n - 1)) ** 2
    return w / w.sum()


def generate_kaiser(n: int, beta: float) -> np.ndarray:
    # centre the index range at 0
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
    x = i / ((n - 1) / 2.0)          # scaled to [-1, 1]
    w = i0(beta * np.sqrt(1.0 - x ** 2)) / i0(beta)
    w = np.maximum(w, 0.0)          # guard against tiny negatives
    return w / w.sum()


# ----------------------------------------------------------------------
# Gradient of the flattening (L2) loss
# ----------------------------------------------------------------------


def flatten_gradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Gradient of ‖conv(a,a) – avg‖₂² where avg = 1/(2n‑1) and Σa=1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient: 4 * (a ⊛ diff_rev)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)          # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def adam_flatten(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.35,
) -> Tuple[np.ndarray, float]:
    """Adam optimisation on the L₂ flattening loss."""
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        grad, loss = flatten_gradient(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 30_000 == 0:
            print(
                f"    [Flatten] iter={it:,d} loss={loss:.2e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth surrogate of the max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate B) where
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv‑max)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights on convolution entries
    B = max_c + np.log(sum_exp) / lam

    # weighted sum of sub‑gradients:
    # ∂/∂a Σ_k w_k (a*a)_k = 2 * (a ⊛ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_max(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 2000.0,
    noise_every: int = 1000,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed max‑convolution surrogate."""
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # quadratic temperature schedule (slow annealing)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.6)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 50_000 == 0:
            print(
                f"    [Smooth‑max] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} lam={lam:.1f}"
            )
    return best_a, best_sc

# Exact sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # back‑tracking line‑search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 60_000 == 0:
            print(
                f"    [Sub‑grad] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc

def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 400_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian random walk on the simplex with a tiny uphill‑move probability.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = compute_score(a_prop)

        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 80_000 == 0:
            print(
                f"    [Random walk] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc

def quick_sampling(
    n: int,
    time_budget: float,
    rng: np.random.Generator,
) -> Tuple[np.ndarray, float]:
    """
    Very fast sweep of a large variety of families; returns the best
    candidate (as a NumPy array) found within `time_budget` seconds.
    """
    start = time.time()
    best = None
    best_sc = float("inf")

    def consider(v: np.ndarray):
        nonlocal best, best_sc
        sc = compute_score(v)
        if sc < best_sc:
            best_sc = sc
            best = v.copy()

    # deterministic starters
    consider(uniform_shape(n))
    consider(arcsine_initial(n))
    consider(generate_hann(n))
    consider(generate_hamming(n))
    consider(generate_raised_cosine(n))
    for beta in (2.0, 4.0, 8.0, 12.0):
        consider(generate_kaiser(n, beta))
    # modest grid of symmetric Beta shapes
    for alpha in np.linspace(0.2, 1.5, 13):
        consider(generate_beta_shape(n, alpha, alpha, eps=1e-6))

    # random exploration
    while time.time() - start < time_budget:
        r = rng.random()
        if r < 0.40:      # pure beta shape
            a = rng.uniform(0.02, 5.0)
            b = rng.uniform(0.02, 5.0)
            eps = rng.uniform(1e-8, 0.02 * n)
            cand = generate_beta_shape(n, a, b, eps)
        elif r < 0.70:    # beta mixture
            w = rng.random()
            a1 = rng.uniform(0.02, 4.0)
            b1 = rng.uniform(0.02, 4.0)
            e1 = rng.uniform(1e-8, 0.02 * n)
            a2 = rng.uniform(0.02, 4.0)
            b2 = rng.uniform(0.02, 4.0)
            e2 = rng.uniform(1e-8, 0.02 * n)
            cand = generate_beta_mixture(n, a1, b1, e1, a2, b2, e2, w)
        else:                # Dirichlet draw (encourages sparsity)
            conc = rng.uniform(0.1, 1.0)
            gamma = rng.gamma(conc, 1.0, size=n)
            s = gamma.sum()
            if s == 0.0:
                continue
            cand = gamma / s
        consider(cand)
    return best, best_sc

def propose_candidate(

        samp_budget = min(8.0, remaining * 0.07)   # at most 8 s or ~7 % of time

        )
        a_flat = symmetrize_and_project(a_flat)

        # ‑‑ Smooth‑max stage (log‑sum‑exp surrogate)
        if time.time() < deadline - 2.0:
            a_max, _ = adam_max(
                initial_a=a_flat,
                time_deadline=deadline,
                max_iter=1_200_000,
                lr0=0.10,
                lam_start=1.0,
                lam_end=2000.0,

                noise_scale=2e-5,
            )
            a_max = symmetrize_and_project(a_max)
        else:
            a_max = a_flat

        # ‑‑ Exact sub‑gradient descent

            a_sub, _ = subgradient_optimize(
                a_max,
                time_deadline=deadline,
                step0=0.25,
                max_iter=500_000,
            )
            a_sub = symmetrize_and_project(a_sub)
        else:
            a_sub = a_max

        # ‑‑ Stochastic local walk
        if time.time() < deadline - 1.0:
            a_final, _ = random_local_search(
                a_sub,
                time_deadline=deadline,
                init_step=0.017,
                max_iter=500_000,
            )
            a_final = symmetrize_and_project(a_final)
        else:
            a_final = a_sub

        # ------------------------------------------------------------------
        # 3.  Official evaluation
        # ------------------------------------------------------------------
        val = evaluate_sequence(a_final.tolist())
        print(f"  → official score for length {n} = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()
            print(f"  *** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 4.  Global polishing of the best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the overall best sequence ===")
        a = np.array(best_seq, dtype=float)

        # −‑ Flattening (longer)
        a, _ = adam_flatten(
            initial_a=a,
            time_deadline=deadline,
            max_iter=1_200_000,
            lr0=0.55,
        )
        a = symmetrize_and_project(a)

        # −‑ Smooth‑max (very high temperature)
        if time.time() < deadline - 4.0:
            a, _ = adam_max(
                initial_a=a,
                time_deadline=deadline,
                max_iter=1_500_000,
                lr0=0.08,
                lam_start=1.0,
                lam_end=3000.0,
                noise_every=1500,
                noise_scale=1e-5,
            )
            a = symmetrize_and_project(a)

        # −‑ Sub‑gradient descent
        if time.time() < deadline - 4.0:
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.18,
                max_iter=600_000,
            )
            a = symmetrize_and_project(a)

        # −‑ Final random walk
        if time.time() < deadline - 2.0:
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.010,
                max_iter=800_000,
            )
            a = symmetrize_and_project(a)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Safe fallback
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector, score =", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
