def local_random_refine(a, deadline, max_iter=200000, step_scale=0.01):

        delta = step_scale * rng.random()
        # ensure delta <= best[i] to stay non-negative

"""
propose_candidate – 2024‑11‑27‑v2
================================

A three‑stage optimiser with several enhancements over the previous
implementation.

Stage 1 – Diverse seeding
    Generate a few thousand candidates from many parametric families:
    arcsine, beta, power, triangular, sine and mixtures.  All vectors are
    normalised to Σa = 1 and evaluated with a fast surrogate
    ``score = 2·n·max(conv(a,a))`` (the official objective after normalising).

Stage 2 – Smooth‑max Adam optimisation
    Optimises the smooth approximation
        Φτ(a)= (1/τ)·log Σk exp(τ·conv_k)
    using Adam on the logits (softmax parametrisation).  The temperature τ
    grows exponentially from 0.5 to 150 and the learning rate follows
    ηt = 0.25 / √t.

Stage 3 – Exact polishing
    * Euclidean projected sub‑gradient descent on the true max‑convolution.
    * Exponentiated‑gradient (mirror‑descent) polishing.
    * A simple stochastic “mass‑transfer” local search that moves tiny
      amounts of probability between two coordinates while keeping the sum
      fixed.  All steps respect the global time budget (1000 s).

The routine always returns a non‑empty sequence; if anything fails it falls
back to a uniform distribution.  The best sequence found (according to the
official ``evaluate_sequence``) is printed and returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector `a` is assumed to be normalised to Σa = 1, therefore the
    denominator disappears and the score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators (used for seeding)

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """(i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_sine_shape(n: int) -> np.ndarray:
    """Smooth sine shape a_i ∝ sin(π·(i+0.5)/(n+1))."""
    i = np.arange(n, dtype=float) + 0.5
    vals = np.sin(np.pi * i / (n + 1.0))
    total = vals.sum()
    return vals / total


def generate_mixture_shape(n: int, w_arc: float, w_sine: float) -> np.ndarray:
    """Mixture of arcsine, sine and uniform."""
    arc = arcsine_initial(n)
    sine = generate_sine_shape(n)
    uni = np.full(n, 1.0 / n)
    mix = w_arc * arc + w_sine * sine + (1.0 - w_arc - w_sine) * uni
    return mix / mix.sum()

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 30_000,
                             tau_start: float = 0.5,
                             tau_end: float = 150.0,
                             base_lr: float = 0.25) -> np.ndarray:
    """
    Optimises the smooth approximation of max(conv) using Adam on the logits.
    The temperature τ grows exponentially from `tau_start` to `tau_end`.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits (softmax parametrisation)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        # softmax to obtain a
        w_shift = w - np.max(w)               # for numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth max (log‑sum‑exp) weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t a
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (learning rate ∝ 1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report (fast surrogate)
        if it % 5000 == 0:
            score = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={score:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(a_init: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.10) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective max(conv).
    """
    a = project_to_simplex(a_init.copy())
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of indices where the maximum is attained (within tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·(a * mask) reversed convolution
        rev_mask = mask[::-1]
        conv_rev = convolution_fft(a, rev_mask)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a_init: np.ndarray,
              deadline: float,
              max_iter: int = 120_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient polishing (mirror‑descent with KL geometry).
    """
    a = a_init.copy()
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev_mask = mask[::-1]
        conv_rev = convolution_fft(a, rev_mask)
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * grad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def local_random_refine(a_init: np.ndarray,
                        deadline: float,
                        max_iter: int = 300_000,
                        init_step: float = 0.02) -> np.ndarray:
    """
    Very simple stochastic refinement: move a tiny amount of mass from one
    coordinate to another while keeping the simplex constraint.
    The step size decays geometrically.
    """
    rng = np.random.default_rng()
    a = a_init.copy()
    n = a.shape[0]
    best = a
    best_score = compute_score(best)
    step = init_step

    it = 0
    while time.time() < deadline - 0.05 and it < max_iter:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # propose moving `delta` from i to j
        delta = step * rng.random()
        delta = min(delta, best[i])          # stay non‑negative
        if delta <= 0:
            continue
        candidate = best.copy()
        candidate[i] -= delta
        candidate[j] += delta
        # numerical noise may violate the simplex – re‑project
        candidate = project_to_simplex(candidate)

        score = compute_score(candidate)
        if score < best_score:
            best = candidate
            best_score = score
            # whenever we improve, reset the step a little
            step = init_step
        else:
            # slowly shrink step when we fail to improve
            step *= 0.9995

        it += 1

    return best

def propose_candidate(seed: int = 42,

    """

    non‑empty sequence.

    deadline = start + budget_s - 1.0            # small safety margin

    if "height_sequence_1" in globals():

            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")

                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Seeding (fast surrogate)
    # ------------------------------------------------------------------

    candidate_lengths = [2048]

        candidate_lengths.append(len(best_seq))

    pool: List[Tuple[float, np.ndarray]] = []

    for n in candidate_lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Seeding length {n} ---")

        # deterministic starters
        uniform = np.full(n, 1.0 / n)
        pool.append((compute_score(uniform), uniform.copy()))
        arc = arcsine_initial(n)
        pool.append((compute_score(arc), arc.copy()))
        sine = generate_sine_shape(n)
        pool.append((compute_score(sine), sine.copy()))

        # random beta families
        for _ in range(250):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            pool.append((compute_score(cand), cand))

        # random power‑shape
        for _ in range(250):
            if time.time() >= deadline - 5.0:

            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((compute_score(cand), cand))

        # mixture of arcsine, sine and uniform
        for _ in range(150):
            if time.time() >= deadline - 5.0:
                break
            w_arc = rng.random()
            w_sine = rng.random() * (1.0 - w_arc)
            cand = generate_mixture_shape(n, w_arc, w_sine)

        # triangular bumps
        for _ in range(150):
            if time.time() >= deadline - 5.0:
                break
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

    # keep a few best seeds for the expensive Adam stage
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:10]]
    print(f"\nSelected {len(top_seeds)} top seeds for Adam optimisation "

    # ------------------------------------------------------------------

    refined_candidates: List[Tuple[float, np.ndarray]] = []
