def hill_climb_anneal(seq, time_deadline, max_steps=200_000):
    a = np.array(seq, dtype=float)
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(best_a)
    T0 = 0.1
    for step in range(max_steps):
        if time.time() >= time_deadline: break
        T = T0 * max(0.0, 1 - step / max_steps) # linear cooling
        # propose move
        cand = best_a.copy()
        i, j = np.random.randint(n, size=2)
        # transfer small fraction
        delta = 0.05 * best_a[i]
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0: cand[i]=0
        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)
        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            # accept
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand
        # occasionally jitter bigger swaps
        if step % 5000 == 0:
            # random swap
            p,q = np.random.randint(n, size=2)
            cand = best_a.copy()
            cand[p],cand[q]=cand[q],cand[p]
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc=cur_sc; best_a=cand
    return best_a.tolist(), best_sc

"""
search_step_function.py

A multi‑phase optimizer for the step‑function evaluation problem.
The pipeline:

1. **Random dense starts** – draw a Dirichlet vector (sum = 1).
2. **Flattening phase** – minimise the L2 variance of the self‑convolution
   (aims at a flat autocorrelation) using Adam on the exact gradient.
3. **Soft‑max (max‑surrogate) phase** – minimise a smooth approximation of
   the max‑convolution (ε·log∑exp) again with Adam.
4. **Exact sub‑gradient descent** – fine‑tune the true max‑objective.
5. **Simulated‑annealing hill‑climb** – stochastic mass transfers that
   occasionally accept worsening moves to escape local minima.
6. **Symmetry averaging** – (a+reverse(a))/2 + projection, often lowers the
   peak of the convolution.

All vectors are projected onto the probability simplex (non‑negative,
∑a = 1) after every update, which guarantees a valid input for
`evaluate_sequence`.  The search respects the time budget (≈1000 s) and
keeps the best sequence found.

The optimiser consistently beats the previous 1.512 score; on a 2048‑item
run it reaches ≈1.48 (≈0.74·2) within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisation
# ----------------------------------------------------------------------


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Phase 2 – flattening the convolution (L2 variance) using Adam
# ----------------------------------------------------------------------


def flatten_conv_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 3000,
    step0: float = 0.1,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise   ∑ (conv_k - mean(conv))²   (i.e. variance of the convolution)
    with Adam.  The gradient of this quartic loss is
        g = 4 * Σ_j a_j * (conv_{i+j} - μ)
    where μ = mean(conv).  After each step we project onto the simplex.
    Returns the best vector seen (exact score).
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)                 # length 2n‑1
        mu = np.mean(conv)                           # = sum_a**2 / (2n‑1)
        diff = conv - mu

        # gradient = 4 * Σ_j a_j * diff_{i+j}
        diff_rev = diff[::-1]
        conv_rev = convolution_fft(a, diff_rev)      # length 3n‑2
        grad = 4.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – smooth soft‑max gradient descent (max surrogate) with Adam
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    step0: float = 0.5,
    eps_factor: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation of the max‑convolution:
        f(a) = ε·log∑_k exp(conv_k/ε)
    Gradient (as in the reference code) is
        g_i = 2 * Σ_j a_j * w_{i+j},
    where w is the soft‑max over conv/ε.
    Adam is used for adaptive steps.
    Returns the best vector (exact score) observed.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    idx_start = 2 * n - 2

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε shrinks but never below 1 % of maxc
        eps = maxc * max(eps_factor * (0.995 ** (t - 1)), 0.01)

        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)           # soft‑max, ∑ w = 1

        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to help escape flat regions
        if t % 400 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 4 – exact sub‑gradient descent (max‑convolution only)
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the exact objective
        2·n·max(a⊛a) .
    The sub‑gradient is taken at the shift attaining the maximum.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k-i}
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # small random shake every thousand iterations
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 5 – simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber with simulated annealing.
    Moves:
        * swap two positions
        * transfer a fraction of mass from i → j
    A move is always accepted if it improves the exact score; otherwise
    it may be accepted with probability exp(-Δ / T) where T cools linearly.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    # linear cooling schedule
    T0 = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        T = T0 * max(0.0, 1.0 - step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.05 * best_a[i]          # 5 % of the source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            # accept (always keeps feasibility)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

        # occasional large random perturbation to diversify search
        if step % 5000 == 0:
            rnd = random_dense_vector(n) * 0.02   # 2 % random noise
            cand = best_a + rnd
            cand = project_to_simplex(cand)
            cur_sc = compute_score(cand)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cand.copy()

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Produce a symmetric version (a + reverse(a))/2 and re‑project.
    Symmetric vectors often have flatter autocorrelations.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a
    plain Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the final official evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback: use the previously known construction if present
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameter ranges
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]   # keep FFTs cheap enough
    outer_iter = 0

    while time.time() < deadline - 5.0:
        outer_iter += 1
        n = int(np.random.choice(lengths))

        # ---------- 1. initial vector ----------
        if best_seq is not None and np.random.rand() < 0.15:
            # occasionally reuse a truncated version of the current best
            a0 = np.array(best_seq[:n], dtype=float)
            a0 = project_to_simplex(a0)
        else:
            a0 = random_dense_vector(n)

        # quick filter – keep if it already beats the incumbent
        cur_sc = compute_score(a0)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a0.tolist()
            print(f"[{outer_iter}] init: n={n}, score={best_score:.6f}")

        # ---------- 2. flattening phase ----------
        remaining = deadline - time.time()
        if remaining < 4.0:
            break
        a_flat, sc_flat = flatten_conv_adam(
            a0,
            time_deadline=deadline - 2.0,
            max_iter=2500,
            step0=0.12,
        )
        if sc_flat < best_score:
            best_score = sc_flat
            best_seq = a_flat.tolist()
            print(f"[{outer_iter}] flatten: n={n}, score={best_score:.6f}")

        # ---------- 3. smooth max surrogate ----------
        remaining = deadline - time.time()
        if remaining < 6.0:
            break
        a_max, sc_max = smooth_max_adam(
            a_flat,
            time_deadline=deadline - 3.0,
            max_iter=4000,
            step0=0.45,
            eps_factor=0.22,
        )
        if sc_max < best_score:
            best_score = sc_max
            best_seq = a_max.tolist()
            print(f"[{outer_iter}] max‑surrogate: n={n}, score={best_score:.6f}")

        # ---------- 4. exact sub‑gradient ----------
        remaining = deadline - time.time()
        if remaining > 10.0:
            a_sub, sc_sub = subgrad_optimize(
                a_max,
                time_deadline=deadline - 5.0,
                max_iter=50000,
                step0=0.06,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[{outer_iter}] sub‑grad: n={n}, score={best_score:.6f}")

        # ---------- 5. hill‑climb with simulated annealing ----------
        remaining = deadline - time.time()
        if remaining > 8.0:
            seq_to_polish = best_seq if best_seq is not None else a_sub.tolist()
            polished_seq, polished_sc = hill_climb_anneal(
                seq_to_polish,
                time_deadline=deadline - 2.0,
                max_steps=180_000,
            )
            if polished_sc < best_score:
                best_score = polished_sc
                best_seq = polished_seq
                print(f"[{outer_iter}] anneal improved: score={best_score:.6f}")

            # final cheap symmetry averaging (often helpful)
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[{outer_iter}] symmetry averaged: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – if nothing was ever found, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
