"""
Improved search for a low‑score step‑function.

The algorithm builds on the previous “beta‑family → flatten → smooth‑max → sub‑grad
→ random walk” pipeline but adds a few new ideas:

* **Symmetry enforcement** – the optimal sequence is symmetric; after each
  update we replace a by (a + a[::-1])/2 before projection onto the simplex.
* **Dynamic length growth** – we first explore modest lengths (512‑2048) to get a
  good shape, then up‑sample the best candidate to a larger length (up to
  8192) and continue optimisation.  Larger n moves the theoretical lower bound
  2·n/(2n‑1) closer to 1.
* **Longer and more aggressive flattening** – a higher initial learning rate
  (0.4) and a larger iteration budget are used.
* **Higher temperature for the smooth‑max stage** – λ is annealed up to 2000,
  giving a tighter approximation of the true max.
* **Momentum‑free mirror‑descent style updates** – after each Adam step the
  vector is symmetrised and projected onto the probability simplex.
* **Time‑driven iteration budgeting** – each phase receives as many iterations
  as the remaining wall‑time permits (with a small safety margin).  The code
  always returns a feasible sequence (uniform 256‑vector) even if the time
  budget is exhausted early.

The objective used during the optimisation is the fast surrogate
`2·n·max(conv(a,a))` (the sum is kept equal to 1).  The official
`evaluate_sequence` is called only for the final candidate(s).
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of `a` (full length)."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate score: 2·n·max(conv(a,a)) for probability vectors."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) – O(n log n).
    """
    n = v.shape[0]
    # if all non‑positive, return uniform
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce even symmetry: v ← (v + v[::-1]) / 2."""
    return 0.5 * (v + v[::-1])

# Shape generators

def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Unnormalised symmetric beta density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta densities (renormalised)."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) density."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Resize `a` to `new_n` points by linear interpolation and renormalise."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_up = np.interp(x_new, x_old, a)
    a_up = a_up / a_up.sum()
    return a_up


# ----------------------------------------------------------------------
# Gradient of the L2 flattening loss
# ----------------------------------------------------------------------


def flatten_gradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Gradient of ||conv(a,a) - avg||_2^2 where avg = 1/(2n-1) and Σa = 1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient: 4 * (a ⊛ diff_rev)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)      # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def adam_flatten(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.25,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the flattening L2 loss.  Symmetry is enforced after each
    update.  Returns the best vector seen (projected) and its fast surrogate
    score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        grad, loss = flatten_gradient(a)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        # enforce symmetry and simplex
        a_new = symmetrize(a_new)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 20000 == 0:
            print(
                f"[Flatten] it={it:,d} loss={loss:.2e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth surrogate of the max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value B) where
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv‑max)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max on convolution entries
    B = max_c + np.log(sum_exp) / lam

    # weighted convolution gradient
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_max(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 800.0,
    noise_every: int = 1000,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution surrogate.
    Symmetry is enforced after each update.
    Returns the best vector (projected) and its fast surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # quadratic schedule for λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = symmetrize(a_new)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"[Adam‑max] it={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f} λ={lam:.1f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max; back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    deadline: float,
    step0: float = 0.3,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Symmetry is enforced after each update.
    Returns best vector seen and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # sub‑gradient w.r.t. the active entry
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # back‑tracking line‑search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = symmetrize(a_new)
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"[Sub‑grad] it={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap stochastic Gaussian walk on the simplex
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with an almost‑always‑downhill rule.
    Symmetry is preserved.
    Returns best vector and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = symmetrize(a_proposed)
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # accept only improvements (tiny uphill probability to escape)
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"[Walk] it={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine performs a quick exploration on modest lengths, then
    up‑samples the best shape to a larger length and spends the remaining
    budget on extensive flattening, smooth‑max Adam, sub‑gradient descent
    and a final stochastic walk.  All optimisation respects the global
    time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # leave a 1 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback (previous best if available)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[Fallback] official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Quick sampling on small‑to‑medium lengths
    # ------------------------------------------------------------------
    lengths_small = [512, 1024, 2048]
    global_best_a = None
    global_best_n = None
    global_best_sc = float("inf")

    for n in lengths_small:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Sampling length n = {n} ===")

        # deterministic starters
        starters: List[np.ndarray] = [
            np.full(n, 1.0 / n),          # uniform
            arcsine_initial(n),           # arcsine
        ]
        # symmetric beta grid
        for alpha in np.linspace(0.15, 0.85, 13):
            starters.append(generate_beta_shape(n, alpha, alpha, eps=1e-6))

        # evaluate starters
        for cand in starters:
            sc = compute_score(cand)
            if sc < global_best_sc:
                global_best_sc = sc
                global_best_a = cand.copy()
                global_best_n = n

        # random beta / mixtures
        max_samples = int(3000 * (budget_s / 1000.0))
        for _ in range(max_samples):
            if time.time() >= deadline - 30.0:
                break
            if rng.random() < 0.35:  # mixture
                w = rng.random()
                a1 = generate_beta_shape(
                    n,
                    rng.uniform(0.02, 3.5),
                    rng.uniform(0.02, 3.5),
                    eps=rng.uniform(1e-8, max(0.01, 0.02 * n)),
                )
                a2 = generate_beta_shape(
                    n,
                    rng.uniform(0.02, 3.5),
                    rng.uniform(0.02, 3.5),
                    eps=rng.uniform(1e-8, max(0.01, 0.02 * n)),
                )
                cand = w * a1 + (1.0 - w) * a2
                cand /= cand.sum()
            else:  # single beta
                cand = generate_beta_shape(
                    n,
                    rng.uniform(0.02, 3.5),
                    rng.uniform(0.02, 3.5),
                    eps=rng.uniform(1e-8, max(0.01, 0.02 * n)),
                )
            sc = compute_score(cand)
            if sc < global_best_sc:
                global_best_sc = sc
                global_best_a = cand.copy()
                global_best_n = n

    # ------------------------------------------------------------------
    # 2.  Upsample to a larger length if time permits
    # ------------------------------------------------------------------
    target_large_n = 8192
    if global_best_a is None:
        # as a safety net, start from arcsine of moderate size
        global_best_n = 1024
        global_best_a = arcsine_initial(global_best_n)

    # decide final length (capped by target_large_n and remaining budget)
    remaining = deadline - time.time()
    if remaining > 60.0 and global_best_n < target_large_n:
        large_n = min(target_large_n, global_best_n * 4)
        print(f"\n[Upsample] Enlarging from {global_best_n} → {large_n}")
        global_best_a = upsample(global_best_a, large_n)
        global_best_n = large_n
    else:
        large_n = global_best_n

    # ------------------------------------------------------------------
    # 3.  Intensive optimisation on the (possibly) larger vector
    # ------------------------------------------------------------------
    # ---- Flattening ----------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_iter_flat = int(min(200_000, remaining * 25))
        print(f"[Flatten] max_iter={max_iter_flat}")
        a_flat, _ = adam_flatten(
            initial_a=global_best_a,
            deadline=deadline,
            max_iter=max_iter_flat,
            lr0=0.40,
        )
    else:
        a_flat = global_best_a

    # ---- Smooth‑max Adam -----------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_iter_adam = int(min(300_000, remaining * 28))
        print(f"[Adam‑max] max_iter={max_iter_adam}")
        a_max, _ = adam_max(
            initial_a=a_flat,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.10,
            lam_start=1.0,
            lam_end=2000.0,          # more aggressive temperature
            noise_every=1200,
            noise_scale=5e-5,
        )
    else:
        a_max = a_flat

    # ---- Exact sub‑gradient descent ------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_iter_sub = int(min(200_000, remaining * 30))
        print(f"[Sub‑grad] max_iter={max_iter_sub}")
        a_sub, _ = subgradient_optimize(
            a_max,
            deadline=deadline,
            step0=0.30,
            max_iter=max_iter_sub,
        )
    else:
        a_sub = a_max

    # ---- Stochastic local walk -----------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        max_iter_walk = int(min(250_000, remaining * 35))
        print(f"[Walk] max_iter={max_iter_walk}")
        a_final, _ = random_local_search(
            a_sub,
            deadline=deadline,
            init_step=0.012,
            max_iter=max_iter_walk,
        )
    else:
        a_final = a_sub

    # ------------------------------------------------------------------
    # 4.  Official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(a_final.tolist())
        print(f"[Official] final score for length {global_best_n} = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()
    except Exception as exc:
        print(f"[Error] evaluation failed: {exc}")

    # ------------------------------------------------------------------
    # 5.  Global polishing (optional second pass) if any time left
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_flat = int(min(180_000, remaining * 22))
        print(f"[Global flatten] max_iter={max_iter_flat}")
        a, _ = adam_flatten(
            initial_a=a,
            deadline=deadline,
            max_iter=max_iter_flat,
            lr0=0.35,
        )

        remaining = deadline - time.time()
        if remaining > 4.0:
            max_iter_adam = int(min(350_000, remaining * 27))
            print(f"[Global Adam‑max] max_iter={max_iter_adam}")
            a, _ = adam_max(
                initial_a=a,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.09,
                lam_start=1.0,
                lam_end=2500.0,
                noise_every=1500,
                noise_scale=4e-5,
            )

        remaining = deadline - time.time()
        if remaining > 4.0:
            max_iter_sub = int(min(180_000, remaining * 28))
            print(f"[Global sub‑grad] max_iter={max_iter_sub}")
            a, _ = subgradient_optimize(
                a,
                deadline=deadline,
                step0=0.25,
                max_iter=max_iter_sub,
            )

        remaining = deadline - time.time()
        if remaining > 2.0:
            max_iter_walk = int(min(250_000, remaining * 38))
            print(f"[Global walk] max_iter={max_iter_walk}")
            a, _ = random_local_search(
                a,
                deadline=deadline,
                init_step=0.010,
                max_iter=max_iter_walk,
            )

        try:
            final_val = evaluate_sequence(a.tolist())
            print(f"[Global] final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val = final_val
                best_seq = a.tolist()
                print("*** Updated global best after polishing ***")
        except Exception as exc:
            print(f"[Error] final evaluation failed: {exc}")

    # ------------------------------------------------------------------
    # 6.  Ensure a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("[Fallback] uniform 256‑vector used.", best_val)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
