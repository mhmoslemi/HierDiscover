"""
Improved step‑function optimizer.

The evaluation metric (after normalising the sequence to Σa = 1) is
    score = 2·n·max(conv(a,a))
where conv is a linear convolution.  The objective is non‑convex because the
maximum is taken after a quadratic form.  The algorithm follows three ideas:

1️⃣  **Diverse seeding** – a large pool of candidates from many parametric
    families (arcsine, beta, power, triangular, mixtures, Dirichlet).
    The fast surrogate ``compute_score`` (the exact objective for Σa=1) is
    used to keep only a few promising seeds.

2️⃣  **Smooth‑max Adam** – each seed is refined by minimising a smooth
    approximation of the max‑convolution
        Φτ(a)=τ⁻¹ log Σk exp(τ·convk)
    using Adam on the soft‑max logits (the simplex constraint is handled by
    the softmax).  The temperature τ is raised from 0.5 to 150 so the surrogate
    gradually becomes the true max.

3️⃣  **Escaping local minima** – a simulated‑annealing local search that
    repeatedly transfers a random amount of mass between two coordinates.
    The schedule cools the temperature linearly to zero.  The best vector from
    this stage is polished with the exact sub‑gradient descent and an
    exponentiated‑gradient (mirror‑descent) step exactly as in the previous
    state‑of‑the‑art code.

All phases respect the global 1000 s budget, keep the globally best sequence
and finally return it (as a plain Python ``list[float]``).  The returned
sequence has Σa = 1, therefore the official score coincides with the fast
surrogate used throughout the search.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real, length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex Σx = 1, x ≥ 0.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic symmetric arcsine shape."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def sample_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a point from Dirichlet(α,…,α) and normalise to Σ=1."""
    g = rng.gamma(alpha, 1.0, size=n)
    if g.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return g / g.sum()

# Exact sub‑gradient descent & exponentiated‑gradient polish
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """Euclidean projected sub‑gradient descent on the true max‑convolution."""
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.3) -> np.ndarray:
    """Exponentiated‑gradient (mirror descent with KL geometry) polishing."""
    a = a.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (used on the surrogate)
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 30_000,
                             tau_start: float = 0.5,
                             tau_end: float = 150.0,
                             base_lr: float = 0.3) -> np.ndarray:
    """
    Adam optimiser on the smooth max‑convolution Φτ.
    The optimisation variable is the logits w, a = softmax(w).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)          # initialise logits

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax probabilities (a lives on the simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        conv = convolution_fft(a, a)

        # smooth‑max weights (log‑sum‑exp gradient)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp                     # shape (2n‑1,)

        # gradient w.r.t. a
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if it % 5000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a

# Simulated‑annealing local search (non‑gradient)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000) -> Tuple[np.ndarray, float]:
    """
    Simple temperature‑controlled random walk on the simplex.
    Moves transfer a random amount of mass from one coordinate to another.
    Returns the best vector seen and its surrogate score.
    """
    rng = np.random.default_rng()
    a = a_start.copy()
    n = len(a)
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    # temperature starts as a fraction of the current score and goes to zero
    T0 = cur_score * 0.2
    start_time = time.time()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear cooling
        T = T0 * max(0.0, 1.0 - it / max_iters)

        # propose a transfer
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] == 0.0:
            continue
        delta = rng.random() * a[i]  # up to the whole mass at i
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = compute_score(a_new)

        # Metropolis accept/reject
        accept = False
        if new_score < cur_score:
            accept = True
        elif T > 1e-12:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        # occasional checkpoint (helps with time‑outs)
        if it % 50_000 == 0:
            now = time.time()
            elapsed = now - start_time
            print(f"       SA it={it:,d} best={best_score:.6f} elapsed={elapsed:.1f}s")

    return best_a, best_score

    # random families
    for _ in range(300):
        if time.time() >= deadline - 5.0:
            break
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        eps = rng.uniform(0.0, max(1.0, 0.02 * n))
        cand = generate_beta_shape(n, a, b, eps)
        pool.append((compute_score(cand), cand))

    for _ in range(300):
        if time.time() >= deadline - 5.0:
            break
        p = rng.uniform(0.1, 2.5)
        eps = rng.uniform(0.0, 2.0)
        cand = generate_power_shape(n, p, eps)
        pool.append((compute_score(cand), cand))

    for _ in range(200):
        if time.time() >= deadline - 5.0:
            break
        w = rng.random()
        cand = generate_mixture_shape(n, w)
        pool.append((compute_score(cand), cand))

    for _ in range(200):
        if time.time() >= deadline - 5.0:
            break
        peak = rng.integers(0, n)
        cand = generate_triangular_shape(n, peak)
        pool.append((compute_score(cand), cand))

    # Dirichlet‑type seeds with various concentration parameters
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            cand = sample_dirichlet(n, alpha, rng)
            pool.append((compute_score(cand), cand))

    # keep a handful of the best seeds for further optimisation
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:10]]
    print(f"\nSelected {len(top_seeds)} top seeds for Adam optimisation "
          f"(surrogate range {pool[0][0]:.6f} – {pool[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 3.  Smooth‑max Adam optimisation on the top seeds
    # ------------------------------------------------------------------
    adam_refined: List[np.ndarray] = []
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 4.0:
            break
        remaining = deadline - time.time()
        max_iter = int(min(30_000, remaining * 30))
        print(f"\n=== Adam refinement of seed {idx}/{len(top_seeds)} (max_iter={max_iter}) ===")
        a_adam = adam_optimize_smooth_max(seed_vec,
                                          deadline,
                                          max_iters=max_iter,
                                          tau_start=0.5,
                                          tau_end=150.0,
                                          base_lr=0.3)
        surrogate = compute_score(a_adam)
        print(f"    Post‑Adam surrogate = {surrogate:.6f}")
        adam_refined.append(a_adam)
        pool.append((surrogate, a_adam))

    # ------------------------------------------------------------------
    # 4.  Simulated‑annealing local search starting from the best Adam seed
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    best_adam = pool[0][1]                    # best surrogate after Adam stage
    print(f"\n--- Starting simulated‑annealing refinement (n={n}) ---")
    remaining = deadline - time.time()
    max_sa_iters = int(min(300_000, remaining * 150))
    sa_best, sa_score = simulated_annealing(best_adam,
                                            deadline,
                                            max_iters=max_sa_iters)
    print(f"    Simulated‑annealing best surrogate = {sa_score:.6f}")
    pool.append((sa_score, sa_best))

    # ------------------------------------------------------------------
    # 5.  Exact refinement (sub‑gradient + exponentiated‑gradient)
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    candidate = pool[0][1]
    n = candidate.shape[0]
    print(f"\n--- Exact refinement (n={n}) ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(250_000, remaining * 30))
    refined = subgradient_descent_exact(candidate,
                                        deadline,
                                        max_iter=max_iter_sub,
                                        step0=0.10)

    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        max_iter_eg = int(min(150_000, remaining * 30))
        refined = eg_refine(refined, deadline,
                            max_iter=max_iter_eg,
                            eta0=0.22)

    # ------------------------------------------------------------------
    # 6.  Official evaluation and global best bookkeeping
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(refined.tolist())
        print(f"Official score after final refinement = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            print("*** New global best found! ***")
    except Exception as e:
        print(f"Evaluation failed after refinement: {e}")

    # ------------------------------------------------------------------
    # 7.  Optional extra polishing on the global best (time permitting)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Global polishing of the best sequence ---")
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.08)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(200_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.18)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 8.  Guard – ensure a non‑empty sequence is always returned
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = n
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
