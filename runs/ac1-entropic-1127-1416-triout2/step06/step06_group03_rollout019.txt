def solve_lp_max_sum(f, M):
    n = len(f)
    A = build_conv_matrix(f)

    c = -np.ones(n)  # minimize -sum g => maximize sum

    if not res.success:

    # returns best t, resulting max conv.

"""
Propose a low‑value step sequence.

Algorithm
---------

1.  Initialise a non‑negative shape on the probability simplex
    (arcsine distribution) for a chosen length *n*.

2.  **LP‑driven Frank‑Wolfe step**  
    Repeatedly solve a linear programme

        maximise   Σ g_j
        subject to  (f * g)_i ≤ M   (i = 0…2n‑2)
                    g_j ≥ 0

    where *M* is the current maximum convolution value of *f*.
    The LP finds a direction *g* that pushes mass into indices where
    the current shape *f* is small without violating the bound *M*.
    The new shape is a convex combination *f←(1‑t)f+t g*.
    The optimal mixing coefficient *t* is obtained by an inexpensive
    1‑D line‑search exploiting the exact formula for the convolution of
    the mixture.

3.  Iterate the LP‑step a few times (until no improvement or time runs
    out).  This implements a Frank‑Wolfe type optimisation for the
    non‑convex objective
        2·n·maxₖ (f*f)ₖ   (∑f = 1).

4.  **Smooth‑max Adam polishing**  
    With the remaining time run the soft‑max/Adam optimiser from the
    previous state‑of‑the‑art code to fine‑tune the shape.

5.  Return the best shape found (as a Python list).  All entries are
    non‑negative and the vector is normalised to sum = 1, so the official
    ``evaluate_sequence`` sees exactly the same objective value as our
    fast surrogate ``2·n·max(conv)``.
"""

# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(n log n))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a)) – assumes Σa = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple, diverse initial shapes
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Discretised arcsine density on {0,…,n‑1} (already normalised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Pre‑compute data layout for the convolution matrix (used by the LP)
# ----------------------------------------------------------------------
def _precompute_conv_pattern(n: int):
    """
    Return (rows, cols, m) describing the (2n‑1)×n matrix A such that
    A @ g = conv(f, g) for any vector f of length n.
    """
    m = 2 * n - 1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    idx = 0
    for j in range(n):
        rows[idx:idx + n] = np.arange(j, j + n, dtype=np.int32)
        cols[idx:idx + n] = j
        idx += n
    return rows, cols, m


def _build_conv_matrix(f: np.ndarray, rows: np.ndarray,
                       cols: np.ndarray, m: int):
    """Construct the sparse convolution matrix for the current f."""
    data = np.tile(f, f.shape[0])               # f repeated n times
    return sp.csr_matrix((data, (rows, cols)), shape=(m, f.shape[0]))


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, m: int) -> np.ndarray:
    """
    Solve   maximise Σ g_j   s.t.   (f * g)_i ≤ M,   g ≥ 0.
    Returns the optimal g (not normalised).
    """
    n = f.shape[0]
    A = _build_conv_matrix(f, rows, cols, m)
    b = np.full(m, M, dtype=float)
    c = -np.ones(n)                     # minimise –Σg  ⇔ maximise Σg
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                 bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# Line‑search for the optimal mixing coefficient t
# ----------------------------------------------------------------------
def _best_mix(t_candidates, conv_ff, conv_gg, conv_fg):
    """Return (t, new_max) that minimises the mixed convolution maximum."""
    best_t = 0.0
    best_max = np.max(conv_ff)
    for t in t_candidates:
        mixed = ((1.0 - t) ** 2) * conv_ff + (t ** 2) * conv_gg + \
                2.0 * t * (1.0 - t) * conv_fg
        mx = mixed.max()
        if mx < best_max - 1e-15:
            best_max = mx
            best_t = t
    return best_t, best_max


# ----------------------------------------------------------------------
# Soft‑max + Adam optimiser (the "smooth‑max" polish phase)
# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _smooth_adam_optimize(z0: np.ndarray, deadline: float,
                          n: int, beta0: float = 5.0,
                          beta_max: float = 120.0,
                          ramp_iters: int = 20000,
                          lr: float = 0.05) -> np.ndarray:
    """Adam on the log‑softmax parametrisation with a temperature‑schedule."""
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0
    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear temperature increase
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        w_norm = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

        # gradient w.r.t. a : 2·(a ⋆ reverse(w_norm))
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            sc = fast_score(_softmax(z))
            print(f"    Adam it={t:,d} β={beta:.1f} score={sc:.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Return a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine first performs a few Frank‑Wolfe‑style LP steps
    (fast, shape‑optimising) and then uses a smooth‑max Adam refinement.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  2048 is a comfortable trade‑off between
    # LP size (≈4 M non‑zeros) and achievable bound.
    # ------------------------------------------------------------------
    n = 2048 if budget_s < 600 else 4096
    print(f"=== Optimisation started (n={n}) ===")

    # ------------------------------------------------------------------
    # Initial shape on the simplex
    # ------------------------------------------------------------------
    f = arcsine_initial(n)           # already sums to 1
    f = np.maximum(f, 0.0)
    f /= f.sum()

    # ------------------------------------------------------------------
    # Pre‑compute sparse pattern for the convolution matrix (used by LP)
    # ------------------------------------------------------------------
    if _HAVE_SCIPY:
        rows_pat, cols_pat, m_pat = _precompute_conv_pattern(n)
    else:
        rows_pat = cols_pat = None
        m_pat = None

    # ------------------------------------------------------------------
    # Current objective value
    # ------------------------------------------------------------------
    conv_ff = convolution_fft(f, f)
    M = conv_ff.max()
    best_score = 2.0 * n * M
    best_f = f.copy()
    print(f"Initial surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Frank‑Wolfe LP loop (skip if SciPy is unavailable)
    # ------------------------------------------------------------------
    if _HAVE_SCIPY:
        t_grid = [0.0, 0.01, 0.02, 0.04, 0.06, 0.08, 0.10,
                  0.15, 0.20, 0.30, 0.40, 0.50, 0.70, 0.90]
        max_outer = 30
        for it in range(1, max_outer + 1):
            if time.time() >= deadline - 8.0:
                print("Time limit approaching – exit LP loop")
                break

            # ---- solve LP to obtain direction g -----------------------
            try:
                g_raw = _solve_lp_max_sum(f, M, rows_pat, cols_pat, m_pat)
            except Exception as exc:                       # pragma: no cover
                print(f"LP failure at iter {it}: {exc}")
                break

            g = np.maximum(g_raw, 0.0)
            s = g.sum()
            if s <= 0.0:
                print(f"LP returned zero‑mass vector at iter {it}")
                break
            g /= s                                     # normalise to Σg = 1

            # ---- evaluate mixed convolutions -------------------------
            conv_gg = convolution_fft(g, g)
            conv_fg = convolution_fft(f, g)

            # ---- best mixing coefficient -----------------------------
            t_opt, new_M = _best_mix(t_grid, conv_ff, conv_gg, conv_fg)

            if t_opt == 0.0:
                print(f"No improving mix found (iter {it}) – stop")
                break

            # ---- update the current shape ----------------------------
            f = (1.0 - t_opt) * f + t_opt * g
            f = np.maximum(f, 0.0)
            f /= f.sum()

            conv_ff = convolution_fft(f, f)
            M = conv_ff.max()
            score = 2.0 * n * M
            print(f"Iter {it:2d}: t={t_opt:.3f}  score={score:.6f}")

            if score + 1e-12 < best_score:
                best_score = score
                best_f = f.copy()
            else:                                   # no further improvement
                print("Score stagnated – exiting LP loop")
                break
    else:                                           # pragma: no cover
        print("SciPy unavailable – skipping LP stage")

    # ------------------------------------------------------------------
    # Adam polishing with the remaining budget
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        print(f"Polishing with Adam (≈{time_left:.1f}s left)")
        eps = 1e-12
        z0 = np.log(best_f + eps)
        polished = _smooth_adam_optimize(z0, deadline, n=n,
                                         beta0=5.0, beta_max=120.0,
                                         ramp_iters=20000, lr=0.05)
        # re‑normalise (soft‑max already guarantees Σ=1, but guard against drift)
        polished = np.maximum(polished, 0.0)
        if polished.sum() > 0:
            polished /= polished.sum()
        best_f = polished
        best_score = fast_score(best_f)
        print(f"After Adam polish: surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks & official evaluation
    # ------------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    if best_f.sum() == 0:
        best_f = np.full(n, 1.0 / n)
    else:
        best_f /= best_f.sum()

    try:
        official = evaluate_sequence(best_f.tolist())
        print(f"Official evaluation = {official:.6f}")
    except Exception as exc:                         # pragma: no cover
        print(f"Official evaluation failed: {exc}")
        official = None

    return best_f.tolist()
