    a = np.array(seq, dtype=float)

    best = a.copy(); best_score = fast_score(a)

    cur_score = best_score

    step = 0; max_steps = 400_000  # or time based

    T0 = max(0.1, (best_score - 1.0) * 0.5)  # set based on gap

    T_final = 0.001

    while time.time() < deadline and step < max_steps:

        i, j = np.random.randint(n, size=2)

        if i == j: continue

        # choose delta proportionally to a[i] (random factor)
        delta = np.random.rand() * a[i] * 0.5  # up to 50% of mass of i

        a_candidate = a.copy()

        a_candidate[i] -= delta; a_candidate[j] += delta

        # ensure nonnegative:
        if a_candidate[i] < 0:

def convolution_fft(a, b):

    m = len(a) + len(b) - 1

    n_fft = next_pow2(m)

    A = np.fft.rfft(a, n_fft)

    B = np.fft.rfft(b, n_fft)

    C = A * B

    conv = np.fft.irfft(C, n_fft)[:m]

    return conv

    for step in range(max_steps):

        if time.time() >= deadline - 0.5: break

        # choose move type

        if np.random.rand() < 0.3:  # swap

            i, j = np.random.randint(n, size=2)

            cand = a.copy()

            cand[i], cand[j] = cand[j], cand[i]

        else:
            i, j = np.random.randint(n, size=2)

            if i == j: continue

            # transfer fraction up to step_fraction (maybe 0.3)

            frac = np.random.rand() * 0.3

            delta = frac * a[i]  # move from i to j

            cand = a.copy()

            cand[i] -= delta

            cand[j] += delta

            # ensure nonnegative; if cand[i] <0, adjust small epsilon

            if cand[i] < 0:
                # put back to zero; adjust j accordingly
                delta = a[i]  # move all
                cand[i] = 0.0
                cand[j] = a[j] + delta

        total = np.sum(cand)

            if cur_score < best_score:

"""
Search for a low‑score non‑negative step function.
The algorithm runs a few random restarts, each consisting of:
  * a soft‑max (exponential‑gradient) optimisation that smoothly
    approximates the ∞‑norm of the self‑convolution,
  * a long simulated‑annealing hill‑climb that moves mass between
    coordinates, occasionally swapping or re‑randomising a pair.
Both stages respect the 1000 s wall‑time budget and use only
numpy’s FFT for cheap evaluation of the exact objective
(the same quantity returned by `evaluate_sequence` when the
vector is normalised to sum = 1).  The best vector found is
returned unchanged – scaling does not affect the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    if x <= 0:
        return 1
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Result length is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # uniform fallback
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (fast when Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute the exact value returned by `evaluate_sequence` for a
    vector whose entries sum to 1:
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Exponential‑gradient (soft‑max) optimisation
# ----------------------------------------------------------------------


def exponential_gradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    iters: int = 150_000,
    beta0: float = 0.5,
    beta_max: float = 80.0,
    eta0: float = 0.6,
    report_every: int = 0,
):
    """
    Minimise a soft‑max approximation of the max‑convolution.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    L = next_pow2(3 * n - 2)          # length sufficient for all convolutions
    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, iters + 1):
        if time.time() >= time_deadline - 0.5:
            break

        # --- a*a -------------------------------------------------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = np.max(conv)

        # --- soft‑max weights ------------------------------------
        beta = min(beta_max, beta0 * np.sqrt(t))
        exp_terms = np.exp(beta * (conv - max_c))   # stabilised
        w = exp_terms / np.sum(exp_terms)

        # --- gradient = 2·(w ⋆ a_rev) ----------------------------
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # --- exponentiated update --------------------------------
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        # --- keep best -------------------------------------------
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if report_every and t % report_every == 0:
                print(f"  iter {t:8d} – new best {cur_val:.6f}")

    # final check
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def annealed_hill_climb(
    seq,
    time_deadline,
    max_steps: int = 300_000,
    init_temp: float = 0.05,
    final_temp: float = 0.001,
):
    """
    Stochastic hill‑climber with a temperature schedule.
    Moves are of three types:
        * swap two coordinates,
        * transfer a fraction of mass from i to j,
        * repartition the mass of a pair (i,j) randomly.
    Only moves that improve the exact objective are always accepted;
    worsening moves are accepted according to the Metropolis rule.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    cur_score = fast_score(a)

    best_a = a.copy()
    best_score = cur_score

    # re‑normalise every few iterations to avoid drift
    renorm_interval = 1000

    for step in range(max_steps):
        if time.time() >= time_deadline - 0.5:
            break

        move_type = np.random.rand()

        # copy current vector
        cand = a.copy()

        if move_type < 0.1:
            # random repartition of the mass of two positions
            i, j = np.random.randint(n, size=2)
            total = cand[i] + cand[j]
            # draw a new split from a uniform Dirichlet(1,1)
            split = np.random.rand()
            cand[i] = split * total
            cand[j] = (1.0 - split) * total
        elif move_type < 0.6:
            # transfer a fraction of mass from i to j (small step)
            i, j = np.random.randint(n, size=2)
            if cand[i] == 0.0:
                continue
            # up to 20 % of the source mass
            frac = np.random.rand() * 0.2
            delta = frac * cand[i]
            cand[i] -= delta
            cand[j] += delta
        else:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        # Floating‑point drift – re‑normalise periodically
        if (step % renorm_interval) == 0:
            s = np.sum(cand)
            if s != 0.0:
                cand /= s

        cand_score = fast_score(cand)

        # temperature schedule (exponential decay)
        T = init_temp * (final_temp / init_temp) ** (step / max_steps)
        # acceptance decision
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a.tolist(), best_score

                best_seq = climbed_seq

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing if time permits
    # ------------------------------------------------------------------

"""
Search for a low‑score non‑negative step function.

The algorithm respects the 1000 s wall‑time limit and proceeds as follows:

1.  Try a supplied fallback (if any).
2.  Perform a small number of random Dirichlet restarts.
    * Each restart is refined by a soft‑max (exponential‑gradient) stage
      that approximates the ∞‑norm of the self‑convolution.
    * The refined vector is then polished by a long simulated‑annealing
      hill‑climb that moves mass between coordinates, occasionally swaps
      or repartitions a pair.  Down‑hill moves are always accepted; uphill
      moves are accepted with Metropolis probability.
3.  If time permits, a final polishing hill‑climb is run.
4.  The best vector found (or a tiny uniform fallback) is returned.
   The score is invariant under scaling, so the vector is kept with
   unit ℓ¹‑norm – exactly the quantity used by `fast_score`, which matches
   `evaluate_sequence` for such normalised inputs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    if x <= 0:
        return 1
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using an FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # collapse to the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective (fast when Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Exact value returned by `evaluate_sequence` for a vector whose
    entries sum to 1:  2·n·max(a * a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Exponential‑gradient (soft‑max) optimisation
# ----------------------------------------------------------------------


def exponential_gradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    iters: int = 150_000,
    beta0: float = 0.5,
    beta_max: float = 80.0,
    eta0: float = 0.6,
    report_every: int = 0,
):
    """
    Minimise a soft‑max approximation of the max‑convolution.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    L = next_pow2(3 * n - 2)          # length sufficient for all FFT‑based convolutions
    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, iters + 1):
        if time.time() >= time_deadline - 0.5:
            break

        # ----- a*a -------------------------------------------------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = np.max(conv)

        # ----- soft‑max weights ------------------------------------
        beta = min(beta_max, beta0 * np.sqrt(t))
        exp_terms = np.exp(beta * (conv - max_c))     # stabilised exponentials
        w = exp_terms / np.sum(exp_terms)

        # ----- gradient = 2·(w ⋆ a_rev) ----------------------------
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- exponentiated (mirror) update -----------------------
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        # ----- keep track of the best -------------------------------
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if report_every and t % report_every == 0:
                print(f"  iter {t:8d} – new best {cur_val:.6f}")

    # final check
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def annealed_hill_climb(
    seq,
    time_deadline,
    max_steps: int = 300_000,
    init_temp: float = 0.05,
    final_temp: float = 0.001,
):
    """
    Stochastic hill‑climber with a temperature schedule.
    Three move types are used:
        * random repartition of the mass of a pair,
        * transfer of a fraction of mass from i to j,
        * simple swap of two coordinates.
    Down‑hill moves are always accepted; uphill moves are accepted
    with Metropolis probability.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    cur_score = fast_score(a)

    best_a = a.copy()
    best_score = cur_score

    # re‑normalise every few hundred iterations to avoid drift
    renorm_interval = 1000

    for step in range(max_steps):
        if time.time() >= time_deadline - 0.5:
            break

        move_type = np.random.rand()
        cand = a.copy()

        if move_type < 0.1:
            # ---- random repartition of the mass of a pair ----
            i, j = np.random.randint(n, size=2)
            total = cand[i] + cand[j]
            split = np.random.rand()               # uniform Dirichlet(1,1)
            cand[i] = split * total
            cand[j] = (1.0 - split) * total
        elif move_type < 0.6:
            # ---- small transfer from i to j ----
            i, j = np.random.randint(n, size=2)
            if cand[i] == 0.0:
                # nothing to transfer
                continue
            frac = np.random.rand() * 0.2          # up to 20 % of source mass
            delta = frac * cand[i]
            cand[i] -= delta
            cand[j] += delta
        else:
            # ---- simple swap ----
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        # occasional renormalisation to keep the vector on the simplex
        if (step % renorm_interval) == 0:
            s = np.sum(cand)
            if s != 0.0:
                cand /= s

        cand_score = fast_score(cand)

        # exponential temperature decay
        T = init_temp * (final_temp / init_temp) ** (step / max_steps)

        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a.tolist(), best_score
