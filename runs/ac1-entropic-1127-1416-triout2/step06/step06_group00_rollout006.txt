def adam_optimize(start_a, time_deadline, max_iters=300000, beta_start=5.0, beta_end=200.0, step0=0.4):
    a = start_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    for it in range(1, max_iters+1):
        if time.time() >= time_deadline: break
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t  # exponential annealing
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()
        # compute gradient
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = project_to_simplex(a_new)
        a = a_new
        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score
```

"""
Optimise a non‑negative step‑function to minimise

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The objective is invariant to scaling, therefore we keep the vector
on the probability simplex (sum = 1).  The algorithm consists of four
stages and deliberately trades a little extra wall‑clock time for a
much tighter final bound:

1. **Initial pool** – generate several promising starts:
   * uniform (all entries equal);
   * a “U‑shaped” arcsine‐like distribution (large mass at the
     two ends);
   * sparse random vectors with Dirichlet‑distributed masses.

2. **Adam‑accelerated smoothed descent** – minimise a soft‑max
   approximation of `max(conv)`.  The smoothing temperature β is
   annealed from β≈5 to β≈200, the gradient is obtained via a
   convolution and the update follows the Adam scheme.  After each
   step the iterate is projected onto the simplex.

3. **Pure sub‑gradient phase** – with β at its maximum the weight
   vector becomes a (soft) Kronecker, i.e. the gradient coincides
   with a sub‑gradient of the true max‑objective.  An Adam update
   with a decaying stepsize continues to refine the solution.

4. **Stochastic hill‑climb** – finally a cheap mass‑transfer
   random walk (swaps or small transfers) is performed, always
   projecting back onto the simplex, to escape possible tiny
   plateaus.

All heavy work uses only NumPy (FFT‑based convolutions) and stays
well within the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x)=1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=float)


def arcsine_like_init(n: int) -> np.ndarray:
    """U‑shaped distribution ≈ 1 / sqrt(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def random_sparse_init(n: int) -> np.ndarray:
    """Random support of size ~[5%, 60%] with Dirichlet masses."""
    p = np.random.uniform(0.05, 0.6)
    k = max(2, int(round(p * n)))
    idx = np.random.choice(n, size=k, replace=False)
    vals = np.random.rand(k) + 1e-8
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑accelerated smoothed optimisation
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 200.0,
                         step0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max surrogate
        f_β(a) = (1/β) log Σ exp(β·conv_i)
    with Adam.  The smoothing parameter β is annealed exponentially.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed max via soft‑max weights
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        # avoid overflow/underflow
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:            # extreme under‑flow (should never happen)
            w = np.zeros_like(w)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient = 2·(a_rev ⊛ w) slice
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final exact evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑accelerated pure sub‑gradient phase
# ----------------------------------------------------------------------


def adam_subgrad_optimize(start_a: np.ndarray,
                          deadline: float,
                          max_iters: int = 200_000,
                          step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Adam on the true max‑objective.  The gradient is a sub‑gradient
    obtained from the index where the convolution attains its maximum.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient w.r.t. the max entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # occasional tiny random jitter to avoid stagnation
        if it % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps: int = 500_000):
    """
    Random local moves (swap or mass transfer) with simplex projection.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_frac = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_frac * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_frac = min(step_frac * 1.06, 0.6)
        else:
            step_frac *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score
