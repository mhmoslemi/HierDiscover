"""
Propose a low‑dimensional yet powerful candidate for the step‑function
optimization problem.

Algorithm Overview
------------------
1. **Diverse Seeding** – Generate a large pool of candidate vectors for
   three lengths (512, 1024, 2048).  Candidates come from several analytic
   families (arcsine, beta, power, triangular, mixtures) and the uniform
   vector.  All candidates are normalised to Σa = 1 and scored with a fast
   surrogate   2·n·max(conv(a,a)).

2. **Smooth‑max Adam Refinement** – The best few seeds are refined with an
   Adam optimiser that minimises a smooth approximation of the max‑convolution:
   Φτ(a)= (1/τ)·log Σk exp(τ·convk).  The temperature τ is increased from
   0.5 to 200, turning the surrogate gradually into the true max.  The
   optimisation variable is the soft‑max logits, guaranteeing non‑negativity
   and the simplex constraint.

3. **Exact Sub‑gradient / Exponentiated‑Gradient Polishing** – The top
   refined candidates are further improved with a projected sub‑gradient
   descent on the exact objective (max‑convolution) followed by a
   mirror‑descent (exponentiated‑gradient) step.  These stages dominate the
   remaining time budget.

4. **Global Polishing** – If time permits, the current best sequence is
   polished again with the exact methods.

All stages respect the global 1000 s deadline, frequently checking the
clock and keeping the best official score (via `evaluate_sequence`) found so
far.

The implementation is fully self‑contained, uses only NumPy and the
provided `evaluate_sequence`, and returns a list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities

def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no padding artefacts)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector a is assumed to satisfy Σa = 1, hence the denominator of the
    official objective disappears and the surrogate is simply 2·n·max(conv).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x≥0, Σx = 1}.
    Duchi et al. (2008).  Handles the degenerate case of all‑zero input.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric families for seeding
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√[(i+½)(n‑½‑i)]."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type shape (i+ε)^{α‑1}(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if tot <= 0 or not np.isfinite(tot):
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine a_i ∝ [(i+½)(n‑½‑i)+ε]^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if tot <= 0 or not np.isfinite(tot):
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at `peak`."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def upscale_shape(a_small: np.ndarray, n_target: int) -> np.ndarray:
    """
    Linear interpolation from a small vector to a larger one,
    followed by re‑normalisation to Σ=1.
    """
    n_small = a_small.shape[0]
    x_small = np.linspace(0, 1, n_small, endpoint=False)
    x_target = np.linspace(0, 1, n_target, endpoint=False)
    a_large = np.interp(x_target, x_small, a_small)
    if a_large.sum() == 0:
        a_large = np.full(n_target, 1.0 / n_target)
    else:
        a_large = a_large / a_large.sum()
    return a_large


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 30_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.25) -> np.ndarray:
    """
    Minimises Φτ(a)= (1/τ)·log Σk exp(τ·conv_k) using Adam on soft‑max logits.
    The temperature τ grows linearly from ``tau_start`` to ``tau_end``.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits (log‑softmax)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax to get a on the simplex
        w_shift = w - np.max(w)               # for stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max weights = softmax(tau·conv)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)   # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # back‑prop through softmax
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update with decreasing step size
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if it % 5000 == 0:
            surrogate = compute_score(a)
            print(f"    [Adam] it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected) and exponentiated‑gradient polish
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of indices attaining the max (within tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(a, mask_rev) restricted to original length
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.2) -> np.ndarray:
    """Exponentiated‑gradient (mirror‑descent) polishing on the exact objective."""
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return a

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine combines diverse random seeding, smooth‑max Adam optimisation,
    and exact sub‑gradient + exponentiated‑gradient polishing, all respecting
    the global 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # Optional fallback from a previous run
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # --------------------------------------------------------------
    # Stage 0 – generate a pool of diverse seeds for several lengths
    # --------------------------------------------------------------
    lengths = [512, 1024, 2048]
    pool: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Seeding length {n} ---")

        # deterministic landmarks
        uniform = np.full(n, 1.0 / n)
        pool.append((compute_score(uniform), uniform.copy()))
        arc = arcsine_initial(n)
        pool.append((compute_score(arc), arc.copy()))

        # random beta family
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            pool.append((compute_score(cand), cand))

        # random power family
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((compute_score(cand), cand))

        # mixtures of arcsine & uniform
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((compute_score(cand), cand))

        # triangular bumps
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

    # Keep the very best seeds (across all lengths) for the heavy optimisation.
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:max(8, len(pool) // 20)]]
    print(f"\nSelected {len(top_seeds)} top seeds for Adam refinement "
          f"(surrogate best = {pool[0][0]:.6f})")

    # --------------------------------------------------------------
    # Stage 1 – Smooth‑max Adam optimisation on each top seed
    # --------------------------------------------------------------
    polished_candidates: List[Tuple[float, np.ndarray]] = []

    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 4.0:
            break
        n = seed_vec.shape[0]
        print(f"\n=== Adam refinement seed {idx}/{len(top_seeds)} (n={n}) ===")

        # allocate a modest but flexible number of iterations
        remaining = deadline - time.time()
        max_iters = int(min(80_000, remaining * 30))   # approx 30 it / sec budget
        a_adam = adam_optimize_smooth_max(seed_vec,
                                          deadline,
                                          max_iters=max_iters,
                                          tau_start=0.5,
                                          tau_end=200.0,
                                          base_lr=0.25)

        surrogate = compute_score(a_adam)
        print(f"    Post‑Adam surrogate = {surrogate:.6f}")
        polished_candidates.append((surrogate, a_adam))

    # Merge Adam‑refined vectors with the original pool and keep a few best.
    all_candidates = pool + polished_candidates
    all_candidates.sort(key=lambda x: x[0])
    top_for_exact = [vec for _, vec in all_candidates[:5]]

    # --------------------------------------------------------------
    # Stage 2 – Exact refinement (sub‑gradient + EG) on the best candidates
    # --------------------------------------------------------------
    for idx, cand in enumerate(top_for_exact, start=1):
        if time.time() >= deadline - 5.0:
            break
        n = cand.shape[0]
        print(f"\n--- Exact refinement candidate {idx}/{len(top_for_exact)} (n={n}) ---")
        remaining = deadline - time.time()
        # allocate the bulk of the remaining budget to the very best candidate,
        # a smaller share to the others.
        if idx == 1:
            max_iter_sub = int(min(400_000, remaining * 25))
        else:
            max_iter_sub = int(min(150_000, remaining * 15))
        refined = subgradient_descent_exact(cand, deadline,
                                            max_iter=max_iter_sub,
                                            step0=0.12)

        # Exponentiated‑gradient polishing if time permits
        if time.time() < deadline - 1.5:
            it_eg = int(min(200_000, (deadline - time.time()) * 25))
            refined = eg_refine(refined, deadline, max_iter=it_eg, eta0=0.20)

        # Official evaluation (expensive but reliable)
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print("*** New global best! ***")
        except Exception as e:
            print(f"    Evaluation failed: {e}")

    # --------------------------------------------------------------
    # Stage 3 – Final global polish (if time remains)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\n--- Final global polish ---")
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        it_sub = int(min(300_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline, max_iter=it_sub, step0=0.09)

        if time.time() < deadline - 1.0:
            it_eg = int(min(200_000, (deadline - time.time()) * 30))
            a = eg_refine(a, deadline, max_iter=it_eg, eta0=0.18)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # Guard – ensure a non‑empty sequence is always returned
    # --------------------------------------------------------------
    if not best_seq:
        # fallback to a simple arcsine of moderate size
        fallback = arcsine_initial(1024)
        best_seq = fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback used, official score = {best_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
