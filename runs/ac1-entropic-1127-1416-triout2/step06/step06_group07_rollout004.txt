    best_score = float('inf')

    n = 4096
    if deadline - time.time() < 20:
        n = 2048

    print(f"Using n={n}")

    # Define possible initializations
    init_strategies = []

    # Uniform
    init_strategies.append(np.full(n, 1.0/n))

    # Arc sine shape (beta distribution)
    def beta_init(alpha):
        xs = (np.arange(n)+0.5)/n
        eps = 1e-12
        a = np.power(np.clip(xs, eps, 1-eps), alpha-1) * np.power(np.clip(1-xs, eps, 1-eps), alpha-1)
        a /= a.sum()
        return a

    # Add several beta shapes with random alphas
    for _ in range(3):
        alpha = np.random.uniform(0.2, 1.8)
        init_strategies.append(beta_init(alpha))

    # Random Dirichlet restarts (densities)
    for _ in range(2):
        alpha = np.random.uniform(0.2, 2.0)
        a = np.random.gamma(alpha, 1.0, size=n)
        a = a / a.sum()

    # For each init, perform Adam optimization

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
* Choose a relatively large length (default 4096).  The objective
  `evaluate_sequence` is scale‑invariant, so we work with a probability
  vector (sum = 1) and minimise the supremum of the self‑convolution.
* Initialise several candidates:
    – uniform,
    – a family of symmetric Beta‑distributions (U‑shaped when α<1),
    – a few random Dirichlet samples.
* For each start run an Adam optimiser on a smooth log‑sum‑exp
  approximation of the ∞‑norm:
        fβ(a) = (2·n/β)·log Σₖ exp(β·(a*a)[k]).
  The gradient is obtained by one extra FFT; the parameters are kept
  in log‑space (`y = log a`) and are normalised after each step.
* Keep the best vector seen during optimisation.
* Finally apply a simple stochastic hill‑climber (swap / mass‑transfer
  moves) for a few hundred thousand steps to polish the solution.
* Return the best sequence (any scaling is fine because the score is
  invariant under positive scaling).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast exact score for vectors that sum to 1
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a probability vector (Σa = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on the smooth (log‑sum‑exp) objective
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    time_deadline: float,
    max_iters: int = 150_000,
    beta0: float = 0.5,
    beta_max: float = 80.0,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    report_every: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the soft‑max approximation of max(a*a).

    Returns the best vector (still summing to 1) and its exact score.
    """
    n = a0.shape[0]
    L = next_pow2(3 * n - 2)

    # work in log‑space to keep positivity automatically
    y = np.log(a0 + 1e-30)
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)

    best_a = a0.copy()
    best_val = fast_score(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline - 1.0:
            break

        # ----- obtain current probability vector -----
        a = np.exp(y)
        a_sum = a.sum()
        if a_sum == 0.0:               # numerical safety
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ----- convolution a*a -----
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = np.max(conv)

        # ----- soft‑max weights -----
        beta = min(beta_max, beta0 * np.sqrt(t))
        # shift for numerical stability
        exp_terms = np.exp(beta * (conv - max_c))
        w = exp_terms / np.sum(exp_terms)

        # ----- gradient of the smooth objective (w.r.t a) -----
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        # slice contains the needed n components
        grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- convert to gradient w.r.t log‑variables -----
        grad_y = grad_a * a

        # ----- Adam update on y -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- record best -----
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val

    # final check
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val

# Simple stochastic hill‑climber (swap / mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Tiny hill‑climber that keeps the vector on the simplex.
    Uses fast_score for quick acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = fast_score(best_a)

    step_scale = 0.1  # fraction of mass moved on a transfer move

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two entries – sum unchanged, always feasible
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j (also keeps the sum)
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        # numeric safety: clip tiny negatives, renormalise
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / cand_sum

        cur_score = fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # enlarge step after success
        else:
            step_scale *= 0.96                         # shrink step after failure

        max_steps -= 1

    return best_a.tolist(), best_score
