def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                       # soft‑max weights
    B = max_c + np.log(sum_exp) / lam            # smooth approximation of the max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B

def l_bfgs_optimize(a0, deadline, maxfun=2000, lam=2000):
    n = a0.shape[0]
    # parameterize by t for softmax
    t0 = np.log(a0 + 1e-12)
    def obj_and_grad(t):
        # compute a via softmax
        exp_t = np.exp(t - np.max(t))
        a = exp_t / exp_t.sum()
        grad_a, B = smooth_gradient(a, lam)  # gradient of surrogate objective w.r.t a
        # compute gradient w.r.t t
        inner = np.dot(a, grad_a)
        grad_t = a * (grad_a - inner)
        # compute objective value: f = 2 * n * B (since sum=1)
        f = 2 * n * B
        return f, grad_t.astype(float)
    res = scipy.optimize.minimize(
        fun=lambda t: obj_and_grad(t)[0],
        x0=t0,
        jac=lambda t: obj_and_grad(t)[1],
        method='L-BFGS-B',
        options={'maxfun': maxfun, 'maxiter': maxfun}
    )
    t_opt = res.x
    exp_t = np.exp(t_opt - np.max(t_opt))
    a_opt = exp_t / exp_t.sum()
    # compute final score using fast_score
    return a_opt, fast_score(a_opt)
```

"""
Improved search for a low‑value step‑function sequence.

Main ideas
----------
* Sample many beta‑family shapes (including two‑component mixtures) on several
  lengths and keep a few survivors with the smallest *fast* score
  (2·n·max(conv(a,a)) which is the official objective for a normalized
  vector Σa=1).
* Refine each survivor with a high‑precision L‑BFGS optimisation of the
  smooth log‑sum‑exp surrogate of the max‑convolution.  The variables are
  stored in a soft‑max (logit) parametrisation so that the simplex constraints
  are satisfied automatically.
* Perform a short exact sub‑gradient descent to guarantee that the true
  objective never worsens.
* Keep the globally best vector and polish it once more with a longer
  L‑BFGS run followed by a final stochastic walk.
* All stages respect the global 1000 s budget and fall back to a uniform
  256‑vector if nothing is found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for all n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector is assumed to be normalised to Σa = 1, hence
    the denominator is 1 and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int,
    alpha: float,
    beta: float,
    eps: float,
) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density (used in the original baseline)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def sin_shape(n: int) -> np.ndarray:
    """Positive sinusoid – zero at the ends, maximal in the middle."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate of the max‑convolution and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the smooth log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value) where the gradient is taken w.r.t.
    the vector a (which is assumed to satisfy Σa>0).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)        # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    B = max_c + np.log(sum_exp) / lam    # smooth approximation of the max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def lbfgs_optimize(
    init_a: np.ndarray,
    deadline: float,
    maxfun: int = 1500,
    lam: float = 2000.0,
) -> Tuple[np.ndarray, float]:
    """
    L‑BFGS optimisation of the smooth surrogate.
    The variables are stored as logits (t) and projected to the simplex
    via a soft‑max, which guarantees non‑negativity and Σa=1.
    """
    n = init_a.shape[0]
    # initialise logits – a small offset avoids log(0)
    t0 = np.log(np.clip(init_a, 1e-12, None))

    try:
        import scipy.optimize as opt
    except Exception:  # fallback to Adam if SciPy is absent
        from math import inf
        return init_a, fast_score(init_a)

    def obj_and_grad(t: np.ndarray) -> Tuple[float, np.ndarray]:
        # a = softmax(t)
        max_t = np.max(t)
        exp_t = np.exp(t - max_t)
        a = exp_t / exp_t.sum()
        grad_a, B = smooth_gradient(a, lam)

        # chain rule: ∂f/∂t = J^T ∂f/∂a, where J is the Jacobian of soft‑max
        inner = np.dot(a, grad_a)                # scalar = aᵀ grad_a
        grad_t = a * (grad_a - inner)            # shape (n,)

        # surrogate objective (Σa=1 ⇒ denominator = 1)
        f_val = 2.0 * n * B
        return f_val, grad_t.astype(np.float64)

    # wrappers required by scipy.optimize.minimize
    def fun(t):
        return obj_and_grad(t)[0]

    def jac(t):
        return obj_and_grad(t)[1]

    # run the optimiser; we keep a very small tolerance to stop early
    options = {"maxfun": maxfun, "maxiter": maxfun, "disp": False}
    res = opt.minimize(fun, t0, jac=jac, method="L-BFGS-B", options=options)

    t_opt = res.x
    max_t = np.max(t_opt)
    exp_t = np.exp(t_opt - max_t)
    a_opt = exp_t / exp_t.sum()
    return a_opt, fast_score(a_opt)


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the official objective.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term (sub‑gradient)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        # simple back‑tracking line‑search
        for _ in range(12):
            a_new = a - step * grad
            # project onto simplex
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        best_val = official_score(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq


def random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Cheap Gaussian walk on the simplex.
    Occasionally accepts a small uphill move to escape plateaus.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = fast_score(a_prop)

        # accept if strictly better, otherwise with a tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50_000 == 0:
            print(
                f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc
