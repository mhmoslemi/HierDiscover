def simulated_hill_climb(a, conv, n, deadline, initial_temp, final_temp, max_steps=None):
    # a (np.ndarray), conv (np.ndarray), n len.
    # returns best_a, best_score.
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_score = 2.0 * n * cur_conv.max()
    best_a = cur_a.copy()
    best_score = cur_score

    T0 = initial_temp
    Tf = final_temp
    # steps approx (deadline - now) * steps_per_sec (estimated maybe 2000 steps per sec). We'll just loop while time < deadline.

    step_counter = 0
    accepted = 0
    while time.time() < deadline:
        step_counter += 1
        # degrade temperature:
        progress = (step_counter) / 1000000  # not accurate
        T = T0 * max(Tf / T0, 1.0 - progress)  # linear cooling to Tf

        # pick indices:
        i = np.random.randint(n)
        if cur_a[i] <= 0:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j += 1   # ensure j != i

        delta = cur_a[i] * np.random.rand() * 0.5   # step up to 0.5 of source mass
        # limit delta small for stability.
        # you could also limit to e.g., step_scale * cur_a[i]
        # use step_scale adaptively.
        # ensure delta > 0:
        if delta <= 0:
            continue

        a_old = cur_a.copy()  # O(n)
        # compute inc array
        inc = np.zeros_like(cur_conv)
        inc[i:i+n] -= 2 * delta * a_old
        inc[j:j+n] += 2 * delta * a_old
        # handle delta^2 terms
        idx2i = 2*i
        if idx2i < inc.shape[0]:
            inc[idx2i] += delta*delta
        idx2j = 2*j
        if idx2j < inc.shape[0]:
            inc[idx2j] += delta*delta
        idxij = i+j
        inc[idxij] -= 2 * delta*delta

        new_conv = cur_conv + inc
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max

        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / max(T, 1e-12)):
            # accept
            cur_a[i] -= delta
            cur_a[j] += delta
            cur_conv = new_conv
            cur_score = new_score
            accepted += 1
            if new_score < best_score:

def hill_climb_simulated_annealing(a, n, deadline):
    cur_a = a.copy()
    cur_conv = np.convolve(cur_a, cur_a)
    cur_score = 2.0 * n * cur_conv.max()
    best_a = cur_a.copy()
    best_score = cur_score
    steps = 0
    accepted = 0
    step_scale = 0.1  # fraction of source mass
    T0 = cur_score * 0.01  # 1% of current score
    Tf = cur_score * 0.0001  # final tiny temperature

    while time.time() < deadline:
        steps += 1
        # decaying temperature
        progress = steps / 500000  # arbitrarily assume up to 500k steps
        T = max(Tf, T0 * (1 - progress))
        # make sure T > 0
        i = np.random.randint(n)
        if cur_a[i] <= 1e-15:
            continue
        # pick j distinct
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # delta fraction
        delta = cur_a[i] * step_scale * np.random.rand()
        if delta <= 0:
            continue

        a_old = cur_a.copy()
        new_conv = cur_conv.copy()
        # apply updates
        new_conv[i:i+n] -= 2 * delta * a_old
        new_conv[j:j+n] += 2 * delta * a_old
        # squared terms
        idx2i = 2*i
        idx2j = 2*j
        idxij = i + j
        if idx2i < new_conv.shape[0]:
            new_conv[idx2i] += delta*delta
        if idx2j < new_conv.shape[0]:
            new_conv[idx2j] += delta*delta
        new_conv[idxij] -= 2 * delta*delta

        new_max = new_conv.max()
        new_score = 2.0 * n * new_max

        # acceptance
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / max(T, 1e-12)):
            # accept
            cur_a[i] -= delta
            cur_a[j] += delta
            cur_conv = new_conv
            cur_score = new_score
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"SA step {steps}: new best {best_score:.6f}")

        # adapt step_scale every few steps
        if steps % 5000 == 0:
            # recompute exact conv to reduce drift (optional)
            cur_conv = np.convolve(cur_a, cur_a)

            total = cur_a.sum()
            if abs(total - 1.0) > 1e-12:
                cur_a /= total

    return best_a, best_score

"""
Propose a low‑score step‑function using a three‑phase pipeline:

1.  Load the previously best known sequence (if available) and put it on the
    probability simplex.
2.  A short smooth‑soft‑max projected gradient descent (FFT based) to get a
    reasonably flat autocorrelation.
3.  A long simulated‑annealing hill climber that updates the convolution
    incrementally (O(n) per move) – this explores the non‑convex landscape
    far better than the original greedy hill‑climber.
4.  Optional symmetry averaging and a final exact evaluation.

The algorithm stays entirely in the simplex (sum = 1, non‑negative) so the
objective 2·n·max(conv) is exactly the value returned by
``evaluate_sequence`` (the scaling of the vector does not matter).
The search respects the given wall‑time budget (≈1000 s) and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv) for a vector of length n whose sum = 1."""
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth soft‑max gradient descent (for a quick “warm‑start”)
# ----------------------------------------------------------------------


def smooth_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 5000,
    step0: float = 0.5,
    eps_factor: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Minimise a smooth log‑sum‑exp surrogate of max‑convolution."""
    a = a0.astype(float).copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score_from_conv(convolution_fft(a, a), n)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = conv.max()
        eps = maxc * max(eps_factor * (0.99 ** (t - 1)), 0.01)

        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / exp_vals.sum()               # soft‑max weights, sum = 1
        w_rev = w[::-1]
        grad_rev = convolution_fft(a, w_rev)        # length 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * grad_rev[idx_start - np.arange(n)]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * grad)

        cur_sc = compute_score_from_conv(convolution_fft(a, a), n)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Simulated‑annealing hill‑climber with incremental convolution updates
# ----------------------------------------------------------------------


def simulated_annealing_hill_climb(
    a0: np.ndarray,
    deadline: float,
    initial_temp_factor: float = 0.001,
    final_temp_factor: float = 1e-6,
    max_window: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing on the exact objective.
    Moves transfer a random fraction of mass from one index to another.
    Convolution is updated incrementally (O(n) per move) to keep the search fast.
    """
    a = a0.astype(float).copy()
    n = a.size
    conv = np.convolve(a, a)          # exact linear convolution, length 2n‑1
    cur_score = compute_score_from_conv(conv, n)

    best_a = a.copy()
    best_score = cur_score

    # temperature schedule (linear in the number of accepted steps)
    T0 = cur_score * max(initial_temp_factor, 1e-12)
    Tf = cur_score * max(final_temp_factor, 1e-12)

    # temporary array reused each iteration to avoid repeated allocations
    tmp_conv = np.empty_like(conv)

    step_scale = 0.10                 # initial fraction of source mass to move
    accept_cnt = 0
    total_cnt = 0
    recent_accept = 0
    recent_total = 0

    while time.time() < deadline:
        total_cnt += 1
        recent_total += 1

        # pick source index with positive mass
        i = np.random.randint(n)
        if a[i] <= 1e-15:
            continue
        # pick distinct target index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount to move (up to step_scale of the source weight)
        delta = a[i] * step_scale * np.random.rand()
        if delta <= 0.0:
            continue

        # ---- incremental convolution update ---------------------------------
        # copy current convolution (fast using pre‑allocated buffer)
        tmp_conv[:] = conv

        # linear part: 2 * (a ⊛ Δ)   with Δ having -δ at i and +δ at j
        #   -> subtract 2·δ·a from slice i:i+n, add 2·δ·a to slice j:j+n
        tmp_conv[i : i + n] -= 2.0 * delta * a
        tmp_conv[j : j + n] += 2.0 * delta * a

        # quadratic part Δ⊛Δ
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        if idx2i < tmp_conv.size:
            tmp_conv[idx2i] += delta * delta
        if idx2j < tmp_conv.size:
            tmp_conv[idx2j] += delta * delta
        tmp_conv[idxij] -= 2.0 * delta * delta

        new_score = compute_score_from_conv(tmp_conv, n)

        # temperature (linear decay)
        progress = min(1.0, total_cnt / 300_000.0)     # assume we do ≤300k moves
        T = max(Tf, T0 * (1.0 - progress))

        # Metropolis acceptance
        if (new_score < cur_score) or (np.random.rand() < np.exp((cur_score - new_score) / max(T, 1e-12))):
            # accept
            a[i] -= delta
            a[j] += delta
            conv = tmp_conv.copy()
            cur_score = new_score
            accept_cnt += 1
            recent_accept += 1

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[SA] step {total_cnt:7d} – new best = {best_score:.6f}")

        # periodic housekeeping ------------------------------------------------
        if total_cnt % max_window == 0:
            # re‑normalise to compensate rounding errors
            s = a.sum()
            if abs(s - 1.0) > 1e-12:
                a /= s

            # recompute convolution from scratch to avoid drift
            conv = np.convolve(a, a)

            # adapt step scale to keep acceptance ratio ~30‑40%
            if recent_total > 0:
                acc_ratio = recent_accept / recent_total
                if acc_ratio > 0.45:
                    step_scale = min(step_scale * 1.07, 0.5)
                elif acc_ratio < 0.25:
                    step_scale = max(step_scale * 0.90, 1e-4)
            recent_accept = recent_total = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Symmetry averaging (often a cheap polish)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a+reverse(a))/2 and re‑project onto the simplex."""
    b = (a + a[::-1]) * 0.5
    return project_to_simplex(b)
