"""
Improved step‑function optimizer.

We minimise
        score(a) = 2·n·max(conv(a,a)) / (∑a)²
which is invariant to scaling, so we work on the probability simplex
(∑a_i = 1, a_i ≥ 0).  The objective on the simplex is
        F(a) = 2·n·max(conv(a,a))          (conv = linear convolution)

The optimiser consists of three parts

1️⃣  **Random restarts** – many Dirichlet starts with a random concentration.
2️⃣  **Adam on the soft‑max smoothed max** – we minimise a smooth surrogate
    f_λ(a) = (1/λ)·log∑_k exp(λ·conv_k) using the re‑parameterisation
    a = softmax(u).  The gradient of f_λ is
        g_a = 2·(a_rev ∗ w)                      (w = soft‑max weights)
    and the gradient w.r.t. the logits u is
        g_u = a ⊙ (g_a – ⟨g_a,a⟩)                (⊙ = element‑wise)
    Adam’s adaptive steps allow fast convergence; λ is increased
    gradually from a small value to a large one (≈200) so that the
    surrogate tightens to the true max.
3️⃣  **Fine‑grained hill‑climbing** – stochastic mass‑transfer moves
    (move a fraction of the probability mass from one index to another)
    that are accepted only when they improve the exact objective.

All work is performed inside the given time budget (≈1000 s).  The
best sequence seen is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b via a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation of the scaled objective (sum(a) = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate of the max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    lam_start: float = 1.0,
    lam_end: float = 200.0,
    lambda_ramp_iters: int = 30_000,
    base_lr: float = 0.10,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-12,
    eval_interval: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector `a` (∑a = 1) with Adam on the
    smoothed max‑convolution objective.

    Parameters
    ----------
    init_a : np.ndarray
        Starting point (already on the simplex).
    deadline : float
        UNIX time at which the optimisation must stop.
    lam_start / lam_end : float
        Soft‑max temperature schedule (slow → sharp).
    lambda_ramp_iters : int
        Number of iterations over which λ is increased linearly.
    base_lr : float
        Base learning rate for Adam (applied in log‑space).
    eval_interval : int
        How often the true (non‑smoothed) score is evaluated.

    Returns
    -------
    best_a, best_score
        The best vector encountered and its exact (non‑smoothed) score.
    """
    # work in the logit (unconstrained) space: a = softmax(u)
    u = np.log(init_a + eps)          # avoid log(0)
    n = init_a.shape[0]

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)

    best_a = init_a.copy()
    best_score = compute_score(best_a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---------- build the probability vector ----------
        # softmax with a stable shift
        u_shift = u - np.max(u)
        exp_u = np.exp(u_shift)
        a = exp_u / np.sum(exp_u)               # Σa = 1, a_i > 0

        # ---------- convolution ----------
        conv = convolution_fft(a, a)            # length 2n‑1

        # ---------- λ schedule ----------
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max(1, lambda_ramp_iters))

        # ---------- soft‑max weights ----------
        max_c = np.max(conv)
        # exp(lam * (conv - max_c)) is ≤ 1, safe from overflow
        exp_vals = np.exp(lam * (conv - max_c))
        w = exp_vals / np.sum(exp_vals)        # length 2n‑1, Σw = 1

        # ---------- gradient w.r.t. a ----------
        a_rev = a[::-1]                         # reversed a
        g_full = convolution_fft(a_rev, w)      # length 3n‑2
        # central n entries correspond to ∂/∂a_i of the smoothed objective
        g_a = 2.0 * g_full[n - 1 : n - 1 + n]

        # ---------- gradient w.r.t. logits u (softmax Jacobian) ----------
        dot = np.dot(g_a, a)                    # scalar ⟨g_a, a⟩
        g_u = a * (g_a - dot)                  # shape (n,)

        # ---------- Adam update ----------
        m_adam = beta1 * m_adam + (1.0 - beta1) * g_u
        v_adam = beta2 * v_adam + (1.0 - beta2) * (g_u ** 2)

        m_hat = m_adam / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        u -= step                               # gradient descent on u

        # ---------- periodic evaluation ----------
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}]  iter {it:,}  new best {best_score:.6f}")

    # final check (in case loop ends without evaluation)
    final_score = compute_score(best_a)
    if final_score < best_score:
        best_score = final_score
        best_a = best_a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climber (mass‑transfer moves) on the exact objective
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps: int = 800_000):
    """
    Stochastic hill‑climber that moves a fraction of probability mass
    from one index to another.  Moves are kept only when they improve
    the true objective (the exact `compute_score`).  The transfer
    fraction adapts to the recent success rate.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.15  # initial fraction of the source mass to move

    steps = 0
    while steps < max_steps and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # amount of probability that will be moved
        delta = step_scale * best_a[i]
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # ensure no negative entries and re‑normalise
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:          # extremely unlikely, but guard against division by zero
            continue
        cand /= cand_sum                # keep sum exactly 1

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            # increase step size a little – we are on a promising direction
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # failed move – shrink the step
            step_scale *= 0.98

        steps += 1

    return best_a.tolist(), best_score
