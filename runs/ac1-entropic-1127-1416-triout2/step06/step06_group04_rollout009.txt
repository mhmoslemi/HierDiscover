"""
Improved heuristic for minimizing 2·n·max(conv(a,a)) with Σa = 1.

The algorithm follows a multi‑phase scheme:

0️⃣  Warm‑start from the best known construction (``height_sequence_1``) or,
    if unavailable, from the classic arcsine Beta(0.5,0.5) density.

1️⃣  *Flattening* – minimise the L₂ distance of the self‑convolution to the
    flat target 1/(2n‑1) using Adam.  This step moves the mass towards a
    flatter spectrum without changing the optimisation target.

2️⃣  *Smooth‑max* – minimise a log‑sum‑exp surrogate of the true maximum.
    The surrogate is differentiable; its gradient is simply the
    convolution of ``a`` with the soft‑max weights of the convolution
    entries.  Crucially we **ignore the denominator derivative** because the
    sequence is always projected onto the probability simplex (Σa = 1),
    so the denominator is constant.  The temperature λ is annealed from
    1 to a large value (≈800) so that the surrogate approaches the true
    max.

3️⃣  *Exact sub‑gradient* – a few iterations of projected sub‑gradient descent
    on the true max‑convolution (the active index provides a sub‑gradient).

4️⃣  *Stochastic walk* – cheap Gaussian perturbations on the simplex with a
    tiny uphill‑acceptance probability to escape shallow local minima.

The three stages are performed for several candidate lengths
(512, 1024, 2048, 4096).  The best sequence found across all lengths is
polished once more with longer runs of the same stages.  All operations
respect the 1000 s global budget and fall back to a uniform 256‑vector if
no improvement is obtained.

The code is deliberately modular: every helper is a top‑level function,
no closures or lambdas are used, and all heavy loops check the remaining
time budget.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT‑based convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return float(np.max(_conv_fft(a, a)))


def _fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score for vectors already on the simplex:
    2·n·max(conv(a,a))  (Σa = 1 ⇒ denominator = 1).
    """
    n = a.shape[0]
    return 2.0 * n * _max_convolution(a)


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (beta families, arcsine)
# ----------------------------------------------------------------------


def _generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Unnormalised symmetric beta‑family density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def _arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (Beta(0.5,0.5)) density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Gradient of the L₂ “flattening” loss
# ----------------------------------------------------------------------


def _flatten_gradient(a: np.ndarray) -> (np.ndarray, float):
    """
    Gradient of ‖conv(a,a) – avg‖₂² where avg = 1/(2n‑1) and Σa = 1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)                 # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient = 4·(a ⊛ diff_rev)
    diff_rev = diff[::-1]
    conv_rev = _conv_fft(a, diff_rev)      # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def _adam_flatten(initial_a: np.ndarray,
                  deadline: float,
                  max_iter: int = 120_000,
                  lr0: float = 0.30) -> (np.ndarray, float):
    """Adam optimiser on the L₂ flattening loss."""
    a = _project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = _fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        grad, loss = _flatten_gradient(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = _project_to_simplex(a_new)

        cur_sc = _fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 20000 == 0:
            print(f"    Adam‑flatten it={it:,d} loss={loss:.2e} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient (ignoring denominator)
# ----------------------------------------------------------------------


def _smooth_max_gradient(a: np.ndarray, lam: float) -> (np.ndarray, float):
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)):
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv−max)).
    The gradient w.r.t. a is 2·(a ⊛ w_rev) where w are the soft‑max
    weights on the convolution entries.
    Returns (grad, B).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)                     # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max on the shifted values
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                     # weights sum to 1
    B = max_c + np.log(sum_exp) / lam

    # weighted convolution gradient
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)             # length 3n‑2
    L = 2 * n - 1
    grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, B


def _adam_smooth_max(initial_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 200_000,
                     lr0: float = 0.12,
                     lam_start: float = 1.0,
                     lam_end: float = 800.0,
                     noise_every: int = 1000,
                     noise_scale: float = 5e-5) -> (np.ndarray, float):
    """
    Adam on the smooth‑max surrogate B (ignoring the denominator term).
    The temperature λ is increased from ``lam_start`` to ``lam_end``.
    """
    a = _project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = _fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # quadratic annealing of λ (more aggressive than linear)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, B = _smooth_max_gradient(a, lam)

        # Adam update (learning‑rate does not decay with λ)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = _project_to_simplex(a_new)

        cur_sc = _fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam‑smooth it={it:,d} λ={lam:.1f} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max (projected)
# ----------------------------------------------------------------------


def _subgradient_descent(initial_a: np.ndarray,
                         deadline: float,
                         step0: float = 0.30,
                         max_iter: int = 150_000) -> (np.ndarray, float):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Simple back‑tracking line‑search guarantees non‑increase.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = _fast_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # Sub‑gradient of the active entry:
        # (grad)_i = 2·a_{idx‑i}  if 0 ≤ idx‑i < n else 0
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # back‑tracking line‑search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = _project_to_simplex(a_new)
            new_val = _max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = _fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(f"    Sub‑grad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap stochastic Gaussian walk (projected) with occasional uphill moves
# ----------------------------------------------------------------------


def _random_local_walk(initial_a: np.ndarray,
                       deadline: float,
                       init_step: float = 0.015,
                       max_iter: int = 250_000) -> (np.ndarray, float):
    """
    Gaussian walk on the simplex, accepting every improvement and a tiny
    fraction of worsening steps (to avoid getting stuck).
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = _fast_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = _project_to_simplex(a_proposed)

        cur_sc = _fast_score(a_proposed)

        # accept if better, or with a very small uphill probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(f"    Random walk it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine explores a few lengths, applies flattening → smooth‑max
    → exact sub‑gradient → stochastic walk, and finally performs a longer
    global polishing on the best candidate found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use the historic best construction if present (fallback)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Candidate lengths (trade‑off between expressivity & speed)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]

    # Allocate a modest share of the total time to each length.
    # We'll stop early if the budget runs out.
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # ---- 1a. Initialise from arcsine, keep a small set of random seeds ----
        seeds = [_arcsine_initial(n)]

        # Add a few random beta shapes (both symmetric and asymmetric)
        for _ in range(4):
            if time.time() >= deadline - 5.0:
                break
            alpha = rng.uniform(0.02, 3.5)
            beta = rng.uniform(0.02, 3.5)
            eps = rng.uniform(1e-8, max(0.01, 0.02 * n))
            cand = _generate_beta_shape(n, alpha, beta, eps)
            seeds.append(cand)

        # Pick the seed with the best *fast* score as starting point.
        cur_best = None
        cur_best_sc = float("inf")
        for s in seeds:
            sc = _fast_score(s)
            if sc < cur_best_sc:
                cur_best_sc = sc
                cur_best = s.copy()
        if cur_best is None:
            continue

        # ------------------------------------------------------------------
        # 2.  Stage A – flattening (moderate number of iterations)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        max_iter_flat = int(min(80_000, remaining * 15))
        print(f"  Flattening stage (max_iter={max_iter_flat}) …")
        a_flat, _ = _adam_flatten(
            initial_a=cur_best,
            deadline=deadline,
            max_iter=max_iter_flat,
            lr0=0.30,
        )

        # ------------------------------------------------------------------
        # 3.  Stage B – smooth‑max Adam (log‑sum‑exp surrogate)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        max_iter_smooth = int(min(240_000, remaining * 30))
        print(f"  Smooth‑max Adam stage (max_iter={max_iter_smooth}) …")
        a_smooth, _ = _adam_smooth_max(
            initial_a=a_flat,
            deadline=deadline,
            max_iter=max_iter_smooth,
            lr0=0.12,
            lam_start=1.0,
            lam_end=800.0,
            noise_every=1300,
            noise_scale=5e-5,
        )

        # ------------------------------------------------------------------
        # 4.  Stage C – exact sub‑gradient descent (few iterations)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        max_iter_sub = int(min(120_000, remaining * 20))
        print(f"  Sub‑gradient descent (max_iter={max_iter_sub}) …")
        a_sub, _ = _subgradient_descent(
            initial_a=a_smooth,
            deadline=deadline,
            step0=0.28,
            max_iter=max_iter_sub,
        )

        # ------------------------------------------------------------------
        # 5.  Stage D – stochastic local walk
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        max_iter_walk = int(min(200_000, remaining * 25))
        print(f"  Random walk polishing (max_iter={max_iter_walk}) …")
        a_final, _ = _random_local_walk(
            initial_a=a_sub,
            deadline=deadline,
            init_step=0.017,
            max_iter=max_iter_walk,
        )

        # ------------------------------------------------------------------
        # 6.  Official evaluation for this length
        # ------------------------------------------------------------------
        val = evaluate_sequence(a_final.tolist())
        print(f"  → official score for length {n} = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()
            print(f"  *** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 7.  Global polishing of the overall best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # Stage A – longer flattening
        remaining = deadline - time.time()
        max_iter_flat = int(min(180_000, remaining * 20))
        print(f"  Global flattening (max_iter={max_iter_flat}) …")
        a, _ = _adam_flatten(
            initial_a=a,
            deadline=deadline,
            max_iter=max_iter_flat,
            lr0=0.35,
        )

        # Stage B – high‑temperature smooth‑max Adam
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_smooth = int(min(350_000, remaining * 28))
            print(f"  Global smooth‑max Adam (max_iter={max_iter_smooth}) …")
            a, _ = _adam_smooth_max(
                initial_a=a,
                deadline=deadline,
                max_iter=max_iter_smooth,
                lr0=0.10,
                lam_start=1.0,
                lam_end=1200.0,
                noise_every=1500,
                noise_scale=4e-5,
            )

        # Stage C – sub‑gradient descent
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(180_000, remaining * 30))
            print(f"  Global sub‑gradient (max_iter={max_iter_sub}) …")
            a, _ = _subgradient_descent(
                initial_a=a,
                deadline=deadline,
                step0=0.22,
                max_iter=max_iter_sub,
            )

        # Stage D – final random walk
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_walk = int(min(250_000, remaining * 40))
            print(f"  Global random walk (max_iter={max_iter_walk}) …")
            a, _ = _random_local_walk(
                initial_a=a,
                deadline=deadline,
                init_step=0.012,
                max_iter=max_iter_walk,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 8.  Guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
