"""
Hybrid optimizer for the step‑function attack (v2).

Goal:
    minimise  score(a) = 2·n·max(conv(a,a)) / (sum(a))**2
which is invariant under scaling.  We therefore work on the simplex
(sum == 1, a_i ≥ 0) and minimise the surrogate 2·n·max(conv(a,a)).

Method (all stages run within the 1000 s budget)
------------------------------------------------
1️⃣  **Random restarts** – Dirichlet draws with a random concentration
    give diverse starts.
2️⃣  **Sub‑gradient descent on the true max‑conv objective**.
    At each iteration we locate the worst convolution shift k,
    compute the sub‑gradient g_i = 2·a_{k‑i} (zero outside the range)
    and perform a multiplicative update a ← a·exp(−η·g).  After each
    step the vector is projected back onto the probability simplex.
3️⃣  **Optional smoothed exponentiated‑gradient warm‑up** – a short
    run of the soft‑max surrogate (identical to the previous version)
    can help escape plateaus.
4️⃣  **Simulated‑annealing hill‑climber** – stochastic mass‑transfer
    moves are accepted with a Metropolis probability that gradually
    cools.  This fine‑tunes the solution and can pass shallow local
    minima.

The routine keeps the best sequence seen across all restarts,
prints intermediate scores and finally returns the best list of
floats.  All helper functions are top‑level; no I/O is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real FFT for speed)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT.
    The result length is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform point if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for `evaluate_sequence` when Σa == 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.7,
    eval_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative sub‑gradient descent on
        f(a) = max_k (a*a)[k]

    The gradient of the active shift k* is
        g_i = 2·a_{k*‑i}   (if the index is inside the vector, else 0).

    Returns the best vector encountered and its fast score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # convolution and index of the worst shift
        conv = convolution_fft(a, a)          # length 2n‑1
        max_val = np.max(conv)
        # choose a random index among the (possibly many) maximisers
        cand_inds = np.where(conv >= max_val * (1.0 - 1e-14))[0]
        k = np.random.choice(cand_inds)

        # sub‑gradient for this k
        g = np.zeros_like(a)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        g[idx] = 2.0 * a[k - idx]

        # step size (diminishing)
        eta = eta0 / np.sqrt(it)

        # multiplicative update with safety clipping
        step = -eta * g
        step = np.clip(step, -50, 50)          # avoid overflow in exp
        a = a * np.exp(step)

        # back onto the simplex
        a = project_to_simplex(a)

        # periodic evaluation of the true (non‑smoothed) score
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] subgrad iter {it:,}  new best {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent (warm‑up)
# ----------------------------------------------------------------------


def smooth_exp_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    lam_start: float = 1.0,
    lam_end: float = 30.0,
    eval_interval: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Same as the original implementation – a softmax surrogate of
    max(conv) is used, yielding a smooth gradient.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while True:
        if time.time() >= deadline:
            break
        it += 1

        conv = convolution_fft(a, a)                # length 2n‑1
        M = np.max(conv)

        # temperature schedule (λ) – slowly increasing
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / 2_000_000)

        # softmax weights (stable)
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)

        # gradient = 2 * (a_rev * w) central part
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)         # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        eta = eta0 / np.sqrt(it)
        step = -eta * g
        step = np.clip(step, -50, 50)
        a = a * np.exp(step)

        a = project_to_simplex(a)

        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}]  smooth iter {it:,}  new best {best_score:.6f}")

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (fine polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(seq, deadline, max_steps: int = 800_000,
                 init_temp: float = 0.05, final_temp: float = 1e-6) -> tuple[list, float]:
    """
    Stochastic hill‑climbing with a Metropolis acceptance rule.
    Moves transfer a random fraction of mass from i to j.
    Temperature decays linearly from `init_temp` to `final_temp`.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        t_frac = step / max_steps
        temp = init_temp * (1 - t_frac) + final_temp * t_frac

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose moving a random fraction (up to 20 %) of source mass
        delta = (np.random.rand() * 0.20) * a[i]
        if delta <= 0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # enforce feasibility
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        diff = cand_score - cur_score

        # Metropolis acceptance
        if diff < 0 or np.random.rand() < np.exp(-diff / max(temp, 1e-12)):
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a.tolist(), best_score
