"""
Improved step‑function optimiser.

The score is     S(a) = 2·n·max(conv(a,a)) / (sum(a))²
and is invariant under scaling, so we work on the probability simplex
  Σ a_i = 1 , a_i ≥ 0  and minimise  F(a)=2·n·max(conv(a,a)).

The optimiser consists of three stages (all respecting a global deadline):

1️⃣  **Smoothed exponentiated‑gradient descent** – a soft‑max over the
    convolution entries yields a smooth surrogate.
    The (multiplicative) mirror‑descent update
        a ← a·exp(−η·g)   (g = 2·(a_rev ∗ w))
    is projected back onto the simplex.
    λ (soft‑max temperature) is increased from ≈1 to ≈80,
    η decays as η₀/√t.

2️⃣  **Targeted sub‑gradient hill‑climbing** – using the true
    sub‑gradient of the non‑smooth max‐convolution
    (weights w are uniform over the active maximal shifts),
    we repeatedly move a small fraction of mass from the index with
    largest gradient (most responsible for the current peak) to the
    index with smallest gradient.  The step size adapts to the
    recent success rate.

3️⃣  **Final polishing** – a few thousand random mass‑transfer moves
    (the classic hill‑climber) are performed to catch any remaining
    improvements.

All phases keep the best sequence ever seen.  The code works for any
length `n` (the fallback `height_sequence_1` is used when present) and
conforms to the 1000 s time budget.
"""

import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real FFT, power‑of‑2 padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * b using a real FFT.
    Result length = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Fast version of `evaluate_sequence` for Σa = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # uniform point if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Stage 1 – Smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    lam_start: float = 1.0,
    lam_end: float = 80.0,
    eval_interval: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed surrogate
        f_λ(a) = log Σ_k exp( λ·conv_k(a) )
    The true gradient of the soft‑max surrogate is
        g = 2·(a_rev ∗ w)  (central n entries only)
    where w is the softmax weight vector.
    Returns the best vector seen and its true unsmoothed score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # full convolution a*a (length 2n‑1)
        conv = convolution_fft(a, a)
        M = np.max(conv)

        # λ increases linearly (more sharpening as iterations grow)
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / 2_000_000)

        # softmax weights – numerically stable
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)  # length 2n‑1, Σ w = 1

        # gradient (central n entries of a_rev * w)
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)          # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # step size schedule
        eta = eta0 / np.sqrt(it)

        # multiplicative mirror‑descent update
        step = -eta * g
        step = np.clip(step, -50, 50)               # protect exp()
        a = a * np.exp(step)

        # simplex projection
        a = project_to_simplex(a)

        # occasional evaluation of the exact (non‑smoothed) objective
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] exp‑GD iter {it:,} new best {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – Targeted sub‑gradient hill‑climbing
# ----------------------------------------------------------------------


def subgradient_hill_climb(
    init_seq,
    deadline: float,
    max_steps: int = 1_000_000,
    init_step: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a fraction of probability mass from the index with
    largest sub‑gradient (most responsible for the current peak) to the
    index with smallest sub‑gradient.  The move size adapts to recent
    successes.
    """
    a = np.array(init_seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = init_step
    steps = 0
    while steps < max_steps and time.time() < deadline:
        # current convolution and active max‑indices
        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        act = np.where(conv >= max_val - 1e-15)[0]   # active set (tight constraints)

        # construct sub‑gradient (average over active shifts)
        w = np.zeros_like(conv)
        w[act] = 1.0 / len(act)
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)
        g = 2.0 * g_full[n - 1 : n - 1 + n]          # length n

        i = int(np.argmax(g))   # index to give mass away from
        j = int(np.argmin(g))   # index to receive mass

        delta = step_scale * a[i]
        if delta <= 0:
            # nothing to move – shrink step and continue
            step_scale *= 0.95
            steps += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # projection back onto the simplex (ensures non‑negativity & Σ=1)
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand                                   # accept as current state
            step_scale = min(step_scale * 1.07, 0.5)    # become bolder
        else:
            # reject – become more cautious
            step_scale *= 0.97

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – Light random hill‑climber (fallback polish)
# ----------------------------------------------------------------------


def random_hill_climb(seq, deadline, max_steps: int = 300_000):
    """
    Very lightweight random mass‑transfer hill‑climber.
    Moves are accepted only when they lower the true objective.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.15
    steps = 0
    while steps < max_steps and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        cand = best_a.copy()
        delta = step_scale * best_a[i]
        if delta <= 0:
            continue
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.98

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point demanded by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine runs several random restarts of a
    smoothed exponentiated‑gradient descent, refines the current best
    with a targeted sub‑gradient hill‑climber and finishes with a brief
    random polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Determine sequence length (fallback if available)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]
    except Exception:
        n = 2048

    # ------------------------------------------------------------------
    # Initialise best sequence (use provided fallback if it exists)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    if best_seq is None:
        # uniform distribution on the simplex
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Phase 1 – multiple random restarts of smoothed exp‑gradient descent
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline - 30.0:   # keep ~30 s for later phases
        restart_counter += 1
        alpha = np.random.uniform(0.1, 5.0)
        a0 = np.random.dirichlet(np.full(n, alpha))

        # Allocate roughly 60 % of the remaining time to this run,
        # but always leave at least 20 s for the final polish.
        remaining = deadline - time.time()
        sub_deadline = time.time() + max(12.0, remaining * 0.6)

        print(f"[{time.time():.1f}] Restart {restart_counter}, α={alpha:.3f}")

        refined_a, refined_score = exp_gradient_optimize(
            a0,
            deadline=sub_deadline,
            eta0=0.55,
            lam_start=0.9,
            lam_end=80.0,
            eval_interval=300,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"  ► new global best after exp‑GD: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – targeted sub‑gradient hill‑climb from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        print("\nStarting targeted sub‑gradient hill‑climb")
        sub_deadline = deadline - 8.0
        polished_a, polished_val = subgradient_hill_climb(
            best_seq,
            deadline=sub_deadline,
            max_steps=800_000,
            init_step=0.12,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_a.tolist()
            print(f"  ► after sub‑gradient climb: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final random polishing (lightweight)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("\nFinal random hill‑climb polishing")
        polished_seq, polished_val = random_hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=300_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"  ► polishing improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety net – ensure we return a valid list
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # Final check (uses the official evaluator)
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq
