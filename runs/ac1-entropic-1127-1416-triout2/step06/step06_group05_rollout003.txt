"""
Search for a non‑negative step sequence that minimizes

    2·n·max(conv(a,a)) / (Σ a)²

The algorithm mixes several ideas:

1. **Multi‑resolution hierarchy** – start with short vectors, find a good
   shape, then up‑sample it to a longer length and continue refining.
2. **Population based exploration** – a small Differential Evolution (DE)
   population (projected onto the probability simplex) is evolved using a
   fast surrogate ``fast_score`` (the denominator is 1 because the vectors are
   normalised).  DE rapidly crawls the landscape and provides diverse good
   candidates.
3. **Smooth Adam phase** – a log‑sum‑exp surrogate of the max‑convolution is
   optimised with Adam.  The temperature λ is annealed up to a large value
   (≈500) so the surrogate becomes almost the true max.
4. **Exact sub‑gradient descent** – after Adam we switch to the true
   max‑convolution and perform a projected sub‑gradient descent with a
   back‑tracking line‑search; this guarantees that the official objective never
   increases.
5. **Final random walk** – a cheap Gaussian walk on the simplex (with rare
   uphill moves) fine‑tunes the result.
6. **Global polishing** – the best vector found over all lengths is again
   polished with the three stages above.

All stages respect the global time budget (default 1000 s) and the function
always returns a non‑empty list.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for all n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator is 1 and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (beta families, mixture, arcsine)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density (the shape used in the original baseline)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation up‑sampling of a simplex‑projected vector."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    return project_to_simplex(up)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (for Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # Stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                 # soft‑max weights
    B = max_c + np.log(sum_exp) / lam      # smooth approximation of the max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 400.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector found (projected on the simplex) together with
    its fast score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc

# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the true objective.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc

def random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Cheap Gaussian walk on the simplex.
    Occasionally accepts a small uphill move to escape plateaus.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = fast_score(a_prop)

        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

    return best_a, best_sc

def differential_evolution_optimize(
    init_pop: np.ndarray,
    deadline: float,
    max_generations: int = 2000,
    F: float = 0.8,
    CR: float = 0.9,
) -> Tuple[np.ndarray, float]:
    """
    Simple DE that works on vectors projected onto the probability simplex.
    Returns the best individual found and its fast score.
    """
    rng = np.random.default_rng()
    pop = init_pop.astype(float)
    popsize, n = pop.shape
    scores = np.array([fast_score(p) for p in pop])
    best_idx = np.argmin(scores)
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]

    for gen in range(max_generations):
        if time.time() >= deadline:
            break
        improved = False
        for i in range(popsize):
            # choose three distinct individuals different from i
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])

            # crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_simplex(trial)

            trial_score = fast_score(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                improved = True
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
        if not improved:
            break

    return best_vec, best_score

                )
            pop.append(cand)

        pop = np.array([project_to_simplex(p) for p in pop])
        # ------------------------------------------------------------------
        # 2.  Differential Evolution (fast surrogate)
        # ------------------------------------------------------------------
        local_best, local_best_sc = differential_evolution_optimize(
            init_pop=pop,
            deadline=deadline,
            max_generations=1500,
            F=0.8,
            CR=0.9,
        )
        print(f"  DE best fast score = {local_best_sc:.6f}")

        # ------------------------------------------------------------------
        # 3.  Adam polishing (smooth surrogate)
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_adam = int(min(600_000, remaining * 35))
            print(f"  Adam polishing (max_iter={max_iter_adam}) …")
            local_best, _ = adam_optimize(
                init_a=local_best,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.10,
                lam_start=1.0,
                lam_end=500.0,
                noise_every=1300,
                noise_scale=3e-5,
            )

        # ------------------------------------------------------------------
        # 4.  Exact sub‑gradient descent
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(250_000, remaining * 30))
            print(f"  Sub‑gradient descent (max_iter={max_iter_sub}) …")
            local_best, _ = subgradient_optimize(
                init_a=local_best,
                deadline=deadline,
                step0=0.22,
                max_iter=max_iter_sub,
            )

        # ------------------------------------------------------------------
        # 5.  Final random walk (fine‑tuning)
        # ------------------------------------------------------------------
        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            max_iter_rw = int(min(300_000, remaining * 35))
            print(f"  Random walk (max_iter={max_iter_rw}) …")
            local_best, _ = random_local_search(
                init_a=local_best,
                deadline=deadline,
                init_step=0.012,
                max_iter=max_iter_rw,
            )

        # ------------------------------------------------------------------
        # 6.  Official evaluation for this length
        # ------------------------------------------------------------------
        val = official_score(local_best)
        print(f"  → official score for length {n} = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = local_best.tolist()
            print(f"  *** New global best (len={n}) = {best_val:.6f} ***")

        prev_best = local_best.copy()   # seed next length

    # ------------------------------------------------------------------
    # 7.  Global polishing of the overall best sequence
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # A – long Adam run
        remaining = deadline - time.time()
        if remaining > 5.0:
            max_iter_adam = int(min(800_000, remaining * 40))
            print(f"  Adam polishing (max_iter={max_iter_adam}) …")
            a, _ = adam_optimize(
                init_a=a,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.09,
                lam_start=1.0,
                lam_end=500.0,
                noise_every=1300,
                noise_scale=2e-5,
            )

        # B – exact sub‑gradient descent
        remaining = deadline - time.time()
        if remaining > 5.0:
            max_iter_sub = int(min(300_000, remaining * 30))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                init_a=a,
                deadline=deadline,
                step0=0.20,
                max_iter=max_iter_sub,
            )

        # C – final stochastic walk
        remaining = deadline - time.time()
        if remaining > 5.0:
            max_iter_rw = int(min(400_000, remaining * 45))
            print(f"  Random walk polishing (max_iter={max_iter_rw}) …")
            a, _ = random_local_search(
                init_a=a,
                deadline=deadline,
                init_step=0.010,
                max_iter=max_iter_rw,
            )

        final_val = official_score(a)
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 8.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = official_score(best_seq)
        print("Fallback to uniform 256‑vector, official score =", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
