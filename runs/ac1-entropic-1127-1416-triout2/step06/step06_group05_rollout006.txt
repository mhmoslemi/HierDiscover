"""
propose_candidate
-----------------

A multi‑stage stochastic optimiser that searches for a non‑negative step
sequence (a “height” vector) with a low official score

    2·n·max(conv(a,a)) / (∑a)² .

The algorithm works in three phases

1. **Exploratory restarts** – for a chosen length (default 2048) a set of
   random initial vectors is drawn from several families (uniform,
   arcsine, beta‑shapes, sparse Dirichlet‑like vectors, mixtures).  Each
   start is polished by a relatively long Adam optimisation of the
   smooth log‑sum‑exp surrogate of the max‑convolution.  The best few
   survivors are retained.

2. **Exact polishing** – the survivors undergo a projected sub‑gradient
   descent on the true max‑convolution, followed by a cheap stochastic
   Gaussian walk that occasionally accepts uphill moves.  These stages
   guarantee that the official objective never increases.

3. **Global refinement** – the current global best vector is run through
   a final, very long Adam pass (λ up to 1000 so the surrogate is almost
   the true max), another sub‑gradient phase and a final random walk.

All stages respect the overall 1000 s budget and keep the best sequence
found (according to the official ``evaluate_sequence``).  If nothing
valid is produced a uniform fallback of length 256 is returned.
"""

import time
from typing import List, Tuple
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for arbitrary lengths)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of *a*."""
    return float(np.max(convolution_fft(a, a)))


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator is 1 and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    {x≥0 , Σx = 1} (Duchi et al. 2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (beta families, arcsine, sparse vectors)
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """Beta‑family density a_i ∝ (i+ε)^{α-1}·(n-1-i+ε)^{β-1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family shapes."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density (the shape used in the baseline)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_sparse_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random sparse vector: pick a random number of positions
    (between 5 % and 40 % of *n*) and distribute the mass uniformly.
    """
    frac = rng.uniform(0.05, 0.40)
    k = max(2, int(frac * n))
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def sample_initial_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random starting vector for optimisation.
    The sampler mixes several families to give a diverse search space.
    """
    p = rng.random()
    if p < 0.15:
        # uniform
        return np.full(n, 1.0 / n)
    if p < 0.35:
        # arcsine
        return arcsine_shape(n)
    if p < 0.65:
        # pure beta shape
        alpha = rng.uniform(0.02, 2.5)
        beta = rng.uniform(0.02, 2.5)
        eps = rng.uniform(1e-9, max(0.01, 0.02 * n))
        return generate_beta_shape(n, alpha, beta, eps)
    if p < 0.85:
        # beta mixture
        w = rng.random()
        alpha1 = rng.uniform(0.02, 2.5)
        beta1 = rng.uniform(0.02, 2.5)
        eps1 = rng.uniform(1e-9, max(0.01, 0.02 * n))
        alpha2 = rng.uniform(0.02, 2.5)
        beta2 = rng.uniform(0.02, 2.5)
        eps2 = rng.uniform(1e-9, max(0.01, 0.02 * n))
        return generate_beta_mixture(
            n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
        )
    # otherwise a sparse Dirichlet‑like vector
    return generate_sparse_vector(n, rng)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (for Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # soft‑max weights over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    B = max_c + np.log(sum_exp) / lam            # smooth approximation of max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 800.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # anneal λ quadratically (smooth → hard)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress report (does not affect timing)
        if it % 50_000 == 0:
            print(
                f"    [Adam] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the official objective.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term (ordered pairs)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step size decaying like 1/sqrt(t)
        step = step0 / np.sqrt(it)

        # back‑tracking line‑search: ensure a genuine decrease of the max
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40_000 == 0:
            print(
                f"    [Sub‑grad] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap stochastic local walk (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Cheap Gaussian walk on the simplex.
    Small uphill moves are accepted with a tiny probability to escape
    plateaux.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = fast_score(a_prop)

        # accept if better, otherwise with a tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50_000 == 0:
            print(
                f"    [Walk] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine performs many short Adam restarts, keeps a few promising
    survivors, refines them by exact sub‑gradient descent and a final
    stochastic walk, and finally runs a long global polishing pass.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[Fallback] official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Focused optimisation on a single length (2048) – many restarts
    # ------------------------------------------------------------------
    length = 2048
    print(f"\n=== Exploring length {length} with many restarts ===")

    # Upper bound on the number of restarts based on remaining time.
    # Empirically one restart (≈30 s) consumes ~60 k Adam iterations.
    time_per_restart = 30.0
    restarts = max(1, int((deadline - time.time()) // time_per_restart))
    restarts = min(restarts, 20)  # do not exceed a reasonable count

    survivors: List[Tuple[float, np.ndarray]] = []

    for r in range(restarts):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n-- Restart {r + 1}/{restarts} --")
        init = sample_initial_vector(length, rng)

        # --------------------------------------------------------------
        # 1a. Short Adam polishing (fast but still respects constraints)
        # --------------------------------------------------------------
        max_iter_adam = 60_000
        a_adam, _ = adam_optimize(
            init_a=init,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.10,
            lam_start=1.0,
            lam_end=800.0,
            noise_every=1200,
            noise_scale=3e-5,
        )
        val_adam = evaluate_sequence(a_adam.tolist())
        print(f"    Adam official score = {val_adam:.6f}")

        # Keep the best few across restarts
        survivors.append((val_adam, a_adam))
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.tolist()
            print(f"    *** New global best (restart) = {best_val:.6f} ***")

        # keep only the top 3 survivors to limit later work
        survivors.sort(key=lambda x: x[0])
        survivors = survivors[:3]

    # ------------------------------------------------------------------
    # 2.  Exact polishing of the best survivors
    # ------------------------------------------------------------------
    for rank, (cur_val, cur_vec) in enumerate(survivors, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Polishing survivor #{rank} (score {cur_val:.6f}) ---")

        # Sub‑gradient descent on the true max‑convolution
        a_sub, _ = subgradient_optimize(
            cur_vec,
            deadline,
            step0=0.30,
            max_iter=150_000,
        )
        val_sub = evaluate_sequence(a_sub.tolist())
        print(f"    After sub‑gradient score = {val_sub:.6f}")
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.tolist()
            print(f"    *** New global best (sub‑grad) = {best_val:.6f} ***")

        # Final random walk
        a_walk, _ = random_local_search(
            a_sub,
            deadline,
            init_step=0.014,
            max_iter=200_000,
        )
        val_walk = evaluate_sequence(a_walk.tolist())
        print(f"    After walk score = {val_walk:.6f}")
        if val_walk < best_val:
            best_val = val_walk
            best_seq = a_walk.tolist()
            print(f"    *** New global best (walk) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best sequence
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # A – long Adam run with a very high temperature (λ≈1000)
        remaining = deadline - time.time()
        max_iter_adam = int(min(800_000, remaining * 40))
        print(f"    Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.09,
            lam_start=1.0,
            lam_end=1000.0,
            noise_every=1300,
            noise_scale=2e-5,
        )

        # B – exact sub‑gradient descent
        if time.time() < deadline - 4.0:
            max_iter_sub = int(min(250_000, (deadline - time.time()) * 30))
            print(f"    Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                deadline,
                step0=0.25,
                max_iter=max_iter_sub,
            )

        # C – final stochastic walk
        if time.time() < deadline - 2.0:
            max_iter_loc = int(min(350_000, (deadline - time.time()) * 45))
            print(f"    Random walk polishing (max_iter={max_iter_loc}) …")
            a, _ = random_local_search(
                a,
                deadline,
                init_step=0.010,
                max_iter=max_iter_loc,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # safe uniform fallback (length 256)
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
