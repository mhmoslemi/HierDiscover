"""
Propose a non‑negative step sequence with a small official score.

The algorithm works as follows

1. Choose a target length (default 2048, later 4096 if time permits).
2. Generate a modest pool of random shapes (beta families, mixtures,
   Dirichlet draws, the classic arcsine and the uniform vector).  Keep the
   few best according to the fast surrogate 2·n·max(conv) (the denominator
   is 1 because the vectors are projected onto the simplex).
3. Refine each survivor with a long Adam run on a smooth log‑sum‑exp
   surrogate of the max‑convolution.  The temperature λ is annealed up
   to a very large value (≈2000) so the surrogate is almost the true
   max.
4. Switch to true sub‑gradient descent on the exact max‑convolution with
   a back‑tracking line‑search, guaranteeing that the official objective
   never increases.
5. Perform a cheap Gaussian walk (with occasional uphill moves) for a
   final fine‑tuning.
6. Keep the globally best vector seen (according to the official
   ``evaluate_sequence``) and, if time permits, repeat the whole
   refinement on this best vector with even more aggressive Adam/L‑BFGS
   style parameters.

All stages respect the 1000 s budget; a simple uniform fallback is
returned if nothing better is found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for any length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector is assumed to be normalised to Σa = 1,
    therefore the denominator is 1 and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine distribution (used by the original baseline)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the smooth log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value) where the objective is
        f(a) = 2·n·B / (∑a)²    with   B ≈ max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                       # soft‑max weights
    B = max_c + np.log(sum_exp) / lam            # smoothed max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 400.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam‑style optimiser on the smooth surrogate.
    Returns the best vector (projected onto the simplex) seen and its
    fast_score (2·n·max(conv) with Σa=1).
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # quadratic annealing of the temperature λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 50_000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution) with back‑tracking
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the official objective.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step size that decays like 1/√t
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40_000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap stochastic Gaussian walk (with occasional uphill moves)
# ----------------------------------------------------------------------


def random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Simple random walk on the simplex. Accepts a small probability of
    uphill moves to avoid becoming trapped.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = fast_score(a_prop)

        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50_000 == 0:
            print(
                f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine concentrates on a single length (2048, then 4096 if time
    permits), generates a pool of promising shapes, refines each of them
    with a long Adam run (high temperature → accurate surrogate),
    follows up with exact sub‑gradient descent and a stochastic walk,
    and finally performs a global polishing of the best vector found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Fallback to a previously discovered construction, if any.
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Main optimisation loop (first 2048, then maybe 4096).
    # --------------------------------------------------------------
    target_lengths = [2048, 4096]  # second length only if time remains
    for n in target_lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # ---- 1a. Sampling stage -------------------------------------------------
        top_k = 5
        top_vecs: List[np.ndarray] = []
        top_scores: List[float] = []

        # a few deterministic starters
        for starter in (np.full(n, 1.0 / n), arcsine_shape(n)):
            sc = fast_score(starter)
            top_vecs.append(starter.copy())
            top_scores.append(sc)

        max_samples = 12_000
        for _ in range(max_samples):
            if time.time() >= deadline - 30.0:
                break

            # random shape selector
            selector = rng.integers(0, 5)
            if selector == 0:  # simple beta
                alpha = rng.uniform(0.02, 1.5)
                beta = rng.uniform(0.02, 1.5)
                eps = rng.uniform(1e-9, max(0.01, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)
            elif selector == 1:  # mixture of two betas
                w = rng.random()
                a1 = rng.uniform(0.02, 1.5)
                b1 = rng.uniform(0.02, 1.5)
                e1 = rng.uniform(1e-9, max(0.01, 0.02 * n))
                a2 = rng.uniform(0.02, 1.5)
                b2 = rng.uniform(0.02, 1.5)
                e2 = rng.uniform(1e-9, max(0.01, 0.02 * n))
                cand = generate_beta_mixture(
                    n, a1, b1, e1, a2, b2, e2, w
                )
            elif selector == 2:  # Dirichlet draw (sparse)
                conc = rng.uniform(0.05, 1.0)
                gam = rng.gamma(conc, 1.0, size=n)
                if gam.sum() == 0 or not np.isfinite(gam.sum()):
                    cand = np.full(n, 1.0 / n)
                else:
                    cand = gam / gam.sum()
            elif selector == 3:  # convex combo arcsine–uniform
                t = rng.random()
                cand = t * arcsine_shape(n) + (1.0 - t) * np.full(n, 1.0 / n)
                cand = cand / cand.sum()
            else:  # random perturbation of a previously good vector (if any)
                if top_vecs:
                    base = rng.choice(top_vecs)
                    delta = rng.normal(scale=0.02, size=n)
                    cand = base + delta
                    cand = project_to_simplex(cand)
                else:
                    cand = np.full(n, 1.0 / n)

            sc = fast_score(cand)
            # keep only the top_k vectors
            if len(top_vecs) < top_k:
                top_vecs.append(cand.copy())
                top_scores.append(sc)
            else:
                worst = int(np.argmax(top_scores))
                if sc < top_scores[worst]:
                    top_vecs[worst] = cand.copy()
                    top_scores[worst] = sc

        # ---- 1b. Refine each survivor -----------------------------------------
        for idx, vec in enumerate(top_vecs):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Refining survivor {idx + 1}/{len(top_vecs)}")
            # a) long Adam run with a very high temperature
            remaining = deadline - time.time()
            max_iter_adam = int(min(800_000, remaining * 35))
            a_ref, _ = adam_optimize(
                init_a=vec,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.09,
                lam_start=1.0,
                lam_end=2000.0,
                noise_every=1000,
                noise_scale=2e-5,
            )
            # b) exact sub‑gradient descent
            if time.time() < deadline - 5.0:
                max_iter_sub = int(min(300_000, (deadline - time.time()) * 30))
                a_ref, _ = subgradient_optimize(
                    a_ref,
                    deadline,
                    step0=0.28,
                    max_iter=max_iter_sub,
                )
            # c) final random walk
            if time.time() < deadline - 3.0:
                max_iter_rw = int(min(300_000, (deadline - time.time()) * 35))
                a_ref, _ = random_local_search(
                    a_ref,
                    deadline,
                    init_step=0.012,
                    max_iter=max_iter_rw,
                )
            # official evaluation
            val = evaluate_sequence(a_ref.tolist())
            print(f"    → official score for survivor {idx + 1} = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_ref.tolist()
                print(f"    *** New global best (len={n}) = {best_val:.6f} ***")

    # --------------------------------------------------------------
    # 2. Global polishing of the current best vector (if any)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # A – very long Adam with an extremely high temperature
        remaining = deadline - time.time()
        max_iter_adam = int(min(1_200_000, remaining * 45))
        a, _ = adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.07,
            lam_start=1.0,
            lam_end=3000.0,
            noise_every=800,
            noise_scale=1e-5,
        )

        # B – aggressive sub‑gradient descent
        if time.time() < deadline - 4.0:
            max_iter_sub = int(min(400_000, (deadline - time.time()) * 30))
            a, _ = subgradient_optimize(
                a,
                deadline,
                step0=0.22,
                max_iter=max_iter_sub,
            )

        # C – final random walk with very small steps
        if time.time() < deadline - 2.0:
            max_iter_rw = int(min(400_000, (deadline - time.time()) * 45))
            a, _ = random_local_search(
                a,
                deadline,
                init_step=0.008,
                max_iter=max_iter_rw,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # 3. Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
