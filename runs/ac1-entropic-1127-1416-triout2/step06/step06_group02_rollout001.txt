assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where \((a*a)_k\) is the linear convolution of the sequence with itself.  
Because the denominator is homogeneous of degree 2, we can restrict the search to the **probability simplex**  
\(\sum_i a_i = 1\); the returned list is therefore directly comparable to `evaluate_sequence`.

The code implements a **multi‑phase stochastic optimiser**:

1. **Initialisation** – a mixture of sparse random vectors and smooth Gaussian‑shaped vectors.
2. **Adam‑based smoothed optimisation** – we minimise a smooth approximation of the
   \(\max\) operator  

   \[
   F_\varepsilon(a)=\varepsilon\log\!\Bigl(\sum_k e^{(a*a)_k/\varepsilon}\Bigr) .
   \]

   The gradient is obtained with a single extra FFT, and an Adam optimiser
   (biased‑first‑moment / RMS‑prop) updates the vector while it is continuously
   projected onto the simplex.
3. **Exact sub‑gradient descent** – once the smooth surrogate is close to the true
   maximum we switch to the genuine sub‑gradient (the gradient of the active
   convolution shift) for fine tuning.
4. **Targeted hill‑climbing** – the shift that attains the current maximum is
   identified; mass is transferred away from the pair of indices that contribute
   most to this shift.  Only improving moves are kept.
5. **Symmetry averaging** – averaging the vector with its reverse often reduces
   the peak of the autocorrelation without breaking feasibility.

The whole routine runs as many independent restarts as the time budget (≈ 1000 s)
allows, each restart using a different length (`n ∈ {1024, 2048, 4096, 8192}`).
The best sequence found over all restarts is finally evaluated with the provided
`evaluate_sequence` and returned.

The optimiser is fully deterministic apart from the initial random seed, uses only
NumPy and the standard library, and respects the 1‑CPU constraint.

```python
"""
Multi‑phase optimiser for the step‑function score.

1. Random / Gaussian initialisation on the simplex.
2. Adam optimisation of a soft‑max (log‑sum‑exp) surrogate of max‑convolution.
3. Exact sub‑gradient descent on the true max‑convolution.
4. Targeted hill‑climbing that steals mass from the most dominant pair of indices.
5. Optional symmetric averaging.

All vectors stay on the probability simplex (non‑negative, sum = 1).  The routine
runs as many independent restarts as the 1000 s budget permits and finally
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a ⊛ b using real FFT (full length)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact score for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    Guarantees non‑negativity and sum(v) = 1.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_sparse_vector(n: int) -> np.ndarray:
    """
    Random probability vector of length n.
    A random subset (density 0.1–0.9) receives equal mass.
    """
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def gaussian_vector(n: int, sigma_factor: float = 0.15) -> np.ndarray:
    """
    Smooth symmetric vector with a Gaussian shape, normalised to sum = 1.
    """
    sigma = sigma_factor * n
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - n / 2) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 2 – Adam on a smooth soft‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_opt(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 5000,
    lr: float = 0.05,
    eps_factor: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max‑convolution.

    The surrogate is
        F_ε(a) = ε·log Σ_k exp((a*a)_k / ε) .
    Its gradient is
        g = 2·(w_rev ⊛ a) ,    w_i = softmax( (a*a)_i / ε ).

    The vector is kept on the simplex after each update.
    Returns the best exact score seen during the run.
    """
    a = a0.astype(float).copy()
    n = len(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # epsilon shrinks but never below 0.5 % of maxc
        eps = maxc * max(eps_factor * (0.98 ** (t - 1)), 0.005)

        # soft‑max weights (stable)
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)

        # gradient of the surrogate
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)                # length = 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to escape plateaus
        if t % 600 == 0:
            jitter = np.random.randn(n) * (lr * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on the exact objective 2·n·max(a⊛a)."""
    a = init_a.astype(float).copy()
    n = len(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k-i} where the index is inside the array
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # small random perturbation every 1000 iterations
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc

def hill_climb_targeted(
    seq,
    deadline,
    max_steps: int = 300_000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber that always steals a tiny fraction of mass from
    the index participating most heavily in the current maximal convolution shift.
    Only strictly improving moves are kept.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    # initial transfer magnitude (as a fraction of the donor mass)
    transfer_frac = 0.08

    while max_steps > 0 and time.time() < deadline:
        # current maximal shift
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # contributions of each index to that shift
        j = k - np.arange(n)
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j[mask]]

        # choose donor index proportional to its contribution
        total_c = contrib.sum()
        if total_c > 0:
            probs = contrib / total_c
            donor = np.random.choice(n, p=probs)
        else:
            donor = np.random.randint(n)

        # Choose a receiver that is far enough not to increase the same shift
        # (just pick a random index that is not the donor)
        receiver = np.random.randint(n)
        while receiver == donor:
            receiver = np.random.randint(n)

        delta = transfer_frac * best_a[donor]
        if delta <= 0:
            max_steps -= 1
            continue

        cand = best_a.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        cand = project_to_simplex(cand)

        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            transfer_frac = min(transfer_frac * 1.07, 0.30)   # be bolder
        else:
            transfer_frac *= 0.94                               # be cautious

        max_steps -= 1

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a + reverse(a))/2 and project back onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
```
