```python
def exponentiated_gradient_polish(a0: np.ndarray, deadline: float, max_iter: int = 50000, step: float = 0.01) -> np.ndarray:
    """
    Mirror descent on the original objective using KL divergence,
    i.e. multiplicative updates a_i ← a_i * exp(-η * subgrad_i) and renormalise.
    """
    a = np.clip(a0, 1e-15, None)  # avoid zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for active constraints
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # multiplicative update
        a = a * np.exp(-step * subgrad)
        # renormalize to simplex
        a_sum = a.sum()
        if a_sum <= 0 or not np.isfinite(a_sum):
            # fallback to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # optional periodic logging
        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a
```

"""
Propose a step function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Diverse seeding** – generate a small pool of candidates for each
   length (4096, 2048, 1024) using analytic shapes (arcsine, power,
   beta families, mixtures) and, if available, the previously best
   known sequence (`height_sequence_1`).

2. **Smooth‑max Adam phase** – each seed is refined with Adam on a
   convex surrogate of the max‑convolution:
        max(conv) ≈ (1/β)·log sum exp(β·conv).
   The temperature β is increased linearly from 5 up to 200 so the
   optimisation first moves on a smooth landscape and then sharpens
   the true objective.

3. **Mirror‑descent polishing** – the candidate from Adam is further
   improved by KL‑mirror (exponentiated‑gradient) descent directly on
   the original non‑smooth objective.  The update is
        a_i ← a_i·exp(−η·subgrad_i)  and a is re‑normalised to the simplex.
   This respects the non‑negativity/normalisation constraints
   automatically.

4. **Exact sub‑gradient polishing** – a final projected sub‑gradient
   descent (the same routine used in the reference solution) is run
   for a long horizon to squeeze any remaining improvement.

5. **Selection** – the best sequence seen according to the official
   ``evaluate_sequence`` is kept and returned.  The routine respects
   a 1000 s wall‑clock budget and always returns a non‑empty valid
   sequence.

The combination of a smooth surrogate, an entropy‑based mirror step,
and a final exact sub‑gradient pass gives a better chance of escaping
local minima and reaching a lower bound than the prior implementation.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (no circular wrap)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the objective when the vector already sums to 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Generators for diverse initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like envelope, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑distribution‑like shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (inverse of a polynomial envelope)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Projection utilities for the simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.05,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # identify all near‑maximal entries (numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a

# Smooth max surrogate + Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 120.0,
    ramp_iters: int = 20000,
    lr: float = 0.05,
) -> np.ndarray:
    """
    Minimise the smooth log‑sum‑exp surrogate of max(conv) using Adam.
    The iterates are kept on the simplex via the soft‑max map.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                       # a ≥ 0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linearly raise the temperature β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth max via log‑sum‑exp weighting
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2 * (a ⋆ w_rev) (restricted to original length)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # length n

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final vector on the simplex
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# KL‑mirror (exponentiated‑gradient) polishing on the true objective
# ----------------------------------------------------------------------


def exponentiated_gradient_polish(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 50_000,
    step: float = 0.02,
) -> np.ndarray:
    """
    Mirror descent on the original max‑convolution objective using the
    KL divergence (multiplicative updates).  The iterate stays on the simplex.
    """
    a = np.clip(a0, 1e-15, None)   # avoid exact zeros
    a = a / a.sum()
    n = a.size
    L = 2 * n - 1
    it = 0

    while it < max_iter and time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for the set of active (maximal) indices
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient step
        a = a * np.exp(-step * subgrad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a
