def adam_optimize(init_a, deadline, eta0=0.3, beta1=0.9, beta2=0.999, eps=1e-8, lam_start=1.0, lam_end=30.0, eval_interval=500):
    a = init_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        M = np.max(conv)
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / 500000)  # schedule
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)
        g = 2.0 * g_full[len(a)-1 : len(a)-1 + len(a)]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = - eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a + step
        a = project_to_simplex(a)
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam iter {it:,} best {best_score:.6f}")
    # final check
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score

def simulated_hill_climb(a, deadline, max_iters=500000):
    n = len(a)
    cur_a = a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule: start high, end low
    T0 = cur_score * 0.05  # relative scale
    Tmin = 1e-6
    start_time = time.time()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # linear schedule based on elapsed proportion
        progress = (time.time() - start_time) / (deadline - start_time)
        T = max(Tmin, T0 * (1 - progress))

        # move: transfer mass from i to j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # choose delta as uniform fraction of source or based on temperature
        max_transfer = cur_a[i]
        if max_transfer <= 1e-12:
            continue
        # sample fraction in [0, min(step_scale, 1)] maybe adapt
        fraction = np.random.rand() * 0.5  # up to 0.5 of mass
        delta = fraction * max_transfer
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure numerical sum = 1 - may drift, project to simplex for safety
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        delta_score = cand_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / T):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # Optionally print progress
                if it % 5000 == 0:
                    print(f"[{time.time():.1f}] SA iter {it:,} new best {best_score:.6f}")
    return best_a.tolist(), best_score

```
def lp_step(f, M, deadline):
    n = len(f)
    # Build constraint matrix A where A[k,i] = f[i] if 0 <= k-i < n else 0
    # Build as dense matrix for simplicity (2n-1) x n
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        # column i contributes f[i] to rows i .. i+n-1
        A[i:i+n, i] = f[i]
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # inequality constraints: A @ b <= M * np.ones(rows)
    res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 1e-12:
        return None
    g = b / sum_b  # normalized to sum 1
    return g
```

"""
Hybrid optimizer for the step‑function attack.

The score to minimise is
    score(a) = 2·n·max(conv(a,a)) / (sum(a))**2 .
Because the expression is homogeneous we restrict ourselves to the
probability simplex Σa_i = 1, a_i ≥ 0 and minimise the simpler
objective 2·n·max(conv(a,a)).

The routine consists of five phases that run within the 1000 s budget:

1️⃣ **Quick parameter sweep** – a few analytically‑motivated families
   (geometric, sinusoidal, two‑spike) are tested; the best of them is taken
   as an initial point.

2️⃣ **Adam‑based soft‑max gradient descent** – a smooth surrogate of the
   max‑convolution is optimised with an Adam optimiser and a projection
   onto the simplex after every step.

3️⃣ **LP‑refinement** – a linear programme solves
        max Σb_j   s.t.  f * b ≤ M ,
   where f is the current vector, * is linear convolution and M is the
   current worst convolution value.  The solution is mixed with f.

4️⃣ **Simulated‑annealing hill‑climbing** – stochastic mass‑transfer moves
   are accepted with a Metropolis rule, allowing the search to escape
   shallow local minima.

5️⃣ **Final polishing** – a longer annealing run on the best vector found.

All helpers are defined at top level, the code uses only NumPy,
SciPy and the provided `evaluate_sequence`.  The best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b via a real FFT.
    Result length = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective and simplex projection
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Fast evaluation for a simplex‑normalised vector (Σa = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameter sweep (geometric, sinusoidal, two‑spike)
# ----------------------------------------------------------------------


def param_search(n: int, deadline: float):
    """Search a few low‑dimensional families; return best vector and its score."""
    best_seq = np.full(n, 1.0 / n)
    best_val = compute_score(best_seq)

    # 1) geometric (truncated exponential)
    for r in np.linspace(0.1, 0.99, 30):
        a = (1.0 - r) * (r ** np.arange(n))
        total = a.sum()
        if total == 0.0:
            continue
        a /= total
        val = compute_score(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[param] geom r={r:.3f}  score={val:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist(), best_val

    # 2) half‑sine with a power
    x = np.arange(n) + 0.5
    for p in [0.5, 1, 2, 3, 4, 5]:
        a = np.sin(np.pi * x / n) ** p
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[param] sin p={p}  score={val:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist(), best_val

    # 3) two‑spike family
    for L in range(100, n, max(100, n // 20)):
        for w in np.linspace(0.01, 0.99, 20):
            a = np.zeros(n)
            a[0] = w
            a[L] = 1.0 - w
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[param] two‑spike L={L} w={w:.2f}  score={val:.6f}")
            if time.time() >= deadline:
                return best_seq.tolist(), best_val

    return best_seq.tolist(), best_val


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate of max(conv)
# ----------------------------------------------------------------------


def adam_optimize(init_a: np.ndarray,
                  deadline: float,
                  eta0: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  lam_start: float = 1.0,
                  lam_end: float = 30.0,
                  eval_interval: int = 300):
    """Adam optimiser of 2·n·max(conv) using a soft‑max smoothing."""
    a = init_a.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)                     # length 2n‑1
        M = np.max(conv)

        # temperature for the soft‑max (increases slowly)
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / 500000)

        # soft‑max weights
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)                 # sum‑to‑1, length 2n‑1

        # gradient of the smooth surrogate
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)               # length 3n‑2
        g = 2.0 * g_full[len(a) - 1 : len(a) - 1 + len(a)]

        # Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = -eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a + step
        a = project_to_simplex(a)

        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[adam] iter {it:,}  best {best_score:.6f}")

    # final check
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP refinement (maximise Σb  s.t. f*b ≤ M)
# ----------------------------------------------------------------------


# cache for sparse matrix pattern (identical for a given n)
_LP_cache = {}


def _lp_pattern(n: int):
    """Return (row_ind, col_ind, rows) pre‑computed for the convolution pattern."""
    if n not in _LP_cache:
        rows = 2 * n - 1
        # each column i contributes the value f[i] to rows i … i+n‑1
        row_ind = np.concatenate([np.arange(i, i + n) for i in range(n)])
        col_ind = np.repeat(np.arange(n), n)
        _LP_cache[n] = (row_ind, col_ind, rows)
    return _LP_cache[n]


def lp_step(f: np.ndarray, M: float):
    """
    Solve   max Σb_j   s.t.   f * b ≤ M ,  b ≥ 0.
    Returns a normalised vector g (Σg = 1) or None on failure.
    """
    n = f.shape[0]
    row_ind, col_ind, rows = _lp_pattern(n)

    data = np.repeat(f, n)                     # f[i] repeated n times
    A = csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))

    c = -np.ones(n)                            # minimise –Σb  ⇔ maximise Σb
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 1e-12:
        return None
    g = b / sum_b
    return g


def alternating_lp_optimize(init_f: np.ndarray,
                            deadline: float,
                            step_size: float = 0.4,
                            max_iter: int = 12):
    """
    Alternating refinement: solve the LP, mix the solution with the current
    vector and repeat.  Returns the best vector encountered.
    """
    f = init_f.astype(float).copy()
    best_f = f.copy()
    best_score = compute_score(f)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = convolution_fft(f, f)
        M = np.max(conv)

        g = lp_step(f, M)
        if g is None:
            break

        # convex combination and projection
        f = (1.0 - step_size) * f + step_size * g
        f = project_to_simplex(f)

        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[LP] iter {it:,}  best {best_score:.6f}")

    return best_f, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber
# ----------------------------------------------------------------------


def simulated_hill_climb(start_a: np.ndarray,
                         deadline: float,
                         max_steps: int = 500_000):
    """
    Stochastic mass‑transfer moves accepted with a Metropolis rule.
    The temperature decays linearly to a very small value.
    """
    n = start_a.shape[0]
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    start_time = time.time()
    # initial temperature proportional to the current score
    T0 = max(1e-6, cur_score * 0.02)
    Tmin = 1e-7

    steps = 0
    while steps < max_steps and time.time() < deadline:
        steps += 1
        # linear temperature schedule
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if cur_a[i] <= 1e-12:
            continue

        # move a random fraction (up to 50 %) of the source mass
        frac = np.random.rand() * 0.5
        delta = frac * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # projection to keep a valid point on the simplex
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        delta_score = cand_score - cur_score

        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / T):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                if steps % 5000 == 0:
                    print(f"[SA] step {steps:,}  new best {best_score:.6f}")

    return best_a.tolist(), best_score
