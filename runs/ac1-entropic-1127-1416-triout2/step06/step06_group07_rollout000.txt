def compute_grad(a, beta, L=None):
    n = len(a)
    # compute conv = a*a length 2n-1 using FFT (padded length L)
    L = next_pow2(3*n -2) # length maybe 4*n? Use same earlier.

    A_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(A_fft*A_fft, n=L)[:2*n-1]
    max_c = np.max(conv_full) # maybe for stability
    exp_terms = np.exp(beta*(conv_full - max_c))
    w = exp_terms/np.sum(exp_terms)
    # compute gradient
    a_rev = a[::-1]
    A_rev_fft = np.fft.rfft(a_rev, n=L)
    W_fft = np.fft.rfft(w, n=L)
    grad_full = np.fft.irfft(A_rev_fft*W_fft, n=L)
    grad = 2.0*grad_full[n-1:n-1+n]
    return grad, max_c

def hill_climb(seq, deadline, max_steps=300000):
    a = np.array(seq)
    best_score = fast_score(a)
    best_a = a.copy()
    cur_score = best_score
    T0 = 0.1  # starting temperature relative to score difference
    for step in range(max_steps):
        if time.time() >= deadline:
            break
        # generate candidate
        cand = a.copy()
        # choose move type
        move_type = np.random.rand()
        n = len(cand)
        if move_type < 0.4:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.8:
            # transfer mass
            i, j = np.random.randint(n, size=2)
            if cand[i] > 1e-12:
                delta = np.random.rand() * cand[i] * 0.3 # up to 30% of i's mass
                cand[i] -= delta
                cand[j] += delta
        else:
            # small gaussian noise redistribution among 3 indices
            idx = np.random.choice(n, size=3, replace=False)
            total = cand[idx].sum()
            # add small random perturbation preserving sum
            perturb = np.random.randn(3) * 0.01
            perturb = perturb - np.mean(perturb)
            new_vals = np.maximum(total * (cand[idx]/total + perturb), 0)
            # renormalize to be same total mass
            new_vals = new_vals * (total/ new_vals.sum())
            cand[idx] = new_vals
        # project to simplex - but swaps and transfers preserve sum maybe need to enforce nonneg.
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()  # ensure sum=1
        # compute score
        cand_score = fast_score(cand)
        # acceptance
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        else:
            # accept with probability exp(-(cand_score - cur_score)/T)
            # temperature schedule
            T = T0 * (1 - step / max_steps)
            if T > 0 and np.random.rand() < np.exp(-(cand_score - cur_score) / T):
                a = cand
                cur_score = cand_score
    return best_a.tolist(), best_score

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. Choose a length (2048 by default, 4096 if enough time remains).
2. Run several random Dirichlet restarts.
   For each restart:
      * Initialise a probability vector a (sum = 1).
      * Minimise a smooth soft‑max approximation of the max‑convolution
        using an Adam‑style optimiser on the simplex.
        The temperature β grows like √t up to a large β_max,
        so the objective converges to the true ∞‑norm.
      * Keep the best vector seen during this run.
3. Polish the best vector with a simulated‑annealing hill‑climber that
   performs swaps, mass‑transfer and three‑point redistribution moves.
4. Return the best sequence (any scaling works; we keep sum = 1).
All work respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT, output length len(a)+len(b)-1."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective (sum = 1) – matches evaluate_sequence
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a probability vector a."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------


def gradient_softmax(a: np.ndarray, beta: float, L_fft: int) -> tuple[np.ndarray, float]:
    """
    Compute gradient of the smooth objective
        fβ(a) = (2·n/β)·log Σ exp(β·(a*a)[k])
    and also return the current maximum convolution value.
    """
    n = a.shape[0]

    # ----- a*a --------------------------------------------------------
    A = np.fft.rfft(a, n=L_fft)
    conv_full = np.fft.irfft(A * A, n=L_fft)[: 2 * n - 1]
    max_c = np.max(conv_full)

    # ----- soft‑max weights -------------------------------------------
    # stabilize by subtracting max_c
    exp_terms = np.exp(beta * (conv_full - max_c))
    w = exp_terms / np.sum(exp_terms)

    # ----- gradient = 2·(w ⋆ a_rev) ------------------------------------
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=L_fft)
    W = np.fft.rfft(w, n=L_fft)
    grad_full = np.fft.irfft(A_rev * W, n=L_fft)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]

    return grad, max_c


# ----------------------------------------------------------------------
# Adam‑style optimisation on the simplex
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    beta0: float = 0.5,
    beta_max: float = 80.0,
    alpha0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max approximation using Adam on the simplex.
    Returns the best vector encountered and its exact score.
    """
    n = a0.shape[0]
    a = a0.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    L_fft = next_pow2(3 * n - 2)  # length sufficient for all FFTs
    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline - 0.5:
            break

        # temperature schedule
        beta = min(beta_max, beta0 * np.sqrt(t))

        # gradient of the smooth loss
        grad, _ = gradient_softmax(a, beta, L_fft)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha0 / np.sqrt(t)   # decreasing step size
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project back to the simplex
        a = project_to_simplex(a)

        # evaluate exact score every few thousand iterations
        if t % 10_000 == 0 or t == max_iters:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"    Adam iter {t:7d} – new best score {cur_val:.6f}")

    # final check
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber
# ----------------------------------------------------------------------


def hill_climb_anneal(
    seq: list[float],
    deadline: float,
    max_steps: int = 250_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climber with occasional uphill moves (SA).
    Moves:
        * swap two entries
        * transfer a fraction of mass
        * 3‑point redistribution
    """
    a = np.array(seq, dtype=float)
    a /= a.sum()                # ensure exact simplex
    n = a.shape[0]

    best_a = a.copy()
    best_score = fast_score(best_a)

    cur_score = best_score
    cur_a = a.copy()

    # initial temperature (scaled to typical score differences)
    T0 = 0.02 * best_score

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        r = np.random.rand()

        if r < 0.4:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.8:                   # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0:
                frac = np.random.rand() * 0.4      # up to 40 % of i’s mass
                delta = frac * cand[i]
                cand[i] -= delta
                cand[j] += delta
        else:                           # 3‑point redistribution
            idx = np.random.choice(n, size=3, replace=False)
            total = cand[idx].sum()
            if total > 0:
                perturb = np.random.randn(3) * 0.02
                perturb -= perturb.mean()
                new_vals = np.maximum(total * (cand[idx] / total + perturb), 0.0)
                # renormalise the three entries to keep their total
                if new_vals.sum() > 0:
                    new_vals = new_vals * (total / new_vals.sum())
                    cand[idx] = new_vals

        # re‑normalise to the simplex (tiny numerical drift)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum

        cand_score = fast_score(cand)

        # acceptance rule
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # temperature linearly decays
            T = T0 * (1.0 - step / max_steps)
            if T > 0 and np.random.rand() < np.exp(-(cand_score - cur_score) / T):
                accept = True

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                # occasional log
                if step % 20_000 == 0:
                    print(f"    SA step {step:7d} – improved to {cand_score:.6f}")

    return best_a.tolist(), best_score

    best_seq = None

    for r in range(restarts):

        if init_score < best_val:
            best_val = init_score
            best_seq = a0.tolist()
            print(f"Restart {r}: init score = {init_score:.6f}")

            deadline - 5.0,

            beta0=0.5,
            beta_max=80.0,
            alpha0=0.25,
        )

            best_val = opt_score
            best_seq = opt_a.tolist()
            print(f"Restart {r}: Adam best score = {opt_score:.6f}")

    # ---- final polishing with simulated annealing -------------------------
    if best_seq is not None and time.time() < deadline - 5.0:

    # ----------------------------------------------------------------------

    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
