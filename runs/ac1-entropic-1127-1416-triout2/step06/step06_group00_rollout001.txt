"""
Propose a low‑score non‑negative step‑function.

Algorithm outline
-----------------
1. **Coarse search on a tiny problem** (n≈256).  
   Many random sparse starts are optimised with a smoothed
   log‑sum‑exp formulation of ``max(conv(a,a))``.
   The cheap size lets us perform dozens of restarts.

2. **Hierarchical up‑sampling**.  
   The best tiny vector is interpolated to a larger size
   (first 1024, then 4096).  Each up‑sampled version is refined
   again with the same smoothed optimiser and a short projected
   sub‑gradient phase.

3. **Final polishing** – a stochastic hill‑climb that repeatedly
   swaps entries or moves a small fraction of mass between two
   positions, always projecting back onto the simplex.

All optimisation respects the simplex constraint (non‑negative,
sum = 1), hence the evaluation ``evaluate_sequence`` is invariant
to scaling.  The routine is time‑aware: it stops before the 1000 s
budget expires and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b using real FFTs."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a ⊛ a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp optimiser
# ----------------------------------------------------------------------


def smooth_optimize(start_a: np.ndarray,
                    deadline: float,
                    max_iters: int = 150_000,
                    beta_start: float = 5.0,
                    beta_end: float = 80.0,
                    step0: float = 0.6,
                    momentum: float = 0.0) -> tuple[np.ndarray, float]:
    """
    Minimise (1/β)·log Σ exp(β·conv_i) by projected gradient descent.
    Optional momentum (β‑scaled) can be activated.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    # momentum buffer (same shape as a)
    vel = np.zeros_like(a) if momentum > 0 else None

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- anneal β -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- smoothed objective & softmax weights --------------------
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()        # softmax, length 2n‑1

        # ----- gradient (cross‑correlation) ----------------------------
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1: n - 1 + n]

        # ----- step size & momentum ------------------------------------
        step = step0 / np.sqrt(it)
        if momentum > 0:
            vel = momentum * vel - step * grad
            a_new = a + vel
        else:
            a_new = a - step * grad

        a = project_to_simplex(a_new)

        # ----- keep best exact score -----------------------------------
        if it % 250 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    # final check
    final = compute_score(a)
    if final < best_score:
        best_score, best_a = final, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = start_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: 2·a shifted by the maximizing index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a = project_to_simplex(a_new)

        # keep best
        cur = compute_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()

        # occasional jitter for escaping plateaus
        if t % 2000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    # final check
    final = compute_score(a)
    if final < best_score:
        best_score, best_a = final, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq,
               deadline: float,
               max_steps: int = 400_000) -> tuple[list[float], float]:
    """Random swaps / mass transfers with simplex projection."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12  # initial fraction moved

    for _ in range(max_steps):
        if time.time() >= deadline:
            break

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur = compute_score(cand)

        if cur < best_score:
            best_score = cur
            best_a = cand
            step_scale = min(step_scale * 1.04, 0.5)
        else:
            step_scale *= 0.97

    return best_a.tolist(), best_score

# Helper utilities

def random_sparse_vector(n: int, min_density=0.05, max_density=0.45) -> np.ndarray:
    """Create a random sparse probability vector of length n."""
    p = np.random.uniform(min_density, max_density)
    k = max(2, int(round(p * n)))
    idx = np.random.choice(n, size=k, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / k
    return project_to_simplex(vec)


def upsample(seq: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation up‑sampling to *target_n* points,
    followed by projection onto the simplex.
    """
    cur_n = seq.shape[0]
    if cur_n == target_n:
        return seq.copy()
    x_old = np.arange(cur_n)
    x_new = np.linspace(0, cur_n - 1, target_n)
    up = np.interp(x_new, x_old, seq)
    return project_to_simplex(up)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises
    ``evaluate_sequence``.  The routine caps its wall‑clock usage
    at *budget_s* seconds and returns the best list of floats found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. fallback to a previously discovered sequence, if available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. coarse optimisation on a tiny problem (n≈256)
    # ------------------------------------------------------------------
    n_small = 256
    print(f"\n--- Stage 1: tiny optimisation (n={n_small}) ---")
    tiny_deadline = deadline - 300.0
    if tiny_deadline > time.time() + 30:
        best_small = None
        best_small_score = float("inf")
        # allocate a fixed number of restarts (cheap)
        for restart in range(30):
            if time.time() >= tiny_deadline:
                break
            a0 = random_sparse_vector(n_small)
            a_opt, sc = smooth_optimize(
                a0,
                deadline=tiny_deadline,
                max_iters=120_000,
                beta_start=4.0,
                beta_end=70.0,
                step0=0.7,
                momentum=0.0,
            )
            # quick sub‑gradient polish (very cheap)
            a_ref, sc_ref = subgrad_optimize(
                a_opt,
                deadline=tiny_deadline,
                max_iter=100_000,
                step0=0.05,
            )
            if sc_ref < best_small_score:
                best_small_score = sc_ref
                best_small = a_ref
                print(f"  Tiny restart {restart+1}: new best {best_small_score:.6f}")

        if best_small is not None:
            best_small = project_to_simplex(best_small)
    else:
        best_small = None

    # ------------------------------------------------------------------
    # 2. hierarchical up‑sampling & refinement (1024 → 4096)
    # ------------------------------------------------------------------
    # target lengths we would like to reach (increase as time permits)
    target_lengths = [1024, 2048, 4096]
    current_vec = best_small if best_small is not None else random_sparse_vector(256)

    for target_n in target_lengths:
        if time.time() >= deadline - 150:
            break   # not enough time for another up‑sample stage
        print(f"\n--- Stage 2: up‑sample to {target_n} ---")
        # up‑sample current best to the new size
        a_up = upsample(current_vec, target_n)

        # smooth optimisation on the new size
        stage_deadline = deadline - 100   # keep reserve for later stages
        a_opt, scr = smooth_optimize(
            a_up,
            deadline=stage_deadline,
            max_iters=180_000,
            beta_start=5.0,
            beta_end=90.0,
            step0=0.55,
            momentum=0.4,
        )
        # sub‑gradient refinement
        a_ref, scr_ref = subgrad_optimize(
            a_opt,
            deadline=deadline - 50,
            max_iter=250_000,
            step0=0.07,
        )
        # keep the best seen so far
        if scr_ref < best_val:
            best_val = scr_ref
            best_seq = a_ref.tolist()
            print(f"  New global best after refinement (n={target_n}): {best_val:.6f}")

        # prepare for next up‑sample (if any)
        current_vec = a_ref

    # ------------------------------------------------------------------
    # 3. final hill‑climb polishing on the current best (or fallback)
    # ------------------------------------------------------------------
    if best_seq is None:
        # as a last resort use a modest uniform vector
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    print("\n--- Stage 3: final hill‑climb ---")
    polished_seq, polished_score = hill_climb(
        best_seq,
        deadline=deadline,
        max_steps=500_000,
    )
    if polished_score < best_val:
        best_val = polished_score
        best_seq = polished_seq
        print(f"Improved by hill‑climb: final score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. final evaluation & return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    return best_seq
