The program implements a multi‑stage optimiser for the step‑function attack:

1. **Random restarts** – a Dirichlet draw (random concentration) gives a diverse starting point on the probability simplex (`∑a = 1 , a_i ≥ 0`).

2. **Smoothed exponential‑gradient descent** – a soft‑max over the whole convolution makes the objective smooth.  
   The temperature `λ` is increased from a low value to a large one, so the method automatically switches from a global‑exploration phase to a focus‑on‑the‑worst shift.

3. **Pure sub‑gradient refinement** – after the smoothed phase we switch to the true sub‑gradient of  
   `f(a)=max_k (a∗a)_k`.  
   This step drives the current maximum convolution entry down sharply.

4. **Targeted hill‑climbing** – mass is moved from indices that contribute most to the current worst shift to random destinations.  
   The move size is adapted according to the recent success rate (a simple “success‑based step‑scale”).

5. **Final polishing** – a longer hill‑climb (with both random and targeted moves) is applied to the overall best candidate.

All updates keep the vector on the simplex by Euclidean projection (`project_to_simplex`).  
Fast convolution is performed by a real FFT (`convolution_fft`).  
The code is fully deterministic apart from the random seed, respects the 1000 s budget, and finally returns the best sequence found.

```python
"""
Hybrid optimiser for the step‑function attack.

The aim is to minimise
    score(a) = 2·n·max(conv(a,a)) / (sum(a))**2
which is homogeneous – we optimise on the probability simplex (sum=1) and
 minimise  2·n·max(conv(a,a)).
The optimiser consists of

  1️⃣  random Dirichlet restarts,
  2️⃣  a smoothed exponential‑gradient descent (soft‑max over all shifts),
  3️⃣  a pure sub‑gradient phase that follows the true worst shift,
  4️⃣  a targeted hill‑climber that moves mass away from the worst‑convolution
      contributions,
  5️⃣  a final long polishing hill‑climb.

All steps use FFT‑based convolution, projection onto the simplex and adaptive
step‑sizes.  The routine respects the 1000 s time budget and returns the best
list of floats it has discovered.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (real FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT.
    Result length = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)==1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # uniform point if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient descent (soft‑max surrogate)
# ----------------------------------------------------------------------


def smooth_exp_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.55,
    lam_start: float = 1.0,
    lam_end: float = 80.0,
    eval_interval: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Smoothed exponential‑gradient descent on
        f(a) = max_k (a*a)[k]   (a on the simplex).

    A softmax with temperature λ yields a smooth surrogate.
    The gradient of the surrogate is
        g = 2 * (a_rev  *  w)[central n entries]

    Returns the best vector encountered and its (unscaled) score
    2·n·max(conv).
    """
    a = project_to_simplex(init_a.astype(float).copy())
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # full convolution a*a
        conv = convolution_fft(a, a)          # length 2n‑1
        M = np.max(conv)

        # temperature schedule (linear ramp, capped)
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / 2_000_000)

        # softmax weights (stable)
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)       # length 2n‑1, sums to 1

        # gradient = 2 * (a_rev * w) centred on the original n coordinates
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)    # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # step size (decreasing)
        eta = eta0 / np.sqrt(it)

        # multiplicative mirror‑descent update (log‑domain)
        step = -eta * g
        step = np.clip(step, -50, 50)         # keep exp() numerically safe
        a = a * np.exp(step)

        # re‑project onto simplex
        a = project_to_simplex(a)

        # occasional exact evaluation
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}]  iter {it:,}  new best (smooth) {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Pure sub‑gradient refinement (focus on the worst shift)
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.30,
    eval_interval: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  f(a)=max(conv(a,a)).
    Uses the true sub‑gradient at the current worst shift.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = convolution_fft(a, a)
        k_max = int(np.argmax(conv))

        # sub‑gradient: g_i = 2 * a_{k_max-i}   if the index exists, else 0
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = project_to_simplex(a)

        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}]  subgrad iter {it:,}  new best {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climbing (mass transfers away from the worst contribution)
# ----------------------------------------------------------------------


def advanced_hill_climb(
    init_seq,
    deadline,
    max_steps: int = 800_000,
    start_scale: float = 0.12,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climber that preferentially moves mass from indices
    that contribute heavily to the current worst convolution entry.
    """
    a = np.array(init_seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = start_scale
    steps = 0
    while steps < max_steps and time.time() < deadline:
        # current worst shift and contributions
        conv = convolution_fft(best_a, best_a)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best_a)
        contrib[mask] = best_a[mask] * best_a[idx[mask]]

        # pick source index i – usually a heavy contributor
        if contrib.sum() > 0 and np.random.rand() < 0.85:
            probs = contrib / contrib.sum()
            i = np.random.choice(n, p=probs)
        else:
            i = np.random.randint(n)

        # destination j (different from i, and also preferably not the partner)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * best_a[i]
        if delta <= 0:
            # nothing to move – try again
            steps += 1
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.55)   # enlarge step after success
        else:
            step_scale *= 0.985                         # shrink after failure

        steps += 1

    return best_a.tolist(), best_score

        # pick the better of the two phases as the candidate for hill‑climbing
        candidate = a_sub if score_sub < score_smooth else a_smooth
        cand_score = min(score_sub, score_smooth)

        # 3️⃣ brief targeted hill‑climb before the next restart
        now = time.time()
        hill_deadline = now + 5.0   # small polishing window
        cand_seq, cand_score = advanced_hill_climb(
            candidate,
            deadline=hill_deadline,
            max_steps=150_000,
        )
        print(f"  → restart {restart_counter} best interim score {cand_score:.6f}")

        # Keep global best (using the true evaluate_sequence for safety)
        true_val = evaluate_sequence(cand_seq)
        if true_val < best_val:
            best_val = true_val
            best_seq = cand_seq
            print(f"*** New global best after restart {restart_counter}: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the overall best sequence
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print("\nFinal polishing phase")
        polished_seq, polished_score = advanced_hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=600_000,
            start_scale=0.15,
        )
        true_polished = evaluate_sequence(polished_seq)
        if true_polished < best_val:
            best_val = true_polished
            best_seq = polished_seq
            print(f"Polish improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but guarantees a non‑empty answer.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
```
