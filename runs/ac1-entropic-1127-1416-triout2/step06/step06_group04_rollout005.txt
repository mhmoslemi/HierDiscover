"""
Hybrid random‑beta + Adam + up‑sampling optimizer.

The search proceeds in three logical layers:

1️⃣  **Exploratory sampling** – a handful of deterministic starters
    (uniform, arcsine, symmetric beta grid) and a large pool of random
    beta‑family / mixture shapes.  The best candidate (according to the
    fast proxy 2·n·max(conv)) is kept.

2️⃣  **Local polishing** – a cascade of gradient‑based stages working on
    the probability simplex:
        • flattening   – minimise ‖conv(a,a) – 1/(2n‑1)‖₂² (Adam)
        • smooth‑max  – minimise a smooth log‑sum‑exp surrogate of
                       max(conv) (Adam with an annealed temperature λ
                       up to 2000)
        • exact sub‑gradient descent on the true max (projected)
        • cheap stochastic Gaussian walk.

    All stages respect the global time budget and keep the best vector.

3️⃣  **Upsampling & final polish** – the best vector from step 2 is
    linearly interpolated to a larger resolution (4096 → 8192 points)
    which empirically lowers the max‑convolution.  The upsampled vector
    receives a short round of the smooth‑max Adam optimiser and a final
    random walk.

The routine returns a non‑negative list whose official score is the
minimum found before the 1000 s deadline.  A uniform 256‑vector is used
as a safety net in case nothing else finishes in time.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (real FFT, power‑of‑two padding)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of self‑convolution a*a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for the official score.
    The vector a is assumed to lie on the simplex (sum(a)=1).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (beta families, arcsine, mixtures)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Symmetric beta‑family density on {0,…,n‑1} (unnormalised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta families (renormalised)."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) density."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Gradient of the L2 flattening loss
# ----------------------------------------------------------------------


def flatten_gradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """Gradient of ‖conv(a,a) – avg‖₂², avg = 1/(2n‑1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)                      # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient = 4 * (a ⊛ diff_rev)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)           # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def adam_flatten(
    init: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    lr0: float = 0.30,
) -> Tuple[np.ndarray, float]:
    """Adam optimiser on the L2 flattening loss."""
    a = project_to_simplex(init.astype(float).copy())
    best = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        grad, loss = flatten_gradient(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a_new.copy()

        a = a_new

        if it % 20000 == 0:
            print(
                f"    Adam‑flatten it={it:,d} loss={loss:.3e} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best, best_sc


# ----------------------------------------------------------------------
# Log‑sum‑exp surrogate of the max (smooth‑max) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the smoothed max:
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv‑max))

    Returns (grad, B).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights (size 2n‑1)
    B = max_c + np.log(sum_exp) / lam

    # weighted sub‑gradient: 2 * (a ⊛ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # Chain rule for the official score: 2·n·max/Σ² (Σ=1) → grad = (2·n/Σ²)·∂max
    # Here we directly minimise 2·n·B (the λ‑smoothed version)
    grad = 2.0 * n * weighted_grad
    return grad, B


def adam_max(
    init: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 2000.0,
    noise_every: int = 1200,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed max‑convolution surrogate."""
    a = project_to_simplex(init.astype(float).copy())
    best = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # quadratic schedule for lambda (slow start, fast finish)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, B = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam‑max it={it:,d} lam={lam:.1f} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max objective
# ----------------------------------------------------------------------


def subgradient_optimize(
    init: np.ndarray,
    deadline: float,
    step0: float = 0.28,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = init.copy()
    best = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # active gradient: indices (i, j) with i+j = idx
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # simple back‑tracking line‑search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best, best_sc

def random_local_search(
    init: np.ndarray,
    deadline: float,
    init_step: float = 0.017,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """Gaussian perturbations with occasional uphill acceptance."""
    a = init.copy()
    best = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = compute_score(a_prop)

        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = a_prop.copy()

        if it % 50000 == 0:
            print(
                f"    Random walk it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best, best_sc


# ----------------------------------------------------------------------
# Upsampling utilities
# ----------------------------------------------------------------------


def upsample_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger size."""
    old_n = len(v)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, v)
    up = up.clip(min=0.0)                     # numerical safety
    up = up / up.sum() if up.sum() > 0 else np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Core search for a single length
# ----------------------------------------------------------------------


def search_length(
    n: int,
    rng: np.random.Generator,
    deadline: float,
) -> Tuple[np.ndarray, float]:
    """
    Sample many beta‑family candidates, keep the best according to the fast
    proxy, and run the full polishing pipeline.
    Returns (best_vector, fast_score).
    """
    print(f"\n=== Searching length n={n} ===")

    # ----- 0. deterministic starters & coarse grid -----
    starters = [
        np.full(n, 1.0 / n),    # uniform
        arcsine_initial(n),    # classic arcsine
    ]

    # symmetric beta grid (α=β)
    for a in np.linspace(0.15, 0.85, 13):
        starters.append(generate_beta_shape(n, a, a, eps=1e-6))

    # keep best of deterministic pool
    best_vec = None
    best_sc = float("inf")
    for st in starters:
        sc = compute_score(st)
        if sc < best_sc:
            best_sc = sc
            best_vec = st.copy()

    # ----- 1. random beta / mixture sampling -----
    max_samples = 4000
    for _ in range(max_samples):
        if time.time() >= deadline - 20.0:
            break

        if rng.random() < 0.33:      # mixture
            w = rng.random()
            a1 = rng.uniform(0.02, 3.5)
            b1 = rng.uniform(0.02, 3.5)
            e1 = rng.uniform(1e-8, max(0.01, 0.02 * n))
            a2 = rng.uniform(0.02, 3.5)
            b2 = rng.uniform(0.02, 3.5)
            e2 = rng.uniform(1e-8, max(0.01, 0.02 * n))
            cand = generate_beta_mixture(n, a1, b1, e1, a2, b2, e2, w)
        else:                        # single beta
            a = rng.uniform(0.02, 3.5)
            b = rng.uniform(0.02, 3.5)
            e = rng.uniform(1e-8, max(0.01, 0.02 * n))
            cand = generate_beta_shape(n, a, b, e)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()

    # ----- 2. polishing pipeline -----
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    # budgets for the four stages (seconds)
    now = time.time()
    remaining = deadline - now
    # Stage A – flattening
    if remaining > 8.0:
        time_a_dead = now + remaining * 0.30
        print(f"  Flattening stage (time≈{remaining*0.30:.1f}s)…")
        best_vec, _ = adam_flatten(
            init=best_vec,
            deadline=time_a_dead,
            max_iter=150_000,
            lr0=0.35,
        )
        now = time.time()
        remaining = deadline - now

    # Stage B – smooth‑max Adam
    if remaining > 8.0:
        time_b_dead = now + remaining * 0.35
        print(f"  Smooth‑max Adam stage (time≈{remaining*0.35:.1f}s)…")
        best_vec, _ = adam_max(
            init=best_vec,
            deadline=time_b_dead,
            max_iter=250_000,
            lr0=0.12,
            lam_start=1.0,
            lam_end=2000.0,
            noise_every=1300,
            noise_scale=5e-5,
        )
        now = time.time()
        remaining = deadline - now

    # Stage C – exact sub‑gradient descent
    if remaining > 6.0:
        time_c_dead = now + remaining * 0.30
        print(f"  Sub‑gradient descent (time≈{remaining*0.30:.1f}s)…")
        best_vec, _ = subgradient_optimize(
            init=best_vec,
            deadline=time_c_dead,
            step0=0.28,
            max_iter=150_000,
        )
        now = time.time()
        remaining = deadline - now

    # Stage D – random walk polishing
    if remaining > 4.0:
        time_d_dead = now + remaining * 0.40
        print(f"  Random walk (time≈{remaining*0.40:.1f}s)…")
        best_vec, _ = random_local_search(
            init=best_vec,
            deadline=time_d_dead,
            init_step=0.015,
            max_iter=250_000,
        )
        now = time.time()
        remaining = deadline - now

    # final fast score
    final_fast = compute_score(best_vec)
    return best_vec, final_fast


# ----------------------------------------------------------------------
# Global polishing after upsampling
# ----------------------------------------------------------------------


def global_polish(
    vec: np.ndarray,
    deadline: float,
) -> np.ndarray:
    """
    Upsample to a higher resolution, then perform a short smooth‑max
    Adam run followed by a tiny random walk.  The function respects the
    remaining budget and returns the improved vector.
    """
    n0 = len(vec)
    targets = [n0 * 2, n0 * 4]     # e.g. 4096 → 8192 → 16384 (if time permits)

    for tgt in targets:
        if time.time() >= deadline - 3.0:
            break
        print(f"\n=== Upsampling {n0} → {tgt} ===")
        up = upsample_vector(vec, tgt)

        # a quick Adam on the smooth‑max surrogate (very few iterations)
        time_up_dead = time.time() + min(30.0, deadline - time.time() - 2.0)
        up, _ = adam_max(
            init=up,
            deadline=time_up_dead,
            max_iter=80_000,
            lr0=0.08,
            lam_start=1.0,
            lam_end=3000.0,
            noise_every=2000,
            noise_scale=3e-5,
        )
        # tiny random walk to escape minor local traps
        time_up_dead = time.time() + min(20.0, deadline - time.time() - 1.0)
        up, _ = random_local_search(
            init=up,
            deadline=time_up_dead,
            init_step=0.010,
            max_iter=120_000,
        )
        vec = up
        n0 = tgt

    return vec


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine combines random beta sampling, several Adam optimiser
    phases, exact sub‑gradient descent, stochastic walk and finally an
    up‑sampling polishing step.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # 0.5 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Search on a moderate length (2048) – this is fast and gives a
    #     solid base shape.
    # ------------------------------------------------------------------
    base_n = 2048
    if time.time() < deadline - 30.0:
        vec, fast_sc = search_length(base_n, rng, deadline)
        print(f"Base length {base_n} fast proxy = {fast_sc:.6f}")
        # official evaluation
        val = evaluate_sequence(vec.tolist())
        print(f"Base length {base_n} official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.tolist()

    # ------------------------------------------------------------------
    # 2.  Upsample & global polishing if time permits
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global up‑sampling & polish ===")
        a = np.array(best_seq, dtype=float)
        a = global_polish(a, deadline)
        final_val = evaluate_sequence(a.tolist())
        print(f"Final official score after up‑sample polish = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()

    # ------------------------------------------------------------------
    # 3.  Guarantee a fallback answer
    # ------------------------------------------------------------------
    if not best_seq:
        print("No improvement found – returning uniform 256‑vector.")
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback official score = {best_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
