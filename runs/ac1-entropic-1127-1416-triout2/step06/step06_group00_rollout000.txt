def smooth_optimize_adam(start_a, deadline, max_iters, beta_start, beta_end, lr):
    a = start_a.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    for it in range(1, max_iters+1):
        if time.time()>=deadline:
            break
        # annealing beta
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start)**t
        conv = convolution_fft(a, a) # length 2n-1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w) # length 3n-2
        grad = 2.0 * conv_rev[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = project_to_simplex(a_new)
        # occasional exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    # final evaluation
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score

def hill_climb_targeted(seq, deadline, max_steps=400_000):
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.2 # start moving 20% of mass from high to low
    for step in range(max_steps):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # gradient for max shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # choose source = index with largest grad, target = smallest
        src = int(np.argmax(grad))
        # Possibly pick target among low grad, avoid src
        # We can pick random low grad:
        low_candidates = np.where(grad == grad.min())[0]
        if low_candidates.size > 0:
            tgt = int(np.random.choice(low_candidates))
        else:
            # fallback random
            tgt = int(np.random.randint(n))
        if src == tgt:
            continue
        # transfer fraction
        delta = step_scale * a[src]
        a[src] -= delta
        a[tgt] += delta
        # keep nonnegative
        a = project_to_simplex(a)    # ensures sum=1
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97
    return best_a.tolist(), best_score

"""
Algorithm
---------

Search for a non‑negative step‑function (probability vector) that
minimises the evaluation function

    2 * n * max(conv(a, a))

where `conv` is the full discrete autocorrelation.
The search runs inside the 1000 s wall‑clock limit and consists of
the following stages (all operate on vectors with sum = 1, i.e. a
simplex):

1. **Random sparse restarts** – many candidates are drawn with a random
   support size roughly proportional to √(2 n) and random positive
   masses.  Each candidate is refined by a smooth log‑sum‑exp
   optimisation (a smooth approximation of the max‑operator) using a
   projected Adam optimiser.

2. **Sub‑gradient refinement** – the best candidate from stage 1 is fed
   to a projected Adam optimiser that works on the exact max‑objective.
   The sub‑gradient is the usual autocorrelation sub‑gradient
   (shifted copy of the current vector).

3. **Targeted hill‑climb** – a cheap greedy local search that repeatedly
   moves a fraction of mass from the index that contributes most to the
   peak autocorrelation to the index that contributes least, always
   keeping the vector on the simplex.

The stages respect a global time budget; the final vector is returned
as a plain Python list of floats.  The routine works for any length
(up to a few‑ten‑thousands) and uses only NumPy and the provided
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a ⊛ b using a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Random initial vector
# ----------------------------------------------------------------------


def random_initial_vector(n: int) -> np.ndarray:
    """
    Produce a random probability vector of length n.
    The support size is drawn around √(2 n) and the masses are Gamma‑distributed.
    """
    scale = np.sqrt(2.0 * n)
    # support size between 0.5·scale and 2·scale
    k = max(2, int(np.random.uniform(0.5, 2.0) * scale))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = np.random.gamma(shape=1.0, scale=1.0, size=k)
    a = project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp) with Adam
# ----------------------------------------------------------------------


def smooth_optimize_adam(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 5.0,
                         beta_end: float = 45.0,
                         lr: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv(a, a)) using Adam.
    The smooth objective is f_β(a) = (1/β)·log Σ exp(β·conv_i).
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # full autocorrelation
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))        # stabilised
        w = exp_w / exp_w.sum()                     # weights (probability vector)

        # gradient = 2 * (a_rev * w)_{relevant slice}
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = project_to_simplex(a - step)

        # occasional exact evaluation
        if it % 500 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
    # final check
    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()
    return best_a, best_score

# Sub‑gradient optimisation of the exact max‑objective with Adam

def subgrad_optimize_adam(start_a: np.ndarray,
                          deadline: float,
                          max_iters: int = 300_000,
                          lr: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected Adam descent on the genuine objective
          2·n·max(conv(a, a)).
    The sub‑gradient corresponds to the shift attaining the maximal
    autocorrelation.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = project_to_simplex(a - step)

        # occasional exact evaluation
        if it % 500 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
    # final check
    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb that moves mass from “bad” to “good” positions
# ----------------------------------------------------------------------


def hill_climb_targeted(seq: list[float],
                        deadline: float,
                        max_steps: int = 600_000) -> tuple[list[float], float]:
    """
    Greedy local improvement.
    At each step the shift k giving the maximal autocorrelation is found.
    The index with the largest contribution to this shift loses a
    fraction of its mass, which is transferred to an index with the
    smallest contribution.  The total mass stays constant, so the
    simplex constraint is preserved without an explicit projection.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.20  # start moving 20 % of the source mass each step

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active constraint (same as sub‑gradient)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        src = int(np.argmax(grad))          # most responsible index
        # target: any index with minimal responsibility (could be many)
        min_val = grad.min()
        candidates = np.where(grad == min_val)[0]
        tgt = int(np.random.choice(candidates))

        if src == tgt:
            continue

        delta = step_scale * a[src]          # move a fraction of the source mass
        if delta <= 0:
            continue

        a[src] -= delta
        a[tgt] += delta

        # Numerical safety – enforce non‑negativity and exact sum of 1
        a = np.where(a < 0.0, 0.0, a)
        a_sum = a.sum()
        if not np.isclose(a_sum, 1.0):
            a = a / a_sum

        cur = compute_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            step_scale = min(step_scale * 1.05, 0.5)   # become more aggressive
        else:
            step_scale *= 0.97                         # otherwise become more conservative

    return best_a.tolist(), best_score

    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine follows a multi‑restart Adam‑based optimisation pipeline
    and respects a 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin (2 s) for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a reasonable length (larger gives more freedom)
    # ------------------------------------------------------------------
    # Use 8192 if we have sufficient head‑room, otherwise 4096.
    if time.time() < deadline - 300:
        n = 8192
    else:
        n = 4096
    print(f"Chosen sequence length = {n}")

    # ------------------------------------------------------------------
    # 1. Random restarts with smooth + sub‑gradient refinement
    # ------------------------------------------------------------------
    # Reserve roughly 200 s for later polishing phases.
    while time.time() < deadline - 200:
        a0 = random_initial_vector(n)

        # smooth log‑sum‑exp stage
        a_s, score_s = smooth_optimize_adam(
            a0,
            deadline - 120,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=45.0,
            lr=0.2,
        )

        # sub‑gradient stage
        a_sub, score_sub = subgrad_optimize_adam(
            a_s,
            deadline - 80,
            max_iters=300_000,
            lr=0.07,
        )

        if score_sub < best_val:
            best_val = score_sub
            best_seq = a_sub.tolist()
            print(f"New best after restart: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Further fine‑tuning of the globally best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        # a short extra smooth optimisation with a tighter β‑range
        a_fine, score_fine = smooth_optimize_adam(
            np.array(best_seq),
            deadline - 70,
            max_iters=300_000,
            beta_start=20.0,
            beta_end=80.0,
            lr=0.15,
        )
        if score_fine < best_val:
            best_val = score_fine
            best_seq = a_fine.tolist()
            print(f"Fine‑smooth improvement: score={best_val:.6f}")

        # a second sub‑gradient Adam pass with a slightly smaller step
        a_refine, score_ref = subgrad_optimize_adam(
            np.array(best_seq),
            deadline - 40,
            max_iters=400_000,
            lr=0.06,
        )
        if score_ref < best_val:
            best_val = score_ref
            best_seq = a_refine.tolist()
            print(f"Second sub‑grad improvement: score={best_val:.6f}")

        # ------------------------------------------------------------------
        # 3. Final targeted hill‑climb polishing
        # ------------------------------------------------------------------
        if time.time() < deadline - 5:
            polished_seq, polished_score = hill_climb_targeted(
                best_seq,
                deadline,
                max_steps=600_000,
            )
            if polished_score < best_val:
                best_val = polished_score
                best_seq = polished_seq
                print(f"Hill‑climb final improvement: score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: fallback to a modest uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 256
        best_seq = [1.0 / n0] * n0
        best_val = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided `evaluate_sequence`)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
