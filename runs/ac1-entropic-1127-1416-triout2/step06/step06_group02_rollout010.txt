assistantfinal**Solution Overview**

The task is to find a non‑negative vector `a` (the “step heights”) that minimises  

\[
F(a)=\frac{2\,n\;\max (a\star a)}{(\sum_i a_i)^2},
\qquad n=\text{len}(a),
\]

where `a⋆a` is the ordinary (linear) convolution.  
`F` is homogeneous, therefore we may restrict ourselves to probability vectors
(`∑a_i = 1`).  The optimisation problem is non‑convex; we use a combination
of fast smooth optimisation, exact sub‑gradient steps and a stochastic
hill‑climber.  Two important ideas improve on the previous implementation:

1. **Better initialisation** – a discrete *arcsine*‑like profile  
   `a_i ∝ 1/√(i·(n‑1‑i))` is already close to the theoretical square‑root
   of the uniform distribution and yields a flatter convolution than a
   random sparse start.

2. **Adam‑accelerated soft‑max descent** – the soft‑max surrogate  
   \(\displaystyle f_\varepsilon(a)=\varepsilon\log\sum_k e^{c_k/\varepsilon}\)  
   (with \(c=a\star a\)) gives a smooth approximation of the max‑operator.
   Using the Adam optimiser together with an exponentially decreasing
   temperature `ε` drives the solution quickly toward a low‑peak convolution.

The whole search is time‑bounded (≈ 1000 s).  Several independent starts,
different lengths (`n∈{1024,2048,4096,8192}`) and a final symmetry
averaging step give a reliable chance of beating the target score
`1.503`.

```python
"""
Step‑function optimisation for the evaluation metric

    score = 2·n·max(conv) / (sum(a)**2)

The algorithm combines:

* a discrete arcsine‑type initialization,
* Adam‑accelerated smooth soft‑max gradient descent,
* exact sub‑gradient descent on the true max‑conv,
* a stochastic hill‑climber with mass transfers,
* an optional symmetry averaging step.

All vectors are kept on the probability simplex (non‑negative, sum=1)
so that the cheap fast score `2·n·max(conv)` equals the exact
`evaluate_sequence` value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – FFT convolution and simplex projection
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def arcsine_initialisation(n: int) -> np.ndarray:
    """
    Discrete arcsine–like density: a_i ∝ 1/√(i·(n‑1‑i)).
    The distribution is symmetric and already gives a relatively flat
    self‑convolution.
    """
    i = np.arange(n, dtype=float)
    # avoid division by zero at the boundaries
    eps = 1e-12
    w = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    return project_to_simplex(w)


def random_sparse_vector(n: int) -> np.ndarray:
    """Random probability vector with a modest number of non‑zero entries."""
    density = np.random.uniform(0.15, 0.85)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# 1. Smooth soft‑max descent (Adam optimiser)
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f(a) = eps·log Σ_k exp(conv_k/eps)
    where conv = a⋆a.

    Returns  grad = 2 * (w_rev ⊛ a),
    with w = softmax(conv/eps).
    """
    conv = convolution_fft(a, a)
    maxc = np.max(conv)

    # stable soft‑max
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / np.sum(exp_vals)                 # Σ w = 1

    # cross‑correlation w_rev ⊛ a gives Σ_j a_j w_{i+j}
    w_rev = w[::-1]
    # length of conv(a, w_rev) = len(a) + len(w_rev) - 1 = 2n-1
    corr = convolution_fft(a, w_rev)
    # we need the part that aligns with the original indices
    # indices n‑1 … 2n‑2 correspond to shift i
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad


def adam_softmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 4000,
    step0: float = 0.2,
    eps_factor: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth soft‑max surrogate.
    ε is decreased exponentially; the algorithm stops when the time budget
    expires.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    # Adam parameters
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature – never smaller than 1 % of the current max
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), 0.01)

        grad = softmax_gradient(a, eps)

        # Adam update (biased estimates)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)                       # basic decay
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 500 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent (max‑conv only)
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the true objective 2·n·max(a⋆a).
    The sub‑gradient is taken at the shift where the convolution attains its
    maximum.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k‑i} (zero outside the support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # tiny random shake every 1500 iterations
        if t % 1500 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climbing (swap / mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    time_deadline: float,
    max_steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climber that only accepts strictly improving moves.
    Moves:
        * swap two entries,
        * transfer a fraction of mass from one index to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # transfer size adapts during the run
    step_frac = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_frac * cand[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            # numerical safeguard
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # become bolder
        else:
            step_frac *= 0.96                       # become more cautious

        max_steps -= 1

    return best_a, best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Enforce (approx.) symmetry: (a + reverse(a))/2 and re‑project."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)
```
