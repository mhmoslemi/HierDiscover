"""
Improved search for a low‑score non‑negative step function.

The script performs several random restarts with diverse
initialisations (Dirichlet, Gaussian, sparse and sinusoidal) and
optimises each candidate with a projected Adam optimiser applied to a
log‑sum‑exp (soft‑max) surrogate of the convolution ∞‑norm.  After the
gradient phase a short stochastic hill‑climber (mass‑transfer /
swap moves) is used to polish the best vector found.

All work respects the 1000 s time budget.  The returned sequence is a
list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact score (the exact objective when Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute 2·n·max(a*a) for a vector whose entries sum to 1.
    This matches the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate of the max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    beta0: float = 0.5,
    beta_max: float = 80.0,
    eta0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    report_every: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smooth log‑sum‑exp approximation

        fβ(a) = (2·n/β)·log Σₖ exp(β·(a*a)[k])

    The routine returns the best vector encountered and its exact score.
    """
    n = a0.shape[0]
    a = a0.copy()
    m = np.zeros_like(a)          # first‑moment estimate
    v = np.zeros_like(a)          # second‑moment estimate
    best_a = a.copy()
    best_val = fast_score(best_a)

    # Length needed for all FFTs (a, a_rev, and w)
    L = next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline - 0.5:
            break

        # ---------- self‑convolution ----------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        max_c = np.max(conv_full)

        # ---------- soft‑max weights ----------
        beta = min(beta_max, beta0 * np.sqrt(t))
        # stable log‑sum‑exp
        exp_terms = np.exp(beta * (conv_full - max_c))
        w = exp_terms / np.sum(exp_terms)

        # ---------- gradient ----------
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(np.fft.rfft(a_rev, n=L) * W, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = eta0 / np.sqrt(t)          # decreasing base learning rate
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex (maintains Σa = 1, a ≥ 0)
        a = project_to_simplex(a)

        # occasional exact evaluation
        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  iter {t:8d} – new best exact score {cur_val:.6f}")

    # final check
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (mass‑transfer / swap moves)
# ----------------------------------------------------------------------


def hill_climb(
    seq: list[float],
    time_deadline: float,
    max_steps: int = 300_000,
    init_step: float = 0.2,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climber that performs two types of moves:
    * swap two entries,
    * transfer a fraction of mass from one position to another.
    The true objective `evaluate_sequence` is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    # keep the vector normalised (Σa = 1) at all times
    a = a / a.sum()
    best_a = a.copy()
    best_score = fast_score(best_a)

    step_frac = init_step

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # ---- swap ----
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # ---- mass transfer ----
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 1e-12:
                delta = step_frac * cand[i]          # amount moved from i to j
                cand[i] -= delta
                cand[j] += delta
        # re‑normalise (project onto simplex)
        cand = project_to_simplex(cand)

        cur_score = fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # enlarge step on success
        else:
            step_frac = max(step_frac * 0.93, 1e-4)   # shrink on failure

        max_steps -= 1

    return best_a.tolist(), best_score
