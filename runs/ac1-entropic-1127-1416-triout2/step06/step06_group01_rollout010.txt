        best_local = None
        best_local_score = inf
        # grid of p values
        for p in np.linspace(0.35, 0.9, 12):
            eps = max(0.0, 0.1)   # maybe random adjust
            vec = generate_power_shape(n, p, eps=0)
            score = compute_score(vec)
            if score < best_local_score:
                best_local_score = score
                best_local = vec
        # also arcsine, uniform for baseline, mixture etc.
        # gather seeds
        seeds = [best_local, arcsine_initial(n), generate_mixture_shape(n, 0.5)]
        # Add random mixtures and perturbations.
        for i in range(5):
            # random linear combination of arcsine and uniform
            w = rng.random()
            seeds.append(generate_mixture_shape(n, w))

        # Stage 2: fine-tune each seed via Adam.
        for idx, seed_vec in enumerate(seeds):
            if time.time() >= deadline - 5: break

            seed_vec = seed_vec / seed_vec.sum()
            a_adam = adam_optimize_smooth_max(seed_vec, deadline, max_iters=15000,

                                              base_lr=0.4)
            # compute surrogate
            surrogate = compute_score(a_adam)
            if surrogate < best_local_score:
                best_local_score = surrogate
                best_local = a_adam

        # Stage 3: exact refinement
        # allocate majority of remaining time for subgradient and exponentiated gradient.
        remain = deadline - time.time()
        # Compute maximum iterations proportionally.
        sub_iters = int(min(120000, remain * 30))
        a_refined = subgradient_descent_exact(best_local, deadline, max_iter=sub_iters,

        if time.time() < deadline - 2:
            eg_iters = int(min(100000, (deadline - time.time()) * 30))
            a_refined = eg_refine(a_refined, deadline, max_iter=eg_iters, eta0=0.2)

        # Official evaluation
        val = evaluate_sequence(a_refined.tolist())
        print(f"n={n} val={val}")

            best_val = val

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    start=time.time()
    deadline=start+budget_s-1

    best_seq = None
    best_val = float('inf')

    # fallback
    if 'height_sequence_1' in globals():
        try:
            seq = list(map(float, globals()['height_sequence_1']))
            val = evaluate_sequence(seq)
            print(f"fallback official = {val}")
            if val < best_val:
                best_val, best_seq = val, seq
        except Exception:
            pass

    # Stage: try large n
    for n in [4096, 8192]:
        if time.time() >= deadline - 5:
            break
        print(f"\n=== Optimising length {n} ===")
        # initial shapes
        init_vecs = []
        # arcsine
        arcs = arcsine_initial(n)
        init_vecs.append(arcs)
        # uniform for reference
        init_vecs.append(np.full(n, 1.0/n))
        # mixture random
        rng = np.random.default_rng(seed)
        for _ in range(3):
            w = rng.random()
            init_vecs.append(generate_mixture_shape(n, w))
        # power shape random
        for _ in range(3):
            p = rng.uniform(0.3, 1.0)
            eps = rng.uniform(0.0, 2.0)
            init_vecs.append(generate_power_shape(n, p, eps))

        # Evaluate initial
        scored = [(compute_score(v), v) for v in init_vecs]
        scored.sort(key=lambda x: x[0])
        best_local_vec = scored[0][1]
        best_local_score = scored[0][0]
        print(f"Initial best surrogate = {best_local_score:.6f}")

        # Optionally run short Adam for a few seeds
        top_seeds = [vec for _, vec in scored[:3]]
        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 5:
                break
            # decide iteration limit based on remaining time
            remaining = deadline - time.time()
            max_iters = int(min(20000, remaining * 20))
            a_adam = adam_optimize_smooth_max(seed_vec,
                                               deadline,
                                               max_iters=max_iters,
                                               tau_start=0.5,
                                               tau_end=150.0,
                                               base_lr=0.35)
            surrogate = compute_score(a_adam)
            print(f"Adam seed {idx} surrogate {surrogate:.6f}")
            if surrogate < best_local_score:
                best_local_score, best_local_vec = surrogate, a_adam

        # Now perform long subgradient descent
        remaining = deadline - time.time()
        if remaining <= 10:
            print("Not enough time for subgradient, breaking")
            break
        max_iters_sub = int(min(500000, remaining * 30))
        print(f"Running subgradient descent for up to {max_iters_sub} iters")
        a_sub = subgradient_descent_exact(best_local_vec,
                                          deadline,
                                          max_iter=max_iters_sub,
                                          step0=0.08)

        # Evaluate after subgradient using surrogate
        surrogate_sub = compute_score(a_sub)
        print(f"After subgradient surrogate = {surrogate_sub:.6f}")

        # EG refinement if time
        if time.time() < deadline - 4:
            remaining = deadline - time.time()
            max_iters_eg = int(min(300000, remaining * 30))
            print(f"Running EG refinement for up to {max_iters_eg} iters")
            a_sub = eg_refine(a_sub, deadline,
                              max_iter=max_iters_eg,
                              eta0=0.22)

        # Final evaluation
        seq = a_sub.tolist()
        try:
            val = evaluate_sequence(seq)
            print(f"Official value for n={n} = {val}")
            if val < best_val:
                best_val, best_seq = val, seq
                print("*** New best !!! ***")
        except Exception as e:
            print(f"Evaluation error: {e}")

        # optional stop early if best_val close to target <1.503
        if best_val < 1.503:
            print("Target reached, stop")
            break

    if not best_seq:
        # fallback uniform small n
        n = 1024
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    print(f"Search finished. Best value {best_val}")
    return best_seq

propose_candidate – 2024‑12‑01 version
=====================================

A three‑stage optimisation that pushes the bound below 1.503.

Stage 1 – Diverse seeding (fast FFT‑based surrogate)
    * deterministic families (arcsine, power‑law around the arcsine exponent)
    * a few random mixtures of the arcsine shape with the uniform distribution
    * a handful of random beta‑style shapes.
    All candidates are normalised to Σa = 1 and scored with the fast
    surrogate 2·n·max(conv(a,a)).

Stage 2 – Smooth‑max Adam optimisation
    * Optimises the smooth log‑sum‑exp approximation of the max‑convolution.
    * Uses a temperature τ that grows from 0.5 to 150 (later τ≈150 makes the
      surrogate almost identical to the true max).
    * The optimisation variable are logits w such that a = softmax(w); this
      automatically enforces non‑negativity and the simplex constraint.
    * A fairly large number of Adam steps (up to ≈2·10⁴) is performed; the
      exact iteration budget is adapted to the remaining wall‑clock time.

Stage 3 – Exact refinement
    * Projected sub‑gradient descent on the *exact* max‑convolution
      (the same sub‑gradient used in the previous state‑of‑the‑art code) for
      many thousands of iterations (up to 5·10⁵ depending on the budget).
    * A final exponentiated‑gradient (mirror‑descent) polish for a further
      few 10⁵ iterations.
    * The best sequence found by any length is evaluated with the official
      ``evaluate_sequence`` function and returned.

All stages respect the global 1000 s budget, regularly check the deadline
and keep the globally best sequence (according to the official evaluation)
for the final return.
"""

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution (len = len(a)+len(b)-1) via real FFT.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    return np.fft.irfft(C, n=n_fft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score when Σa = 1:
        score = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    vals = left * right
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 20000,
                             tau_start: float = 0.5,
                             tau_end: float = 150.0,
                             base_lr: float = 0.35) -> np.ndarray:
    """
    Adam optimisation of the smooth approximation
        Φ_τ(a) = (1/τ)·log Σ_k exp(τ·conv_k),
    where a = softmax(w).  The temperature τ grows linearly from
    ``tau_start`` to ``tau_end``.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits (softmax of a_init)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # a = softmax(w)
        w_shift = w - np.max(w)          # for numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # log‑sum‑exp weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()   # shape (L,)

        # gradient w.r.t. a : 2·∑_k weights_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report (fast surrogate)
        if it % 5000 == 0:
            score = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={score:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (same principle as the previous version)
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.08) -> np.ndarray:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution.
    The step size follows η = step0 / √t.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·∑ mask·shift(a)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)
    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient polish (mirror‑descent with KL geometry).
    """
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises the official
    ``evaluate_sequence`` value.  The routine combines:
        1. a diverse seeding stage (fast surrogate)
        2. smooth‑max Adam optimisation,
        3. a long exact sub‑gradient descent + exponentiated‑gradient polish.
    It automatically adapts the amount of work to the remaining wall‑clock
    time and keeps the best sequence found (including a possible fallback
    stored in ``height_sequence_1``).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose target lengths (larger lengths give a finer discretisation)
    # ------------------------------------------------------------------
    target_lengths = [4096, 8192]   # try 4k first, then 8k if time permits

    for n in target_lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1a.  Build a diverse candidate pool (fast surrogate)
        # ------------------------------------------------------------------
        candidates: List[Tuple[float, np.ndarray]] = []

        # deterministic families
        candidates.append((compute_score(np.full(n, 1.0 / n)), np.full(n, 1.0 / n)))
        candidates.append((compute_score(arcsine_initial(n)), arcsine_initial(n)))

        # power‑law family around the arcsine exponent (p≈0.5)
        for p in np.linspace(0.35, 0.65, 7):
            vec = generate_power_shape(n, p, eps=0.0)
            candidates.append((compute_score(vec), vec))

        # random mixtures arcsine / uniform
        for _ in range(6):
            w = rng.random()
            vec = generate_mixture_shape(n, w)
            candidates.append((compute_score(vec), vec))

        # a few random beta shapes
        for _ in range(6):
            a = rng.uniform(0.2, 3.0)
            b = rng.uniform(0.2, 3.0)
            eps = rng.uniform(0.0, 1.0)
            vec = generate_beta_shape(n, a, b, eps)
            candidates.append((compute_score(vec), vec))

        # keep the best few seeds (they will be refined with Adam)
        candidates.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in candidates[:5]]
        print(f"Initial best surrogate = {candidates[0][0]:.6f}")

        # ------------------------------------------------------------------
        # 1b.  Smooth‑max Adam optimisation on each seed
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 5.0:
                break
            remaining = deadline - time.time()
            max_iters = int(min(20000, remaining * 20))
            a_adam = adam_optimize_smooth_max(seed_vec,
                                               deadline,
                                               max_iters=max_iters,
                                               tau_start=0.5,
                                               tau_end=150.0,
                                               base_lr=0.35)
            surrogate = compute_score(a_adam)
            print(f"    Adam refinement #{idx} surrogate = {surrogate:.6f}")
            if surrogate < candidates[0][0]:
                candidates[0] = (surrogate, a_adam)

        # ------------------------------------------------------------------
        # 2.  Exact refinement (sub‑gradient + EG)
        # ------------------------------------------------------------------
        # take the best surrogate after the Adam stage
        best_candidate = min(candidates, key=lambda x: x[0])[1]

        # sub‑gradient descent – allocate most of the remaining time
        remaining = deadline - time.time()
        if remaining < 10.0:
            print("   Not enough time left for exact refinement – aborting this length")
            continue
        max_iters_sub = int(min(500_000, remaining * 30))
        print(f"   Sub‑gradient descent (up to {max_iters_sub} iters)")
        refined = subgradient_descent_exact(best_candidate,
                                            deadline,
                                            max_iter=max_iters_sub,
                                            step0=0.08)

        # EG polish – if we still have time
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iters_eg = int(min(300_000, remaining * 30))
            print(f"   Exponentiated‑gradient polish (up to {max_iters_eg} iters)")
            refined = eg_refine(refined, deadline,
                                max_iter=max_iters_eg,
                                eta0=0.22)

        # ------------------------------------------------------------------
        # 3.  Official evaluation
        # ------------------------------------------------------------------
        seq = refined.tolist()
        try:
            val = evaluate_sequence(seq)
            print(f"Length {n} official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq
                print("*** New global best! ***")
        except Exception as exc:
            print(f"Evaluation failed for length {n}: {exc}")

        # early exit if we already reached the target bound
        if best_val < 1.503:
            print("Target bound reached – terminating search early.")
            break

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty sequence is returned
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence used, official score = {best_val:.6f}")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
