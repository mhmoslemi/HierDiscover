propose_candidate – 2024‑11‑27 (enhanced)

A “wide‑and‑deep” search that tries to beat the previous 1.51212 bound.
The routine works in three phases inside the 1000 s budget:

1️⃣  Diverse seeding – many parametric families (beta, power, mixtures,

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)); assumes Σa = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine density (U‑shaped)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β) on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^(−p)."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def sine_shape(n: int, power: float = 1.0) -> np.ndarray:
    """Sinusoidal bump peaking in the middle (can be flattened with power)."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n) ** power
    total = vals.sum()
    return vals / total


def triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def mixture_shape(n: int, w_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_shape(n)
    uni = np.full(n, 1.0 / n)
    mix = w_arc * arc + (1.0 - w_arc) * uni
    return mix / mix.sum()


def dirichlet_shape(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex via Dirichlet(α,…,α)."""
    rng = np.random.default_rng()
    vec = rng.gamma(alpha, 1.0, size=n)
    vec_sum = vec.sum()
    if vec_sum <= 0:
        return np.full(n, 1.0 / n)
    return vec / vec_sum

        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()      # shape (L,)

        # gradient wrt a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)        # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wrt logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report (fast surrogate)
        if it % 8_000 == 0:
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={fast_surrogate(a):.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(a0: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.1) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = project_to_simplex(a0.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of indices attaining (or almost attaining) the max
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·(a * mask) convolved (see original code)
        rev_mask = mask[::-1]
        grad = 2.0 * convolution_fft(a, rev_mask)[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        a = project_to_simplex(a)

    return a


def exponentiated_gradient(a0: np.ndarray,
                          deadline: float,
                          max_iter: int = 200_000,
                          eta0: float = 0.25) -> np.ndarray:
    """Mirror‑descent polishing with KL geometry."""
    a = a0.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev_mask = mask[::-1]
        grad = 2.0 * convolution_fft(a, rev_mask)[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * grad)

        # renormalise – keep on the simplex
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


def random_perturbation(a: np.ndarray,
                        rng: np.random.Generator,
                        magnitude: float = 1e-3) -> np.ndarray:
    """Move a tiny amount of mass between a few random coordinates."""
    n = a.shape[0]
    # choose 2‑3 indices to modify
    k = rng.integers(2, 5)               # 2–4 indices
    idx = rng.choice(n, size=k, replace=False)
    # generate a random direction that preserves the simplex
    delta = rng.normal(scale=magnitude, size=k)
    delta -= delta.mean()                # zero‑sum adjustment
    a_new = a.copy()
    a_new[idx] += delta
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new = a_new / s
    return a_new

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence`` value.
    The routine mixes random parametric seeding, smooth‑max Adam optimisation,
    exact sub‑gradient polishing and a final stochastic local‑search phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback – use a previously found construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Generate a large, diverse pool (fast surrogate)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]          # three sizes, last only if time permits
    pool: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n--- Seeding length {n} ---")
        # deterministic starters
        uniform = np.full(n, 1.0 / n)
        pool.append((fast_surrogate(uniform), uniform.copy()))
        arc = arcsine_shape(n)
        pool.append((fast_surrogate(arc), arc.copy()))

        # random beta families (including U‑shaped cases α,β<1)
        for _ in range(250):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.05, 3.0)
            b = rng.uniform(0.05, 3.0)
            eps = rng.uniform(0.0, 0.5)
            cand = beta_shape(n, a, b, eps)
            pool.append((fast_surrogate(cand), cand))

        # random power‑shape
        for _ in range(250):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.2, 2.0)
            eps = rng.uniform(0.0, 2.0)
            cand = power_shape(n, p, eps)
            pool.append((fast_surrogate(cand), cand))

        # sine‑shaped (can be raised to a power to flatten the peak)
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            power = rng.uniform(0.5, 3.0)
            cand = sine_shape(n, power)
            pool.append((fast_surrogate(cand), cand))

        # mixtures of arcsine & uniform
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = mixture_shape(n, w)
            pool.append((fast_surrogate(cand), cand))

        # triangular bumps with random apex
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            peak = rng.integers(0, n)
            cand = triangular_shape(n, peak)
            pool.append((fast_surrogate(cand), cand))

        # random Dirichlet draws (purely uniform‑like but give extra diversity)
        for _ in range(200):
            if time.time() >= deadline - 5.0:
                break
            alpha = rng.uniform(0.2, 5.0)
            cand = dirichlet_shape(n, alpha)
            pool.append((fast_surrogate(cand), cand))

    # keep the best few seeds for the expensive Adam stage
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:12]]
    print(f"\nSelected {len(top_seeds)} top seeds for Adam optimisation "
          f"(surrogate range {pool[0][0]:.6f} – {pool[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2.  Smooth‑max Adam optimisation of each seed
    # ------------------------------------------------------------------
    refined_seeds: List[np.ndarray] = []
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 4.0:
            break
        n = seed_vec.shape[0]
        print(f"\n=== Adam refinement of seed {idx}/{len(top_seeds)} (n={n}) ===")
        remaining = deadline - time.time()
        it_budget = int(min(45_000, remaining * 35))   # allocate generously
        a_adam = adam_optimize_smooth_max(seed_vec,
                                          deadline,
                                          max_iters=it_budget,
                                          tau_start=0.5,
                                          tau_end=200.0,
                                          base_lr=0.4)

        surrogate = fast_surrogate(a_adam)
        print(f"    Post‑Adam surrogate = {surrogate:.6f}")
        refined_seeds.append(a_adam)
        pool.append((surrogate, a_adam))

    # ------------------------------------------------------------------
    # 3.  Exact refinement (sub‑gradient + EG) on the globally best seed
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    candidate = pool[0][1]               # the smallest surrogate so far
    n = candidate.shape[0]
    print(f"\n--- Exact refinement (n={n}) ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(250_000, remaining * 30))
    candidate = subgradient_descent_exact(candidate,
                                          deadline,
                                          max_iter=max_iter_sub,
                                          step0=0.09)

    if time.time() < deadline - 3.0:
        remaining = deadline - time.time()
        max_iter_eg = int(min(150_000, remaining * 30))
        candidate = exponentiated_gradient(candidate,
                                           deadline,
                                           max_iter=max_iter_eg,
                                           eta0=0.22)

    # official evaluation after the exact polishing
    try:
        val = evaluate_sequence(candidate.tolist())
        print(f"Official score after exact polish = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = candidate.tolist()
            print("*** New global best! ***")
    except Exception as e:
        print(f"Evaluation failed after exact polish: {e}")

    # ------------------------------------------------------------------
    # 4.  Random‑perturbation local search (simulated annealing style)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n--- Stochastic local search ---")
        a = np.array(best_seq, dtype=float)
        cur_val = best_val
        temperature = 0.02                      # initial step magnitude
        while time.time() < deadline - 2.0:
            # gradually shrink the perturbation magnitude
            temperature *= 0.9995
            a_new = random_perturbation(a, rng, magnitude=temperature)
            new_val = evaluate_sequence(a_new.tolist())
            if new_val < cur_val:
                a = a_new
                cur_val = new_val
                print(f"    Improved to {new_val:.6f}")
                if new_val < best_val:
                    best_val = new_val
                    best_seq = a_new.tolist()
                    print("*** Updated global best (local search) ***")
        # store final local‑search result
        best_seq = a.tolist()
        best_val = cur_val

    # ------------------------------------------------------------------
    # 5.  Final safeguard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
