"""
Propose a step‐function (non‑negative vector) with a low
score = 2·n·max(conv(a,a))/sum(a)^2.

The search uses three ideas that differed from the previous
implementation:

1️⃣  A richer family of initial shapes (beta, exponential,
    sinusoidal, mixtures) and a broader parameter range.
2️⃣  A high‑temperature Adam optimisation of a smooth
    log‑sum‑exp surrogate for the max‑convolution with a
    more aggressive λ‑schedule (λ up to 2000).  After each
    Adam step the vector is projected onto the probability
    simplex and symmetrised – the optimum is known to be
    symmetric.
3️⃣  A final cheap stochastic Gaussian walk on the simplex
    (with tiny uphill moves) to escape any remaining plateau.

All phases respect the global 1000 s budget and the routine
guarantees a non‑empty return (uniform 256‑vector) even if
the time runs out.  The code is fully self‑contained and
uses only numpy and time.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (full length)."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    Works for vectors already normalised to Σa = 1:
    score = 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Simplex / symmetry utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al. (2008), O(n log n).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry (v_i = v_{n‑1‑i}) while preserving the sum."""
    return 0.5 * (v + v[::-1])

def generate_beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-8) -> np.ndarray:
    """Symmetric beta‑family density on {0,…,n‑1} (unnormalised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float, beta1: float, eps1: float,
    alpha2: float, beta2: float, eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities (renormalised)."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) density."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def exponential_shape(n: int, decay: float) -> np.ndarray:
    """Symmetric exponential (geometric) shape."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-decay * i) + np.exp(-decay * (n - 1 - i))
    vals /= vals.sum()
    return vals


def sine_shape(n: int, power: float) -> np.ndarray:
    """Symmetric sine‑power shape (U‑shaped for power>0)."""
    i = np.arange(n, dtype=float) + 0.5
    angle = np.pi * i / n
    vals = np.sin(angle) ** power
    # protect against underflow at edges
    vals[vals < 0] = 0.0
    vals /= vals.sum()
    return vals

# Gradient utilities
# ----------------------------------------------------------------------


def flatten_gradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Gradient of ‖conv(a,a) – avg‖₂² where avg = 1/(2n‑1) and Σa = 1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient: 4 * (a ⊛ diff_rev)  (⊛ = convolution)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)          # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def adam_flatten(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    lr0: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the L₂ flattening loss.
    Returns best vector (projected & symmetrised) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        grad, loss = flatten_gradient(a)

        # Adam update (learning rate decays like 1/√it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_to_simplex(a_new)
        a_new = symmetrize(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 20000 == 0:
            print(
                f"      Adam‑flatten iter={it:,d} loss={loss:.2e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


def smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smooth log‑sum‑exp surrogate for the max‑convolution.
    Returns (grad, surrogate B) where
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv‑max)).
    The gradient corresponds to the official objective
    2·n·max(conv)/sum(a)² (with sum(a) kept at 1 by projection).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights on convolution entries
    B = max_c + np.log(sum_exp) / lam

    # weighted gradient of the *weighted* sum of convolution entries:
    # ∂/∂a Σ_k w_k (a*a)_k = 2 * (a ⊛ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # Gradient of the official score 2·n·max/Σ² (Σ=1) approximated by 2·n·grad of B
    grad = (2.0 * n) * weighted_grad
    return grad, B


def adam_max(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 250_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 2000.0,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth max‑convolution surrogate.
    Returns the best vector (projected & symmetrised) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # Quadratic λ‑schedule (slow start, fast finish)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)
        a_new = symmetrize(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"      Adam‑max iter={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f} lam={lam:.1f}"
            )
    return best_a, best_sc


def random_local_search(
    initial_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap Gaussian walk on the simplex with tiny uphill acceptance.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)
        a_proposed = symmetrize(a_proposed)

        cur_sc = compute_score(a_proposed)

        # Accept if strictly better, otherwise tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"      Random walk iter={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine mixes an enriched shape sampling, a strong smooth‑max
    Adam run (λ up to 2000) and a final stochastic walk.  All phases
    respect the global time budget and a fallback uniform vector is
    guaranteed.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use a known construction if available (fallback)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Shape sampling on a moderate length (2048) – most time is spent here
    # ------------------------------------------------------------------
    n = 2048
    print(f"\n=== Sampling shape space for n = {n} ===")
    local_best = None
    local_best_sc = float("inf")

    # deterministic starters
    starters = [
        np.full(n, 1.0 / n),          # uniform
        arcsine_initial(n),           # classic arcsine
    ]

    # grid on symmetric beta exponents (α = β) – extended range
    for alpha in np.linspace(0.05, 1.20, 17):
        cand = generate_beta_shape(n, alpha, alpha, eps=1e-8)
        starters.append(cand)

    # sinusoidal and exponential templates
    for power in np.linspace(0.5, 3.0, 9):
        starters.append(sine_shape(n, power))
    for decay in np.logspace(-4, -1, 9):
        starters.append(exponential_shape(n, decay))

    # evaluate deterministic starters
    for st in starters:
        st = symmetrize(project_to_simplex(st))
        sc = compute_score(st)
        if sc < local_best_sc:
            local_best_sc = sc
            local_best = st.copy()

    # random sampling (beta, mixtures, exponential, sine)
    max_random_samples = 8000
    for _ in range(max_random_samples):
        if time.time() >= deadline - 30.0:
            break
        if rng.random() < 0.30:
            # mixture of two beta shapes
            w = rng.random()
            a1 = generate_beta_shape(
                n,
                rng.uniform(0.02, 2.0),
                rng.uniform(0.02, 2.0),
                eps=rng.uniform(1e-8, 0.01),
            )
            a2 = generate_beta_shape(
                n,
                rng.uniform(0.02, 2.0),
                rng.uniform(0.02, 2.0),
                eps=rng.uniform(1e-8, 0.01),
            )
            cand = w * a1 + (1.0 - w) * a2
        else:
            # choose a random family
            family = rng.choice(["beta", "exp", "sine"])
            if family == "beta":
                cand = generate_beta_shape(
                    n,
                    rng.uniform(0.02, 2.5),
                    rng.uniform(0.02, 2.5),
                    eps=rng.uniform(1e-8, 0.02)
                )
            elif family == "exp":
                cand = exponential_shape(n, rng.uniform(5e-5, 5e-2))
            else:  # sine
                cand = sine_shape(n, rng.uniform(0.4, 3.5))

        cand = symmetrize(project_to_simplex(cand))
        sc = compute_score(cand)
        if sc < local_best_sc:
            local_best_sc = sc
            local_best = cand

    if local_best is None:
        print("No candidate found during sampling – falling back to uniform.")
        local_best = np.full(n, 1.0 / n)

    print(f"  Best sampled surrogate score = {local_best_sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Heavy optimisation of the best sampled vector
    # ------------------------------------------------------------------
    # allocate remaining time
    remaining = deadline - time.time()
    if remaining < 5.0:
        print("Time ran out before optimisation.")
    else:
        # -------- Stage A – flattening (few hundred thousand Adam steps) --------
        flat_iters = int(min(120_000, remaining * 22))
        print(f"  Flattening stage (max_iter={flat_iters}) …")
        a_flat, _ = adam_flatten(
            initial_a=local_best,
            deadline=deadline,
            max_iter=flat_iters,
            lr0=0.30,
        )
        # -------- Stage B – smooth‑max Adam (high λ) ---------------------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iters_adam = int(min(300_000, remaining * 28))
            print(f"  Smooth‑max Adam (max_iter={max_iters_adam}, λ up to 2000) …")
            a_max, _ = adam_max(
                initial_a=a_flat,
                deadline=deadline,
                max_iter=max_iters_adam,
                lr0=0.12,
                lam_start=1.0,
                lam_end=2000.0,
                noise_every=0,
                noise_scale=0.0,
            )
        else:
            a_max = a_flat

        # -------- Stage C – final stochastic walk -----------------------------
        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            max_iters_walk = int(min(250_000, remaining * 35))
            print(f"  Final random walk (max_iter={max_iters_walk}) …")
            a_final, _ = random_local_search(
                a_max,
                deadline=deadline,
                init_step=0.012,
                max_iter=max_iters_walk,
            )
        else:
            a_final = a_max

        # -------- Official evaluation -----------------------------------------
        val = evaluate_sequence(a_final.tolist())
        print(f"  → official score for length {n} = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()
            print(f"  *** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3.  Global polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # Stage A – long flattening
        remaining = deadline - time.time()
        max_iter_flat = int(min(180_000, remaining * 20))
        print(f"  Global flattening (max_iter={max_iter_flat}) …")
        a, _ = adam_flatten(
            initial_a=a,
            deadline=deadline,
            max_iter=max_iter_flat,
            lr0=0.35,
        )

        # Stage B – high‑temperature Adam on the max‑surrogate
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_adam = int(min(400_000, remaining * 28))
            print(f"  Global Adam‑max (max_iter={max_iter_adam}, λ up to 2000) …")
            a, _ = adam_max(
                initial_a=a,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.10,
                lam_start=1.0,
                lam_end=2000.0,
                noise_every=0,
                noise_scale=0.0,
            )

        # Stage C – final random walk
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_walk = int(min(300_000, remaining * 40))
            print(f"  Global random walk (max_iter={max_iter_walk}) …")
            a, _ = random_local_search(
                a,
                deadline=deadline,
                init_step=0.009,
                max_iter=max_iter_walk,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guarantees a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
