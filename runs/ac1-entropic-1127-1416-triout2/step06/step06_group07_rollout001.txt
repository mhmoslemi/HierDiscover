Propose a low‑score non‑negative step function.

Algorithm Overview
------------------
1.  Choose a sequence length (tries 4096, 2048, 1024 depending on remaining time).
2.  For each length evaluate a few deterministic candidates
    (cosine‑shaped, triangular) and keep the best.
3.  Perform several random restarts:
    * Initialise a probability vector with a Dirichlet draw.
    * Refine it with a soft‑max (log‑sum‑exp) approximation of the
      ∞‑norm of the self‑convolution, using exponentiated mirror updates
      and projection on the probability simplex.
    * Polish the refined vector with a simulated‑annealing hill‑climber
      (swap / mass‑transfer moves) that works directly on the exact
      objective.
4.  Keep the best vector found and return it.
The routine respects the 1000 s budget (2 s safety margin) and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (returns length len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # all non‑positive -> return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a: np.ndarray) -> float:
    """
    Exact objective when Σ a = 1:
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Deterministic candidate generators (non‑random)
# ----------------------------------------------------------------------


def cosine_candidate(n: int) -> np.ndarray:
    """Non‑negative cosine‑shaped distribution, normalised to sum 1."""
    x = np.arange(n, dtype=float)
    # cosine varies from -1 to 1; shift to [0,2] and scale
    a = 1.0 + np.cos(2.0 * np.pi * x / (n - 1))
    a = a / a.sum()
    return a


def triangular_candidate(n: int) -> np.ndarray:
    """Symmetric triangular distribution, normalised to sum 1."""
    half = n // 2
    if n % 2 == 0:
        inc = np.arange(1, half + 1, dtype=float)
        dec = inc[::-1]
    else:
        inc = np.arange(1, half + 2, dtype=float)
        dec = inc[:-1][::-1]
    a = np.concatenate([inc, dec])
    a = a / a.sum()
    return a

    beta0: float = 0.5,
    beta_max: float = 200.0,

    """
    Minimise a smooth approximation of max(a*a) using exponentiated
    updates (mirror descent) and projection onto the simplex.
    Returns the best vector encountered and its exact score.
    """
    n = a.shape[0]
    L = next_pow2(3 * n - 2)  # length sufficient for all FFTs
    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline - 0.5:
            break

        # ----- convolution a*a -------------------------------------------------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = np.max(conv)

        # ----- soft‑max weighting ------------------------------------------------
        beta = min(beta_max, beta0 * np.sqrt(t))
        # stable exponentials
        exp_terms = np.exp(beta * (conv - max_c))
        w = exp_terms / np.sum(exp_terms)

        # ----- gradient of the smooth objective ---------------------------------
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        # slice that corresponds to the gradient w.r.t. each a_i
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- exponentiated (mirror) update ------------------------------------
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        # ----- bookkeeping -------------------------------------------------------
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if report_every and t % report_every == 0:
                print(f"  iter {t:8d} – new best score {cur_val:.6f}")

    # final check (in case the loop exited early)
    cur_val = fast_score(a)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_sa(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber with a simple linear temperature schedule.
    Works directly on the exact objective (fast_score) and always
    respects the simplex constraint via projection.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    while max_steps > 0 and time.time() < deadline:
        cand = cur.copy()

        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a random fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.5 * np.random.rand() * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cand_score = fast_score(cand)

        # linear cooling from 1.0 -> 1e‑4
        elapsed = time.time() - start
        total = deadline - start
        T = max(1e-4, 1.0 - elapsed / total)

        if cand_score < cur_score or np.random.rand() < np.exp((cur_score - cand_score) / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()

        max_steps -= 1

    return best.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The search uses a mixture of deterministic seeds, smooth
    exponentiated‑gradient optimisation and a simulated‑annealing
    hill‑climber.  All work respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin; the harness will terminate the run at 1000 s
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Fallback: any previously‑found construction (if present)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val = fb_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Search over a few lengths (largest first)
    # --------------------------------------------------------------
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5:
            break
        print(f"\n=== Searching length n = {n} ===")

        # ----------------------------------------------------------
        # 1a.  Evaluate cheap deterministic candidates
        # ----------------------------------------------------------
        for maker, name in [(cosine_candidate, "cosine"),
                            (triangular_candidate, "triangular")]:
            a_det = maker(n)
            det_score = fast_score(a_det)
            print(f"Deterministic {name} init score {det_score:.6f}")
            if det_score < best_val:
                best_val = det_score
                best_seq = a_det.tolist()
                print(f"  → new global best (deterministic)")

        # ----------------------------------------------------------
        # 2️⃣  Estimate how many restarts we can afford for this length
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 30:
            print("Not enough time left for this length.")
            continue

        # Rough heuristic for iterations per restart
        if remaining > 300:
            max_iters = 250_000
            restarts = 3
        elif remaining > 150:
            max_iters = 180_000
            restarts = 2
        else:
            max_iters = 80_000
            restarts = 1

        for r in range(restarts):
            if time.time() >= deadline - 5:
                break
            # Dirichlet start (alpha sampled to span dense to sparse)
            alpha = np.random.uniform(0.2, 3.0)
            a0 = np.random.gamma(alpha, 1.0, size=n)
            a0 = a0 / a0.sum()
            init_score = fast_score(a0)
            print(f"Restart {r + 1}/{restarts} – init score {init_score:.6f}")

            # ------------------------------------------------------
            # 3️⃣  Smooth optimisation (soft‑max approximation)
            # ------------------------------------------------------
            opt_deadline = deadline - 5.0
            refined_a, refined_score = softmax_optimize(
                a0,
                deadline=opt_deadline,
                max_iters=max_iters,
                beta0=0.5,
                beta_max=200.0,
                eta0=0.5,
                report_every=0,
            )
            if refined_score < best_val:
                best_val = refined_score
                best_seq = refined_a.tolist()
                print(f"  Refined best score {refined_score:.6f}")

            # ------------------------------------------------------
            # 4️⃣  Final polishing with simulated‑annealing hill‑climb
            # ------------------------------------------------------
            if time.time() < deadline - 5.0:
                hill_deadline = deadline - 2.0
                polished_seq, polished_val = hill_climb_sa(
                    np.array(best_seq, dtype=float),
                    deadline=hill_deadline,
                    max_steps=300_000,
                )
                if polished_val < best_val:
                    best_val = polished_val
                    best_seq = polished_seq
                    print(f"  Hill‑climb improved to {polished_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Safeguard – tiny uniform vector if nothing succeeded
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 6️⃣  Final evaluation (exact) and return
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f} (length {len(best_seq)})")
    return best_seq
