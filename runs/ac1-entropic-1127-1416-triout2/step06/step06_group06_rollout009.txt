def simulated_annealing_hill_climb(seq, deadline, max_steps=1_000_000):

    a = np.array(seq, copy=True)
    n = a.shape[0]
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    step_scale = 0.1 # fraction of mass moved each step maybe.

    T = 0.02 # initial temperature
    T_min = 1e-6
    decay = 0.99995 # per iteration

    steps = 0
    while steps < max_steps and time.time() < deadline:
        # random move: choose i, j
        i, j = np.random.randint(n, size=2)
        if i==j:
            continue

        # adapt move size based on a[i]
        delta = step_scale * a[i] * np.random.rand()
        if delta <= 0:
            continue
        # ensure not moving more than a[i]
        if delta > a[i]:
            delta = a[i]

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # Non-negativity automatically (since delta <= a[i])

        cand_score = compute_score(cand)

        diff = cand_score - cur_score
        accept = False
        if diff <= 0:
            accept = True
        else:
            # accept with probability
            prob = np.exp(-diff / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            # optionally increase step_scale
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            # reject, maybe shrink step_scale
            step_scale *= 0.99

        # temperature decay
        T = max(T * decay, T_min)

        steps += 1

    return best_a.tolist(), best_score

```python
"""...
"""
import time
import numpy as np

# FFT utils
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors using an FFT of power‑of‑2 size."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

# Score for a vector that already sums to 1.
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector a."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)

# --------------------------------------------------------------
# Adam based smoothed max‑convolution optimisation (log‑space)
# --------------------------------------------------------------

def adam_smooth_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 5000,
    eta0: float = 0.15,
    lam_start: float = 1.0,
    lam_end: float = 120.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Optimise the smoothed objective
        f(a) = Σ_k w_k·(a*a)[k],   w = softmax(λ·conv(a,a)),
    using Adam updates on the log‑probability vector.
    The function works on the probability simplex (∑a = 1, a_i ≥ 0).

    Returns the best vector seen (as a NumPy array) and its true score
    (2·n·max(conv)).
    """
    # log‑probability parametrisation
    y = np.log(init_a + 1e-30)        # ensure positivity
    n = init_a.shape[0]

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = None
    best_score = float("inf")

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # a = softmax(y)  (probability simplex)
        a = softmax(y)

        # linear convolution a*a
        conv = convolution_fft(a, a)  # length 2n‑1
        M = np.max(conv)

        # schedule λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iters)

        # soft‑max weights centred at the current maximum
        w = softmax(lam * (conv - M))

        # gradient of the smooth surrogate w.r.t. a
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)  # length 3n‑2
        g_a = 2.0 * g_full[n - 1 : n - 1 + n]   # central n entries

        # gradient in the log‑space (chain rule)
        g_y = a * g_a

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (inverse‑sqrt decay)
        eta = eta0 / np.sqrt(it)

        # parameter update
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        y -= step

        # occasional exact evaluation
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}]  iter {it:,}  new best {best_score:.6f}")

    # final evaluation
    final_a = softmax(y)
    final_score = compute_score(final_a)
    if final_score < best_score:
        best_score = final_score
        best_a = final_a.copy()

    # if we never recorded a better vector, fall back to the final one
    if best_a is None:
        best_a = final_a.copy()

    return best_a, best_score

# Simulated‑annealing hill‑climber (exact objective)
# --------------------------------------------------------------

def simulated_annealing_hill_climb(
    seq,
    deadline,
    max_steps: int = 800_000,
    initial_temp: float = 0.02,
    final_temp: float = 1e-6,
    decay: float = 0.99995,
    step_scale_start: float = 0.12,
    step_scale_max: float = 0.5,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb that accepts worsening moves with a
    temperature that decays exponentially.  Moves consist of transferring
    a fraction of the current mass from one index to another.
    Returns the best vector (as a list) and its true score.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    T = initial_temp
    step_scale = step_scale_start

    steps = 0
    while steps < max_steps and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a random fraction of the mass from i to j
        delta = step_scale * a[i] * np.random.rand()
        if delta <= 0.0:
            continue
        # enforce non‑negativity
        if delta > a[i]:
            delta = a[i]

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = compute_score(cand)

        diff = cand_score - cur_score
        accept = False
        if diff <= 0.0:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-diff / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            # improve the global record
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # slowly enlarge the step size when we are succeeding
                step_scale = min(step_scale * 1.03, step_scale_max)
            else:
                # even on acceptance we shrink a bit to stay stable
                step_scale = max(step_scale * 0.99, step_scale_start / 2.0)
        else:
            # reject – make the proposal a bit more conservative
            step_scale = max(step_scale * 0.995, step_scale_start / 4.0)

        # cooling
        T = max(T * decay, final_temp)

        steps += 1

    return best_a.tolist(), best_score
```

The optimizer works on the probability simplex ( ∑ aᵢ = 1, aᵢ ≥ 0 ).  
The true objective is  

```
score(a) = 2·n·max(conv(a,a))               (n = len(a))
```

which is invariant under scaling, therefore we optimise on the simplex and
use the fast surrogate  

```
fλ(a) = Σk wₖ·(a*a)[k] , w = softmax(λ·conv(a,a))
```

When λ → ∞ the surrogate tends to the true max‑convolution.
The optimiser consists of three stages:

1. **Random restarts** – a Dirichlet draw (concentration α drawn uniformly
   in [0.5, 3]) gives diverse starting points.
2. **Adam‑based smoothed optimisation** – we keep a log‑parameter vector
   `y` (so that `a = softmax(y)` is automatically on the simplex) and
   minimise `fλ` with Adam.  The gradient of `fλ` w.r.t. `a` is  

   ```
   ∇ₐ fλ = 2·(a_rev * w)                (central n entries)
   ```

   The gradient in log‑space is `∇_y = a * ∇ₐ`.  λ is increased linearly
   from `lam_start` to `lam_end` during a run.
3. **Simulated‑annealing hill‑climber** – after the gradient phase we
   perform a stochastic mass‑transfer move (move a fraction of the mass
   from one index to another).  Moves that improve the exact score are
   always accepted; worsening moves are accepted with Metropolis
   probability `exp(−Δ/ T)` where the temperature `T` decays
   exponentially.

All heavy work (convolutions) is performed by a real‑FFT routine,
so each iteration is `O(n log n)`.  The whole search respects the
1000‑second budget and prints intermediate progress.

```python
"""
Hybrid optimiser for the step‑function attack.

The algorithm minimises
        score(a) = 2·n·max(conv(a,a))
under aᵢ≥0, Σaᵢ=1.  It uses:

1️⃣ Random Dirichlet restarts.
2️⃣ Adam‑based smoothed‑max optimisation in log‑space
   (softmax surrogate with an increasing λ).
3️⃣ Simulated‑annealing hill‑climbing on the exact objective.

All helper functions are defined at top level; the entry point is
`propose_candidate`, which returns a list of floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real FFT, power‑of‑2 length)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT.
    Result length = len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast exact objective (sum of a is 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Numerically stable softmax
# ----------------------------------------------------------------------


def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 4000,
    eta0: float = 0.18,
    lam_start: float = 1.0,
    lam_end: float = 120.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate

        fλ(a) = Σₖ wₖ·(a*a)[k],  w = softmax(λ·conv(a,a))

    on the probability simplex (a = softmax(y)).
    Returns the best vector encountered and its exact score.
    """
    # log‑probability vector
    y = np.log(init_a + 1e-30)
    n = init_a.shape[0]

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = None
    best_score = float("inf")

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # a is always a probability vector
        a = softmax(y)

        # linear convolution a*a
        conv = convolution_fft(a, a)          # length 2n‑1
        M = np.max(conv)

        # λ ramps up linearly
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iters)

        # soft‑max weights centred at the current max
        w = softmax(lam * (conv - M))

        # gradient w.r.t. a
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)    # length 3n‑2
        g_a = 2.0 * g_full[n - 1 : n - 1 + n]

        # gradient in the log‑space (chain rule)
        g_y = a * g_a

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (inverse‑sqrt decay)
        eta = eta0 / np.sqrt(it)

        # parameter step
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        y -= step

        # occasional exact evaluation
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}]  iter {it:,}  new best {best_score:.6f}")

    # final check
    final_a = softmax(y)
    final_score = compute_score(final_a)
    if final_score < best_score:
        best_score = final_score
        best_a = final_a.copy()

    if best_a is None:
        best_a = final_a.copy()

    return best_a, best_score
```
