"""
Improved search for a low‑value step sequence.

The algorithm works in two complementary phases:

1️⃣  **LP‑driven mass relocation** – Starting from a simple “U‑shaped’’ arcsine
    distribution we repeatedly solve a linear program that pushes as much
    probability mass as possible onto indices where the current convolution
    is small.  The LP problem is  

        maximise   Σ b_i
        subject to (a ∗ b)_k ≤ M   for all k,
                   b_i ≥ 0,

    where `M = max(a ∗ a)`.  The optimal `b` is renormalised to a new
    distribution `g` (Σ g = 1) and we mix the current vector `a` with `g`.
    Rather than using a tiny mixing factor we perform a *line‑search* on the
    mixing weight `t∈(0,1]` and pick the value that gives the smallest possible
    maximum entry of the resulting self‑convolution.  The steps are
    guaranteed to non‑increase the official score and typically shrink the
    maximum quickly.

2️⃣  **Smooth‑max Adam polishing** – After a handful of LP steps the vector
    is handed to a standard Adam optimiser that minimises the true
    ``max(a ∗ a)`` (via a log‑sum‑exp surrogate, high temperature).  This
    final stage cleans up any remaining irregularities and usually yields a
    better bound than the pure gradient‑only approach used previously.

Both phases respect the global 1000 s budget, fall back to a uniform 256‑vector
if nothing else finishes, and return the best sequence found (as a list of
floats).  The routine is deterministic apart from the optional random seed.
"""

import time
from typing import List

import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT (real‑valued)."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of `a`."""
    return float(np.max(_convolution_fft(a, a)))


def _compute_score(a: np.ndarray) -> float:
    """Fast (official) score for a vector already normalised to Σa = 1."""
    n = a.shape[0]
    return 2.0 * n * _max_convolution(a)


# ----------------------------------------------------------------------
# Simplex projection (euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators – the classic arcsine density (U‑shaped)
# ----------------------------------------------------------------------


def _arcsine_initial(n: int) -> np.ndarray:
    """Return a symmetric arcsine‑type density on {0,…,n‑1}, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (used by Adam)
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray, lam: float) -> (np.ndarray, float):
    """
    Gradient of the log‑sum‑exp surrogate of max(a∗a).

    Returns (grad, surrogate value B) where
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv‑max)).
    """
    n = a.shape[0]
    conv = _convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights on the convolution entries
    B = max_c + np.log(sum_exp) / lam

    # Gradient of Σ_k w_k (a*a)_k = 2 * (a ⊛ w_rev)
    w_rev = w[::-1]
    conv_rev = _convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Adam optimisation of the true max (via the smooth surrogate)
# ----------------------------------------------------------------------


def _adam_max(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 1000.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> (np.ndarray, float):
    """
    Adam optimiser on the smoothed max‑convolution surrogate.
    Returns the best vector (projected onto the simplex) and its fast score.
    """
    a = _project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = _compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # temperature schedule (quadratic in iteration fraction)
        t_frac = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t_frac ** 0.5)

        grad, _ = _smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = _project_to_simplex(a_new)

        cur_sc = _compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5_000 == 0:
            print(
                f"    Adam‑max iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} lam={lam:.1f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# LP‑driven refinement (mass relocation)
# ----------------------------------------------------------------------


def _lp_refine(
    a: np.ndarray,
    t_mix_grid: np.ndarray,
    max_lp_iter: int,
    time_deadline: float,
    lp_tolerance: float = 1e-12,
) -> np.ndarray:
    """
    Perform up to `max_lp_iter` LP‑based refinement steps.

    In each step we:
        • compute the current maximal convolution value M,
        • solve   maximise Σ b_i   subject to (a∗b)_k ≤ M,
        • normalise the LP solution to a simplex vector g,
        • pick the mixing coefficient `t∈t_mix_grid` that minimises
          max((1−t)² a∗a + 2t(1−t) a∗g + t² g∗g),
        • replace a ← (1−t) a + t g (projected onto the simplex).

    The routine returns the final refined vector.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    ones_row = np.full(rows, np.inf)   # placeholder

    # Pre‑allocate the matrix that will be rebuilt each iteration
    A_dense = np.empty((rows, n), dtype=np.float64)

    for it in range(max_lp_iter):
        if time.time() >= time_deadline:
            break

        # current maximal entry of the self‑convolution
        M = np.max(_convolution_fft(a, a))

        # Build the convolution matrix A such that (A @ b) = a ∗ b
        for j in range(n):
            A_dense[j : j + n, j] = a
        # Solve the LP: maximise Σ b_i   ⇔   minimise -Σ b_i
        # Constraints: A @ b ≤ M  (one inequality per convolution entry)
        res = linprog(
            -np.ones(n),
            A_ub=A_dense,
            b_ub=np.full(rows, M),
            bounds=(0, None),
            method="highs",
            options={"presolve": True, "dual_feasibility_tolerance": 1e-9},
        )
        if not res.success:
            print(f"    LP failed at iteration {it}: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= lp_tolerance:
            break

        g = b / sum_b                     # normalised direction on the simplex
        # Pre‑compute the three convolutions needed for the quadratic blend
        conv_a = _convolution_fft(a, a)
        conv_g = _convolution_fft(g, g)
        conv_ag = _convolution_fft(a, g)

        # Search the best mixing coefficient on the supplied grid
        t_vals = t_mix_grid
        t_arr = t_vals[:, None]                     # shape (m,1)
        # quadratic combination of the three convolution vectors
        mix = ((1 - t_arr) ** 2) * conv_a + (2 * t_arr * (1 - t_arr)) * conv_ag + (t_arr ** 2) * conv_g
        max_vals = mix.max(axis=1)
        best_idx = np.argmin(max_vals)
        t_opt = t_vals[best_idx]

        # Form the new candidate and project back onto the simplex (numerical safety)
        a_candidate = (1 - t_opt) * a + t_opt * g
        a_candidate = _project_to_simplex(a_candidate)

        new_M = np.max(_convolution_fft(a_candidate, a_candidate))
        if new_M < M - lp_tolerance:
            a = a_candidate
            print(
                f"    LP iter {it:2d}: M reduced from {M:.6e} to {new_M:.6e} using t={t_opt:.3f}"
            )
        else:
            # No improvement – bail out early
            print(
                f"    LP iter {it:2d}: no improvement (M={M:.6e}), stopping LP phase."
            )
            break

    return a
