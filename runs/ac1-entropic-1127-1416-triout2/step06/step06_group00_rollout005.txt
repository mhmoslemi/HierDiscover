def convolution_fft(a,b):
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def smooth_adam_optimize(start_a, time_deadline, max_iters=100000, beta_start=5.0, beta_end=200.0, lr=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    # start_a positive array sum=1
    n = start_a.shape[0]
    # logits
    z = np.log(start_a + 1e-20)  # small epsilon
    m_vec = np.zeros_like(z)
    v_vec = np.zeros_like(z)
    best_a = np.exp(z - np.max(z))
    best_a = best_a / np.sum(best_a)
    best_score = compute_score(best_a)
    # iteration loop
    for t in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        # compute a from logits
        a = softmax(z)
        # anneal beta
        t_frac = t / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac
        # conv
        conv = convolution_fft(a, a)
        # compute w
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()
        # gradient w.r.t a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n-1:n-1+n]
        # grad wr.t logits (z)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)
        # Adam updates
        m_vec = beta1 * m_vec + (1 - beta1) * grad_z
        v_vec = beta2 * v_vec + (1 - beta2) * (grad_z * grad_z)
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step
        # evaluate occasionally
        if t % 200 == 0:
            cur_a = softmax(z)
            cur_score = compute_score(cur_a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
    # final evaluation
    final_a = softmax(z)
    final_score = compute_score(final_a)
    if final_score < best_score:
        best_score = final_score
        best_a = final_a.copy()
    return best_a, best_score

"""
Improved minimiser for the auto‑convolution bound.

The search works in three phases:

1. **Fast Adam‑based smoothing** – we minimise the smooth
   approximation  fβ(a) = (1/β)·log ∑ exp(β·conv(a,a)).
   The variables are kept on the probability simplex by
   representing them as a soft‑max of unconstrained logits.
   Adam gives per‑coordinate step‑size adaptation, which is
   far more robust than the plain 1/√t step used before.

2. **Projected sub‑gradient refinement** – after the smooth
   phase we switch to true max‑objective (the maximum entry of
   the convolution) and perform many iterations of sub‑gradient
   descent with a simple Euclidean projection onto the simplex.

3. **Targeted hill‑climb** – a short stochastic mass‑transfer
   phase that actively reduces the peak of the convolution by
   moving a small fraction of the weight away from the indices
   that contribute most to the current maximum.

All phases respect the 1000 s wall‑clock limit, run only on a
single CPU and use only NumPy (no external LP‑solvers).  The
function `propose_candidate` is the required entry point.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimisation of the smoothed max‑objective
# ----------------------------------------------------------------------


def smooth_adam_optimize(start_a: np.ndarray,
                         time_deadline: float,
                         max_iters: int = 120_000,
                         beta_start: float = 5.0,
                         beta_end: float = 200.0,
                         lr: float = 0.05,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv(a,a)) with Adam.

    Parameters
    ----------
    start_a : np.ndarray
        Positive vector that already sums to 1.
    time_deadline : float
        Absolute unix time when the routine must stop.
    max_iters : int
        Upper bound on iterations (the routine may stop earlier).
    beta_start, beta_end : float
        Annealing schedule for the log‑sum‑exp smoothing parameter.
    lr : float
        Base learning rate for Adam.
    beta1, beta2, eps : Adam hyper‑parameters.

    Returns
    -------
    best_a, best_score
        Best vector (still on the simplex) and its exact (non‑smooth) score.
    """
    n = start_a.shape[0]

    # initialise logits – they may be any real numbers
    z = np.log(start_a + 1e-20)

    m_vec = np.zeros_like(z)
    v_vec = np.zeros_like(z)

    # keep the best solution seen
    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ----- current point on the simplex -----
        a = softmax(z)

        # ----- anneal β (smoothness) -----
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # ----- smoothed objective (log‑sum‑exp) -----
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))  # stabilised
        w = exp_weights / exp_weights.sum()         # probabilities, length 2n‑1

        # ----- gradient w.r.t. a (see derivation in the description) -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]    # size n

        # ----- chain rule to obtain gradient w.r.t logits -----
        dot = np.dot(a, grad_a)                     # scalar ⟨a,∇f⟩
        grad_z = a * (grad_a - dot)                 # ∇f wrt logits

        # ----- Adam update on logits -----
        m_vec = beta1 * m_vec + (1 - beta1) * grad_z
        v_vec = beta2 * v_vec + (1 - beta2) * (grad_z * grad_z)

        # bias‑corrected moments
        m_hat = m_vec / (1 - beta1 ** it)
        v_hat = v_vec / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # ----- occasional exact evaluation -----
        if it % 200 == 0:
            cur_a = softmax(z)
            cur_score = compute_score(cur_a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()

    # final check
    final_a = softmax(z)
    final_score = compute_score(final_a)
    if final_score < best_score:
        best_score = final_score
        best_a = final_a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 400_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent for  max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient (2·a shifted)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g

        # project onto the probability simplex (Euclidean)
        # --- Euclidean projection onto simplex (O(n log n)) ---
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.where(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
        else:
            theta = 0.0
        a = np.maximum(a_new - theta, 0.0)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape shallow plateaus
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            # re‑project quickly (reuse code above)
            u = np.sort(a)[::-1]
            cssv = np.cumsum(u) - 1.0
            ind = np.arange(1, n + 1)
            cond = u - cssv / ind > 0
            if np.any(cond):
                rho = np.where(cond)[0][-1]
                theta = cssv[rho] / (rho + 1.0)
            else:
                theta = 0.0
            a = np.maximum(a - theta, 0.0)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb polishing (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq: list[float],
               time_deadline: float,
               max_steps: int = 300_000) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb that preferentially attacks the peak of the
    convolution.  Each step either swaps two entries or moves a small
    fraction of mass away from the positions that contribute most to the
    current maximum.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    # keep best
    best_a = a.copy()
    best_score = compute_score(best_a)

    # scale for how much mass we move in a transfer move
    transfer_scale = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        # --- decide move type -----------------------------------------
        if np.random.rand() < 0.6:
            # targeted mass transfer based on the current peak
            conv = convolution_fft(a, a)
            k = int(np.argmax(conv))
            # pick an index that participates strongly in that peak
            i_candidates = np.where((k - np.arange(n) >= 0) &
                                    (k - np.arange(n) < n))[0]
            if i_candidates.size > 0:
                i = np.random.choice(i_candidates)
                j = k - i
                # move a fraction from the larger of the two to a random elsewhere
                if a[i] >= a[j]:
                    src, dst = i, np.random.randint(n)
                else:
                    src, dst = j, np.random.randint(n)
                delta = transfer_scale * a[src]
                cand = a.copy()
                cand[src] -= delta
                cand[dst] += delta
            else:
                # fallback to a random swap
                i, j = np.random.randint(n, size=2)
                cand = a.copy()
                cand[i], cand[j] = cand[j], cand[i]
        else:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # project back onto the simplex (soft‑max style)
        # we reuse the Euclidean projection used in sub‑grad
        u = np.sort(cand)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.where(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
        else:
            theta = 0.0
        cand = np.maximum(cand - theta, 0.0)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
            a = cand
            # become a bit more aggressive
            transfer_scale = min(transfer_scale * 1.07, 0.5)
        else:
            # be a little more conservative
            transfer_scale = max(transfer_scale * 0.97, 0.01)
            # keep current a unchanged
        max_steps -= 1

    return best_a.tolist(), best_score
