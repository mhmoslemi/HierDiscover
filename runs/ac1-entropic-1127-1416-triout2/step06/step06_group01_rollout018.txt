    power, mixtures) and keep the few best according to the surrogate
    score 2·n·max(conv) (the denominator Σa² cancels because the
    vectors are normalised to sum = 1).

2.  LP‑refinement – for the current best vector f (Σf=1) compute
    M = max(f * f).  Solve the linear program
          maximise   Σb
          subject to 0 ≤ b,  f * b ≤ M
    (the convolution is linear in b).  Scale the solution to unit
    sum, mix it back into f (convex combination) and repeat.
    Each LP iteration strictly reduces the supremum of the
    self‑convolution, driving the objective towards the theoretical
    optimum.

3.  Exact polishing – a long projected sub‑gradient descent on the
    true max‑convolution, followed by an exponentiated‑gradient (KL)
    polishing.  The official ``evaluate_sequence`` is called on the
    best candidate found.

All stages stop as soon as the global 1000 s deadline is reached.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate (Σa = 1) → 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """(i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Exact optimisation primitives
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.08) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·(a * mask)  (implemented via convolution)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent) polishing.
    """
    a = a.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (core novelty)
# ----------------------------------------------------------------------


def lp_refine(f: np.ndarray,
              deadline: float,
              max_iters: int = 30,
              mix: float = 0.5) -> np.ndarray:
    """
    Perform up to ``max_iters`` LP‑updates.
    Each iteration:
        * M ← max(f * f)
        * maximise Σb  subject to  b ≥ 0 and  f * b ≤ M
        * normalise b → g, then set f ← (1‑mix)·f + mix·g
    The routine returns the last f (still normalised to sum = 1).
    """
    from scipy.optimize import linprog

    n = f.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iters + 1):
        if time.time() >= deadline - 5.0:
            break

        # current worst‑case convolution value
        conv = convolution_fft(f, f)
        M = float(np.max(conv))

        # build the Toeplitz‑like constraint matrix A (size L×n)
        # row k, column j contains f_{k‑j} if 0≤k‑j<n else 0
        A = np.zeros((L, n), dtype=np.float64)
        # each column j is a copy of f placed starting at row j
        for j in range(n):
            A[j:j + n, j] = f

        # linear program: maximise Σb   ⇔ minimise −Σb
        c = -np.ones(n, dtype=np.float64)      # objective for linprog
        bounds = [(0.0, None)] * n               # b ≥ 0
        b_ub = M * np.ones(L, dtype=np.float64)  # f * b ≤ M

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            print(f"   LP iteration {it} failed ({res.message}); aborting LP.")
            break

        b = res.x
        if b.sum() == 0.0:
            print(f"   LP iteration {it}: zero‑mass solution – stop.")
            break

        # normalise to a probability vector
        g = b / b.sum()

        # convex update
        f = (1.0 - mix) * f + mix * g
        f /= f.sum()   # re‑normalise against tiny numerical drift

        # occasional logging with the official evaluator
        if it % 5 == 0:
            try:
                val = evaluate_sequence(f.tolist())
                print(f"   LP it={it:2d}, official score={val:.6f}, M={M:.8f}")
            except Exception:
                print(f"   LP it={it:2d}, surrogate score={compute_score(f):.6f}")

    return f


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The procedure mixes analytic seeding, a linear‑programming refinement,
    and a long exact polishing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback from a previous run (if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Diverse seeding (fast surrogate)
    # ------------------------------------------------------------------
    lengths = [1024, 2048]          # keep dimensions modest for the LP step
    pool: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break

        print(f"\n--- Seeding with length {n} ---")

        # deterministic starters
        uniform = np.full(n, 1.0 / n)
        pool.append((compute_score(uniform), uniform.copy()))
        arc = arcsine_initial(n)
        pool.append((compute_score(arc), arc.copy()))

        # random beta shapes (symmetric for speed)
        for _ in range(250):
            if time.time() >= deadline - 10.0:
                break
            a = rng.uniform(0.1, 5.0)
            b = a                     # keep symmetry
            eps = rng.uniform(0.0, 0.02 * n)
            cand = generate_beta_shape(n, a, b, eps)
            pool.append((compute_score(cand), cand))

        # random power‑shape (p close to 0.5 is arcsine‑like)
        for _ in range(250):
            if time.time() >= deadline - 10.0:
                break
            p = rng.uniform(0.2, 1.5)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((compute_score(cand), cand))

        # mixtures arcsine / uniform
        for _ in range(150):
            if time.time() >= deadline - 10.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((compute_score(cand), cand))

        # triangular bumps (useful for extreme‑mass patterns)
        for _ in range(150):
            if time.time() >= deadline - 10.0:
                break
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

    # keep the top few seeds (they become starting points for LP)
    pool.sort(key=lambda x: x[0])
    seeds = [vec for _, vec in pool[:6]]
    print(f"\nSelected {len(seeds)} seeds for LP refinement "
          f"(surrogate range {pool[0][0]:.6f} – {pool[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2.  LP‑based refinement (core improvement step)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(seeds, start=1):
        if time.time() >= deadline - 15.0:
            break
        n = seed_vec.shape[0]
        print(f"\n>>> LP refinement of seed {idx}/{len(seeds)} (n={n})")
        refined = lp_refine(seed_vec, deadline,
                            max_iters=30,          # enough for noticeable drop
                            mix=0.5)               # aggressive mixing

        # quick surrogate check
        surrogate = compute_score(refined)
        print(f"    Post‑LP surrogate score = {surrogate:.6f}")

        # store candidate for possible exact polishing later
        pool.append((surrogate, refined))

    # ------------------------------------------------------------------
    # 3.  Choose the best surrogate and perform exact polishing
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    candidate = pool[0][1]
    n = candidate.shape[0]
    print(f"\n--- Exact polishing (n={n}) ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(400_000, remaining * 30))
    candidate = subgradient_descent_exact(candidate,
                                          deadline,
                                          max_iter=max_iter_sub,
                                          step0=0.09)

    if time.time() < deadline - 3.0:
        remaining = deadline - time.time()
        max_iter_eg = int(min(250_000, remaining * 30))
        candidate = eg_refine(candidate,
                              deadline,
                              max_iter=max_iter_eg,
                              eta0=0.20)

    # official evaluation
    try:
        val = evaluate_sequence(candidate.tolist())
        print(f"Official score after polishing = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = candidate.tolist()
            print("*** New global best discovered! ***")
    except Exception as e:
        print(f"Evaluation failed after polishing: {e}")

    # ------------------------------------------------------------------
    # 4.  Final global polish (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\n--- Final global polish ---")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.07)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(200_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.18)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – ensure a non‑empty sequence is returned
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
