"""
Search for a step‑function sequence with a minimal value of
``evaluate_sequence``.  The algorithm proceeds as follows:

1.  Choose a single length (2048 gives a good balance between
    expressiveness and speed).  Generate a rich pool of parametric
    seeds:
        – arcsine shape,
        – uniform,
        – a mixture of arcsine & uniform,
        – “U‑shaped” power families,
        – symmetric beta families (α≈β<1).
    All seeds are normalised to sum to 1.

2.  Keep the few seeds with the smallest exact objective
    ``2·n·max(conv(a,a))`` (the same value returned by
    ``evaluate_sequence`` for Σa=1).

3.  For each selected seed run a *smooth‑max* Adam optimiser.
    The max in the convolution is replaced by a log‑sum‑exp with a
    temperature β that is increased linearly from 5 up to 300.
    The parameters are stored in unconstrained logits ``z`` and
    mapped to a simplex‑point by a soft‑max, ensuring positivity and
    Σa=1 automatically.

4.  The Adam result is refined by a fast *projected sub‑gradient*
    descent.  The sub‑gradient of ``max(conv)`` is obtained from the
    tight convolution entries; after each gradient step the vector is
    projected onto the probability simplex by a cheap clip‑and‑renorm
    operation.

5.  Finally a tiny random local search moves a small amount of mass
    from indices with a large sub‑gradient to those with a small one.
    This helps to escape shallow plateaus.

All steps respect the 1000 s wall‑clock budget; the best sequence
found before the deadline is returned (and stored in the global
``height_sequence_1`` variable for possible reuse).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for moderate n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already satisfies Σa = 1:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑type shape (Beta(½,½) density discretised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Symmetric Beta(α,β) shape (α and β may differ)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """U‑shaped shape a_i ∝ ((i+½)(n‑½‑i)+eps)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Soft‑max parametrisation utilities
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max surrogate
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 300.0,
    ramp_iters: int = 50_000,
    lr: float = 0.10,
) -> np.ndarray:
    """
    Minimise a smooth approximation of max(conv) using Adam.
    The iterates stay on the simplex via the soft‑max map.
    ```
    β(t)  – linear ramp from beta0 to beta_max
    w    – soft‑max weights ∝ exp(β·(conv‑mx))
    grad – 2·(a ⋆ w_rev) (restricted to original length)
    ```
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        # simplex point
        a = softmax(z)

        # convolution a*a
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # temperature
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx               # ≤ 0
        w = np.exp(beta * shifted)        # in (0,1]
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(
                f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}",
                flush=True,
            )

    # final simplex point
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (clip‑and‑renorm)
# ----------------------------------------------------------------------


def simple_simplex_proj(v: np.ndarray) -> np.ndarray:
    """
    Fast (approximate) Euclidean projection onto the simplex:
    clip to ≥0 and renormalise to sum 1.
    This is sufficient for our sub‑gradient descent and far cheaper
    than the exact O(n log n) projection.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

# Projected sub‑gradient descent (fast version)
# ----------------------------------------------------------------------


def subgradient_descent_fast(
    a_init: np.ndarray,
    deadline: float,
    step0: float = 0.20,
    max_iter: int = 500_000,

    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses the cheap clip‑and‑renorm projection.
    """
    a = simple_simplex_proj(a_init.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # consider all indices that are (numerically) tight
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2·(a ⋆ mask_rev) restricted to original length
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = simple_simplex_proj(a)

        if it % 10_000 == 0:
            print(
                f"    SubGrad it={it:,d} score={compute_score(a):.6f}",
                flush=True,
            )

    return a


# ----------------------------------------------------------------------
# Random local mass‑transfer search (tiny refinement)
# ----------------------------------------------------------------------


def random_local_move(
    a: np.ndarray,
    best_score: float,
    deadline: float,
    move_factor: float = 0.02,
    batch: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Perform a number of cheap random mass‑transfer steps guided by the
    sub‑gradient: move a small fraction of probability from a high‑gradient
    index to a low‑gradient index.
    """
    n = a.size
    while time.time() < deadline:
        # compute sub‑gradient once per batch
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        g = 2.0 * rev[n - 1 : n - 1 + n]

        # pick source (largest g) and destination (smallest g)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or a[src] <= 1e-15:
            # fallback to random pair if pathological
            src = np.random.randint(n)
            dst = np.random.randint(n)
            while src == dst:
                dst = np.random.randint(n)

        eps = min(move_factor * a[src], a[src])
        if eps <= 0:
            break

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        # projection (cheap)
        a_new = simple_simplex_proj(a_new)

        val = compute_score(a_new)
        if val < best_score:
            a, best_score = a_new, val
            # report occasional improvements
            print(f"    Random move improved score to {best_score:.6f}", flush=True)

        # limit loop iterations per call
        if batch <= 0:
            break
        batch -= 1

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The routine follows the plan described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ----------------------------------------------------------------------
    # 1. Choose a work‑length (2048 offers a good trade‑off)
    # ----------------------------------------------------------------------
    n = 2048
    print(f"\n=== Starting search (n={n}) ===", flush=True)

    # ----------------------------------------------------------------------
    # 2. Build a large set of diverse seeds
    # ----------------------------------------------------------------------
    seeds = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),
        generate_mixture_shape(n, 0.5),
    ]

    # power‑U family
    for p in np.linspace(0.3, 3.0, 9):
        seeds.append(generate_power_shape(n, p, eps=0.5))

    # symmetric beta family (α=β)
    for alpha in np.linspace(0.2, 0.95, 8):
        seeds.append(generate_beta_shape(n, alpha, alpha, eps=0.2))

    # a few random beta shapes
    for _ in range(6):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        eps = rng.uniform(0.0, 1.0)
        seeds.append(generate_beta_shape(n, a, b, eps))

    # ----------------------------------------------------------------------
    # 3. Keep the most promising seeds according to the exact score
    # ----------------------------------------------------------------------
    scored = [(compute_score(s), s) for s in seeds]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:4]]  # up to four seeds

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ----------------------------------------------------------------------
    # 4. Optimise each seed (most time is given to the best seed)
    # ----------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 10.0:
            break

        # allocate a slice of the remaining budget
        remaining = deadline - time.time()
        # give 60 % of the remaining time to the first seed,
        # then split what is left equally among the rest
        if idx == 1:
            per_seed = 0.60 * remaining
        else:
            per_seed = 0.40 * remaining / (len(top_seeds) - 1)

        seed_deadline = time.time() + per_seed
        print(
            f"\n--- Seed {idx}/{len(top_seeds)} – time budget ≈{per_seed:.1f}s ---",
            flush=True,
        )

        # --- Adam optimisation ------------------------------------------------
        eps = 1e-12
        z0 = np.log(seed_vec + eps)
        a_adam = smooth_adam_optimize(
            z0,
            seed_deadline,
            n=n,
            beta0=5.0,
            beta_max=300.0,
            ramp_iters=50_000,
            lr=0.10,
        )
        val_adam = compute_score(a_adam)
        print(f"    Adam finished – fast score = {val_adam:.6f}", flush=True)
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"    *** New global best (Adam) = {best_val:.6f} ***", flush=True)

        # --- Sub‑gradient refinement -----------------------------------------
        if time.time() < deadline - 5.0:
            sub_deadline = deadline - 2.0
            a_sub = subgradient_descent_fast(
                best_seq if best_seq is not None else a_adam,
                sub_deadline,
                step0=0.25,
                max_iter=400_000,
            )
            val_sub = compute_score(a_sub)
            print(f"    Sub‑gradient finished – score = {val_sub:.6f}", flush=True)
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"    *** New global best (SubGrad) = {best_val:.6f} ***", flush=True)

        # --- Light random local search ----------------------------------------
        if time.time() < deadline - 3.0:
            best_seq, best_val = random_local_move(
                best_seq,
                best_val,
                deadline - 0.5,
                move_factor=0.015,
                batch=2000,
            )
            print(f"    After random moves – best score = {best_val:.6f}", flush=True)

    # ----------------------------------------------------------------------
    # 5. Final polishing (if any time left)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n=== Final polishing with sub‑gradient descent ===", flush=True)
        final_seq = subgradient_descent_fast(
            best_seq,
            deadline,
            step0=0.12,
            max_iter=300_000,
        )
        final_val = compute_score(final_seq)
        print(f"    Final polishing score = {final_val:.6f}", flush=True)
        if final_val < best_val:
            best_val, best_seq = final_val, final_seq
            print("*** Global best improved by final polish ***", flush=True)

    # ----------------------------------------------------------------------
    # 6. Fallback safety net
    # ----------------------------------------------------------------------
    if best_seq is None:
        # a trivial feasible sequence (uniform) – never used in practice
        best_seq = np.full(n, 1.0 / n)
        best_val = compute_score(best_seq)

    # Store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()

    print(
        f"\nSearch finished. Best reported official score = {best_val:.6f}",
        flush=True,
    )
    return best_seq.tolist()
