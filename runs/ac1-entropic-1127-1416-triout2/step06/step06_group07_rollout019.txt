"""
search for a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1.  Choose a problem size `n` (default 2048).
2.  Scan a few low‑dimensional families of non‑negative vectors:
        * geometric  a_i ∝ r^i
        * power‑law  a_i ∝ (i+1)^(-p)
        * beta‑shape a_i ∝ x^{α‑1} (1‑x)^{β‑1}   (x = (i+½)/n)
        * half‑sine a_i ∝ sin(π·(i+½)/n)
   Every vector is normalised to Σa_i = 1 and its exact objective
   (the “fast score”) is evaluated with an FFT‑based convolution.
   The best vector found in this cheap global search becomes the
   starting point for the next phase.
3.  Refine the best candidate with a smooth‑max (soft‑max) mirror‑descent
   optimiser (the same routine used in the previous baseline) but with
   a larger temperature schedule (`β_max = 80`).  The optimiser works
   directly on the exact objective’s smooth approximation and keeps the
   iterate on the probability simplex.
4.  Finally a limited stochastic hill‑climber (swap / mass‑transfer moves)
   is applied to polish the solution.
5.  The best sequence discovered within the 1000 s budget is returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution, real‑valued)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective when Σa = 1
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """2·n·max(a*a) for a vector whose entries sum to 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps=150_000):
    """Stochastic hill‑climber using the *exact* objective."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = fast_score(best_a)

    step_scale = 0.1
    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:                     # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer mass i → j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # success → larger steps
        else:
            step_scale *= 0.97                         # failure → smaller steps

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Smooth‑max optimiser (soft‑max + exponentiated‑gradient)
# ----------------------------------------------------------------------


def smooth_optimize(a: np.ndarray,
                    deadline: float,
                    iters: int = 200_000,
                    beta0: float = 0.5,
                    beta_max: float = 80.0,
                    eta0: float = 0.4,
                    report_every: int = 20_000):
    """
    Mirror‑descent on the soft‑max approximation
        fβ(a) = (2·n/β)·log Σₖ exp(β·(a*a)[k]).
    Returns the best vector seen and its *exact* score.
    """
    n = a.shape[0]
    L = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, iters + 1):
        if time.time() >= deadline - 0.5:
            break

        # a*a  (FFT)
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]

        max_c = np.max(conv)

        # soft‑max weights
        beta = min(beta_max, beta0 * np.sqrt(t))
        # stabilise exponentials
        exp_terms = np.exp(beta * (conv - max_c))
        w = exp_terms / np.sum(exp_terms)

        # gradient = 2·(w ⋆ a_rev)
        a_rev = a[::-1]
        grad_full = np.fft.irfft(np.fft.rfft(a_rev, n=L) *
                                 np.fft.rfft(w, n=L), n=L)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # exponentiated update + simplex projection
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  iter {t:7d} – new best score {cur_val:.6f}")

    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Generator functions for the low‑dimensional families
# ----------------------------------------------------------------------


def make_geometric(r: float, n: int) -> np.ndarray:
    """a_i ∝ r^i  (r∈(0,1))."""
    a = r ** np.arange(n, dtype=float)
    a_sum = a.sum()
    return a / a_sum


def make_power(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (p>0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    return a / a.sum()


def make_beta(alpha: float, beta: float, n: int) -> np.ndarray:
    """Beta‑shape a_i ∝ x^{α‑1}(1‑x)^{β‑1},   x = (i+½)/n."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    # avoid exact zeros for stability
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def make_sine(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/n) (non‑negative on (0,π))."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(np.pi * x)
    return a / a.sum()
