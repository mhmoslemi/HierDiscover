from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp_for_g(f, M):
    n = len(f)
    K = 2*n - 1
    # Build matrix row-wise
    rows = []
    cols = []
    data = []
    for i in range(n):
        fi = f[i]
        if fi == 0:
            continue
        # This fi contributes to constraints k = i + j for each j in 0..n-1
        # so for each j, constraint index k = i + j
        # So for each j in 0..n-1, row = i + j, col = j, data = fi
        # We'll create many entries; better use loops
        for j in range(n):
            rows.append(i + j)
            cols.append(j)
            data.append(fi)
    A_ub = csr_matrix((data, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

"""
Improved step‑function search.
================================
The score to minimise is 2·n·max((a∗a)) under the constraint Σa = 1,
a_i ≥ 0 (the scaling cancels in the original objective).
The algorithm consists of three stages:

1. **Randomised smooth optimisation** – a soft‑max surrogate for the
   ∞‑norm is minimised by exponentiated (mirror) gradient descent
   with an annealing temperature β.  Several Dirichlet restarts are
   performed; the best vector obtained is kept.

2. **Linear‑programming refinement** – given the current vector f,
   we solve a LP that maximises the total mass of a second vector g
   while keeping the convolution f∗g bounded by the present peak M.
   After normalising g, a convex combination  f←(1−t)f+t g  reduces the
   maximum convolution further.  A few LP steps are applied.

3. **Stochastic hill‑climbing polish** – swap / mass‑transfer moves
   are attempted using the exact score (the true max‑convolution);
   successful moves enlarge the step size, otherwise it is shrunk.
   The result is returned as the final candidate.

All work respects the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# FFT utilities ----------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    conv = _conv_fft(a, a)
    return float(np.max(conv))


def _fast_score(a: np.ndarray) -> float:
    """Exact objective returned by `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    return 2.0 * n * _max_convolution(a)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) ----------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth optimisation (soft‑max of the convolution) -------------------------
def _smooth_optimize(
    a0: np.ndarray,
    deadline: float,
    iters: int = 200_000,
    beta0: float = 0.5,
    beta_max: float = 60.0,
    eta0: float = 0.4,
    report_every: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the soft‑max surrogate.
    Returns the best vector seen and its exact max‑convolution score.
    """
    n = a0.shape[0]
    L = _next_pow2(3 * n - 2)          # length sufficient for all FFTs
    a = a0.copy()
    best_a = a.copy()
    best_val = _fast_score(best_a)

    for t in range(1, iters + 1):
        if time.time() >= deadline - 1.0:
            break

        # ----- a*a ---------------------------------------------------------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = np.max(conv)

        # ----- soft‑max weights ---------------------------------------------
        beta = min(beta_max, beta0 * np.sqrt(t))
        # stable soft‑max
        exp_terms = np.exp(beta * (conv - max_c))
        w = exp_terms / np.sum(exp_terms)

        # ----- gradient = 2·(w ⋆ a_rev) ------------------------------------
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- exponentiated (mirror) update -------------------------------
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_to_simplex(a)

        # ----- occasional reporting -----------------------------------------
        if t % report_every == 0:
            cur_val = _fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  smooth iter {t:8d} – new best score {cur_val:.6f}")

    # final check
    final_val = _fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement ----------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> diags:
    """
    Build the (2n‑1)×n matrix A with A[k, j] = f[k‑j] (zero when out of range).
    This matrix encodes the linear map g ↦ f∗g.
    """
    n = f.size
    # each diagonal offset -d (d=0…n‑1) contains n copies of f[d]
    data = [np.full(n, f[d]) for d in range(n)]
    offsets = -np.arange(n)
    return diags(data, offsets, shape=(2 * n - 1, n), format="csr")


def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_j
        subject to  (f∗g)[k] ≤ M   for all k
                    g ≥ 0
    Returns the optimal g (not normalised) or None on failure.
    """
    n = f.size
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M * (1.0 + 1e-12))   # tiny slack for numerical safety
    c = -np.ones(n)                              # maximise Σg → minimise –Σg
    bounds = [(0, None)] * n

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        return None
    return res.x


def _lp_refine(a: np.ndarray, deadline: float, max_steps: int = 8, mix: float = 0.5):
    """
    Repeatedly apply the LP update described in the paper.
    Returns the best vector obtained and its exact score.
    """
    best = a.copy()
    best_val = _fast_score(best)

    for step in range(max_steps):
        if time.time() >= deadline - 1.0:
            break
        M = _max_convolution(best)
        g0 = _lp_step(best, M)
        if g0 is None:
            break
        sum_g = float(np.sum(g0))
        if sum_g <= 1.0 + 1e-12:
            # no improvement possible
            break
        g = g0 / sum_g          # renormalise to Σg = 1
        # convex combination
        best = (1.0 - mix) * best + mix * g
        best = _project_to_simplex(best)
        cur_val = _fast_score(best)
        if cur_val < best_val:
            best_val = cur_val
            print(f"  LP refinement step {step+1}: score = {cur_val:.6f}")
    return best, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climber ----------------------------------------------------
def _hill_climb(seq: list[float], deadline: float, max_steps: int = 250_000):
    """
    Simple hill climbing using swaps and fractional mass transfers.
    The exact (`evaluate_sequence`) score is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    a = _project_to_simplex(a)      # guarantee feasibility
    best_a = a.copy()
    best_score = _fast_score(best_a)

    step_scale = 0.1                # fraction of mass moved in a transfer

    n = a.shape[0]
    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = _project_to_simplex(cand)
        cur_score = _fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # success → larger steps
        else:
            step_scale *= 0.97                         # failure → smaller steps

        max_steps -= 1

    return best_a.tolist(), best_score
