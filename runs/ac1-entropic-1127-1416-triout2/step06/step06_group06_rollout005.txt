    a = np.array(seq)
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)
    step_frac = 0.1
    steps = 0
    start = time.time()
    while steps < max_steps and time.time() - start < time_budget:

        conv = convolution_fft(best_a, best_a)
        M = np.max(conv)
        # active mask (within small relative tolerance)
        tol = 1e-12
        active = np.abs(conv - M) <= tol
        # if multiple active, treat mask

        d = 2.0 * g_full[n-1:n-1+n]
        # choose i with max d, j with min d
        i = int(np.argmax(d))
        j = int(np.argmin(d))
        if i == j:
            # fallback random
            i, j = np.random.randint(n, size=2)
        # amount to move
        delta = step_frac * best_a[i]
        if delta <= 0:
            step_frac *= 0.9
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0)
        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac *= 0.97
        steps += 1
    return best_a.tolist(), best_sc

"""
Hybrid optimizer for the “step‑function attack”.

Goal
----
Minimise
    score(a) = 2·n·max(conv(a,a)) / (sum(a))**2
which is invariant under scaling.  Hence we work on the probability
simplex (∑a_i = 1, a_i ≥ 0) and minimise the surrogate
    f(a) = 2·n·max(conv(a,a))

Algorithm
---------
1. **Random restarts** – a Dirichlet draw with a random concentration
   (α∈[0.1,5]) yields a diverse initial point.

2. **Adam‑based smoothed max descent** – the non‑differentiable max is
   replaced by a soft‑max with a temperature λ that grows from ~1 to
   ~40.  The gradient of the smooth surrogate is
        g = 2·(a_rev ∗ w)          (central block only)
   where w are the soft‑max weights of the convolution entries.
   Adam’s adaptive moment estimates are used and after every step the
   iterate is projected back onto the simplex.

3. **Guided hill‑climbing** – after each descent phase we use the exact
   objective.  The indices that most influence the current maximum are
   obtained from the sub‑gradient
        d = 2·(a_rev ∗ 1_{conv≈max}) .
   A fraction of the mass is moved from the “worst’’ index (largest d)
   to the “best’’ one (smallest d).  The step size adapts to the
   acceptance ratio.

4. **Optional LP polishing** – for the current best vector `a` we
   construct the linear program
        maximise   ∑ g_j
        s.t.       conv(a,g) ≤ M   (M = max(conv(a,a)))
                    g_j ≥ 0
   (the convolution with a fixed `a` is linear in `g`).  If the LP
   returns a vector with a larger total mass we mix it with the current
   candidate; usually this yields a tiny extra drop in the true
   objective.

All heavy work (convolutions) is performed with a real FFT for
speed.  The routine runs for the whole time budget (default 1000 s) and
returns the best list of non‑negative heights found.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT (fast).
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast objective (sum is forced to 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))  –  assumes Σa = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Adam optimisation of the smooth surrogate of max(conv)
# ----------------------------------------------------------------------


def adam_smooth_descent(
    init_a: np.ndarray,
    time_budget: float,
    eta0: float = 0.5,
    lam_start: float = 1.0,
    lam_end: float = 40.0,
    eval_every: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed max of the convolution.
    Returns the best vector encountered and its true score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_sc = compute_score(a)

    it = 0
    start = time.time()
    while time.time() - start < time_budget:
        it += 1

        # convolution of a with itself
        conv = convolution_fft(a, a)            # length 2n‑1
        M = np.max(conv)

        # temperature schedule – slowly sharpen the soft‑max
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / 500_000)

        # soft‑max over the convolution entries (stable version)
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)        # length 2n‑1, sums to 1

        # gradient of the smooth surrogate
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)     # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # Adam update (we minimise → move opposite the gradient)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta0 / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = np.maximum(a, 0.0)                 # keep non‑negative
        a = project_to_simplex(a)               # back onto simplex

        # periodic true‑objective evaluation
        if it % eval_every == 0:
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]  iter {it:,}  new best {best_sc:.6f}")

    # final check
    final_sc = compute_score(a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Guided hill‑climbing using the exact max‑conv sub‑gradient
# ----------------------------------------------------------------------


def guided_hill_climb(
    seq,
    time_budget: float,
    max_steps: int = 800_000,
    init_step: float = 0.15,
) -> tuple[list, float]:
    """
    Moves a fraction of the mass from the index that most contributes to
    the current maximal convolution to the index that contributes the
    least.  The step size adapts to recent successes.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = init_step
    steps = 0
    start = time.time()

    while steps < max_steps and time.time() - start < time_budget:
        # current convolution and its maximum
        conv = convolution_fft(best_a, best_a)
        M = np.max(conv)

        # indices whose value is (almost) maximal – tolerance to catch ties
        tol = max(1e-12, 1e-9 * M)
        active = np.abs(conv - M) <= tol
        w_active = active.astype(float)          # length 2n‑1

        # sub‑gradient d = 2·(a_rev ∗ w_active) (central n entries)
        a_rev = best_a[::-1]
        d_full = convolution_fft(a_rev, w_active)   # 3n‑2 long
        d = 2.0 * d_full[n - 1 : n - 1 + n]

        # pick source (large d) and target (small d)
        i = int(np.argmax(d))
        j = int(np.argmin(d))
        if i == j:
            i, j = np.random.randint(n, size=2)

        delta = step_scale * best_a[i]          # amount to move
        if delta <= 0:
            step_scale *= 0.9
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cur_sc = compute_score(cand)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # grow a little
        else:
            step_scale *= 0.97                         # shrink
        steps += 1

    return best_a.tolist(), best_sc


# ----------------------------------------------------------------------
# Optional LP polishing – exploits linearity of conv(a,·)
# ----------------------------------------------------------------------


def lp_polish(a: np.ndarray, time_budget: float) -> np.ndarray:
    """
    Solves   max ∑g   subject to conv(a,g) ≤ M, g ≥ 0,
    where M = max(conv(a,a)).  The solution is mixed back with `a`
    (simple 50‑50 average) if it yields an improvement.
    """
    n = a.shape[0]
    M = np.max(convolution_fft(a, a))

    # Build the (2n‑1)×n constraint matrix C where
    # (C g)_k = ∑_j a_{k-j} g_j   (conv(a,g)[k])
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(k)
            cols.append(i)
            data.append(a[j])                     # coefficient a_{k-i}
    C = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Linear program: maximise 1ᵀg  ↆ minimise -1ᵀg
    lp_res = scipy.optimize.linprog(
        c=-np.ones(n),
        A_ub=C,
        b_ub=np.full(2 * n - 1, M),
        bounds=[(0, None)] * n,
        method="highs",
        options={"time_limit": max(1.0, time_budget)},
    )
    if not lp_res.success:
        return a

    g_opt = lp_res.x
    if np.sum(g_opt) <= 1.0001:                     # no useful direction
        return a

    # Mix the two vectors (simple averaging) and renormalise
    mixed = 0.5 * (a + g_opt / np.sum(g_opt))
    mixed = project_to_simplex(mixed)

    # Accept only if we truly improve the true objective
    old_sc = compute_score(a)
    new_sc = compute_score(mixed)
    if new_sc < old_sc:
        print(f"[{time.time():.1f}] LP polishing improved score {old_sc:.6f} → {new_sc:.6f}")
        return mixed
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (`list[float]`) minimising
    `evaluate_sequence`.  The routine performs several random restarts,
    Adam‑based smoothed‑max descent, a guided hill‑climber and an
    optional LP polishing step.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # choose a working length (use fallback length if available)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]
    except Exception:
        n = 1024

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Use the provided fallback if it exists
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main optimisation loop (random restarts + Adam descent)
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline - 10.0:
        restart_counter += 1
        alpha = np.random.uniform(0.1, 5.0)
        a0 = np.random.dirichlet(np.full(n, alpha))

        # give each start roughly the same amount of remaining budget,
        # keep a few seconds for the final polishing
        remaining = deadline - time.time()
        run_budget = max(5.0, remaining * 0.6)

        print(f"[{time.time():.1f}] Restart {restart_counter}, α={alpha:.2f}")

        a_opt, sc_opt = adam_smooth_descent(
            a0,
            time_budget=run_budget,
            eta0=0.55,
            lam_start=0.8,
            lam_end=38.0,
            eval_every=400,
        )
        if sc_opt < best_val:
            best_val = sc_opt
            best_seq = a_opt.tolist()
            print(f"   → new global best after descent {best_val:.6f}")

        # a quick guided hill‑climb on the fresh candidate
        if time.time() < deadline - 3.0:
            hc_seq, hc_val = guided_hill_climb(
                best_seq,
                time_budget=deadline - time.time(),
                max_steps=200_000,
            )
            if hc_val < best_val:
                best_val = hc_val
                best_seq = hc_seq
                print(f"   → hill‑climb improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing: a longer guided hill‑climb + optional LP step
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\nFinal polishing phase")
        # longer hill‑climb
        best_seq, best_val = guided_hill_climb(
            best_seq,
            time_budget=deadline - time.time(),
            max_steps=800_000,
        )
        # try LP polishing if time permits
        remaining = deadline - time.time()
        if remaining > 2.0:
            a_arr = np.array(best_seq, dtype=float)
            a_arr = lp_polish(a_arr, time_budget=remaining - 0.5)
            best_seq = a_arr.tolist()
            best_val = compute_score(a_arr)

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely event of failure
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished after {restart_counter} restarts.")
    print(f"Final reported score = {final_score:.6f}")

    return best_seq
