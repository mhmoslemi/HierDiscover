            if val < best_val:
                best_val = val
                best_seq = a_opt.tolist()
                print(f">>> Updated global best after refinement: {best_val:.6f}")

        # final Adam

        max_iter_adam = int(min(800_000, remaining * 40))

        # final walk

            max_iter_rw = int(min(300_000, (deadline - time.time()) * 45))
            a, _ = random_local_search(a, deadline, init_step=0.008, max_iter=max_iter_rw)
        final_val = evaluate_sequence(a.tolist())

        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated best after final polishing ***")

    if not best_seq:
        # fallback uniform
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback uniform used, score =", best_val)

    print(f"Search finished. Best official score = {best_val:.6f}")

"""
Propose a low‑value step sequence.

The search combines three phases:

1. **Broad evolutionary exploration** – for a handful of candidate
   lengths we keep a modest population (≈30 vectors) and evolve it with
   crossover, Gaussian mutation and projection onto the probability
   simplex.  The individuals are forced to be symmetric (a[i] = a[-i‑1])
   which greatly reduces the search space while preserving the objective.

2. **Intensive gradient refinement** – the best survivor of each length is
   refined with a long Adam run on a smooth log‑sum‑exp surrogate of the
   max‑convolution (final temperature λ≈800–1000).  Afterwards a projected
   sub‑gradient descent on the true max‑convolution and a cheap stochastic
   walk are applied.

3. **Global polishing** – the overall best vector found so far is polished
   again with Adam, sub‑gradient descent and a final random walk.

All time is shared across the phases; the routine respects the 1000 s
budget and always returns a non‑empty vector.  The official objective is
evaluated only for the few final candidates to keep the run fast.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective (Σa = 1 is assumed):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (beta families and arcsine)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float, beta1: float, eps1: float,
    alpha2: float, beta2: float, eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density – a useful deterministic starter."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (for Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)            # length 2n‑1
    max_c = np.max(conv)

    # soft‑max weights over convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                 # soft‑max probabilities
    B = max_c + np.log(sum_exp) / lam      # smooth approximation of the max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)   # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 400.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector seen (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees that the official objective never increases.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Simple Gaussian walk on the simplex with occasional uphill moves.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = fast_score(a_prop)

        # strict improvement or rare uphill acceptance
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

    return best_a, best_sc

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with low official score.
    The method combines an evolutionary stage, Adam + sub‑gradient polishing,
    and a final global refinement.  All time is shared across the phases and
    a safe fallback is guaranteed.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    best_seq: List[float] = []
    best_official: float = float("inf")

    # ------------------------------------------------------------------
    # Helper for the official evaluator (may be a bit slower, used sparingly)
    # ------------------------------------------------------------------
    def official_score(vec: np.ndarray) -> float:
        return evaluate_sequence(vec.tolist())

    # ------------------------------------------------------------------
    # 0.   Try the historic construction (if available)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = official_score(np.array(fallback))
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_official:
            best_official = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.   Evolutionary exploration on several lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Evolutionary stage – length {n} ===")

        # ---- initialise a modest population ---------------------------------
        pop: List[np.ndarray] = []
        scores: List[float] = []

        # deterministic starters (uniform and arcsine – both symmetric)
        starters = [
            np.full(n, 1.0 / n),
            arcsine_shape(n),
        ]
        for s in starters:
            pop.append(s)
            scores.append(fast_score(s))

        # random beta‑family shapes (including mixtures)
        for _ in range(15):
            if rng.random() < 0.35:   # mixture
                w = rng.random()
                alpha1 = rng.uniform(0.02, 2.0)
                beta1 = rng.uniform(0.02, 2.0)
                eps1 = rng.uniform(1e-9, 0.1 * n)
                alpha2 = rng.uniform(0.02, 2.0)
                beta2 = rng.uniform(0.02, 2.0)
                eps2 = rng.uniform(1e-9, 0.1 * n)
                cand = generate_beta_mixture(
                    n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
                )
            else:                     # single beta
                alpha = rng.uniform(0.02, 2.0)
                beta = rng.uniform(0.02, 2.0)
                eps = rng.uniform(1e-9, 0.1 * n)
                cand = generate_beta_shape(n, alpha, beta, eps)

            # enforce symmetry before projection
            cand = (cand + cand[::-1]) / 2.0
            cand = project_to_simplex(cand)

            pop.append(cand)
            scores.append(fast_score(cand))

        # fill to a fixed size (≈30)
        while len(pop) < 30:
            cand = rng.dirichlet(np.ones(n))
            cand = (cand + cand[::-1]) / 2.0
            cand = project_to_simplex(cand)
            pop.append(cand)
            scores.append(fast_score(cand))

        pop = np.stack(pop)          # shape (pop, n)
        scores = np.array(scores)

        # keep only the best half (elitism)
        idx = np.argsort(scores)[:pop.shape[0] // 2]
        pop = pop[idx]
        scores = scores[idx]

        # ---- evolutionary loop -------------------------------------------------
        generation = 0
        max_generations = 2000  # enough to consume a fair chunk of the budget
        while time.time() < deadline - 5.0 and generation < max_generations:
            # generate children (same number as current pop)
            children = []
            child_scores = []
            mut_scale = 0.12 / np.sqrt(generation + 1)

            for _ in range(pop.shape[0]):
                # tournament selection of two parents
                i1, i2 = rng.choice(pop.shape[0], size=2, replace=False)
                p1, p2 = pop[i1], pop[i2]

                # blend crossover
                alpha_c = rng.random()
                child = alpha_c * p1 + (1.0 - alpha_c) * p2

                # Gaussian mutation (zero‑mean, keep sum≈1)
                child += rng.normal(scale=mut_scale, size=n)

                # symmetry + simplex projection
                child = (child + child[::-1]) / 2.0
                child = project_to_simplex(child)

                children.append(child)
                child_scores.append(fast_score(child))

            children = np.stack(children)
            child_scores = np.array(child_scores)

            # combine and keep the best individuals
            combined = np.concatenate([pop, children], axis=0)
            combined_scores = np.concatenate([scores, child_scores])
            idx = np.argsort(combined_scores)[:pop.shape[0]]
            pop = combined[idx]
            scores = combined_scores[idx]

            # track global best (fast score)
            if scores[0] < best_official:
                best_official = scores[0]
                best_seq = pop[0].tolist()
                print(f"  *** New global best (len={n}) fast_score={best_official:.6f}")

            generation += 1

        # ---- gradient polishing of the local champion -------------------------
        if time.time() < deadline - 5.0:
            a_start = pop[0].copy()
            remaining = deadline - time.time()

            print(f"  Polishing length {n} with Adam (iters={max_iter_adam}) …")
            a_opt, _ = adam_optimize(
                init_a=a_start,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.10,
                lam_start=1.0,
                lam_end=800.0,
                noise_every=1200,
                noise_scale=3e-5,
            )

            # sub‑gradient descent (exact max)
            if time.time() < deadline - 5.0:
                max_iter_sub = int(min(180_000, (deadline - time.time()) * 30))
                print(f"  Sub‑gradient descent (iters={max_iter_sub}) …")
                a_opt, _ = subgradient_optimize(
                    a_opt,
                    deadline,
                    step0=0.28,
                    max_iter=max_iter_sub,
                )

            # final cheap Gaussian walk
            if time.time() < deadline - 3.0:
                max_iter_rw = int(min(260_000, (deadline - time.time()) * 35))
                print(f"  Random walk polish (iters={max_iter_rw}) …")
                a_opt, _ = random_local_search(
                    a_opt,
                    deadline,
                    init_step=0.012,
                    max_iter=max_iter_rw,
                )

            # official evaluation for this length
            val = official_score(a_opt)
            print(f"  → official score for length {n} = {val:.6f}")
            if val < best_official:
                best_official = val
                best_seq = a_opt.tolist()
                print(f"  *** Updated global best after refinement (len={n}) = {best_official:.6f} ***")

    # ------------------------------------------------------------------
    # 2. Global polishing of the overall best vector (if any left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # A – long Adam run (very high temperature)
        remaining = deadline - time.time()
        max_iter_adam = int(min(800_000, remaining * 40))
        print(f"  Adam polishing (iters={max_iter_adam}) …")
        a, _ = adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.09,
            lam_start=1.0,
            lam_end=1000.0,
            noise_every=1300,
            noise_scale=2e-5,
        )

        # B – exact sub‑gradient descent
        if time.time() < deadline - 4.0:
            max_iter_sub = int(min(220_000, (deadline - time.time()) * 30))
            print(f"  Sub‑gradient polishing (iters={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                deadline,
                step0=0.25,
                max_iter=max_iter_sub,
            )

        # C – final stochastic walk
        if time.time() < deadline - 2.0:
            max_iter_rw = int(min(320_000, (deadline - time.time()) * 45))
            print(f"  Random walk polishing (iters={max_iter_rw}) …")
            a, _ = random_local_search(
                a,
                deadline,
                init_step=0.009,
                max_iter=max_iter_rw,
            )

        final_val = official_score(a)
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_official = official_score(best_seq)
        print("Fallback to uniform 256‑vector, score =", best_official)

    print(f"\nSearch finished. Best reported official score = {best_official:.6f}")
    return best_seq
