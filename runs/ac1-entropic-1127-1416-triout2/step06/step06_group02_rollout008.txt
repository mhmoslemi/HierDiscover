def softmax_gradient_eg(a0, time_deadline, max_iter, lr0=1.0, eps_factor=0.02, epsilon_min=1e-4):
    a = a0.copy()
    loga = np.log(a + 1e-30)  # avoid log 0
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break

        a = np.exp(loga)
        a /= np.sum(a)  # projection onto simplex

        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        eps = max(epsilon_min, maxc * eps_factor * (0.95 ** (t - 1)))

        # stable softmax
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)

        # gradient
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        # get gradient slice
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # learning rate
        eta = lr0 / np.sqrt(t)  # monotonic decreasing

        # update loga: gradient descent on a -> adjust log
        # We need gradient w.r.t. loga? we approximate using chain rule: d f / d loga = (d f/d a) * a
        # Since a = exp(loga) / sum, but after normalization we recompute a fresh.
        # We'll approximate with multiplicative update: loga = loga - eta * grad * a
        # This is standard exponentiated gradient: new a_i ∝ a_i * exp(-eta * grad_i)
        # Equivalent to loga_i = loga_i - eta * grad_i
        # No need to multiply by a.
        loga = loga - eta * grad

        # optional jitter:
        if t % 2000 == 0:
            # random jitter on a
            jitter = np.random.randn(n) * (lr0 * 0.002 / np.sqrt(t))
            loga += jitter

        # Evaluate exact score
        # We'll recompute a after projection
        a_eval = np.exp(loga)
        a_eval /= np.sum(a_eval)
        cur_sc = compute_score(a_eval)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_eval.copy()
            # print progress
            # print...

    return best_a, best_sc

def optimize_sequence(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = 2 * n * np.max(convolution_fft(a, a))
    t=1
    while time.time() < deadline:
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # compute epsilon as scaling factor
        eps = maxc * max(1e-3, (0.9 ** t))
        # softmax
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / exp_vals.sum()

        conv_rev = convolution_fft(a, w_rev)

            best_sc = cur_sc
            best_a = a.copy()
        t += 1
        if t % 1000 == 0:
            # reduce epsilon factor gradually
            pass
    return best_a, best_sc

"""
Optimisation of the step‑function bound.

The algorithm works in three stages:

1.  **Random start** – a dense Dirichlet vector of a chosen length.
2.  **Projected gradient descent on a smooth soft‑max surrogate** of
    the max‑convolution.  An exponentially‑decaying ε drives the surrogate
    to the true max, while each iterate is projected onto the probability
    simplex (positivity + unit sum).  After every update the vector is
    symmetrised (a ← (a+rev(a))/2) – symmetry is known to flatten the
    autocorrelation.
3.  **Stochastic mass‑transfer hill‑climb** – tiny random transfers of
    probability mass between two coordinates are accepted only if they
    improve the exact score.
The whole search respects the 1000 s budget; the best sequence found
is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real‑valued)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – keeps non‑negativity and sum = 1
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # extreme case – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact score (sum of a is assumed to be 1)
# ----------------------------------------------------------------------


def exact_score_from_max(n: int, max_conv: float) -> float:
    """Score = 2·n·max(conv) because sum(a)=1."""
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# 1. Random dense start (Dirichlet / Gamma)
# ----------------------------------------------------------------------


def random_dirichlet(n: int, alpha: float = 0.5) -> np.ndarray:
    """Generate a dense probability vector of length n."""
    a = np.random.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 2. Projected smooth‑max gradient descent
# ----------------------------------------------------------------------


def smooth_max_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.8,
    eps_factor: float = 0.03,
    eps_min_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation
        fε(a) = ε·log∑_k exp(conv_k/ε)
    where conv = a⊛a.
    The gradient is 2·(w_rev ⊛ a) with w = soft‑max(conv/ε).
    Each iterate is projected onto the simplex and then symmetrised.
    """
    a = a0.astype(float).copy()
    n = len(a)

    # initialise best
    conv = convolution_fft(a, a)
    maxc = conv.max()
    best_score = exact_score_from_max(n, maxc)
    best_a = a.copy()

    # pre‑allocate for speed
    idx_slice = np.arange(n)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ---- convolution and soft‑max weights ----
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # ε shrinks geometrically but never below eps_min_factor·maxc
        eps = max(eps_min_factor * maxc, eps_factor * maxc * (0.95 ** (t - 1)))

        # stable soft‑max
        shifted = conv - maxc                     # avoid overflow
        exp_vals = np.exp(shifted / eps)
        w = exp_vals / exp_vals.sum()            # sum(w)=1

        # ---- gradient (2·(w_rev ⊛ a)) ----
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)     # length = 2n‑1
        # pick the slice that aligns with the gradient components
        # the last index of conv_rev is 2n‑2
        grad = 2.0 * conv_rev[(2 * n - 2) - idx_slice]

        # ---- step and projection ----
        step = step0 / np.sqrt(t)                # classic 1/√t decay
        a_new = a - step * grad
        # enforce symmetry (helps flatten autocorrelation)
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new = project_to_simplex(a_new)

        a = a_new

        # record best (exact) score – max(conv) is already known
        cur_score = exact_score_from_max(n, maxc)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional log
            if t % 1000 == 0:
                print(f"[GD] iter={t:,}  score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Simple hill‑climb (mass‑transfer) on the exact objective
# ----------------------------------------------------------------------


def hill_climb_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    max_steps: int = 300_000,
    init_scale: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber that moves a fraction of mass from one index
    to another.  Only moves that strictly improve the exact score are kept.
    The move‑size is adapted during the run.
    """
    a = a0.copy()
    n = len(a)
    conv = convolution_fft(a, a)
    best_score = exact_score_from_max(n, conv.max())
    best_a = a.copy()

    step_scale = init_scale

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        if delta <= 0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # re‑project to simplex (fast: we only disturbed two entries)
        cand = project_to_simplex(cand)

        conv_cand = convolution_fft(cand, cand)
        cur_score = exact_score_from_max(n, conv_cand.max())

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.97                         # become more cautious

        # occasional logging
        if (step + 1) % 30_000 == 0:
            print(f"[HC] steps={step+1:,}  best_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 4. Final symmetry averaging (cheap, sometimes improves)
# ----------------------------------------------------------------------


def symmetry_average(a: np.ndarray) -> np.ndarray:
    """Return (a + rev(a))/2 projected onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a plain
    Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback (if the notebook already contains a known sequence)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameters for the outer search
    # ------------------------------------------------------------------
    # larger lengths give a better theoretical lower bound, but also cost a
    # little more per iteration.  8192 is a good compromise.
    lengths = [4096, 8192, 12288]          # keep them powers of two for FFT speed
    outer_start = time.time()

    # we will try a modest number of restarts – each restart gets a long
    # optimisation budget because the problem is hard.
    max_outer = 6

    for outer in range(max_outer):
        if time.time() >= deadline - 30.0:
            break

        n = int(np.random.choice(lengths))
        # dense Dirichlet start (α<1 encourages some variability)
        a0 = random_dirichlet(n, alpha=0.4)

        # ------------------------------------------------------------------
        # Phase 1 – smooth‑max gradient descent (most of the budget)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 10.0:
            break
        gd_deadline = time.time() + min(remaining * 0.78, 600.0)  # leave time for polishing
        a_gd, sc_gd = smooth_max_gradient_descent(
            a0,
            deadline=gd_deadline,
            max_iter=250_000,
            step0=0.9,
            eps_factor=0.03,
            eps_min_factor=1e-4,
        )
        if sc_gd < best_score:
            best_score = sc_gd
            best_seq = a_gd.tolist()
            print(f"[outer {outer}] after GD (n={n})  score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – hill‑climb polishing (exact objective)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        hc_deadline = time.time() + min(remaining * 0.8, 200.0)
        a_hc, sc_hc = hill_climb_mass_transfer(
            np.array(best_seq, dtype=float),
            deadline=hc_deadline,
            max_steps=150_000,
            init_scale=0.02,
        )
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.tolist()
            print(f"[outer {outer}] after HC   (n={n})  score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Final cheap symmetry averaging (often gives a tiny boost)
        # ------------------------------------------------------------------
        a_sym = symmetry_average(np.array(best_seq, dtype=float))
        conv_sym = convolution_fft(a_sym, a_sym)
        sc_sym = exact_score_from_max(len(a_sym), conv_sym.max())
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[outer {outer}] symmetry avg improved: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if nothing was ever found, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
