"""
Propose a low‑score non‑negative step function.

The objective
    score = 2·n·max(conv(a,a)) / (sum(a)**2)
is scale‑invariant: we may fix Σa = 1 and minimise 2·n·max(conv).
For a geometric (exponential) distribution
    a_i ∝ r^i ,   0<r<1,
the convolution peak can be made arbitrarily small while the sum stays 1.
The code:
* searches the one‑parameter family a_i = (1‑r)·r^i / (1‑rⁿ) for the best r,
* also tries a simple power‑law family a_i ∝ (i+1)^{-p},
* refines the best candidate with a cheap stochastic hill‑climber,
* finally returns the best vector found (as a plain Python list).

All heavy work is done with NumPy and FFT‑based convolution, so the search
fits comfortably into the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution, real‑valued)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact score (the objective when the sum is 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute 2·n·max(a*a) for a vector whose entries sum to 1.
    This equals the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq, time_deadline, max_steps=200_000, seed=0
) -> tuple[list, float]:
    """
    Stochastic hill‑climber that swaps two entries or moves a fraction
    of mass between two positions.  Operates on vectors that already sum
    to 1 (the simplex) – after each move we simply re‑project.
    """
    rng = np.random.default_rng(seed)
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = fast_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if rng.random() < 0.5:
            # swap two positions
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cur_score = fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)   # success → larger steps
        else:
            step_scale *= 0.96                         # failure → smaller steps

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Strategy:
        1. Fix a length (2048 or 4096 if time permits).
        2. Exhaustively scan a one‑parameter exponential family
           a_i ∝ r^i (geometric distribution) – the parameter is
           c = n·(1‑r).  The theoretical optimum lies near c≈2.
        3. Scan a simple power‑law family a_i ∝ (i+1)^{-p}.
        4. Polish the best vector with a short hill‑climber.
    The whole search respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Choose a length (larger length gives a tighter theoretical bound)
    # --------------------------------------------------------------
    n = 4096
    if time.time() + 10 > deadline:
        n = 2048
    print(f"Using sequence length n = {n}")

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 1️⃣  Exponential (geometric) family search
    # --------------------------------------------------------------
    # The parameter we sweep is c = n·(1‑r).  Theory predicts the optimum
    # near c≈2, but we scan a wider window.
    c_grid = np.linspace(0.2, 5.0, 2500)  # 2500 points are cheap
    for c in c_grid:
        if time.time() >= deadline - 2.0:
            break

        r = 1.0 - c / n
        if not (0.0 < r < 1.0):
            continue

        # a_i = (1‑r)·r^i / (1‑rⁿ)  (already normalised to sum=1)
        log_r = np.log(r)
        i_arr = np.arange(n)
        a = (1.0 - r) * np.exp(log_r * i_arr)
        denom = 1.0 - r**n
        a /= denom

        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(
                f"New best (exp)  c={c:.4f}  r={r:.6f}  score={cur_val:.6f}"
            )

    # --------------------------------------------------------------
    # 2️⃣  Power‑law family search (a_i ∝ (i+1)^{-p})
    # --------------------------------------------------------------
    p_grid = np.linspace(0.05, 3.0, 300)
    for p in p_grid:
        if time.time() >= deadline - 2.0:
            break
        a = 1.0 / ((np.arange(n) + 1) ** p)
        a /= np.sum(a)
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"New best (pow)  p={p:.4f}  score={cur_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final hill‑climb polishing (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nRunning final hill‑climb polishing …")
        polished_seq, polished_val = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=deadline,
            max_steps=300_000,
            seed=seed,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = np.array(polished_seq, dtype=float)
            print(f"Hill‑climb improved score to {polished_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Safety net – very small uniform vector if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_val = fast_score(best_seq)

    # --------------------------------------------------------------
    # Report the final score (exact `evaluate_sequence`) and return
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq.tolist()
