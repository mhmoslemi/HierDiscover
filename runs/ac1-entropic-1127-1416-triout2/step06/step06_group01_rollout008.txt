propose_candidate – 2024‑11‑27‑improved

A three‑stage optimiser that builds on the previous implementation
and adds a few new ideas:

*   Searches three lengths (1024, 2048, 4096) – the larger length

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    ``a`` must be normalised to Σa = 1, therefore the denominator
    disappears and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Classic U‑shaped arcsine distribution."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_semicircle_shape(n: int) -> np.ndarray:
    """Wigner‑type semicircle density on the discrete grid."""
    i = np.arange(n, dtype=float)
    x = 2.0 * i / (n - 1) - 1.0          # map to [-1,1]
    vals = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    total = vals.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangle with apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_dirichlet_sparse(n: int, alpha: float, rng) -> np.ndarray:
    """Sparse Dirichlet draw (α<1 gives spiky vectors)."""
    vec = rng.gamma(alpha, 1.0, size=n)
    total = vec.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vec / total


def generate_spike_shape(n: int, k: int, rng) -> np.ndarray:
    """
    k large spikes (equal weight) plus a tiny uniform background.
    The vector is normalised to sum to 1.
    """
    if k >= n:
        return np.full(n, 1.0 / n)
    background = 1e-3
    spikes_weight = 1.0 - background * n
    spike_val = spikes_weight / k
    a = np.full(n, background)
    idx = rng.choice(n, size=k, replace=False)
    a[idx] = spike_val
    return a / a.sum()

    Adam optimiser on the smooth approximation of max(conv) using a
    temperature τ that grows linearly from *tau_start* to *tau_end*.
    The optimisation variable are soft‑max logits, which keep the
    iterate on the simplex (Σa=1, a_i≥0) automatically.
    """
    n = a_init.shape[0]
    L = 2 * n - 1
    # initialise logits w such that softmax(w) = a_init
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax → a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        conv = convolution_fft(a, a)

        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp

        # gradient wrt a (2·∑_k w_k·shift(a,k))
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # back‑prop through softmax
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 10_000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 400_000,
                              step0: float = 0.07) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the exact max‑convolution.
    The step size is step0 / √t.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev = mask[::-1]
        conv_rev = convolution_fft(a, rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 50_000 == 0:
            surrogate = compute_score(a)
            print(f"    Sub‑grad it={it:,d} surrogate={surrogate:.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.22) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent with KL geometry) polishing.
    """
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev = mask[::-1]
        conv_rev = convolution_fft(a, rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        tot = a.sum()
        if tot == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / tot

        if it % 50_000 == 0:
            surrogate = compute_score(a)
            print(f"    EG it={it:,d} surrogate={surrogate:.6f}")

    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence``.  The routine builds a diverse pool of
    parametric seeds, refines the most promising ones with a smooth‑max
    Adam optimiser, then runs a long exact sub‑gradient descent and a
    final exponentiated‑gradient polish.  If time permits, a cheap
    hill‑climbing perturbation loop is performed.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Fallback from a previous run (if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            # normalise to Σ=1 (the surrogate works with that)
            total = sum(fallback)
            if total > 0.0:
                fallback = [x / total for x in fallback]
            val = evaluate_sequence(fallback)
            print(f"Fallback candidate official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Build a diverse pool of seeds (fast surrogate)
    # --------------------------------------------------------------
    lengths = [1024, 2048, 4096]
    pool: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Seeding length {n} ---")

        # deterministic starters
        uniform = np.full(n, 1.0 / n)
        pool.append((compute_score(uniform), uniform.copy()))
        arc = arcsine_initial(n)
        pool.append((compute_score(arc), arc.copy()))
        semi = generate_semicircle_shape(n)
        pool.append((compute_score(semi), semi.copy()))

        # random beta family
        for _ in range(150):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.05, 5.0)
            b = rng.uniform(0.05, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.01 * n))
            cand = generate_beta_shape(n, a, b, eps)
            pool.append((compute_score(cand), cand))

        # random power‑shape
        for _ in range(150):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.05, 3.0)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((compute_score(cand), cand))

        # Dirichlet sparse draws (α < 1 gives spiky vectors)
        for _ in range(100):
            if time.time() >= deadline - 5.0:
                break
            alpha = rng.uniform(0.01, 0.5)
            cand = generate_dirichlet_sparse(n, alpha, rng)
            pool.append((compute_score(cand), cand))

        # Spike mixtures
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            k = rng.integers(2, min(20, n // 2))
            cand = generate_spike_shape(n, int(k), rng)
            pool.append((compute_score(cand), cand))

        # Triangular bumps
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

        # Mixture of arcsine and uniform
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((compute_score(cand), cand))

    # keep the very best seeds for the expensive Adam stage
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:12]]
    print(f"\nSelected {len(top_seeds)} top seeds for Adam optimisation "
          f"(best surrogate {pool[0][0]:.6f})")

    # --------------------------------------------------------------
    # 2.  Smooth‑max Adam optimisation (one pass per seed)
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 3.0:
            break
        n = seed_vec.size
        print(f"\n=== Adam refinement of seed {idx}/{len(top_seeds)} (n={n}) ===")
        remaining = deadline - time.time()
        # give each seed a modest budget; the 30 %‑40 % of remaining time
        # is typically enough for 30‑50 k iterations.
        max_iter = int(min(50_000, remaining * 25))
        a_adam = adam_optimize_smooth_max(seed_vec,
                                          deadline,
                                          max_iters=max_iter,
                                          tau_start=0.5,
                                          tau_end=200.0,
                                          base_lr=0.4)

        surrogate = compute_score(a_adam)
        print(f"    Post‑Adam surrogate = {surrogate:.6f}")

        # keep for final selection
        pool.append((surrogate, a_adam))

        # update best if this candidate already beats the global best
        if surrogate < best_val:
            best_val = surrogate
            best_seq = a_adam.tolist()
            print("*** New global best (surrogate) ***")

    # --------------------------------------------------------------
    # 3.  Exact refinement of the current best surrogate
    # --------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    candidate = pool[0][1]
    n = candidate.size
    print(f"\n--- Exact refinement (n={n}) ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(400_000, remaining * 30))
    refined = subgradient_descent_exact(candidate,
                                        deadline,
                                        max_iter=max_iter_sub,
                                        step0=0.07)

    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        max_iter_eg = int(min(200_000, remaining * 30))
        refined = eg_refine(refined, deadline,
                            max_iter=max_iter_eg,
                            eta0=0.22)

    # official evaluation
    try:
        val = evaluate_sequence(refined.tolist())
        print(f"Official score after exact refinement = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            print("*** New global best (official) ***")
    except Exception as e:
        print(f"Official evaluation failed: {e}")

    # --------------------------------------------------------------
    # 4.  Cheap hill‑climbing perturbation (if time remains)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 8.0:
        print("\n--- Local perturbation search ---")
        a = np.array(best_seq, dtype=float)
        n = a.size
        best_surrogate = compute_score(a)
        attempts = 0
        while time.time() < deadline - 1.0:
            attempts += 1
            i, j = rng.choice(n, size=2, replace=False)
            # move a tiny random amount from i to j
            delta = rng.uniform(0.0, min(a[i], 0.001))
            if delta == 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            s = compute_score(a_new)
            if s < best_surrogate:
                best_surrogate = s
                a = a_new
                best_seq = a.tolist()
        print(f"Hill‑climbing performed {attempts} attempts; "
              f"best surrogate after perturb = {best_surrogate:.6f}")

        try:
            val = evaluate_sequence(best_seq)
            print(f"Official after perturb = {val:.6f}")
            if val < best_val:
                best_val = val
                print("*** New global best after perturb ***")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 5.  Final polish (if any time left)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\n--- Final global polish ---")
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter_sub = int(min(250_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.06)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(120_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.18)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # 6.  Guard – ensure we always return a non‑empty sequence
    # --------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)
        print("Fallback uniform sequence used as a last resort.")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
