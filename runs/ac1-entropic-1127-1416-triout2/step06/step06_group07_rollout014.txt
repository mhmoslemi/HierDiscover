def subgradient_refine(a, deadline, steps=20000, eta0=0.5):
    best = a
    best_val = fast_score(a)
    n = len(a)
    L = next_pow2(3*n-2)
    for step in range(steps):
        if time.time()>deadline: break
        # compute conv
        conv = convolution_fft(a,a)
        max_val = np.max(conv)
        # find indices where value == max_val (allow tolerance)
        idx = np.where(conv >= max_val - 1e-12)[0]  # tolerance
        w = np.zeros_like(conv)
        w[idx] = 1.0/len(idx)
        # gradient via convolution with reversed a and w
        a_rev = a[::-1]
        grad = 2 * convolution_fft(a_rev, w)[n-1:n-1+len(a)]
        # determine step size (maybe decreasing)
        eta = eta0 / np.sqrt(step+1)
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
    return best, best_val

def conv_fft(a,b):
    m = len(a) + len(b) - 1
    L = 1 << (m-1).bit_length()
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = A*B
    return np.fft.irfft(C, n=L)[:m]

def hill_climb_sa(seq, deadline, max_steps=200000):
    a = np.array(seq)
    n = len(a)
    best = a.copy()
    best_val = fast_score(best)
    cur_val = best_val
    cur = a.copy()
    temperature = 0.1  # initial temperature
    cooling = 0.99995   # multiply each step
    for step in range(max_steps):
        if time.time() > deadline: break
        # candidate via random move
        cand = cur.copy()
        # choose move type (swap, transfer)
        if np.random.rand() < 0.5:
            i,j = np.random.randint(n,size=2)
            cand[i],cand[j] = cand[j],cand[i]
        else:
            i,j = np.random.randint(n,size=2)
            if i!=j:
                delta = 0.1 * cand[i]  # step fraction
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0: cand[i]=0
        # project onto simplex (to maintain sum=1)
        cand = project_to_simplex(cand)
        cand_val = fast_score(cand)

        d = cand_val - cur_val
        if d < 0 or np.random.rand() < np.exp(-d/temperature):
            cur = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
        # temperature decay
        temperature = max(temperature*cooling, 1e-6)
    return best.tolist(), best_val
```

assistantfinal```python
"""
Non‑negative step function optimiser.

The code searches for a probability vector `a` (sum = 1, a_i ≥ 0) that
minimises

    f(a) = 2·n·max_k (a*a)[k] ,

i.e. the evaluated score returned by `evaluate_sequence`.  The algorithm
mixes three ideas:

1. **Initialisations** – a few Dirichlet draws and edge‑biased draws.
2. **Two‑phase smooth optimisation**  
   * L2‑error minimisation (pushes the whole self‑convolution towards a
     flat distribution).  
   * Soft‑max approximation of the infinity‑norm (the classic
     log‑sum‑exp smoothing) with an exponentiated‑gradient update on the
     probability simplex.
3. **Final annealed hill‑climb** – stochastic mass transfers with a
   simulated‑annealing acceptance rule, again projected onto the simplex.

All work respects the time budget (≈ 1000 s).  The best sequence found
is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    return np.fft.irfft(C, n=n_fft)[:m]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – keeps a on the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # completely non‑positive – return uniform vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a*a)  (assumes Σa = 1)."""
    n = a.shape[0]
    conv = conv_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# 1. L2‑error optimiser (flattens the whole convolution)
# ----------------------------------------------------------------------


def l2_optimize(
    a: np.ndarray,
    time_deadline: float,
    iters: int = 80_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Minimises the L2 distance between a*a and the perfectly flat target.
    Returns the best vector seen and its exact `fast_score`.
    """
    n = a.shape[0]
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=float)

    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, iters + 1):
        if time.time() >= time_deadline - 0.5:
            break

        # current self‑convolution
        conv = conv_fft(a, a)            # length 2n‑1
        e = conv - target                # error vector

        # gradient of ½‖e‖² w.r.t. a:
        #   ∂L/∂a = 2 * ((e * a_rev) + (e_rev * a))[center]
        a_rev = a[::-1]
        conv1 = conv_fft(e, a_rev)       # length 3n‑2
        conv2 = conv_fft(e[::-1], a)     # length 3n‑2
        grad = 2.0 * (conv1[n - 1 : n - 1 + n] + conv2[n - 1 : n - 1 + n])

        eta = eta0 / np.sqrt(t)
        # exponentiated (mirror) update – clip exponent to avoid overflow
        upd = np.exp(np.clip(-eta * grad, -50, 50))
        a = a * upd
        a = project_to_simplex(a)

        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# 2. Smooth (soft‑max) optimiser – classic log‑sum‑exp smoothing
# ----------------------------------------------------------------------


def smooth_optimize(
    a: np.ndarray,
    time_deadline: float,
    iters: int = 150_000,
    beta0: float = 0.5,
    beta_max: float = 80.0,
    eta0: float = 0.5,
    report_every: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Minimises the soft‑max approximation of max(a*a) using an
    exponentiated‑gradient (mirror descent) scheme.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, iters + 1):
        if time.time() >= time_deadline - 0.5:
            break

        # ----- self‑convolution (full) -----
        conv_full = conv_fft(a, a)          # length 2n‑1
        max_c = np.max(conv_full)

        # ----- soft‑max weights -----
        beta = min(beta_max, beta0 * np.sqrt(t))
        # stabilise exponentials
        exp_terms = np.exp(beta * (conv_full - max_c))
        w = exp_terms / np.sum(exp_terms)   # length 2n‑1, sums to 1

        # ----- gradient of the smooth objective -----
        a_rev = a[::-1]
        # convolution of reversed a with the weight vector
        grad_full = conv_fft(a_rev, w)      # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- exponentiated (mirror) update -----
        eta = eta0 / np.sqrt(t)
        upd = np.exp(np.clip(-eta * grad, -50, 50))
        a = a * upd
        a = project_to_simplex(a)

        # ----- reporting & bookkeeping -----
        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  iter {t:8d} – new best {cur_val:.6f}")

    # final check
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# 3. Simulated‑annealing hill‑climber (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_sa(seq, time_deadline, max_steps: int = 300_000) -> tuple[list, float]:
    """
    Stochastic hill‑climber with a simulated‑annealing acceptance rule.
    Operates directly on the simplex‑projected vector.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_val = fast_score(best_a)

    cur_a = a.copy()
    cur_val = best_val

    # annealing schedule
    temperature = 0.1
    cooling = 0.99995

    for step in range(max_steps):
        if time.time() >= time_deadline - 0.2:
            break

        cand = cur_a.copy()

        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = 0.1 * cand[i]                # 10 % of source mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_val = fast_score(cand)

        delta = cand_val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / max(temperature, 1e-12)):
            cur_a, cur_val = cand, cand_val
            if cur_val < best_val:
                best_val, best_a = cur_val, cur_a.copy()

        temperature = max(temperature * cooling, 1e-6)

    return best_a.tolist(), best_val


# ----------------------------------------------------------------------
# Helper: generate initial vectors
# ----------------------------------------------------------------------


def random_initial(n: int) -> np.ndarray:
    """
    Returns a simplex‑normalised vector.
    Randomly chooses between a Dirichlet sample and an edge‑biased design.
    """
    mode = np.random.choice(["dirichlet", "edge"])
    if mode == "dirichlet":
        # concentration drawn uniformly – encourages both dense and sparse vectors
        alpha = np.random.uniform(0.05, 3.0)
        vec = np.random.gamma(alpha, 1.0, size=n)
    else:
        # edge‑biased: larger mass near the two ends, controlled by an exponent s
        s = np.random.uniform(0.2, 2.0)
        idx = np.arange(1, n + 1, dtype=float)
        w = (idx ** (-s) + (n + 1 - idx) ** (-s))
        vec = w * np.random.rand(n)  # perturb a bit
    vec = np.maximum(vec, 0.0)
    vec /= vec.sum()
    return vec


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with minimal `evaluate_sequence`
    value.  The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Fallback (if a previous construction is available)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val = fb_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Choose a working length (larger length gives more flexibility)
    # --------------------------------------------------------------
    n = 2048
    # if we have plenty of time, try a longer vector
    if deadline - time.time() > 300:
        n = 4096
    print(f"Working length n = {n}")

    # --------------------------------------------------------------
    # 2️⃣  Several random restarts with a two‑phase optimiser
    # --------------------------------------------------------------
    restarts = 6
    while time.time() < deadline - 30 and restarts > 0:
        restarts -= 1

        a0 = random_initial(n)
        init_score = fast_score(a0)
        if init_score < best_val:
            best_val, best_seq = init_score, a0.tolist()
            print(f"Restart init score = {init_score:.6f}")

        # Phase A – L2‑flattening
        a1, score1 = l2_optimize(
            a0,
            time_deadline=deadline - 15,
            iters=60_000,
            eta0=0.5,
        )
        if score1 < best_val:
            best_val, best_seq = score1, a1.tolist()
            print(f"  L2‑phase improved to {score1:.6f}")

        # Phase B – smooth (log‑sum‑exp) refinement
        a2, score2 = smooth_optimize(
            a1,
            time_deadline=deadline - 10,
            iters=120_000,
            beta0=0.5,
            beta_max=80.0,
            eta0=0.5,
            report_every=20_000,
        )
        if score2 < best_val:
            best_val, best_seq = score2, a2.tolist()
            print(f"  Smooth‑phase improved to {score2:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final annealed hill‑climbing (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("\nRunning final annealed hill‑climb")
        polished_seq, polished_val = hill_climb_sa(
            best_seq,
            time_deadline=deadline,
            max_steps=300_000,
        )
        if polished_val < best_val:
            best_val, best_seq = polished_val, polished_seq
            print(f"Hill‑climb finished with best {best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Safeguard – very small uniform fallback if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 5️⃣  Report final score and return the sequence
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
