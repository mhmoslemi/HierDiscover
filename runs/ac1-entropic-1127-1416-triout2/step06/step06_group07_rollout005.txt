- compute L = next_pow2(3*n-2) once.
- for t in range(1, iters+1):
    if time.time() >= deadline-1: break.
    A = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(A * A, n=L)
    conv = conv_full[:2*n-1]
    max_val = np.max(conv)
    if t > iters * switch_iter_frac:
        # use hard max weight
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        beta = min(beta_max, beta0 * np.sqrt(t))
        exp_terms = np.exp(beta * (conv - max_val))
        w = exp_terms / np.sum(exp_terms)

    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=L)

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
* Choose a length n (2048 by default; 4096 when time permits).
* Perform several random Dirichlet restarts.
* For each restart run a mirror‑descent / exponentiated‑gradient optimiser
  that minimises a smooth soft‑max approximation of the ℓ∞‑norm of the
  self‑convolution.  After a warm‑up phase the soft‑max is replaced by a
  hard‑max (sub‑gradient) to focus on the true worst shift.
* Keep the best vector seen across all restarts.
* Finally polish the best vector with a stochastic hill‑climber that
  swaps entries or transfers small fractions of mass while always
  projecting onto the probability simplex.
* Return the best sequence found (scaled to sum = 1 – the objective is
  homogeneous, so any scaling yields the same score).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution, real‑valued)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # uniform vector if everything non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective (fast when Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute 2·n·max(a*a) for a vector whose entries sum to 1.
    This is exactly the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Soft‑max / hard‑max optimisation (mirror descent on the simplex)
# ----------------------------------------------------------------------


def smooth_optimize(
    a: np.ndarray,
    time_deadline: float,
    iters: int = 300_000,
    beta0: float = 0.5,
    beta_max: float = 80.0,
    eta0: float = 0.5,
    switch_frac: float = 0.75,
    report_every: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)).
    Returns the best vector seen and its exact score.
    """
    n = a.shape[0]
    L = next_pow2(3 * n - 2)            # length for all FFTs (a, a_rev, w)
    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, iters + 1):
        # stop if time is exhausted
        if time.time() >= time_deadline - 1.0:
            break

        # ----- a*a (full convolution) -----
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]
        cur_max = np.max(conv)

        # ----- weighting vector w -----
        if t > int(iters * switch_frac):
            # hard‑max (sub‑gradient of the true ℓ∞‑norm)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # soft‑max with temperature β
            beta = min(beta_max, beta0 * np.sqrt(t))
            # stable exponentials (max shifted to zero)
            exp_terms = np.exp(beta * (conv - cur_max))
            w = exp_terms / np.sum(exp_terms)

        # ----- gradient of   <w, a*a>   w.r.t. a -----
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- exponentiated (mirror) update -----
        # step size decays a little faster than 1/√t
        eta = eta0 / (t ** 0.6)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)               # guard against tiny negatives
        a = project_to_simplex(a)

        # ----- occasional reporting -----
        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  iter {t:8d} – new best score {cur_val:.6f}")

    # final check (in case the very last iterate is best)
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=400_000):
    """
    Stochastic hill‑climber that swaps two entries or moves a fraction of mass.
    The true objective `evaluate_sequence` is used only at the very end.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = fast_score(best_a)

    # initial transfer scale (fraction of mass moved)
    step_scale = 0.1

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            # success → allow a larger transfer next time
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # failure → shrink step size
            step_scale *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score
