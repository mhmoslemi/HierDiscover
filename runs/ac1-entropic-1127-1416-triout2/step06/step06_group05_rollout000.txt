def semicircle_shape(n):
    i = np.arange(n, dtype=float)
    x = (2*i+1 - n) / n    # range approx -1 to 1

"""
Propose a non‑negative step sequence with a low official score.

The algorithm expands on the previous “beta‑sampling + Adam + sub‑gradient”
pipeline with several new ideas:

*   A richer family of deterministic seed shapes (semicircle, cosine, power‑law,
    symmetric exponential) is added.
*   Randomised shape generation is broadened (mixtures, symmetric variants,
    random power‑law/exponential families).
*   The Adam stage uses a higher final temperature (λ≈1500), a larger initial
    learning rate and aggressive gradient clipping, which helps escape shallow
    plateaus.
*   After Adam we run a cheap projected sub‑gradient descent (exact max‑convolution)
    with a larger step‑size schedule.
*   A simple coordinate‑swap local search (mass transfer between two indices)
    performs fine‑grained adjustments that are cheap to evaluate.
*   Finally the globally best survivor is polished again with a long Adam run,
    a sub‑gradient pass and a short random walk.

All expensive calls use a fast surrogate score
    2·n·max(conv(a,a))          (the denominator is 1 after projection to the simplex)
so that we can afford millions of iterations within the 1000 s budget.
The official ``evaluate_sequence`` is invoked only for the final candidates.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for all n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector is assumed to be normalised to Σa = 1,
    therefore the denominator is 1 and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (deterministic families)
# ----------------------------------------------------------------------


def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def semicircle_shape(n: int) -> np.ndarray:
    """a_i ∝ sqrt(1 - ((2i+1-n)/n)^2) on [0, n)."""
    i = np.arange(n, dtype=float)
    x = (2.0 * i + 1.0 - n) / n  # maps i∈[0,n-1] → [-1,1)
    vals = np.sqrt(np.maximum(0.0, 1.0 - x ** 2))
    if vals.sum() == 0:
        return uniform_shape(n)
    return vals / vals.sum()


def cosine_shape(n: int) -> np.ndarray:
    """Raised‑cosine (zero at the ends, peak in the centre)."""
    i = np.arange(n, dtype=float)
    vals = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    vals /= vals.sum()
    return vals


def power_law_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (p>0)."""
    i = np.arange(1, n + 1, dtype=float)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def exponential_decay_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), optionally symmetrised."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-lam * i)
    vals /= vals.sum()
    return vals


def symmetric_exponential_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay (peak at centre)."""
    half = (n + 1) // 2
    left = np.exp(-lam * np.arange(half)[::-1])
    if n % 2 == 0:
        right = left[1:][::-1]
    else:
        right = left[1:][::-1]
    vals = np.concatenate([left, right])
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Beta family (kept from the previous implementation)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α-1}·(n-1-i+ε)^{β-1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return uniform_shape(n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return uniform_shape(n)
    return mix / s


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (for Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                       # soft‑max weights (sum to 1)

    B = max_c + np.log(sum_exp) / lam            # smooth approx of max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.20,
    lam_start: float = 1.0,
    lam_end: float = 1500.0,
    noise_every: int = 1200,
    noise_scale: float = 4e-5,
    grad_clip: float = 5.0,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector found (projected on the simplex) together with
    its fast surrogate score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # quadratic (sqrt) schedule for λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        # gradient clipping (by norm)
        gnorm = np.linalg.norm(grad)
        if gnorm > grad_clip:
            grad = grad * (grad_clip / gnorm)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 50_000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} lam={lam:.1f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the official objective (via fast surrogate).
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # sub‑gradient of the active quadratic term (all indices attaining the max)
        # we average over all active indices to obtain a smoother direction
        eps_active = 1e-12
        active_mask = np.abs(conv - max_val) <= eps_active
        active_idxs = np.where(active_mask)[0]

        grad = np.zeros_like(a)
        for k in active_idxs:
            i = np.arange(n)
            j = k - i
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]
        grad /= max(1, len(active_idxs))  # average

        step = step0 / np.sqrt(it)

        # simple back‑tracking line‑search
        for _ in range(10):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 30_000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple coordinate‑swap local search (mass transfer)
# ----------------------------------------------------------------------


def coordinate_swap_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.02,
    max_iter: int = 100_000,
) -> Tuple[np.ndarray, float]:
    """
    Cheap local search: repeatedly move a small amount of mass from one index
    to another while staying on the simplex.  Accepts strictly improving moves.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # Decrease step size slowly
        step = init_step / np.sqrt(it)

        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 0:
            continue
        delta = step * a[i]  # proportion of source mass
        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta
        # No need to re‑project because total mass is preserved and we kept positivity
        a_prop = np.clip(a_prop, 0.0, None)  # safeguard against tiny negatives

        # Simple re‑normalisation to avoid drift
        a_prop /= a_prop.sum()

        cur_sc = fast_score(a_prop)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_prop.copy()
            a = a_prop
        # otherwise keep current a (allow occasional uphill moves with tiny prob)
        elif rng.random() < 1e-4:
            a = a_prop

        if it % 25_000 == 0:
            print(
                f"    Coord‑swap iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc
