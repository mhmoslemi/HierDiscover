    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.5
    best_a = a.copy()
    best_sc = compute_score(a)
    t = 0
    idx_start = 2 * n - 2
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        # epsilon schedule
        eps_smooth = max(1e-7, maxc * max(0.2 * (0.98 ** (t-1)), 0.001))

        conv_full = convolution_fft(a, w_rev)
        grad = 2.0 * conv_full[idx_start - np.arange(n)]
        # Adam updates

            best_sc = cur_sc
            best_a = a_new.copy()

        exp_vec = generate_exponential_vector(n)
        sc = compute_score(exp_vec)
        if sc < best_score:
            best_score = sc
            best_seq = exp_vec.tolist()
            print(f"Exponential n={n} score={sc:.6f}")
        # powerlaw with p=0.5,1.0,1.5
        for p in [0.5, 1.0, 1.5]:
            pl = generate_powerlaw_vector(n, p)
            sc = compute_score(pl)
            if sc < best_score:
                best_score = sc
                best_seq = pl.tolist()
                print(f"Powerlaw p={p} n={n} score={sc:.6f}")
    # Main search loop
    outer = 0
    while time.time() < deadline - 5.0:
        outer += 1
        n = int(np.random.choice(lengths))
        # Choose random init type
        init_type = np.random.choice(['sparse', 'uniform', 'exp', 'pl'])
        if init_type == 'sparse':
            a = random_sparse_vector(n)
        elif init_type == 'uniform':
            a = np.full(n, 1.0/n)
        elif init_type == 'exp':
            a = generate_exponential_vector(n)
        else:
            p = np.random.choice([0.5, 1.0, 1.5, 2.0])
            a = generate_powerlaw_vector(n, p)
        # Quick score
        cur_sc = compute_score(a)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a.tolist()
            print(f"[{outer}] init {init_type} n={n} score={cur_sc:.6f}")

        # Adam optimization phase
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        max_adam_iters = int(min(20000, (remaining * 30)))  # about 30 it per sec
        a_adam, sc_adam = adam_optimize(a, deadline - 2.0, max_iter=max_adam_iters)
        cur_sc = compute_score(a_adam)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a_adam.tolist()
            print(f"[{outer}] after Adam n={n} score={cur_sc:.6f}")

        # Subgradient refinement
        remaining = deadline - time.time()
        if remaining > 10.0:
            a_sub, sc_sub = subgrad_optimize(a_adam, deadline - 4.0, max_iter=50000, step0=0.06)
            cur_sc = compute_score(a_sub)
            if cur_sc < best_score:
                best_score = cur_sc
                best_seq = a_sub.tolist()
                print(f"[{outer}] after subgrad n={n} score={cur_sc:.6f}")

        # Directed hill climb
        remaining = deadline - time.time()
        if remaining > 8.0:
            seq_to_polish = best_seq if best_seq is not None else a_sub.tolist()
            polished_seq, polished_sc = directed_hill_climb(seq_to_polish, deadline - 2.0, max_steps=200_000)
            if polished_sc < best_score:
                best_score = polished_sc
                best_seq = polished_seq
                print(f"[{outer}] hill-climb improved score={polished_sc:.6f}")

            # symmetry average
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[{outer}] symmetry improved score={sym_sc:.6f}")

    # fallback uniform if no result
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq

"""
propose_candidate – stochastic search for a non‑negative step‑function
with a low value of evaluate_sequence.

The algorithm follows several complementary phases
    1. quick evaluation of a few deterministic families (uniform,
       exponential, power‑law) that already beat the naïve uniform bound.
    2. many random restarts (sparse, uniform, exponential, power‑law) on
       a range of lengths.
    3. a smooth “log‑sum‑exp” approximation of the max‑convolution
       minimised with Adam (adaptive moments).  This gives a fast
       descent that respects the simplex constraint.
    4. classic sub‑gradient descent on the exact objective for fine‑tuning.
    5. a directed hill‑climber that moves mass from the index contributing
       most to the current worst convolution entry to the index contributing
       the least – this explicitly attacks the bottleneck.
    6. optional symmetry averaging (a←(a+rev(a))/2) which often flattens
       the autocorrelation further.

All intermediate vectors are projected onto the probability simplex
(non‑negative, sum = 1) using the Euclidean projection.  The routine
stops a few seconds before the supplied time budget, then returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (fast linear convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using a real FFT.
    Result length = len(a) + len(b) – 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (non‑negative, sum = 1).  Works for any real vector.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple families of initial vectors
# ----------------------------------------------------------------------


def random_sparse_vector(n: int) -> np.ndarray:
    """Random probability vector with a random support density 0.1‑0.9."""
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def generate_exponential_vector(n: int) -> np.ndarray:
    """
    Decreasing exponential a_i ∝ r^i, r chosen so that the product
    of the largest and the smallest entry roughly balances the central
    convolution (n·r^{n‑1}=1).  The vector is then normalised to sum = 1.
    """
    # solve n·r^{n‑1}=1  →  r = exp(-log(n)/(n‑1))
    r = math.exp(-math.log(n) / (n - 1))
    powers = r ** np.arange(n, dtype=float)
    a = powers / np.sum(powers)
    return a


def generate_powerlaw_vector(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation of a smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  max_iter: int = 20000,
                  lr: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps_adam: float = 1e-8,
                  eps_factor: float = 0.20,
                  eps_min: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth objective
        f_eps(a) = eps·log Σ_k exp( (a⊛a)_k / eps )
    using Adam.  After each update the vector is projected onto the simplex.
    Returns the best vector (lowest exact score) seen during the run.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    idx_start = 2 * n - 2
    best_a = a.copy()
    best_sc = compute_score(a)

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1

        # convolution of a with itself
        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # epsilon diminishes with the iteration count, never below eps_min
        eps = max(eps_min, maxc * max(eps_factor * (0.99 ** (t - 1)), 0.01))

        # soft‑max weights – numerically stable
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)        # ∑ w = 1
        w_rev = w[::-1]

        # gradient of the smooth surrogate
        full = convolution_fft(a, w_rev)       # length 3·n‑2
        grad = 2.0 * full[idx_start - np.arange(n)]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Classic sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     deadline: float,
                     max_iter: int = 100_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(a⊛a) with projection onto the simplex.
    The sub‑gradient is taken at the shift attaining the maximum entry.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if t % 1000 == 0:
            # tiny random shake to avoid stagnation
            a += np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Directed hill‑climb that attacks the current worst convolution index
# ----------------------------------------------------------------------


def directed_hill_climb(seq,
                        deadline: float,
                        max_steps: int = 200_000) -> tuple[list, float]:
    """
    Greedy mass‑transfer that moves probability mass from the index that
    contributes most to the current worst convolution entry to the index that
    contributes the least.  Only moves that improve the exact score are kept.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    step_frac = 0.10  # initial fraction of the source mass to move

    while max_steps > 0 and time.time() < deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)

        # contribution of each coordinate to the offending entry
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        i_max = int(np.argmax(contrib))          # source (large contribution)
        i_min = int(np.argmin(contrib))          # destination (small contribution)

        epsilon = step_frac * a[i_max]
        if epsilon <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= epsilon
        a_new[i_min] += epsilon
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            a = a_new
            step_frac = min(step_frac * 1.05, 0.5)   # become bolder after success
        else:
            step_frac = max(step_frac * 0.94, 0.001)  # become more cautious

            # occasional random jitter to escape flat plateaus
            if np.random.rand() < 0.05:
                i, j = np.random.randint(n, size=2)
                cand = a.copy()
                cand[i], cand[j] = cand[j], cand[i]
                cand = project_to_simplex(cand)
                cand_sc = compute_score(cand)
                if cand_sc < best_sc:
                    best_sc = cand_sc
                    best_a = cand.copy()
                    a = cand

        max_steps -= 1

    return best_a.tolist(), best_sc


# ----------------------------------------------------------------------
# Symmetry averaging (often reduces the peak of the autocorrelation)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a+reverse(a))/2 and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Deterministic families (quick cheap candidates)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]
    for n in lengths:
        # uniform
        uni = np.full(n, 1.0 / n)
        sc = compute_score(uni)
        if sc < best_score:
            best_score, best_seq = sc, uni.tolist()
            print(f"Uniform n={n}  score={sc:.6f}")

        # exponential decay
        exp_vec = generate_exponential_vector(n)
        sc = compute_score(exp_vec)
        if sc < best_score:
            best_score, best_seq = sc, exp_vec.tolist()
            print(f"Exponential n={n}  score={sc:.6f}")

        # power‑law with a few exponents
        for p in (0.5, 1.0, 1.5, 2.0):
            pl = generate_powerlaw_vector(n, p)
            sc = compute_score(pl)
            if sc < best_score:
                best_score, best_seq = sc, pl.tolist()
                print(f"Power‑law p={p} n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # Main stochastic search
    # ------------------------------------------------------------------
    outer = 0
    while time.time() < deadline - 5.0:
        outer += 1
        n = int(np.random.choice(lengths))

        # pick a random initial family
        init_choice = np.random.choice(["sparse", "uniform", "exp", "pl"])
        if init_choice == "sparse":
            a = random_sparse_vector(n)
        elif init_choice == "uniform":
            a = np.full(n, 1.0 / n)
        elif init_choice == "exp":
            a = generate_exponential_vector(n)
        else:   # power‑law
            p = float(np.random.choice([0.5, 1.0, 1.5, 2.0]))
            a = generate_powerlaw_vector(n, p)

        # quick filter with the cheap score
        cur_sc = compute_score(a)
        if cur_sc < best_score:
            best_score, best_seq = cur_sc, a.tolist()
            print(f"[{outer}] init {init_choice} n={n}  score={cur_sc:.6f}")

        # ------------------------------------------------------------------
        # Phase A – Adam on the smooth log‑sum‑exp surrogate
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 12.0:
            break
        max_adam_iters = int(min(20_000, remaining * 30))   # ~30 it/s
        a_adam, _ = adam_optimize(
            a,
            deadline - 2.0,
            max_iter=max_adam_iters,
        )
        cur_sc = compute_score(a_adam)
        if cur_sc < best_score:
            best_score, best_seq = cur_sc, a_adam.tolist()
            print(f"[{outer}] after Adam n={n}  score={cur_sc:.6f}")

        # ------------------------------------------------------------------
        # Phase B – exact sub‑gradient refinement
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 10.0:
            a_sub, _ = subgrad_optimize(
                a_adam,
                deadline - 4.0,
                max_iter=60_000,
                step0=0.06,
            )
            cur_sc = compute_score(a_sub)
            if cur_sc < best_score:
                best_score, best_seq = cur_sc, a_sub.tolist()
                print(f"[{outer}] after sub‑grad n={n}  score={cur_sc:.6f}")

        # ------------------------------------------------------------------
        # Phase C – directed hill‑climb polishing
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 8.0:
            seq_to_polish = best_seq if best_seq is not None else a_sub.tolist()
            polished_seq, polished_sc = directed_hill_climb(
                seq_to_polish,
                deadline - 2.0,
                max_steps=200_000,
            )
            if polished_sc < best_score:
                best_score, best_seq = polished_sc, polished_seq
                print(f"[{outer}] hill‑climb improved  score={polished_sc:.6f}")

            # symmetry averaging (cheap, often beneficial)
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score, best_seq = sym_sc, a_sym.tolist()
                print(f"[{outer}] symmetry average improved  score={sym_sc:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
