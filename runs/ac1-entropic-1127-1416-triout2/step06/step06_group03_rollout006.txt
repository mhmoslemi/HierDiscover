"""
Search for a low‑value step sequence using a multi‑stage
smooth‑max surrogate minimised by L‑BFGS‑B and a final
projected sub‑gradient polish.

The algorithm:
  • Generate diverse normalised seeds (arcsine, uniform,
    beta‑ and power‑shapes).
  • For each seed optimise the smooth approximation
        f_β(a)= (1/β)·log ∑ₖ exp(β·(a*a)_k)
    with the L‑BFGS‑B optimiser.  The soft‑max
    parametrisation a=softmax(z) guarantees a_i≥0
    and Σa_i=1, so the true objective 2·n·max(a*a) is
    exactly the value returned by ``evaluate_sequence``.
    The temperature β is increased through a continuation
    schedule (5 → 300) so that the optimiser first sees a
    smooth landscape and then focuses on the true max.
  • Keep the best sequence seen.
  • If time permits, polish the best candidate with a
    classic projected sub‑gradient descent (the same routine
    used in the reference implementation).

The code respects the 1000 s wall‑clock budget, always returns a
valid non‑negative sequence and prints progress information.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (real inputs, real output)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple generators for diverse starting points (all normalised)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑type shape – heavy at the ends."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family density on the discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Soft‑max parametrisation and smooth‑max surrogate
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_surrogate_and_grad(z: np.ndarray, beta: float):
    """
    Returns (f, grad_z) where
        f = (1/β)·log∑ₖ exp(β·conv_k)     (smooth approximation of max)
    and grad_z is the gradient of f w.r.t. the unconstrained vector z.
    The soft‑max map a = softmax(z) enforces a_i≥0, Σa_i=1.
    """
    a = softmax(z)                              # a ≥ 0 , Σa = 1
    conv = convolution_fft(a, a)                # length 2n‑1
    n = a.size
    # --- smooth max (log‑sum‑exp) ---
    x = beta * conv
    max_x = np.max(x)
    exp_shift = np.exp(x - max_x)               # stable
    sum_exp = np.sum(exp_shift)
    f = (max_x + np.log(sum_exp)) / beta

    # --- gradient w.r.t. a ---
    w = exp_shift / sum_exp                     # w_i = soft‑max weights
    w_rev = w[::-1]
    rev = convolution_fft(a, w_rev)            # length = n + (2n‑1) - 1 = 3n‑2
    L = 2 * n - 1
    grad_a = 2.0 * rev[L - 1 - np.arange(n)]    # same slice used in sub‑gradient code

    # --- chain rule to obtain grad_z ---
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimiser with time‑aware callback
# ----------------------------------------------------------------------


def lbfgs_optimize(z0: np.ndarray, beta: float, maxiter: int, deadline: float):
    """
    Minimise the smooth surrogate for a fixed temperature β
    using L‑BFGS‑B.  The optimisation stops early if the wall‑clock
    deadline is exceeded.
    """
    # SciPy expects a callable returning (f, grad)
    def fun(z):
        return smooth_surrogate_and_grad(z, beta)

    # callback: return True -> terminate optimisation
    def callback(_):
        return time.time() > deadline

    # Perform the optimisation
    res = minimize(fun,
                   x0=z0,
                   method='L-BFGS-B',
                   jac=True,
                   options={'maxiter': maxiter, 'disp': False},
                   callback=callback)

    # If the optimisation was aborted early we still obtain the
    # current iterate in res.x.
    return res.x


# ----------------------------------------------------------------------
# Projection onto the probability simplex (used by the sub‑gradient phase)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.05,
) -> np.ndarray:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for “tight” indices
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2 * (a ⋆ mask_rev) (restricted to original length)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            # cheap surrogate for monitoring
            surrogate = 2.0 * n * float(np.max(convolution_fft(a, a)))
            print(f"    SubGrad it={it:,d} surrogate={surrogate:.6f}")

    return a

    # 0.  If a previous solution exists, try it as a fallback
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose a set of lengths (larger → more degrees of freedom)
    # ------------------------------------------------------------------
    lengths = [16384, 8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ---- 1a. build a modest pool of seeds -------------------------
        seeds = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            generate_mixture_shape(n, 0.5),
        ]

        # random beta‑family seeds
        for _ in range(4):
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(generate_beta_shape(n, a, b, eps))

        # random power‑law seeds
        for _ in range(4):
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))

        # if we already have a good candidate, try a (truncated/padded) copy
        if best_seq:
            bs = np.array(best_seq, dtype=float)
            if bs.size == n:
                seeds.append(bs.copy())
            elif bs.size > n:
                seeds.append(bs[:n] / bs[:n].sum())
            else:  # bs.size < n → repeat to fill
                reps = int(np.ceil(n / bs.size))
                extended = np.tile(bs, reps)[:n]
                seeds.append(extended / extended.sum())

        # ---- 1b. keep only the most promising seeds (fast surrogate) --
        def quick_score(vec):
            # fast surrogate = 2·n·max(conv)  (vec already sums to 1)
            conv = convolution_fft(vec, vec)
            return 2.0 * n * float(np.max(conv))

        scored = [(quick_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:max(2, len(scored) // 2)]]

        # ---- 1c. allocate a slice of remaining time for each seed -----
        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 5.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n})")
            # initialise the optimisation variables (log‑softmax)
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            # ----- 2. continuation of smooth‑max L‑BFGS ----------------
            beta_schedule = [5.0, 20.0, 80.0, 160.0, 300.0]
            z_cur = z0.copy()
            for b_idx, beta in enumerate(beta_schedule, start=1):
                if time.time() >= deadline - 5.0:
                    break
                # allocate a tiny budget per stage (max 200 it, but stop early if time runs out)
                max_iter_stage = 200
                z_cur = lbfgs_optimize(z_cur, beta, max_iter_stage, deadline)

                # occasional progress report
                if b_idx == len(beta_schedule) or (b_idx % 2 == 0):
                    a_tmp = softmax(z_cur)
                    surrogate = quick_score(a_tmp)
                    print(f"    β={beta:.0f} surrogate={surrogate:.6f}")

            a_opt = softmax(z_cur)

            # ----- 3. official evaluation ---------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ----- 4. light exact polishing (if time permits) ----------
            if time.time() < deadline - 5.0:
                polish_deadline = deadline - 2.0
                cand_arr = np.array(best_seq, dtype=float)
                cand_arr = subgradient_descent_exact(
                    cand_arr,
                    polish_deadline,
                    max_iter=50000,
                    step0=0.12,
                )
                try:
                    val2 = evaluate_sequence(cand_arr.tolist())
                    print(f"   Polished score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, cand_arr.tolist()
                        print(f"   *** Updated best after polish: {best_val:.6f} ***")
                except Exception:
                    pass

    # ------------------------------------------------------------------
    # 5. Final global polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=150_000,
            step0=0.08,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform seq score = {best_val:.6f}")

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
