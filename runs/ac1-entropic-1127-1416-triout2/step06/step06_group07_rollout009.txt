def hardmax_descent(a, time_deadline, max_iters=500_000, eta0=0.1):
    best_a = a.copy()
    best_score = fast_score(best_a)
    n = a.shape[0]
    # reduce eta over time
    for t in range(1, max_iters+1):
        if time.time() >= time_deadline - 0.5:
            break
        # compute convolution via FFT
        # Actually use convolution via np.fft.rfft to compute full conv length 2n-1.
        L = next_pow2(2 * n - 1)
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2*n-1]
        k_max = np.argmax(conv)
        # gradient vector
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # step size schedule
        eta = eta0 / np.sqrt(t)
        # multiplicative update
        a = a * np.exp(-eta * grad)
        # renorm to simplex
        sum_a = a.sum()
        if sum_a == 0:
            # reinitialize to uniform to avoid zero
            a = np.full(n, 1.0/n)
        else:
            a = a / sum_a
        cur_score = fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # maybe adjust eta0 based on improvement?
    return best_a, best_score

"""
Propose a low‑score non‑negative step function.

Algorithm Overview
------------------
1. Choose a moderate length (default 2048).  A longer vector gives more
   freedom, but 2048 fits comfortably into the 1000 s budget.
2. Initialise a few deterministic candidates (uniform, √‑law, symmetric
   arcsine‑law) and a handful of random Dirichlet draws.
3. For each start run a *smooth* exponential‑gradient descent on the
   soft‑max surrogate of the self‑convolution (the same idea as the
   reference code, but with a larger β‑schedule – up to 100 – so the
   surrogate gets very close to the true ∞‑norm).
4. After the soft optimisation switch to a *hard‑max* projected
   multiplicative descent: at every iteration the current maximal
   convolution entry is identified, its exact sub‑gradient is computed
   (which is just a shifted copy of the vector itself), the iterate is
   updated multiplicatively and renormalised to the probability simplex.
   This fine‑tunes the vector once the smooth surrogate has already
   flattened the landscape.
5. Finally a short stochastic hill‑climber (swap / mass‑transfer moves)
   is applied to the best vector discovered.
6. The best sequence found is returned; if everything fails a tiny
   uniform fallback is used.

All steps respect the global time budget (≈ 1000 s).  The code uses only
numpy and the supplied ``evaluate_sequence`` routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution, real‑valued)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.  Returns a vector of length
    len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast exact score (exact when Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute 2·n·max(a*a) for a vector whose entries sum to 1.
    This is exactly the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Smooth optimisation (soft‑max surrogate) – same idea as the reference
# ----------------------------------------------------------------------


def smooth_optimize(
    a0: np.ndarray,
    time_deadline: float,
    iters: int = 200_000,
    beta0: float = 0.5,
    beta_max: float = 100.0,
    eta0: float = 0.4,
    report_every: int = 25_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max approximation of max(a*a).
    Returns the best vector seen and its exact score.
    """
    n = a0.shape[0]
    L = next_pow2(3 * n - 2)  # size sufficient for all FFTs
    a = a0.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    for t in range(1, iters + 1):
        if time.time() >= time_deadline - 1.0:
            break

        # a * a  (full convolution)
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]

        # soft‑max weights
        beta = min(beta_max, beta0 * np.sqrt(t))
        max_c = np.max(conv)
        # Stabilise the exponentials
        exp_terms = np.exp(beta * (conv - max_c))
        w = exp_terms / np.sum(exp_terms)

        # gradient = 2 * (w ⋆ a_rev)
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # exponentiated (mirror) update + renormalisation (simple simplex proj)
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  smooth iter {t:8d} – new best {cur_val:.6f}")

    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Hard‑max (sub‑gradient) descent – fine‑tunes after the smooth phase
# ----------------------------------------------------------------------


def hardmax_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Projected multiplicative sub‑gradient descent that always follows the
    exact maximal convolution entry (β = ∞).  The iterate stays in the
    probability simplex by simple renormalisation (positivity is inherent).
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    # Pre‑compute FFT length for the convolution (fixed throughout)
    L = next_pow2(2 * n - 1)

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline - 0.5:
            break

        # compute convection and locate the maximiser
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # exact sub‑gradient of the selected entry:
        #   g_i = 2 * a_{k_max - i}  (if the index is inside the vector)
        grad = np.zeros_like(a)
        # i ranges where j = k_max - i is valid
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)          # inclusive
        if i_start <= i_end:
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k_max - i_idx
            grad[i_idx] = 2.0 * a[j_idx]

        # multiplicative update
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex (positivity guaranteed)
        s = a.sum()
        if s == 0.0:          # numerical safeguard
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=200_000):
    """
    Stochastic hill‑climber that swaps two entries or moves a fraction of
    mass from one position to another.  Uses the exact score for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = fast_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise (all entries stay non‑negative)
        s = cand.sum()
        if s == 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / s

        cur_score = fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # success → larger steps
        else:
            step_scale *= 0.97                         # failure → smaller steps

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Initialiser helpers (deterministic shapes)
# ----------------------------------------------------------------------


def init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def init_sqrt_law(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)  (the classic √‑law)"""
    idx = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt(idx + 1.0)
    return a / a.sum()


def init_arcsine_law(n: int) -> np.ndarray:
    """a_i ∝ 1/√( (i+½)*(n‑i‑½) )  – symmetric arcsine‑like shape"""
    i = np.arange(n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(i * (n - i + 0.5))
    return a / a.sum()


def init_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Random Dirichlet draw with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()
