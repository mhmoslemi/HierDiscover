def build_conv_matrix(f):
    n = len(f)

    for i in range(n):

import time, numpy as np, scipy.optimize as opt

def convolution_fft(a,b):
    m=len(a)+len(b)-1
    n_fft=1<<(m-1).bit_length()
    A=np.fft.rfft(a,n_fft)
    B=np.fft.rfft(b,n_fft)
    return np.fft.irfft(A*B,n_fft)[:m]

def max_convolution(a):
    return float(np.max(convolution_fft(a,a)))

def compute_score(a):
    n=len(a)
    return float(2.0*n*max_convolution(a))

def project_to_simplex(v):
    # Euclidean projection onto probability simplex
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def build_conv_matrix(f):
    n=len(f)
    rows=2*n-1
    C=np.empty((rows,n),dtype=np.float64)
    C.fill(0.0)
    for i in range(n):
        C[i:i+n,i]=f
    return C

def lp_step(f,M):
    n=len(f)
    C=build_conv_matrix(f)
    b=np.full(2*n-1, M, dtype=np.float64)
    c=-np.ones(n, dtype=np.float64)   # maximize sum(g)
    # SciPy linprog expects minimize c^T x
    res=opt.linprog(c, A_ub=C, b_ub=b, bounds=[(0,None)]*n, method='highs')
    if not res.success:
        return None, None
    g=res.x
    sum_g=g.sum()
    return g,sum_g

def lp_improve(f, deadline, max_iter=30):
    n=len(f)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        M=max_convolution(f)
        cur_score=2.0*n*M
        g,sum_g=lp_step(f,M)
        if g is None:
            break
        if sum_g <= 1.0 + 1e-9:
            # no improvement possible
            break
        f_new=g/sum_g
        new_M=max_convolution(f_new)
        new_score=2.0*n*new_M
        # accept if improvement
        if new_score < cur_score - 1e-12:
            f=f_new
            continue
        # try mixing 50%
        mix=0.5
        f_mix=(1-mix)*f + mix*f_new
        mix_M=max_convolution(f_mix)
        mix_score=2.0*n*mix_M
        if mix_score < cur_score - 1e-12:
            f=f_mix
            continue
        # otherwise stop
        break
    return f

def subgradient_descent(a, deadline, max_iter=20000):
    n=len(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv=convolution_fft(a,a)
        idx=int(np.argmax(conv))
        max_val=conv[idx]
        i=np.arange(n)
        j=idx-i
        mask=(j>=0)&(j<n)
        grad=np.zeros(n, dtype=np.float64)
        grad[mask]=2.0*a[j[mask]]
        # simple diminishing step size
        step=0.5/np.sqrt(it)
        a_new=a-step*grad
        a_new=project_to_simplex(a_new)
        # accept if max conv decreased
        if max_convolution(a_new) < max_val - 1e-15:
            a=a_new
    return a

def propose_candidate(seed=42,budget_s=1000,**kwargs):
    rng=np.random.default_rng(seed)
    start=time.time()
    deadline=start+budget_s-0.5
    best_seq=None

    # fallback
    if not best_seq:

        best_score=evaluate_sequence(best_seq)
    print(f\"Search finished. Best score {best_score:.6f}\")
    return best_seq

"""
Propose a candidate step‑function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

using a two‑phase approach:

1️⃣  Linear‑programming “mass‑augmentation’’:
    For a fixed length n and a current simplex vector f,
    solve   max Σ g_j   s.t.   f * g ≤ M     (M = max(conv(f,f)), g≥0 ).
    The optimal g has a larger ℓ₁–norm; after normalising g back
    to Σ=1 the self‑convolution bound drops.  Repeating this step
    (≤ 30 times) yields a strong reduction of the max‑convolution.

2️⃣  Simple projected sub‑gradient descent on the *true* max‑convolution
    (one pass of the active‑index gradient) to polish the LP result.

Both phases respect the global 1000 s budget, try a few lengths
(1024, 2048, 4096) and keep the best sequence found.  The code
uses only NumPy and SciPy (linprog) – no external data, no filesystem.
"""

import time
import numpy as np
from scipy.optimize import linprog
from typing import List

# ----------------------------------------------------------------------
# Basic convolution utilities (full linear convolution via real FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """Fast official score for a vector already normalised to Σa = 1."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# LP helpers – build the convolution matrix for a fixed left vector f
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return matrix C where (C @ g)[k] = (f * g)[k] (full linear conv)."""
    n = f.size
    rows = 2 * n - 1
    C = np.empty((rows, n), dtype=np.float64)
    C.fill(0.0)
    # each column i contains the vector f shifted by i positions
    for i in range(n):
        C[i:i + n, i] = f
    return C


def _solve_lp_step(f: np.ndarray, M: float) -> (np.ndarray, float):
    """
    One LP step:
        maximise Σ g_j   s.t.   (C @ g) ≤ M , g ≥ 0,
    where C is the convolution matrix of f.
    Returns (g, Σg) on success, otherwise (None, None).
    """
    n = f.size
    C = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # linprog minimises ‑cᵀx → maximise Σg

    # use HiGHS – fast interior‑point LP solver
    res = linprog(
        c,
        A_ub=C,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None, None
    g = res.x
    return g, g.sum()


# ----------------------------------------------------------------------
# Iterative LP improvement (≤ 30 steps) – monotone reduction of max‑conv
# ----------------------------------------------------------------------


def lp_improve(f: np.ndarray, deadline: float, max_iter: int = 30) -> np.ndarray:
    """
    Run the LP “mass‑augmentation’’ loop.  The vector f is assumed to lie
    on the simplex (Σf = 1, f ≥ 0).  The loop stops early when no
    improvement is possible or the time budget is exhausted.
    """
    n = f.size
    for it in range(max_iter):
        if time.time() >= deadline:
            break

        M = max_convolution(f)                # current bound
        cur_score = 2.0 * n * M

        g, sum_g = _solve_lp_step(f, M)
        if g is None or sum_g is None:
            break
        # No increase in ℓ₁‑norm → we cannot improve the bound
        if sum_g <= 1.0 + 1e-9:
            break

        # Normalise the LP solution back to Σ=1
        f_candidate = g / sum_g
        new_M = max_convolution(f_candidate)
        new_score = 2.0 * n * new_M

        # Accept the candidate if it strictly improves the official score
        if new_score < cur_score - 1e-12:
            f = f_candidate
            continue

        # If a pure replacement does not help, try a 50 % convex blend.
        blend = 0.5
        f_mix = (1.0 - blend) * f + blend * f_candidate
        mix_M = max_convolution(f_mix)
        mix_score = 2.0 * n * mix_M
        if mix_score < cur_score - 1e-12:
            f = f_mix
            continue

        # No improvement reachable → stop the LP phase.
        break
    return f


# ----------------------------------------------------------------------
# Very cheap sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a: np.ndarray, deadline: float, max_iter: int = 8000) -> np.ndarray:
    """
    Projected sub‑gradient descent on the *maximum* entry of conv(a,a).
    The gradient is taken at the currently active index.
    """
    n = a.size
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        idx = int(np.argmax(conv))
        max_val = conv[idx]

        # gradient of the active quadratic term (see explanation in analysis)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size and projection
        step = 0.5 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        # keep the move only if we actually improve the max‑convolution
        if max_convolution(a_new) < max_val - 1e-14:
            a = a_new

    return a


# ----------------------------------------------------------------------
# Classical arcsine (U‑shaped) density – a strong generic starter
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) density on {0,…,n‑1}, already normalised."""
    i = np.arange(n, dtype=np.float64)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Entry point required by the competition harness
# ----------------------------------------------------------------------

    """
    rng = np.random.default_rng(seed)

    fallback = [1.0] * 256
