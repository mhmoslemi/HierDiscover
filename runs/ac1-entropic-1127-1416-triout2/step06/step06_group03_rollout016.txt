def random_hill_climb(a, deadline, move_scale=0.01, max_iter=20000):
    a = a.copy()
    n = len(a)
    best = a
    best_score = compute_score(a) # uses fast surrogate
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # choose random i, j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = move_scale * rng.random()
        if a[i] < delta:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renorm sum (should remain 1) but due to delta maintain sum.
        # Keep positivity
        # Compute score
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            a = a_new
            best_score = score
            best = a_new
    return best

    from scipy.optimize import linprog
    n = A.shape[1]
    c = -np.ones(n)
    # bounds
    bounds = [(0, None)] * n
    b_ub = np.full(A.shape[0], M)

    if not res.success:
        return None
    return res.x

"""
search_step_sequence.py

A multi‑stage optimiser for the “max‑convolution” step‑function problem.

Stage 1 – diverse parametric seeds (arcsine, beta‑family, power‑law, mixtures).
Stage 2 – a linear‑programming move:
    given the current distribution a (∑a=1) solve
        max 1ᵀg   s.t.   a∗g ≤ M   ,   g ≥ 0
    where M = max(a∗a).  The LP pushes mass to places that keep the
    self‑convolution below the current peak.  The resulting g is mixed
    back into a with a decreasing step size.
Stage 3 – a smooth‑max Adam refinement (soft‑max parametrisation, β raised
    from 5 to 200).  This fine‑tunes the shape once the LP has flattened
    the convolution.
Stage 4 – cheap random hill‑climbing moves (mass transfer between two
    coordinates) to escape tiny local basins.
Stage 5 – exact sub‑gradient descent (projected onto the simplex) for a
    final polish.

All computations respect a wall‑clock budget (default 1000 s) and
return a non‑negative sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)²)
(the official “evaluate_sequence” function is scale‑invariant, so we
keep the vectors normalised to sum = 1).
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT based convolution (linear, not circular)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(a,a)) when Σa = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all produce vectors that already sum to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Soft‑max + Adam optimiser (smooth approximation of the max)
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 200.0,
    ramp_iters: int = 30000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    Minimise a smooth surrogate of max(conv) using Adam.
    The iterates stay on the simplex via the soft‑max map.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear β schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # soft‑max surrogate weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a  (2 * a ⋆ rev(w))
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)          # length 3n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]     # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    return softmax(z)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x≥0, Σx=1} (Euclidean)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.08,
) -> np.ndarray:
    """
    Projected sub‑gradient descent for f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # indices where the max is attained (within a tiny tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # Sub‑gradient: 2·(a ⋆ rev(mask))
        rev = convolution_fft(a, mask[::-1])
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming move (stage 2)
# ----------------------------------------------------------------------


def build_conv_matrix_dense(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n convolution matrix A such that A @ g = a ∗ g.
    Dense implementation – acceptable for n≈2048.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp_max_sum(A: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max 1ᵀg   s.t.   A g ≤ M·1 , g ≥ 0.
    Returns the optimal g (may be None if LP fails).
    """
    from scipy.optimize import linprog

    n = A.shape[1]
    c = -np.ones(n)                     # maximise sum(g) → minimise –sum(g)
    bounds = [(0.0, None)] * n
    b_ub = np.full(A.shape[0], M * (1.0 + 1e-12))  # tiny slack for numerics

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Light random hill‑climbing moves (stage 4)
# ----------------------------------------------------------------------


def random_hill_climb(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    move_scale: float = 0.005,
    max_iter: int = 20000,
) -> np.ndarray:
    """
    Very cheap stochastic local search: move a tiny amount of mass
    from one coordinate to another; keep the best vector seen.
    """
    n = a.size
    best = a.copy()
    best_score = compute_score(best)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = move_scale * rng.random()
        if best[i] < delta:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta  # sum stays 1
        score = compute_score(cand)
        if score < best_score:
            best = cand
            best_score = score
    return best

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The routine mixes diverse parametric seeds, a linear‑programming
    flattening step, smooth‑max Adam refinement, cheap random
    hill‑climbing and a final exact sub‑gradient polish.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Try to reuse a previously found solution, if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose a working length (2048 is a good compromise)
    # ------------------------------------------------------------------
    n = 2048
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 1a. generate a pool of diverse seeds
    # ------------------------------------------------------------------
    seeds: List[np.ndarray] = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),
        generate_mixture_shape(n, 0.5),
    ]

    # random beta‑family seeds
    for _ in range(8):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        eps = rng.uniform(0.0, max(0.5, 0.02 * n))
        seeds.append(generate_beta_shape(n, a, b, eps))

    # random power‑law seeds
    for _ in range(8):
        p = rng.uniform(0.1, 3.0)
        eps = rng.uniform(0.0, 2.0)
        seeds.append(generate_power_shape(n, p, eps))

    # keep the most promising according to the fast surrogate
    scored = [(compute_score(s), s) for s in seeds]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:5]]

    # ------------------------------------------------------------------
    # 2.  Process each promising seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break

        print(f"\nSeed {idx}/{len(top_seeds)} – time left ≈{deadline - time.time():.1f}s")
        a = seed_vec.astype(float)
        a /= a.sum()                     # ensure Σa = 1

        # ------------------------------
        # Stage 2 – LP flattening moves
        # ------------------------------
        lp_iters = 5
        for lp_it in range(lp_iters):
            if time.time() >= deadline - 5.0:
                break

            conv = convolution_fft(a, a)
            M = float(np.max(conv))

            A = build_conv_matrix_dense(a)            # (2n‑1)×n matrix
            g = solve_lp_max_sum(A, M)
            if g is None or g.sum() <= 0.0:
                break

            # Mix the LP solution back into a (decreasing step size)
            t = 0.5 / (lp_it + 2)                     # 0.25, 0.166…, …
            g_norm = g / g.sum()
            a = (1.0 - t) * a + t * g_norm
            a = a / a.sum()                            # renormalise (numerical safety)

            print(f"  LP‑step {lp_it + 1}/{lp_iters}  score={compute_score(a):.6f}")

        # ------------------------------
        # Stage 3 – smooth‑max Adam refinement
        # ------------------------------
        eps = 1e-12
        z0 = np.log(a + eps)
        adam_deadline = min(deadline - 2.0, time.time() + 30.0)
        a = smooth_adam_optimize(
            z0,
            adam_deadline,
            n=n,
            beta0=5.0,
            beta_max=200.0,
            ramp_iters=30000,
            lr=0.04,
        )
        a = a / a.sum()        # safety normalisation

        # ------------------------------
        # Stage 4 – cheap random hill‑climb
        # ------------------------------
        if time.time() < deadline - 5.0:
            hill_deadline = deadline - 2.0
            a = random_hill_climb(a, hill_deadline, rng, move_scale=0.005, max_iter=20000)

        # ------------------------------
        # Official evaluation
        # ------------------------------
        try:
            val = evaluate_sequence(a.tolist())
            print(f"   Official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a.tolist()
                print(f"   *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"   Evaluation failed: {e}")

        # ------------------------------
        # Light polishing with exact sub‑gradient (if time permits)
        # ------------------------------
        if time.time() < deadline - 5.0:
            polish_deadline = deadline - 2.0
            cand_arr = np.array(best_seq, dtype=float)
            cand_arr = subgradient_descent_exact(
                cand_arr,
                polish_deadline,
                max_iter=50000,
                step0=0.12,
            )
            try:
                val2 = evaluate_sequence(cand_arr.tolist())
                print(f"   Polished score = {val2:.6f}")
                if val2 < best_val:
                    best_val, best_seq = val2, cand_arr.tolist()
                    print(f"   *** Updated best after polish: {best_val:.6f} ***")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 3.  Final global polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=150_000,
            step0=0.08,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
