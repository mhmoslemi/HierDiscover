"""
Step‑function optimizer for the ``evaluate_sequence`` score.

The algorithm is a three‑stage pipeline:

1️⃣  Smooth log‑sum‑exp gradient descent – quickly finds a low‑peak
    probability vector on the simplex.

2️⃣  Frank‑Wolfe / linear‑programming refinement.
    For a current vector f (∑f = 1) we solve
        maximise  1ᵀ g
        s.t.      f * g ≤ M   (pointwise) ,   g ≥ 0
    where * is linear convolution and M = max(f*f).  The optimum g has a
    larger ℓ₁‑norm than f, therefore by scaling g back to the simplex and
    mixing f ← (1‑t)f + t·g we obtain a new vector with a *strictly* smaller
    maximal autocorrelation.  Repeating this step drives the objective
    towards its theoretical lower bound.

3️⃣  Stochastic hill‑climbing (simulated annealing) for a final polish.

All work is done on the probability simplex (∑a = 1).  The final
candidate is returned as a plain Python list of floats; the harness
calls ``evaluate_sequence`` on it.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for probability vectors (∑a = 1).
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (used in hill‑climbing)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random sparse start
# ----------------------------------------------------------------------


def random_sparse_vector(n: int) -> np.ndarray:
    """Generate a random probability vector of length n with density ≈0.1‑0.9."""
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# 1️⃣  Smooth log‑sum‑exp gradient descent
# ----------------------------------------------------------------------


def smooth_gradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    step0: float = 0.5,
    eps_factor: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth surrogate of max‑convolution:
        f(a) = ε·log∑_k exp(conv_k / ε)
    Gradient is obtained via a single FFT per iteration.
    The iterate is projected back onto the simplex after each step.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε shrinks but never falls below 1 % of the current peak
        eps = maxc * max(eps_factor * (0.99 ** (t - 1)), 0.01)

        # stable soft‑max
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)          # ∑ w = 1

        # gradient = 2·(w_rev ⊛ a)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)    # length 3n‑2
        idx_start = 2 * a.shape[0] - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(a.shape[0])]

        # SGD step with a diminishing schedule
        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to avoid plateaus
        if t % 400 == 0:
            jitter = np.random.randn(a.shape[0]) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_sc

# 2️⃣  Linear‑programming (Frank‑Wolfe) refinement
# ----------------------------------------------------------------------


def build_conv_matrix_sparse(f: np.ndarray) -> lil_matrix:
    """
    Build the convolution matrix A such that A @ g = f * g (full linear conv).
    A has shape (2n‑1, n) and is returned as a *lil* matrix for efficient
    construction; the caller may convert it to CSR before passing to linprog.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = lil_matrix((rows, n), dtype=np.float64)
    for i in range(n):
        # column i contributes f (as a vertical strip) starting at row i
        A[i:i + n, i] = f
    return A


def lp_max_sum_given_M(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀ g   subject to   f * g ≤ M  (pointwise),   g ≥ 0.
    Returns the optimal g (as a dense ndarray) or None on failure.
    """
    A = build_conv_matrix_sparse(f).tocsr()
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)          # minimise -∑g  → maximise ∑g
    bounds = [(0, None)] * f.shape[0]

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    # Tiny negative entries due to numerical noise are clipped.
    g = np.maximum(res.x, 0.0)
    return g


def fw_refine(f: np.ndarray, deadline: float, max_steps: int = 100) -> np.ndarray:
    """
    Perform a short Frank‑Wolfe (LP‑based) refinement.
    The algorithm never increases the maximal auto‑convolution.
    """
    while max_steps > 0 and time.time() < deadline:
        conv = convolution_fft(f, f)
        M = np.max(conv)

        # linear program to inflate the ℓ₁‑norm without violating the bound
        g = lp_max_sum_given_M(f, M)
        if g is None:
            break

        sum_g = np.sum(g)
        if sum_g <= 1.000001:           # no improvement possible
            break

        g_norm = g / sum_g               # bring back to the simplex

        # try a few mixing coefficients – a modest step is enough to keep the bound
        improved = False

            f_new = (1.0 - t) * f + t * g_norm
            new_M = np.max(convolution_fft(f_new, f_new))
            if new_M < M - 1e-12:        # strict improvement
                f = f_new
                improved = True
                break
        if not improved:
            break
        max_steps -= 1
    return f

        if np.random.rand() < 0.5:
            # swap
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            delta = step_scale * best_a[i]
            if delta > 0:
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        # acceptance test
        if cur_sc < best_sc or np.random.rand() < np.exp((best_sc - cur_sc) / max(temperature, 1e-12)):
            best_a = cand
            best_sc = cur_sc
            if cur_sc < best_sc:
                step_scale = min(step_scale * 1.05, 0.5)
                temperature = max(temperature * 0.99, 1e-4)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_sc


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine respects the given time budget (≈ 1000 s) and returns a plain
    Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192]      # larger lengths give better potential
    outer_iter = 0

    while time.time() < deadline - 5.0:
        outer_iter += 1
        n = int(np.random.choice(lengths))

        # --- 1️⃣  initialise -------------------------------------------------
        a0 = random_sparse_vector(n)
        sc0 = compute_score(a0)
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.tolist()
            print(f"[{outer_iter}] init n={n}  score={best_score:.6f}")

        # --- 2️⃣  smooth log‑sum‑exp GD ------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # allocate a few seconds per GD run (more iterations for larger time left)
        max_gd_iter = int(min(15_000, remaining * 25))
        a_gd, _ = smooth_gradient_descent(
            a0,
            time_deadline=deadline - 2.0,
            max_iter=max_gd_iter,
            step0=0.7,
            eps_factor=0.15,
        )
        sc_gd = compute_score(a_gd)
        if sc_gd < best_score:
            best_score = sc_gd
            best_seq = a_gd.tolist()
            print(f"[{outer_iter}] after GD n={n}  score={best_score:.6f}")

        # --- 3️⃣  Frank‑Wolfe / LP refinement (only for moderate sizes) ---
        remaining = deadline - time.time()
        if remaining > 8.0 and n <= 4096:
            f = a_gd.copy()
            f = fw_refine(f, deadline - 4.0, max_steps=80)
            sc_lp = compute_score(f)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = f.tolist()
                print(f"[{outer_iter}] after LP n={n}  score={best_score:.6f}")

        # --- 4️⃣  hill‑climbing polish -------------------------------------
        remaining = deadline - time.time()
        if remaining > 7.0 and best_seq is not None:
            polished_seq, polished_sc = hill_climb(best_seq, deadline - 2.0, max_steps=120_000)
            if polished_sc < best_score:
                best_score = polished_sc
                best_seq = polished_seq
                print(f"[{outer_iter}] hill‑climb improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if nothing was found return a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
