```
def exp_mirror_descent(a0, deadline, max_iter=200000, step0=0.2):
    a = np.array(a0, copy=True)
    n = a.size
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = convolution_fft(a, a)
        mx = conv.max()
        # compute subgrad
        tol = mx * 1e-12 + 1e-14
        mask = np.abs(conv - mx) <= tol
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[n-1:n-1+n]

        eta = step0 / np.sqrt(t)
        # multiplicative update
        a = a * np.exp(-eta * subgrad)
        # renormalize to sum=1
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= sum_a

        if t % 20000 == 0:
            cur_score = compute_score(a)
            print(f"    Mirror it={t:,d} score={cur_score:.6f}")
    return a
```

"""
Propose a low‑value step sequence for the ``evaluate_sequence`` challenge.

Algorithm
---------

* **Diverse seeds** – several analytic shapes (arcsine, uniform,
  beta‑family, power‑family, mixture) are generated for a set of
  lengths (up to 16384).  They are normalised to sum to one.

* **Smooth‑max Adam phase** – the non‑differentiable “max‑convolution”
  is approximated by a soft‑max (log‑sum‑exp) with a linearly
  increasing temperature β.  The probabilities are represented by a
  soft‑max of free parameters *z*; Adam updates are performed on *z*.
  β grows from 5 to 500, which yields a very sharp approximation of the
  true maximum.

* **Mirror‑descent polishing** – after the Adam phase we minimise the
  exact objective ``max(conv(a,a))`` using an exponentiated‑gradient
  (entropy‑mirror) update.  The sub‑gradient is the usual one for a
  max‑convolution (sum of shifted copies of *a* over all tight
  constraints).  A decreasing step size ``η_t = η₀/√t`` together with a
  renormalisation onto the probability simplex gives fast convergence.

* **Final global polishing** – if time permits a second mirror‑descent
  run is performed on the best sequence found.

All operations stay inside the time budget (default 1000 s).  The
function always returns a non‑empty list of non‑negative numbers, so
the harness never receives ``-inf``.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast and accurate)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(a,a)) – a already sums to 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed generators (all output vectors summing to one)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑like density – peaks at the two ends."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def triangular_shape(n: int) -> np.ndarray:
    """Simple triangle (peak in the centre)."""
    i = np.arange(n, dtype=float)
    vals = 1.0 - np.abs(i - (n - 1) / 2.0) / ((n - 1) / 2.0)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Soft‑max + Adam optimiser (smooth approximation of max‑conv)
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 500.0,
    ramp_iters: int = 50000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Minimise a smooth surrogate of max(conv) using Adam on the
    soft‑max parametrisation.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linearly increase temperature β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # soft‑max weights for the surrogate
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient wrt a : 2·(a ⋆ w_rev) restricted to length n
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)          # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]     # slice length n

        # chain rule for the soft‑max map
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final probabilities
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# Mirror‑descent polishing (exponentiated gradient on the exact max)
# ----------------------------------------------------------------------


def mirror_descent_max(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.2,
) -> np.ndarray:
    """
    Entropy‑mirror (multiplicative‑weights) sub‑gradient descent on
    f(a)=max(conv(a,a)).  The iterate stays on the probability simplex.
    """
    a = np.array(a0, copy=True, dtype=float)
    n = a.size
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # tolerance to catch all tight indices
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2·shifted‑convolution of a with mask_rev
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[n - 1 : n - 1 + n]

        eta = step0 / np.sqrt(it)

        # multiplicative update (entropy mirror)
        a = a * np.exp(-eta * subgrad)

        # keep strictly positive, renormalise onto the simplex
        a = np.maximum(a, 1e-15)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        if it % 20000 == 0:
            print(f"    Mirror it={it:,d} score={compute_score(a):.6f}")

    return a
