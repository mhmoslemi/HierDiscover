"""
Propose a non‑negative step sequence with a low official score.

The algorithm is an extended version of the baseline:

1.  *Exploratory sampling* – for a handful of lengths we generate many
    parametric shapes (beta, exponential, power‑law, logistic, ramps,
    mixtures) and keep the few best candidates according to a fast
    surrogate `2·n·max(conv(a,a))` (the sum of the vector is forced to 1).

2.  *Refinement* – each survivor is polished by three stages:
        • Adam on a smooth log‑sum‑exp surrogate of the max‑convolution,
          with an aggressive temperature schedule (λ up to 800).
        • Exact projected sub‑gradient descent on the true max‑convolution,
          guaranteeing that the official objective never increases.
        • A cheap stochastic Gaussian walk that occasionally accepts
          uphill moves.

3.  *Global polishing* – the best vector overall is refined again with
    longer runs of the same three stages.

All steps respect the global 1000 s budget, frequently checking the
deadline.  The code is deliberately modular so that the search can be
interrupted at any point without losing the current best solution.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for all n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator is 1 and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)

def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α-1}·(n-1-i+ε)^{β-1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density (the shape used in the original baseline)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def exponential_shape(n: int, rate: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-rate * i)
    return vals / vals.sum()


def powerlaw_shape(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Power‑law a_i ∝ (i+offset)^{-p}."""
    i = np.arange(n, dtype=float) + offset
    vals = i ** (-p)
    return vals / vals.sum()


def ramp_shape(n: int) -> np.ndarray:
    """Linear decreasing ramp."""
    i = np.arange(n, dtype=float)
    vals = (n - i)  # decreasing from n to 1
    return vals / vals.sum()


def logistic_shape(n: int, centre: float, scale: float) -> np.ndarray:
    """Logistic curve a_i ∝ 1/(1+exp((i-centre)/scale))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / (1.0 + np.exp((i - centre) / scale))
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (for Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                       # soft‑max weights
    B = max_c + np.log(sum_exp) / lam            # smooth approximation of the max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 800.0,
    noise_every: int = 1000,
    noise_scale: float = 2e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector found (projected on the simplex) together with
    its fast surrogate score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # anneal λ (quadratic in sqrt‑time)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 50_000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the official objective.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step size decays like 1/sqrt(t)
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40_000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


def random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Cheap Gaussian walk on the simplex.
    Occasionally accepts a tiny uphill move to escape plateaus.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = fast_score(a_prop)

        # accept if strictly better, otherwise with a tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50_000 == 0:
            print(
                f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine samples a rich family of parametric shapes, keeps the
    best few, refines them with Adam → sub‑gradient → random walk, and
    finally polishes the global best.  The search adapts the time budget
    on‑the‑fly and guarantees a non‑empty return.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # Helper to evaluate a candidate with the *official* evaluator.
    def official_score(vec: np.ndarray) -> float:
        return evaluate_sequence(vec.tolist())

    # ------------------------------------------------------------------
    # 0.  Fallback to a known construction if it exists.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = official_score(np.array(fallback))
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Exploratory sampling on several lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Sampling length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Deterministic starter shapes
        # --------------------------------------------------------------
        starters = [
            np.full(n, 1.0 / n),
            arcsine_shape(n),
            ramp_shape(n),
            logistic_shape(n, centre=n / 2, scale=n / 4),
        ]
        local_best = None
        local_best_sc = float("inf")
        for s in starters:
            sc = fast_score(s)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = s.copy()

        # --------------------------------------------------------------
        # 1b. Random parametric shapes
        # --------------------------------------------------------------
        max_samples = int(7_000 * (budget_s / 1000.0))
        for _ in range(max_samples):
            if time.time() >= deadline - 30.0:
                break

            shape_type = rng.integers(0, 6)
            if shape_type == 0:  # beta family
                alpha = rng.uniform(0.02, 5.0)
                beta = rng.uniform(0.02, 5.0)
                eps = rng.uniform(1e-9, max(0.01, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)

            elif shape_type == 1:  # exponential
                # rate scaled so that typical decay length ≈ n/10 … n/2
                rate = rng.uniform(1.0 / n, 5.0 / n)
                cand = exponential_shape(n, rate)

            elif shape_type == 2:  # power‑law
                p = rng.uniform(0.1, 2.5)
                cand = powerlaw_shape(n, p)

            elif shape_type == 3:  # logistic
                centre = rng.uniform(0, n - 1)
                scale = rng.uniform(0.05 * n, 0.3 * n)
                cand = logistic_shape(n, centre, scale)

            elif shape_type == 4:  # linear ramp with random orientation
                if rng.random() < 0.5:
                    cand = ramp_shape(n)
                else:
                    # increasing ramp (mirror the decreasing one)
                    cand = ramp_shape(n)[::-1]
                # optionally add a tiny random perturbation
                cand = cand + 1e-8 * rng.random(n)

            else:  # mixture of two beta families
                w = rng.random()
                a1 = rng.uniform(0.02, 5.0)
                b1 = rng.uniform(0.02, 5.0)
                eps1 = rng.uniform(1e-9, max(0.01, 0.02 * n))
                a2 = rng.uniform(0.02, 5.0)
                b2 = rng.uniform(0.02, 5.0)
                eps2 = rng.uniform(1e-9, max(0.01, 0.02 * n))
                cand = generate_beta_mixture(
                    n, a1, b1, eps1, a2, b2, eps2, w
                )

            # Normalise explicitly to avoid numerical drift
            cand = project_to_simplex(cand)

            sc = fast_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = cand.copy()

        # --------------------------------------------------------------
        # 2. Keep a few top survivors for polishing
        # --------------------------------------------------------------
        if local_best is None:
            continue

        # keep a handful of diverse survivors (we reuse the same best
        # vector several times with different random seeds for robustness)
        survivors = [local_best]
        # generate a few perturbed copies of the best as extra seeds
        for _ in range(3):
            noise = rng.normal(scale=0.02, size=n)
            pert = project_to_simplex(local_best + noise)
            survivors.append(pert)

        # --------------------------------------------------------------
        # 3. Polishing each survivor
        # --------------------------------------------------------------
        for idx, seed_vec in enumerate(survivors):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Polishing survivor {idx + 1}/{len(survivors)} for n={n}")

            # a) Adam phase – allocate as many iterations as time permits
            a_opt, _ = adam_optimize(
                init_a=seed_vec,
                deadline=deadline,
                max_iter=int(min(400_000, (deadline - time.time()) * 30)),
                lr0=0.10,
                lam_start=1.0,
                lam_end=800.0,
                noise_every=1200,
                noise_scale=2e-5,
            )

            # b) Sub‑gradient descent – guarantee non‑increase of true objective
            if time.time() < deadline - 4.0:
                a_opt, _ = subgradient_optimize(
                    a_opt,
                    deadline,
                    step0=0.25,
                    max_iter=int(min(200_000, (deadline - time.time()) * 25)),
                )

            # c) Final random walk
            if time.time() < deadline - 3.0:
                a_opt, _ = random_local_search(
                    a_opt,
                    deadline,
                    init_step=0.014,
                    max_iter=int(min(250_000, (deadline - time.time()) * 35)),
                )

            # Official evaluation
            val = official_score(a_opt)
            print(f"    → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_opt.tolist()
                print(f"    *** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 4. Global polishing of the best vector (if any)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # A – long Adam run with a very high temperature
        a, _ = adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=int(min(800_000, (deadline - time.time()) * 40)),
            lr0=0.09,
            lam_start=1.0,
            lam_end=900.0,
            noise_every=1300,
            noise_scale=1e-5,
        )

        # B – exact sub‑gradient descent
        if time.time() < deadline - 4.0:
            a, _ = subgradient_optimize(
                a,
                deadline,
                step0=0.22,
                max_iter=int(min(250_000, (deadline - time.time()) * 30)),
            )

        # C – final stochastic walk
        if time.time() < deadline - 2.0:
            a, _ = random_local_search(
                a,
                deadline,
                init_step=0.012,
                max_iter=int(min(350_000, (deadline - time.time()) * 45)),
            )

        final_val = official_score(a)
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # safe uniform fallback (length 256)
        best_seq = [1.0 / 256] * 256
        best_val = official_score(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
