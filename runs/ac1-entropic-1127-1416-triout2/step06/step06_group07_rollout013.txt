"""
Improved search for a non‑negative step function that minimises
`evaluate_sequence`.

The objective for a normalised vector (∑a_i = 1) is
    score = 2·n·max(conv(a,a)),
where conv(a,a) is the *full* self‑convolution.
A small score means that the probability mass function of the sum of two
independent draws from the distribution a is as flat as possible.

The algorithm consists of several stages

1️⃣  Several random initialisations (uniform, Dirichlet, Gaussian,
    triangular and double‑Gaussian shapes) on a fairly large length
    (2048 or 4096 depending on the remaining time).

2️⃣  A mirror‑descent optimiser that minimises a smooth soft‑max
    approximation of the max‑convolution.
    The update is *exponential* (multiplicative) and the iterate is
    projected onto the probability simplex after each step.
    The temperature β grows like √t up to a large value (β_max=200) so
    that the optimisation eventually targets the true ℓ∞‑norm.

3️⃣  A stochastic hill‑climber that performs swaps, mass‑transfer moves
    and small redistributions while always re‑projecting onto the simplex.
    Successful moves increase the step size, failures shrink it.

All work respects the 1000 s budget – time‑checks are performed
continuously and the routine falls back to a tiny uniform vector if
nothing else succeeded.

The best sequence found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution, real‑valued)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx_i = 1, x_i≥0
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive – return uniform point
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective when the vector is normalised (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute 2·n·max(a*a) for a vector whose entries sum to 1.
    This equals the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Random initialisers – a small palette of useful shapes
# ----------------------------------------------------------------------


def generate_initial_distribution(n: int, kind: str) -> np.ndarray:
    """Return a normalised (sum=1) non‑negative vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n)

    elif kind == "dirichlet":
        # concentration α drawn uniformly to obtain both dense and sparse starts
        alpha = np.random.uniform(0.2, 2.0)
        a = np.random.gamma(alpha, 1.0, size=n)

    elif kind == "gaussian":
        mu = (n - 1) / 2.0
        sigma = np.random.uniform(0.1 * n, 0.3 * n)
        idx = np.arange(n)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif kind == "triangular":
        idx = np.arange(n)
        a = 1.0 - np.abs(idx - (n - 1) / 2.0) / ((n - 1) / 2.0)
        a = np.maximum(a, 0.0)

    elif kind == "double_gaussian":
        idx = np.arange(n)
        sigma = np.random.uniform(0.05 * n, 0.15 * n)
        c1 = int(0.25 * n)
        c2 = int(0.75 * n)
        a = np.exp(-0.5 * ((idx - c1) / sigma) ** 2) + np.exp(
            -0.5 * ((idx - c2) / sigma) ** 2
        )
    else:
        # fallback – simple Dirichlet with α=1
        a = np.random.rand(n)

    a = a.astype(float)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth optimisation of the soft‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def smooth_optimize_expgrad(
    a0: np.ndarray,
    time_deadline: float,
    iters: int = 500_000,
    beta0: float = 0.5,
    beta_max: float = 200.0,
    eta0: float = 0.5,
    report_every: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the soft‑max approximation of the max‑convolution.
    Uses exponentiated updates and projects onto the simplex each step.
    """
    a = a0.copy()
    n = a.shape[0]
    L = next_pow2(3 * n - 2)  # length sufficient for all FFTs

    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, iters + 1):
        # time guard
        if time.time() >= time_deadline - 0.8:
            break

        # ---- self‑convolution a*a (full) ---------------------------------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]          # length 2n‑1
        max_c = np.max(conv)

        # ---- soft‑max weights (log‑sum‑exp) -------------------------------
        beta = min(beta_max, beta0 * np.sqrt(t))
        # stabilised exponentials
        exp_terms = np.exp(beta * (conv - max_c))
        w = exp_terms / np.sum(exp_terms)      # shape (2n‑1,)

        # ---- gradient of the soft‑max surrogate ---------------------------
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        # the required slice starts at index n‑1
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ---- exponentiated mirror update ---------------------------------
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        # ---- occasional reporting -----------------------------------------
        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  iter {t:8d} – new best score {cur_val:.6f}")

    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass‑transfer / small reshuffle)
# ----------------------------------------------------------------------


def hill_climb(
    seq: list[float],
    time_deadline: float,
    max_steps: int = 200_000,
) -> tuple[list[float], float]:
    """
    Stochastic local search that only accepts improving moves.
    The move set consists of swaps, directed mass transfers and tiny
    random redistributions among three positions.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = fast_score(best_a)

    # step size – fraction of the source mass that may be moved
    step_scale = 0.05

    for step in range(max_steps):
        if time.time() >= time_deadline - 0.5:
            break

        cand = best_a.copy()
        r = np.random.rand()

        if r < 0.33:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        elif r < 0.66:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i] * np.random.rand()
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        else:
            # redistribution among three random indices
            idx = np.random.choice(n, size=3, replace=False)
            total = cand[idx].sum()
            if total > 0:
                rand = np.random.rand(3)
                rand /= rand.sum()
                cand[idx] = total * rand

        # keep feasibility
        cand = project_to_simplex(cand)

        cur_score = fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # success → larger steps
        else:
            step_scale *= 0.96                         # failure → smaller steps

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the time budget, tries several initial shapes,
    refines them with a smooth exponential‑gradient optimiser and finally
    polishes the best candidate with a hill‑climber.
    The returned object is a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Fallback if a previous construction is available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val = fb_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Choose a length keeping an eye on the remaining time
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 650:
        n = 4096
    elif remaining > 300:
        n = 2048
    else:
        n = 1024
    print(f"Using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2️⃣  Random restarts with smooth optimisation
    # ------------------------------------------------------------------
    init_kinds = ["uniform", "dirichlet", "gaussian", "triangular", "double_gaussian"]
    restart = 0

    while time.time() < deadline - 30.0:
        kind = np.random.choice(init_kinds)
        a0 = generate_initial_distribution(n, kind)
        init_sc = fast_score(a0)
        if init_sc < best_val:
            best_val = init_sc
            best_seq = a0.tolist()
            print(f"Restart {restart} – init {kind:<13} score {init_sc:.6f}")

        # smooth optimisation (the heavy part)
        opt_seq, opt_sc = smooth_optimize_expgrad(
            a0,
            time_deadline=deadline - 7.0,
            iters=500_000,
            beta0=0.5,
            beta_max=200.0,
            eta0=0.5,
            report_every=25_000,
        )
        if opt_sc < best_val:
            best_val = opt_sc
            best_seq = opt_seq.tolist()
            print(f"Restart {restart} – refined best {opt_sc:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb polishing (if we still have a few seconds)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nApplying final hill‑climb polishing")
        polished_seq, polished_val = hill_climb(
            best_seq, time_deadline=deadline, max_steps=200_000
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Safeguard – tiny uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Report final score and return the sequence
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
