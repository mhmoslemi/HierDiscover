    n_fft = next_pow2(n)

def simulated_annealing(a, deadline, max_moves=200000):
    a = a.copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score
    T0 = cur_score * 0.1  # initial temperature
    t = 0
    while time.time() < deadline and t < max_moves:
        t += 1
        # linear schedule: T = T0 * (1 - t/Tmax)
        T = max(T0 * (1 - t / max_moves), 1e-8)
        # propose move
        cand = a.copy()
        i = np.random.randint(len(a))
        j = np.random.randint(len(a))
        if i == j:
            continue
        # randomly decide type:
        if np.random.rand() < 0.5:
            # swap
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer delta fraction from larger to smaller? We'll pick delta = step_scale * cand[i]
            delta = np.random.rand() * 0.1 * cand[i]  # up to 10% of i's mass
            cand[i] -= delta
            cand[j] += delta
        # ensure non-negative, reproject to simplex (ensures sum 1)
        cand = project_to_simplex(cand)
        new_score = compute_score(cand)
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score)/max(T, 1e-12)):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
    return best_a, best_score

"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The algorithm combines several ideas that were
successful in earlier attempts and adds a few new ingredients:

1. **Diverse random starts** – either dense Dirichlet samples or sparse
   equal‑mass subsets.
2. **Adam optimisation on a smooth soft‑max surrogate** of the
   max‑convolution.  The surrogate is
        f_ε(a)=ε·log∑_k exp(conv_k/ε) ,
   whose gradient can be obtained with a single FFT per iteration.
3. **Exact sub‑gradient descent** to polish the solution on the true
   objective 2·n·max(conv).
4. **Simulated‑annealing hill‑climber** that accepts occasional
   worsening moves, allowing escape from shallow local minima.
5. **Optional symmetry averaging** (a←(a+rev(a))/2) which often flattens
   the autocorrelation further.

All updates are performed on the probability simplex (sum = 1,
non‑negative) using an Euclidean projection.  The search runs for the
available time budget (≈ 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b using real FFT (real-valued inputs)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to one.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_vector(n: int) -> np.ndarray:
    """Sparse probability vector: a random subset gets equal mass."""
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dirichlet_vector(n: int) -> np.ndarray:
    """Dense Dirichlet (exponential) draw, normalised to sum = 1."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Adam optimisation on the smooth soft‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 8000,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_factor: float = 0.25,
    min_eps_rel: float = 0.01,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth surrogate

         f_ε(a) = ε·log∑_k exp(conv_k/ε),

    where conv = a⊛a.
    The routine keeps the best exact score seen.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # --- soft‑max of the convolution ---------------------------------
        conv = convolution_fft(a, a)                      # length 2n‑1
        maxc = conv.max()
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), min_eps_rel)

        exp_vals = np.exp((conv - maxc) / eps)          # stable soft‑max
        w = exp_vals / exp_vals.sum()                  # ∑ w = 1

        # --- gradient of the surrogate ------------------------------------
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)           # length 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # --- Adam update ---------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 / (np.sqrt(t) + 1e-8)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)

        # project back to the simplex
        a_new = project_to_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to avoid stagnation
        if t % 500 == 0:
            jitter = np.random.randn(n) * (lr0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution objective)
# ----------------------------------------------------------------------


def subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the exact objective
        2·n·max(a⊛a) .
    The sub‑gradient is taken at a shift attaining the current maximum.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient g_i = 2·a_{k‑i} (zero outside support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # occasional random shake
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber
# ----------------------------------------------------------------------


def simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    max_moves: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber with a linear temperature schedule.
    Moves are either swaps or fractional mass transfers.
    Only moves that improve the exact score are always accepted;
    worse moves are accepted with probability exp(-Δ/T).
    """
    a = a0.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    n = len(a)

    T0 = cur_sc * 0.10  # initial temperature ≈10 % of the score
    for move_no in range(1, max_moves + 1):
        if time.time() >= deadline:
            break

        # linear temperature decay
        T = max(T0 * (1.0 - move_no / max_moves), 1e-9)

        cand = a.copy()
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        if np.random.rand() < 0.5:          # swap
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction
            delta = np.random.rand() * 0.25 * cand[i]   # up to 25 % of mass at i
            cand[i] -= delta
            cand[j] += delta

        cand = project_to_simplex(cand)
        new_sc = compute_score(cand)

        accept = False
        if new_sc < cur_sc:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_sc - cur_sc) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = cand.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Symmetry averaging (optional final polish)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a+rev(a))/2 and project back onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that minimises
    `evaluate_sequence`.  The function respects a time budget (≈ 1000 s)
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin so we can still call evaluate_sequence at the end
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Use a provided fallback (if the notebook defines height_sequence_1)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameters governing the outer search
    # ------------------------------------------------------------------
    # Lengths we will try – larger lengths give a lower theoretical bound.
    lengths = [1024, 2048, 4096, 8192]

    # Keep looping until we run out of time.
    outer_iter = 0
    while time.time() < deadline - 5.0:
        outer_iter += 1
        remaining = deadline - time.time()

        # Choose a length that fits the remaining budget.
        if remaining > 350:
            n = int(np.random.choice([4096, 8192]))
        elif remaining > 150:
            n = int(np.random.choice([2048, 4096]))
        elif remaining > 60:
            n = int(np.random.choice([1024, 2048]))
        else:
            n = 1024

        # ---- Random start -------------------------------------------------
        if np.random.rand() < 0.5:
            a = random_sparse_vector(n)
        else:
            a = random_dirichlet_vector(n)

        # quick guard – if the naive score is already terrible we skip heavy work
        init_sc = compute_score(a)
        if init_sc < best_score:
            best_score = init_sc
            best_seq = a.tolist()
            print(f"[{outer_iter}] init (n={n}) improved: score={best_score:.6f}")

        # ---- Phase 1 : Adam on smooth surrogate ----------------------------
        phase_deadline = deadline - 2.0
        if time.time() < phase_deadline:
            max_adam_iter = min(8000, int((phase_deadline - time.time()) * 30))
            a_adam, _ = adam_smooth_descent(
                a,
                deadline=phase_deadline,
                max_iter=max_adam_iter,
                lr0=0.5,
                eps_factor=0.20,
                min_eps_rel=0.01,
            )
            sc_adam = compute_score(a_adam)
            if sc_adam < best_score:
                best_score = sc_adam
                best_seq = a_adam.tolist()
                print(f"[{outer_iter}] after Adam (n={n})  score={best_score:.6f}")

        # ---- Phase 2 : exact sub‑gradient descent -------------------------
        phase_deadline = deadline - 5.0
        if time.time() < phase_deadline:
            max_sub_iter = min(80_000, int((phase_deadline - time.time()) * 15))
            a_sub, _ = subgrad_optimize(
                a_adam,
                deadline=phase_deadline,
                max_iter=max_sub_iter,
                step0=0.06,
            )
            sc_sub = compute_score(a_sub)
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[{outer_iter}] after sub‑grad (n={n}) score={best_score:.6f}")

        # ---- Phase 3 : simulated‑annealing hill‑climber --------------------
        phase_deadline = deadline - 8.0
        if time.time() < phase_deadline:
            max_moves = int(min(300_000, (phase_deadline - time.time()) * 200))
            a_sa, sc_sa = simulated_annealing(
                np.array(best_seq, dtype=float) if best_seq else a_sub,
                deadline=phase_deadline,
                max_moves=max_moves,
            )
            if sc_sa < best_score:
                best_score = sc_sa
                best_seq = a_sa.tolist()
                print(f"[{outer_iter}] SA hill‑climb (n={n})  score={best_score:.6f}")

        # ---- Phase 4 : optional symmetry averaging -------------------------
        a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{outer_iter}] symmetry avg (n={n})   score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – in the unlikely event we never found a candidate.
    # ------------------------------------------------------------------
    if not best_seq:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (the harness will use this result)
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
