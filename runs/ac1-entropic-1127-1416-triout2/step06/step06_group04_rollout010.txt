def convolution_fft(a, b):
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

  optimisation from a very smooth surrogate to an accurate approximation of
  the true maximum.
* Random‑seeded beta‑mixture initialisations broaden the search space.
"""

import time
import numpy as np
from functools import partial
from scipy import optimize as opt

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (returns length len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Shape generators (beta families and mixtures)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Un‑normalised symmetric beta‑family density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex mixture of two beta‑family densities (re‑normalised)."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) density, already normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Gradient of the smooth surrogate of max(conv) (the official score)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smooth max‑convolution surrogate.
    Returns (grad_a, B) where
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv‑max))
    and grad_a = ∂/∂a ( 2·n·B / (sum(a)²) ).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)               # length 2n‑1
    max_c = np.max(conv)

    # log‑sum‑exp in a numerically stable way
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                     # soft‑max weights on convolution entries
    B = max_c + np.log(sum_exp) / lam

    # weighted sub‑gradient of the convolution
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx=1} (Duchi et al., 2008)."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper used by L‑BFGS – objective + gradient w.r.t. soft‑max parameters
# ----------------------------------------------------------------------


def compute_obj_grad(x: np.ndarray, n: int, lam: float) -> tuple[float, np.ndarray]:
    """
    Returns (value, grad) for the smooth surrogate of the official score,
    using a soft‑max parametrisation: a = s·softmax(x) with s = √(2n).
    The returned value is exactly B (the surrogate max) because the sum of a
    is fixed to s, making the prefactor 2·n / s² = 1.
    """
    s = np.sqrt(2.0 * n)
    # soft‑max (numerically stable)
    mx = np.max(x)
    ex = np.exp(x - mx)
    p = ex / np.sum(ex)          # p ≥ 0, Σp = 1
    a = s * p                    # a respects the sum constraint automatically

    grad_a, B = smooth_gradient(a, lam)

    # Chain rule: ∂B/∂x = s·p·(grad_a - ⟨grad_a,p⟩)
    avg = np.dot(grad_a, p)      # ⟨grad_a,p⟩
    grad_x = s * p * (grad_a - avg)

    return B, grad_x


# ----------------------------------------------------------------------
# Simple stochastic walk on the simplex (final polishing)
# ----------------------------------------------------------------------


def stochastic_simplex_walk(
    a_start: np.ndarray,
    deadline: float,
    init_step: float = 0.02,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Tiny Gaussian walk on the simplex (project after each move).
    Returns the best vector seen and its official score.
    """
    n = a_start.size
    target_sum = np.sqrt(2.0 * n)
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    rng = np.random.default_rng()

    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed) * target_sum
        cur_val = evaluate_sequence(a_proposed.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_proposed.copy()
            a = a_proposed
        # occasional uphill acceptance (tiny probability)
        elif rng.random() < 1e-4:
            a = a_proposed
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step sequence.
    The routine:
       1. Tries a handful of lengths (1024, 2048, 4096) – larger lengths give
          a finer discretisation and usually a smaller maximal convolution.
       2. For each length generates a small portfolio of seeds:
          arcsine, uniform, random beta‑family shapes and a few Gaussian seeds.
       3. Each seed is refined by a cascade of L‑BFGS runs on the smooth
          surrogate, with a schedule of increasing smoothing parameters λ.
       4. The best vector overall is optionally polished by a short stochastic
          walk on the simplex.
       5. If time runs out, a uniform 256‑vector is returned as a safe fallback.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Candidate lengths (larger lengths typically allow a flatter convolution)
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n=== Optimising length n = {n} ===")
        target_sum = np.sqrt(2.0 * n)

        # ----- Build a collection of initial seeds -----------------------
        seeds: list[np.ndarray] = []

        # uniform and arcsine (both already normalised to sum = 1)
        seeds.append(np.full(n, 1.0 / n) * target_sum)          # uniform
        seeds.append(arcsine_initial(n) * target_sum)          # arcsine

        # a few random beta‑family and mixture shapes
        for _ in range(6):
            if time.time() >= deadline - 5.0:
                break
            alpha = rng.uniform(0.1, 3.5)
            beta = rng.uniform(0.1, 3.5)
            eps = rng.uniform(1e-8, 0.02 * n)
            a = generate_beta_shape(n, alpha, beta, eps) * target_sum
            seeds.append(a)

        for _ in range(6):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            a1 = generate_beta_shape(
                n,
                rng.uniform(0.1, 3.5),
                rng.uniform(0.1, 3.5),
                rng.uniform(1e-8, 0.02 * n),
            )
            a2 = generate_beta_shape(
                n,
                rng.uniform(0.1, 3.5),
                rng.uniform(0.1, 3.5),
                rng.uniform(1e-8, 0.02 * n),
            )
            a = (w * a1 + (1.0 - w) * a2) * target_sum
            seeds.append(a)

        # a few pure Gaussian soft‑max seeds
        for _ in range(4):
            if time.time() >= deadline - 5.0:
                break
            x = rng.normal(scale=1.0, size=n)
            mx = np.max(x)
            p = np.exp(x - mx)
            p /= p.sum()
            seeds.append(p * target_sum)

        # ----------------------------------------------------------------
        # Refine each seed with a cascade of L‑BFGS runs
        # ----------------------------------------------------------------
        for si, a0 in enumerate(seeds):
            if time.time() >= deadline - 2.0:
                break

            # initialise soft‑max parameters x from a0 (a0 already has the correct sum)
            p0 = a0 / target_sum
            # avoid exact zeros (log would blow up)
            p0 = np.clip(p0, 1e-15, None)
            x_cur = np.log(p0)

            # increasing smoothing schedule (more accurate surrogate later)
            lam_schedule = [5.0, 30.0, 150.0, 800.0]

            for lam in lam_schedule:
                if time.time() >= deadline - 2.0:
                    break

                # allocate a modest number of L‑BFGS iterations – the remaining
                # budget determines how many we can afford.
                remaining = deadline - time.time()
                max_iter = int(min(2000, remaining * 10))
                if max_iter < 20:
                    break

                # L‑BFGS on the smooth surrogate (value+gradient supplied)
                res = opt.minimize(
                    fun=partial(compute_obj_grad, n=n, lam=lam),
                    x0=x_cur,
                    jac=True,
                    method="L-BFGS-B",
                    options={"maxiter": max_iter, "disp": False},
                )
                x_cur = res.x

                # evaluate the *official* score at the current iterate
                mx = np.max(x_cur)
                ex = np.exp(x_cur - mx)
                p = ex / np.sum(ex)
                a = target_sum * p
                val = evaluate_sequence(a.tolist())
                if val < best_val:
                    best_val = val
                    best_seq = a.tolist()
                    print(
                        f"  New global best (len={n}, seed={si}, λ={lam:.0f}) → {val:.6f}"
                    )

            # ----------------------------------------------------------------
            # Optional tiny stochastic polish for this seed (if time permits)
            # ----------------------------------------------------------------
            if time.time() < deadline - 1.0:
                # Convert final x_cur to a vector
                mx = np.max(x_cur)
                ex = np.exp(x_cur - mx)
                p = ex / np.sum(ex)
                a = target_sum * p
                a_polished, val_polished = stochastic_simplex_walk(
                    a, deadline, init_step=0.015, max_iter=5000
                )
                if val_polished < best_val:
                    best_val = val_polished
                    best_seq = a_polished.tolist()
                    print(
                        f"  After stochastic walk (len={n}, seed={si}) → {val_polished:.6f}"
                    )

    # ------------------------------------------------------------------
    # Fallback in the unlikely event no candidate was accepted
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector, official score =", best_val)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
