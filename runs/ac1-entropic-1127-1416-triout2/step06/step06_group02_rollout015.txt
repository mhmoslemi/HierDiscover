```
def hill_climb(seq, time_deadline, max_steps=200000):
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(best_a)
    step_frac = 0.1
    while max_steps > 0 and time.time() < time_deadline:
        # pick two indices i, j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # decide move type
        if np.random.rand() < 0.5:
            # swap
            cand = best_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer fraction
            delta = step_frac * best_a[i]
            if delta == 0:
                continue
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac *= 0.97
        max_steps -= 1
    return best_a.tolist(), best_sc
```

```
def adam_optimize(a0, time_deadline, max_iter=10000, lr=0.5, eps_factor=0.2):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    best_a = a.copy()
    best_sc = compute_score(a)
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        eps = maxc * max(eps_factor * (0.98 ** (t - 1)), 0.01)
        # softmax
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(t) + 1)   # optional decaying learning rate (maybe use fixed lr)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_to_simplex(a_new)
        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_sc
```

The score

```
score = 2·n·max(a⊛a) / (∑a)²
```

is *scale‑invariant* – only the shape of the non‑negative vector `a` matters.  
The problem is to flatten the linear convolution `a⊛a` as much as possible.  
Our search consists of three stages

1. **Randomised starts** – Dirichlet‑distributed vectors of various sparsities.  
2. **Smooth max‑optimisation** – minimise a smooth surrogate  
   `fε(a)=ε·log Σk exp(convk/ε)` with Adam’s adaptive steps and a projection
   onto the probability simplex after every update.  The soft‑max weight
   `w` gives the exact gradient of the surrogate in one FFT.
3. **Exact refinement** – sub‑gradient descent on the true objective and a
   stochastic hill‑climber that moves small amounts of mass between two
   coordinates (or swaps them).  After hill‑climbing we
   apply a symmetry average (`(a+rev(a))/2`) – this is cheap and often reduces the peak further.

All heavy work (convolution, gradient) is performed by FFTs, which makes the
algorithm fast even for vectors of length 4096.  The routine respects the
time‑budget (`budget_s`) and always returns the best sequence found before the
deadline.

```python
"""
Fast search for a low‑score non‑negative step function.

* generate many random Dirichlet‑type starts (different densities)
* minimise the smooth soft‑max surrogate of max(a⊛a) with Adam + projection
* refine with exact sub‑gradient descent and a stochastic hill‑climber
* finish with a symmetry averaging step

The code keeps track of the best vector (lowest score) seen so far and
stops automatically when the given time budget (≈ 1000 s) is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution (real FFT) – used everywhere
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b via real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors that already satisfy
    ∑a = 1 (probability simplex).  Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Random start generator
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int) -> np.ndarray:
    """
    Produce a random probability vector of length n.
    The concentration parameter α is sampled uniformly from [0.2, 2.0] to
    obtain both dense and sparse patterns.  Afterwards a random fraction
    of entries is forced to zero (up to 30 %).
    """
    alpha = np.random.uniform(0.2, 2.0)
    raw = np.random.gamma(alpha, 1.0, size=n)
    # force a random fraction of zeros
    zero_frac = np.random.uniform(0.0, 0.3)
    if zero_frac > 0.0:
        mask = np.random.rand(n) < zero_frac
        raw[mask] = 0.0
        # avoid division by zero
        if raw.sum() == 0.0:
            raw[np.random.randint(n)] = 1.0
    vec = raw / raw.sum()
    return vec.astype(float)


# ----------------------------------------------------------------------
# 2. Smooth max optimisation with Adam
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 20000,
    lr: float = 0.4,
    eps_factor: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate  ε·log Σ exp(conv/ε)  using Adam.
    After each update the vector is projected onto the simplex.
    Returns the best (lowest exact score) vector encountered.
    """
    a = a0.copy()
    n = len(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # convolution and smooth‑max weights
        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        eps = maxc * max(eps_factor * (0.98 ** (t - 1)), 0.01)

        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)          # soft‑max, Σw = 1

        # gradient of the smooth surrogate:
        #   ∇ = 2 * (w_rev ⊛ a)   where w_rev = w reversed
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)     # length 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update (learning‑rate decays very slowly)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to escape shallow plateaux
        if t % 800 == 0:
            jitter = np.random.randn(n) * (lr * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Exact sub‑gradient descent (max‑only)
# ----------------------------------------------------------------------


def subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the true objective 2·n·max(a⊛a).
    The sub‑gradient is taken at a shift attaining the current maximum.
    """
    a = a0.copy()
    n = len(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                     # shift giving the max

        # sub‑gradient g_i = 2·a_{k-i}  (zero outside the support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # tiny random shake every thousand iterations
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Stochastic hill‑climbing polish
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    deadline: float,
    max_steps: int = 200_000,
) -> tuple[list, float]:
    """
    Local search that either swaps two coordinates or transfers a fraction
    of mass from i → j.  Only moves that improve the exact score are kept.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.10   # initial fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder
        else:
            step_scale *= 0.97                         # become more cautious

        max_steps -= 1

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Replace a by (a + reverse(a)) / 2 and re‑project.
    Symmetric vectors often have flatter self‑convolutions.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback to a known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Main multi‑restart loop
    # ------------------------------------------------------------------
    lengths = [1024, 1536, 2048, 3072, 4096]   # lengths where FFTs stay cheap
    outer_start = time.time()

    while time.time() < deadline - 5.0:
        # pick a length (occasionally try a larger one)
        n = int(np.random.choice(lengths))

        # random Dirichlet start
        a0 = random_dirichlet_start(n)

        cur_sc = compute_score(a0)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a0.tolist()
            print(f"[init] n={n:4d}  score={cur_sc:.6f}")

        # ------------------------------------------------------------------
        # 2. Adam optimiser on the smooth surrogate
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 12.0:
            break
        max_adam_iter = min(30_000, int(remaining * 35))   # ~35 it/s on a typical CPU
        a_adam, sc_adam = adam_optimize(
            a0,
            deadline - 2.0,
            max_iter=max_adam_iter,
            lr=0.45,
            eps_factor=0.12,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.tolist()
            print(f"[adam] n={n:4d}  score={sc_adam:.6f}")

        # ------------------------------------------------------------------
        # 3. Exact sub‑gradient descent for fine‑tuning
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 10.0:
            a_sub, sc_sub = subgrad_optimize(
                a_adam,
                deadline - 5.0,
                max_iter=70_000,
                step0=0.06,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[sub‑grad] n={n:4d}  score={sc_sub:.6f}")

        # ------------------------------------------------------------------
        # 4. Hill‑climbing polishing + symmetry averaging
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 8.0:
            seq_to_polish = best_seq if best_seq is not None else a_sub.tolist()
            polished_seq, polished_sc = hill_climb(
                seq_to_polish,
                deadline - 2.0,
                max_steps=150_000,
            )
            if polished_sc < best_score:
                best_score = polished_sc
                best_seq = polished_seq
                print(f"[hill‑climb] n={n:4d}  score={polished_sc:.6f}")

            # final cheap symmetry averaging (often improves the peak)
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[sym‑avg] n={n:4d}  score={sym_sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Safeguard – if nothing succeeded, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6. Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Best reported score = {final_score:.6f}")
    return best_seq
```
