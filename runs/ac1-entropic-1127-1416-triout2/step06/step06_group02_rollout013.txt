"""
Propose a low‑score step function.

Algorithm outline
-----------------
1. **Parameterised seeds** – generate a handful of vectors of different
   analytical shapes (arcsine‑like, exponential, power‑law, Dirichlet,
   uniform).  The length is taken large (≈ 8000) to give many degrees of
   freedom; the vectors are normalised to have ℓ₁‑norm 1 (the objective is
   scale‑invariant).

2. **Fast screening** – keep the best few candidates according to
   `compute_score` (the exact objective for unit sum).

3. **Smoothed max‑convolution descent** – each kept candidate is refined
   with a projected gradient method on the log‑sum‑exp surrogate of the
   max‑convolution.  An Adam‑style momentum update and a rapidly
   decreasing smoothing parameter ε are used; after every step the vector
   is projected onto the probability simplex.

4. **Simulated‑annealing permutation polishing** – the refined vector is
   further improved by stochastic swaps (and tiny mass‑transfer moves)
   accepted with a Metropolis criterion.  This explores the space of
   permutations of the (already‑optimised) values, which is crucial
   because the convolution depends on the ordering.

5. **Optional symmetry averaging** – occasionally averaging the vector
   with its reverse can lower the peak of the convolution; the improvement
   is kept if it helps.

All steps respect the global time budget (≈ 1000 s).  The routine falls
back to a uniform vector if nothing better is found.  The final candidate
is returned as a plain Python list – ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Exact score (for ℓ₁‑norm 1 vectors) ----------------------------------

    """
    Exact evaluation of `evaluate_sequence` for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Seed generators -------------------------------------------------------
# ----------------------------------------------------------------------


def arcsine_vector(n: int) -> np.ndarray:
    """
    Discrete analogue of the arcsine density.
    Heavier mass near the ends, lighter near the centre.
    """
    i = np.arange(n, dtype=float)
    # shift by 0.5 to avoid division by zero at the boundaries
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def exponential_vector(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def powerlaw_vector(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)  # start at 1 to avoid 0⁰
    w = 1.0 / (i ** p)
    w /= w.sum()
    return w


def dirichlet_vector(n: int, alpha: float) -> np.ndarray:
    w = np.random.gamma(alpha, 1.0, size=n)
    w /= w.sum()
    return w


def uniform_vector(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=float)


# ----------------------------------------------------------------------
# Smoothed max‑convolution gradient descent (Adam‑like) ---------------
# ----------------------------------------------------------------------


def smoothed_gradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 20000,
    step0: float = 0.45,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_factor: float = 0.25,
    min_eps_ratio: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Optimises the log‑sum‑exp surrogate of the max‑convolution.
    Uses Adam‑style momentum and projects onto the simplex after each step.
    Returns the best vector (lowest exact score) encountered.
    """
    a = a0.astype(float).copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # ----- convolution and smooth max ----------
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # shrinking epsilon, never smaller than a tiny fraction of maxc
        eps = max_eps = maxc * max(eps_factor * (0.995 ** (t - 1)), min_eps_ratio * maxc)

        # stable soft‑max over the full convolution
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / exp_vals.sum()           # ∑ w = 1

        # ----- gradient of the smooth max (as derived in the baseline) -----
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)    # length 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # ----- Adam update -----
        step = step0 / np.sqrt(t)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to help escape plateaus
        if t % 800 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing permutation polish -------------------------------
# ----------------------------------------------------------------------


def annealing_permutation(
    a0: np.ndarray,
    time_deadline: float,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over permutations (and tiny mass‑transfer moves).
    Only moves that improve the exact score are accepted with a
    Metropolis probability depending on a slowly cooling temperature.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc

    # initial temperature – a fraction of the current score
    T = cur_sc * 0.12
    cooling = 0.99995                # exponential cooling factor

    step_scale = 0.07                # typical transferred fraction

    for step in range(max_steps):
        if time.time() >= time_deadline:
            break

        # decide move type
        if np.random.rand() < 0.6:
            # swap two positions (pure permutation move)
            i, j = np.random.randint(n, size=2)
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # tiny mass transfer between two distinct indices
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * a[i]
            if delta <= 0:
                continue
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            # re‑project to keep the simplex constraint precise
            cand = project_to_simplex(cand)

        cand_sc = compute_score(cand)
        Δ = cand_sc - cur_sc

        if Δ < 0 or np.random.rand() < np.exp(-Δ / max(T, 1e-12)):
            a = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = cand.copy()

        # cooling schedule
        T *= cooling
        # shrink the transfer magnitude as we cool
        step_scale = max(0.001, step_scale * (0.9999 if T < cur_sc * 0.02 else 1.0))

    return best_a, best_sc


# ----------------------------------------------------------------------
# Symmetry averaging (optional cheap polish) ---------------------------
# ----------------------------------------------------------------------


def symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a + reverse(a))/2 and re‑project onto simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a plain
    Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback (the known best construction, if any)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a length – larger gives more freedom, but keep FFT cheap
    # ------------------------------------------------------------------
    lengths = [4096, 8192, 12288]          # a few candidates, ordered by difficulty
    # Pick the largest length that fits comfortably into the remaining time.
    # We start with the biggest and fallback if we run out of time.
    for n in lengths:
        if time.time() >= deadline - 30.0:
            break

        print(f"\n=== Working with length n = {n} ===")

        # ------------------------------------------------------------------
        # 2. Build a diverse pool of seed vectors
        # ------------------------------------------------------------------
        seed_vectors = []

        # a) arcsine‑like deterministic shape
        seed_vectors.append(arcsine_vector(n))

        # b) uniform (pure baseline)
        seed_vectors.append(uniform_vector(n))

        # c) several random Dirichlet draws with varying concentration
        for _ in range(4):
            alpha = np.random.uniform(0.02, 3.0)
            seed_vectors.append(dirichlet_vector(n, alpha))

        # d) several exponential decays with random rates
        for _ in range(4):
            lam = 10 ** np.random.uniform(-5, -1) * (2.0 / n)   # scale with n
            seed_vectors.append(exponential_vector(n, lam))

        # e) power‑law tails
        for _ in range(4):
            p = np.random.uniform(0.3, 2.5)
            seed_vectors.append(powerlaw_vector(n, p))

        # ------------------------------------------------------------------
        # 3. Quick screening – keep the 5 best according to the exact score
        # ------------------------------------------------------------------
        screened = []
        for vec in seed_vectors:
            sc = compute_score(vec)
            screened.append((sc, vec))
        screened.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in screened[:5]]
        print(f"Screened {len(seed_vectors)} seeds, best raw score = {screened[0][0]:.6f}")

        # ------------------------------------------------------------------
        # 4. Refine each seed with smoothed gradient descent
        # ------------------------------------------------------------------
        refined = []
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline - 40.0:
                break
            remaining = deadline - time.time()
            max_it = int(min(20000, remaining * 40))   # approx. 40 iter/s
            a_opt, sc_opt = smoothed_gradient_descent(
                seed_vec,
                time_deadline=deadline - 5.0,
                max_iter=max_it,
                step0=0.45,
                eps_factor=0.20,
            )
            refined.append((sc_opt, a_opt))
            print(f"[seed {idx}] after GD: score = {sc_opt:.6f}")

        # ------------------------------------------------------------------
        # 5. Pick the best refined vector and run simulated‑annealing polish
        # ------------------------------------------------------------------
        if refined:
            refined.sort(key=lambda x: x[0])
            cur_best_vec = refined[0][1]
            cur_best_sc = refined[0][0]

            if cur_best_sc < best_score:
                best_score = cur_best_sc
                best_seq = cur_best_vec.tolist()
                print(f"New overall best (pre‑anneal) score = {best_score:.6f}")

            # Annealing phase (more time‑intensive)
            if time.time() < deadline - 20.0:
                a_polished, sc_polished = annealing_permutation(
                    cur_best_vec,
                    time_deadline=deadline - 5.0,
                    max_steps=200_000,
                )
                if sc_polished < best_score:
                    best_score = sc_polished
                    best_seq = a_polished.tolist()
                    print(f"Annealing improved score = {best_score:.6f}")

                # Optional cheap symmetry averaging
                a_sym = symmetry_average(np.array(best_seq, dtype=float))
                sym_sc = compute_score(a_sym)
                if sym_sc < best_score:
                    best_score = sym_sc
                    best_seq = a_sym.tolist()
                    print(f"Symmetry averaging improved score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 6. If we still have enough budget, try the next larger length
        # ------------------------------------------------------------------
        if time.time() >= deadline - 30.0:
            break

    # ------------------------------------------------------------------
    # 7. Fallback to a uniform vector if nothing better was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 8. Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
