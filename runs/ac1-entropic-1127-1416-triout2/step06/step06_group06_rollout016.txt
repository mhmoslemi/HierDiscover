def targeted_hill_climb(a, deadline, max_iters=200000):
    n = len(a)
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.1
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute conv and index of max
        conv = convolution_fft(best, best)
        k = int(np.argmax(conv))
        # compute contributions vector
        contrib = np.zeros(n)
        # for i in [0,n): j=k-i if 0 <= j < n
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # vectorized:
        i = np.arange(i_start, i_end+1)
        j = k - i
        contrib[i] = best[i] * best[j]
        # choose i with largest contribution
        i_max = i[np.argmax(contrib[i])]
        # pick random j_target that is not i_max to move mass to
        j_target = np.random.randint(n)
        if j_target == i_max:
            continue
        # compute amount to transfer based on step_scale
        delta = step_scale * best[i_max]
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i_max] -= delta
        cand[j_target] += delta
        # projection onto simplex (small adjustments might produce negative)
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97
        it += 1
    return best.tolist(), best_score
```

"""
Hybrid optimizer for the step‑function “attack”.

Goal: minimise
    score(a) = 2·n·max(conv(a,a)) / (sum(a))**2
which is scale‑invariant.  We therefore work on the probability simplex
(sum = 1, a_i ≥ 0) and minimise 2·n·max(conv(a,a)).  The search consists of
three phases:

1️⃣ **Random restarts** – a Dirichlet draw (concentration α sampled
   uniformly in [0.1, 5]) supplies a diverse start point.
2️⃣ **Smoothed / hard‑max Adam optimiser** – a smooth soft‑max approximation
   of the max is used initially (λ grows from 1 to ≈80).  The gradient of the
   surrogate is 2·(a_rev ∗ w) (central part only).  An Adam‑style adaptive
   step is applied with multiplicative updates  a ← a·exp(‑η·g)  and the
   vector is renormalised to stay on the simplex.
   After enough iterations λ is set to a huge value, effectively switching
   to the true (hard) max and allowing a final fine‑tuning phase.
3️⃣ **Targeted hill‑climbing** – repeatedly moves a small fraction of mass
   from an index that contributes heavily to the current worst convolution
   entry to a random destination; the move is accepted iff the true score
   (the non‑smoothed objective) improves.  The transfer fraction adapts
   automatically.

The whole routine runs for at most `budget_s` seconds, keeps the best
sequence ever seen, prints occasional progress messages and finally returns
the best list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real FFT, fast)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT.
    The result length is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa == 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    """
    # If all entries are non‑positive the uniform vector is the projection.
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Adam‑style smoothed‑max optimiser
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.8,
    lam_start: float = 1.0,
    lam_end: float = 80.0,
    eval_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate of  max(conv(a,a)).
    Returns the best vector encountered and its (non‑smoothed) score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    # we use a loose schedule for λ: after 1 000 000 iterations it reaches lam_end
    lam_iters = 1_000_000

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # convolution a*a (length 2n‑1)
        conv = convolution_fft(a, a)
        M = np.max(conv)

        # λ schedule (smooth → hard max)
        if it < lam_iters:
            lam = lam_start + (lam_end - lam_start) * (it / lam_iters)
        else:
            # effectively a hard max (softmax becomes a delta)
            lam = 1e12

        # softmax (or near‑hard) weights
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)  # length 2n‑1, sums to 1

        # gradient of the surrogate:
        #   g = 2 * (a_rev ∗ w)  (central n entries)
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # Adam update (multiplicative)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / np.sqrt(it)          # basic decay
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # multiplicative update; clip to avoid overflow in exp
        step = np.clip(step, -50, 50)
        a = a * np.exp(-step)

        # renormalise to the simplex (positivity is guaranteed)
        a_sum = np.sum(a)
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # occasional true‑objective evaluation
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] iter {it:,}  new best {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climber (focuses on the current worst convolution entry)
# ----------------------------------------------------------------------


def targeted_hill_climb(
    seq,
    deadline,
    max_iters: int = 250_000,
    init_step: float = 0.2,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climber that always attacks the index contributing most
    to the current maximal convolution value.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = init_step
    it = 0

    while it < max_iters and time.time() < deadline:
        # current convolution and its worst shift
        conv = convolution_fft(best_a, best_a)
        k = int(np.argmax(conv))

        # contributions of each position to this shift
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        contrib = np.zeros(n, dtype=float)
        contrib[i_idx] = best_a[i_idx] * best_a[j_idx]

        if np.all(contrib == 0):
            # fallback to random move if somehow all zero
            i = np.random.randint(n)
        else:
            probs = contrib / contrib.sum()
            i = np.random.choice(n, p=probs)   # index with high contribution

        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * best_a[i]
        if delta <= 0:
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # project back onto the simplex (positivity + sum‑1)
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # grow a bit on success
        else:
            step_scale *= 0.97                         # shrink on failure

        it += 1

    return best_a.tolist(), best_score
