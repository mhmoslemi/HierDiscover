def smooth_gradient_descent_adam(a0, time_deadline, max_iter=5000, lr=0.5, eps_factor=0.2):
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    t = 0
    for it in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        maxc = conv.max()
        eps_smooth = maxc * max(eps_factor * (0.95 ** (it-1)), 0.01)  # decreasing epsilon
        exp_vals = np.exp((conv - maxc) / eps_smooth)
        w = exp_vals / exp_vals.sum()
        # gradient
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        idx_start = 2 * len(a) - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(len(a))]
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps) * m_hat
        a_new = a - step
        # projection onto simplex
        a_new = project_to_simplex(a_new)
        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new
        # optional jitter
        if it % 500 == 0:
            jitter = np.random.randn(len(a)) * (0.001 * (lr / (np.sqrt(it)))) 
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc

"""
Improved step‑function optimizer.

The search proceeds in four phases for each random start:

1.  Generate a diverse seed (uniform Dirichlet, sparse, linear, Gaussian).
2.  Adam‑based smooth‑max gradient descent (log‑sum‑exp surrogate of max‑conv).
3.  Exact sub‑gradient descent on the true objective.
4.  Simulated‑annealing hill‑climbing (mass transfers) with a final symmetry average.

All iterates are kept on the probability simplex (non‑negative, sum = 1), so the
fast `compute_score` (which implements the exact evaluation for sum = 1) can be
used throughout.  The routine respects the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# FFT‑based utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a⋆b using real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors that already satisfy
    sum(a) == 1 (the algorithm always keeps this invariant).
    Returns 2·n·max(a⋆a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum=1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def random_sparse_vector(n: int) -> np.ndarray:
    """Random probability vector with a random dense block (10‑90 % density)."""
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dense_vector(n: int) -> np.ndarray:
    """Dirichlet(1,…,1) → fully dense probability vector."""
    a = np.random.dirichlet(np.ones(n))
    return a


def linear_vector(n: int) -> np.ndarray:
    """Deterministic increasing linear profile, normalised to sum = 1."""
    x = np.arange(1, n + 1, dtype=float)
    a = x / x.sum()
    return a


def gaussian_vector(n: int) -> np.ndarray:
    """Discretised Gaussian centred in the middle, with σ≈n/4."""
    centre = (n - 1) / 2.0
    sigma = n / 4.0
    x = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 2. Smooth‑max (log‑sum‑exp) gradient descent with Adam
# ----------------------------------------------------------------------


def smooth_adam_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    base_lr: float = 0.05,
    eps_factor: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(a⋆a)

        f(a) = ε·log Σ_k exp( (a⋆a)_k / ε )

    using Adam. After each step the iterate is projected onto the simplex.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    t = 0

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # ----- smooth max -------------------------------------------------
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # epsilon shrinks but never below 1 % of the current maximum
        eps_s = maxc * max(eps_factor * (0.95 ** (it - 1)), 0.01)

        # stable soft‑max weights
        exp_vals = np.exp((conv - maxc) / eps_s)
        w = exp_vals / exp_vals.sum()          # Σ w = 1

        # gradient of the smoothed objective:
        #   ∇f = 2·(w_rev ⋆ a)   (where rev denotes reversal)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # ----- Adam update ------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = np.random.randn(n) * (base_lr * 0.003 / np.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on 2·n·max(a⋆a)."""
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient = 2·a_{k‑i} (zero outside the support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # tiny random shake every thousand iterations
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Simulated‑annealing hill climbing (mass‑transfer moves)
# ----------------------------------------------------------------------


def simulated_annealing(
    a0: np.ndarray,
    time_deadline: float,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber with Metropolis acceptance.
    Moves consist of transferring a fraction of mass from one index to another.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]

    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc

    # initial temperature – a small fraction of the current score
    temperature = cur_sc * 0.08
    decay = 0.99997  # slow exponential cooling

    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        steps += 1

        # pick two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue

        # transfer a random fraction of the mass at i to j
        frac = np.random.rand() * 0.15   # ≤15 % of the source weight
        delta = frac * a[i]
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        # no need for projection – sum is preserved and all entries stay ≥ 0
        cand_sc = compute_score(a_cand)

        if cand_sc < cur_sc:
            a = a_cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = a_cand.copy()
        else:
            # Metropolis acceptance
            prob = np.exp(-(cand_sc - cur_sc) / (temperature + 1e-12))
            if np.random.rand() < prob:
                a = a_cand
                cur_sc = cand_sc

        # cool the temperature
        temperature = max(temperature * decay, 1e-8)

        # occasional re‑symmetrisation (helps in many runs)
        if steps % 40000 == 0:
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Symmetry averaging (cheap polish)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a + reverse(a))/2 and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

    # Parameter pool
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]
    init_strategies = [random_sparse_vector,
                       random_dense_vector,
                       linear_vector,
                       gaussian_vector]

    outer_iter = 0
    # continue launching new starts while we have time
    while time.time() < deadline - 5.0:
        outer_iter += 1
        n = int(random.choice(lengths))
        init_fn = random.choice(init_strategies)
        a = init_fn(n)

        # quick score
        cur_sc = compute_score(a)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a.tolist()
            print(f"[{outer_iter}] init n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – smooth‑max Adam descent
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 6.0:
            break
        max_gd_iter = int(min(20000, remaining * 25))  # ~25 iters per sec
        a_gd, sc_gd = smooth_adam_descent(
            a,
            time_deadline=deadline - 3.0,
            max_iter=max_gd_iter,
            base_lr=0.06,
            eps_factor=0.2,
        )
        exact_sc = compute_score(a_gd)
        if exact_sc < best_score:
            best_score = exact_sc
            best_seq = a_gd.tolist()
            print(f"[{outer_iter}] after Adam n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – exact sub‑gradient refinement
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 10.0:
            a_sub, sc_sub = subgradient_descent(
                a_gd,
                time_deadline=deadline - 5.0,
                max_iter=50000,
                step0=0.06,
            )
            exact_sc = compute_score(a_sub)
            if exact_sc < best_score:
                best_score = exact_sc
                best_seq = a_sub.tolist()
                print(f"[{outer_iter}] after sub‑grad n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 4 – simulated‑annealing hill‑climb + symmetry polish
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 12.0:
            base = np.array(best_seq if best_seq is not None else a_sub, dtype=float)
            a_sa, sc_sa = simulated_annealing(
                base,
                time_deadline=deadline - 4.0,
                max_steps=250_000,
            )
            exact_sc = compute_score(a_sa)
            if exact_sc < best_score:
                best_score = exact_sc
                best_seq = a_sa.tolist()
                print(f"[{outer_iter}] after SA    n={n} score={best_score:.6f}")

            # final cheap symmetry average
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[{outer_iter}] symmetry avg improved: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if nothing was ever found, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
