"""
Hybrid Adam‑based optimizer for the aperiodic self‑convolution problem.

Goal:
    minimise  score(a) = 2·n·max(conv(a,a))   with   a_i ≥ 0 , Σa_i = 1

Approach
---------
* Work on the probability simplex (Σa = 1).  Optimise the smooth
  approximation  log‑sum‑exp_λ(conv)  of the infinity‑norm.
* Optimisation is performed in log‑space so that a_i stay positive:
        log_a ← log_a – η·∇_log f   (Adam step)
  After each step we exponentiate and project back onto the simplex.
* A temperature λ is increased from a small value (≈1) to a large one
  (≈200) so that the surrogate gradually turns into the true max.
* Many random restarts are executed within the time budget; every
  restart receives a slice of the remaining time.
* The best vector found is refined by a stochastic “mass‑transfer”
  hill‑climber that accepts only improving moves (with an adaptive
  step‑size).
* If a fallback sequence `height_sequence_1` is present it is used as an
  initial candidate.

The code is pure NumPy, uses FFT for O(n log n) convolutions and
respects the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real input → real output)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors via real FFT.
    Length of result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σx_i = 1}
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence for Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    lr: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 200.0,
    warmup_iters: int = 150_000,
    eval_interval: int = 800,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of a on the simplex.
    The objective is a smooth approximation of max(conv(a,a)):
        f_λ(a) = (1/λ)·log Σ_k exp(λ·conv_k)

    Parameters
    ----------
    init_a : np.ndarray (sum = 1, non‑negative)
        Starting point.
    deadline : float
        Unix time after which the routine must stop.
    lr : float
        Base learning‑rate for the Adam update (applied in log‑space).
    lam_start, lam_end : float
        Soft‑max temperature schedule.
    warmup_iters : int
        Number of iterations over which λ is linearly increased.
    eval_interval : int
        How often the true (non‑smoothed) score is evaluated.

    Returns
    -------
    best_a : np.ndarray
        Best vector found (still on the simplex).
    best_score : float
        Corresponding 2·n·max(conv) value.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    # log‑space representation (avoid log(0))
    eps = 1e-30
    log_a = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    it = 0

    best_a = a.copy()
    best_score = compute_score(a)

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1.  Convolution & soft‑max weights
        # --------------------------------------------------------------
        a = np.exp(log_a)                     # back to primal
        a = project_to_simplex(a)              # numerical drift correction
        log_a = np.log(a + eps)

        conv = convolution_fft(a, a)          # length 2n‑1
        M = conv.max()
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / warmup_iters)

        # stable softmax: subtract the max before exp
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

        # --------------------------------------------------------------
        # 2.  Gradient of the smoothed objective (w.r.t. a)
        #    g = 2 * (rev(a) * w)   – keep only the n central entries
        # --------------------------------------------------------------
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)    # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # --------------------------------------------------------------
        # 3.  Convert to gradient w.r.t. log‑a (chain rule)
        #    ∂f/∂log_a = (∂f/∂a) ⊙ a
        # --------------------------------------------------------------
        grad_log = g * a

        # --------------------------------------------------------------
        # 4.  Adam update in log‑space
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_a -= step

        # --------------------------------------------------------------
        # 5.  Periodic evaluation of the true (non‑smoothed) objective
        # --------------------------------------------------------------
        if it % eval_interval == 0:
            a_tmp = np.exp(log_a)
            a_tmp = project_to_simplex(a_tmp)
            cur_score = compute_score(a_tmp)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}]  iter {it:,}  new best {best_score:.6f}")

    # Final check with the latest point
    a_final = np.exp(log_a)
    a_final = project_to_simplex(a_final)
    final_score = compute_score(a_final)
    if final_score < best_score:
        best_score = final_score
        best_a = a_final.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (mass transfer)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps: int = 800_000):
    """
    Refine a candidate by moving a fraction of mass between two indices.
    Only improving moves are accepted; the transfer fraction adapts to
    the observed success rate.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.15   # initial transferred fraction

    steps = 0
    while steps < max_steps and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # amount to move (proportional to source mass)
        delta = step_scale * best_a[i]
        if delta <= 0:
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # keep numerical feasibility
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.08, 0.55)   # become bolder
        else:
            step_scale *= 0.97                          # become more cautious

        steps += 1

    return best_a.tolist(), best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 2025, budget_s: int = 1000, **_):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine launches a number of random restarts, each optimised by an
    Adam‑based smoothed‑max descent, and finishes with a short hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a safety margin

    # ------------------------------------------------------------------
    # Determine a base length.  Use the length of the supplied fallback if
    # it exists, otherwise fall back to 2048 (a good trade‑off).
    # ------------------------------------------------------------------
    try:
        n_base = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n_base = 2048

    # ------------------------------------------------------------------
    # Keep track of the best candidate seen so far.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback (may already be a decent solution)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore[name-defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main optimisation loop – keep launching restarts until we run out of
    # time.  Different lengths are tried (n = n_base, 2·n_base) to give the
    # optimizer more flexibility.
    # ------------------------------------------------------------------
    restart_counter = 0
    lengths = [n_base, n_base * 2]               # e.g. 2048 and 4096
    while time.time() < deadline - 12.0:         # reserve ~12 s for polishing
        restart_counter += 1
        n = int(np.random.choice(lengths))

        # Dirichlet draw with a random concentration (controls sparsity)
        alpha = np.random.uniform(0.3, 3.0)
        a0 = np.random.dirichlet(np.full(n, alpha))

        # Allocate a slice of the remaining budget to this restart.
        remaining = deadline - time.time()
        # At least 6 s for the optimiser, the rest for the final hill‑climb.
        sub_deadline = time.time() + max(6.0, remaining * 0.55)

        print(f"[{time.time():.1f}] Restart {restart_counter:3d} – n={n}, α={alpha:.2f}")

        refined_a, refined_score = adam_optimize(
            a0,
            deadline=sub_deadline,
            lr=0.12,
            lam_start=1.0,
            lam_end=200.0,
            warmup_iters=200_000,
            eval_interval=600,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"  ** new global best (raw) {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with the exact objective (hill‑climbing).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print("\n--- Final hill‑climb polishing ---")
        polished_seq, polished_val = hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=600_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Polish improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the extreme case that nothing was produced.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = n_base
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final verification (the harness will call evaluate_sequence again,
    # but we print the score for our own information).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished – final reported score = {final_score:.6f}")

    return best_seq
