def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # margin

    def fast_score(arr):
        n = arr.size
        conv = np.fft.irfft(np.fft.rfft(arr, n=next_pow2(2*n-1))*np.fft.rfft(arr, n=next_pow2(2*n-1)), n=next_pow2(2*n-1))
        maxc = np.max(conv[:2*n-1])
        return 2.0*n*maxc

    # shapes
    def uniform_shape(n):
        return np.full(n, 1.0/n)

    def arcsine_shape(n):
        i = np.arange(n, dtype=float)
        eps = 0.25
        vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
        vals /= np.sum(vals)
        return vals

    def beta_shape(n, a, eps):
        i = np.arange(n, dtype=float)
        left = (i + eps) ** (a - 1.0)
        right = (n - 1 - i + eps) ** (a - 1.0)
        base = left * right
        total = np.sum(base)
        if total == 0 or not np.isfinite(total):
            return uniform_shape(n)
        return base / total

    # evaluate mixture of shape with uniform
    lengths = [2048, 4096, 8192]   # choose maybe up to 8192 if time
    best_val = float('inf')
    best_vec = None

    # grid parameters
    alphas = [0.05,0.07,0.09,0.11,0.13,0.15,0.18,0.2,0.25,0.3,0.35,0.4,0.45,0.5,0.55,0.6,0.7,0.8,0.9,1.0,1.2,1.5,2.0]
    eps_vals = [0.0, 0.5, 1.0, 2.0]   # additive epsilon
    w_vals = np.linspace(0,1,21) # 0 to 1 step 0.05

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"Evaluating length {n}")
        # precompute uniform
        u = uniform_shape(n)
        # arcsine shape
        arcs = arcsine_shape(n)
        # baseline best candidates list
        candidates = []
        # evaluate arcsine alone
        s = fast_score(arcs)
        candidates.append((s, arcs))
        # uniform alone
        s = fast_score(u)
        candidates.append((s, u))
        # Loop over alphas and eps and w
        for a in alphas:
            if time.time() > deadline: break
            for eps in eps_vals:
                shape = beta_shape(n, a, eps)
                for w in w_vals:
                    if time.time() > deadline: break
                    # mixture w*shape + (1-w)*u
                    cand = w*shape + (1.0 - w)*u
                    # due to combination sum remains 1
                    sc = fast_score(cand)
                    if sc < best_val:
                        best_val = sc
                        best_vec = cand.copy()
                        print(f"New best {best_val:.6f} n={n} a={a} eps={eps} w={w}")
                    # store top few candidates (maybe for later local optimization)
                    candidates.append((sc, cand))
        # Keep top k for further refinement
        candidates.sort(key=lambda x: x[0])
        topk = [vec for _, vec in candidates[:5]]
        # run local refinement on topk using Adam with high lambda
        for init in topk:
            if time.time() > deadline - 5:
                break
            max_iter = int(min(200_000, (deadline - time.time())*30))
            # Adam optimizer similar to earlier
            a_opt, sc_opt = adam_optimize(init, deadline, max_iter=max_iter, lr0=0.1, lam_end=2000.0)
            # Evaluate official
            val = evaluate_sequence(a_opt.tolist())
            if val < best_val:
                best_val = val
                best_vec = a_opt.copy()
                print(f"New best after Adam {best_val:.6f}")

    # final fallback
    if best_vec is None:
        best_vec = uniform_shape(256)
        best_val = evaluate_sequence(best_vec.tolist())
    print(f"Best final value {best_val}")
    return best_vec.tolist()

"""
Propose a step sequence with a low official score.
The algorithm works as follows (total time ≤ budget_s seconds):

1. Try the historic fallback sequence (if present).
2. For a few candidate lengths (8192, 4096, 2048) generate cheap
   parametric shapes (uniform, arcsine, symmetric Beta(α,α) with a tiny
   regulariser).  All shapes are already normalised (Σa = 1).
3. Sweep α (and a few ε values) on a fine grid, keep the best shape
   according to the *exact* fast score 2·n·max(conv(a,a)).
4. Refine the best shape of each length by a long Adam optimisation
   on a smooth log‑sum‑exp surrogate of the max‑convolution.
   The temperature λ is annealed up to 2000 so the surrogate is
   virtually the true max.  After each Adam step the vector is projected
   onto the simplex and forced to be symmetric (a ← (a+a[::-1])/2).
5. A short exact sub‑gradient descent (with back‑tracking line‑search)
   guarantees that the official objective never increases.
6. Keep the globally best vector seen (according to the official scorer)
   and return it.  The routine never returns an empty list – a uniform
   fallback is used as a last resort.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for all n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def fast_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    # all non‑positive → uniform point
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact left‑right symmetry (preserves the sum)."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Shape factories
# ----------------------------------------------------------------------


def uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution – Σa = 1."""
    return np.full(n, 1.0 / n)


def arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine density (the classic baseline)."""
    i = np.arange(n, dtype=float)
    # tiny regulariser avoids a zero denominator at the extreme ends
    eps = 0.0
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def beta_shape(n: int, alpha: float, eps: float = 0.0) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on {0,…,n‑1} with a small additive regulariser.
    The vector is already normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (alpha - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return uniform_shape(n)
    return base / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (for Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate value B).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max weights over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp

    # smooth approx of the max
    B = max_c + np.log(sum_exp) / lam

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)            # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.10,
    lam_start: float = 1.0,
    lam_end: float = 2000.0,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (still normalised to Σa=1) and its fast score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    a = symmetrize(a)

    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # quadratic annealing of λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)
        a_new = symmetrize(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress report (kept cheap)
        if it % 50_000 == 0:
            print(f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected) on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the official objective.
    """
    a = project_to_simplex(init_a.copy())
    a = symmetrize(a)

    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # decaying step size
        step = step0 / np.sqrt(it)

        # simple back‑tracking to ensure descent
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            a_new = symmetrize(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40_000 == 0:
            print(f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

    # ------------------------------------------------------------------
    # Helper that evaluates a candidate with the *official* scorer.
    # ------------------------------------------------------------------
    def official_score(vec: np.ndarray) -> float:
        return evaluate_sequence(vec.tolist())

    # ------------------------------------------------------------------
    # 0.  Try the historic fallback (if it exists).
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = official_score(np.array(fallback))
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Scan a few vector lengths.
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048]   # larger lengths first – they give a finer discretisation
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Length n = {n} ===")

        # ----- Baselines (uniform & arcsine) -----
        uni = uniform_shape(n)
        uni_sc = fast_score(uni)
        arc = arcsine_shape(n)
        arc_sc = fast_score(arc)

        # keep the best of the two as a quick reference
        if uni_sc < best_val:
            best_val, best_seq = uni_sc, uni.tolist()
        if arc_sc < best_val:
            best_val, best_seq = arc_sc, arc.tolist()

        print(f" Uniform fast score = {uni_sc:.6f}")
        print(f" Arcsine fast score = {arc_sc:.6f}")

        # ----- Grid search over symmetric Beta shapes -----
        # we keep the top few candidates for a local refinement step
        alphas = np.linspace(0.40, 1.20, 41)          # step ≈0.02
        eps_candidates = [0.0, 0.5, 1.0]             # tiny regularisers
        candidates: List[Tuple[float, np.ndarray]] = []

        for eps in eps_candidates:
            for a in alphas:
                if time.time() >= deadline - 5.0:
                    break
                shape = beta_shape(n, a, eps)
                sc = fast_score(shape)
                candidates.append((sc, shape))

        # include the deterministic starters as well
        candidates.append((uni_sc, uni))
        candidates.append((arc_sc, arc))

        # keep the few best candidates (they will be refined later)
        candidates.sort(key=lambda x: x[0])
        top_vectors = [vec for _, vec in candidates[:5]]

        # ----- Local refinement of each survivor -----
        for vec in top_vectors:
            if time.time() >= deadline - 3.0:
                break

            # Adam stage – large temperature → almost exact max
            remaining = deadline - time.time()
            max_iter_adam = int(min(300_000, remaining * 30))
            print(f"  Adam polishing (max_iter={max_iter_adam}) …")
            a_adam, _ = adam_optimize(
                init_a=vec,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.09,
                lam_start=1.0,
                lam_end=2000.0,
                noise_every=0,
            )

            # Sub‑gradient stage – guarantees monotone improvement
            if time.time() < deadline - 2.0:
                remaining = deadline - time.time()
                max_iter_sub = int(min(100_000, remaining * 20))
                print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
                a_sub, _ = subgradient_optimize(
                    init_a=a_adam,
                    deadline=deadline,
                    step0=0.22,
                    max_iter=max_iter_sub,
                )
                a_refined = a_sub
            else:
                a_refined = a_adam

            # Official evaluation
            val = official_score(a_refined)
            print(f"   → official score after refinement = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_refined.tolist()
                print(f"   *** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Global polishing of the current best (if any time left).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_adam = int(min(400_000, remaining * 35))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.08,
            lam_start=1.0,
            lam_end=2500.0,
            noise_every=0,
        )

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(120_000, remaining * 20))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                init_a=a,
                deadline=deadline,
                step0=0.20,
                max_iter=max_iter_sub,
            )

        final_val = official_score(a)
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_shape(256).tolist()
        best_val = official_score(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
