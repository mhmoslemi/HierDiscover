"""
Propose a step‑function (non‑negative list) with a low official score.

Algorithm
---------

1. **Very fast exploration** – for several lengths (256 … 8192) we draw
   many beta‑family shapes (single, mixture, and a “sparse” Dirac‑like
   pattern) and keep the few best according to the *fast* surrogate
   `2·n·max(conv(a,a))` (the denominator is set to 1 by normalising to a
   probability simplex).  This stage is deliberately cheap.

2. **Adam on a smooth log‑sum‑exp surrogate** – each survivor is refined
   by Adam on the differentiable surrogate of `max(conv)`.  The temperature
   λ is annealed from 1 up to a very large value (≈2000) so that the
   surrogate becomes virtually the true max.

3. **Multiplicative (mirror‑descent) polishing** – a few hundred thousand
   iterations of an exponentiated‑gradient update (which respects the
   non‑negativity automatically) are performed on the exact max‑convolution
   sub‑gradient.  This often reduces the peak further than plain
   sub‑gradient descent.

4. **Exact projected sub‑gradient descent** – a classic sub‑gradient
   method with a back‑tracking line‑search guarantees that the official
   objective never increases.

5. **Stochastic Gaussian walk** – a cheap random walk on the simplex,
   with occasional uphill moves, gives the final fine‑tuning.

All stages respect the global 1000 s budget and continually keep the
best vector seen (according to the *official* `evaluate_sequence`).  If
nothing is found, a uniform fallback of length 256 is returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for any length)."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of self‑convolution."""
    return float(np.max(_conv_fft(a, a)))


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    # Duchi‑et‑al. (2008) – robust and O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv) – assumes Σa = 1."""
    n = a.shape[0]
    return 2.0 * n * _max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def _beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def _beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    a1 = _beta_shape(n, alpha1, beta1, eps1)
    a2 = _beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def _sparse_random(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Place equal mass on m random distinct positions."""
    if m >= n:
        return np.full(n, 1.0 / n)
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and Adam optimiser
# ----------------------------------------------------------------------


def _smooth_grad(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate value).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)                     # length 2n‑1
    mval = np.max(conv)

    # numerically stable soft‑max
    exp_term = np.exp(lam * (conv - mval))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    B = mval + np.log(sum_exp) / lam           # smooth approximation of max

    # gradient of Σ_k w_k (a*a)_k  = 2 * (a ★ rev(w))
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)            # length 3n‑2
    L = 2 * n - 1
    weighted = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def _adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.10,
    lam_start: float = 1.0,
    lam_end: float = 2000.0,
    noise_every: int = 2000,
    noise_scale: float = 2e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth log‑sum‑exp surrogate.
    Returns the best vector (projected onto the simplex) seen.
    """
    a = _project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = _fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # quadratic annealing of λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = _smooth_grad(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = _project_to_simplex(a_new)

        cur_sc = _fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 50_000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
# ----------------------------------------------------------------------


def _mirror_descent_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    decay: float = 0.9995,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Multiplicative mirror descent using the exact sub‑gradient of the
    max‑convolution.  Keeps the vector on the simplex by simple
    renormalisation after each step.
    """
    a = _project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = _fast_score(a)

    n = a.size
    eta = eta0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active peak
        i = np.arange(n)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # exponentiated‑gradient update (mirror descent w.r.t. KL)
        update = np.exp(-eta * grad)
        a = a * update
        a = _project_to_simplex(a)

        cur_sc = _fast_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        eta *= decay

        if it % 40_000 == 0:
            print(
                f"    Mirror‑descent iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (with back‑tracking)
# ----------------------------------------------------------------------


def _subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the official objective.
    """
    a = _project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = _fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)

        for _ in range(12):
            a_new = a - step * grad
            a_new = _project_to_simplex(a_new)
            new_max = float(np.max(_conv_fft(a_new, a_new)))
            if new_max < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = _fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40_000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc

# Stochastic local Gaussian walk (final polishing)
# ----------------------------------------------------------------------


def _random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Cheap random walk on the simplex, with a vanishing step‑size.
    Occasionally accepts a tiny uphill move to escape plateaus.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = _fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = _project_to_simplex(a_prop)

        cur_sc = _fast_score(a_prop)

        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50_000 == 0:
            print(
                f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc

    # 1.  Exploratory sampling on several lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Sampling length n = {n} ===")

        # --------------------------------------------------------------
        # deterministic starters
        # --------------------------------------------------------------
        starters = [
            np.full(n, 1.0 / n),
            _arcsine_shape(n),
        ]
        # a few random sparse seeds (use a handful of points)
        for m in (int(np.sqrt(n) // 2), int(np.sqrt(n)), int(2 * np.sqrt(n))):
            starters.append(_sparse_random(n, max(2, min(n, m)), rng))

        local_best = None
        local_best_sc = float("inf")
        for s in starters:
            sc = _fast_score(s)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = s.copy()

        # --------------------------------------------------------------
        # random beta‑family and mixture shapes
        # --------------------------------------------------------------
        max_samples = int(5_000 * (budget_s / 1000.0))
        for _ in range(max_samples):
            if time.time() >= deadline - 30.0:
                break

            if rng.random() < 0.4:   # mixture
                w = rng.random()
                alpha1 = rng.uniform(0.01, 3.0)
                beta1 = rng.uniform(0.01, 3.0)
                eps1 = rng.uniform(1e-9, max(0.01, 0.02 * n))
                alpha2 = rng.uniform(0.01, 3.0)
                beta2 = rng.uniform(0.01, 3.0)
                eps2 = rng.uniform(1e-9, max(0.01, 0.02 * n))
                cand = _beta_mixture(
                    n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
                )
            else:                     # plain beta
                alpha = rng.uniform(0.01, 3.0)
                beta = rng.uniform(0.01, 3.0)
                eps = rng.uniform(1e-9, max(0.01, 0.02 * n))
                cand = _beta_shape(n, alpha, beta, eps)

            sc = _fast_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = cand.copy()

        # --------------------------------------------------------------
        # 2.  Polish the locally best candidate (if we have any time)
        # --------------------------------------------------------------
        if local_best is not None and time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            # allocate the majority of the remaining budget to Adam
            max_iter_adam = int(min(800_000, remaining * 40))
            print(f"  Polishing with Adam (max_iter={max_iter_adam}) …")
            a_polished, _ = _adam_optimize(
                init_a=local_best,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.09,
                lam_start=1.0,
                lam_end=2000.0,
                noise_every=1300,
                noise_scale=2e-5,
            )

            # Mirror‑descent stage – often reduces the peak further
            if time.time() < deadline - 5.0:
                max_iter_mirror = int(min(300_000, (deadline - time.time()) * 35))
                print(f"  Mirror‑descent (max_iter={max_iter_mirror}) …")
                a_polished, _ = _mirror_descent_optimize(
                    a_polished,
                    deadline,
                    eta0=0.6,
                    decay=0.9995,
                    max_iter=max_iter_mirror,
                )

            # Exact sub‑gradient descent
            if time.time() < deadline - 4.0:
                max_iter_sub = int(min(200_000, (deadline - time.time()) * 30))
                print(f"  Sub‑gradient descent (max_iter={max_iter_sub}) …")
                a_polished, _ = _subgradient_optimize(
                    a_polished,
                    deadline,
                    step0=0.28,
                    max_iter=max_iter_sub,
                )

            # Final random walk
            if time.time() < deadline - 2.0:
                max_iter_rw = int(min(350_000, (deadline - time.time()) * 45))
                print(f"  Random walk (max_iter={max_iter_rw}) …")
                a_polished, _ = _random_local_search(
                    a_polished,
                    deadline,
                    init_step=0.012,
                    max_iter=max_iter_rw,
                )

            # Official evaluation for this length
            val = _official_score(a_polished)
            print(f"  → official score for length {n} = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_polished.tolist()
                print(f"  *** New global best (len={n}) = {best_val:.6f} ***")
        else:
            # fallback: evaluate the raw best sampler
            if local_best is not None:
                val = _official_score(local_best)
                print(f"  Raw candidate official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = local_best.tolist()
                    print("*** Updated global best (raw) ***")

    # ------------------------------------------------------------------
    # 3. Global polishing of the overall best vector (if we still have time)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # Adam polishing – very high λ to approximate the true max
        remaining = deadline - time.time()
        max_iter_adam = int(min(900_000, remaining * 45))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = _adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.08,
            lam_start=1.0,
            lam_end=2500.0,
            noise_every=1400,
            noise_scale=1e-5,
        )

        # Mirror‑descent polishing
        if time.time() < deadline - 4.0:
            max_iter_mirror = int(min(350_000, (deadline - time.time()) * 40))
            print(f"  Mirror‑descent polishing (max_iter={max_iter_mirror}) …")
            a, _ = _mirror_descent_optimize(
                a,
                deadline,
                eta0=0.5,
                decay=0.9998,
                max_iter=max_iter_mirror,
            )

        # Exact sub‑gradient polishing
        if time.time() < deadline - 3.0:
            max_iter_sub = int(min(250_000, (deadline - time.time()) * 30))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = _subgradient_optimize(
                a,
                deadline,
                step0=0.22,
                max_iter=max_iter_sub,
            )

        # Final stochastic walk
        if time.time() < deadline - 2.0:
            max_iter_rw = int(min(400_000, (deadline - time.time()) * 50))
            print(f"  Random walk polishing (max_iter={max_iter_rw}) …")
            a, _ = _random_local_search(
                a,
                deadline,
                init_step=0.011,
                max_iter=max_iter_rw,
            )

        final_val = _official_score(a)
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = _official_score(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
