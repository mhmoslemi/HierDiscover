def exp_gradient_optimize(start_a, time_deadline, max_iters=200_000, beta_start=5., beta_end=80., step0=0.5):
    a = start_a.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # compute smoothed gradient
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        a_rev = a[::-1]

        grad = 2.0 * conv_rev[n-1: n-1 + n]

        # multiplicative update
        step = step0 / np.sqrt(it)

        a = a * update_factor
        # renormalize to sum=1

    # final check

def upsample(a_small, n_large):
    x_small = np.linspace(0, 1, len(a_small), endpoint=False)
    x_large = np.linspace(0, 1, n_large, endpoint=False)

    a_large = np.maximum(a_large, 0)

def smooth_adam_optimize(start_a, time_deadline, max_iters=200_000,
                         beta_start=5.0, beta_end=80.0,
                         step0=0.05,
                         adam_beta1=0.9, adam_beta2=0.999, epsilon=1e-8):
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    for it in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        # annealing beta
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac
        # compute gradient of smoothed objective
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n-1: n-1 + n]
        # Adam updates
        m = adam_beta1 * m + (1 - adam_beta1) * grad
        v = adam_beta2 * v + (1 - adam_beta2) * (grad * grad)
        m_hat = m / (1 - adam_beta1 ** it)
        v_hat = v / (1 - adam_beta2 ** it)
        step = step0 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + epsilon)
        # project
        a = project_to_simplex(a)
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_a, best_score

        conv = convolution_fft(a, a)
        maxc = conv.max()

        grad = 2.0 * conv_rev[n-1: n-1 + n]

        step = step0 / np.sqrt(it)
        # multiplicative update
        update = np.exp(-step * grad)
        a = a * update
        a = np.maximum(a, eps)  # avoid zeros

        if it % 200 == 0:

        best_a = a.copy()
    return best_a, best_score

"""
Propose a low‑score step function.

The search consists of three phases:

1. **Multiple random restarts on a modest size (≈2048)** –  
   each restart runs a fast Adam‑based minimisation of a smoothed
   `max(conv)` objective (log‑sum‑exp smoothing).  The best vector
   found is kept.

2. **Upsample the best vector to the full length (≈4096) and refine** –  
   a mirror‑descent (exponentiated‑gradient) optimiser on the same
   smoothed objective is applied, followed by a classic sub‑gradient
   descent on the exact `max`‑objective.

3. **Stochastic hill‑climb polishing** – random swaps / mass‑transfer
   moves are accepted if they improve the exact score.

All optimisation respects the non‑negative simplex (∑a = 1); the
official evaluation `evaluate_sequence` is scale‑invariant, so the
final list can be returned as‑is.  The routine respects a 1000 s
wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a ⊛ b using a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for Σa = 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed optimisation – Adam (additive) version
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 80.0,
    step0: float = 0.05,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    epsilon: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of `max(conv(a,a))` using Adam.
    The smoothed objective is
        f_β(a) = (1/β)·log Σ exp(β·conv_i)

    Returns the best vector seen and its exact (non‑smooth) score.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal smoothing parameter β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # smoothed gradient
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + epsilon)

        # projection onto simplex
        a = project_to_simplex(a)

        # occasional exact evaluation
        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed optimisation – Mirror‑descent (exponentiated‑gradient) version
# ----------------------------------------------------------------------


def smooth_exp_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 250_000,
    beta_start: float = 6.0,
    beta_end: float = 100.0,
    step0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) optimisation of the same smoothed
    objective.  Updates are multiplicative, keeping the iterate inside
    the simplex without an explicit Euclidean projection.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        step = step0 / np.sqrt(it)

        # multiplicative update, keep positivity
        a = a * np.exp(-step * grad)
        a = np.maximum(a, eps)
        a = a / a.sum()

        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimisation of the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    2·n·max(conv(a,a)).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max (2·a shifted)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 200_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb that swaps entries or transfers a fraction
    of mass between two positions, always projecting back onto the simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial mass‑transfer fraction

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def upsample_sequence(seq_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linearly interpolate a probability vector from a small to a
    larger length, preserving the simplex constraint.
    """
    seq_small = np.asarray(seq_small, dtype=float)
    n_small = seq_small.shape[0]

    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_large = np.linspace(0.0, 1.0, n_large, endpoint=False)

    seq_large = np.interp(x_large, x_small, seq_small, left=0.0, right=0.0)
    seq_large[seq_large < 0] = 0.0
    total = seq_large.sum()
    if total == 0.0 or not np.isfinite(total):
        seq_large = np.full(n_large, 1.0 / n_large)
    else:
        seq_large = seq_large / total
    return seq_large
