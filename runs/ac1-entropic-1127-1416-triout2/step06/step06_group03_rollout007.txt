"""
Optimises a non‑negative step sequence to minimise

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / sum(seq)^2

The implementation mixes several ideas that proved useful for the
previous version:

* FFT‑based convolution (real‑valued rfft) for speed.
* A smooth log‑sum‑exp surrogate of the max‑operator.  The temperature
  β is annealed from a modest start (β≈5) to a large final value
  (β≈300).  The surrogate is convex for fixed β and its gradient can
  be computed analytically via a weighted convolution.
* The vector is parametrised by a soft‑max (z → a); this enforces
  non‑negativity and the simplex constraint Σa = 1 automatically,
  allowing us to use the unconstrained Adam optimiser.
* A diverse pool of initial seeds (arcsine, uniform, mixtures,
  beta‑family, power‑family and Dirichlet draws) is generated for each
  candidate length.  The cheap surrogate 2·n·max(conv) is used to keep
  only the most promising seeds.
* After the Adam phase the best candidate is polished by an exact
  projected sub‑gradient method for the true max‑objective.
* Larger sequence lengths (8192, 4096, 2048) are tried first – the
  asymptotic bound improves with n.

The routine respects the wall‑clock budget (default 1000 s) and always
returns a valid non‑negative sequence.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(init_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 100_000,
                              step0: float = 0.1) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance to treat near‑max values as active
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # Sub‑gradient = 2 * shift(a, K) summed over all active K
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)          # length = n + L - 1
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional progress report (quiet for speed)
        # if it % 5000 == 0:
        #     print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Soft‑max helpers
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def smooth_adam_optimize(z0: np.ndarray,
                         deadline: float,
                         n: int,
                         beta0: float = 5.0,
                         beta_max: float = 300.0,
                         ramp_iters: int = 80_000,
                         lr: float = 0.03) -> np.ndarray:
    """
    Minimise the smooth approximation of max(conv) using Adam.

    Parameters
    ----------
    z0 : np.ndarray
        Starting point for the log‑soft‑max parametrisation.
    deadline : float
        Unix timestamp after which the optimiser must stop.
    n : int
        Length of the vector (the optimiser keeps Σa = 1).
    beta0, beta_max : float
        Initial and final temperature for the log‑sum‑exp surrogate.
    ramp_iters : int
        Over how many iterations β is increased linearly.
    lr : float
        Base learning rate for Adam.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥ 0, Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # log‑sum‑exp weights  w ∝ exp(β·(conv−mx))
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2 * (a ⋆ w_rev) restricted to original length
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = n + (2n-1) -1 = 3n-2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through the soft‑max map
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional debug output every few thousand steps
        # if t % 5000 == 0:
        #     print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final a (should already be on the simplex)
    return softmax(z)

# Seed generators – give the optimiser diverse starting points
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑type shape: heavy at the ends, light in the middle."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def uniform_initial(n: int) -> np.ndarray:
    """Purely uniform distribution."""
    return np.full(n, 1.0 / n)


def mixture_shape(n: int, weight_arc: float = 0.5) -> np.ndarray:
    """Convex combination of arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = uniform_initial(n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def dirichlet_initial(n: int, alpha: float = 0.3, rng=None) -> np.ndarray:
    """Sparse Dirichlet draw (α < 1 gives many near‑zero entries)."""
    if rng is None:
        rng = np.random.default_rng()
    draws = rng.gamma(alpha, 1.0, size=n)
    total = draws.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return draws / total


def beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family shape on a discrete grid."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return uniform_initial(n)
    return base / total


def power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape – essentially a discretised arcsine with exponent p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return uniform_initial(n)
    return vals / total

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The routine mixes several diverse seeds, a smooth‑max Adam phase,
    and a short exact sub‑gradient polishing phase.  Larger lengths are
    tried first because the asymptotic bound improves with n.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously discovered sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Try progressively smaller lengths (larger n → better asymptotic)
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a modest but diverse set of seeds
        # --------------------------------------------------------------
        seeds: List[np.ndarray] = [
            arcsine_initial(n),
            uniform_initial(n),
            mixture_shape(n, 0.5),
        ]

        # Random beta‑family shapes
        for _ in range(5):
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(beta_shape(n, a, b, eps))

        # Random power‑law shapes
        for _ in range(5):
            p = rng.uniform(0.1, 4.0)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(power_shape(n, p, eps))

        # Dirichlet (sparse) draws
        for _ in range(5):
            alpha = rng.uniform(0.01, 0.5)
            seeds.append(dirichlet_initial(n, alpha=alpha, rng=rng))

        # Keep the most promising seeds according to the cheap surrogate
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]  # up to five per length

        # --------------------------------------------------------------
        # 1b. Allocate a slice of time for each seed (reserve a little
        #     for final polishing)
        # --------------------------------------------------------------
        remaining = deadline - time.time() - 10.0
        if remaining <= 0:
            break
        per_seed = max(2.0, remaining / (len(top_seeds) * 2))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 5.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed:.1f}s")
            seed_deadline = time.time() + per_seed

            # Initialise Adam with a log‑softmax (avoids zeros)
            eps = 1e-14
            z0 = np.log(seed_vec + eps)

            # ----------------------------------------------------------
            # 2. Smooth‑max Adam optimisation
            # ----------------------------------------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=300.0,
                ramp_iters=80_000,
                lr=0.03,
            )

            # ----------------------------------------------------------
            # 3. Official evaluation
            # ----------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   evaluation failed: {e}")

            # ----------------------------------------------------------
            # 4. Light exact polishing (if time permits)
            # ----------------------------------------------------------
            if time.time() < deadline - 5.0:
                polish_deadline = deadline - 2.0
                cand_arr = np.array(best_seq, dtype=float)
                cand_arr = subgradient_descent_exact(
                    cand_arr,
                    polish_deadline,
                    max_iter=120_000,
                    step0=0.12,
                )
                try:
                    val2 = evaluate_sequence(cand_arr.tolist())
                    print(f"   polished score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, cand_arr.tolist()
                        print(f"   *** Updated best after polish: {best_val:.6f} ***")
                except Exception:
                    pass

    # ------------------------------------------------------------------
    # 5. Final global polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=200_000,
            step0=0.08,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 1024
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
