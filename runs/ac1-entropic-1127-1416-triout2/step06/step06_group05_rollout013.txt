"""
Propose a non‑negative step sequence with a low official score.

The search works in three stages:

1. **Fast parametric exploration** – for several lengths (up to 8192) we
   generate beta‑family shapes a_i ∝ (i+ε)^{α‑1}(n‑1‑i+ε)^{β‑1}
   (including symmetric β=α).  A small grid together with a few random
   perturbations is evaluated with a fast FFT‑based surrogate
   ``fast_score = 2·n·max(conv(a,a))`` (the denominator is 1 because the
   vectors are normalised to Σa=1).  The best shape of each length is kept.

2. **Local polishing** – the surviving shape is refined by Adam on a
   smooth log‑sum‑exp surrogate of the max‑convolution (λ is annealed up
   to 800) and finally by a projected sub‑gradient descent on the true
   max‑convolution.  All steps respect the remaining time budget.

3. **Global polishing** – the overall best vector found so far is again
   polished with Adam, sub‑gradient descent and a cheap random walk.
   The final candidate is evaluated with the official ``evaluate_sequence``.

The routine respects the 1000 s wall‑clock budget, prints progress and
always returns a non‑empty list of non‑negative numbers.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT based convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for any n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official objective.
    The vector is assumed to be normalised to Σa = 1,
    therefore the denominator is 1 and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (beta families and arcsine)
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic discrete arcsine density (the shape used in the baseline)."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # tiny regulariser to avoid division by zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (for Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value).  The vector a is assumed to lie on the
    simplex (Σa = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                       # soft‑max weights
    B = max_c + np.log(sum_exp) / lam            # smooth approximation of the max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)  # should be 1, kept for safety
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 500.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth log‑sum‑exp surrogate.
    Returns the best vector found (projected onto the simplex) together
    with its fast score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # quadratic annealing of the temperature λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected) with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 80_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the true objective.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term (active index)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # decaying step size
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Cheap Gaussian walk on the simplex.  Occasionally accepts a tiny
    uphill move to escape plateaus.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = fast_score(a_prop)

        # accept if strictly better, otherwise a tiny chance
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

    return best_a, best_sc

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine combines a fast parametric search (beta‑family shapes)
    with short Adam / sub‑gradient refinements and a final polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # helper to evaluate the official objective
    def official_score(vec: np.ndarray) -> float:
        return evaluate_sequence(vec.tolist())

    # ------------------------------------------------------------------
    # 0.  Fallback to a known construction if it exists
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]       # type: ignore
        val = official_score(np.array(fallback))
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Fast parametric search over several lengths
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]
    TARGET = 1.503  # we try to beat this

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Exploring length {n} ===")

        # start from the classic arcsine shape
        best_local = arcsine_shape(n)
        best_local_sc = fast_score(best_local)

        # ---- systematic grid on (α,ε) with β = α (symmetric) ----
        alphas = np.linspace(0.30, 0.80, 11)        # 0.30 … 0.80
        eps_vals = np.geomspace(1e-4, 2.0, 12)     # 10⁻⁴ … 2

        for alpha in alphas:
            for eps in eps_vals:
                if time.time() >= deadline - 5.0:
                    break
                cand = generate_beta_shape(n, alpha, alpha, eps)
                sc = fast_score(cand)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = cand

        # ---- random beta‑family samples (including asymmetric) ----
        max_samples = 2000
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break
            alpha = rng.uniform(0.02, 2.0)
            beta = rng.uniform(0.02, 2.0)
            eps = rng.uniform(1e-6, max(0.5, 0.01 * n))
            cand = generate_beta_shape(n, alpha, beta, eps)
            sc = fast_score(cand)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local = cand

        # ---- few mixtures of two beta shapes ----
        for _ in range(500):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            a1 = generate_beta_shape(
                n,
                rng.uniform(0.02, 2.0),
                rng.uniform(0.02, 2.0),
                rng.uniform(1e-6, max(0.5, 0.01 * n)),
            )
            a2 = generate_beta_shape(
                n,
                rng.uniform(0.02, 2.0),
                rng.uniform(0.02, 2.0),
                rng.uniform(1e-6, max(0.5, 0.01 * n)),
            )
            mix = w * a1 + (1.0 - w) * a2
            mix /= mix.sum()
            sc = fast_score(mix)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local = mix

        # ---- brief Adam polishing of the best local candidate ----
        if time.time() < deadline - 5.0:
            sub_dead = min(deadline, time.time() + 30.0)  # at most 30 s
            best_local, _ = adam_optimize(
                init_a=best_local,
                deadline=sub_dead,
                max_iter=120_000,
                lr0=0.10,
                lam_start=1.0,
                lam_end=800.0,
                noise_every=1300,
                noise_scale=3e-5,
            )

        # official evaluation for this length
        val = official_score(best_local)
        print(f"Length {n} candidate official score = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = best_local.tolist()
            print(f"*** New global best (len={n}) = {best_val:.6f} ***")

        if best_val <= TARGET:
            print("Target reached – stopping early.")
            break

    # ------------------------------------------------------------------
    # 2.  Global polishing of the overall best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 10.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # A – extended Adam run
        sub_dead = deadline
        max_iter_adam = int(min(400_000, (deadline - time.time()) * 35))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.09,
            lam_start=1.0,
            lam_end=1000.0,
            noise_every=1500,
            noise_scale=2e-5,
        )

        # B – exact sub‑gradient descent
        if time.time() < deadline - 4.0:
            max_iter_sub = int(min(150_000, (deadline - time.time()) * 30))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                deadline,
                step0=0.22,
                max_iter=max_iter_sub,
            )

        # C – final random walk
        if time.time() < deadline - 2.0:
            max_iter_loc = int(min(250_000, (deadline - time.time()) * 45))
            print(f"  Random walk polishing (max_iter={max_iter_loc}) …")
            a, _ = random_local_search(
                a,
                deadline,
                init_step=0.012,
                max_iter=max_iter_loc,
            )

        final_val = official_score(a)
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # uniform fallback (length 256) – guaranteed to be valid
        n = 256
        best_seq = [1.0 / n] * n
        best_val = official_score(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
