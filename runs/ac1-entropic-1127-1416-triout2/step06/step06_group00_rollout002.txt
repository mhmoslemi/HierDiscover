"""
Propose a low‑score step‑function.

The algorithm works in three phases:

1. **Multi‑scale smoothed optimisation** –  Starting from many random
   sparse vectors on a small grid (length 512) we minimise a smooth
   proxy of the max‑convolution (log‑sum‑exp) using the Adam optimiser.
   The best result is up‑sampled to the next grid size (1024 → 2048 →
   4096) and refined again.  This multi‑scale approach gives the
   optimiser a good initial shape and considerably expands the search
   space without a large time penalty.

   optimum.

3. **Stochastic hill‑climb polishing** –  Finally we perform a cheap
   hill‑climb that transfers a small fraction of mass between two
   positions (or swaps them) while always projecting back onto the
   probability simplex.  The best sequence seen during the whole run
   is returned.

All steps respect the 1000 s wall‑clock limit, use only NumPy and

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a ⊛ b using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation up‑sampling to `new_len` elements."""
    old_len = len(seq)
    if old_len == new_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    up = np.interp(x_new, x_old, seq)
    up = up / up.sum()
    return up

                    adam_beta2: float = 0.999,
                    eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate

        f_beta(a) = (1/β)·log Σ_i exp(β·conv_i(a))

    with Adam on the simplex.  Returns the best vector (sum=1) and its
    exact score 2·n·max(conv).
    """
    a = project_to_simplex(start_a.astype(float))
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ---- anneal β -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- smoothed objective ------------------------------------------------
        conv = convolution_fft(a, a)                      # length 2n-1
        maxc = conv.max()
        # stable softmax weights
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # ---- gradient (cross‑correlation) -----------------------------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)             # length 3n‑2
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # ---- Adam update ---------------------------------------------
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad ** 2)

        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_to_simplex(a_new)

        a = a_new

        # ---- occasional exact evaluation -------------------------------
        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective (with momentum)
# ----------------------------------------------------------------------


def subgrad_opt_momentum(init_a: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 300_000,
                         step0: float = 0.07,
                         mu: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent (with momentum) on the exact objective
    `2·n·max(conv(a,a))`.  Returns the best vector seen and its score.
    """
    a = project_to_simplex(init_a.astype(float))
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    vel = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # step size diminishing like 1/√t
        step = step0 / np.sqrt(t)

        # momentum update
        vel = mu * vel + step * g
        a_new = a - vel
        a_new = project_to_simplex(a_new)

        a = a_new

        if t % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb polishing (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq: list[float],
               time_deadline: float,
               max_steps: int = 400_000) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb that swaps entries or moves a fraction of mass
    between two positions, always re‑projecting onto the simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10   # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative sequence minimising `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback on any previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Multi‑scale smoothed optimisation
    # ------------------------------------------------------------------
    # Allocate ~60 % of the total budget to the smooth phase
    smooth_deadline = start_time + 0.60 * budget_s

    scale_lengths = [512, 1024, 2048, 4096]

    # per‑scale configuration
    scale_cfg = {
        512:  {"restarts": 12, "max_iters": 150_000, "beta_end": 80.0},
        1024: {"restarts": 9,  "max_iters": 180_000, "beta_end": 100.0},
        2048: {"restarts": 6,  "max_iters": 220_000, "beta_end": 130.0},
        4096: {"restarts": 4,  "max_iters": 300_000, "beta_end": 180.0},
    }

    cur_best = None

    for L in scale_lengths:
        cfg = scale_cfg[L]
        restarts = cfg["restarts"]
        max_iters = cfg["max_iters"]
        beta_end = cfg["beta_end"]
        print(f"\n=== Scale {L} (restarts={restarts}, max_iters={max_iters}) ===")

        # compute a deadline for this scale – share the remaining smooth time
        remaining_smooth = smooth_deadline - time.time()
        if remaining_smooth <= 5.0:
            print("insufficient time for further smoothing – exiting scales")
            break
        # give each remaining scale an equal share of the remaining smooth budget
        scales_left = len(scale_lengths) - scale_lengths.index(L)
        scale_deadline = time.time() + remaining_smooth / scales_left

        best_local_a = None
        best_local_score = float("inf")

        for r in range(restarts):
            if time.time() >= scale_deadline - 3.0:
                break   # out of time for this scale

            if cur_best is not None:
                # start from up‑sampled version of the previous best + jitter
                start = upsample(cur_best, L)
                jitter = np.random.randn(L) * 0.02
                start = start + jitter
                start = np.maximum(start, 0.0)
            else:
                # completely random sparse start
                p = np.random.uniform(0.05, 0.55)
                k = max(2, int(round(p * L)))
                idx = np.random.choice(L, size=k, replace=False)
                vals = np.random.rand(k) + 0.1   # avoid exact zeros on support
                start = np.zeros(L, dtype=float)
                start[idx] = vals
                start = start / start.sum()

            # run the Adam‑based smoothed optimiser
            a_opt, score = smooth_opt_adam(
                start,
                time_deadline=scale_deadline - 1.0,
                max_iters=max_iters,
                beta_start=5.0,
                beta_end=beta_end,
                lr0=0.22,
            )

            if score < best_local_score:
                best_local_score = score
                best_local_a = a_opt.copy()
                print(f"   Restart {r + 1}/{restarts}: new local best score = {score:.6f}")

        # keep the best for the next scale
        if best_local_a is not None:
            cur_best = best_local_a
            if best_local_score < best_val:
                best_val = best_local_score
                best_seq = best_local_a.tolist()
                print(f"*** Best overall after scale {L}: score = {best_val:.6f}")

        # early exit if we are already out of total time
        if time.time() >= deadline - 30.0:
            break

    # ------------------------------------------------------------------
    # 2.  Sub‑gradient refinement (momentum)
    # ------------------------------------------------------------------
    subgrad_deadline = start_time + 0.85 * budget_s
    if cur_best is not None and time.time() < subgrad_deadline - 5.0:
        print("\n--- Sub‑gradient refinement with momentum ---")
        refined_a, refined_score = subgrad_opt_momentum(
            np.array(cur_best, dtype=float),
            time_deadline=subgrad_deadline - 2.0,
            max_iter=250_000,
            step0=0.08,
            mu=0.9,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            cur_best = refined_a
            print(f"    Sub‑grad improvement: score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print("\n--- Hill‑climb polishing ---")
        polished_seq, polished_score = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=600_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"    Hill‑climb improvement: final score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Safeguard fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        # a modest uniform vector as absolute last resort
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # final official evaluation (uses the provided `evaluate_sequence`)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
