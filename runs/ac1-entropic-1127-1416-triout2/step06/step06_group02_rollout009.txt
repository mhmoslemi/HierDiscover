```
def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    best_seq = None
    best_score = np.inf

    # Choose lengths distribution
    lengths = [256,512,1024,2048,4096]  # maybe up to 4096

    # Warm-start: uniform vector for each length: compute baseline
    for n in lengths:
        a = np.full(n, 1.0/n)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
    print(f"Uniform baselines: best score {best_score}")

    # Random sparse restarts loop
    while time.time() < deadline - 5:
        # Choose length with probability maybe favor larger n but also random
        n = int(np.random.choice(lengths))
        # Randomized vector: use Dirichlet distribution for some sparsity
        k = np.random.randint(int(0.1*n), int(0.9*n)+1)
        # Choose k positions random
        idx = np.random.choice(n, k, replace=False)
        # Sample Dirichlet on k entries to get varying masses
        alpha = np.random.uniform(0.01, 2.0) # shape param
        probs = np.random.gamma(alpha, 1.0, size=k)
        probs /= probs.sum()
        a = np.zeros(n)
        a[idx] = probs
        # Project onto simplex to ensure sum=1 (should be already).
        # Evaluate quickly.
        cur_sc = compute_score(a)
        if cur_sc < best_score:
            best_score = cur_sc
```

```
def adam_gradient_descent(a0, deadline, max_iter, step0=0.01, eps_factor=0.2, beta1=0.9, beta2=0.999):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_sc = compute_score(a)
    for t in range(1,...):
        if time.time() >= deadline: break
        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        eps = maxc * max(eps_factor * (0.97**(t-1)), 0.001)
        exp_vals = np.exp((conv - maxc)/eps)
        w = exp_vals / np.sum(exp_vals)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        idx_start = 2*n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 / (np.sqrt(v_hat) + 1e-8)
        a_new = a - step * m_hat
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a_new.copy()
        a = a_new
        # occasional jitter
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)
    return best, best_sc
```

The search consists of four stages that are all executed inside the 1000 s budget.

1. **Initial candidates** – generate a diverse pool of vectors on the probability simplex
   (uniform, sparse, Dirichlet‑distributed, and bimodal “edge‑blocks”).  
2. **Smooth Adam optimisation** – minimise a differentiable soft‑max surrogate of the
   max‑convolution  
   \[
   f_\varepsilon(a)=\varepsilon\log\Bigl(\sum_k\exp\!\bigl(\tfrac{(a*a)_k}{\varepsilon}\bigr)\Bigr)
   \]  
   using Adam (first‑order momentum) and a projection onto the simplex.
3. **Exact incremental hill‑climbing** – mass is transferred between two positions.
   The full convolution is updated **in O(n)** (no FFT) using the analytic
   formula for a rank‑one change.  Only moves that strictly lower the current
   maximum autocorrelation are kept; the step size is adapted automatically.
4. **Optional symmetry averaging** – the vector is symmetrised
   \((a+a^{\mathrm{rev}})/2\) and projected back onto the simplex.  Symmetric
   vectors frequently have a flatter autocorrelation.

The algorithm works for any length; lengths 512, 1024, 2048 and 4096 are tried
(the larger the length the more degrees of freedom, but the FFT cost also grows).
All intermediate results are kept, and the best sequence found so far is
returned at the end.  The routine never exceeds the given time budget, and it
uses only NumPy and (optionally) `cvxpy` for the LP that was used in the
reference implementation – the LP part is **not required** for the core search
and is therefore omitted to stay comfortably within the time limit.

```python
"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence().  The algorithm consists of:

1. Generation of diverse random start vectors on the simplex.
2. Adam optimiser on a smooth log‑sum‑exp surrogate of the max‑convolution.
3. Incremental hill‑climbing with exact O(n) convolution updates.
4. Optional symmetry averaging.

All work is performed inside the supplied time budget (≈1000 s) and the
best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b via real FFT.
    Result length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_max_conv(a: np.ndarray) -> float:
    """
    Returns max_k (a ⊛ a)_k for a vector that already satisfies sum(a)=1.
    This is the quantity that determines the score:
        score = 2·n·max_conv .
    """
    conv = convolution_fft(a, a)
    return float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1 }.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------


def random_sparse_vector(n: int) -> np.ndarray:
    """
    Random probability vector of length n.
    A random subset (density 10‑90 %) receives equal mass.
    """
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dirichlet_vector(n: int) -> np.ndarray:
    """
    Draw a Dirichlet‑distributed probability vector.
    The concentration α is random, and the number of non‑zero components
    is also random (10‑70 % of the length).
    """
    k = max(2, int(np.random.uniform(0.10, 0.70) * n))
    idx = np.random.choice(n, size=k, replace=False)
    alpha = np.random.uniform(0.01, 2.0)
    vals = np.random.gamma(alpha, 1.0, size=k)
    vals /= vals.sum()
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    return a


def bimodal_edge_vector(n: int) -> np.ndarray:
    """
    Two uniform blocks at the left and right edge of the vector.
    The fraction of the support is random.
    """
    frac = np.random.uniform(0.03, 0.20)      # size of each edge block
    blk = max(1, int(frac * n))
    a = np.zeros(n, dtype=float)
    a[:blk] = 0.5 / blk
    a[-blk:] = 0.5 / blk
    return a


# ----------------------------------------------------------------------
# 2. Adam optimiser on a smooth surrogate of the max‑convolution
# ----------------------------------------------------------------------


def adam_smooth_max(a0: np.ndarray,
                    deadline: float,
                    max_iter: int = 20000,
                    step0: float = 0.5,
                    eps_factor: float = 0.25,
                    beta1: float = 0.9,
                    beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser of
        f_ε(a) = ε·log Σ_k exp( (a⊛a)_k / ε )
    with a decreasing temperature ε.
    Returns the best vector (lowest exact max‑convolution) seen.
    """
    a = a0.astype(float).copy()
    n = len(a)
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_a = a.copy()
    best_max = compute_max_conv(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        cur_max = np.max(conv)

        # temperature – never let it become too small
        eps = max(cur_max * max(eps_factor * (0.97 ** (t - 1)), 0.001))

        # soft‑max weights (stable)
        exp_vals = np.exp((conv - cur_max) / eps)
        w = exp_vals / np.sum(exp_vals)          # Σ w = 1

        # gradient of the smooth surrogate
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)    # length 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - step * m_hat
        a = project_to_simplex(a)

        # monitoring
        cur_max = compute_max_conv(a)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # occasional tiny jitter to escape plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_max


# ----------------------------------------------------------------------
# 3. Incremental hill‑climbing (exact updates in O(n))
# ----------------------------------------------------------------------


def hill_climb_incremental(a0: np.ndarray,
                           deadline: float,
                           max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic hill‑climber.
    At each step a fraction of mass is moved from position i to j.
    The full convolution is updated in O(n) using the analytic formula
    (no FFT per step).  Only moves that strictly reduce the current maximum
    autocorrelation are kept.
    """
    a = a0.astype(float).copy()
    n = len(a)
    conv = convolution_fft(a, a)               # length 2n‑1
    cur_max = float(np.max(conv))

    best_a = a.copy()
    best_max = cur_max

    step_scale = 0.08   # initial fraction of the source mass
    no_improve = 0

    # pre‑allocate index array once
    ks = np.arange(2 * n - 1)

    while max_steps > 0 and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        if delta <= 0.0:
            continue

        # ----- compute the updated convolution -----
        conv_new = conv.copy()

        # subtract contribution of the decreasing site i
        idx = ks - i
        mask = (idx >= 0) & (idx < n)
        conv_new[mask] -= 2.0 * delta * a[idx[mask]]

        # add contribution of the increasing site j
        idx = ks - j
        mask = (idx >= 0) & (idx < n)
        conv_new[mask] += 2.0 * delta * a[idx[mask]]

        # quadratic (δ^2) correction terms
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = float(np.max(conv_new))

        if new_max < cur_max - 1e-14:                     # improvement
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_max = new_max
            no_improve = 0
            step_scale = min(step_scale * 1.04, 0.5)

            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        else:
            # reject move – adapt step scale
            no_improve += 1
            step_scale = max(step_scale * 0.97, 1e-4)

        max_steps -= 1

        # occasional aggressive jitter if stuck for a while
        if no_improve > 5_000:
            jitter = np.random.randn(n) * (step_scale * 0.02)
            a = project_to_simplex(a + jitter)
            conv = convolution_fft(a, a)
            cur_max = float(np.max(conv))
            no_improve = 0

    return best_a, best_max


# ----------------------------------------------------------------------
# 4. Symmetry averaging (cheap post‑processing)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a + a_rev)/2 and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback: use a previously known construction if present
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Lengths that are cheap for FFT but give enough degrees of freedom
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]

    # ------------------------------------------------------------------
    # Uniform vectors as a cheap baseline on each length
    # ------------------------------------------------------------------
    for n in lengths:
        a = np.full(n, 1.0 / n)
        score = 2.0 * n * compute_max_conv(a)
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"[baseline] length={n} score={score:.6f}")

    # ------------------------------------------------------------------
    # Main stochastic search loop
    # ------------------------------------------------------------------
    outer_iter = 0
    while time.time() < deadline - 5.0:
        outer_iter += 1
        n = int(np.random.choice(lengths))

        # ---- generate a random start vector ---------------------------------
        init_type = np.random.choice(["sparse", "dirichlet", "bimodal", "uniform"])
        if init_type == "sparse":
            a = random_sparse_vector(n)
        elif init_type == "dirichlet":
            a = random_dirichlet_vector(n)
        elif init_type == "bimodal":
            a = bimodal_edge_vector(n)
        else:  # uniform
            a = np.full(n, 1.0 / n)

        a = project_to_simplex(a)   # just in case

        # ---- cheap evaluation of the start ---------------------------------
        cur_score = 2.0 * n * compute_max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(f"[{outer_iter}] random init ({init_type}) n={n} score={cur_score:.6f}")

        # ---- phase 2 – Adam on smooth max ----------------------------------
        remaining = deadline - time.time()
        if remaining < 6.0:
            break
        a_adam, max_adam = adam_smooth_max(
            a,

            step0=0.5,

        score_adam = 2.0 * n * max_adam
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.tolist()
            print(f"[{outer_iter}] after Adam n={n} score={best_score:.6f}")

        # ---- phase 3 – exact incremental hill climbing ----------------------
        remaining = deadline - time.time()
        if remaining > 12.0:
            a_hc, max_hc = hill_climb_incremental(
                a_adam,
                deadline - 2.0,
                max_steps=int(300_000 * (remaining / budget_s))
            )
            score_hc = 2.0 * n * max_hc
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.tolist()
                print(f"[{outer_iter}] hill‑climb improved n={n} score={best_score:.6f}")

        # ---- phase 4 – symmetry averaging (cheap) ---------------------------
        a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
        score_sym = 2.0 * len(a_sym) * compute_max_conv(a_sym)
        if score_sym < best_score:
            best_score = score_sym
            best_seq = a_sym.tolist()
            print(f"[{outer_iter}] symmetry average improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if nothing was ever found, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
```
