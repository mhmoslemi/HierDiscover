def subgradient_descent(a, deadline, step0=0.6):
    n = len(a)
    best_a = a.copy()
    best_score = compute_score_fast(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = conv.max()
        # find one index; optionally handle ties average
        idx = int(conv.argmax())
        # compute gradient
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # step size schedule
        step = step0 / np.sqrt(it)
        # backtracking line search
        # attempt step, project, compute new max; ensure improvement else halve
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_max = convolution_fft(a_new, a_new).max()
            if new_max < max_val - 1e-15:
                break
            step *= 0.5
        # compute score
        cur_score = 2 * n * new_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
        # occasional progress output
        if it % 50000 == 0:
            print(f"    Subgrad iter={it:,d} best_score={best_score:.6f}")
    return best_a, best_score

def random_walk(a, deadline, init_step=0.01):
    n = len(a)
    best_a = a.copy()
    best_score = compute_score_fast(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)
        cur_score = compute_score_fast(a_prop)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_prop.copy()
            a = a_prop
        # else maybe accept with low prob? optional
        if it % 50000 == 0:
            print(f"    Random walk iter={it:,d} best={best_score:.6f}")
    return best_a, best_score

def convolution_fft(a,b):
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv
def max_convolution(a):
    return float(convolution_fft(a,a).max())
def compute_score_fast(a):

"""
Simple yet powerful search for a step sequence that minimises
    2·n·max(a⋆a) / (∑a)².

The algorithm works in three phases (all respect the global 1000 s budget):

1️⃣  Initialise a promising shape – the classical arcsine β(½,½) density,
    optionally a random β‑family shape – and normalise it to Σa = 1.

2️⃣  Long projected sub‑gradient descent on the *true* ℓ∞‑objective.
    The sub‑gradient of max(a⋆a) at the active shift is known analytically,
    so each iteration needs only one FFT‑based convolution and a cheap line‑search.
    The step size follows  ηₜ = η₀ / √t with a back‑tracking safeguard.

3️⃣  A cheap stochastic Gaussian walk on the simplex (simulated‑annealing style)
    to polish the final vector.

At the end the candidate is handed to the official ``evaluate_sequence``.
The code never returns an empty list – a uniform 256‑vector is used as a
fallback.  No external files or network access are required.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based convolution)
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(_conv_fft(a, a).max())


def _fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score for a vector normalised
    to Σa = 1:  2·n·max(a⋆a).
    """
    n = a.shape[0]
    return 2.0 * n * _max_convolution(a)


# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1} (Duchi et al.)."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Standard shapes (arcsine, beta‑family)
# ------------------------------------------------------------

def arcsine_initial(n: int) -> np.ndarray:
    """
    Discrete arcsine (β(½,½)) density on {0,…,n‑1},
    normalised to Σ = 1.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25                     # small regulariser against div‑by‑0
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Unnormalised symmetric beta‑family density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ------------------------------------------------------------

def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    step0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact L∞‑objective.
    Returns the best vector seen and its fast score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _fast_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # 1) current convolution and the active shift
        conv = _conv_fft(a, a)                     # length 2n‑1
        max_val = float(conv.max())
        idx = int(conv.argmax())

        # 2) sub‑gradient of the active entry (see analysis)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # 3) line‑search (back‑tracking) to guarantee descent
        step = step0 / np.sqrt(it)
        for _ in range(12):          # at most 12 halvings
            a_new = a - step * grad
            a_new = _project_to_simplex(a_new)
            new_max = _max_convolution(a_new)
            if new_max < max_val - 1e-15:
                break
            step *= 0.5

        cur_score = 2.0 * n * new_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if it % 50000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} best_score={best_score:.6f}"
            )

    return best_a, best_score


# ------------------------------------------------------------
# Cheap stochastic walk on the simplex (simulated‑annealing flavour)
# ------------------------------------------------------------

def _random_walk(
    a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
) -> tuple[np.ndarray, float]:
    """Gaussian perturbations with a simple acceptance rule."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = _fast_score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = _project_to_simplex(a_prop)

        cur_score = _fast_score(a_prop)

        # accept only improvements (tiny uphill probability omitted for speed)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_prop.copy()
            a = a_prop

        if it % 50000 == 0:
            print(
                f"    Random walk iter={it:,d} best_score={best_score:.6f}"
            )
    return best_a, best_score


# ------------------------------------------------------------
# Main entry point required by the competition harness
# ------------------------------------------------------------

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine uses an arcsine start, long projected sub‑gradient descent,
    and a final stochastic walk.  All phases respect the global time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------
    # 0.  Try the previously known construction (if available)
    # --------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------
    # 1.  Main optimisation loop over a few lengths
    # --------------------------------------------------------
    for n in (2048, 4096):                # larger n gives a finer discretisation
        if time.time() >= deadline - 5.0:
            break

        print(f"\n=== Optimising length n = {n} ===")

        # ---- 1a. initialise with a promising shape -----------------
        # we keep two starters: arcsine and a random β‑shape
        starters = []

        arcs = arcsine_initial(n)
        arcs /= arcs.sum()
        starters.append(arcs)

        # random β‑shape (as a second seed)
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        eps = rng.uniform(1e-6, 0.01 * n)
        random_beta = generate_beta_shape(n, alpha, beta, eps)
        random_beta /= random_beta.sum()
        starters.append(random_beta)

        # keep the best starter (fast surrogate score)
        local_best = None
        local_best_sc = float("inf")
        for st in starters:
            sc = _fast_score(st)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = st.copy()

        # ---- 1b. allocate time: most of the budget to sub‑gradient ----
        remaining = deadline - time.time()
        if remaining < 10.0:
            break

        # 70 % of the remaining budget for sub‑gradient, the rest for polishing
        sub_deadline = time.time() + remaining * 0.70
        print(f"  Sub‑gradient descent (target iter≈{int((sub_deadline-time.time())*1500):,d}) …")
        a_opt, sub_sc = _subgradient_descent(local_best, sub_deadline, step0=0.6)
        print(f"  Sub‑gradient best score (fast) = {sub_sc:.6f}")

        # ---- 1c. final stochastic walk (use whatever time is left) ----
        if time.time() < deadline - 1.0:
            walk_deadline = deadline
            print(f"  Random walk polishing (remaining ≈{int(deadline-time.time())} s) …")
            a_opt, walk_sc = _random_walk(a_opt, walk_deadline, init_step=0.015)
            print(f"  Random‑walk best score (fast) = {walk_sc:.6f}")

        # ---- 1d. official evaluation ---------------------------------
        seq = a_opt.tolist()
        val = evaluate_sequence(seq)
        print(f"  → official score for length {n} = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"  *** New global best (len={n}) = {best_val:.6f} ***")

        # early exit if we have already beaten the target
        if best_val < 1.503:
            print("Target reached – exiting early.")
            break

    # --------------------------------------------------------
    # 2.  Guarantee a non‑empty return
    # --------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
