def conv_constraint_matrix(f):
    n = len(f)
    L = 2*n - 1
    A = np.zeros((L, n), dtype=float)
    for i in range(n):
        start = i
        end = i + n
        A[start:end, i] = f  # length n
    return A

def local_search(a, iterations, step_size, deadline):
    # a: np array sum=1
    best = a.copy()
    best_score = compute_score(best)
    n = len(a)
    for it in range(iterations):
        if time.time() >= deadline: break
        # pick two indices i, j randomly; transfer mass delta = step_size * best[i]
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if i == j: continue
        delta = step_size * best[i] * rng.random()  # or random factor
        if delta == 0: continue
        new = best.copy()
        new[i] -= delta
        new[j] += delta
        if new[i] < 0: continue
        # renorm not needed as sum unchanged.
        new_score = compute_score(new)
        if new_score < best_score:
            best = new
            best_score = new_score
    return best

def lp_refine(f, deadline, max_iters=20, t=0.5):
    n = len(f)
    sum_f = f.sum()
    # ensure sum = 1 for simplicity
    f = f / sum_f
    for it in range(max_iters):
        if time.time() >= deadline: break
        conv = convolution_fft(f, f)
        M = np.max(conv)
        # build constraint matrix
        A = np.zeros((2*n-1, n), dtype=float)
        for i in range(n):
            A[i:i+n, i] = f
        # solve LP
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)] * n, method='highs')
        if not res.success:
            print("LP failed iteration", it, res.message)
            break
        b = res.x
        sum_b = b.sum()
        # if not improvement, break
        if sum_b <= 1 + 1e-12:
            break
        # scale to sum=1
        g = b / sum_b
        # convex combination
        f = (1 - t) * f + t * g
        # ensure sum 1
        f = f / f.sum()
    return f
```

A three‑stage optimiser that builds on the previous “smooth‑max Adam”
approach but adds a cheap linear‑programming refinement (a classic
alternating‑LP scheme) on a modest problem size and an additional
random local‑search polish.  The pipeline is:

0.  optional fallback from a previous run.
1.  Diverse seeding (several parametric families, many lengths).
2.  Smooth‑max Adam optimisation of the log‑sum‑exp surrogate.
3.  LP refinement on the *small* problem (n≈256) that provably never
    increases the maximal autocorrelation.
4.  Upsample the refined small‑n shape to the target length and run a
    long Euclidean sub‑gradient descent followed by exponentiated‑gradient
    polishing.
5.  A cheap stochastic neighbourhood search (pairwise mass transfers)
    for a final tweak.

All phases respect the global 1000 s deadline and keep the best
sequence (according to the official ``evaluate_sequence``) for the
final return.
"""

import time
import numpy as np
from typing import List, Tuple
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate – assumes Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type density (i+ε)^{α‑1}(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_gaussian_shape(n: int, mu: float, sigma: float) -> np.ndarray:
    """Gaussian density on the discrete grid, truncated to non‑negative."""
    x = np.linspace(0.0, 1.0, n)
    vals = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return vals / vals.sum()


def generate_exponential_shape(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-lam * i)
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Adam optimiser of the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 30_000,
                             tau_start: float = 0.5,
                             tau_end: float = 50.0,
                             base_lr: float = 0.3) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    ``a_init`` must already be normalised to Σa = 1.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # logits → softmax gives a simplex point
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # log‑sum‑exp weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a (2·∑_k w_k·shift(a,k))
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional monitoring (fast surrogate)
        if it % 5000 == 0:
            score = compute_score(a)
            print(f"    [Adam] it={it:,d} τ={tau:.2f} surrogate={score:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (Euclidean projection)
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient: 2·∑ mask·shift(a)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.3) -> np.ndarray:
    """Exponentiated‑gradient polishing (mirror‑descent with KL geometry)."""
    a = a.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (alternating LP scheme)
# ----------------------------------------------------------------------


def lp_refine(f: np.ndarray,
              deadline: float,
              max_iters: int = 15,
              mix: float = 0.5) -> np.ndarray:
    """
    Alternating LP improvement on a small problem.
    ``f`` is assumed to be on the simplex (Σf = 1, f ≥ 0).
    The routine never increases max(conv(f,f)).
    """
    n = f.shape[0]
    L = 2 * n - 1

    # Pre‑allocate the constraint matrix – same for every iteration
    A = np.zeros((L, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f  # each column = shifted copy of f

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(f, f)
        M = float(np.max(conv))

        # Solve  max Σb  s.t. A·b ≤ M·1,  b ≥ 0
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(L, M),
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            print(f"    [LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # no improvement possible
            break

        # scale to keep the simplex
        g = b / sum_b

        # convex combination
        f = (1.0 - mix) * f + mix * g
        f = f / f.sum()   # numerical safety

        # update A for the next iteration – columns are shifted copies of the new f
        for i in range(n):
            A[i:i + n, i] = f

        # diagnostics
        new_score = compute_score(f)
        print(f"    [LP] it={it} → surrogate={new_score:.6f}")

    return f


# ----------------------------------------------------------------------
# Random local‑search neighbourhood optimisation
# ----------------------------------------------------------------------


def local_search(a: np.ndarray,
                 rng: np.random.Generator,
                 deadline: float,
                 steps: int = 25_000,
                 max_step_frac: float = 0.25) -> np.ndarray:
    """
    Very cheap stochastic hill‑climbing: repeatedly pick two indices
    (i,j) and transfer a random fraction of a[i] to a[j].
    Only moves that improve the fast surrogate are kept.
    """
    best = a.copy()
    best_score = compute_score(best)
    n = len(a)

    for it in range(steps):
        if time.time() >= deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a random amount of mass from i to j
        delta = best[i] * rng.random() * max_step_frac
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep non‑negativity automatically (delta ≤ a[i])
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

    print(f"    [LocalSearch] finished with surrogate={best_score:.6f}")
    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official
    ``evaluate_sequence`` value.  The routine mixes random seeding,
    smooth‑max Adam, a cheap LP refinement on a small problem, a
    final Euclidean sub‑gradient + EG polish, and a stochastic local
    search at the end.  All phases respect the 1000 s global deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Possible fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"[Fallback] official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Diverse seeding (fast surrogate)
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024]          # smaller sizes for cheap LP, larger for final polish
    pool: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Seeding length {n} ---")

        # deterministic starters
        uniform = np.full(n, 1.0 / n)
        pool.append((compute_score(uniform), uniform.copy()))
        arc = arcsine_initial(n)
        pool.append((compute_score(arc), arc.copy()))

        # random beta family
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            pool.append((compute_score(cand), cand))

        # random power‑shape
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((compute_score(cand), cand))

        # mixture of arcsine & uniform
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((compute_score(cand), cand))

        # triangular bumps
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

        # Gaussian shapes
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            mu = rng.random()
            sigma = rng.uniform(0.05, 0.3)
            cand = generate_gaussian_shape(n, mu, sigma)
            pool.append((compute_score(cand), cand))

        # Exponential decay
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            lam = rng.uniform(0.001, 0.02)
            cand = generate_exponential_shape(n, lam)
            pool.append((compute_score(cand), cand))

    # keep a few best seeds (by surrogate) for the heavy stages
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:12]]
    print(f"\nSelected {len(top_seeds)} top seeds (surrogate range {pool[0][0]:.6f} – {pool[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2.  Smooth‑max Adam optimisation on each seed
    # ------------------------------------------------------------------
    refined_candidates: List[np.ndarray] = []
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 4.0:
            break
        n = seed_vec.shape[0]
        print(f"\n=== Adam refinement of seed {idx}/{len(top_seeds)} (n={n}) ===")
        remaining = deadline - time.time()
        # allocate a reasonable fraction of the remaining budget
        max_iter = int(min(35_000, remaining * 30))
        a_adam = adam_optimize_smooth_max(seed_vec,
                                          deadline,
                                          max_iters=max_iter,
                                          tau_start=0.5,
                                          tau_end=50.0,
                                          base_lr=0.3)

        surrogate = compute_score(a_adam)
        print(f"    Post‑Adam surrogate = {surrogate:.6f}")
        refined_candidates.append(a_adam)

    # ------------------------------------------------------------------
    # 3.  Choose the best surrogate and perform LP refinement (on small n)
    # ------------------------------------------------------------------
    # pick the best among the Adam‑refined vectors (by fast surrogate)
    best_idx = int(np.argmin([compute_score(v) for v in refined_candidates]))
    candidate = refined_candidates[best_idx]
    n_candidate = candidate.shape[0]
    print(f"\n--- LP refinement on candidate (n={n_candidate}) ---")
    # we only run LP on a *small* problem; if the candidate is large,
    # we down‑sample it to n=256, run LP, then up‑sample back.
    target_lp_n = 256
    if n_candidate != target_lp_n:
        # simple linear interpolation to 256 points
        x_src = np.linspace(0.0, 1.0, n_candidate)
        x_dst = np.linspace(0.0, 1.0, target_lp_n)
        candidate_lp = np.interp(x_dst, x_src, candidate)
        candidate_lp = candidate_lp / candidate_lp.sum()
    else:
        candidate_lp = candidate.copy()

    # LP refinement (takes a few seconds for n=256)
    candidate_lp = lp_refine(candidate_lp, deadline, max_iters=12, mix=0.5)

    # Upsample back to the original size (or to the largest size we used)
    up_n = max(lengths)  # i.e. 1024 in our configuration
    if candidate_lp.shape[0] != up_n:
        x_src = np.linspace(0.0, 1.0, candidate_lp.shape[0])
        x_dst = np.linspace(0.0, 1.0, up_n)
        candidate_up = np.interp(x_dst, x_src, candidate_lp)
        candidate_up = candidate_up / candidate_up.sum()
    else:
        candidate_up = candidate_lp.copy()

    # ------------------------------------------------------------------
    # 4.  Exact refinement (sub‑gradient + EG) on the up‑sampled shape
    # ------------------------------------------------------------------
    n = candidate_up.shape[0]
    print(f"\n--- Exact refinement on length {n} ---")
    remaining = deadline - time.time()
    max_iter_sub = int(min(200_000, remaining * 30))
    refined = subgradient_descent_exact(candidate_up,
                                        deadline,
                                        max_iter=max_iter_sub,
                                        step0=0.12)

    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        max_iter_eg = int(min(120_000, remaining * 30))
        refined = eg_refine(refined, deadline,
                            max_iter=max_iter_eg,
                            eta0=0.25)

    # ------------------------------------------------------------------
    # 5.  Final stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("\n--- Final stochastic neighbourhood search ---")
        refined = local_search(refined, rng, deadline, steps=30_000, max_step_frac=0.15)

    # ------------------------------------------------------------------
    # 6.  Official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(refined.tolist())
        print(f"Official score after full pipeline = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            print("*** New global best! ***")
    except Exception as e:
        print(f"Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 7.  Global polishing of the best sequence (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Global polishing of the best sequence ---")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(250_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.09)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(150_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.20)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 8.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
