def smooth_adam(a0, time_deadline, max_iter=2000, alpha=0.5, beta1=0.9, beta2=0.999, eps_factor=0.2):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        eps = maxc * max(eps_factor * (0.95 ** (t-1)), 0.01)  # maybe gradually shrink.
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        idx_start = 2 * len(a) - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(len(a))]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step
        # enforce symmetry optionally
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        # occasional jitter
        if t % 500 == 0:
            jitter = np.random.randn(len(a)) * (alpha * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)
    return best_a, best_sc

def advanced_hill_climb(a0, time_deadline, max_steps=200000):
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    # scaling factor for delta based on 0.05 * sqrt(best_sc?) but just constant first
    delta_frac = 0.1
    while time.time() < time_deadline and max_steps>0:
        # Choose two indices i,j
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        # Choose move: swap or transfer
        if np.random.rand() < 0.4:
            cand = best_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer random fraction up to delta_frac of mass from i to j
            max_transfer = delta_frac * best_a[i]
            if max_transfer <= 0:
                continue
            delta = np.random.rand() * max_transfer
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
        cand = project_to_simplex(cand)
        # optional symmetry averaging after candidate generation?
        # evaluate
        cur_sc = compute_score(cand)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            delta_frac = min(delta_frac * 1.1, 0.5)
        else:
            delta_frac = max(delta_frac * 0.99, 0.01)
        max_steps -= 1
    return best_a, best_sc
```

def convolution_fft(a, b):
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A*B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

"""
Propose a step‑function (non‑negative vector) that minimizes
`evaluate_sequence`.  The algorithm combines several complementary
strategies within the 1000 s budget:

1. **Diverse random starts** – sparse, exponential and Gaussian‑shaped
   vectors, symmetrised and projected onto the probability simplex.
2. **Smooth Adam descent** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution using Adam with an
   exponential ε‑schedule; after each update the vector is
   symmetrised and re‑projected.
3. **Exact sub‑gradient refinement** – classic sub‑gradient descent on
   the true objective 2·n·max(a⊛a) with a √t step‐size.
4. **Stochastic hill‑climbing** – swaps or mass‑transfer moves that
   are accepted only if they improve the exact score.
5. **Final symmetry averaging** – (a + reverse(a))/2 followed by a
   projection onto the simplex.

All intermediate vectors live on the simplex (sum = 1), which makes
`compute_score` exact for the evaluation function.
The search stops before the deadline and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact replica of `evaluate_sequence` for vectors whose entries sum to 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees v_i ≥ 0 and sum(v) = 1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisers (sparse, exponential, Gaussian)
# ----------------------------------------------------------------------


def random_sparse_vector(n: int) -> np.ndarray:
    """Uniform mass on a random subset of coordinates."""
    density = np.random.uniform(0.10, 0.80)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    a = project_to_simplex(a)
    a = (a + a[::-1]) * 0.5      # enforce symmetry early
    return a


def random_exponential_vector(n: int) -> np.ndarray:
    """Decaying exponential shape, randomly shifted."""
    lam = np.random.uniform(0.001, 5.0)
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * idx)
    shift = np.random.randint(n)
    a = np.roll(a, shift)
    a = project_to_simplex(a)
    a = (a + a[::-1]) * 0.5
    return a


def random_gaussian_vector(n: int) -> np.ndarray:
    """Truncated Gaussian centred in the middle."""
    sigma = np.random.uniform(0.1, 0.4) * n
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a = project_to_simplex(a)
    a = (a + a[::-1]) * 0.5
    return a


# ----------------------------------------------------------------------
# 1. Smooth ADAM descent on log‑sum‑exp surrogate of max‑convolution
# ----------------------------------------------------------------------


def smooth_adam(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 3000,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_factor: float = 0.4,
    min_eps_frac: float = 0.005,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        f_ε(a) = ε·log Σ_k exp(conv_k / ε)
    using Adam.  After every update the vector is symmetrised and
    projected onto the simplex.
    Returns the best vector (lowest exact score) seen.
    """
    a = a0.astype(float).copy()
    n = len(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    # Since sum(a)=1, exact score = 2·n·max(conv)
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # convolution and max value
        conv = convolution_fft(a, a)
        maxc = float(np.max(conv))

        # ε schedule – start large, shrink exponentially, never below a
        # fixed fraction of the current max
        eps = maxc * max(eps_factor * (0.95 ** (t - 1)), min_eps_frac)

        # soft‑max weights
        exp_vals = np.exp((conv - maxc) / eps)          # stable
        w = exp_vals / np.sum(exp_vals)                # Σ w = 1

        # gradient of the smooth objective
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)           # length = 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step

        # enforce symmetry and feasibility
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        # exact score (no extra FFT needed)
        cur_sc = 2.0 * n * maxc
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (alpha * 0.001 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on 2·n·max(a⊛a).
    The sub‑gradient is taken at the index where the convolution attains its
    maximum.  After each step we project back onto the simplex and enforce
    symmetry.
    """
    a = init_a.astype(float).copy()
    n = len(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient g_i = 2 * a_{k-i} (where defined)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional small random shake
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Stochastic hill‑climbing (swap / mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline,
    max_steps: int = 200_000,
) -> tuple[list, float]:
    """
    Hill‑climbing that proposes swaps or transfers of a fraction of mass
    between two positions.  Only improving moves are kept.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    # initial transfer fraction
    frac = 0.12

    while max_steps > 0 and time.time() < time_deadline:
        # choose move type
        move_type = np.random.rand()
        cand = best_a.copy()

        i, j = np.random.randint(n, size=2)
        if i == j:
            max_steps -= 1
            continue

        if move_type < 0.45:
            # swap
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            delta = frac * best_a[i]
            if delta <= 0:
                max_steps -= 1
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5   # keep symmetry
        cand = project_to_simplex(cand)

        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            frac = min(frac * 1.07, 0.5)   # become bolder after success
        else:
            frac = max(frac * 0.95, 0.005)  # become more cautious

        max_steps -= 1

    return best_a.tolist(), best_sc


# ----------------------------------------------------------------------
# 4. Post‑processing: symmetry averaging
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Return (a + reverse(a)) / 2 projected onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

        if cur_sc < best_score:
