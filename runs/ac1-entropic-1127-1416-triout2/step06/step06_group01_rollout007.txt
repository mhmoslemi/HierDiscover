def lp_improve(a: np.ndarray, M: float, t: float) -> np.ndarray:

   n = len(a)

   b = cp.Variable(n, nonneg=True)  # default bounds

   constraints = [cp.conv(a, b) <= M]  # cp.conv returns length L

   prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

   prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=1000) # maybe set high iteration.

   b_val = b.value

   if b_val is None: fallback to a.

   sum_b = np.sum(b_val)

   if sum_b <= 0: return a

   g = b_val / sum_b

   new_a = (1 - t) * a + t * g

   return new_a

Implement function lp_step_linprog(a, M, t):

   n = len(a); L = 2*n-1

   # Build A_ub (shape L x n). We'll fill for each row.

   data = []
   rows = []
   cols = []

   for k in range(L):
       # j index range: max(0, k - (n-1)) <= i <= min(k, n-1)
       # we need coefficients a_i where i = k - j? Wait we want sum_i a_i * b_{k-i}. Equivalent to convolution: (a * b)_k = sum_i a_i b_{k-i}.

       # So for each i where 0<=i<n and 0<=k-i<n => i in [max(0, k-(n-1)), min(k, n-1)]

       i_start = max(0, k - (n - 1))
       i_end = min(k, n - 1)   # inclusive

       # For i = i_start ... i_end, coefficient on b_{k-i} is a_i.

       # So for each i, j = k - i.

       for i in range(i_start, i_end+1):
           j = k - i
           rows.append(k)
           cols.append(j)
           data.append(a[i])

   # Build A_ub:
   A_ub = csr_matrix((data, (rows, cols)), shape=(L, n))

   # b_ub = M * np.ones(L)
   b_ub = np.full(L, M)

   # Objective: maximize sum b => minimize -sum b
   c = -np.ones(n)

   # bounds (0, None) default.

   # Call linprog
   res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs', options={'presolve': True})

   if not res.success:
        return a

   b_val = res.x
   sum_b = b_val.sum()
   if sum_b <= 0:
        return a
   g = b_val / sum_b

   new_a = (1 - t) * a + t * g

   return new_a

def lp_iterate(a, max_iters, deadline, step_decay=0.5):
    n = a.shape[0]
    # cvxpy model
    a_param = cp.Parameter(n)
    M_param = cp.Parameter()
    b = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(a_param, b) <= M_param]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    curr_M = compute_max_conv(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a_param.value = a
        M_param.value = float(curr_M)
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        if b.value is None:
            break
        b_vec = np.asarray(b.value).flatten()
        sum_b = b_vec.sum()
        if sum_b <= 0:
            break
        g = b_vec / sum_b
        step = step_decay / (it + 1)  # decreasing step
        a_new = (1 - step) * a + step * g
        # ensure numeric stability: clip tiny negatives
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        new_M = compute_max_conv(a_new)
        # if no improvement, break
        if new_M >= curr_M * (1 - 1e-12):
            break
        a = a_new
        curr_M = new_M
        # optionally print progress
        if it % 2 == 0:
            print(f"    LP it={it}, step={step:.4f}, max_conv={new_M:.8f}")
    return a, curr_M

"""
Propose a step‑function sequence with a low official score.
================================================================
The algorithm has three distinct phases:

1️⃣  **Diverse seeding** – generate a handful of promising probability
    vectors (arcsine, uniform, beta / power families, mixed arcsine‑uniform,
    triangular bumps).  All vectors are normalised to sum‑to‑1, because the
    official score 2·n·max(conv) is invariant under scaling.

2️⃣  **LP‑based monotone improvement** – starting from each seed we solve
    a small linear program

        maximise   Σ b_j
        subject to  a * b ≤ M          (component‑wise)
                    b_j ≥ 0,

    where `a` is the current vector, `M = max(a*a)` and `*` denotes linear
    convolution.  The LP (implemented with CVXPY) finds a feasible vector
    `b` whose total mass is as large as possible without violating the
    convolution bound.  Mixing `a←(1‑τ)a+τ·(b/Σb)` with a decreasing step
    `τ` (τ = 0.5/(iteration+1)) strictly reduces `M`.  Repeating a few
    iterations yields a shape with a substantially smaller peak convolution.

3️⃣  **Exact polishing** – the best LP‑refined vector is finally polished
    with a sub‑gradient descent on the true non‑smooth objective followed
    by an exponentiated‑gradient (mirror‑descent) stage.  Both stages use
    the exact convolution (FFT‑based) and respect the simplex constraint.

All phases respect a global 1000 s wall‑time budget and keep the globally
best sequence (according to the official `evaluate_sequence`).  The code
contains many time‑checks and prints progress information.
"""

import time
from typing import List, Tuple

import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a*a (fast via FFT)."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate for the official score (Σa is assumed to be 1)."""
    n = a.shape[0]
    return 2.0 * n * compute_max_conv(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Exact sub‑gradient & exponentiated‑gradient polishers
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.05,
) -> np.ndarray:
    """Euclidean projected sub‑gradient descent on max(conv(a,a))."""
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # points where the max is attained (with tiny tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·conv(a, mask_rev) evaluated at the central positions
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    eta0: float = 0.3,
) -> np.ndarray:
    """Exponentiated‑gradient (mirror‑descent) polishing."""
    a = a.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Linear‑programming based monotone improvement
# ----------------------------------------------------------------------


def lp_improve(
    a_init: np.ndarray,
    max_iters: int,
    deadline: float,
    step_decay: float = 0.5,
) -> Tuple[np.ndarray, float]:
    """
    Run a few LP iterations that monotonically decrease max(conv(a,a)).
    Returns the improved vector and its final max‑convolution value.
    """
    n = a_init.shape[0]

    # CVXPY parametrisation – rebuilt once per length for efficiency
    a_param = cp.Parameter(n)
    M_param = cp.Parameter()
    b = cp.Variable(n, nonneg=True)
    # linear convolution constraint (full length = 2n‑1)
    constraints = [cp.conv(a_param, b) <= M_param]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    a = a_init.copy()
    cur_M = compute_max_conv(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # update parameters
        a_param.value = a
        M_param.value = float(cur_M)

        # solve the LP – ECOS is fast on this size
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)

        if b.value is None:
            # Solver failure – abort LP phase
            break

        b_vec = np.asarray(b.value).flatten()
        sum_b = b_vec.sum()
        if sum_b <= 0.0:
            break

        # feasible candidate (normalised)
        g = b_vec / sum_b

        # decreasing mixing weight
        step = step_decay / (it + 1)          # e.g. 0.5, 0.25, 0.166, …
        a_new = (1.0 - step) * a + step * g
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        new_M = compute_max_conv(a_new)

        # monotone improvement check (tiny tolerance to avoid endless loops)
        if new_M >= cur_M * (1.0 - 1e-12):
            # no progress – stop LP phase
            break

        a = a_new
        cur_M = new_M

        if it % 2 == 0:
            print(
                f"    LP it={it:3d} step={step:.4f} max_conv={cur_M:.8e}"
            )
    return a, cur_M

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The search combines diverse seeding, a small LP‑based monotone improvement
    and a final exact polishing (sub‑gradient + exponentiated‑gradient).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose a single length (larger lengths give lower theoretical bound)
    # ------------------------------------------------------------------
    lengths = [2048]          # can be extended to [1024, 2048] if time permits
    # ------------------------------------------------------------------
    # 2.  Process each length independently
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n={n} ===")
        # ----- build a modest pool of seeds ---------------------------------
        seeds: List[np.ndarray] = []

        # deterministic starters
        seeds.append(arcsine_initial(n))
        seeds.append(np.full(n, 1.0 / n))

        # mixtures of arcsine & uniform
        for _ in range(3):
            w = rng.random()
            seeds.append(generate_mixture_shape(n, w))

        # random beta‑family shapes
        for _ in range(3):
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            eps = rng.uniform(0.0, 0.02 * n)
            seeds.append(generate_beta_shape(n, a, b, eps))

        # random power‑shape families
        for _ in range(3):
            p = rng.uniform(0.2, 2.5)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))

        # few random triangular bumps
        for _ in range(2):
            peak = rng.integers(0, n)
            seeds.append(generate_triangular_shape(n, peak))

        # ------------------------------------------------------------------
        # 3.  Examine each seed: LP improvement → exact polish → evaluate
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(seeds, start=1):
            if time.time() >= deadline - 5.0:
                break

            a = seed_vec.copy()
            a /= a.sum()                     # ensure simplex
            init_score = compute_score(a)
            print(
                f"\nSeed {idx}/{len(seeds)} – initial surrogate score = {init_score:.6f}"
            )

            # -------- LP‑based monotone improvement ------------------------
            # allocate a modest number of LP iterations (time‑aware)
            remaining = deadline - time.time()
            max_lp_iters = int(min(12, remaining * 5))   # at most ~12 solves
            if max_lp_iters <= 0:
                break

            a_lp, max_conv_lp = lp_improve(
                a,
                max_iters=max_lp_iters,
                deadline=deadline,
                step_decay=0.5,
            )
            surrogate_lp = compute_score(a_lp)
            print(f"    After LP – surrogate score = {surrogate_lp:.6f}")

            # -------- Exact polishing (sub‑gradient + EG) ------------------
            # Use the remaining budget for a short but effective polish
            if time.time() < deadline - 5.0:
                # sub‑gradient descent – the heavy part
                remaining = deadline - time.time()
                max_sub_iters = int(min(100_000, remaining * 30))
                a_refined = subgradient_descent_exact(
                    a_lp,
                    deadline,
                    max_iter=max_sub_iters,
                    step0=0.12,
                )
                # optional EG polish if time permits
                if time.time() < deadline - 2.0:
                    remaining = deadline - time.time()
                    max_eg_iters = int(min(80_000, remaining * 30))
                    a_refined = eg_refine(
                        a_refined,
                        deadline,
                        max_iter=max_eg_iters,
                        eta0=0.25,
                    )
            else:
                a_refined = a_lp

            # -------- Official evaluation ------------------------------------
            try:
                val = evaluate_sequence(a_refined.tolist())
                print(f"    Official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = a_refined.tolist()
                    print("*** New global best ***")
            except Exception as e:
                print(f"    Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 4.  Final global polish (if time still remains)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n--- Final global polishing ---")
        a = np.array(best_seq, dtype=float)

        # a short sub‑gradient phase
        remaining = deadline - time.time()
        max_sub_iters = int(min(150_000, remaining * 30))
        a = subgradient_descent_exact(
            a,
            deadline,
            max_iter=max_sub_iters,
            step0=0.09,
        )
        # final EG polish
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_eg_iters = int(min(100_000, remaining * 30))
            a = eg_refine(
                a,
                deadline,
                max_iter=max_eg_iters,
                eta0=0.20,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a non‑empty sequence
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
