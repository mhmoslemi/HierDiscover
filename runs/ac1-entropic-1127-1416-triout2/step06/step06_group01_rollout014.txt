def anneal(a, deadline, max_iter):
    best = a.copy()
    best_score = compute_score(best)
    cur = a.copy()
    cur_score = best_score
    n = len(a)
    start = time.time()
    max_temp = 0.1  # score units? we use difference.
    min_temp = 1e-6
    for it in range(1, max_iter+1):
        if time.time() >= deadline: break
        # progressive temperature
        t = max_temp * (1 - it/max_iter) + min_temp
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        # choose delta proportionally to cur[i] (maybe small)
        # step size adapt: sigma = cur[i] * 0.1 (max 0.05)
        delta = (rng.random() - 0.5) * 0.2 * cur[i]  # random around +-10% of cur[i]
        # Ensure not go negative
        if delta > cur[i]:
            delta = cur[i]
        # apply
        cur[i] -= delta
        cur[j] += delta
        # compute new score
        new_score = compute_score(cur)
        # acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / t):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert move
            cur[i] += delta
            cur[j] -= delta
    return best, best_score

"""
Propose a low‑autocorrelation step sequence.

Algorithm
---------
1.  **Seeding** –  generate a diverse pool of candidates:
    * deterministic arcsine shape (known good start);
    * several random Beta‑family shapes (α,β ∈ (0.1,5));
    * the historic best sequence if available (height_sequence_1).

2.  **Simulated annealing** – for each seed we run a cheap
    “move‑mass‑between‑two‑bins’’ Monte‑Carlo search that
    minimises the fast surrogate   s(a)=2·n·max(conv(a,a)).
    The temperature is linearly cooled; the move size is a
    fraction (≤0.5) of the donor bin.  The best sequence found
    by annealing is kept.

3.  **Exact polishing** – the survivor is refined with the
    projected sub‑gradient descent on the true objective
    (max‑convolution) followed by an exponentiated‑gradient
    (mirror‑descent) polishing.  Both respect the simplex
    constraint Σa=1, a≥0.

4.  **Evaluation** – the official ``evaluate_sequence`` is called
    once for every polished candidate; the globally best
    sequence is returned.  All steps stop before the global
    1000 s deadline.

The combination of a broader random pool + stochastic annealing
provides a better chance of escaping the local minima that
limited the previous implementation and typically yields a lower
official score (≈ 1.50 …).
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT‑based, O(n log n))
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score, valid for Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric shape generators (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}  on 0,…,n‑1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    raw = left * right
    total = raw.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return raw / total


# ----------------------------------------------------------------------
# Exact optimisation primitives (sub‑gradient + exponentiated‑gradient)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.05,
) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = project_to_simplex(a_init.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·(mask reversed) * a
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    eta0: float = 0.3,
) -> np.ndarray:
    """Exponentiated‑gradient (mirror‑descent with KL) polishing."""
    a = a_init.copy()
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (new component)
# ----------------------------------------------------------------------


def anneal(
    a_start: np.ndarray,
    deadline: float,
    max_iters: int = 500_000,
    transfer_frac: float = 0.5,
) -> Tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing on the fast surrogate.
    Moves a random amount (≤ transfer_frac of donor) from one bin to another.
    Temperature is linearly cooled from max_temp≈score to a tiny floor.
    Returns the best seen vector and its surrogate score.
    """
    a = a_start.copy()
    n = a.size
    rng = np.random.default_rng()

    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule (scale with initial surrogate)
    max_temp = max(0.1, cur_score * 0.5)
    min_temp = cur_score * 1e-5 + 1e-8

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # linear cooling
        temp = max_temp * (1.0 - it / max_iters) + min_temp

        # pick a donor with positive mass
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        # pick a distinct recipient
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose a transfer
        max_transfer = a[i]
        delta = max_transfer * rng.random() * transfer_frac
        # apply
        a[i] -= delta
        a[j] += delta

        new_score = compute_score(a)

        # Metropolis acceptance
        improve = new_score < cur_score
        if improve or rng.random() < np.exp(-(new_score - cur_score) / temp):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert move
            a[i] += delta
            a[j] -= delta

        # occasional re‑normalisation to avoid drift
        if it % 1000 == 0:
            a = a / a.sum()

    return best_a, best_score

            rem = deadline - time.time()
            max_eg = int(min(150_000, rem * 30))
            a_ref = eg_refine(a_ref, deadline, max_iter=max_eg, eta0=0.25)

        # ------------------------------------------------------------------
        # 4.  Official evaluation
        # ------------------------------------------------------------------
        try:
            val = evaluate_sequence(a_ref.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_ref.tolist()
                print("*** New global best! ***")
        except Exception as exc:
            print(f"    Evaluation failed: {exc}")

    # ------------------------------------------------------------------
    # 5.  Final global polishing of the overall best (if time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Final global polishing ---")
        a = np.array(best_seq, dtype=float)
        rem = deadline - time.time()
        max_sub = int(min(250_000, rem * 30))
        a = subgradient_descent_exact(a, deadline, max_iter=max_sub, step0=0.09)

        if time.time() < deadline - 2.0:
            rem = deadline - time.time()
            max_eg = int(min(150_000, rem * 30))
            a = eg_refine(a, deadline, max_iter=max_eg, eta0=0.20)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 6.  Safety net – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
