        param_candidates.append(shape)

    for frac in [0.05,0.08,0.12,0.18,0.25,0.35,0.45]:
        sigma = max(1.0, frac * n)
        shape = generate_gaussian_shape(n, sigma)
        param_candidates.append(shape)

    # symmetric beta shape sweep
    for a in np.arange(0.2, 1.1, 0.1):
        shape = generate_beta_shape(n, a, a, eps=1e-12)
        param_candidates.append(shape)

    # random seed shapes
    for _ in range(10):
        # random beta parameters
        a = rng.uniform(0.1, 2.0)
        b = rng.uniform(0.1, 2.0)
        eps = rng.uniform(0, 1.0)
        shape = generate_beta_shape(n, a, b, eps)
        param_candidates.append(shape)

    # evaluate fast scores, keep top K
    scored = [(compute_score(s), s) for s in param_candidates]
    scored.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [vec for _, vec in scored[:top_k]]

    # optionally add best global fallback
    if best_seq:
        top_candidates.append(np.array(best_seq, dtype=float))

    # ---------------------------------
    # 2. refine each top candidate with Adam
    # ---------------------------------
    for idx, seed_vec in enumerate(top_candidates, start=1):
        if time.time() >= deadline - 10.0:
            break
        # allocate time per candidate uniformly
        remaining = deadline - time.time()
        cand_deadline = time.time() + remaining / (len(top_candidates) - idx + 1) - 2.0
        # ensure at least few seconds left after

   keep the best few (default 5).
4. Refine each of these seeds with a smooth‑max surrogate
   minimised by the Adam optimiser.  The surrogate replaces the
   non‑differentiable max by a log‑sum‑exp with a temperature β
   that is increased linearly from 5 to 250.  The optimisation runs
   until the per‑seed time budget expires.
5. The best Adam‑refined candidate is evaluated with the official
   `evaluate_sequence`.  Afterwards a *projected sub‑gradient descent*
   (exact max‑convolution sub‑gradient) is performed for a final polish.
6. The routine always returns a non‑empty sequence; if everything fails
   a simple uniform 512‑vector is used.

The whole search respects the 1000 s wall‑clock limit and prints
progress information.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (no wrap‑around).
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector `a` must be non‑negative and sum to 1.
    Returns   2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple parametric generators (all return a normalised probability vector)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine shape ∝ 1/√(i·(n‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_power_shape(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """
    a_i ∝ 1 / ((i+½)*(n‑½‑i)+eps)ᵖ.
    For p=0.5 this coincides (up to scaling) with the arcsine shape.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float = 1e-12
) -> np.ndarray:
    """
    Symmetric (or not) Beta‑family:
        a_i ∝ (i+eps)^{α‑1}·(n‑1‑i+eps)^{β‑1}.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_gaussian_shape(n: int, sigma: float, eps: float = 1e-12) -> np.ndarray:
    """
    Discrete centred Gaussian, later normalised.
    """
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - mu) / sigma) ** 2) + eps
    vals /= vals.sum()
    return vals


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture arcsine·weight + uniform·(1‑weight)."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Projection utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector with non‑negative entries that sum to 1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.05,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterates stay on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance to treat close‑to‑max entries as active
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2·(a ⋆ mask_rev) restricted to original length
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)          # length = n+L‑1 = 3n‑2
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


# ----------------------------------------------------------------------
# Smooth max + Adam optimiser
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 250.0,
    ramp_iters: int = 40_000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    Minimise the smooth surrogate of `max(conv(a,a))` using Adam.
    The variable is the log‑it vector `z`; after each update `a=softmax(z)`.

    Parameters
    ----------
    z0 : np.ndarray
        Initial log‑it vector.
    deadline : float
        Unix time after which the optimiser must stop.
    n : int
        Length of the step function (required for slicing).
    beta0, beta_max : float
        Temperature schedule for the log‑sum‑exp surrogate.
    ramp_iters : int
        Number of iterations over which β grows linearly.
    lr : float
        Base learning rate for Adam.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ∈ simplex
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear temperature increase
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # soft‑max weights for the surrogate
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ⋆ w_rev)  (take the central n entries)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = n + (2n‑1) – 1 = 3n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule through the soft‑max parametrisation
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

    # final a on the simplex
    a = _softmax(z)
    return a


# ----------------------------------------------------------------------
# Main entry point for the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a low‑value step sequence within the given time budget.
    Returns a list of non‑negative numbers (the heights of the step function).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use a previously known construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose a working length (4096 is a good trade‑off)
    # ------------------------------------------------------------------
    n = 4096
    if deadline - time.time() < 60:   # not enough time – fall back to 2048
        n = 2048
    print(f"\n=== Optimisation length set to {n} ===")

    # ------------------------------------------------------------------
    # 2.  Build a diverse pool of parametric seeds
    # ------------------------------------------------------------------
    seed_pool: List[np.ndarray] = []

    # deterministic families
    seed_pool.append(arcsine_initial(n))
    for w in np.linspace(0.0, 1.0, 11):      # mixture arcsine / uniform
        seed_pool.append(generate_mixture_shape(n, w))

    # power‑law sweep
    for p in np.arange(0.20, 0.95, 0.05):
        seed_pool.append(generate_power_shape(n, p, eps=1e-12))

    # symmetric Beta sweep (α=β)
    for a in np.arange(0.20, 1.10, 0.10):
        seed_pool.append(generate_beta_shape(n, a, a, eps=1e-12))

    # centred Gaussian sweep
    for frac in [0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.45]:
        sigma = max(1.0, frac * n)
        seed_pool.append(generate_gaussian_shape(n, sigma))

    # a few random beta‑family seeds
    for _ in range(10):
        a = rng.uniform(0.1, 2.0)
        b = rng.uniform(0.1, 2.0)
        eps = rng.uniform(0.0, 1.0)
        seed_pool.append(generate_beta_shape(n, a, b, eps))

    # ------------------------------------------------------------------
    # 3.  Quick surrogate ranking – keep the most promising few
    # ------------------------------------------------------------------
    scored = [(compute_score(s), s) for s in seed_pool]
    scored.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [vec for _, vec in scored[:top_k]]

    # also keep the best fallback (if any)
    if best_seq:
        top_candidates.append(np.array(best_seq, dtype=float))

    print(f"\nSelected {len(top_candidates)} top seeds for Adam refinement:")

    # ------------------------------------------------------------------
    # 4.  Adam refinement (one pass per seed, time‑budget aware)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_candidates, start=1):
        if time.time() >= deadline - 8.0:
            print("Time limit almost reached – leaving refinement loop.")
            break

        # allocate the remaining time roughly equally among the still‑to‑run seeds
        candidates_left = len(top_candidates) - idx + 1
        remaining_time = deadline - time.time()
        per_seed_budget = max(8.0, remaining_time / candidates_left - 2.0)
        seed_deadline = time.time() + per_seed_budget

        print(f"\n--- Refine seed {idx}/{len(top_candidates)} (budget ≈{per_seed_budget:.1f}s) ---")
        # initialise Adam with log‑softmax of the seed (avoids division by zero)
        eps = 1e-12
        z0 = np.log(seed_vec + eps)

        a_opt = smooth_adam_optimize(
            z0,
            seed_deadline,
            n=n,
            beta0=5.0,
            beta_max=250.0,
            ramp_iters=40_000,
            lr=0.04,
        )

        # cheap surrogate check
        surrogate = compute_score(a_opt)
        print(f"   Surrogate after Adam: {surrogate:.6f}")

        # official evaluation (may be a bit slower but still cheap)
        try:
            val = evaluate_sequence(a_opt.tolist())
            print(f"   Official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                print(f"   *** New global best! ***")
        except Exception as e:
            print(f"   Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 5.  Final polishing with exact sub‑gradient descent (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\nFinal exact sub‑gradient polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=200_000,
            step0=0.08,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception as e:
            print(f"   Final polishing error: {e}")

    # ------------------------------------------------------------------
    # 6.  Safety net – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        print("\nNo valid sequence found – falling back to uniform 512‑vector.")
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
