"""
Propose a low‑score non‑negative step function.

Algorithm Overview
------------------
1.  Work with a fixed length `n = 2048` (fits comfortably in memory and
    gives a theoretical lower bound ≈ 1).
2.  Perform several restarts – one uniform start and a few random
    Dirichlet draws.  Each restart consists of:
        *  a “soft‑max” mirror‑descent optimiser (the original routine),
           which quickly drives the convolution towards a flat shape;
        *  a “hard‑max” sub‑gradient optimiser that focuses exclusively on
           the current maximum entry of the convolution and uses an
           exponentiated update; this usually pushes the maximum a little
           further down.
3.  The best vector obtained so far is polished by a stochastic hill‑climber
    (swap / mass‑transfer moves) that uses the exact `evaluate_sequence`
    score for acceptance.
4.  If everything fails a tiny uniform fallback is returned.

All operations respect the 1000 s time budget; the code checks the
deadline frequently and stops early if necessary.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector whose entries sum to 1:
        2·n·max(a*a)
    (the same value returned by `evaluate_sequence` after normalisation).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Soft‑max (smooth) optimiser – same as the reference implementation
# ----------------------------------------------------------------------


def smooth_optimize(
    a0: np.ndarray,
    time_deadline: float,
    iters: int = 180_000,
    beta0: float = 0.5,
    beta_max: float = 30.0,
    eta0: float = 0.4,
    report_every: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimiser that minimises a soft‑max approximation of the
    infinity‑norm of the self‑convolution.
    """
    n = a0.shape[0]
    L = next_pow2(3 * n - 2)          # length needed for all FFTs
    a = a0.copy()
    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, iters + 1):
        if time.time() >= time_deadline - 1.0:
            break

        # ----- a*a ----------------------------------------------------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = np.max(conv)

        # ----- soft‑max weights ---------------------------------------
        beta = min(beta_max, beta0 * np.sqrt(t))
        exp_terms = np.exp(beta * (conv - max_c))
        w = exp_terms / np.sum(exp_terms)

        # ----- gradient = 2·(w ⋆ a_rev) -------------------------------
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- exponentiated (mirror) update -------------------------
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  smooth iter {t:8d} – new best {cur_val:.6f}")

    # final check
    cur_val = fast_score(a)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Hard‑max (sub‑gradient) optimiser
# ----------------------------------------------------------------------


def hardmax_optimize(
    a0: np.ndarray,
    time_deadline: float,
    max_iters: int = 120_000,
    eta0: float = 0.5,
    report_every: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Optimises the *true* infinity norm by repeatedly moving away from the
    current maximal convolution entry.
    The update is exponentiated (mirroring the soft‑max optimiser) and the
    vector is projected back onto the simplex after each step.
    """
    n = a0.shape[0]
    L = next_pow2(3 * n - 2)
    a = a0.copy()
    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline - 1.0:
            break

        # ----- convolution -------------------------------------------------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]

        # ----- index of the current maximum -------------------------------
        k_max = int(np.argmax(conv))

        # ----- sub‑gradient for the maximal entry -------------------------
        # grad_j = 2 * a_{k_max - j}  (zero when the index is out of range)
        a_rev = a[::-1]
        shift = n - 1 - k_max                 # distance between a_rev[0] and the
                                              # element that contributes to conv[k_max]
        grad = np.zeros_like(a)
        if shift >= 0:
            grad[: n - shift] = 2.0 * a_rev[shift:]
        else:
            s = -shift
            grad[s:] = 2.0 * a_rev[: n - s]

        # ----- exponentiated update ---------------------------------------
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  hard   iter {t:8d} – new best {cur_val:.6f}")

    # final check
    cur_val = fast_score(a)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climber that performs swaps or transfers of a
    fraction of mass between two positions.  The exact `evaluate_sequence`
    score is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    best_a = a.copy()
    best_score = fast_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer move

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(len(a), size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(len(a), size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # enlarge step after success
        else:
            step_scale *= 0.97                         # shrink after failure

        max_steps -= 1

    return best_a.tolist(), best_score

            report_every=15_000,
        )
        if soft_score < best_val:
            best_val = soft_score
            best_seq = a_soft.tolist()
            print(f"Restart {r_id} – after soft‑opt  {soft_score:.6f}")

        # ----- hard‑max refinement -----------------------------------------
        a_hard, hard_score = hardmax_optimize(
            np.array(best_seq),
            time_deadline=deadline - 5.0,
            max_iters=120_000,
            eta0=0.5,
            report_every=10_000,
        )
        if hard_score < best_val:
            best_val = hard_score
            best_seq = a_hard.tolist()
            print(f"Restart {r_id} – after hard‑opt  {hard_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final hill‑climber polishing (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print("\nRunning final hill‑climb polishing")
        polished_seq, polished_val = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=300_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Safeguard – tiny uniform fallback if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 6️⃣  Report final score and return the sequence
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
