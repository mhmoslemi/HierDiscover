def lp_step(a, target_sum, M, t=0.05, max_active=30):

    n = len(a)

    conv_self = np.convolve(a, a)

    idxs = np.argpartition(-conv_self, max_active-1)[:max_active]

    # optionally add any near-M

    eps_rel = 1e-6

    mask = conv_self >= M * (1 - eps_rel)

    idxs = np.unique(np.concatenate([idxs, np.where(mask)[0]]))

    # Build A

    rows = []

    for k in idxs:

        row = np.zeros(n)

        start = max(0, k-(n-1))

        end = min(n-1, k)

        # iterate i from start..end

        # j = k - i

        # row[j] = a[i]

        # Use vectorization: for i range

        for i in range(start, end+1):

            j = k - i

            row[j] = a[i]

        rows.append(row)

    A = np.vstack(rows)

    # Build bounds: b >= 0

    bounds = [(0, None)] * n

    # Objective c: maximize sum(b) -> minimize -sum(b)

    c = -np.ones(n)

    # Solve

    res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M), bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})

    if not res.success:

        return None

    b = res.x

    # Rescale

    sum_b = b.sum()

    if sum_b <= 0:

        return None

    scale = target_sum / sum_b

    b_scaled = b * scale

    # Mix

    a_new = (1 - t) * a + t * b_scaled

    return a_new

"""
propose_candidate – search for a low‑value step sequence.

The algorithm mixes several ideas:

1. **Random shape sampling** – beta‑family densities (single or mixtures),
   the classic arcsine shape and the uniform vector are generated for a few
   lengths.  Fast surrogate score
   `2·n·max(conv(a,a))` (with the vector normalised to Σa=1) is used to keep
   the best few candidates.

2. **Alternating LP refinement** – for each survivor we run a short
   alternating optimisation inspired by the LP‑based construction from the
   problem statement.  With the current vector `f` we compute `M = max(f*f)`,
   pick the indices where the self‑convolution is maximal (a small active
   set) and solve a tiny linear programme

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   for the active k,
                    b_j ≥ 0.

   The solution `b` is rescaled to the normalisation
   `Σb = sqrt(2·n)` and blended into `f`.  A few iterations quickly drive
   the max‑convolution down while preserving the total sum.

3. **Local polishing** – the refined vectors are further optimised with the
   smooth‑gradient Adam optimiser, an exact projected sub‑gradient descent,
   and a cheap stochastic Gaussian walk (the same components that were used
   in the previous state‑of‑the‑art code).

4. **Global polishing** – after all lengths have been processed the current
   best vector receives a final round of Adam → sub‑gradient → random walk.

All stages respect the global 1000 s budget and the best vector seen
according to the *official* `evaluate_sequence` is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – fast for all n)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (FFT version)."""
    return float(np.max(convolution_fft(a, a)))


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector is normalised to Σa = 1, therefore the denominator is 1
    and the score equals 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    a_norm = a / s
    return 2.0 * a.shape[0] * max_convolution_fft(a_norm)


# ----------------------------------------------------------------------
# Shape generators (beta families and arcsine)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density – the shape used in the original baseline."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # tiny regulariser to avoid division by zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – for Adam
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                           # soft‑max weights
    B = max_c + np.log(sum_exp) / lam                # smooth max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)             # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 400.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector found (projected on the simplex) together
    with its fast surrogate score.
    """
    a = init_a.astype(float).copy()
    a = a / a.sum()                     # project to simplex
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # anneal the temperature λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        # simplex projection
        a_new = a_new - np.max(a_new) + 1e-12   # shift to avoid negatives
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc

# Exact sub‑gradient descent with simple back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the true (official) objective.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step size decays like 1/√t
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            new_val = max_convolution_fft(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc

def random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap Gaussian walk on the simplex with a tiny uphill‑move acceptance
    probability, useful for final polishing.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = np.maximum(a_prop, 0.0)
        a_prop = a_prop / a_prop.sum()

        cur_sc = fast_score(a_prop)

        # accept if better, otherwise tiny probability to escape plateaus
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

    return best_a, best_sc

def alternating_optimize(
    init_a: np.ndarray,
    deadline: float,
    iter_max: int = 30,
    step_size: float = 0.15,
    active_K: int = 30,
    eps_rel: float = 1e-6,
) -> np.ndarray:
    """
    Short alternating scheme based on the LP described in the problem
    statement.

    1. Normalise the vector to Σa = sqrt(2·n).
    2. Repeatedly:
        * compute M = max(f*f)
        * pick at most ``active_K`` positions where the self‑convolution is
          maximal (plus any indices within a relative tolerance ``eps_rel``)
        * solve a tiny LP:
              maximise Σ b_j  subject to (f*b)_k ≤ M on the active set,
              b_j ≥ 0.
        * rescale b to the same normalisation and blend:
              f ← (1‑τ)·f + τ·b   with a decaying τ.
    The function returns the best vector (according to the fast surrogate)
    seen during the iterations.
    """
    n = init_a.size
    target_sum = np.sqrt(2.0 * n)

    # initialise and normalise to the target sum
    a = init_a.astype(float).copy()
    if a.sum() == 0.0 or not np.isfinite(a.sum()):
        a = np.full(n, target_sum / n)
    else:
        a = a * (target_sum / a.sum())

    best = a.copy()
    best_sc = fast_score(a)

    for it in range(iter_max):
        if time.time() >= deadline:
            break

        # self‑convolution and its maximum
        conv_self = np.convolve(a, a)          # length 2n‑1, dense but cheap
        M = conv_self.max()

        # active indices – top K values and any that are within eps_rel of the max
        if active_K < conv_self.size:
            top_idx = np.argpartition(-conv_self, active_K - 1)[:active_K]
        else:
            top_idx = np.arange(conv_self.size)
        mask = conv_self >= M * (1.0 - eps_rel)
        idxs = np.unique(np.concatenate([top_idx, np.where(mask)[0]]))

        # build the (tiny) inequality matrix A·b ≤ M
        rows = np.zeros((len(idxs), n), dtype=float)
        for r, k in enumerate(idxs):
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            i_range = np.arange(start, end + 1)
            j_range = k - i_range
            rows[r, j_range] = a[i_range]

        # linear programme: maximise Σ b_j  ⇔ minimise −Σ b_j
        c = -np.ones(n, dtype=float)
        bounds = [(0.0, None) for _ in range(n)]
        # a tiny slack avoids numerical infeasibility
        b_ub = np.full(rows.shape[0], M + 1e-12)

        res = linprog(
            c,
            A_ub=rows,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        b = np.maximum(res.x, 0.0)
        sum_b = b.sum()
        if sum_b <= 0.0 or not np.isfinite(sum_b):
            # fallback to uniform on the simplex
            b = np.full(n, target_sum / n)
        else:
            b = b * (target_sum / sum_b)

        # decaying blending factor τ
        tau = step_size * (0.95 ** it)

        a_new = (1.0 - tau) * a + tau * b
        # re‑normalise to keep the exact target sum (numerical drift)
        a_new = a_new * (target_sum / a_new.sum())

        sc = fast_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best = a_new.copy()

        a = a_new

    return best

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine combines random beta‑family sampling, a short alternating
    LP refinement and the classic Adam → sub‑gradient → random‑walk polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate with the *official* evaluator.
    # ------------------------------------------------------------------
    def official_score(vec: np.ndarray) -> float:
        return evaluate_sequence(vec.tolist())

    # ------------------------------------------------------------------
    # 0.  Use a known construction if it exists (fallback).
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = official_score(np.array(fallback))
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Explore a handful of lengths.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n=== Length {n} ===")

        # --------------------------------------------------------------
        # 1a. Deterministic starters.
        # --------------------------------------------------------------
        starters = [
            np.full(n, 1.0 / n),   # uniform
            arcsine_shape(n),      # classic arcsine
        ]

        # --------------------------------------------------------------
        # 1b. Random beta‑family shapes (including mixtures).
        # --------------------------------------------------------------
        candidates = starters.copy()
        max_samples = int(2500 * (budget_s / 1000.0) * (256 / n))  # fewer for larger n
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break

            if rng.random() < 0.35:   # mixture
                w = rng.random()
                a1 = generate_beta_shape(
                    n,
                    rng.uniform(0.02, 1.5),
                    rng.uniform(0.02, 1.5),
                    rng.uniform(1e-9, max(0.01, 0.02 * n)),
                )
                a2 = generate_beta_shape(
                    n,
                    rng.uniform(0.02, 1.5),
                    rng.uniform(0.02, 1.5),
                    rng.uniform(1e-9, max(0.01, 0.02 * n)),
                )
                cand = w * a1 + (1.0 - w) * a2
            else:
                cand = generate_beta_shape(
                    n,
                    rng.uniform(0.02, 1.5),
                    rng.uniform(0.02, 1.5),
                    rng.uniform(1e-9, max(0.01, 0.02 * n)),
                )
            candidates.append(cand)

        # --------------------------------------------------------------
        # 1c. Keep the few best according to the fast surrogate.
        # --------------------------------------------------------------
        scores = [fast_score(c) for c in candidates]
        top_indices = np.argsort(scores)[:5]   # keep at most five survivors
        survivors = [candidates[i] for i in top_indices]

        # --------------------------------------------------------------
        # 2. Refine each survivor.
        # --------------------------------------------------------------
        for sid, vec in enumerate(survivors, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Survivor {sid}/{len(survivors)} (n={n})")

            # ---- Alternating LP refinement (new ingredient) ----
            a = alternating_optimize(vec, deadline, iter_max=25, step_size=0.18)

            # ---- Adam fine‑tuning ----
            if time.time() < deadline - 3.0:
                remaining = deadline - time.time()
                max_iter_adam = int(min(180_000, remaining * 30))
                a, _ = adam_optimize(
                    init_a=a,
                    deadline=deadline,
                    max_iter=max_iter_adam,
                    lr0=0.09,
                    lam_start=1.0,
                    lam_end=400.0,
                    noise_every=1200,
                    noise_scale=3e-5,
                )

            # ---- Exact sub‑gradient descent ----
            if time.time() < deadline - 2.0:
                remaining = deadline - time.time()
                max_iter_sub = int(min(120_000, remaining * 25))
                a, _ = subgradient_optimize(
                    init_a=a,
                    deadline=deadline,
                    step0=0.22,
                    max_iter=max_iter_sub,
                )

            # ---- Final random walk ----
            if time.time() < deadline - 1.0:
                remaining = deadline - time.time()
                max_iter_rw = int(min(150_000, remaining * 30))
                a, _ = random_local_search(
                    init_a=a,
                    deadline=deadline,
                    init_step=0.013,
                    max_iter=max_iter_rw,
                )

            # ---- Official evaluation ----
            val = official_score(a)
            print(f"    official score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print(f"    *** New global best = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3. Global polishing of the overall best vector (if any time left).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # A – long Adam run with a very high temperature
        remaining = deadline - time.time()
        max_iter_adam = int(min(800_000, remaining * 40))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.08,
            lam_start=1.0,
            lam_end=500.0,
            noise_every=1300,
            noise_scale=2e-5,
        )

        # B – exact sub‑gradient descent
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(200_000, remaining * 30))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                init_a=a,
                deadline=deadline,
                step0=0.20,
                max_iter=max_iter_sub,
            )

        # C – final stochastic walk
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_rw = int(min(300_000, remaining * 45))
            print(f"  Random walk polishing (max_iter={max_iter_rw}) …")
            a, _ = random_local_search(
                init_a=a,
                deadline=deadline,
                init_step=0.012,
                max_iter=max_iter_rw,
            )

        final_val = official_score(a)
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4. Guard – guarantee a non‑empty return.
    # ------------------------------------------------------------------
    if not best_seq:
        # safe uniform fallback (length 256)
        best_seq = [1.0] * 256
        best_val = official_score(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    # ensure non‑negative output
    best_seq = [max(0.0, float(v)) for v in best_seq]
    return best_seq
