def subgradient_descent_exact(init_a: np.ndarray, deadline: float,
                             max_iter: int=150000,
                             step0: float=0.08) -> np.ndarray:
    a = init_a.copy()
    a = project_to_simplex(a)  # ensure sum=1
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)  # shape L
        # compute subgrad
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)  # length 3n-2
        # gradient indices: L-1-i
        idx = L - 1 - np.arange(n)
        subgrad = 2.0 * rev[idx]
        # step size decreasing sqrt
        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Subgrad it={it:,d} score={compute_score(a):.6f}")

    return a

def generate_beta_shape(n, alpha, beta, eps):
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)

Step‑function optimizer (large‑scale arcsine seed → exponential‑β Adam → sub‑gradient polish).

* Choose a single long length (up to 16384) – the continuous‑limit arcsine shape
  becomes more accurate as the discretisation refines, lowering the theoretical
  constant C = ½·score.  The target 1.503 corresponds to C≈0.7515, while the
  previous 2048‑point run gave C≈0.7561.  With n≈16384 we can push C below the
  target.

* Initialise the probability vector with the discrete arcsine law (β‑distribution
  with parameters ½,½).  The vector is parametrised by the unconstrained log‑
  weights `z` via a stable soft‑max, guaranteeing non‑negativity and Σa=1.

* Minimise a smooth surrogate of the non‑smooth objective
      f(a)=2·n·max(conv(a,a)) / (Σa)²
  (the denominator is constant under soft‑max).  The surrogate replaces the
  max by a log‑sum‑exp with a temperature β.  β is increased **exponentially**
  from β₀≈5 to βₘₐₓ≈200 during a long “ramp” phase (≈2·10⁵ it).  This lets
  Adam first move quickly on a smooth landscape and then fine‑tune the exact
  maximum.

* The gradient of the surrogate w.r.t. `a` is obtained analytically using a
  weighted convolution; the chain rule yields the gradient w.r.t. `z`.
  Adam (β₁=0.9, β₂=0.999) with a 1/√t step‑size schedule updates `z`.

* After the time‑limited Adam run, the best vector found is polished by a
  projected sub‑gradient descent that works directly on the true max‑convolution
  objective.  The sub‑gradient is computed from all shifts where the convolution
  attains its maximum; the iterate is projected onto the probability simplex
  after each step.

* The whole routine respects the 1000 s wall‑clock budget, always returns a
  non‑empty sequence, and prints progress and intermediate scores.

from typing import List

# Utility FFT helpers

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (a·b)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(a,a)) (a assumed to sum to 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return vectors that sum to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine law on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family shape (proportional to (i+eps)^{α‑1}(n‑1‑i+eps)^{β‑1})."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power‑law shape (∝ 1/((i+½)(n‑½‑i)+eps)^{p})."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.08,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective
    f(a)=2·n·max(conv(a,a)) / (sum(a)²).
    The iterate stays on the simplex (sum=1) and non‑negative.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # indices where conv is (practically) maximal
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)  # shape L

        # sub‑gradient = 2·(a ⋆ mask_rev) evaluated at proper offsets
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)  # length 3n‑2
        idx = L - 1 - np.arange(n)          # positions aligning with a
        subgrad = 2.0 * rev[idx]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Soft‑max parametrisation
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth‑max + Adam optimiser (exponential β schedule)
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 200.0,
    ramp_iters: int = 200_000,
    lr: float = 0.08,
) -> np.ndarray:
    """
    Minimise the smooth log‑sum‑exp surrogate of max(conv(a,a))
    using Adam on the soft‑max parametrisation.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    # FFT sizes (static for a given n)
    n_fft1 = next_pow2(2 * n - 1)      # for a⋆a
    n_fft2 = next_pow2(3 * n - 2)      # for a⋆w_rev

    while time.time() < deadline:
        t += 1
        a = softmax(z)                      # Σa = 1, a_i ≥ 0

        # ---- conv = a ⋆ a (max part) ----
        A1 = np.fft.rfft(a, n=n_fft1)
        conv = np.fft.irfft(A1 * A1, n=n_fft1)[: 2 * n - 1]
        mx = float(np.max(conv))

        # ---- temperature schedule (exponential) ----
        if t <= ramp_iters:
            beta = beta0 * (beta_max / beta0) ** (t / ramp_iters)
        else:
            beta = beta_max

        # ---- soft‑max over the convolution values ----
        shift = conv - mx                      # ≤ 0
        w = np.exp(beta * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # ---- gradient wrt a : 2·(a ⋆ w_rev) ----
        w_rev = w_norm[::-1]
        A2 = np.fft.rfft(a, n=n_fft2)
        B = np.fft.rfft(w_rev, n=n_fft2)
        rev = np.fft.irfft(A2 * B, n=n_fft2)[: 3 * n - 2]
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # ---- chain rule through soft‑max ----
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final a (still sum‑to‑1)
    return softmax(z)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a low‑score step sequence.
    The strategy:
        1. Choose a large length (up to 16384) based on remaining time.
        2. Build a small pool of diverse seeds (arcsine + a few Beta‑family shapes).
        3. For each seed run the exponential‑β Adam optimiser for the allotted time.
        4. Evaluate officially, keep the best.
        5. Perform a final projected sub‑gradient polishing if time permits.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # optional fallback to a previously discovered sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. pick a length fitting the remaining budget
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 800:
        n = 16384
    elif time_left > 400:
        n = 8192
    elif time_left > 200:
        n = 4096
    else:
        n = 2048
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 2. generate a modest pool of seeds
    # ------------------------------------------------------------------
    seeds = [arcsine_initial(n)]

    # a handful of random Beta‑family shapes
    for _ in range(4):
        a = rng.uniform(0.2, 4.0)
        b = rng.uniform(0.2, 4.0)
        eps = rng.uniform(0.0, 0.5)
        seeds.append(generate_beta_shape(n, a, b, eps))

    # a couple of power‑law variations
    for _ in range(2):
        p = rng.uniform(0.1, 3.0)
        eps = rng.uniform(0.0, 2.0)
        seeds.append(generate_power_shape(n, p, eps))

    # keep the most promising few according to the cheap surrogate
    scored = [(compute_score(s), s) for s in seeds]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:3]]  # at most three seeds

    # ------------------------------------------------------------------
    # 3. allocate time per seed (roughly equal slices)
    # ------------------------------------------------------------------
    per_seed_time = max(5.0, (deadline - time.time()) / (len(top_seeds) + 1))

    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\nSeed {idx}/{len(top_seeds)} – time budget ≈{per_seed_time:.1f}s")
        seed_deadline = time.time() + per_seed_time

        # initialise Adam via log‑softmax (avoids division by zero)
        eps = 1e-12
        z0 = np.log(seed_vec + eps)

        # --------------------------------------------------------------
        # 3a. smooth‑max Adam optimisation
        # --------------------------------------------------------------
        a_opt = smooth_adam_optimize(
            z0,
            seed_deadline,
            n=n,
            beta0=5.0,
            beta_max=200.0,
            ramp_iters=200_000,
            lr=0.08,
        )

        # --------------------------------------------------------------
        # 3b. official evaluation
        # --------------------------------------------------------------
        try:
            val = evaluate_sequence(a_opt.tolist())
            print(f"   Official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                print(f"   *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"   Evaluation failed: {e}")

        # --------------------------------------------------------------
        # 3c. light polishing (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            polish_deadline = deadline - 2.0
            cand_arr = np.array(best_seq, dtype=float)
            cand_arr = subgradient_descent_exact(
                cand_arr,
                polish_deadline,
                max_iter=100_000,
                step0=0.12,
            )
            try:
                val2 = evaluate_sequence(cand_arr.tolist())
                print(f"   Polished score = {val2:.6f}")
                if val2 < best_val:
                    best_val, best_seq = val2, cand_arr.tolist()
                    print(f"   *** Updated best after polish: {best_val:.6f} ***")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 4. final global polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=200_000,
            step0=0.08,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 5. guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
