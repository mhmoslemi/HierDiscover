"""
Improved step‑function optimizer.

The search proceeds in three phases:
1.  **Diverse restarts** – generate a handful of sensible initial
    vectors (uniform, geometric, Gaussian, random sparse) on the
    probability simplex.
2.  **Smoothed log‑sum‑exp Adam descent** – minimise a smooth
    approximation of the max‑convolution using an Adam optimiser.
    The smoothness parameter β is annealed (0.5 → 200) so the
    optimisation gradually focuses on the true ℓ∞ objective.
3.  **Refinement** – a classic projected sub‑gradient descent followed
    by a stochastic hill‑climb (mass‑transfer moves) that accepts
    occasional uphill steps (simulated‑annealing style).

All heavy work is performed with FFT‑based convolutions
(O(n log n)) and the algorithm respects the 1000 s wall‑clock limit.
The returned list of floats is ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a ⊛ b using a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp optimisation (Adam)
# ----------------------------------------------------------------------


def smooth_optimize_adam(start_a: np.ndarray,
                         time_deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 0.5,
                         beta_end: float = 200.0,
                         lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv(a,a)) using Adam.
    The smooth objective is
        f_β(a) = (1/β)·log Σ exp(β·conv_k)
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β (exponential schedule)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smooth objective weights (softmax)
        conv = convolution_fft(a, a)                 # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()          # ∑w = 1

        # gradient of the smooth objective (cross‑correlation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)         # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]       # slice of length n

        # Adam update with a 1/√t step‑size scaling
        lr = lr0 / np.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a_new)

        # occasional exact evaluation
        if it % 300 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    # final check
    final = compute_score(a)
    if final < best_score:
        best_score = final
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (projected SGD)
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 300_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the true max‑objective.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max_k conv_k  (2·a shifted)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a = project_to_simplex(a_new)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # tiny jitter to escape plateaus
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a)

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb_sa(seq,
                  time_deadline,
                  max_steps: int = 400_000,
                  init_temp: float = 0.02,
                  final_temp: float = 1e-4) -> tuple[list[float], float]:
    """
    Simple stochastic hill‑climb that moves a fraction of mass between
    two positions.  Accepts uphill moves with a temperature schedule.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # exponential temperature decay
    temp_factor = (final_temp / init_temp) ** (1.0 / max_steps)
    T = init_temp
    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        i, j = np.random.randint(n, size=2)
        if i != j:
            delta = step_scale * cand[i]                # move a fraction of i
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0
        else:  # rare swap when i==j (no effect)
            pass

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # accept with probability exp(-(Δ)/T)
            if np.random.rand() < np.exp(-(cur_score - best_score) / max(T, 1e-12)):
                best_a = cand
                best_score = cur_score
                step_scale = max(step_scale * 0.97, 0.01)

        T *= temp_factor
        max_steps -= 1

    return best_a.tolist(), best_score

        C = (c / n) / (1.0 - r ** n)
        a = C * (r ** np.arange(n))
        a = a / a.sum()
        init_candidates.append(a)

    for sigma_factor in [0.07, 0.12, 0.20]:

        a = project_to_simplex(a)
        init_candidates.append(a)

    # random sparse starts (5 %–50 % density)
    for _ in range(8):
        p = np.random.uniform(0.05, 0.5)
        k = max(2, int(round(p * n)))

        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k
