```python
def next_pow2(x): return 1 << (x-1).bit_length()
def convolution_fft(a,b):
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    C = A*B
    conv = np.fft.irfft(C, n_fft)[:m]
    return conv
def max_convolution(a):
    return float(np.max(convolution_fft(a,a)))
def compute_score(a):
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)
def project_to_simplex(v):
    # Duchi's O(n log n) algorithm
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0/n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
def symmetrize(v):
    return (v + v[::-1]) / 2.0
```

```python
def flatten_gradient(a):
    n = a.shape[0]
    conv = convolution_fft(a, a)
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff**2))
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)  # length 3n-2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L-1 - np.arange(n)]
    return grad, loss
```

```python
def adam_flatten(initial_a, time_deadline, max_iter=150_000, lr0=0.25):
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        grad, loss = flatten_gradient(a)
        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad**2)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_to_simplex(a_new)
        a_new = symmetrize(a_new)
        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new
        if it % 20000 == 0:
            print(f"    Adam-flatten iter={it} loss={loss:.3e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}")
    return best_a, best_sc
```

Now smooth_gradient:

We've defined earlier.

Note we need to compute sum a (should be 1). The formula uses (2 n / s^2) * weighted_grad - (4n B / s^3) * 1 (vector). With s=1 simplifies: term1 = 2n * weighted_grad, term2 = -4n * B * ones.

Thus we can simplify perhaps.

We'll keep general.

```python
def smooth_gradient(a, lam):
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_c = np.max(conv)
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    B = max_c + np.log(sum_exp) / lam
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 3n-2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L-1 - np.arange(n)]
    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B
```

```python
def adam_max(initial_a, time_deadline, max_iter=200_000,
             lr0=0.12, lam_start=1.0, lam_end=500.0,
             noise_every=1000, noise_scale=4e-5):
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    n = a.shape[0]
    for it in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)
        grad, _ = smooth_gradient(a, lam)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale
        a_new = project_to_simplex(a_new)
        a_new = symmetrize(a_new)
        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new
        if it % 5000 == 0:
            print(f"    Adam-max iter={it} cur={cur_sc:.6f} best={best_sc:.6f} lam={lam:.1f}")
    return best_a, best_sc
```

```python
def subgradient_optimize(initial_a, time_deadline, step0=0.3, max_iter=200_000):
    a = initial_a.copy()
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    for it in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))
        # gradient: 2 * a_{idx - i} where idx-i in [0,n)
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            a_new = symmetrize(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5
        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new
        if it % 40000 == 0:
            print(f"    Subgrad iter={it} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc
```

```python
def random_local_search(initial_a, time_deadline,
                       init_step=0.015, max_iter=250_000):
    a = initial_a.copy()
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    for it in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)
        a_proposed = symmetrize(a_proposed)
        cur_sc = compute_score(a_proposed)
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()
        if it % 50000 == 0:
            print(f"    Random walk iter={it} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc
```

```python
def generate_beta_shape(n, alpha, beta, eps):
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total
def generate_beta_mixture(n, alpha1, beta1, eps1, alpha2, beta2, eps2, w):
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum
def arcsine_initial(n):
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals
```

Now upsample function:

```python
def upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    # ensure nonnegative and sum to 1
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / np.sum(a_new)
    return a_new
```

Define ConvIneq class & SumEq.

```python
class ConvIneq:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        a = x[:self.n]
        t = x[self.n]
        conv = convolution_fft(a, a)
        return conv - t
    def jac(self, x):
        a = x[:self.n]
        n = self.n
        m = 2*n - 1
        J = np.zeros((m, n+1))
        # fill gradient for a variables
        for i in range(n):
            ks = np.arange(i, i+n)
            J[ks, i] = 2.0 * a[ks - i]
        J[:, n] = -1.0
        return J

class SumEq:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        a = x[:self.n]
        return np.sum(a) - 1.0
    def jac(self, x):
        n = self.n
        grad = np.zeros(n+1)
        grad[:n] = 1.0
        return grad
```

```python
def slsqp_optimize_small(n, time_deadline, seeds):
    """
    Perform SLSQP optimization for length n with given initial seeds (list of arrays).
    Returns best a (sum=1) and min t (max conv) found.
    """
    best_a = None
    best_t = np.inf
    best_score = np.inf
    for seed_idx, init_a in enumerate(seeds):
        if time.time() >= time_deadline:
            break
        init_a = project_to_simplex(init_a)
        t0 = max_convolution(init_a)
        x0 = np.append(init_a, t0)
        # Bounds: a_i >=0, no upper bound; t>=0
        bounds = [(0, None)] * n + [(0, None)]
        ineq = ConvIneq(n)
        eq = SumEq(n)
        max_iter = 3000  # moderate
        # We'll set options with maxiter and ftol.
        try:
            res = minimize(fun=lambda x: x[n],
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type':'ineq','fun':ineq,'jac':ineq.jac},
                                        {'type':'eq','fun':eq,'jac':eq.jac}],
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        except Exception as e:
            print(f"SLSQP failed for seed {seed_idx}: {e}")
            continue
        if not res.success:
            print(f"SLSQP seed {seed_idx} not successful: {res.message}")
        a_opt = res.x[:n]
        t_opt = res.x[n]
        # Normalize explicitly (might have small numerical diff)
        a_opt = np.maximum(a_opt, 0.0)
        a_opt = a_opt / np.sum(a_opt)
        score = 2.0 * n * t_opt  # official score guess
        if score < best_score:
            best_score = score
            best_a = a_opt
            best_t = t_opt
            print(f"SLSQP seed {seed_idx} improved score to {score:.6f}")
    return best_a, best_t, best_score
```

"""
Combined SLSQP + gradient‐based polishing for the max‑convolution / Σ² problem.

1.  Small‑scale (n=256) non‑convex optimisation:
    – Solve  min t  s.t.  (a*a)_k ≤ t ,   Σa = 1 , a≥0
      using SciPy’s SLSQP (quadratic constraints are supplied via Jacobians).
    – Several seeds (uniform, arcsine, random β‑shapes, mixtures) are tried
      and the best feasible vector is kept.

2.  Upsample the best small vector to a larger length (2048) by linear
    interpolation and re‑project onto the probability simplex.

3.  Refine the large vector with a fast three‑stage pipeline:
    – Adam on the L₂ “flattening’’ loss (makes the self‑convolution as flat
      as possible).
    – Adam on a smooth log‑sum‑exp surrogate of the max‑convolution
      (annealed temperature λ).
    – Projected sub‑gradient descent on the true max‑convolution
      (back‑tracking line‑search).
    – Cheap stochastic Gaussian walk on the simplex.

   All optimisation phases enforce symmetry (a ↔ a[::-1]) after every
   projection, because the optimal step function is known to be symmetric.
   The total runtime respects the 1000 s budget and all intermediate steps
   check the remaining time.

If SciPy is unavailable, the SLSQP stage is skipped and the algorithm falls
back to the purely gradient‑based pipeline (still guaranteeing a valid
return).  A uniform 256‑vector is used as a final safety net.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate score: 2·n·max(conv(a,a)) (sum a is assumed to be 1)."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1} (Duchi et al. 2008)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry a_i = a_{n‑1‑i}."""
    return (v + v[::-1]) / 2.0

# Shape generators (beta families & arcsine)

def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals

def flatten_gradient(a: np.ndarray):
    """
    Gradient of ‖conv(a,a) – avg‖₂² where avg = 1/(2n‑1) and Σa = 1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # grad = 4·(a ⊛ diff_rev)   (⊛ = convolution)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)          # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def smooth_gradient(a: np.ndarray, lam: float):
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate value B).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights
    B = max_c + np.log(sum_exp) / lam

    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B

def adam_flatten(initial_a: np.ndarray,
                 time_deadline: float,
                 max_iter: int = 150_000,
                 lr0: float = 0.25) -> (np.ndarray, float):
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        grad, loss = flatten_gradient(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_to_simplex(a_new)
        a_new = symmetrize(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 20000 == 0:
            print(f"    Adam‑flatten iter={it:,d} loss={loss:.3e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}")

    return best_a, best_sc


def adam_max(initial_a: np.ndarray,
             time_deadline: float,
             max_iter: int = 200_000,
             lr0: float = 0.12,
             lam_start: float = 1.0,
             lam_end: float = 500.0,
             noise_every: int = 1000,
             noise_scale: float = 4e-5) -> (np.ndarray, float):
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)
        a_new = symmetrize(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam‑max iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} lam={lam:.1f}")

    return best_a, best_sc


def subgradient_optimize(initial_a: np.ndarray,
                         time_deadline: float,
                         step0: float = 0.3,
                         max_iter: int = 200_000) -> (np.ndarray, float):
    a = symmetrize(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            a_new = symmetrize(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def random_local_search(initial_a: np.ndarray,
                        time_deadline: float,
                        init_step: float = 0.015,
                        max_iter: int = 250_000) -> (np.ndarray, float):
    a = symmetrize(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)
        a_prop = symmetrize(a_prop)

        cur_sc = compute_score(a_prop)

        # accept improvements, otherwise a tiny uphill chance
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50000 == 0:
            print(f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation and re‑normalisation to the simplex."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= np.sum(a_new)
    return a_new

def objective_t(x, n):
    """Objective for SLSQP: minimise the bound t (the max‑convolution)."""
    return x[n]


class ConvIneq:
    """Inequality: (a*a)_k ≤ t for all k."""

    def __init__(self, n: int):
        self.n = n

    def __call__(self, x):
        a = x[: self.n]
        t = x[self.n]
        conv = convolution_fft(a, a)
        return conv - t

    def jac(self, x):
        a = x[: self.n]
        n = self.n
        m = 2 * n - 1
        J = np.zeros((m, n + 1))
        for i in range(n):
            ks = np.arange(i, i + n)         # all k where k-i is in range
            J[ks, i] = 2.0 * a[ks - i]
        J[:, n] = -1.0
        return J


class SumEq:
    """Equality Σa = 1."""

    def __init__(self, n: int):
        self.n = n

    def __call__(self, x):
        a = x[: self.n]
        return np.sum(a) - 1.0

    def jac(self, x):
        n = self.n
        grad = np.zeros(n + 1)
        grad[:n] = 1.0
        return grad


def slsqp_optimize_small(n: int,
                         time_deadline: float,
                         seeds: List[np.ndarray]) -> (np.ndarray, float, float):
    """
    Run a few SLSQP starts on length n and return the best (a, t, official_score).
    """
    best_a = None
    best_t = np.inf
    best_score = np.inf

    for idx, init_a in enumerate(seeds):
        if time.time() >= time_deadline:
            break
        a0 = project_to_simplex(init_a)
        t0 = max_convolution(a0)
        x0 = np.append(a0, t0)

        bounds = [(0, None)] * n + [(0, None)]
        ineq = ConvIneq(n)
        eq = SumEq(n)

        # modest iteration budget – enough for a quick refinement
        max_iter = 3000
        try:
            res = minimize(fun=objective_t,
                           x0=x0,
                           args=(n,),
                           method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'ineq', 'fun': ineq, 'jac': ineq.jac},
                                        {'type': 'eq',   'fun': eq,  'jac': eq.jac}],
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        except Exception as e:                     # pragma: no cover
            print(f"  SLSQP failed (seed {idx}): {e}")
            continue

        if not res.success:
            print(f"  SLSQP seed {idx} not successful: {res.message}")

        a_opt = res.x[:n]
        t_opt = res.x[n]

        # re‑normalise numerically
        a_opt = np.maximum(a_opt, 0.0)
        a_opt /= np.sum(a_opt)

        score = 2.0 * n * t_opt
        if score < best_score:
            best_score = score
            best_a = a_opt
            best_t = t_opt
            print(f"  SLSQP seed {idx} improved to score {score:.6f}")

    return best_a, best_t, best_score

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    Combines a tiny SLSQP global search on n=256 with a multi‑phase
    gradient‑based polishing on a larger length (2048).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    best_seq: List[float] = []
    best_val: float = float('inf')

    # ------------------------------------------------------------------
    # 0. Simple fallback (uniform 256‑vector)
    # ------------------------------------------------------------------
    fallback = [1.0] * 256
    try:
        fallback_val = evaluate_sequence(fallback)
    except Exception:                      # pragma: no cover
        fallback_val = float('inf')
    best_seq = fallback
    best_val = fallback_val
    print(f"Uniform fallback score = {fallback_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Use any previously supplied construction if it exists
    # ------------------------------------------------------------------
    try:
        seq1 = [float(x) for x in height_sequence_1]   # type: ignore
        val1 = evaluate_sequence(seq1)
        print(f"height_sequence_1 score = {val1:.6f}")
        if val1 < best_val:
            best_val = val1
            best_seq = seq1
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Small‑scale SLSQP optimisation (n=256)
    # ------------------------------------------------------------------
    n_small = 256
    seeds_small = []

    # deterministic starters
    seeds_small.append(np.full(n_small, 1.0 / n_small))
    seeds_small.append(arcsine_initial(n_small))

    # a few random beta shapes
    for _ in range(6):
        alpha = rng.uniform(0.1, 4.0)
        beta = rng.uniform(0.1, 4.0)
        eps = rng.uniform(1e-8, 0.1)
        seeds_small.append(generate_beta_shape(n_small, alpha, beta, eps))

    # a few random mixtures
    for _ in range(6):
        w = rng.random()
        a1 = generate_beta_shape(n_small,
                                 rng.uniform(0.1, 4.0),
                                 rng.uniform(0.1, 4.0),
                                 rng.uniform(1e-8, 0.1))
        a2 = generate_beta_shape(n_small,
                                 rng.uniform(0.1, 4.0),
                                 rng.uniform(0.1, 4.0),
                                 rng.uniform(1e-8, 0.1))
        seeds_small.append(w * a1 + (1 - w) * a2)

    if SCIPY_AVAILABLE:
        print("\n=== SLSQP optimisation on n=256 ===")
        small_a, small_t, small_score = slsqp_optimize_small(
            n_small, deadline, seeds_small
        )
        if small_a is not None:
            print(f"  SLSQP best score (fast estimate) = {small_score:.6f}")
            # evaluate officially
            try:
                official_small = evaluate_sequence(small_a.tolist())
                print(f"  Official score after SLSQP (256) = {official_small:.6f}")
                if official_small < best_val:
                    best_val = official_small
                    best_seq = small_a.tolist()
            except Exception:
                pass
    else:
        print("SciPy not available – skipping SLSQP stage.")

    # ------------------------------------------------------------------
    # 3. Upsample to a larger length and polish
    # ------------------------------------------------------------------
    target_lengths = [2048, 1024]   # try the larger one first
    for target_n in target_lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Polishing length {target_n} ===")

        # upsample the current best vector
        base = np.array(best_seq, dtype=float)
        if base.shape[0] != target_n:
            base = upsample(base, target_n)
        base = project_to_simplex(base)

        # ---- flattening phase ----
        remaining = deadline - time.time()
        if remaining < 20.0:
            break
        max_iter_flat = int(min(80_000, remaining * 30))
        print(f"  Flattening (max_iter={max_iter_flat}) …")
        a_flat, _ = adam_flatten(base, deadline, max_iter=max_iter_flat, lr0=0.30)

        # ---- smooth‑max Adam phase ----
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_max = int(min(120_000, remaining * 30))
            print(f"  Smooth‑max Adam (max_iter={max_iter_max}) …")
            a_max, _ = adam_max(a_flat, deadline,
                                max_iter=max_iter_max,
                                lr0=0.12,
                                lam_start=1.0,
                                lam_end=800.0,
                                noise_every=1300,
                                noise_scale=5e-5)
        else:
            a_max = a_flat

        # ---- sub‑gradient descent phase ----
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(60_000, remaining * 30))
            print(f"  Sub‑gradient descent (max_iter={max_iter_sub}) …")
            a_sub, _ = subgradient_optimize(a_max, deadline,
                                            step0=0.28,
                                            max_iter=max_iter_sub)
        else:
            a_sub = a_max

        # ---- stochastic local walk ----
        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            max_iter_walk = int(min(100_000, remaining * 30))
            print(f"  Random walk (max_iter={max_iter_walk}) …")
            a_final, _ = random_local_search(a_sub, deadline,
                                             init_step=0.017,
                                             max_iter=max_iter_walk)
        else:
            a_final = a_sub

        # official evaluation
        try:
            val = evaluate_sequence(a_final.tolist())
            print(f"  Official score for length {target_n} = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_final.tolist()
                print(f"  *** New global best ({target_n}) = {best_val:.6f} ***")
        except Exception as e:                   # pragma: no cover
            print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 4. Global polishing of the overall best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # Flattening (long run)
        remaining = deadline - time.time()
        max_iter_flat = int(min(180_000, remaining * 20))
        print(f"  Global flatten (max_iter={max_iter_flat}) …")
        a, _ = adam_flatten(a, deadline, max_iter=max_iter_flat, lr0=0.35)

        # High‑temperature Adam on smooth max
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_max = int(min(300_000, remaining * 25))
            print(f"  Global Adam‑max (max_iter={max_iter_max}) …")
            a, _ = adam_max(a, deadline,
                            max_iter=max_iter_max,
                            lr0=0.10,
                            lam_start=1.0,
                            lam_end=1000.0,
                            noise_every=1500,
                            noise_scale=4e-5)

        # Sub‑gradient descent
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(150_000, remaining * 25))
            print(f"  Global sub‑gradient (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(a, deadline,
                                        step0=0.22,
                                        max_iter=max_iter_sub)

        # Final random walk
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_walk = int(min(250_000, remaining * 30))
            print(f"  Global random walk (max_iter={max_iter_walk}) …")
            a, _ = random_local_search(a, deadline,
                                       init_step=0.012,
                                       max_iter=max_iter_walk)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5. Guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = fallback
        best_val = fallback_val
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
