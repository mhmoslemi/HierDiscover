def exgrad_softmax(a0, deadline):
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = 0.7
    eps_factor_initial = 0.5
    eps_decay = 0.99
    iteration = 0
    while True:
        if time.time() >= deadline: break
        iteration += 1
        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        eps = maxc * max(eps_factor_initial * (eps_decay ** (iteration - 1)), 0.01)
        # stable softmax
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)
        # gradient
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[2*n - 2 - np.arange(n)]

        # exponentiated update
        # compute log a
        log_a = np.log(a + 1e-30)  # avoid log(0)
        log_a -= eta * grad
        a = np.exp(log_a)
        # reproject: simplex normalization
        a_sum = np.sum(a)
        if a_sum == 0:
            # fallback to uniform
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum
        # record best
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        # decay eta slowly
        eta = max(eta * 0.999, 0.02)  # small min step
        # occasional jitter
        if iteration % 500 == 0:
            a += np.random.randn(n) * 0.001
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    return best_a, best_sc

def hill_climb_anneal(seq, deadline, max_steps=200000):
    a = np.array(seq, dtype=float)
    best = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    # temperature schedule: start T0 = 0.1 maybe; reduce linearly over steps
    start_T = 0.1
    end_T = 0.001
    step = 0
    while step < max_steps and time.time() < deadline:
        frac = step / max_steps
        T = start_T * (1 - frac) + end_T * frac
        # propose move
        i, j = np.random.randint(n, size=2)
        if np.random.rand() < 0.5:
            # swap
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer mass
            cand = best.copy()
            if cand[i] > 0:
                delta = np.random.rand() * cand[i] * 0.5  # random fraction up to 0.5
                cand[i] -= delta
                cand[j] += delta
        # project to simplex (ensures sum=1)
        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)
        delta_sc = cur_sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            # accept
            best = cand
            if cur_sc < best_sc:
                best_sc = cur_sc
        step += 1
    return best.tolist(), best_sc

"""
Advanced search for a low‑score step function.

The algorithm combines several complementary ideas:

1. **Diverse random starts** – dense Dirichlet vectors, sparse equal‑mass
   vectors and clustered (“clumped”) vectors, all kept on the probability
   simplex (sum = 1).

2. **Exponentiated‑gradient descent on a smooth soft‑max surrogate** of
   the max‑convolution.  The surrogate is
        f_ε(a) = ε·log∑_k exp(conv_k/ε)
   whose gradient can be obtained with a single FFT.  The update is
   multiplicative (mirror‑descent on the simplex) which respects positivity
   without an explicit projection.

3. **Exact sub‑gradient refinement** – classic sub‑gradient descent on the
   true objective 2·n·max(a⊛a) to polish the solution obtained from the
   smooth phase.

4. **Simulated‑annealing hill‑climbing** – stochastic swaps / mass
   transfers are accepted with probability exp(−Δ/ T) where the temperature
   T is slowly cooled.  This allows occasional uphill moves to escape local
   minima.

5. **Optional symmetry averaging** – (a + a[::-1])/2 often flattens the
   autocorrelation further.

All stages respect the 1000 s budget; the routine stops early if the
deadline is near and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT‑based convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a ⊛ b using a real FFT.
    Length of the result is len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors already normalised
    to sum(a)=1.  Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex (non‑negative,
    sum = 1).  Standard O(n log n) algorithm.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_vector(n: int) -> np.ndarray:
    """
    Random probability vector where a random subset (density 10‑90 %) gets
    equal mass.  Guarantees at least two non‑zero entries.
    """
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dense_vector(n: int) -> np.ndarray:
    """Draw a Dirichlet(1,…,1) – a uniform random point on the simplex."""
    a = np.random.dirichlet(np.ones(n))
    return a.astype(float)


def random_clumped_vector(n: int, max_cluster: int = 20) -> np.ndarray:
    """
    A few adjacent positions receive most of the probability mass.
    The cluster size is random up to ``max_cluster``.
    """
    k = np.random.randint(2, max_cluster + 1)
    start = np.random.randint(0, n - k + 1)
    a = np.zeros(n, dtype=float)
    a[start:start + k] = np.random.rand(k)
    a /= np.sum(a)
    return a


# ----------------------------------------------------------------------
# 1. Exponentiated‑gradient descent on a smooth soft‑max surrogate
# ----------------------------------------------------------------------


def exgrad_softmax_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.70,
    eps_factor_initial: float = 0.50,
    eps_decay: float = 0.99,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution.

    The surrogate is  f_ε(a) = ε·log Σ_k exp(conv_k / ε)
    whose gradient is  g = 2·(w_rev ⊛ a)  where  w = softmax(conv/ε).

    The update is multiplicative (mirror descent on the simplex):
          log a ← log a – η·g
    followed by normalisation to enforce Σ a = 1.
    """
    a = a0.astype(float).copy()
    n = len(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    eta = eta0
    iteration = 0

    while iteration < max_iter and time.time() < deadline:
        iteration += 1

        # ---------- soft‑max surrogate ----------
        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        eps = maxc * max(eps_factor_initial * (eps_decay ** (iteration - 1)), 0.01)

        # stable soft‑max
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)  # Σ w = 1

        # ---------- gradient ----------
        w_rev = w[::-1]                     # reverse for cross‑correlation
        conv_rev = convolution_fft(a, w_rev)   # length = 3n‑2
        grad = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]

        # ---------- exponentiated (mirror) update ----------
        # work in log‑space to keep positivity
        log_a = np.log(a + 1e-30)           # avoid log(0)
        log_a -= eta * grad
        a = np.exp(log_a)

        # renormalise onto the simplex
        s = np.sum(a)
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # keep the best exact score seen so far
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # schedule for η (slowly decaying, never too small)
        eta = max(eta * 0.999, 0.02)

        # occasional tiny random jitter to help escape flat plateaux
        if iteration % 500 == 0:
            a += np.random.randn(n) * 0.001
            a = np.maximum(a, 0.0)
            a /= np.sum(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 2. Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the exact objective
        F(a) = 2·n·max(a⊛a) .
    The sub‑gradient is taken at the shift attaining the current maximum.
    """
    a = init_a.astype(float).copy()
    n = len(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k-i} on the overlapping region
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional random shake to escape shallow plateaux
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc

def hill_climb_anneal(

    max_steps: int = 150_000,
    start_temp: float = 0.12,
    end_temp: float = 0.001,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber with a simple linear temperature schedule.
    Proposes swaps or mass transfers; moves that improve the exact score
    are always accepted, worsening moves are accepted with probability
    exp(−Δ/ T).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    step = 0
    while step < max_steps and time.time() < deadline:
        frac = step / max_steps
        T = start_temp * (1.0 - frac) + end_temp * frac

        cand = best_a.copy()

        # pick move type
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a random fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0:
                delta = cand[i] * np.random.rand() * 0.5  # up to 50 % of the source mass
                cand[i] -= delta
                cand[j] += delta

        # projection keeps the vector on the simplex
        cand = project_to_simplex(cand)

        cur_sc = compute_score(cand)
        delta_sc = cur_sc - best_sc

        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(T, 1e-12)):
            # accept
            best_a = cand
            if cur_sc < best_sc:
                best_sc = cur_sc

        step += 1

    return best_a.tolist(), best_sc


# ----------------------------------------------------------------------
# 4. Symmetry averaging (cheap post‑processing)
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a + reverse(a))/2 and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (~1000 s) and returns the best sequence found.

    The search follows the outline described in the module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a known construction (if the global exists)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Helper: quick cheap score (already normalised)
    # ------------------------------------------------------------------
    def cheap_score(v):
        return compute_score(v)

    # ------------------------------------------------------------------
    # Parameter space
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    max_outer_iters = 120  # will be cut by the deadline

    for outer in range(max_outer_iters):
        if time.time() >= deadline - 5.0:
            break

        # ---- choose length and initialisation ----
        n = int(np.random.choice(lengths))
        init_mode = np.random.choice(["sparse", "dense", "clumped", "sym"])
        if init_mode == "sparse":
            a0 = random_sparse_vector(n)
        elif init_mode == "dense":
            a0 = random_dense_vector(n)
        elif init_mode == "clumped":
            a0 = random_clumped_vector(n)
        else:  # symmetric dense start
            a0 = random_dense_vector(n)
            a0 = (a0 + a0[::-1]) / 2.0

        cur_sc = cheap_score(a0)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a0.tolist()
            print(f"[{outer}] init n={n:5d}, mode={init_mode:8s}, score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – exponentiated‑gradient on smooth surrogate
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        gd_deadline = deadline - 3.0
        a_gd, sc_gd = exgrad_softmax_descent(
            a0,
            deadline=gd_deadline,
            max_iter=200_000,
            eta0=0.70,
            eps_factor_initial=0.50,
            eps_decay=0.99,
        )
        if sc_gd < best_score:
            best_score = sc_gd
            best_seq = a_gd.tolist()
            print(f"[{outer}] after exp‑GD: n={n:5d}, score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – exact sub‑gradient refinement
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 12.0:
            sub_deadline = deadline - 2.0
            a_sub, sc_sub = subgrad_optimize(
                a_gd,
                deadline=sub_deadline,
                max_iter=250_000,
                step0=0.07,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[{outer}] after sub‑grad: n={n:5d}, score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – hill‑climbing with simulated annealing + symmetry
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 8.0:
            hill_deadline = deadline - 1.0
            polished_seq, polished_sc = hill_climb_anneal(
                best_seq,
                deadline=hill_deadline,
                max_steps=150_000,
                start_temp=0.12,
                end_temp=0.001,
            )
            if polished_sc < best_score:
                best_score = polished_sc
                best_seq = polished_seq
                print(f"[{outer}] hill‑climb improved: score={best_score:.6f}")

            # optional symmetry averaging (very cheap)
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sym_sc = cheap_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym.tolist()
                print(f"[{outer}] symmetry average improved: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a uniform vector if nothing was discovered
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
