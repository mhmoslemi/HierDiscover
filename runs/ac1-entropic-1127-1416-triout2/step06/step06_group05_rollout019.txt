"""
Propose a non‑negative step sequence with a low official score.

The search proceeds in three phases:

1. **Broad random sampling** –  many low‑dimensional families of shapes
   (beta, mixtures, exponential, power‑law, linear ramp, block‑wise)
   are sampled for a chosen length (default 2048).  
   The 20 best candidates according to the fast surrogate
   `2·n·max(conv(a,a))` are kept.

2. **Local refinement** – every survivor is refined by a long Adam run
   on a log‑sum‑exp surrogate of the max‑convolution (temperature is
   annealed up to a very large λ).  After Adam a short projected
   sub‑gradient descent guarantees that the true objective never
   increases.

3. **Global polishing** – the current overall best vector is polished
   again with Adam, sub‑gradient descent and a tiny random walk.
   The routine respects the 1000 s budget and always returns a non‑empty
   sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for all n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator is 1 and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α-1}·(n-1-i+ε)^{β-1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density (the shape used in the original baseline)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """Strictly decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-lam * i)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def powerlaw_shape(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """Power‑law a_i ∝ (i+offset+1)^{-p}."""
    i = np.arange(n, dtype=float)
    vals = (i + offset + 1.0) ** (-p)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def linear_ramp_shape(n: int, slope: float) -> np.ndarray:
    """Linear ramp a_i = max(0, slope·(n-1-i)).  slope>0 makes a decreasing shape."""
    i = np.arange(n, dtype=float)
    vals = np.maximum(0.0, slope * (n - 1 - i))
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def block_constant_shape(n: int, rng: np.random.Generator,
                         max_blocks: int = 5) -> np.ndarray:
    """Uniform mass on a random set of disjoint blocks."""
    a = np.zeros(n, dtype=float)
    num_blocks = rng.integers(1, max_blocks + 1)
    # avoid overlap by sampling start positions in order
    starts = rng.choice(n, size=num_blocks, replace=False)
    starts.sort()
    for st in starts:
        length = rng.integers(1, n // max_blocks + 1)
        end = min(n, st + length)
        a[st:end] = 1.0
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    a /= a.sum()
    return a


def dirichlet_shape(n: int, rng: np.random.Generator,
                    concentration: float = 0.1) -> np.ndarray:
    """Sparse Dirichlet draw (all entries ≥0, sum = 1)."""
    g = rng.gamma(concentration, 1.0, size=n)
    if g.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return g / g.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (for Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                       # soft‑max weights
    B = max_c + np.log(sum_exp) / lam            # smooth approximation of the max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 800.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector found (projected on the simplex) together with
    its fast score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # quadratic annealing of the temperature λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected) with back‑tracking
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 120_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the true objective.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step size that decays like 1/√t
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc

        "linear_ramp",
        "block",
        "dirichlet",
    ]

    max_samples = 25000  # total number of random candidates (adjusted to time)
    for _ in range(max_samples):
        if time.time() >= deadline - 120.0:
            break

        choice = rng.choice(families)
        if choice == "beta":
            alpha = np.exp(rng.uniform(-7, 2))   # log‑uniform in [~1e‑3, ~7.4]
            beta = np.exp(rng.uniform(-7, 2))
            eps = rng.uniform(1e-9, 0.02 * n)
            cand = generate_beta_shape(n, alpha, beta, eps)
        elif choice == "beta_mix":
            w = rng.random()
            a1 = np.exp(rng.uniform(-7, 2))
            b1 = np.exp(rng.uniform(-7, 2))
            e1 = rng.uniform(1e-9, 0.02 * n)
            a2 = np.exp(rng.uniform(-7, 2))
            b2 = np.exp(rng.uniform(-7, 2))
            e2 = rng.uniform(1e-9, 0.02 * n)
            cand = generate_beta_mixture(n, a1, b1, e1, a2, b2, e2, w)
        elif choice == "exponential":
            lam = rng.uniform(0.0, 5.0)          # λ≈0 gives uniform, larger → steeper decay
            cand = exponential_shape(n, lam)
        elif choice == "powerlaw":
            p = rng.uniform(0.0, 5.0)
            offset = rng.uniform(0.0, 5.0)
            cand = powerlaw_shape(n, p, offset)
        elif choice == "linear_ramp":
            slope = rng.uniform(0.0, 3.0 / n)
            cand = linear_ramp_shape(n, slope)
        elif choice == "block":
            cand = block_constant_shape(n, rng, max_blocks=4)
        else:  # dirichlet
            conc = rng.uniform(0.01, 1.0)
            cand = dirichlet_shape(n, rng, concentration=conc)

        sc = fast_score(cand)
        if sc < best_val:
            # quick official check – inexpensive compared with a full optimisation run
            val = official_score(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.tolist()
                print(f"New raw global best (score={val:.6f})")
        # keep in survivor list if among the current best
        if len(survivors) < top_k or sc < survivors[-1][1]:
            survivors.append((cand, sc))
            survivors = sorted(survivors, key=lambda x: x[1])[:top_k]

    print(f"Sampling finished – {len(survivors)} survivors kept (best fast score {survivors[0][1]:.6f})")

    # ------------------------------------------------------------------
    # 2. Local refinement of survivors
    # ------------------------------------------------------------------
    for idx, (cand, sc) in enumerate(survivors):
        if time.time() >= deadline - 50.0:
            break
        remaining = deadline - time.time()
        # give each survivor a time slice proportional to the remaining budget
        slice_deadline = time.time() + min(remaining * 0.6, 200.0)

        print(f"\n--- Refining survivor {idx + 1}/{len(survivors)} (fast score {sc:.6f}) ---")
        a_opt, _ = adam_optimize(
            init_a=cand,
            deadline=slice_deadline,
            max_iter=200_000,
            lr0=0.11,
            lam_start=1.0,
            lam_end=800.0,
            noise_every=1200,
            noise_scale=3e-5,
        )
        if time.time() < slice_deadline - 2.0:
            a_opt, _ = subgradient_optimize(
                init_a=a_opt,
                deadline=slice_deadline,
                step0=0.28,
                max_iter=80_000,
            )
        # official evaluation
        val = official_score(a_opt)
        print(f"  Official score after polishing = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_opt.tolist()
            print(f"*** New global best from survivor {idx + 1} = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3. Global polishing of the best vector found so far
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the overall best sequence ===")
        a = np.array(best_seq, dtype=float)

        # A – longer Adam run with a very high temperature
        a, _ = adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=500_000,
            lr0=0.09,
            lam_start=1.0,
            lam_end=1200.0,
            noise_every=1300,
            noise_scale=2e-5,
        )
        # B – exact sub‑gradient descent
        if time.time() < deadline - 3.0:
            a, _ = subgradient_optimize(
                init_a=a,
                deadline=deadline,
                step0=0.22,
                max_iter=150_000,
            )
        # C – final cheap random walk (occasionally uphill)
        if time.time() < deadline - 1.0:
            # reuse the earlier random walk implementation – lightweight
            a, _ = random_local_search(
                init_a=a,
                deadline=deadline,
                init_step=0.011,
                max_iter=200_000,
            )

        final_val = official_score(a)
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = official_score(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
