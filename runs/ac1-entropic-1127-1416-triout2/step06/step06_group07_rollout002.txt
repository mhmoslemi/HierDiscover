The goal is to find a non‑negative vector `a` (the “step heights”) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max\limits_k (a*a)[k]}{(\sum_i a_i)^2},
\]

where `*` denotes the discrete linear convolution and `n = len(a)`.
Because the score is homogeneous, we may restrict ourselves to the probability
simplex  

\[
\sum_i a_i = 1,\qquad a_i\ge 0.
\]

For a vector on the simplex the score simplifies to `2·n·max(a*a)`.  
The problem is non‑convex; we therefore use a **three‑phase hybrid search**:

1. **Smooth soft‑max optimisation** – exponentiated‑gradient descent on a
   smooth approximation of the `max` (soft‑max with an increasing temperature
   β).  Several random Dirichlet restarts give diverse starting points.
2. **Maximum‑gradient descent** – a direct sub‑gradient step that reduces the
   current maximal convolution entry.  At iteration *t* we move the
   distribution away from the “peak” index `k_max` using the exact sub‑gradient  
   \[
   g_i =\begin{cases}
        2\,a_{k_{\max}-i}&0\le k_{\max}-i<n,\\[2pt]
        0 &\text{otherwise}.
        \end{cases}
   \]
   After each step we project back onto the simplex.
3. **Targeted hill‑climbing** – stochastic swaps / mass‑transfers that
   locally improve the exact objective.  The step size adapts to recent
   successes.

All phases stop automatically when the time budget (≈ 1000 s) is exhausted.
The best vector found is returned.

The implementation uses only NumPy (FFT for fast convolutions) and pure‑Python
logic; no external solvers are required.

```python
"""
Hybrid optimisation for the L∞ self‑convolution problem.

Three phases:
1. Random Dirichlet restarts + exponentiated‑gradient (soft‑max) descent.
2. Direct sub‑gradient descent on the true max entry.
3. Stochastic hill‑climb (swap / mass‑transfer) with adaptive step size.

All updates stay on the probability simplex via Euclidean projection.
The code respects a 1000 s CPU budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution, real‑valued)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # corner case: all non‑positive – return uniform vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact score for probability vectors (fast variant)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute the exact value returned by `evaluate_sequence` when Σ a = 1:
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Soft‑max (smooth) optimisation – exponentiated‑gradient
# ----------------------------------------------------------------------


def smooth_optimize(
    a0: np.ndarray,
    time_deadline: float,
    iters: int = 200_000,
    beta0: float = 0.5,
    beta_max: float = 60.0,
    eta0: float = 0.4,
    report_every: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the soft‑max approximation of max(a*a).

    Returns the best vector (projected on the simplex) and its exact score.
    """
    n = a0.shape[0]
    L = next_pow2(3 * n - 2)  # length sufficient for all FFTs

    a = a0.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    for t in range(1, iters + 1):
        if time.time() >= time_deadline - 0.8:
            break

        # ----- a*a ----------------------------------------------------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = np.max(conv)

        # ----- soft‑max weights ---------------------------------------
        beta = min(beta_max, beta0 * np.sqrt(t))
        exp_terms = np.exp(beta * (conv - max_c))  # stabilised
        w = exp_terms / np.sum(exp_terms)

        # ----- gradient of the smooth objective -----------------------
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- exponentiated (mirror) update -------------------------
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        # ----- occasional reporting ------------------------------------
        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  smooth iter {t:8d} – new best score {cur_val:.6f}")

    # final check
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Direct sub‑gradient descent on the true max entry
# ----------------------------------------------------------------------


def max_gradient_optimize(
    a0: np.ndarray,
    time_deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that directly reduces the maximal convolution entry.

    At each iteration the index `k` attaining the current maximum of a*a is
    identified, the exact sub‑gradient (2·a_{k‑i}) is formed, a step is taken,
    and the result is projected back onto the simplex.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline - 0.8:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()
        k = int(conv.argmax())

        # build sub‑gradient g_i = 2·a_{k‑i} for valid indices
        g = np.zeros_like(a)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        g[idx] = 2.0 * a[k - idx]

        # step size decreasing like 1/√t
        eta = eta0 / np.sqrt(t)
        a = a - eta * g
        a = project_to_simplex(a)

        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # success → slightly larger steps next time
            eta0 = min(eta0 * 1.05, 2.0)

    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climber (swap / mass‑transfer moves) with adaptive step size
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 500_000,
) -> tuple[list[float], float]:
    """
    Simple stochastic hill‑climber that swaps two entries or moves a fraction
    of mass from one position to another.  The true objective `evaluate_sequence`
    (via `fast_score`) is used for acceptance.

    Returns the best sequence (as a plain list) and its score.
    """
    a = np.asarray(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = fast_score(best_a)

    step_scale = 0.07  # start with a modest fraction of the source mass

    steps = 0
    while steps < max_steps and time.time() < time_deadline - 0.2:
        cand = best_a.copy()

        if np.random.rand() < 0.55:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 1e-12:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cur_score = fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)    # success → larger steps
        else:
            step_scale *= 0.96                         # failure → smaller steps

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.

    The search consists of:
        * several random Dirichlet restarts with soft‑max gradient descent,
        * an aggressive sub‑gradient phase that directly attacks the current
          maximal convolution entry,
        * a final hill‑climb polishing.
    All work respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Optional fallback (previous best construction, if provided)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val = fb_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Choose a length – 2048 gives a good trade‑off
    # --------------------------------------------------------------
    n = 2048
    if time.time() + 5 > deadline:
        n = 1024
    print(f"Using sequence length n = {n}")

    # --------------------------------------------------------------
    # 2️⃣  Random restarts with smooth optimisation (exponentiated‑grad)
    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 30.0:  # leave room for later phases
        # random Dirichlet start – α sampled to obtain both dense & sparse
        alpha = np.random.uniform(0.2, 2.0)
        a0 = np.random.gamma(alpha, 1.0, size=n)
        a0 = a0 / a0.sum()                     # simplex projection

        init_score = fast_score(a0)
        if init_score < best_val:
            best_val = init_score
            best_seq = a0.tolist()
            print(f"Restart {restart}: init score = {init_score:.6f}")

        # smooth optimisation – length of the inner loop is capped by remaining time
        refined_a, refined_score = smooth_optimize(
            a0,
            time_deadline=deadline - 20.0,
            iters=200_000,
            beta0=0.5,
            beta_max=60.0,
            eta0=0.4,
            report_every=25_000,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Restart {restart}: refined best = {refined_score:.6f}")

        restart += 1

    # --------------------------------------------------------------
    # 3️⃣  Direct max‑gradient descent (fine‑tuning)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15.0:
        print("\nStarting max‑gradient fine‑tuning")
        a_start = np.asarray(best_seq, dtype=float)
        tuned_a, tuned_score = max_gradient_optimize(
            a_start,
            time_deadline=deadline - 8.0,
            max_iters=400_000,
            eta0=0.5,
        )
        if tuned_score < best_val:
            best_val = tuned_score
            best_seq = tuned_a.tolist()
            print(f"Max‑gradient improvement to {tuned_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final hill‑climb polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        print("\nApplying final hill‑climb polishing")
        polished_seq, polished_val = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=800_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {polished_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Safeguard – tiny uniform vector if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 6️⃣  Final reporting
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq
```

This implementation respects all constraints, explores the search space with complementary optimisation techniques, and is tuned to extract the last few tenths of a point needed to beat the target score.
