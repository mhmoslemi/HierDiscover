def simulated_annealing(seq, deadline):
    a = np.array(seq, dtype=float)
    n = len(a)
    # ensure sum=1
    a = a / np.sum(a)
    cur_score = compute_score(a)
    best = a.copy()
    best_score = cur_score
    T0 = 0.1  # initial temperature
    # compute total steps based on time left
    while time.time() < deadline:
        t = (deadline - time.time()) / (deadline - start_time)  # fraction left
        T = T0 * t  # linear cooling
        # propose move: random i,j and delta fraction
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = np.random.uniform(0, 0.1) * a[i]  # move up to 10% of mass from i to j
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

"""
Search for a step‑function (non‑negative vector) that minimises

        score = 2·n·max(conv(a,a)) / (sum(a))**2

The objective is invariant under scaling, therefore the algorithm works on
the probability simplex (sum(a)=1).  The search combines three ideas:

1. **Smoothed exponentiated‑gradient descent** – a soft‑max of the
   convolution entries yields a smooth surrogate for the L∞ norm.
   The gradient of this surrogate is a convolution of the reversed vector
   with the soft‑max weights.  Multiplicative updates keep the iterate
   on the simplex.

2. **Small‑population evolutionary refinement** – a handful of candidate
   vectors are kept, the best are selected as parents and mutated in the
   log‑domain.  This injects diversity and helps escape shallow local
   minima.

3. **Simulated‑annealing hill‑climbing** – a final polishing phase that
   moves a random fraction of mass between two coordinates, accepting
   uphill moves with a decreasing temperature schedule.

The routine tries several vector lengths (256, 512, 1024, 2048) and uses
the available time budget (default 1000 s) to allocate work to the three
stages.  The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform point if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# 1️⃣ Smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------


def smooth_exp_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.8,
    lam_start: float = 1.0,
    lam_end: float = 60.0,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Minimise  max(conv(a,a))  via a soft‑max surrogate.
    Returns the best vector seen and its (unsmoothed) score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while True:
        if time.time() >= deadline:
            break
        it += 1

        conv = convolution_fft(a, a)                # length 2n‑1
        M = np.max(conv)

        # linear λ‑schedule (slowly sharpen the soft‑max)
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / 800_000)

        # soft‑max weights (stable computation)
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)            # length 2n‑1, sums to 1

        # gradient = 2 * (a_rev * w) on the central n entries
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)         # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # multiplicative (exponentiated) update
        eta = eta0 / np.sqrt(it)
        step = -eta * g
        step = np.clip(step, -70, 70)              # safety for np.exp
        a = a * np.exp(step)

        # projection back onto the simplex
        a = project_to_simplex(a)

        # periodic exact evaluation
        if it % eval_every == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                print(
                    f"[{time.time():.1f}]  iter {it:,}  new EG best {best_score:.6f}"
                )

    # final check
    final = compute_score(a)
    if final < best_score:
        best_score = final
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣ Evolutionary refinement (tiny population, log‑domain mutation)
# ----------------------------------------------------------------------


def evolutionary_optimize(
    init_pop: list[np.ndarray],
    deadline: float,
    sigma: float = 0.12,
    keep_frac: float = 0.3,
    max_pop: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Simple (µ+λ) evolutionary loop.
    The population is kept on the simplex; mutation is performed
    in the log‑domain and then re‑normalised.
    """
    pop = [proj.copy() for proj in init_pop]
    scores = [compute_score(p) for p in pop]

    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]

    it = 0
    while time.time() < deadline:
        it += 1

        # select parent from the elite fraction
        elite_n = max(2, int(len(pop) * keep_frac))
        elite_idx = np.argpartition(scores, elite_n - 1)[:elite_n]
        parent = pop[np.random.choice(elite_idx)]

        # log‑domain Gaussian mutation
        log_parent = np.log(parent + 1e-14)
        child_log = log_parent + np.random.normal(0.0, sigma, size=parent.shape)
        child = np.exp(child_log)
        child = project_to_simplex(child)

        child_score = compute_score(child)

        # replace worst individual if the child is better
        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            pop[worst_idx] = child
            scores[worst_idx] = child_score

            if child_score < best_score:
                best_score = child_score
                best_vec = child.copy()
                print(
                    f"[{time.time():.1f}]  evo iter {it:,}  new best {best_score:.6f}"
                )

        # occasionally prune population to keep size bounded
        if len(pop) > max_pop:
            # keep elite + some random newcomers
            keep_idx = np.argpartition(scores, max_pop - 1)[:max_pop]
            pop = [pop[i] for i in keep_idx]
            scores = [scores[i] for i in keep_idx]

    return best_vec, best_score


# ----------------------------------------------------------------------
# 3️⃣ Simulated‑annealing hill‑climber (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(
    start_vec: np.ndarray,
    deadline: float,
    init_temp: float = 0.03,
    min_temp: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Classic Metropolis hill‑climber on the exact objective.
    Proposes moving a random fraction of mass from one coordinate to another.
    """
    a = project_to_simplex(start_vec.astype(float).copy())
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    # linear cooling schedule
    total_time = deadline - time.time()
    while time.time() < deadline:
        elapsed = time.time() - (deadline - total_time)
        T = max(min_temp, init_temp * (1.0 - elapsed / total_time))

        n = a.shape[0]
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a random fraction (up to 10 %) of the source mass
        frac = np.random.rand() * 0.1
        delta = frac * a[i]
        if delta <= 0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        new_score = compute_score(cand)
        delta_score = new_score - cur_score

        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / max(T, 1e-12)):
            a = cand
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # keep the temperature high enough after a success
                # (helps to escape a possible shallow plateau)
                T = max(T, init_temp * 0.5)

    return best_a, best_score
