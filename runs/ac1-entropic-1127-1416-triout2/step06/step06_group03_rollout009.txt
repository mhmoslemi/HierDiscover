"""
Propose a low‑value step sequence.

Algorithm
---------

1. **Small‑scale LP refinement** – work with a moderate length
   (n≈1024).  Starting from a few diverse seeds we repeatedly
   *solve a linear program* that maximises the total mass of a new
   candidate `b` under the convex constraints
   `conv(a, b) ≤ max(conv(a, a))`.  The solution is mixed with the
   current iterate (`a ← (1‑t)·a + t·b/∑b`) with a decreasing
   mixing factor `t`.  This is a deterministic implementation of the
   alternating scheme described in the statement and quickly drives
   down the max‑convolution value.

2. **Upsampling & smooth‑max Adam** – the best shape found for the
   small length is interpolated to a larger length (2048/4096) and
   refined with the log‑sum‑exp surrogate optimiser used previously.
   The soft‑max parametrisation guarantees positivity and the simplex
   constraint.

3. **Exact sub‑gradient polishing** – a short projected sub‑gradient
   descent (operating on the true `max(conv)`) finishes the search.

The whole routine respects a 1000 s wall‑clock budget, prints progress
and always returns a feasible non‑negative sequence.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, accurate for our sizes)."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score when Σa = 1.
    Returns 2·n·max(conv(a, a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all produce vectors that sum to 1)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_dirichlet_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    # random concentration parameters between 0.1 and 5
    conc = rng.uniform(0.1, 5.0, size=n)
    vec = rng.dirichlet(conc)
    return vec


# ----------------------------------------------------------------------
# Soft‑max and smooth‑max Adam optimiser (same as the previous code)
# ----------------------------------------------------------------------
def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 150.0,
    ramp_iters: int = 30000,
    lr: float = 0.05,
) -> np.ndarray:
    """
    Adam optimisation of the smooth (log‑sum‑exp) surrogate of max(conv).
    The iterate stays on the probability simplex via the soft‑max map.
    """
    import math

    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # temperature schedule (linear)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # log‑sum‑exp weights (smooth max)
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient wrt a : 2 * (a ⋆ w_rev) restricted to original length
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)          # length = 3n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]

        # chain rule for the soft‑max map
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    [Adam] it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final a on the simplex
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected onto simplex)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.08,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # indices where conv is (almost) maximal – sub‑gradient support
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2·(a ⋆ mask_rev) restricted to length n
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)          # diminishing stepsize
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    [SubGrad] it={it:,d} score={2.0 * n * mx:.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming based improvement for a fixed small length
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return the (2n‑1)×n matrix C such that C @ b = conv(a, b).
    """
    n = a.size
    L = 2 * n - 1
    C = np.zeros((L, n), dtype=float)
    for j in range(n):
        C[j : j + n, j] = a[:]          # shift the whole a‑vector
    return C


def lp_refine(
    a0: np.ndarray,
    max_iters: int,
    time_deadline: float,
    init_mix: float = 0.4,
) -> np.ndarray:
    """
    Alternate: solve   max Σ b_j   s.t.   conv(a, b) ≤ M ,  b ≥ 0,
    then mix the solution into a.  Repeatedly reduces the max‑convolution.
    """
    from scipy.optimize import linprog

    a = a0.copy()
    t_mix = init_mix
    n = a.size

    for it in range(max_iters):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        # Build linear constraints conv(a, b) ≤ M
        A = _build_conv_matrix(a)
        b_ub = np.full(2 * n - 1, M)

        # Linear programme: maximise Σ b_j  <=> minimise -Σ b_j
        c = -np.ones(n)

        res = linprog(
            c, A_ub=A, b_ub=b_ub, bounds=(0, None), method="highs", options={"presolve": True}
        )
        if not res.success:
            # If LP fails, abort the refinement loop.
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            break
        b_norm = b_opt / sum_b

        # Convex combination
        a = (1.0 - t_mix) * a + t_mix * b_norm
        t_mix = max(0.05, t_mix * 0.7)     # decay, but keep a small positive mixing

        # occasional progress report
        if (it + 1) % 2 == 0:
            cur_score = compute_score(a)
            print(f"    [LP] iter {it+1}/{max_iters} – score≈{cur_score:.6f}")

    return a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine follows three stages:
    1. small‑scale LP refinement,
    2. up‑sampling + smooth‑max Adam,
    3. exact sub‑gradient polishing.
    """
    import sys

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously discovered good sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"[Fallback] previous score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception as e:
            print(f"[Fallback] failed: {e}")

    # ------------------------------------------------------------------
    # Stage 1 – LP refinement on a modest length (n = 1024)
    # ------------------------------------------------------------------
    n_small = 1024
    seeds = []

    # deterministic diverse seeds
    seeds.append(arcsine_initial(n_small))
    seeds.append(np.full(n_small, 1.0 / n_small))
    for p in (0.5, 0.8, 1.0, 1.5):
        seeds.append(generate_power_shape(n_small, p, eps=1e-6))
    for _ in range(5):
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        eps = rng.uniform(0.0, 1.0)
        seeds.append(generate_beta_shape(n_small, a, b, eps))
    for _ in range(4):
        seeds.append(generate_dirichlet_shape(n_small, rng))

    # keep only the most promising few (fast surrogate)
    scored = [(compute_score(s), s) for s in seeds]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:5]]

    per_seed_time = max(2.0, (deadline - time.time()) / (len(top_seeds) * 5))

    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 10.0:
            break

        print(f"\n[Stage 1] Seed {idx}/{len(top_seeds)} – length {n_small}")
        # normalise each seed to Σ=1
        a = seed_vec / seed_vec.sum()

        # LP refinement (up to 6 alternating steps)
        seed_deadline = time.time() + per_seed_time
        a_refined = lp_refine(a, max_iters=6, time_deadline=seed_deadline, init_mix=0.4)

        # official evaluation of the refined candidate
        try:
            val = evaluate_sequence(a_refined.tolist())
            print(f"    Official score after LP = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_refined.tolist()
                print(f"    *** New global best (LP) = {best_val:.6f} ***")
        except Exception as e:
            print(f"    Evaluation failed (LP candidate): {e}")

    # ------------------------------------------------------------------
    # Stage 2 – Upsample + smooth‑max Adam optimisation
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        # Pick a larger length that still fits the remaining budget
        remaining = deadline - time.time()
        final_len = 4096 if remaining > 30.0 else 2048

        print(f"\n[Stage 2] Upsampling to length {final_len}")
        a_best = np.array(best_seq, dtype=float)
        # linear interpolation to the new grid
        src_x = np.linspace(0.0, 1.0, a_best.size)
        dst_x = np.linspace(0.0, 1.0, final_len)
        a_upsampled = np.interp(dst_x, src_x, a_best)
        a_upsampled = a_upsampled / a_upsampled.sum()   # enforce Σ=1

        # Adam optimisation with the remaining budget (keep a 15 % safety margin)
        adam_deadline = time.time() + remaining * 0.85
        eps = 1e-12
        z0 = np.log(a_upsampled + eps)
        a_opt = smooth_adam_optimize(
            z0,
            adam_deadline,
            n=final_len,
            beta0=5.0,
            beta_max=180.0,
            ramp_iters=25000,
            lr=0.05,
        )

        try:
            val = evaluate_sequence(a_opt.tolist())
            print(f"    Official score after Adam = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
                print(f"    *** New global best (Adam) = {best_val:.6f} ***")
        except Exception as e:
            print(f"    Evaluation failed (Adam candidate): {e}")

    # ------------------------------------------------------------------
    # Stage 3 – final exact sub‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\n[Stage 3] Exact sub‑gradient polishing")
        a_arr = np.array(best_seq, dtype=float)
        polish_deadline = deadline
        a_polished = subgradient_descent_exact(
            a_arr, polish_deadline, max_iter=120_000, step0=0.07
        )
        try:
            val = evaluate_sequence(a_polished.tolist())
            print(f"    Polished official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_polished.tolist()
                print(f"    *** New global best (polish) = {best_val:.6f} ***")
        except Exception as e:
            print(f"    Polishing evaluation failed: {e}")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty, feasible return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 512
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)
        print("[Guard] fallback uniform sequence used.")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    # Ensure the returned list is pure Python floats
    return [float(x) for x in best_seq]
