"""
LP‑refinement + gradient polishing for the step‑function minimisation.

The algorithm works in four phases:

1. **Random sparse start** – a short support with equal mass, projected
   onto the simplex (∑a=1).

2. **Linear‑programming refinement** – for the current sequence `f`
   compute `M = max(f*f)`.  Solve the LP

        max   Σ g_j
        s.t.  (f * g)_k ≤ M   for all k = 0 … 2n‑2
              g_j ≥ 0

   The constraints are linear because (f * g)_k = Σ_i f_i·g_{k‑i}.
   The optimal `g` can carry more total mass than `f`.  Scaling it back
   to unit sum reduces the maximal convolution value by the factor
   1/‖g‖₁².  The LP is solved with SciPy’s high‑performance “highs”
   backend.

3. **Smooth log‑sum‑exp gradient descent** – a few thousand iterations
   of the smoothed max objective to fine‑tune the sequence.

4. **Projected sub‑gradient + stochastic hill‑climb** – final polishing
   that drives the exact `max(conv)` down again.

All steps respect the 1000 s wall‑clock limit and return a plain list
of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (LP solver & sparse matrices)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy import sparse
    SCIPY_OK = True
except Exception:                     # pragma: no cover
    SCIPY_OK = False

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact score used by `evaluate_sequence` when the vector is already
    normalised to Σa = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (∑x = 1, x ≥ 0)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth log‑sum‑exp gradient descent (same as the reference code)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(start_a: np.ndarray,
                             time_deadline: float,
                             max_iters: int = 250_000,
                             beta_start: float = 5.0,
                             beta_end: float = 40.0,
                             step0: float = 0.5) -> tuple[np.ndarray, float]:
    a = start_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)                     # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        step = step0 / np.sqrt(it)
        a = project_to_simplex(a - step * grad)

        if it % 200 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 300_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        cur = compute_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Simple stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq,
                           time_deadline,
                           max_steps: int = 400_000) -> tuple[list[float], float]:
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                           # transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur = compute_score(cand)

        if cur < best_score:
            best_score = cur
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# 4. Linear‑programming refinement utilities
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray):
    """
    Build the (2n‑1)×n sparse matrix A with
        A_{k,j} = f_{k‑j}   if 0≤k‑j<n else 0.
    This matrix satisfies   A @ g = f * g   (full convolution).
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    # iterate over output index k
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over the non‑zero f‑entries that contribute to row k
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(k - i)            # j = k‑i
            data.append(f[i])
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_refine_once(f: np.ndarray, M: float):
    """
    Solve the LP   max Σ g  s.t.   (f * g) ≤ M, g ≥ 0.
    Returns the optimal g (or None on failure) and the achieved total mass.
    """
    if not SCIPY_OK:                     # pragma: no cover
        return None, None

    A = _build_conv_matrix(f)
    n = f.shape[0]
    c = -np.ones(n)                     # maximize Σ g  → minimize -Σ g
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'disp': False})
    except Exception as e:               # pragma: no cover
        print('linprog raised', e)
        return None, None

    if not res.success:
        return None, None

    g = np.maximum(res.x, 0.0)
    return g, g.sum()
