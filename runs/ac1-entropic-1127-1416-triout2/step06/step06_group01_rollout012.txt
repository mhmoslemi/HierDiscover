def lp_step(f, t, M, deadline):
    n = len(f)
    # Compute constraint matrix needed for convolution with f <= M.
    # Build A_ub shape (2n-1, n)
    rows = np.arange(2*n-1)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols
    mask = (idx >= 0) & (idx < n)
    # Create matrix using np.zeros then fill.
    A = np.zeros((2*n-1, n), dtype=np.float64)
    # Fill A[mask] = f[idx[mask]].
    A[mask] = f[idx[mask]]
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(b)
    # Solve using linprog
    try:
        from scipy.optimize import linprog
    except ImportError:
        return f  # fallback: cannot solve LP, return unchanged

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return f
    b_opt = res.x
    if b_opt.sum() == 0:
        return f
    g = b_opt / b_opt.sum()
    # Mix
    f_new = (1 - t) * f + t * g
    # Normalise to sum=1
    f_new /= f_new.sum()
    return f_new

"""
propose_candidate – 2024‑12‑01 version
=====================================

New strategy
------------
1. **Diverse seeds** – generate several hundred vectors of length 2048 from
   a handful of parametric families (arcsine, beta, power, mixtures,
   triangular, Dirichlet).  All vectors are normalised to Σa = 1.

2. **Linear‑programming “tighten‑upper‑bound” step** – for each seed we run a
   few cheap LP refinements based on the classic scheme
      maximise Σb  subject to b ≥ 0  and  f∗b ≤ M,
   where M = ‖f∗f‖∞.  The LP is solved with ``scipy.optimize.linprog``.
   The result is mixed back into the current vector (mixing coefficient t)
   and re‑normalised.  This quickly lowers the max‑convolution bound.

3. **Smooth‑max Adam polishing** – the best few LP‑refined seeds are
   further optimised with the smooth‑max surrogate (log‑sum‑exp) using Adam
   on the soft‑max logits.  The temperature τ is ramped from 0.5 to 80,
   giving a tighter approximation of the true max.

4. **Exact refinement** – the best Adam result is polished with the
   sub‑gradient descent and exponentiated‑gradient (mirror‑descent) from the
   previous state‑of‑the‑art code.

5. **Local stochastic search** – a short random‑perturbation phase
   (Gaussian noise + projection onto the simplex) is performed while
   time permits.

All phases respect the global 1000 s budget, constantly checking the
deadline and keeping the globally best sequence (according to the official
``evaluate_sequence``).  The routine returns a non‑empty list of non‑negative
heights.
"""

import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    ``a`` is assumed to be normalised to Σa = 1, therefore the denominator
    disappears and the score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σx = 1, x ≥ 0."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators (used for seeding)

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑like shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample from Dirichlet(1,…,1) (i.e. normalised exponential)."""
    x = rng.exponential(scale=1.0, size=n)
    total = x.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return x / total

# Linear‑programming improvement step
# ----------------------------------------------------------------------


def lp_step(f: np.ndarray, t: float, M: float, deadline: float) -> np.ndarray:
    """
    One LP improvement iteration for a given vector ``f``.
    Maximises Σb under the linear constraints  f∗b ≤ M  and  b ≥ 0,
    then mixes the result back with ``f`` using coefficient ``t``.
    The returned vector is normalised to Σ=1.
    """
    n = f.shape[0]
    # build the Toeplitz convolution matrix T (size (2n‑1)×n)
    rows = np.arange(2 * n - 1)[:, None]               # shape (2n‑1,1)
    cols = np.arange(n)[None, :]                      # shape (1,n)
    idx = rows - cols                                  # i = k‑j
    mask = (idx >= 0) & (idx < n)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    T[mask] = f[idx[mask]]

    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                 # maximise Σb → minimise −Σb

    # try to import SciPy only when needed
    try:
        from scipy.optimize import linprog
    except Exception:
        # SciPy unavailable – just skip the LP step
        return f

    # run the LP; we cap the solver's time indirectly via the global deadline
    # (the LP itself is very fast for our sizes)
    res = linprog(c, A_ub=T, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs')
    if not res.success:
        return f

    b_opt = res.x
    if b_opt.sum() <= 0.0:
        return f
    g = b_opt / b_opt.sum()
    f_new = (1.0 - t) * f + t * g
    f_new /= f_new.sum()
    # safety check for the deadline (in case LP took long)
    if time.time() >= deadline:
        return f_new
    return f_new

# Smooth‑max Adam optimiser (new component)
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 30_000,
                             tau_start: float = 0.5,
                             tau_end: float = 80.0,
                             base_lr: float = 0.3) -> np.ndarray:
    """
    Optimises the smooth max‑approximation Φ_τ(a) = (1/τ)·log∑exp(τ·conv)
    by running Adam on the logits that generate a via a soft‑max.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)          # logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        conv = convolution_fft(a, a)

        # log‑sum‑exp weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()          # shape (L,)

        # gradient wrt a : 2·∑_k w_k·shift(a,k)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # chain rule through soft‑max
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / math.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # optional monitor
        if it % 10_000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final soft‑max
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a

def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """Euclidean projected sub‑gradient descent on the exact max‑convolution."""
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient: 2·∑ mask·shift(a)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a -= eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.3) -> np.ndarray:
    """Exponentiated‑gradient polishing (mirror‑descent with KL geometry)."""
    a = a.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (math.sqrt(it) + 1.0)
        a *= np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return a

def propose_candidate(seed: int = 42,

    """

    for _ in range(150):
        a = rng.uniform(0.1, 5.0)

        cand = generate_beta_shape(n, a, b, eps)

        cand = generate_power_shape(n, p, eps)
        pool.append((compute_score(cand), cand))

    # mixtures arcsine / uniform

        w = rng.random()
        cand = generate_mixture_shape(n, w)
        pool.append((compute_score(cand), cand))

    # triangular bumps
    for _ in range(120):
        peak = rng.integers(0, n)
        cand = generate_triangular_shape(n, peak)
        pool.append((compute_score(cand), cand))

    # Dirichlet draws
    for _ in range(120):
        cand = random_dirichlet(n, rng)
        pool.append((compute_score(cand), cand))

    # keep the most promising seeds (smallest surrogate)
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:8]]
    print(f"Selected {len(top_seeds)} top seeds for LP/Adam stage "

    # --------------------------------------------------------------

    for idx, seed_vec in enumerate(top_seeds, start=1):

        print(f"\n=== Seed {idx}/{len(top_seeds)} (LP + Adam) ===")
        f = seed_vec.copy()

        # --- LP tightening (up to 4 iterations) ---
        lp_iters = 4
        for lp_it in range(lp_iters):
            if time.time() >= deadline - 2.0:
