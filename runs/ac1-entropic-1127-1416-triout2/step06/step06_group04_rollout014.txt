def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    best_seq = []
    best_val = float('inf')

    # try fallback sequence
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f'Fallback sequence (height_sequence_1) official score = {val:.6f}')
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # candidate lengths
    lengths = [2048, 4096, 8192]  # try longer lengths if time permits

    for n in lengths:
        if time.time() >= deadline - 5:
            break
        print(f"\n=== Optimizing length n={n} ===")

        # Sampling initial candidates
        # deterministic starters
        starters = [
            np.full(n, 1.0 / n),
            arcsine_initial(n),
        ]
        # symmetric beta grid
        for alpha in np.linspace(0.3, 0.7, 21):
            cand = generate_beta_shape(n, alpha, alpha, eps=1e-6)
            starters.append(cand)

        # Dirichlet random
        for _ in range(200):
            if time.time() >= deadline - 5:
                break
            x = rng.exponential(scale=1.0, size=n)
            cand = x / x.sum()
            starters.append(cand)

        # random beta mixture
        for _ in range(800):
            if time.time() >= deadline - 5:
                break
            if rng.random() < 0.4:
                # mixture
                w = rng.random()
                α1 = rng.uniform(0.1, 3.0)
                β1 = rng.uniform(0.1, 3.0)
                eps1 = rng.uniform(1e-8, 0.02 * n)
                α2 = rng.uniform(0.1, 3.0)
                β2 = rng.uniform(0.1, 3.0)
                eps2 = rng.uniform(1e-8, 0.02 * n)
                cand = generate_beta_mixture(n, α1, β1, eps1, α2, β2, eps2, w)
            else:
                α = rng.uniform(0.1, 3.0)
                β = rng.uniform(0.1, 3.0)
                eps = rng.uniform(1e-8, 0.02 * n)
                cand = generate_beta_shape(n, α, β, eps)
            starters.append(cand)

        # evaluate fast scores
        local_best = None
        local_best_sc = float('inf')
        for st in starters:
            sc = compute_score(st)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = st.copy()
        print(f'  best sampled fast score = {local_best_sc:.6f}')

        if local_best is None:
            continue

        # allocate remaining time
        remaining = deadline - time.time()

Propose a low‑score step‑function sequence.

The routine performs a multi‑stage heuristic search on a chosen length
(e.g. 2048, 4096, 8192).  For each length it

1.  Samples a diverse pool of candidate shapes (uniform, arcsine, symmetric
    beta‑families, random Dirichlet vectors and random mixtures of two beta
    densities).

2.  Picks the best sampled vector (fast score 2·n·max(conv) – the official
    score when the vector is L1‑normalised) and refines it with a sequence
    of projected optimisation phases that respect the simplex constraint:
      • flattening – minimise the L₂ distance of the self‑convolution to the
        theoretical flat level 1/(2n‑1) using Adam;
      • smooth‑max – minimise a log‑sum‑exp surrogate of the max‑convolution
        with a temperature that is annealed up to λ≈5 000;
      • exact sub‑gradient descent on the true max‑convolution;
      • cheap stochastic Gaussian walk on the simplex.

3.  Evaluates the polished vector with the official `evaluate_sequence`
    function and keeps the best overall result.

4.  If time permits a final global polishing (longer Adam / sub‑gradient
    runs + a final random walk) is performed.

All steps respect a global time budget (default 1000 s) and the routine
falls back to a uniform 256‑vector if no candidate could be produced.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (returns length len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (full length)."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score.
    Works for vectors already normalised to Σa = 1.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) – O(n log n).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (beta families and mixtures)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Symmetric beta‑family density on {0,…,n‑1} (unnormalised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities (renormalised)."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) density."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Flattening (L2) loss and its gradient
# ----------------------------------------------------------------------


def flatten_gradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Gradient of ‖conv(a,a) – avg‖₂² where avg = 1/(2n‑1) and Σa = 1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient: 4 * (a ⊛ diff_rev)  (⊛ = convolution)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)          # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def adam_flatten(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.25,
) -> Tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the *flattening* L₂ loss.
    Returns the best vector (projected) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        grad, loss = flatten_gradient(a)

        # Adam update (learning rate decays like 1/√it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 20000 == 0:
            print(
                f"    Adam‑flatten iter={it:,d} loss={loss:.6e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth surrogate of the max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value B) where
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv‑max)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights on convolution entries
    B = max_c + np.log(sum_exp) / lam

    # weighted sum of sub‑gradients:
    # ∂/∂a Σ_k w_k (a*a)_k = 2 * (a ⊛ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_max(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 500.0,
    noise_every: int = 1000,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution surrogate.
    Returns the best vector (projected) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # quadratic temperature schedule
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam‑max iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} lam={lam:.1f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max; back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Simple back‑tracking line‑search guarantees non‑increase.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # back‑tracking line‑search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (cheap Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a small uphill‑move probability.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # accept if improvement, otherwise with a tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine mixes extensive shape sampling, a prolonged flattening stage,
    a high‑temperature Adam optimisation of a log‑sum‑exp surrogate, an
    exact sub‑gradient descent and a final stochastic walk.  All phases
    honour the global time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to any previously known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Work on a few candidate lengths
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # ----- 1.1  Broad sampling of initial shapes --------------------
        starters = [
            np.full(n, 1.0 / n),          # uniform
            arcsine_initial(n),           # classic arcsine
        ]

        # symmetric beta grid (fine)
        for alpha in np.linspace(0.30, 0.70, 21):
            starters.append(generate_beta_shape(n, alpha, alpha, eps=1e-6))

        # random Dirichlet draws
        for _ in range(200):
            if time.time() >= deadline - 5.0:
                break
            x = rng.exponential(scale=1.0, size=n)
            starters.append(x / x.sum())

        # random beta‑mixture draws
        for _ in range(800):
            if time.time() >= deadline - 5.0:
                break
            if rng.random() < 0.4:
                w = rng.random()
                α1 = rng.uniform(0.1, 3.0)
                β1 = rng.uniform(0.1, 3.0)
                eps1 = rng.uniform(1e-8, 0.02 * n)
                α2 = rng.uniform(0.1, 3.0)
                β2 = rng.uniform(0.1, 3.0)
                eps2 = rng.uniform(1e-8, 0.02 * n)
                cand = generate_beta_mixture(
                    n, α1, β1, eps1, α2, β2, eps2, w
                )
            else:
                α = rng.uniform(0.1, 3.0)
                β = rng.uniform(0.1, 3.0)
                eps = rng.uniform(1e-8, 0.02 * n)
                cand = generate_beta_shape(n, α, β, eps)
            starters.append(cand)

        # ----- 1.2  Pick the best sampled candidate --------------------
        local_best = None
        local_best_sc = float("inf")
        for st in starters:
            sc = compute_score(st)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = st.copy()
        print(f"  best sampled fast score = {local_best_sc:.6f}")

        if local_best is None:
            continue

        # ------------------------------------------------------------------
        # 2.  Hierarchical polishing of the chosen candidate
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 10.0:
            break

        # ---- 2.1 Flattening (L2) stage ---------------------------------
        t_flat_end = time.time() + min(remaining * 0.25, 120.0)
        a_flat, _ = adam_flatten(
            initial_a=local_best,
            time_deadline=t_flat_end,
            max_iter=200_000,
            lr0=0.45,
        )
        print(f"  after flattening fast score = {compute_score(a_flat):.6f}")

        # ---- 2.2 Smooth‑max Adam (high λ) -----------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            t_max_end = time.time() + min(remaining * 0.35, 300.0)
            a_max, _ = adam_max(
                initial_a=a_flat,
                time_deadline=t_max_end,
                max_iter=400_000,
                lr0=0.12,
                lam_start=1.0,
                lam_end=5000.0,
                noise_every=800,
                noise_scale=3e-5,
            )
            print(f"  after smooth‑max fast score = {compute_score(a_max):.6f}")
        else:
            a_max = a_flat

        # ---- 2.3 Exact sub‑gradient descent ----------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            t_sub_end = time.time() + min(remaining * 0.25, 200.0)
            a_sub, _ = subgradient_optimize(
                a_max,
                time_deadline=t_sub_end,
                step0=0.20,
                max_iter=300_000,
            )
            print(f"  after sub‑gradient fast score = {compute_score(a_sub):.6f}")
        else:
            a_sub = a_max

        # ---- 2.4 Stochastic Gaussian walk -------------------------------
        remaining = deadline - time.time()
        if remaining > 2.0:
            t_rw_end = time.time() + min(remaining - 0.5, 180.0)
            a_final, _ = random_local_search(
                a_sub,
                time_deadline=t_rw_end,
                init_step=0.015,
                max_iter=400_000,
            )
            print(f"  after random walk fast score = {compute_score(a_final):.6f}")
        else:
            a_final = a_sub

        # ---- 2.5 Official evaluation ------------------------------------
        val = evaluate_sequence(a_final.tolist())
        print(f"  → official score for length {n} = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()
            print(f"  *** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best vector (time permitting)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # Stage A – prolonged flattening
        rem = deadline - time.time()
        t_flat_end = time.time() + min(rem * 0.20, 180.0)
        a, _ = adam_flatten(
            initial_a=a,
            time_deadline=t_flat_end,
            max_iter=300_000,
            lr0=0.55,
        )
        print(f"  after global flattening fast score = {compute_score(a):.6f}")

        # Stage B – high‑temperature Adam on the max‑surrogate
        rem = deadline - time.time()
        if rem > 5.0:
            t_max_end = time.time() + min(rem * 0.30, 300.0)
            a, _ = adam_max(
                initial_a=a,
                time_deadline=t_max_end,
                max_iter=500_000,
                lr0=0.09,
                lam_start=1.0,
                lam_end=8000.0,
                noise_every=1200,
                noise_scale=2e-5,
            )
            print(f"  after global smooth‑max fast score = {compute_score(a):.6f}")

        # Stage C – exact sub‑gradient
        rem = deadline - time.time()
        if rem > 5.0:
            t_sub_end = time.time() + min(rem * 0.25, 250.0)
            a, _ = subgradient_optimize(
                a,
                time_deadline=t_sub_end,
                step0=0.18,
                max_iter=400_000,
            )
            print(f"  after global sub‑gradient fast score = {compute_score(a):.6f}")

        # Stage D – final random walk
        rem = deadline - time.time()
        if rem > 2.0:
            t_rw_end = time.time() + min(rem - 0.5, 200.0)
            a, _ = random_local_search(
                a,
                time_deadline=t_rw_end,
                init_step=0.012,
                max_iter=500_000,
            )
            print(f"  after final random walk fast score = {compute_score(a):.6f}")

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform 256‑vector official score = {best_val:.6f}")

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
