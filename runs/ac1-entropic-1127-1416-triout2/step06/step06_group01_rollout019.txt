"""
propose_candidate – enhanced optimizer
=====================================

This version builds on the previous three‑stage approach but adds:

* a larger search space (optionally 4096 coefficients);
* a much longer Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate;
* an extended Euclidean sub‑gradient phase with adaptive step size;
* a final mirror‑descent (exponentiated‑gradient) polish;
* a cheap hill‑climbing / simulated‑annealing post‑process that works on the
  fast surrogate (it is exactly the official objective when Σa=1);
* several extra random seed families (Dirichlet, sparse extreme points).

All stages respect the global 1000 s budget, the best feasible vector
encountered (according to the official ``evaluate_sequence``) is returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ---- utilities -----------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate when Σa = 1:
    return 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation.
    """
    if v.size == 0:
        return v
    # quick path for already feasible point
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ---- seed generators -----------------------------------------------------------

def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    s = base.sum()
    if s <= 0 or not np.isfinite(s):
        return uniform_shape(n)
    return base / s


def power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    s = vals.sum()
    if s <= 0 or not np.isfinite(s):
        return uniform_shape(n)
    return vals / s


def triangular_shape(n: int, peak: int) -> np.ndarray:
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def mixture_shape(n: int, w_arc: float) -> np.ndarray:
    arc = arcsine_shape(n)
    uni = uniform_shape(n)
    mix = w_arc * arc + (1.0 - w_arc) * uni
    return mix / mix.sum()


def dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet vector with concentration alpha (scalar)."""
    if alpha <= 0:
        alpha = 0.1
    sample = rng.gamma(alpha, 1.0, size=n)
    total = sample.sum()
    if total <= 0:
        return uniform_shape(n)
    return sample / total


# ---- optimisation primitives ---------------------------------------------------

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 150_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log∑_k exp(τ·conv_k)
    with τ linearly increasing from ``tau_start`` to ``tau_end``.
    a is kept on the simplex via the soft‑max re‑parameterisation.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)          # logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax a = exp(w) / sum(exp(w))
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # Jacobian of softmax
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 20_000 == 0:
            surrogate = compute_score(a)
            print(f"    [Adam] it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def subgradient_descent_exact(a_init: np.ndarray,
                              deadline: float,
                              max_iter: int = 500_000,
                              step0: float = 0.08) -> np.ndarray:
    """
    Euclidean projected sub‑gradient descent on the exact max‑convolution.
    """
    a = project_to_simplex(a_init.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # mask of indices attaining (or close to) the maximum
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a -= eta * subgrad
        a = project_to_simplex(a)

        if it % 100_000 == 0:
            surrogate = compute_score(a)
            print(f"    [Subgrad] it={it:,d} surrogate={surrogate:.6f}")

    return a


def eg_refine(a_init: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror descent with KL geometry) polishing.
    """
    a = a_init.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:

        else:
            a /= total

        if it % 100_000 == 0:

            print(f"    [EG] it={it:,d} surrogate={surrogate:.6f}")

    return a


def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_iters: int = 200_000,
               step_scale: float = 0.01) -> np.ndarray:
    """
    Simple stochastic hill‑climbing that moves a small amount of mass
    between two random coordinates if the surrogate (which equals the
    official objective when Σa=1) improves.
    """

    n = a.shape[0]
    best_score = compute_score(a)

    rng = np.random.default_rng()
    while time.time() < deadline and max_iters > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose to shift a small fraction of total mass from i to j
        delta = step_scale * rng.random()
        if a[i] < delta:
            delta = a[i] * 0.5
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # already stays on simplex, non‑negative
        new_score = compute_score(a_new)
        if new_score < best_score:
            a = a_new
            best_score = new_score
            # occasional logging
            if max_iters % 10_000 == 0:
                print(f"    [Hill] improved to {best_score:.6f}")

propose_candidate – enhanced optimizer
=====================================

A three‑stage optimizer (random seeding → smooth‑max Adam → exact

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate when Σa = 1:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1} (Duchi et al., 2008).
    """
    if v.size == 0:
        return v
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    s = base.sum()
    if s <= 0 or not np.isfinite(s):
        return uniform_shape(n)
    return base / s


def power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    s = vals.sum()
    if s <= 0 or not np.isfinite(s):
        return uniform_shape(n)
    return vals / s


def triangular_shape(n: int, peak: int) -> np.ndarray:
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def mixture_shape(n: int, w_arc: float) -> np.ndarray:
    arc = arcsine_shape(n)
    uni = uniform_shape(n)
    mix = w_arc * arc + (1.0 - w_arc) * uni
    return mix / mix.sum()


def dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet vector with scalar concentration ``alpha``."""
    if alpha <= 0:
        alpha = 0.1
    sample = rng.gamma(alpha, 1.0, size=n)
    total = sample.sum()
    if total <= 0:
        return uniform_shape(n)
    return sample / total

def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 150_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.5) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate
        Φ_τ(a) = (1/τ)·log∑_k exp(τ·conv_k)
    with τ linearly increasing from ``tau_start`` to ``tau_end``.
    ``a`` stays on the simplex via the soft‑max parametrisation.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    w = np.log(a_init + 1e-12)          # logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        conv = convolution_fft(a, a)

        # smooth‑max weights
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp          # shape (L,)

        # gradient w.r.t. a
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # Jacobian of softmax
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 20_000 == 0:
            surrogate = compute_score(a)
            print(f"    [Adam] it={it:,d} τ={tau:.1f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a

        # Adam phase (smooth‑max)
        adam_iters = int(min(150_000, (deadline - time.time()) * 120))
        a_adam = adam_optimize_smooth_max(seed_vec,
                                          deadline,
                                          max_iters=adam_iters,
                                          tau_start=0.5,
                                          tau_end=200.0,
                                          base_lr=0.5)
        surrogate = compute_score(a_adam)
        print(f"    After Adam surrogate = {surrogate:.6f}")

        # Sub‑gradient phase
        sub_iters = int(min(400_000, (deadline - time.time()) * 80))
        a_sub = subgradient_descent_exact(a_adam,
                                          deadline,
                                          max_iter=sub_iters,
                                          step0=0.08)

        # EG polishing
        eg_iters = int(min(150_000, (deadline - time.time()) * 40))
        a_eg = eg_refine(a_sub,
                         deadline,
                         max_iter=eg_iters,
                         eta0=0.25)

        final_surrogate = compute_score(a_eg)
        print(f"    After EG surrogate = {final_surrogate:.6f}")

        # store for final selection
        pool.append((final_surrogate, a_eg))

    # ------------------------------------------------------------------
    # 4.  Pick the best surrogate and run a cheap hill‑climb
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    best_candidate = pool[0][1]
    print(f"\nBest surrogate before hill‑climb = {pool[0][0]:.6f}")

    if time.time() < deadline - 5.0:
        print("[Hill‑climb] starting")
        hc_iters = int(min(200_000, (deadline - time.time()) * 30))
        best_candidate = hill_climb(best_candidate, deadline,
                                    max_iters=hc_iters, step_scale=0.02)
        print(f"    After hill‑climb surrogate = {compute_score(best_candidate):.6f}")

    # ------------------------------------------------------------------
    # 5.  Official evaluation (official = 2·n·max(conv) because Σa=1)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_candidate.tolist())
        print(f"Official score after full pipeline = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = best_candidate.tolist()
    except Exception as e:
        print(f"Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 6.  Guard – always return a non‑empty sequence
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_shape(n).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
