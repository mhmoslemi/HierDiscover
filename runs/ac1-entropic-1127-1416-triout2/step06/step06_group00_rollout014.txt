def balance_phase(a, time_deadline, max_steps=200000):
    n=len(a); best_a=a.copy(); best_score=compute_score(a)
    step_frac=0.05
    while time.time()<time_deadline and steps<max_steps:
        conv=convolution_fft(a,a)
        k=np.argmax(conv)
        # compute contributions
        i_arr = np.arange(n)
        j = k - i_arr
        mask = (j>=0)&(j<n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j[mask]]
        # if total contributal mass sum is zero maybe continue
        prob = contrib / contrib.sum() if contrib.sum()>0 else None
        # sample i according to prob
        if prob is not None and prob.max()>0:
            i = np.random.choice(n, p=prob)
            # also choose a target j2 where a is low
            low_idx = np.where(a < np.quantile(a, 0.2))[0]
            if low_idx.size>0:
                t = np.random.choice(low_idx)
            else:
                t = np.random.randint(n)
            delta = step_frac * a[i]
            a[i] -= delta
            a[t] += delta
            a = project_to_simplex(a)
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                step_frac = min(step_frac*1.05, 0.5)
            else:
                step_frac *= 0.97
        steps += 1
    return best_a, best_score

def balance_phase(a, time_deadline, max_steps=200_000):
    a = a.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    step_frac = 0.05
    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # compute contributions
        i_arr = np.arange(n)
        j = k - i_arr
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j[mask]]
        total = contrib.sum()
        if total == 0:
            break
        prob = contrib / total
        # sample source index i according to contribution distribution
        i = np.random.choice(n, p=prob)
        # choose target index with low value
        low_inds = np.where(a < np.quantile(a, 0.2))[0]
        if low_inds.size > 0:
            t = np.random.choice(low_inds)
        else:
            t = np.random.randint(n)
        delta = step_frac * a[i]
        a[i] -= delta
        a[t] += delta
        a = project_to_simplex(a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            step_frac = min(step_frac * 1.07, 0.5)
        else:
            step_frac *= 0.96
        steps += 1
    return best_a, best_score

"""
Propose a low‑score step‑function.

Algorithm (≈ 1000 s budget)
---------------------------
0.  Choose a problem size n (2048 or 4096) depending on the available time.
1.  Initialise a probability vector a (∑a=1) – several random schemes are tried:
    * uniform,
    * Dirichlet with random concentration,
    * exponential decay.
2.  Smooth optimisation of the log‑sum‑exp surrogate of max(a∗a) using
    Adam‑style adaptive steps and an annealed smoothness parameter β.
3.  Projected sub‑gradient descent with Nesterov momentum on the true
    objective 2·n·max(a∗a).
4.  Targeted “balance” phase – repeatedly move a small fraction of the mass
    from indices that contribute most to the current maximal convolution
    towards low‑mass indices.
5.  Stochastic hill‑climb with swaps and mass‑transfer moves (the step size
    adapts to successes).
6.  Keep the best vector seen during the whole run, evaluate it with the
    provided `evaluate_sequence`, and return it as a plain list of floats.
"""

import time
import numpy as np
import math
import random

# ----------------------------------------------------------------------
# Primitive utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b using real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors already normalised to sum=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Smooth optimisation (Adam on log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def smooth_adam_optimize(start_a: np.ndarray,
                         time_deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 5.0,
                         beta_end: float = 300.0,
                         base_lr: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smoothed objective
        f_β(a) = (1/β)·log Σ exp(β·conv_i)
    with an exponential annealing of β from beta_start to beta_end.
    The vector stays on the simplex after every step.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # --- anneal β -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # --- smoothed objective (log‑sum‑exp) ------------------------
        conv = convolution_fft(a, a)                 # length 2n‑1
        maxc = conv.max()
        # stable soft‑max weights
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()          # Σ w_i = 1

        # --- gradient of f_β -----------------------------------------
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]       # slice of length n

        # --- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = project_to_simplex(a)

        # occasional exact evaluation
        if it % 250 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Sub‑gradient descent with Nesterov momentum
# ----------------------------------------------------------------------


def subgrad_optimize_nesterov(init_a: np.ndarray,
                              time_deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        2·n·max(a⊛a)
    with Nesterov momentum.  The vector stays on the simplex.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    v = np.zeros_like(a)   # momentum term
    mu = 0.9

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max conv (2·a shifted)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Nesterov momentum update
        v = mu * v + (1.0 - mu) * g
        step = step0 / np.sqrt(t)
        a = a - step * v
        a = project_to_simplex(a)

        cur = compute_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            a += np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Targeted balance phase – move mass away from the peak convolution
# ----------------------------------------------------------------------


def balance_phase(a: np.ndarray,
                  time_deadline: float,
                  max_steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Heuristic that repeatedly transfers a small fraction of the mass from
    indices that contribute heavily to the current maximal convolution value
    towards low‑mass indices.  The vector stays on the simplex.
    """
    a = a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    step_frac = 0.05
    steps = 0

    while steps < max_steps and time.time() < time_deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # contribution of each index to conv[k]
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]
        total = contrib.sum()
        if total == 0.0:
            break

        prob = contrib / total
        src = np.random.choice(n, p=prob)

        # pick a low‑mass destination
        low = np.where(a < np.quantile(a, 0.2))[0]
        if low.size > 0:
            dst = np.random.choice(low)
        else:
            dst = np.random.randint(n)

        delta = step_frac * a[src]
        a[src] -= delta
        a[dst] += delta
        a = project_to_simplex(a)

        cur = compute_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            step_frac = min(step_frac * 1.07, 0.5)
        else:
            step_frac *= 0.96

        steps += 1

    return best_a, best_score

# 4. Final hill‑climb polishing (stochastic mass transfers)
# ----------------------------------------------------------------------


def hill_climb_targeted(seq,
                        time_deadline,
                        max_steps: int = 300_000) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb that uses swaps and targeted mass‑transfer moves.
    The best vector seen is returned.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10
    steps = 0

    while steps < max_steps and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.6:
            # targeted mass transfer from a high‑contributing index
            conv = convolution_fft(cand, cand)
            k = int(np.argmax(conv))
            i = np.arange(n)
            j = k - i
            mask = (j >= 0) & (j < n)
            contrib = np.zeros(n)
            contrib[mask] = cand[i[mask]] * cand[j[mask]]
            total = contrib.sum()
            if total > 0:
                prob = contrib / total
                src = np.random.choice(n, p=prob)
                low = np.where(cand < np.quantile(cand, 0.2))[0]
                if low.size > 0:
                    dst = np.random.choice(low)
                else:
                    dst = np.random.randint(n)
                delta = step_scale * cand[src]
                cand[src] -= delta
                cand[dst] += delta
        else:
            # simple swap of two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        cand = project_to_simplex(cand)
        cur = compute_score(cand)

        if cur < best_score:
            best_score = cur
            best_a = cand.copy()
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a.tolist(), best_score
