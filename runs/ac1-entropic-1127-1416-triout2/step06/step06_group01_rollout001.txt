"""
propose_candidate – 2024‑11‑27‑Improved
=======================================

A three‑stage optimiser that builds on the previous implementation and adds
new shape families, a longer Adam optimisation with an exponential temperature
schedule, and a cheap stochastic local‑search that evaluates the *official*
objective directly.  The algorithm works for any sequence length, but the
default lengths (1024 – 4096) give a good trade‑off between quality and running
time.

Stage 1 – Diverse seeding
    Several parametric families (arcsine, beta, power‑shape, raised‑cosine,
    exponential, mixtures, triangular) are sampled.  All candidates are
    normalised to sum = 1 and scored with the fast surrogate
    ``2·n·max(conv)``; the best few are kept.

Stage 2 – Smooth‑max Adam optimisation
    The surrogate
        Φ_τ(a)= (1/τ)·log∑_k exp(τ·conv_k)
    approximates the true max.  Adam is run on the logits of a soft‑max
    parametrisation (so the simplex constraint is satisfied automatically).
    The temperature τ grows *exponentially* from 0.5 to 200, turning the
    smooth objective into the true max at the end of the run.

Stage 3 – Exact refinement + stochastic polishing
    A projected sub‑gradient descent (Euclidean projection onto the simplex)
    minimises the exact max‑convolution, followed by an exponentiated‑gradient
    (mirror‑descent) polish.  Finally a cheap random‑walk on the simplex is
    performed; each perturbation is evaluated with the *official* function
    ``evaluate_sequence`` and kept if it improves the score.

All stages respect the global 1000 s budget and keep the globally best
sequence (according to the official objective) for the return value.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    ``a`` is assumed to be normalised to Σa = 1, therefore the denominator
    disappears and the score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators (used for seeding)

def arcsine_initial(n: int) -> np.ndarray:
    """Classic discrete arcsine shape."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_raised_cosine_shape(n: int, p: float) -> np.ndarray:
    """
    Raised‑cosine hump: a_i ∝ (cos(π·(i/(n‑1)‑0.5)))_+^{p}.
    The “_+” denotes truncation at zero; for the full range the cosine is
    already non‑negative.
    """
    i = np.arange(n, dtype=float) / (n - 1)
    vals = np.cos(np.pi * (i - 0.5))
    vals = np.maximum(vals, 0.0) ** p
    total = vals.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_exponential_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(i - centre))
    total = vals.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total

def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 120_000,
                              step0: float = 0.10) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact objective max(conv(a,a)).
    Uses Euclidean projection onto the simplex.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·∑_{k active} shift(a, k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 80_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent with KL geometry) polish.
    """
    a = a.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (main engine)
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 40_000,
                             tau_start: float = 0.5,
                             tau_end: float = 200.0,
                             base_lr: float = 0.30) -> np.ndarray:
    """
    Optimises the smooth approximation Φ_τ using Adam on the logits.
    Temperature τ grows *exponentially* from ``tau_start`` to ``tau_end``.
    This gradually morphs the smooth loss into the true max.
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from the starting probability vector
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        progress = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** progress

        # softmax a = softmax(w)
        w_shift = w - np.max(w)          # numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max weights = softmax(tau * conv)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp   # shape (L,)

        # gradient w.r.t. a
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # back‑prop through softmax
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update with decreasing step size
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_lr / np.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 5000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam iter {it:,d}  τ={tau:.2f}  surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a

def local_random_search(a_start: np.ndarray,
                        deadline: float,
                        max_evals: int = 300,
                        sigma0: float = 0.02) -> np.ndarray:
    """
    Simple random walk on the simplex that keeps any improvement according to
    the official ``evaluate_sequence``.  Very cheap (a few hundred evaluations)
    and often squeezes the final score a little.
    """
    rng = np.random.default_rng()
    a_best = a_start.copy()
    try:
        best_val = evaluate_sequence(a_best.tolist())
    except Exception:
        best_val = float("inf")

    n = len(a_start)
    evals = 0
    while time.time() < deadline and evals < max_evals:
        sigma = sigma0 * (1.0 - evals / max_evals)  # anneal step size
        delta = rng.normal(scale=sigma, size=n)
        cand = project_to_simplex(a_best + delta)
        try:
            val = evaluate_sequence(cand.tolist())
            evals += 1
            if val < best_val:
                best_val = val
                a_best = cand
                print(f"    Random‑search improvement: {best_val:.6f} "
                      f"after {evals} evals")
        except Exception:
            # any malformed candidate is simply discarded
            evals += 1
            continue
    return a_best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine combines diverse seeding, a long Adam smooth‑max phase,
    exact sub‑gradient + exponentiated‑gradient polishing, and a cheap
    stochastic local search.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback from a previous run (if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Build a diverse pool of seed vectors (fast surrogate)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]               # progressively larger
    pool: List[Tuple[float, np.ndarray]] = []  # (surrogate, vector)

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Seeding length {n} ---")

        # deterministic starters
        uniform = np.full(n, 1.0 / n)
        pool.append((compute_score(uniform), uniform.copy()))
        arc = arcsine_initial(n)
        pool.append((compute_score(arc), arc.copy()))

        # random beta‑family
        for _ in range(150):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            pool.append((compute_score(cand), cand))

        # random power‑shape (including steep U‑shapes)
        for _ in range(150):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.2, 4.0)          # allow larger exponents
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            pool.append((compute_score(cand), cand))

        # raised‑cosine bumps
        for _ in range(100):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.5, 6.0)
            cand = generate_raised_cosine_shape(n, p)
            pool.append((compute_score(cand), cand))

        # exponential decays from the centre
        for _ in range(100):
            if time.time() >= deadline - 5.0:
                break
            lam = rng.uniform(0.001, 0.02) * n
            cand = generate_exponential_shape(n, lam)
            pool.append((compute_score(cand), cand))

        # mixtures of arcsine & uniform
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            pool.append((compute_score(cand), cand))

        # triangular bumps at random positions
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            pool.append((compute_score(cand), cand))

    # keep only the very best seeds for the expensive Adam phase
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:12]]
    print(f"\nSelected {len(top_seeds)} top seeds for Adam optimisation "
          f"(surrogate range {pool[0][0]:.6f} – {pool[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2.  Adam refinement (smooth‑max) of each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 4.0:
            break
        n = seed_vec.shape[0]
        print(f"\n=== Adam refinement of seed {idx}/{len(top_seeds)} (n={n}) ===")
        remaining = deadline - time.time()
        # allocate a proportion of the remaining budget (≈70 %)
        max_iters = int(min(40_000, remaining * 30))
        a_adam = adam_optimize_smooth_max(seed_vec,
                                          deadline,
                                          max_iters=max_iters,
                                          tau_start=0.5,
                                          tau_end=200.0,
                                          base_lr=0.30)

        surrogate = compute_score(a_adam)
        print(f"    Post‑Adam surrogate score = {surrogate:.6f}")

        pool.append((surrogate, a_adam))

    # ------------------------------------------------------------------
    # 3.  Pick the globally best surrogate and perform exact polishing
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    candidate = pool[0][1]                # best after Adam
    n = candidate.shape[0]
    print(f"\n--- Starting exact refinement (n={n}) ---")

    remaining = deadline - time.time()
    max_iter_sub = int(min(200_000, remaining * 30))
    refined = subgradient_descent_exact(candidate,
                                        deadline,
                                        max_iter=max_iter_sub,
                                        step0=0.09)

    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        max_iter_eg = int(min(120_000, remaining * 30))
        refined = eg_refine(refined, deadline,
                            max_iter=max_iter_eg,
                            eta0=0.22)

    # ------------------------------------------------------------------
    # 4.  Cheap stochastic polishing with the official objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        refined = local_random_search(refined,
                                      deadline,
                                      max_evals=300,
                                      sigma0=0.018)

    # ------------------------------------------------------------------
    # 5.  Official evaluation of the polished candidate
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(refined.tolist())
        print(f"Official score after full pipeline = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            print("*** New global best found! ***")
    except Exception as e:
        print(f"Official evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 6.  Final global polishing (if time remains)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Final global polish (extra pass) ---")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(250_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.07)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(150_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.18)

        # one more cheap random‑search pass
        if time.time() < deadline - 1.0:
            a = local_random_search(a, deadline, max_evals=200, sigma0=0.015)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 7.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
