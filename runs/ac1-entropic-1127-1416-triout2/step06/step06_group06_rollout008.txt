"""
Hybrid optimizer for the step‑function “attack”.

Goal:
    minimise  score(a) = 2·n·max(conv(a,a)) / (sum(a))**2
which is invariant under scaling.
We therefore optimise a on the probability simplex (sum = 1, a_i ≥ 0)
and minimise  f(a)=2·n·max(conv(a,a)).

Algorithm
---------

1️⃣  **Random restarts** – each restart draws a Dirichlet vector
    (concentration α∈[0.1,5]).

2️⃣  **Accelerated exponentiated‑gradient descent** – a smooth surrogate
    of the max is built with a soft‑max over all convolution entries:
        w_k ∝ exp( λ·conv_k )
    The gradient of the surrogate is
        g = 2·(a_rev * w)        (central n entries only)
    Multiplicative updates  a ← a·exp(−η·g) are performed,
    followed by Euclidean projection onto the simplex.
    • λ grows fast (λ≈200) so the surrogate quickly concentrates
      on the true worst shift.
    • η decays as η₀/√t (η₀≈0.7) with clipping to avoid overflow.
    The best vector seen during the descent is recorded.

3️⃣  **Targeted hill‑climbing** – after the gradient phase we know the
    shift k⁎ that attains the current maximum.  The contribution
    a_i·a_{i+k⁎} is largest for some index i⁎.  A stochastic move
    transfers a fraction of the mass from i⁎ to a random distant index
    j.  The move is accepted only if the true objective decreases.
    The transfer fraction adapts to the recent success rate.

The whole search runs inside the 1000 s budget; the remaining time at
the end is reserved for the polishing hill‑climb.  The best sequence
encountered is returned as a plain Python ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        # fallback to uniform point
        return np.full_like(v, 1.0 / v.shape[0])

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (fast replica of evaluate_sequence for sum=1 vectors)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a on the simplex."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Accelerated smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------


def smooth_exp_descent(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.70,
    lambda_start: float = 0.5,
    lambda_max: float = 200.0,
    lambda_ramp: int = 20000,
    eval_interval: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Run exponentiated‑gradient descent on
        f(a) = max_k (a*a)[k]   (a lies on the simplex).

    The soft‑max temperature λ is increased quickly (up to ``lambda_max``)
    so that the surrogate rapidly focuses on the true worst shift.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # convolution a*a (length 2n-1)
        conv = convolution_fft(a, a)

        # λ schedule – fast ramp to the large value
        lam = lambda_start + (lambda_max - lambda_start) * min(1.0, it / lambda_ramp)

        # soft‑max weights (stable)
        M = np.max(conv)
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)                     # length 2n-1

        # gradient = 2 * (a_rev * w)[n-1 : n-1+n]
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)                 # length 3n-2
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # step size
        eta = eta0 / np.sqrt(it)

        # multiplicative update (with clipping for safety)
        step = -eta * g
        step = np.clip(step, -50.0, 50.0)                   # avoid overflow
        a = a * np.exp(step)

        # project back onto simplex
        a = project_to_simplex(a)

        # periodic evaluation of the true objective
        if it % eval_interval == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]  iter {it:,}  new best {best_val:.6f}")

    # final check
    final_val = compute_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted hill‑climbing (focuses on the current worst convolution shift)
# ----------------------------------------------------------------------


def hill_climb_targeted(start_seq, deadline, max_steps: int = 800_000):
    """
    Stochastic hill‑climbing that always addresses the current worst shift
    (the one attaining max(conv)).  A fraction of the mass of the most
    contributing index is moved to a random distant index; the move is
    kept only if it improves the true score.
    """
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(best_a)

    # adaptive transfer fraction
    step_scale = 0.12
    success_counter = 0

    steps = 0
    while steps < max_steps and time.time() < deadline:
        # current convolution and worst shift
        conv = convolution_fft(a, a)
        k_star = int(np.argmax(conv))

        # compute contributions for this shift
        if k_star < n:
            # contributions = a[i] * a[i + k_star] for i = 0..n-k_star-1
            prod = a[: n - k_star] * a[k_star:]
            i_star = int(np.argmax(prod))
        else:
            # shift on the right side (symmetrical)
            shift = 2 * n - 2 - k_star
            prod = a[shift:] * a[: n - shift]
            i_star = int(np.argmax(prod)) + shift

        # propose a move from i_star to a random distant index j
        j = np.random.randint(n)
        # ensure we really move mass somewhere else
        if j == i_star:
            continue

        delta = step_scale * a[i_star]  # amount to relocate
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i_star] -= delta
        cand[j] += delta

        # enforce non‑negativity before projection
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cur_val = compute_score(cand)

        if cur_val < best_val:
            # accept
            best_val = cur_val
            best_a = cand
            a = cand
            step_scale = min(step_scale * 1.08, 0.5)
            success_counter += 1
        else:
            # reject – shrink step
            step_scale *= 0.97

        # occasional reset of step_scale if we have a long streak of failures
        if steps % 10_000 == 0 and success_counter == 0:
            step_scale = max(step_scale * 0.5, 0.02)
        if steps % 10_000 == 0:
            success_counter = 0

        steps += 1

    return best_a.tolist(), best_val
