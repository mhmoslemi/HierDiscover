def hill_climb(a, deadline, max_iter=1_000_000):
    n = a.shape[0]
    cur = a.copy()
    cur_score = fast_score(cur)
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    iteration = 0
    # set initial step size as maybe sqrt(1/n) I'll just use scale = 0.01
    init_scale = 0.02 # typical move magnitude
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        # temperature schedule: decreasing linearly from init_scale to near zero
        t = 1.0 - iteration / max_iter
        scale = init_scale * t
        i = rng.integers(n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        # propose delta
        max_delta = cur[i]
        delta = rng.uniform(-max_delta, max_delta) * scale
        # Actually use magnitude limited by prob. We can propose delta = rng.uniform(0, max_delta*scale)
        # direction random.
        delta = rng.uniform(0, max_delta * scale)
        # random direction
        if rng.random() < 0.5:
            # move from i to j
            a_prop = cur.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
        else:
            # move from j to i
            max_delta_j = cur[j]
            delta = rng.uniform(0, max_delta_j * scale)
            a_prop = cur.copy()
            a_prop[j] -= delta
            a_prop[i] += delta
        # evaluate
        sc = fast_score(a_prop)
        # accept if improvement or with probability
        if sc < cur_score or rng.random() < 0.01:  # small prob for uphill
            cur = a_prop
            cur_score = sc
            if sc < best_score:
                best = cur.copy()
                best_score = sc
        if iteration % 50000 == 0:
            print(f"    Hill climb iter={iteration:,d} best={best_score:.6f}")
    return best, best_score

"""
Propose a step sequence with low official score.

Algorithm
---------

1. Choose a modest length (n = 1024) – large enough to allow a fine
   shape, small enough for very fast convolution.

2. Initialise the vector with the classic arcsine density (the current
   state‑of‑the‑art starter).

3. Refine the starter with three complementary local optimisers,
   each respecting the simplex constraint (non‑negative, sum = 1).

   * **Sub‑gradient descent** – follows the exact gradient of the
     active convolution entry (the current max).  A simple back‑tracking
     line‑search guarantees that the true objective never increases.

   * **Pairwise hill‑climb (simulated annealing)** – repeatedly moves a
     random amount of mass from one index to another while keeping the
     total sum unchanged.  The step size is gradually cooled; occasional
     uphill moves help escape plateaus.

   * **Adam on a smooth log‑sum‑exp surrogate** – a final high‑temperature
     Adam run (λ up to 500) polishes the solution with a smooth approximation
     of the max‑convolution.  The surrogate gradient is computed analytically
     and the parameters are projected onto the simplex after each step.

4. The best vector encountered (according to the official evaluator) is
   returned.  All stages respect the global 1000 s budget and a tiny safety
   margin is kept.

The combination of an analytically good starter, a deterministic
sub‑gradient phase, a stochastic mass‑transfer walk, and a high‑precision
Adam polish consistently beats the previous baseline (≈ 1.512) and
aims for the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (fast for all n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def fast_score(a: np.ndarray) -> float:
    """Fast surrogate of the official objective (Σa = 1)."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_shape(n: int, eps: float = 0.25) -> np.ndarray:
    """Classic arcsine density (the strong baseline)."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.40,
    max_iter: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)
    n = a.shape[0]

    it = 1
    while time.time() < deadline and it <= max_iter:
        conv = convolution_fft(a, a)
        max_val = float(conv.max())
        idx = int(conv.argmax())

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        # back‑tracking line‑search (up to 12 halvings)
        for _ in range(12):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 0:
                a_new /= s
            new_val = max_convolution(a_new)

            best_a = a_new.copy()

        a = a_new
        it += 1

        if it % 50_000 == 0:
            print(f"    Sub‑grad iter={it:,d} best={best_sc:.6f}")

    return best_a, best_sc

# Pairwise hill‑climb (simulated annealing) on the simplex
# ----------------------------------------------------------------------


def hill_climb(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk that preserves Σa = 1."""
    a = init_a.copy()
    cur_sc = fast_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    n = a.size
    rng = np.random.default_rng()
    it = 0
    init_scale = 0.20  # maximal relative transfer at the start

    while time.time() < deadline and it < max_iter:
        it += 1
        # linear cooling of the transfer scale
        scale = init_scale * (1.0 - it / max_iter)

        # pick a source with positive mass
        src = rng.integers(n)
        while a[src] <= 0.0:
            src = rng.integers(n)

        dst = rng.integers(n)
        while dst == src:
            dst = rng.integers(n)

        # propose moving a random fraction of src's mass to dst
        max_transfer = a[src]
        delta = rng.uniform(0.0, max_transfer * scale)

        if rng.random() < 0.5:
            # src → dst
            prop = a.copy()
            prop[src] -= delta
            prop[dst] += delta
        else:
            # occasionally move the opposite way (using dst's mass)
            max_transfer_dst = a[dst]
            delta2 = rng.uniform(0.0, max_transfer_dst * scale)
            prop = a.copy()
            prop[dst] -= delta2
            prop[src] += delta2

        sc = fast_score(prop)

        # accept if better; otherwise accept tiny fraction of uphill moves
        if sc < cur_sc or rng.random() < 0.001:
            a = prop
            cur_sc = sc
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        if it % 50_000 == 0:
            print(f"    Hill‑climb iter={it:,d} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and Adam optimiser
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                      # soft‑max weights
    B = max_c + np.log(sum_exp) / lam           # smooth approximation of the max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.09,
    lam_start: float = 1.0,
    lam_end: float = 500.0,
    noise_every: int = 1300,
    noise_scale: float = 2e-5,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smooth surrogate."""
    a = init_a.copy()
    # keep a on the simplex
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full_like(a, 1.0 / a.size)

    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # quadratic annealing of λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        # projection onto the simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            a_new = np.full_like(a_new, 1.0 / a_new.size)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 50_000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

    print(f"  Hill‑climb (until {hill_deadline - start_time:.1f}s)…")
    a_hc, sc_hc = hill_climb(a_sg, hill_deadline)

    # 4. Final Adam polish (use whatever time is left)
    if time.time() < deadline - 0.5:
        print(f"  Final Adam polish (remaining ≈ {deadline - time.time():.1f}s)…")
        a_fin, _ = adam_optimize(
            a_hc,
            deadline,
            max_iter=200_000,
            lr0=0.08,
            lam_start=1.0,
            lam_end=500.0,
            noise_every=1500,
            noise_scale=1e-5,
        )
    else:
        a_fin = a_hc

    # --------------------------------------------------------------
    # Official evaluation of the final candidate
    # --------------------------------------------------------------
    final_val = official_score(a_fin)
    print(f"Final official score = {final_val:.6f}")

    if final_val < best_val:
        best_val = final_val
        best_seq = a_fin.tolist()
        print("*** New global best from this run ***")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = official_score(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
