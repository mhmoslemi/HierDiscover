"""
Optimiser for the max‑convolution / Σ² problem.

Main ideas
---------
* Use a *large* sequence length (up to 8192).  The theoretical lower bound
  2·n/(2·n‑1) ≈ 1.0001 for n=8192 is far below the target 1.503, so a good
  flat self‑convolution can easily beat it.
* Initialise the shape with the classic arcsine (U‑shaped) distribution,
  which is already close to the constant‑convolution ideal.
* Refine the shape in three stages, each respecting the probability simplex:
    – Adam optimiser on the L₂ loss between the self‑convolution and the flat
      target (the “flattening” stage).  This drives the whole convolution
      towards a constant.
    – Projected sub‑gradient descent on the true max‑convolution (the
      objective of `evaluate_sequence`).  After flattening, the max is already
      low, and sub‑gradient steps give a final push.
    – A cheap stochastic Gaussian walk on the simplex to escape tiny local
      irregularities.
* All stages are *time‑driven*: they receive the global deadline and stop
  early if needed, guaranteeing a non‑empty return (uniform 256‑vector) even
  when the budget is exhausted.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of `a`."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official score for a vector already on the simplex.
    The official `evaluate_sequence` returns 2·n·max(conv)/sum², and for a on the
    simplex sum=1 this reduces to 2·n·max(conv).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) – O(n log n).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine (U‑shaped) density on {0,…,n‑1}.
    The formula approximates the Beta(½,½) distribution on a discrete grid.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Flattening (L₂) optimisation
# ----------------------------------------------------------------------


def flatten_gradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Gradient of ‖conv(a,a) – avg‖₂² where avg = 1/(2n‑1) and Σa = 1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)               # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient: 4·(a ⊛ diff_rev)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)    # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def adam_flatten(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.35,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the L₂ flattening loss.
    Returns the best vector (projected) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        grad, loss = flatten_gradient(a)

        # Adam update (learning rate decays like 1/√it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = project_to_simplex(a - step)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 50000 == 0:
            print(
                f"[Flatten] iter={it:,d} loss={loss:.3e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.30,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Simple back‑tracking line‑search guarantees non‑increase.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # Gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        # For i where both terms exist we get a[j] + a[i]; the simple
        # implementation below captures the symmetric contribution via the mask.
        grad[mask] = 2.0 * a[j[mask]]

        # Back‑tracking line‑search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_candidate = project_to_simplex(a - step * grad)
            new_max = np.max(convolution_fft(a_candidate, a_candidate))
            if new_max < max_val - 1e-14:
                break
            step *= 0.5

        a = a_candidate
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 60000 == 0:
            print(
                f"[Sub‑grad] iter={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap stochastic Gaussian walk on the simplex
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move acceptance probability.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = project_to_simplex(a + noise)

        cur_sc = compute_score(a_prop)

        # Accept improvements; occasionally accept uphill moves to escape flat spots
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 80000 == 0:
            print(
                f"[Random walk] iter={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc
