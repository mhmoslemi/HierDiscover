    scale = np.sqrt(2.0 * n) / sumg

"""
LP‑based alternating optimisation + stochastic polishing.

We minimise the score returned by ``evaluate_sequence`` which (up to a
scale‐invariant factor) is simply the maximum entry of the self‑convolution
of a non‑negative vector a, provided we keep the ℓ1‑norm fixed to
√(2·n).  The algorithm proceeds as follows:

1️⃣  **Initialisation** – a random Dirichlet draw (concentration α
    uniformly chosen in [0.1,5]) is scaled to ℓ1‑norm √(2·n).

2️⃣  **Linear‑programming step** – for the current vector f we compute
    M = max(conv(f,f)).  The convolution conv(f,g) is linear in g, so we
    solve the LP

        maximise   Σ g_i
        subject to conv(f,g) ≤ M   (elementwise)
                   g_i ≥ 0

    The solution g is then rescaled to norm √(2·n) and mixed with f:
    f ← (1‑t)·f + t·g  (t≈0.2).  Because both f and g respect the bound M,
    the mixed vector cannot increase the objective.  This step is repeated
    until the allotted time for the restart expires.

3️⃣  **Stochastic hill‑climbing** – the best vector found by the LP
    phase is polished by repeatedly transferring a fraction of the mass
    from one index to another.  Moves that lower the true maximum
    convolution are kept; the transfer size is adapted to the success
    rate.

Multiple restarts are performed until the global time budget (≈1000 s)
is exhausted.  The best sequence encountered is finally returned.
"""

import time

# ----------------------------------------------------------------------
# FFT based linear convolution (fast, uses real‑FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT.
    Length of the result is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_self_convolution(a: np.ndarray) -> float:
    """Return max_{k} (a*a)[k] .  a is assumed to be a 1‑D array."""
    return float(np.max(convolution_fft(a, a)))


# ----------------------------------------------------------------------
# Helper: project on the probability simplex (used only by the fallback).
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# LP step (requires cvxpy).  Returns a new vector g with ℓ1‑norm √(2n)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, sqrt2n: float) -> np.ndarray:
    """
    Solve  max Σ g_i   s.t.  conv(f,g) ≤ M , g ≥ 0.
    The returned g is scaled to have ℓ1‑norm sqrt(2·n).
    """
    n = f.shape[0]

    # Build the full convolution matrix T (size (2n-1) × n)
    # T @ g == conv(f, g)
    rows = 2 * n - 1
    T = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = f  # each column i is f placed at offset i

    # Linear program
    g = cp.Variable(n, nonneg=True)
    constraints = [T @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # Solve – ECOS works well for dense problems of this size
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-9, reltol=1e-9)
    except Exception:  # pragma: no cover
        # any solver trouble – fall back to the input vector
        return f

    if g.value is None:
        return f

    g_val = np.asarray(g.value, dtype=np.float64)
    s = np.sum(g_val)
    if s <= 0:
        return f
    scale = sqrt2n / s
    return g_val * scale


# ----------------------------------------------------------------------
# Stochastic mass‑transfer hill‑climber (optimises max self‑convolution)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float, max_moves: int = 500_000):
    """
    Simple hill‑climber that moves a fraction of mass from one index to
    another whenever it reduces the true objective (max self‑convolution).
    """
    n = a.shape[0]
    sqrt2n = math.sqrt(2.0 * n)
    best = a.copy()
    best_val = max_self_convolution(best)

    step_scale = 0.15
    moves = 0
    while moves < max_moves and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a fraction of the mass from i to j
        delta = step_scale * best[i]
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep feasibility
        cand = np.maximum(cand, 0.0)
        # renormalise (should stay √(2n) up to rounding)
        cand = cand * (sqrt2n / np.sum(cand))
        val = max_self_convolution(cand)
        if val < best_val - 1e-12:
            best_val = val
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.98
        moves += 1
    return best, best_val

# The main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.

    The routine performs several random restarts, each consisting of a
    fast LP‑based alternating optimisation (if cvxpy is available) followed
    by a short stochastic hill‑climbing polishing phase.  The best vector
    seen over all restarts is returned.
    """
    np.random.seed(seed)
    start = time.time()
    # small safety margin for the harness (keep a few seconds free)
    deadline = start + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose a length.  Prefer the length of a provided fallback,
    # otherwise default to 2048 (a good trade‑off between quality and speed).
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2048

    sqrt2n = math.sqrt(2.0 * n)

    best_seq = None
    best_val = float("inf")   # stores max(conv) i.e. the true objective

    # --------------------------------------------------------------
    # If a fallback sequence is present, evaluate it immediately.
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        fb_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_val:.6f}")
        # convert to the optimisation scale (ℓ1‑norm √(2n))
        fb_arr = np.array(fallback, dtype=float)
        fb_arr = fb_arr * (sqrt2n / np.sum(fb_arr))
        fallback_obj = max_self_convolution(fb_arr)
        if fallback_obj < best_val:
            best_val = fallback_obj
            best_seq = fb_arr.tolist()
    except Exception:
        pass

    # --------------------------------------------------------------
    # Helper for a single restart
    # --------------------------------------------------------------
    def _run_restart():
        nonlocal best_seq, best_val

        # random Dirichlet start – concentration α controls sparsity
        alpha = np.random.uniform(0.1, 5.0)
        init = np.random.dirichlet(np.full(n, alpha))
        # scale to ℓ1‑norm √(2n)
        f = init * (sqrt2n / np.sum(init))

        # ----------  LP‑based alternating optimisation ----------
        iter_deadline = time.time() + max(8.0, (deadline - time.time()) * 0.6)
        t_mix = 0.20                     # convex‑combination weight
        best_local = f.copy()
        best_local_val = max_self_convolution(f)

        while time.time() < iter_deadline:
            M = max_self_convolution(f)

            # solve the LP; if cvxpy is missing we break out early
            if not _CVXPY_AVAILABLE:
                break
            g = _lp_step(f, M, sqrt2n)

            # convex combination – guarantees non‑increase of the bound
            f = (1.0 - t_mix) * f + t_mix * g
            # renormalise to keep the ℓ1‑norm exact (numerical drift)
            f = f * (sqrt2n / np.sum(f))

            cur_val = max_self_convolution(f)
            if cur_val < best_local_val - 1e-12:
                best_local_val = cur_val
                best_local = f.copy()
                print(f"[{time.time():.1f}] LP iter new local best M={cur_val:.6f}")

            # simple schedule for the mixing weight – slowly shrink
            t_mix = max(t_mix * 0.995, 0.02)

        # keep the best from this restart
        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = best_local.tolist()
            print(f"*** Global best improved to M={best_val:.6f}")

        # ----------  final hill‑climb polishing ----------
        polish_deadline = deadline - 2.0
        if time.time() < polish_deadline:
            polished, polished_val = _hill_climb(np.array(best_seq, dtype=float),
                                                deadline=polish_deadline,
                                                max_moves=400_000)
            if polished_val < best_val:
                best_val = polished_val
                best_seq = polished.tolist()
                print(f"Hill‑climb polishing improved global M to {best_val:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop – keep launching restarts until time runs out.
    # --------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline - 5.0:   # leave a few seconds for final output
        restart_counter += 1
        print(f"[{time.time():.1f}] Starting restart {restart_counter}")
        _run_restart()

    # --------------------------------------------------------------
    # Safety net – if nothing succeeded, fall back to a uniform vector.
    # --------------------------------------------------------------
    if best_seq is None:
        uniform = np.full(n, 1.0 / n)
        best_seq = (uniform * (sqrt2n / np.sum(uniform))).tolist()
        best_val = max_self_convolution(np.array(best_seq, dtype=float))

    # Final reporting (the harness will call evaluate_sequence on the result)
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
