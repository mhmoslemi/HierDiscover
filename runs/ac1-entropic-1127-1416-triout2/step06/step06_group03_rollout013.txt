def mirror_descent_optimize(a_init: np.ndarray, deadline: float, step0: float = 0.1) -> np.ndarray:
    a = a_init.copy()
    n = a.size
    step = step0
    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = np.max(conv)
        # tolerance for max entries
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # subgrad
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[n-1:n-1+n]

        # Exponential update (mirror descent)
        # a_i' ∝ a_i * exp(-step * subgrad_i)
        # Avoid underflow/overflow: subtract max
        update = np.exp(-step * subgrad)
        a = a * update
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # adapt step: decrease gradually
        step *= 0.999  # meets convergence
    return a

    * a smooth‑max surrogate (log‑sum‑exp) is minimised by Adam while
      the temperature β is increased from 5 up to 300.
    * the parameters are the soft‑max of an unconstrained vector, so the
      simplex and non‑negativity constraints are satisfied automatically.

3.  After Adam, a mirror‑descent step (exponential‑gradient) uses the
    exact sub‑gradient of the true max‑convolution to push the solution
    further down.

4.  A final projected sub‑gradient descent (the same routine used in
    the previous state‑of‑the‑art) polishes the best sequence found.

All phases respect the global time budget (default 1000 s).  The routine
always returns a valid non‑negative sequence; if nothing better is
found a simple uniform sequence is used as a safe fallback.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate used for pre‑selection (scale‑invariant)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/sum(a)^2 (no clipping, cheap)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Analytic generators for diverse starting shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    tot = base.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return base / tot


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    tot = vals.sum()
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / tot


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max, used for final polish)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.07,
) -> np.ndarray:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))/sum(a)^2."""
    a = _project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient of the max‑convolution (see comments above)
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = _project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    Sub‑grad it={it:,d}  score≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max + Adam optimiser (continuation on β)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning probabilities that sum to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 300.0,
    ramp_iters: int = 40_000,
    lr: float = 0.04,
) -> np.ndarray:
    """
    Minimise the smooth approximation  L_β(a) = log‑sum‑exp(β·conv(a,a))
    while keeping a on the simplex via the soft‑max map.
    β grows linearly from β0 to β_max (continuation).
    """
    z = np.array(z0, dtype=float, copy=True)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = _softmax(z)                     # a ≥0 , Σa = 1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linear schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2·(a ★ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = 3·n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5_000 == 0:
            print(f"    Adam it={t:,d}  β={beta:.1f}  score≈{fast_score(a):.6f}")

    return _softmax(z)


# ----------------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) with the exact max‑subgradient
# ----------------------------------------------------------------------


def mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.12,
    decay: float = 0.9995,
) -> np.ndarray:
    """
    Exponential‑gradient descent on f(a)=2·n·max(conv(a,a))/sum(a)^2.
    The iterate stays on the simplex automatically.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    step = step0

    while time.time() < deadline:
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        # multiplicative update (mirror step w.r.t. KL divergence)
        factor = np.exp(-step * subgrad)
        a = a * factor
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        step *= decay
        # occasional debug
        if int(step * 1e6) % 100_000 == 0:
            print(f"    Mirror step={step:.6f}  score≈{fast_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The routine combines analytical seeds,
    a continuation Adam phase, an exponential‑gradient mirror step and
    a final projected sub‑gradient polish.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # --------------------------------------------------------------
    # (optional) use a previously discovered sequence if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Scan a few lengths (larger first)
    # --------------------------------------------------------------
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ----------------------------------------------------------
        # 1a. Generate a modest pool of analytically‑motivated seeds
        # ----------------------------------------------------------
        seed_vectors: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            generate_mixture_shape(n, 0.5),
        ]

        # systematic sweep over the power exponent p
        for p in np.linspace(0.2, 2.0, 12):
            seed_vectors.append(generate_power_shape(n, p, eps=1e-8))

        # random beta‑family samples
        for _ in range(6):
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            eps = rng.uniform(0.0, 0.01 * n)
            seed_vectors.append(generate_beta_shape(n, a, b, eps))

        # keep the most promising (according to the fast surrogate)
        scored = [(fast_score(vec), vec) for vec in seed_vectors]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:3]]   # three per length at most

        # ----------------------------------------------------------
        # 1b. Allocate a slice of the remaining budget for each seed
        # ----------------------------------------------------------
        per_seed_budget = max(2.0, (deadline - time.time()) / (len(top_seeds) * 2))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 3.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_budget:.1f}s")
            seed_deadline = time.time() + per_seed_budget

            # initialise Adam via log‑softmax (avoids division by zero)
            z0 = np.log(seed_vec + 1e-12)

            # ------------------------------------------------------
            # 2. Continuation Adam optimisation (smooth‑max)
            # ------------------------------------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=300.0,
                ramp_iters=40_000,
                lr=0.04,
            )

            # ------------------------------------------------------
            # 3. Official evaluation of the Adam result
            # ------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ------------------------------------------------------
            # 4. Mirror‑descent refinement (exact max sub‑gradient)
            # ------------------------------------------------------
            if time.time() < deadline - 2.0:
                md_deadline = deadline - 1.0
                a_mirror = mirror_descent_optimize(
                    np.array(best_seq, dtype=float),
                    md_deadline,
                    step0=0.12,
                    decay=0.9995,
                )
                try:
                    val2 = evaluate_sequence(a_mirror.tolist())
                    print(f"   Mirror‑refined score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, a_mirror.tolist()
                        print(f"   *** Updated best after mirror step: {best_val:.6f} ***")
                except Exception:
                    pass

    # --------------------------------------------------------------
    # 5. Final global polishing with projected sub‑gradient descent
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=200_000,
            step0=0.07,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 6. Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
