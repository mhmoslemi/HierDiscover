    best_a = a.copy()

    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment

            break

        # compute convolution conv = a*a via FFT using length L
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[:2*n-1]

        beta = min(beta_max, beta0 * np.sqrt(t))

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. Choose a length `n` (2048 by default).  
2. Perform several random Dirichlet restarts.  For each restart:
   * A smooth “soft‑max’’ approximation of the max‑convolution is minimised
     using an Adam‑style exponentiated‑gradient descent (mirror descent
     on the probability simplex).  The temperature β is increased as √t,
     so the optimiser gradually focuses on the true ∞‑norm.
   * The obtained vector is further refined by a fast *incremental*
     hill‑climber that moves a small amount of mass from one entry to
     another while updating the self‑convolution in O(n) time (no FFTs).
3. The best sequence over all restarts is finally polished by a longer
   incremental hill‑climb.
4. The best sequence found is returned.  All work respects the 1000 s
   time budget; the score is printed with `evaluate_sequence` for the
   harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution, real‑valued)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact score (the exact objective when Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute 2·n·max(a*a) for a vector whose entries sum to 1.
    This is exactly the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Adam‑style smooth optimisation (soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_optimize_adam(
    a: np.ndarray,
    time_deadline: float,
    max_iters: int = 150_000,
    beta0: float = 0.5,
    beta_max: float = 200.0,
    eta0: float = 0.8,
    beta1: float = 0.9,
    beta2: float = 0.999,
    epsilon: float = 1e-8,
    report_every: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max approximation of the self‑convolution using an
    Adam update on the exponential‑gradient (mirror) parametrisation.
    Returns the best vector seen and its exact score.
    """
    n = a.shape[0]
    # length needed for all FFTs (a, a_rev, w)
    L = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_val = fast_score(best_a)

    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline - 1.0:
            break

        # ----- a*a ----------------------------------------------------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]

        max_c = np.max(conv)

        # ----- soft‑max weights ---------------------------------------
        beta = min(beta_max, beta0 * np.sqrt(t))
        exp_terms = np.exp(beta * (conv - max_c))  # stabilised exponentials
        w = exp_terms / np.sum(exp_terms)

        # ----- gradient = 2·(w ⋆ a_rev) -------------------------------
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        # slice the part that corresponds to the original length
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- Adam exponential‑gradient update ----------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = eta0 / np.sqrt(t)

        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + epsilon))
        a = project_to_simplex(a)

        # ----- occasional reporting ------------------------------------
        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  ADAM iter {t:8d} – new best score {cur_val:.6f}")

    # final check
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Incremental hill‑climber (mass‑transfer moves, O(n) update)
# ----------------------------------------------------------------------


def hill_climb_incremental(
    seq,
    time_deadline,
    max_steps: int = 300_000,
    init_step_scale: float = 0.2,
) -> tuple[list[float], float]:
    """
    Fast hill‑climber that repeatedly transfers a fraction of mass from one
    position to another, updating the self‑convolution incrementally
    (no FFTs). Only moves that improve the exact score are kept.
    Returns the best sequence encountered and its score.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    # guarantee feasibility
    a = project_to_simplex(a)

    conv = convolution_fft(a, a)          # exact linear convolution
    cur_score = 2.0 * n * float(np.max(conv))

    best_a = a.copy()
    best_score = cur_score

    step_scale = init_step_scale
    steps = 0

    while steps < max_steps and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        ai = a[i]
        if ai <= 0.0:
            continue

        # random fraction of the mass at i (small but variable)
        delta = step_scale * ai * np.random.rand()
        if delta == 0.0:
            continue

        # store the old vector for a cheap revert (only two entries change)
        aj = a[j]

        # keep a copy of the *old* whole vector for the convolution update
        a_old = a.copy()

        # ----- apply the move in the primal space --------------------
        a[i] = ai - delta
        a[j] = aj + delta

        # ----- update convolution incrementally -----------------------
        # conv_i contribution is removed, conv_j contribution added
        conv[i : i + n] -= 2.0 * delta * a_old
        conv[j : j + n] += 2.0 * delta * a_old
        # quadratic correction terms
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta

        # ----- compute new score ------------------------------------
        new_max = float(np.max(conv))
        new_score = 2.0 * n * new_max

        # ----- acceptance test ---------------------------------------
        if new_score < cur_score:
            cur_score = new_score
            # successful step ⇒ increase step size a little
            step_scale = min(step_scale * 1.03, 0.5)

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # reject ⇒ revert both a and conv
            a[i] = ai
            a[j] = aj
            # reverse the convolution changes
            conv[i : i + n] += 2.0 * delta * a_old
            conv[j : j + n] -= 2.0 * delta * a_old
            conv[2 * i] -= delta * delta
            conv[2 * j] -= delta * delta
            conv[i + j] += 2.0 * delta * delta
            # shrink step scale after a failure
            step_scale = max(step_scale * 0.97, 1e-4)

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The search consists of:
      * several random Dirichlet restarts,
      * an Adam‑based smooth optimisation of the soft‑max of the convolution,
      * an incremental hill‑climber for fine‑tuning,
      * a final polishing hill‑climb.
    All work respects the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Try the provided fallback (if any)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val = fb_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Choose a length (2048 works well; shrink if time is tight)
    # --------------------------------------------------------------
    n = 2048
    if time.time() + 5.0 > deadline:
        n = 1024
    print(f"Using sequence length n = {n}")

    # --------------------------------------------------------------
    # 2️⃣  Random restarts with Adam‑based smooth optimisation
    # --------------------------------------------------------------
    # Number of restarts is adapted to the remaining budget
    remaining = deadline - time.time()
    num_restarts = max(1, int(remaining // 130.0))   # rough 130 s per restart
    num_restarts = min(num_restarts, 8)             # cap to keep things reasonable
    print(f"Planned restarts: {num_restarts}")

    for restart in range(num_restarts):
        if time.time() >= deadline - 30.0:
            break

        # Dirichlet start – random concentration encourages diversity
        alpha = np.random.uniform(0.2, 2.0)
        a0 = np.random.gamma(alpha, 1.0, size=n)
        a0 = a0 / a0.sum()

        init_score = fast_score(a0)
        if init_score < best_val:
            best_val = init_score
            best_seq = a0.tolist()
            print(f"Restart {restart}: initial score = {init_score:.6f}")

        # ---- smooth Adam optimisation ---------------------------------
        iters = int( (deadline - time.time() - 20.0) // num_restarts )  # split remaining time
        iters = max(30_000, min(iters, 200_000))
        refined_a, refined_score = smooth_optimize_adam(
            a0,
            time_deadline=deadline - 20.0,
            max_iters=iters,
            beta0=0.5,
            beta_max=200.0,
            eta0=0.8,
            report_every=20_000,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Restart {restart}: refined score = {refined_score:.6f}")

        # ---- short incremental hill‑climb ----------------------------
        polished_a, polished_score = hill_climb_incremental(
            refined_a,
            time_deadline=deadline - 5.0,
            max_steps=120_000,
            init_step_scale=0.15,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_a
            print(f"Restart {restart}: hill‑climb improved to {polished_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final long hill‑climb polishing (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nFinal polishing hill‑climb")
        final_a, final_score = hill_climb_incremental(
            best_seq,
            time_deadline=deadline,
            max_steps=300_000,
            init_step_scale=0.2,
        )
        if final_score < best_val:
            best_val = final_score
            best_seq = final_a
            print(f"Final polishing improved best to {final_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Safeguard – tiny uniform vector if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 5️⃣  Report the final score and return the sequence
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
