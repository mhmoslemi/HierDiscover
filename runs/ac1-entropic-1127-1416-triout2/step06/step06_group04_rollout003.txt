"""
Propose a high‑quality step‑function sequence.

The search works on a single (large) length, explores a few
parameterised families (uniform, arcsine, triangular, Hann,
random beta mixtures), enforces symmetry (a_i = a_{n‑1‑i})
and projects onto the probability simplex after every update.
The optimisation pipeline for each start consists of:

1️⃣  Optional L2‑flattening (Adam on ‖conv(a,a) – avg‖₂)
2️⃣  Annealed log‑sum‑exp surrogate minimisation (Adam)
3️⃣  Projected sub‑gradient descent on the true max‑convolution
4️⃣  A cheap stochastic Gaussian walk on the simplex.

All stages respect a global 1000 s budget and stop early if needed.
The best vector found (according to the official ``evaluate_sequence``)
is returned.
"""

import time
from typing import List, Tuple
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT, length = len(a)+len(b)-1."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate “2·n·max(conv)”; assumes Σa = 1."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = 1} (Duchi et al.)."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n‑1‑i} while preserving the simplex sum."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Unnormalised symmetric beta density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta densities (renormalised)."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) density on a discrete grid."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular bump (max at centre)."""
    i = np.arange(n, dtype=float)
    vals = (i + 1) * (n - i)
    vals /= vals.sum()
    return vals


def hann_shape(n: int) -> np.ndarray:
    """Discrete Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=float)
    vals = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# L2‑flattening loss and gradient
# ----------------------------------------------------------------------


def flatten_gradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Gradient of ‖conv(a,a) – avg‖₂² where avg = 1/(2n‑1), Σa = 1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient = 4 * (a ⊛ diff_rev)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)    # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def adam_flatten(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.25,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the L₂ flattening loss.
    Returns the best vector (projected & symmetric) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        grad, loss = flatten_gradient(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_to_simplex(a_new)
        a_new = symmetrize(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 20000 == 0:
            print(
                f"    Adam‑flatten it={it:,d} loss={loss:.2e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth surrogate of the max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate for max(conv(a,a)).
    Returns (grad, surrogate B) with
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv – max)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)               # length 2n‑1
    max_c = np.max(conv)

    # log‑sum‑exp in a numerically stable way
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                     # soft‑max weights over conv entries
    B = max_c + np.log(sum_exp) / lam

    # Weighted convolution gradient (2 * (a ⊛ w_rev))
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)                               # =1 after projection
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_max(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 2000.0,
    noise_every: int = 1000,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution surrogate.
    Returns the best vector (projected & symmetric) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # quadratic temperature schedule (slow at the start, fast at the end)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)
        a_new = symmetrize(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam‑max it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} λ={lam:.1f}"
            )
    return best_a, best_sc

def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Simple back‑tracking line‑search guarantees non‑increase.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # Sub‑gradient w.r.t. the active index idx
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        # back‑tracking
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            a_new = symmetrize(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc

def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    a = symmetrize(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)
        a_prop = symmetrize(a_prop)

        cur_sc = compute_score(a_prop)

        # accept if better; otherwise accept a tiny uphill move with prob 1e‑4
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50000 == 0:
            print(
                f"    Random walk it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc

                )
                remaining = deadline - time.time()
                if remaining < 5.0:
                    break

            # Stage C – exact sub‑gradient descent
            if time.time() < deadline - 6.0:
                remaining = deadline - time.time()
                it_sub = int(min(180_000, remaining * 28))
                print(f"    Sub‑gradient (max_iter={it_sub}) …")
                a_cur, _ = subgradient_optimize(
                    a_cur,
                    time_deadline=deadline,
                    step0=0.25,
                    max_iter=it_sub,
                )
                remaining = deadline - time.time()
                if remaining < 4.0:
                    break

            # Stage D – cheap stochastic walk
            if time.time() < deadline - 3.0:
                remaining = deadline - time.time()
                it_walk = int(min(220_000, remaining * 35))
                print(f"    Random walk (max_iter={it_walk}) …")
                a_cur, _ = random_local_search(
                    a_cur,
                    time_deadline=deadline,
                    init_step=0.012,
                    max_iter=it_walk,
                )

            # ------------------------------------------------------------------
            # 1c.  Official evaluation of the candidate
            # ------------------------------------------------------------------
            val = evaluate_sequence(a_cur.tolist())
            print(f"    → official score (len={n}) = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = a_cur.tolist()
                print(f"    *** New global best (len={n}) = {best_val:.6f} ***")

            # Stop early if we have already reached the target bound
            if best_val < 1.503:
                print("Target bound achieved – exiting early.")
                return best_seq

    # ------------------------------------------------------------------
    # 2.  Global polishing of the overall best vector (if time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # Global flattening (long run)
        remaining = deadline - time.time()
        it_flat = int(min(200_000, remaining * 20))
        print(f"  Global flattening (max_iter={it_flat}) …")
        a, _ = adam_flatten(
            initial_a=a,
            time_deadline=deadline,
            max_iter=it_flat,
            lr0=0.35,
        )

        # Global high‑temperature Adam on the max surrogate
        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            it_adam = int(min(400_000, remaining * 30))
            print(f"  Global Adam‑max (max_iter={it_adam}) …")
            a, _ = adam_max(
                initial_a=a,
                time_deadline=deadline,
                max_iter=it_adam,
                lr0=0.09,
                lam_start=1.0,
                lam_end=3000.0,
                noise_every=1500,
                noise_scale=4e-5,
            )

        # Global exact sub‑gradient descent
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            it_sub = int(min(200_000, remaining * 28))
            print(f"  Global sub‑gradient (max_iter={it_sub}) …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.22,
                max_iter=it_sub,
            )

        # Final random walk
        if time.time() < deadline - 1.0:
            remaining = deadline - time.time()
            it_walk = int(min(250_000, remaining * 35))
            print(f"  Global random walk (max_iter={it_walk}) …")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.010,
                max_iter=it_walk,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
