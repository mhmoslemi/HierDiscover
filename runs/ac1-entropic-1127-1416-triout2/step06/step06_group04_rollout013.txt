"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a streamlined version of the previous “beta‑sampling +
Adam + sub‑gradient” pipeline, with a few key changes that usually push the
official score a little lower:

* **Symmetrisation** – the optimum is empirically symmetric, so after every
  projection we replace the vector by (a+rev(a))/2.  This halves the effective
  dimensionality and improves the quality of the gradient steps.

* **Longer annealed Adam on the smooth max‑surrogate** – a log‑sum‑exp
  approximation of the max is used, its temperature λ is increased
  exponentially from 1 to 2000, and the Adam run is allowed to consume as
  many iterations as the remaining budget permits.

* **More aggressive flattening** – the L²‑flattening stage runs for many more
  iterations (controlled only by the time budget) before the max‑stage.

* **Extended final polishing** – after the main Adam run we perform a
  projected sub‑gradient descent (exact max) and a cheap Gaussian walk,
  both with generous iteration limits.

* **Multiple lengths** – we try a few lengths (1024, 2048, 4096, 8192) and keep
  the best official score seen.  Larger lengths give a smaller theoretical
  lower bound (≈1) and often improve the result.

All steps respect the global 1000 s wall‑clock budget; a uniform 256‑vector
is returned as a safety net if the budget expires before any better sequence
is found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT‑based convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate score used during optimisation.
    Vector a is assumed to lie on the simplex (sum = 1).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al.
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, Σx = 1}."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry: (v + rev(v))/2."""
    return (v + v[::-1]) * 0.5


def project_and_symmetrize(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex and enforce mirror symmetry."""
    v = project_to_simplex(v)
    v = symmetrize(v)
    # projection may disturb the sum by a tiny amount due to rounding;
    # enforce the simplex again.
    return project_to_simplex(v)


# ----------------------------------------------------------------------
# Shape generators – beta families (used for initial sampling)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Unnormalised symmetric beta‑family density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) density."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Flattening loss (L2 distance to the flat target) and its gradient
# ----------------------------------------------------------------------


def flatten_gradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Gradient of ‖conv(a,a) – avg‖₂² where avg = 1/(2n‑1) and Σa = 1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient: 4 * (a ⊛ diff_rev)  (⊛ = convolution)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)          # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def adam_flatten(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.30,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the L₂ flattening loss.
    The vector is kept symmetric and on the simplex.
    """
    a = project_and_symmetrize(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        grad, loss = flatten_gradient(a)

        # Adam update (learning‑rate decays ∝ 1/√it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_and_symmetrize(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 50000 == 0:
            print(
                f"[flatten] it={it:,d} loss={loss:.2e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth surrogate for max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value B) where

        B = max(conv) + (1/lam)·log Σ exp(lam·(conv‑max)) .

    The gradient is the exact gradient of B (see the code comments below).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights over convolution entries
    B = max_c + np.log(sum_exp) / lam

    # Weighted sum of sub‑gradients:
    #   ∂/∂a Σ_k w_k (a*a)_k = 2 * (a ⊛ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_max(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 500_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 2000.0,
    noise_every: int = 1000,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution surrogate.
    The temperature λ grows exponentially from lam_start to lam_end.
    """
    a = project_and_symmetrize(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # exponential schedule: λ = lam_start * (lam_end/lam_start) ** (t)
        t = it / max_iter
        lam = lam_start * (lam_end / lam_start) ** t

        grad, B = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_and_symmetrize(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 50000 == 0:
            print(
                f"[adam‑max] it={it:,d} lam={lam:.1f} B={B:.6e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max (active shift only)
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on the max‑convolution.
    The active shift (where the maximum is attained) provides a cheap sub‑gradient.
    """
    a = project_and_symmetrize(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # sub‑gradient of the active term: (a * rev_shift)_i = a_{idx-i}
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # back‑tracking line‑search (step ∝ 1/√it)
        step = step0 / np.sqrt(it)
        for _ in range(8):
            a_new = a - step * grad
            a_new = project_and_symmetrize(a_new)
            if max_convolution(a_new) < max_val - 1e-15:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 60000 == 0:
            print(
                f"[sub‑grad] it={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f} max_val={max_val:.6e}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Tiny stochastic Gaussian walk (final polishing)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Very cheap Gaussian perturbation on the simplex (with symmetry enforced).
    Accepts any improvement; occasionally accepts a slight uphill move to
    escape shallow plateaus.
    """
    a = project_and_symmetrize(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_and_symmetrize(a_proposed)

        cur_sc = compute_score(a_proposed)

        # strict improvement, or a tiny uphill chance
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"[random‑walk] it={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point – the function called by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 12345,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine explores several lengths, creates a handful of beta‑shape
    seeds, then performs a long flattening + annealed‑Adam + sub‑gradient
    + random‑walk polishing pipeline.  All vectors are kept symmetric
    (mirror‑symmetry) to accelerate convergence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.8   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  If a previously discovered sequence exists, use it as a fallback
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] known sequence official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Explore a few lengths (increasing order).  The longer the length,
    #     the lower the theoretical bound, but also the more expensive.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break

        print(f"\n=== Length n = {n} ===")
        # ------------------- initial candidates -------------------------
        candidates: List[np.ndarray] = []

        # deterministic starters
        candidates.append(np.full(n, 1.0 / n))          # uniform
        candidates.append(arcsine_initial(n))          # classic arcsine

        # grid of symmetric beta shapes (α = β)
        for alpha in np.linspace(0.2, 0.8, 12):
            cand = generate_beta_shape(n, alpha, alpha, eps=1e-6)
            candidates.append(cand)

        # a handful of random beta shapes (including mixtures)
        for _ in range(200):
            if time.time() >= deadline - 30.0:
                break
            if rng.random() < 0.33:  # mixture
                w = rng.random()
                a1 = generate_beta_shape(
                    n, rng.uniform(0.02, 3.5), rng.uniform(0.02, 3.5), eps=1e-5
                )
                a2 = generate_beta_shape(
                    n, rng.uniform(0.02, 3.5), rng.uniform(0.02, 3.5), eps=1e-5
                )
                cand = w * a1 + (1.0 - w) * a2
            else:
                cand = generate_beta_shape(
                    n,
                    rng.uniform(0.02, 3.5),
                    rng.uniform(0.02, 3.5),
                    eps=rng.uniform(1e-8, 0.02 * n),
                )
            candidates.append(cand)

        # ------------------- keep a few promising seeds ---------------
        best_local_sc = float("inf")
        seed_vectors: List[np.ndarray] = []

        for vec in candidates:
            vec = project_and_symmetrize(vec)
            sc = compute_score(vec)
            if sc < best_local_sc:
                best_local_sc = sc
                seed_vectors = [vec]
            elif sc < best_local_sc * 1.05:   # within 5 % of the best
                seed_vectors.append(vec)

        print(f"  Found {len(seed_vectors)} seed vectors (best fast score = {best_local_sc:.6f})")

        # ------------------------------------------------------------------
        # 2.  Polishing each seed (the heavy work).  We allocate the time
        #     proportionally to the remaining budget.
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(seed_vectors):
            if time.time() >= deadline - 10.0:
                break
            print(f"\n  ++ Polishing seed {idx + 1}/{len(seed_vectors)} for n={n}")

            # ---- flattening stage -------------------------------------------------
            remaining = deadline - time.time()
            max_iter_flat = int(min(180_000, remaining * 22))
            a_flat, _ = adam_flatten(
                initial_a=seed_vec,
                deadline=deadline,
                max_iter=max_iter_flat,
                lr0=0.35,
            )
            # ---- annealed Adam on smooth max ---------------------------------------
            remaining = deadline - time.time()
            max_iter_adam = int(min(300_000, remaining * 25))
            a_max, _ = adam_max(
                initial_a=a_flat,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.12,
                lam_start=1.0,
                lam_end=2000.0,
                noise_every=1200,
                noise_scale=4e-5,
            )
            # ---- exact sub‑gradient descent ----------------------------------------
            remaining = deadline - time.time()
            max_iter_sub = int(min(200_000, remaining * 30))
            a_sub, _ = subgradient_optimize(
                a_max,
                deadline=deadline,
                step0=0.28,
                max_iter=max_iter_sub,
            )
            # ---- final stochastic walk ---------------------------------------------
            remaining = deadline - time.time()
            max_iter_walk = int(min(200_000, remaining * 30))
            a_final, _ = random_local_search(
                a_sub,
                deadline=deadline,
                init_step=0.014,
                max_iter=max_iter_walk,
            )

            # ---- official evaluation ------------------------------------------------
            val = evaluate_sequence(a_final.tolist())
            print(f"   → official score for this seed = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = a_final.tolist()
                print(f"   *** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # a) flattening (long)
        remaining = deadline - time.time()
        max_iter_flat = int(min(250_000, remaining * 20))
        a, _ = adam_flatten(
            initial_a=a,
            deadline=deadline,
            max_iter=max_iter_flat,
            lr0=0.40,
        )

        # b) annealed Adam (high λ)
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_adam = int(min(400_000, remaining * 25))
            a, _ = adam_max(
                initial_a=a,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.10,
                lam_start=1.0,
                lam_end=3000.0,
                noise_every=1500,
                noise_scale=3e-5,
            )

        # c) sub‑gradient descent
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(250_000, remaining * 30))
            a, _ = subgradient_optimize(
                a,
                deadline=deadline,
                step0=0.22,
                max_iter=max_iter_sub,
            )

        # d) final random walk
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_walk = int(min(300_000, remaining * 35))
            a, _ = random_local_search(
                a,
                deadline=deadline,
                init_step=0.010,
                max_iter=max_iter_walk,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Fallback – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector, score =", best_val)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
