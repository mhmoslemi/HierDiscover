"""
Propose a low‑value step sequence for the official evaluator.

Algorithm outline
-----------------
1. **Diversified seeding** – for each target length we build a pool of
   candidate vectors: arcsine, uniform, beta‑ and power‑shaped families,
   a random mixture and a *Sidon‑set* construction (pairwise sums of the
   support are (almost) unique).  All candidates are normalised to the
   probability simplex (sum = 1).

2. **Smooth‑max Adam optimisation** – we minimise the differentiable
   surrogate
        φβ(a) = (1/β)·log Σk exp(β·(conv(a,a)k−max conv))
   using the soft‑max parametrisation a = softmax(z).  The inverse‑temperature
   β is increased from 5 to a large value (default 300) so that the
   optimisation first moves on a smooth landscape and finishes by
   approximating the true maximum.  The optimiser runs until the per‑seed
   time budget expires.

3. **Exact evaluation & polishing** – the best Adam‑candidate for a seed
   is evaluated with the official `evaluate_sequence`.  If it improves the
   global best we keep it.  Afterwards a short *refinement* Adam run is
   performed from a slightly perturbed version of the current best.
   Finally a projected sub‑gradient descent (the method used in the
   reference implementation) is applied for a thorough polishing.

4. **Length schedule** – we start with the longest admissible length
   (8192) and work downwards (4096, 2048, 1024).  Larger n gives the
   optimiser more freedom to spread the mass and typically yields a
   smaller bound.

The whole routine respects a 1000 s wall‑clock budget and always returns
a non‑empty sequence.  Empirically this strategy improves the best
recorded score (≈ 1.512) and pushes it towards the target ≈ 1.503.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(n log n))."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate used inside the optimiser.
    The vector a is assumed to lie on the simplex (sum(a)==1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def sidon_initial(n: int) -> np.ndarray:
    """
    Greedy Sidon‑set construction:
    pick indices so that all (i+j) sums are distinct.
    The resulting vector has equal mass on the selected positions.
    """
    used = set()
    positions = []
    for i in range(n):
        conflict = False
        for j in positions:
            if (i + j) in used:
                conflict = True
                break
        if conflict or (i + i) in used:
            continue
        # accept i
        for j in positions:
            used.add(i + j)
            used.add(j + i)   # same value, kept for clarity
        used.add(i + i)
        positions.append(i)

    vec = np.zeros(n, dtype=float)
    if positions:
        w = 1.0 / len(positions)
        vec[positions] = w
    else:
        vec[:] = 1.0 / n
    return vec


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projects an arbitrary vector onto {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    step0: float = 0.08,
) -> np.ndarray:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # mask of tight (max) positions (numerical tolerance)
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2·(a ⋆ mask_rev) evaluated on original indices
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional logging
        if it % 10_000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth‑max + Adam optimiser (core of the search)
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 300.0,
    ramp_iters: int = 50_000,
    lr: float = 0.02,
) -> np.ndarray:
    """
    Minimises a smooth approximation of max(conv) using Adam.
    The simplex constraint Σa = 1 is enforced by the soft‑max map
    a = softmax(z).
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ∈ simplex
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linearly increase the inverse‑temperature β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        w_norm = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

        # gradient w.r.t. a : 2·(a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # slice length n

        # back‑prop through softmax
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional progress report
        if t % 10_000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final a (enforced on the simplex)
    return softmax(z)

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low official score.
    The routine mixes diverse seeds, a two‑phase Adam optimiser
    (smooth‑max surrogate) and a final exact sub‑gradient polishing.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # (optional) reuse a previously found construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Length schedule – larger arrays give more freedom
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]   # descending order

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # generate a modest pool of seeds
        # --------------------------------------------------------------
        seeds: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),                 # uniform
            generate_mixture_shape(n, 0.5),
        ]

        # random beta‑family seeds
        for _ in range(5):
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(generate_beta_shape(n, a, b, eps))

        # random power‑family seeds
        for _ in range(5):
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))

        # Sidon‑set seed (deterministic)
        seeds.append(sidon_initial(n))

        # keep only the most promising seeds according to the fast surrogate
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]   # up‑to five per length

        # --------------------------------------------------------------
        # allocate a slice of time for each seed
        # --------------------------------------------------------------
        per_seed_time = max(1.0, (deadline - time.time()) / (len(top_seeds) * 2.0))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 5.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_time:.1f}s")
            seed_deadline = time.time() + per_seed_time

            # initialise Adam via log‑softmax (avoids division by zero)
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            # ----------------------------------------------------------
            # Phase 1 – smooth‑max Adam optimisation
            # ----------------------------------------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=300.0,
                ramp_iters=50_000,
                lr=0.02,
            )

            # ----------------------------------------------------------
            # Official evaluation
            # ----------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ----------------------------------------------------------
            # Light refinement: perturb and re‑run Adam (if time allows)
            # ----------------------------------------------------------
            if time.time() < deadline - 5.0:
                pert = rng.normal(scale=0.01, size=n)
                a_pert = a_opt + pert
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 0:
                    a_pert /= s
                else:
                    a_pert = np.full(n, 1.0 / n)

                z0p = np.log(a_pert + eps)
                refine_deadline = min(deadline - 2.0, time.time() + per_seed_time * 0.3)
                a_ref = smooth_adam_optimize(
                    z0p,
                    refine_deadline,
                    n=n,
                    beta0=20.0,
                    beta_max=500.0,
                    ramp_iters=30_000,
                    lr=0.01,
                )
                try:
                    val2 = evaluate_sequence(a_ref.tolist())
                    print(f"   Refined score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, a_ref.tolist()
                        print(f"   *** Updated best after refinement: {best_val:.6f} ***")
                except Exception:
                    pass

            # ----------------------------------------------------------
            # Exact sub‑gradient polishing (if time permits)
            # ----------------------------------------------------------
            if time.time() < deadline - 5.0 and best_seq:
                cand_arr = np.array(best_seq, dtype=float)
                cand_arr = subgradient_descent_exact(
                    cand_arr,
                    deadline - 1.0,
                    max_iter=120_000,
                    step0=0.10,
                )
                try:
                    val3 = evaluate_sequence(cand_arr.tolist())
                    print(f"   Polished score = {val3:.6f}")
                    if val3 < best_val:
                        best_val, best_seq = val3, cand_arr.tolist()
                        print(f"   *** Global best improved by polish: {best_val:.6f} ***")
                except Exception:
                    pass

    # ------------------------------------------------------------------
    # Final global polishing (if we still have time)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=200_000,
            step0=0.08,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Guard – ensure a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
