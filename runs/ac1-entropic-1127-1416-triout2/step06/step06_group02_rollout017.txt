assistantfinal**Solution Overview**

The score of a step function  

\[
S(a)=\frac{2\,n\;\max (a\star a)}{(\sum a)^2}
\]

is homogeneous – scaling the whole vector does **not** change the score.  
Therefore we can always keep the vector on the probability simplex  
\(\sum a_i=1,\;a_i\ge 0\) and minimise  

\[
F(a)=2\,n\;\max (a\star a) .
\]

* `a★a` is the linear convolution of `a` with itself (full length `2n‑1`).
* The worst term in the convolution is the **maximum** entry.
* For a fixed maximal value `M` the linear constraints  

  \[
  (a\star g)_k \le M\qquad\forall k
  \]

  are **linear** in `g`.  
  Maximising the ℓ₁‑norm of `g` under those constraints inflates the total
  mass while keeping the peak `M`.  After renormalising to sum = 1 the peak
  is divided by the square of the gained mass – the score drops.

The optimiser consists of three stages

1. **Smooth soft‑max gradient descent** – quickly drives the maximum of the
   convolution down.  The smooth surrogate  

   \[
   f_\varepsilon(a)=\varepsilon\log\!\Bigl(\sum_k\exp\!\bigl((a\star a)_k/\varepsilon\bigr)\Bigr)
   \]

   has a closed‑form gradient that needs only one FFT per iteration.
2. **Linear‑programming (LP) refinement** – with the current maximum `M`
   we solve  

   \[
   \max_{g\ge0}\;\sum_i g_i \qquad
   \text{s.t. } a\star g \le M ,
   \]

   obtain `g`, mix it with the current vector (`t=0.5`) and repeat while the
   score keeps improving.
3. **Stochastic hill‑climbing** – a cheap final polish that swaps or moves a
   small amount of mass between two positions and accepts only improving
   moves.

All operations are performed on the simplex, the score is evaluated with a
fast FFT based routine, and the whole search respects a 1000 s wall‑time
budget.

```python
"""
Non‑negative step‑function optimiser.

The algorithm combines:
  * a smooth soft‑max gradient descent (FFT based, O(n log n) per iter),
  * an LP “mass‑inflation” refinement (maximise ℓ₁‑norm of a vector g under
    the linear constraints a★g ≤ max(a★a)),
  * a final stochastic hill‑climber.

All vectors are kept on the probability simplex (non‑negative, sum = 1),
which makes the evaluation fast:
      score = 2·n·max(a★a)
The routine returns the best sequence found within the given time budget.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ★ b using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast replica of evaluate_sequence for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisation
# ----------------------------------------------------------------------


def random_dirichlet_vector(n: int, alpha: float = 0.5) -> np.ndarray:
    """Dirichlet draw (α < 1 gives sparse vectors)."""
    a = np.random.dirichlet(np.full(n, alpha))
    return a.astype(float)


# ----------------------------------------------------------------------
# 1. Smooth soft‑max gradient descent
# ----------------------------------------------------------------------


def smooth_gradient_descent(a0: np.ndarray,
                            deadline: float,
                            max_iter: int = 5000,
                            step0: float = 0.5,
                            eps_factor: float = 0.15) -> np.ndarray:
    """
    Minimises the log‑sum‑exp surrogate of max(a★a).

    eps shrinks during optimisation, the gradient is
        grad = 2 * (w_rev ★ a)   with   w = softmax(conv/eps)
    After each step the result is projected onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # epsilon never smaller than 1 % of maxc, shrinks geometrically
        eps = maxc * max(eps_factor * (0.995 ** (t - 1)), 0.01)

        # stable soft‑max
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)               # ∑ w = 1

        # gradient: 2·(w_rev ★ a)  (only the central n entries are needed)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)          # length 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # step size (standard AdaGrad schedule)
        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        a = a_new

    return a


# ----------------------------------------------------------------------
# 2. Linear‑programming refinement (mass inflation)
# ----------------------------------------------------------------------


def conv_matrix(a: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix M such that M @ g = a ★ g .
    Each diagonal k (k = 0 … n‑1) contains the constant a[k].
    """
    n = a.shape[0]
    m = 2 * n - 1
    offsets = np.arange(n)                # 0 … n‑1
    # every diagonal consists of the same value a[k]
    data = np.tile(a[:, None], (1, n))    # shape (n, n)
    return sparse.diags(data, offsets, shape=(m, n), format='csr')


def lp_refine(a: np.ndarray,
              deadline: float,
              mix_factor: float = 0.5,
              max_inner: int = 10) -> np.ndarray:
    """
    Repeatedly solve

        max   Σ g_i
        s.t. a ★ g ≤ M ,   g ≥ 0

    where M = max(a★a).
    The solution g is mixed with the current vector (mix_factor) and the
    process stops when no further improvement is observed or the time budget
    expires.
    """
    n = a.shape[0]

    # current maximal convolution value
    cur_max = np.max(convolution_fft(a, a))
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_score = cur_score

    for _ in range(max_inner):
        if time.time() >= deadline - 0.5:
            break

        # avoid exact zeros – they would make the LP unbounded
        a_pos = np.maximum(best_a, 1e-6)
        a_pos = a_pos / a_pos.sum()

        A = conv_matrix(a_pos)                    # (2n‑1)×n
        b = np.full(2 * n - 1, cur_max)

        # maximise Σ g_i  → minimise -Σ g_i
        c = -np.ones(n)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'disp': False})

        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-9:          # no mass inflation possible
            break

        # normalized candidate
        g_norm = g / sum_g               # sum = 1
        # gentle mixing with the old vector
        cand = (1.0 - mix_factor) * best_a + mix_factor * g_norm
        cand = project_to_simplex(cand)

        # evaluate
        new_max = np.max(convolution_fft(cand, cand))
        new_score = 2.0 * n * new_max

        if new_score < best_score - 1e-9:          # improvement found
            best_score = new_score
            best_a = cand
            cur_max = new_max
        else:
            break                                 # cannot improve further

    return best_a


# ----------------------------------------------------------------------
# 3. Hill‑climbing polish
# ----------------------------------------------------------------------


def hill_climb(seq,
               deadline,
               max_steps: int = 200_000) -> np.ndarray:
    """Simple stochastic hill‑climber that accepts only improving moves."""
    a = np.array(seq, copy=True)
    n = a.shape[0]

    best_score = compute_score(a)
    step_scale = 0.10

    while max_steps > 0 and time.time() < deadline:
        # pick move type
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * a[i]
            if delta <= 0:
                continue
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        sc = compute_score(cand)

        if sc < best_score:
            best_score = sc
            a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # be bolder
        else:
            step_scale *= 0.97                         # be more cautious

        max_steps -= 1

    return a


def final_symmetrisation(a: np.ndarray) -> np.ndarray:
    """Replace a by (a + reverse(a))/2 and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (~1000 s) and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the final call to evaluate_sequence
    deadline = start + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optionally start from a known construction if available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb_sc = evaluate_sequence(fallback)
        print(f"[fallback] score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]   # powers of two – cheap FFTs
    outer_limit = 60                    # stop earlier if out of time

    for outer in range(outer_limit):
        if time.time() >= deadline - 5.0:
            break

        n = int(np.random.choice(lengths))
        a = random_dirichlet_vector(n, alpha=0.5)

        # quick score on the raw start
        cur_sc = compute_score(a)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a.tolist()
            print(f"[{outer}] init n={n}, score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – smooth soft‑max gradient descent
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        gd_iters = min(8000, int(remaining * 25))      # ~25 it/s
        a = smooth_gradient_descent(a,
                                    deadline - 2.0,
                                    max_iter=gd_iters,
                                    step0=0.5,
                                    eps_factor=0.15)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{outer}] after GD n={n}, score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – LP mass‑inflation refinement
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 10.0:
            a = lp_refine(a, deadline - 5.0, mix_factor=0.5, max_inner=8)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[{outer}] after LP n={n}, score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – hill‑climb polishing (+ optional symmetrisation)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 8.0:
            a = hill_climb(a, deadline - 2.0, max_steps=250_000)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[{outer}] after hill‑climb n={n}, score={best_score:.6f}")

            a_sym = final_symmetrisation(a)
            sc_sym = compute_score(a_sym)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = a_sym.tolist()
                print(f"[{outer}] after symmetrisation n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
```
