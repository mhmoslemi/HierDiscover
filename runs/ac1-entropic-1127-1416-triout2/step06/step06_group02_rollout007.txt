def refined_subgrad(a0, time_deadline, max_iter=100_000, step0=0.07, top_frac=0.02):
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # choose top indices within tolerance
        # maybe using fraction of n: top_k = max(1, int(top_frac * len(conv)))
        # find indices with conv >= maxc - tol where tol = some fraction of maxc?
        # We'll use top_k indices (largest values)
        top_k = max(1, int(top_frac * len(conv)))
        # Get indices of top_k values
        # Use np.argpartition
        idx_top = np.argpartition(-conv, top_k-1)[:top_k]
        # For each k compute subgradient contribution.
        g = np.zeros_like(a)
        for k in idx_top:
            # compute shift vector: reversed a shifted
            # subgrad from this k is: 2 * a_{k-i} (but a_j=0 if out of bounds)
            j = k - np.arange(n)
            mask = (j >=0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # average over top_k
        grad = g / top_k

        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        # projection to simplex using Euclidean projection (project_to_simplex)
        a_new = project_to_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # add jitter occasionally
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)
    return best_a, best_sc

def hill_climb_sa(a0, time_deadline, max_steps=300_000):
    a = a0.copy()
    n = len(a)
    cur_score = compute_score(a)
    best_a = a.copy()
    best_sc = cur_score

    # initial temperature
    T0 = 0.01 * cur_score  # scaling
    # schedule: T decays linearly to near zero over steps
    # We'll compute t fraction = step / max_steps
    for step in range(max_steps):
        if time.time() >= time_deadline:
            break
        # compute temperature at this step: T = T0 * (1 - step/max_steps)
        T = T0 * (1.0 - step / max_steps)
        # propose move:
        # randomly choose move type: swap or transfer, or random reassign a fraction.
        move_type = np.random.rand()
        cand = a.copy()
        if move_type < 0.33:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.66:
            # mass transfer fraction
            i, j = np.random.randint(n, size=2)

    a = a0.copy()
    v = np.zeros_like(a)  # momentum buffer

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # Determine top set size
        top_k = max(1, int(top_frac * conv.size))
        # Get indices of top_k highest conv values
        idx_top = np.argpartition(-conv, top_k-1)[:top_k]

* Phase 3 – **simulated‑annealing hill‑climbing**: stochastic swaps,
  mass‑transfer and occasional random re‑initialisation are accepted
  according to a linear cooling schedule, allowing occasional uphill
  moves to escape plateaus.
* Phase 4 – **symmetrisation**: replacing `a` by `(a+rev(a))/2` and
  re‑projecting often reduces the maximal autocorrelation further.
* Multiple random restarts (different lengths, different seeds) are
  performed until the time budget (≈ 1000 s) runs out.  The best
  sequence seen is returned.

All heavy operations (convolution, gradient computation) use FFTs, and
every iteration checks the global deadline to guarantee termination
within the allotted time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a⊛b (real FFT, returns length len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast proxy of evaluate_sequence for a simplex vector (sum = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.all(v <= 0):
        # degenerate case – return uniform vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initial vectors
# ----------------------------------------------------------------------


def random_dirichlet_vector(n: int, alpha: float = 0.5) -> np.ndarray:
    """Sparse Dirichlet (Gamma) sample, normalised to sum = 1."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def random_sparse_vector(n: int) -> np.ndarray:
    """Uniform mass on a random subset (density 0.1–0.9)."""
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Phase 1 – soft‑max multiplicative gradient descent (mirror descent)
# ----------------------------------------------------------------------


def mirror_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 20000,
    step0: float = 1.0,
    eps_factor: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on a soft‑max surrogate of the max‑conv."""
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # temperature (ε) – never smaller than 1 % of maxc
        eps = maxc * max(eps_factor * (0.99 ** (t - 1)), 0.01)

        # stable soft‑max weights
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)  # Σ w = 1

        # gradient = 2·(a ⊛ w_rev) on the original index set
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # multiplicative (mirror) update
        step = step0 / np.sqrt(t)
        # clip gradient to avoid overflow in exp
        clip_thr = 50.0 / max(step, 1e-12)
        grad_clipped = np.clip(grad, -clip_thr, clip_thr)
        a *= np.exp(-step * grad_clipped)
        a = a / a.sum()  # projection onto simplex (simple normalisation)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 2 – refined sub‑gradient descent averaging the top‑k peaks
# ----------------------------------------------------------------------


def refined_subgrad_topk(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 200000,
    step0: float = 0.07,
    top_frac: float = 0.02,
    momentum: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent where the gradient is averaged over the largest
    `top_frac` convolution entries.  Momentum (exponential moving average) is
    employed to speed up convergence."""
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    v = np.zeros_like(a)  # momentum buffer

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = conv.max()
        m = conv.size

        # number of top entries to average (at least one)
        top_k = max(1, int(top_frac * m))

        # indices of the top‑k values (unsorted – argpartition is O(m))
        idx_top = np.argpartition(-conv, top_k - 1)[:top_k]

        # weight vector: uniform on the selected indices
        w = np.zeros_like(conv)
        w[idx_top] = 1.0 / top_k

        # gradient via convolution with the reversed weight vector
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # momentum update
        v = momentum * v + (1.0 - momentum) * grad

        step = step0 / np.sqrt(t)
        a_new = a - step * v
        a_new = project_to_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        # small random jitter to avoid stagnation
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            a_new += jitter
            a_new = project_to_simplex(a_new)

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – simulated‑annealing hill climbing
# ----------------------------------------------------------------------


def hill_climb_sa(
    a0: np.ndarray,
    time_deadline: float,
    max_steps: int = 300000,
) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climber with Metropolis acceptance.
    Moves: swaps, mass transfers, occasional random re‑initialisation."""
    a = a0.astype(float).copy()
    n = a.shape[0]
    cur_sc = compute_score(a)

    best_a = a.copy()
    best_sc = cur_sc

    # initialise temperature as a fraction of the current score
    T0 = max(cur_sc * 0.03, 1e-4)

    for step in range(max_steps):
        if time.time() >= time_deadline:
            break

        # linear temperature schedule
        T = T0 * (1.0 - step / max_steps)

        cand = a.copy()
        move_type = np.random.rand()

        if move_type < 0.33:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.66:
            # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                frac = np.random.rand() * 0.25      # up to 25 % of the source mass
                delta = frac * cand[i]
                cand[i] -= delta
                cand[j] += delta
        else:
            # small random perturbation of a few coordinates
            k = np.random.randint(1, 4)            # 1‑3 positions
            idxs = np.random.choice(n, size=k, replace=False)
            # draw fresh Dirichlet mass for these coordinates
            new_mass = np.random.dirichlet(np.ones(k))
            # remove current mass at idxs, keep the rest proportionally
            rest = cand.copy()
            rest[idxs] = 0.0
            rest_sum = rest.sum()
            if rest_sum > 0:
                rest *= (1.0 - new_mass.sum()) / rest_sum
                cand = rest
                cand[idxs] = new_mass
            else:
                # all mass was at the chosen indices – just replace
                cand = np.zeros_like(cand)
                cand[idxs] = new_mass

        cand = project_to_simplex(cand)
        cand_sc = compute_score(cand)

        delta = cand_sc - cur_sc
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            # accept
            a = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = cand.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 4 – optional symmetrisation
# ----------------------------------------------------------------------


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """Replace a by (a + rev(a))/2 and re‑project onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine respects a ~1000 s budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # reserve a tiny safety margin for the final evaluation call
    deadline = start + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"Fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]          # FFTs stay cheap
    outer_limit = 12                     # max different random starts

    for outer in range(outer_limit):
        if time.time() >= deadline - 15.0:
            break

        n = int(np.random.choice(lengths))
        print(f"\n[restart {outer}] length = {n}")

        # random start – blend of Dirichlet and sparse patterns
        if np.random.rand() < 0.5:
            a = random_dirichlet_vector(n, alpha=0.5)
        else:
            a = random_sparse_vector(n)

        # quick pre‑filter
        cur_sc = compute_score(a)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a.tolist()
            print(f"  initial score {cur_sc:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – mirror (soft‑max) descent
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            md_iters = min(15000, int(remaining * 25))   # about 25 iters / sec
            a_md, sc_md = mirror_descent(
                a,
                time_deadline=deadline - 4.0,
                max_iter=md_iters,
                step0=1.0,
                eps_factor=0.20,
            )
            sc_md_exact = compute_score(a_md)
            if sc_md_exact < best_score:
                best_score = sc_md_exact
                best_seq = a_md.tolist()
                print(f"  after mirror descent: score {best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – refined sub‑gradient (top‑k averaging)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 10.0:
            sub_iters = min(180000, int(remaining * 120))  # ~120 iters / sec
            a_sub, sc_sub = refined_subgrad_topk(
                a_md,
                time_deadline=deadline - 5.0,
                max_iter=sub_iters,
                step0=0.07,
                top_frac=0.015,
                momentum=0.85,
            )
            sc_sub_exact = compute_score(a_sub)
            if sc_sub_exact < best_score:
                best_score = sc_sub_exact
                best_seq = a_sub.tolist()
                print(f"  after refined sub‑grad: score {best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – simulated‑annealing hill climbing
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 12.0:
            steps = min(350000, int(remaining * 3000))   # roughly 3000 attempts / sec
            a_hc, sc_hc = hill_climb_sa(
                a_sub,
                time_deadline=deadline - 4.0,
                max_steps=steps,
            )
            sc_hc_exact = compute_score(a_hc)
            if sc_hc_exact < best_score:
                best_score = sc_hc_exact
                best_seq = a_hc.tolist()
                print(f"  after hill‑climb SA: score {best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 4 – final symmetrisation (cheap)
        # --------------------------------------------------------------
        if best_seq is not None:
            a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
            sc_sym = compute_score(a_sym)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = a_sym.tolist()
                print(f"  after symmetry: score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if everything failed, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
