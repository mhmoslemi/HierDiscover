"""
Candidate generator for the “step‑function” challenge.

The objective is      2·n·max(a⊛a) / (∑a)² .
Since it is homogeneous we keep the vector on the probability simplex
(∑a = 1) and minimise   2·n·max(conv) .
The code uses three complementary optimisation phases:

1. **Adam on a smooth soft‑max approximation**  
   f_β(a) = (1/β)·log Σ exp(β·conv) .  When β → ∞ the smooth
   objective converges to the true max.  The gradient can be
   expressed as a second FFT‑based convolution, so every iteration
   costs only O(n log n).  Adam’s adaptive steps accelerate convergence.

2. **Projected sub‑gradient descent** on the exact max‑objective.  
   The sub‑gradient for the active index k is the shifted copy of a.
   A 1/√t step‑size together with a simplex projection drives the
   solution further downhill.

3. **Stochastic hill‑climbing** (mass‑transfer moves).  
   Random swaps or small transfers of probability mass are accepted
   only when they improve the exact score; the step‑size adapts
   automatically.

Multiple random restarts are performed while time allows; the best
candidate seen over the whole run is finally returned (after an
optional last round of Adam with a very large β).  The routine respects
the 1000 s wall‑clock limit imposed by the competition harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFTs (fast for the lengths we need).
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a⊛a) .
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        # corner case – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Adam optimisation of the smooth soft‑max objective
# ----------------------------------------------------------------------


def adam_optimize(start_a: np.ndarray,
                  time_deadline: float,
                  max_iters: int = 200_000,
                  beta_start: float = 1.0,
                  beta_end: float = 200.0,
                  lr: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed objective
        f_β(a) = (1/β)·log Σ exp(β·conv)

    The gradient is
        ∇f_β = 2·conv(a_rev, w)
    where w_i = softmax(β·conv_i).
    The iterate is projected onto the simplex after every update.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ---- anneal β -------------------------------------------------
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # ---- smooth soft‑max -----------------------------------------
        conv = convolution_fft(a, a)                 # length 2n‑1
        maxc = conv.max()
        # stabilize exponentials
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                     # soft‑max weights

        # ---- gradient -----------------------------------------------
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)      # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]    # slice length n

        # ---- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = project_to_simplex(a)

        # ---- occasional exact evaluation ------------------------------
        if (it & 511) == 0:               # every 512 iterations
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Classic projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 300_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   2·n·max(conv(a,a)) .
    """
    a = init_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max entry (shift k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # small jitter every 1000 steps to avoid stagnation
        if (t % 1000) == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    # final check
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq,
               time_deadline,
               max_steps: int = 400_000) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb that repeatedly transfers a fraction of mass
    between two randomly chosen positions (or swaps them).  After each
    move the vector is projected onto the simplex.  The best solution
    ever seen is returned.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two entries (keeps the distribution but may relocate mass)
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # become bolder after success
        else:
            step_scale *= 0.97                         # become more cautious

        max_steps -= 1

    return best_a.tolist(), best_score

        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very last high‑β Adam pass (tiny steps) – optional but cheap
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        # a short burst with an enormous β to squeeze a last bit more
        a_last, last_score = adam_optimize(
            np.array(best_seq, dtype=np.float64),
            time_deadline=deadline,
            max_iters=60_000,
            beta_start=200.0,
            beta_end=1_200.0,
            lr=0.25,
        )
        if last_score < best_val:
            best_val = last_score
            best_seq = a_last.tolist()
            print(f"Final Adam tweak improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Safeguard: if nothing was found, fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 7. Official evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
