    a = a0.copy()
    a = a / a.sum() # ensure sum=1

    # initialization of parameters
    t_iter = 0
    step0 = 0.2  # initial step size
    L = None (maybe compute later)
    while time.time() < deadline:
        t_iter += 1
        # compute conv
        conv = convolution_fft(a, a)
        mx = conv.max()
        # smooth beta: linearly increase with t_iter up to max_iters == ramp_iters*some factor.
        # We can simply compute beta = β_start + (β_end - β_start) * min(1.0, t_iter / ramp_iters)
        beta = beta0 + (beta_max - beta0) * min(1.0, t_iter / ramp_iters)

        # compute weights
        shifted = conv - mx
        # using stable exp: w = np.exp(beta * shifted)  (shifted <=0)
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # compute gradient
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)  # length 3n-2

        # step size schedule: step = step0 / np.sqrt(t_iter)
        step = step0 / np.sqrt(t_iter)

        # gradient step
        a = a - step * grad_a

        # projection onto simplex
        a = project_to_simplex(a)

        # optionally optionally break early if no improvement in surrogate value (smooth). Might not necessary.

    # final normalization to sum=1 (to avoid slight rounding)
    a = a / a.sum()
    return a

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max\_{k}(a*a)[k]}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where `*` denotes the linear (zero‑padded) convolution.  
Because the objective is *scale‑invariant* we can restrict ourselves to
probability vectors `a` (`∑a_i = 1`).  For such vectors  

\[
F(a)=2\,n\;\max_k (a*a)[k] .
\]

The optimisation is highly non‑convex (the convolution terms are
quadratic products).  The implemented search combines three complementary
ideas:

1. **Diverse seeds** – arcsine, uniform, mixtures, random β‑/power‑
   families, Dirichlet and Gaussian‑shaped vectors.
2. **Smooth‑max Adam optimisation** – replace the hard `max` by a
   temperature‑controlled log‑sum‑exp.  For a fixed temperature β the
   surrogate is smooth and its gradient is obtained analytically by a
   second convolution.  Adam together with a linear β‑schedule drives the
   iterate quickly on a smooth landscape and then fine‑tunes the true
   maximum.
3. **Exact sub‑gradient polishing** – after the Adam phase we run a
   projected sub‑gradient descent on the original (non‑smooth) objective
   to squeeze out the last bits of improvement.

The whole procedure respects the 1000 s wall‑clock budget: a quick
exploration of several seeds consumes ≈60 % of the time, the best seed
receives a long fine‑tuning run, and the remaining time is used for the
final exact polish.  All helper functions are top‑level, there are no
closures or file accesses, and the code works with the `evaluate_sequence`
function provided by the harness.

```python
"""
Optimise a non‑negative step sequence for the evaluation function.

Algorithm
---------
1. Choose a moderate length (4096) – large enough to give many degrees
   of freedom, still cheap to convolve with FFT.
2. Generate a diverse pool of seeds (arcsine, uniform, mixtures,
   random beta / power shapes, Dirichlet, Gaussian).  Normalise each
   seed to sum = 1.
3. Quick exploration:
   * For each seed run a short Adam optimisation on the smooth
     surrogate  Sβ(a)= (1/β)·log Σ exp(β·conv_k) .
   * β grows linearly from 5 to 200, learning‑rate = 0.07.
   * Keep the best candidate (official evaluation).
4. Deep refinement of the current best seed:
   * Continue Adam with a higher final temperature (β_max = 500) and
     a slower learning‑rate (0.04) for the remaining budget.
5. Final polishing:
   * Projected sub‑gradient descent on the true objective (max‑convolution)
     with a diminishing step size.
6. Return the best sequence found (guaranteed to be non‑empty).

All convolutions are performed via real FFTs for speed.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using real FFT (fast for large vectors)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast surrogate score (sum = 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(a,a))  (a already normalised to 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple seed generators (all return a vector that sums to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped distribution – often a good starting point."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def uniform_initial(n: int) -> np.ndarray:
    """Uniform vector."""
    return np.full(n, 1.0 / n)


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return uniform_initial(n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Power law shape 1/((i+½)(n‑½‑i)+eps)^p."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return uniform_initial(n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture between arcsine and uniform."""
    arc = arcsine_initial(n)
    uni = uniform_initial(n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sample from a Dirichlet(α,…,α) distribution."""
    draws = rng.gamma(alpha, 1.0, size=n)
    if draws.sum() == 0:
        return uniform_initial(n)
    return draws / draws.sum()


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth‑max Adam optimisation (operates on log‑softmax variables)
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 200.0,
    ramp_iters: int = 30000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Minimise the smooth (log‑sum‑exp) surrogate of max(conv) with Adam.
    z is the log‑softmax parameter (a = softmax(z) ⇒ Σa = 1, a≥0).
    The temperature β is increased linearly from β0 to β_max.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a lives on the simplex
        conv = convolution_fft(a, a)       # linear convolution a*a
        mx = float(np.max(conv))

        # temperature schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # soft‑max weights for the smooth‑max
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)         # may under‑flow, that's fine
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a  (see derivation in the analysis)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)   # length = n + (2n‑1) – 1 = 3n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]   # slice of length n

        # chain rule through the softmax parametrisation
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

    # final a (projected automatically by softmax)
    a = softmax(z)
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true objective (max‑convolution)
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    step0: float = 0.08,
) -> np.ndarray:
    """
    Projected sub‑gradient descent for  f(a)=2·n·max(conv(a,a)).
    The iterate is always kept on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1                     # length of a*a

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # tolerance for deciding which entries are “tight”
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2 * (a ⋆ mask_rev) restricted to original length
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # optional occasional logging (commented out for speed)
        # if it % 5000 == 0:
        #     print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a

    # 5.  Phase 2 – deep refinement of the current best candidate
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n--- Deep refinement of the current best ---")
        a_start = np.array(best_seq, dtype=float)
        a_start /= a_start.sum()
        z0 = np.log(a_start + 1e-12)

        a_refined = smooth_adam_optimize(
            z0,
            deadline - 2.0,          # leave a couple of seconds for polish
            n=n,
            beta0=50.0,
            beta_max=500.0,
            ramp_iters=120000,
            lr=0.04,
        )

        try:
            val = evaluate_sequence(a_refined.tolist())
            print(f"   Refined official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_refined.tolist()
                print(f"   *** Best improved after deep refinement: {best_val:.6f} ***")
        except Exception as e:
            print(f"   Refinement evaluation error: {e}")

    # ------------------------------------------------------------------
    # 6.  Phase 3 – final exact sub‑gradient polishing (use any remaining time)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Final exact polishing ---")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline - 1.0,
            max_iter=200_000,
            step0=0.08,
        )
        try:
            val = evaluate_sequence(a_arr.tolist())
            print(f"   Polished official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_arr.tolist()
                print("*** Global best after polishing ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 7.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        print("\nFallback to uniform sequence (should never happen).")
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
```
