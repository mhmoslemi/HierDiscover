def smooth_optimize_momentum(start_a, deadline, max_iters=200_000, beta_start=5.0, beta_end=200.0, step0=0.6, mu=0.9):
    a = start_a.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()
        a_rev = a[::-1]
        grad_rev = convolution_fft(a_rev, w)
        grad = 2.0 * grad_rev[len(a)-1: len(a)-1 + len(a)]
        step = step0 / np.sqrt(it)
        v = mu * v - step * grad
        a_new = a + v
        a_new = project_to_simplex(a_new)
        a = a_new
        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

"""
Improved search for step‑function heights that minimise the
`evaluate_sequence` objective.

Algorithm
---------
* Choose a length n (2048 / 4096 / 8192) according to remaining time.
* Perform many random restarts.  Each restart draws a Dirichlet vector
  (controlling sparsity with a random concentration) and runs a
  momentum version of the smoothed log‑sum‑exp optimisation.  The
  smoothing parameter β is annealed from a small value to a large one
  (β≈200) so that the final iterates are close to the true max‑norm.
* The best vector from the first stage is refined with Adam on the
  true nonsmooth objective (projected sub‑gradient descent with adaptive
  moments).  This yields very fast reductions of the peak of the
  auto‑convolution.
* A final stochastic hill‑climb moves a small fraction of mass between
  positions that are responsible for the current peak; the move is
  accepted only when it improves the exact score.
* The procedure respects the 1000 s wall‑clock limit; if nothing is
  found a uniform vector is returned.
"""

import time
import numpy as np

# ------------------- low‑level primitives -------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact replica of `evaluate_sequence` for a probability vector
    (i.e. Σa == 1).  Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ------------------- optimisation primitives -------------------

def smooth_optimize_momentum(start_a: np.ndarray,
                             time_deadline: float,
                             max_iters: int = 250_000,
                             beta_start: float = 5.0,
                             beta_end: float = 200.0,
                             step0: float = 0.7,
                             mu: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Smoothed log‑sum‑exp minimisation with Nesterov‑type momentum.
    Returns the best vector (on the simplex) and its exact score.
    """
    a = start_a.astype(float).copy()
    v = np.zeros_like(a)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β exponentially
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed objective – compute soft‑max weights
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))  # stabilised
        w = exp_weights / exp_weights.sum()         # probability vector, length 2n‑1

        # gradient = 2 * conv_rev(w)  (size n)
        a_rev = a[::-1]
        grad_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * grad_rev[n - 1: n - 1 + n]

        # Nesterov‑type update
        step = step0 / np.sqrt(it)
        v = mu * v - step * grad
        a_new = a + v
        a = project_to_simplex(a_new)

        # record progress occasionally
        if it % 250 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    final = compute_score(a)
    if final < best_score:
        best_score = final
        best_a = a.copy()

    return best_a, best_score


def adam_subgrad_optimize(init_a: np.ndarray,
                          time_deadline: float,
                          max_iter: int = 300_000,
                          lr0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the exact nonsmooth objective
    (max‑convolution).  Works directly on the simplex.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the peak (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr0 / np.sqrt(t)

        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a_new)

        if t % 400 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            eps_j = lr0 * 0.03 / np.sqrt(t)
            a += np.random.randn(n) * eps_j
            a = project_to_simplex(a)

    final = compute_score(a)
    if final < best_score:
        best_score = final
        best_a = a.copy()
    return best_a, best_score


def hill_climb(seq: np.ndarray,
               time_deadline: float,
               max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a small fraction of mass from
    indices that contribute heavily to the current peak towards
    indices that contribute little.
    """
    a = seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    # keep a small running scale for the transferred fraction
    delta_frac = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        # compute current peak and its contributing indices
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # indices that appear in the peak sum
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        # pick a source index proportional to its contribution
        if contrib.sum() > 0:
            src = np.random.choice(n, p=contrib / contrib.sum())
        else:
            src = np.random.randint(n)

        # pick a destination that is not the source
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        delta = delta_frac * a[src]
        if delta <= 0:
            max_steps -= 1
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
            a = cand
            # enlarge step a bit when we succeed
            delta_frac = min(delta_frac * 1.07, 0.25)
        else:
            # shrink step when we fail
            delta_frac = max(delta_frac * 0.97, 0.01)
            # occasionally try a pure random perturbation to escape
            if np.random.rand() < 0.001:
                a = project_to_simplex(a + np.random.randn(n) * 0.005)

        max_steps -= 1

    return best_a, best_score

    Works directly on the simplex.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the peak (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr0 / np.sqrt(t)
        a = project_to_simplex(a - step * m_hat / (np.sqrt(v_hat) + eps))

        if t % 400 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

        # occasional jitter to shake out of plateaus
        if t % 2000 == 0:
            jitter = lr0 * 0.03 / np.sqrt(t)
            a += np.random.randn(n) * jitter
            a = project_to_simplex(a)

    final = compute_score(a)
    if final < best_score:
        best_score = final
        best_a = a.copy()
    return best_a, best_score


def hill_climb(seq: np.ndarray,
               time_deadline: float,
               max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that transfers a small fraction of mass from
    indices heavily involved in the current maximal convolution entry to
    indices that are hardly involved.
    """
    a = seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    # transfer fraction – adapts during the walk
    frac = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # contributions of each ordered pair to the peak
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        # source index – sample proportionally to its contribution
        if contrib.sum() > 0:
            src = np.random.choice(n, p=contrib / contrib.sum())
        else:
            src = np.random.randint(n)

        # destination – any other index
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        delta = frac * a[src]
        if delta <= 0:
            max_steps -= 1
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = project_to_simplex(cand)

        cur = compute_score(cand)
        if cur < best_score:
            best_score = cur
            best_a = cand.copy()
            a = cand
            # increase the step size when we are successful
            frac = min(frac * 1.07, 0.25)
        else:
            # shrink when we fail
            frac = max(frac * 0.97, 0.01)
            # tiny random perturbation now and then
            if np.random.rand() < 0.001:
                a = project_to_simplex(a + np.random.randn(n) * 0.005)

        max_steps -= 1

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a reasonable length (more points give more flexibility)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        n = 8192
    elif remaining > 400:
        n = 4096
    else:
        n = 2048
    print(f"Chosen sequence length = {n}")

    # ------------------------------------------------------------------
    # 2. Random restarts + momentum‑smooth optimisation
    # ------------------------------------------------------------------
    # number of inner iterations scaled with the problem size
    smooth_iters = 200_000 if n <= 4096 else 120_000

    def one_restart():
        """Run a single restart: random start → smooth momentum optimisation."""
        # draw a Dirichlet vector; concentration α controls sparsity
        alpha = np.random.uniform(0.1, 5.0)
        a0 = np.random.gamma(alpha, 1.0, size=n)
        a0 /= a0.sum()

        a_opt, scr = smooth_optimize_momentum(
            a0,
            time_deadline=deadline - 130.0,   # keep a generous margin for later phases
            max_iters=smooth_iters,
            beta_start=6.0,
            beta_end=200.0,
            step0=0.7,
            mu=0.9,
        )
        return a_opt, scr

    # launch restarts while we have enough time left for the remaining stages
    while time.time() < deadline - 260.0:
        cand_a, cand_score = one_restart()
        if cand_score < best_val:
            best_val = cand_score
            best_seq = cand_a.tolist()
            print(f"New best after smooth stage: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam sub‑gradient refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print("\nAdam sub‑gradient refinement")
        refined_a, refined_score = adam_subgrad_optimize(
            np.array(best_seq, dtype=float),
            time_deadline=deadline - 80.0,
            max_iter=300_000,
            lr0=0.12,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Improved by Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nHill‑climb polishing")
        polished_a, polished_score = hill_climb(
            np.array(best_seq, dtype=float),
            time_deadline=deadline,
            max_steps=600_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_a.tolist()
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Safeguard: fall back to a modest uniform vector if nothing else
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided `evaluate_sequence`)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
