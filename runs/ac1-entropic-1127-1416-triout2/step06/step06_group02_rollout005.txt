Algorithm

    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple hill‑climber (same as the baseline, kept for polishing)
# ----------------------------------------------------------------------
def hill_climb_permutation(seq, time_deadline, max_steps: int = 300_000):
    """
    Stochastic hill‑climber that performs swaps or transfers a fraction of
    mass between two positions.  Only moves that improve the *exact* score are
    accepted.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.10   # initial fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # be bolder
        else:
            step_scale *= 0.97                         # be more cautious

        max_steps -= 1

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Produce a symmetric version (a + reverse(a))/2 and re‑project.
    Symmetric vectors often have flatter autocorrelations.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------
def build_conv_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the (2·n‑1) × n matrix A such that (A @ g)[k] = (f ⊛ g)[k].
    The i‑th column of A contains the coefficient f[i] at rows i … i+n‑1.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def lp_flatten_step(f: np.ndarray, t: float = 0.30):
    """
    One iteration of the LP‑based flattening:
        * solve maximize Σ g_j   s.t.   g ≥ 0,   f ⊛ g ≤ M   (elementwise)
        * rescale g to sum‑1
        * return the convex combination (1‑t)·f + t·g.
    Returns the new vector and the (optional) LP solution status.
    """
    n = f.shape[0]
    conv = convolution_fft(f, f)
    M = float(np.max(conv))

    # Build constraints  (f ⊛ g)_k ≤ M  for k = 0 … 2n‑2
    A = build_conv_constraint_matrix(f)
    b = np.full(2 * n - 1, M, dtype=float)

    # Linear programme: maximise Σ g_j  →  minimise -Σ g_j
    c = -np.ones(n, dtype=float)

    # Solve (highs is fast and robust)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if not res.success:
        return f, False

    g = res.x
    sum_g = float(np.sum(g))
    if sum_g <= 1e-12:
        return f, False

    # Rescale to the simplex
    g = g / sum_g

    # Convex combination
    f_new = (1.0 - t) * f + t * g
    f_new = project_to_simplex(f_new)
    return f_new, True


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈ 1000 s) and returns a plain
    Python list of floats ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback: use a previously known construction if present
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Search over a few lengths (larger n does not change the asymptotic
    # constant but gives a richer search space)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]  # keep them modest – memory for the LP grows ~ n²
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        # ---- initialise with the uniform distribution -----------------
        f = np.full(n, 1.0 / n, dtype=float)
        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[init] n={n} uniform score={cur_score:.6f}")

        # ---- LP‑based flattening loop ---------------------------------
        # The mixing coefficient t is kept modest to guarantee monotone
        # descent; we adaptively shrink it when progress stalls.
        t = 0.30
        no_improve = 0
        while time.time() < deadline - 5.0:
            f_new, ok = lp_flatten_step(f, t=t)
            if not ok:
                break
            new_score = compute_score(f_new)

            if new_score + 1e-9 < best_score:          # strict improvement
                best_score = new_score
                best_seq = f_new.copy()
                f = f_new
                no_improve = 0
                print(f"[LP] n={n} t={t:.2f} score={new_score:.6f}")
                # occasionally enforce symmetry – helps to keep the distribution flat
                if np.random.rand() < 0.15:
                    f = final_symmetry_average(f)
            else:
                no_improve += 1
                if no_improve >= 3:                    # stagnation → reduce step size
                    t *= 0.5
                    no_improve = 0
                    if t < 0.02:                       # too small → stop LP phase
                        break

        # ---- final polishing with stochastic hill‑climb ---------------
        if time.time() < deadline - 8.0:
            seq_to_polish = best_seq if best_seq is not None else f.tolist()
            polished_seq, polished_sc = hill_climb_permutation(
                seq_to_polish,
                time_deadline=deadline - 2.0,
                max_steps=200_000,
            )
            if polished_sc < best_score:
                best_score = polished_sc
                best_seq = np.array(polished_seq, dtype=float)
                print(f"[hill] n={n} improved score={polished_sc:.6f}")

            # symmetry averaging (cheap, often improves)
            a_sym = final_symmetry_average(best_seq)
            sym_sc = compute_score(a_sym)
            if sym_sc < best_score:
                best_score = sym_sc
                best_seq = a_sym
                print(f"[sym] n={n} improved score={sym_sc:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if nothing was ever found, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq.tolist()
