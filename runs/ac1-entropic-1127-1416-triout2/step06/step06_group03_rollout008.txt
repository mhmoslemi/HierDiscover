```python
def conv_matrix(f):
```

```
def lp_improve(a, t, max_iter):
    # compute M = max(conv(a,a))
    conv = convolution_fft(a, a)
    M = conv.max()
    n = len(a)
    # build conv matrix
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    # constraints: A @ g <= M
    b = np.full(m, M, dtype=np.float64)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if not res.success:
        return a
    g = res.x
    # renormalize g to sum 1
    if g.sum() == 0:
        return a
    g = g / g.sum()
    # mix
    a_new = (1 - t) * a + t * g
    # symmetrize
    a_new = (a_new + a_new[::-1]) / 2
    # ensure sum 1 (normalize)
    a_new = a_new / a_new.sum()
    return a_new
```

assistantfinal**Solution Explanation**

The goal is to minimise  

\[
E(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

where `a*a` is the linear convolution of the step‑heights with
themselves.  
Because the evaluator first normalises the sum, we can **fix the
simplex constraint**  

\[
\sum_i a_i = 1 ,
\]

and then minimise only  

\[
F(a)=2\,n\;\max (a*a).
\]

The problem is non‑convex because of the `max`.  The programme below
combines three ideas that work well together:

1. **Smooth surrogate** – replace the hard `max` by a soft‑max
   (log‑sum‑exp) with an inverse temperature β.
   For a fixed β the surrogate is smooth and its gradient can be
   obtained by a *weighted convolution*.

2. **Symmetry** – the objective is invariant under reversal of the
   sequence, therefore we restrict the search to *palindromic*
   vectors.  After each optimisation step the candidate is symmetrised
   (`a ← (a + a[::-1])/2`).  This halves the effective number of free
   variables and gives a slightly flatter convolution.

3. **Linear‑programming refinement** – for a current vector `f` let  
   \(M=\max(f*f)\).  The linear programme  

   \[
   \max_{g\ge 0}\;\sum_i g_i \quad
   \text{s.t.}\;(f*g)_k\le M,\;k=0,\dots,2n-2
   \]

   finds a new feasible direction `g`.  Mixing a small amount of `g`
   into `f` never increases the maximal convolution, but often
   strictly decreases it.  The LP is solved with `scipy.optimize.linprog`
   (HiGHS backend).

The complete search consists of

* a pool of diverse seeds (arcsine, uniform, beta‑shapes, power‑shapes);
* for each seed a **soft‑max + Adam** phase (≈ 15 k iterations) on the
  smooth surrogate, with the symmetry projection;
* a few LP refinement steps;
* a final **projected sub‑gradient descent** on the true objective to
  polish the result;
* optional global polishing after all lengths have been tried.

All computation is performed with FFT‑based convolutions (`O(n log n)`),
so even a few hundred thousand iterations easily fit into the
1000 s wall‑clock budget.

```python
"""
search_step_function.py

Find a non‑negative step sequence that minimises the official
evaluate_sequence() score.

Algorithm
---------
* generate several diverse seeds (arcsine, uniform, beta‑ and power‑families)
* for each seed run a smooth‑max Adam optimiser.
  – the vector is kept on the probability simplex via a softmax parametrisation
  – after each softmax we enforce exact symmetry a←(a+a[::-1])/2
  – the smooth surrogate uses a log‑sum‑exp with a linearly increasing
    inverse temperature β (5 → 300)
* after Adam, improve the candidate by a small number of linear‑programming
  steps: maximise Σg subject to (f*g)_k ≤ M (M = max(f*f));
  mix the solution back with the current candidate.
* polish the best vector with a projected sub‑gradient method
  (projection onto the simplex and a symmetry projection)
* keep the best sequence ever seen and return it.

All heavy work is done with FFT‑based convolutions; the LP uses
scipy.optimize.linprog (HiGHS).  The routine respects a 1000 s budget
and always returns a valid non‑negative sequence.
"""

import time
import math
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional linear‑programming import (may be unavailable on some hosts)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog  # type: ignore
    _LP_AVAILABLE = True
except Exception:  # pragma: no cover
    _LP_AVAILABLE = False


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast, O(n log n))."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate of the official score (assumes sum(a)=1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact palindrome: (v + reversed(v)) / 2 .  Preserves the sum."""
    return (v + v[::-1]) * 0.5


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max surrogate (with symmetry)
# ----------------------------------------------------------------------
def smooth_adam_optimize_sym(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 300.0,
    ramp_iters: int = 15000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimisation of the smooth (log‑sum‑exp) approximation of
    max(conv(a,a)).  The iterates stay on the simplex via a soft‑max,
    and after each soft‑max the vector is symmetrised.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥ 0 , Σa = 1
        a_sym = symmetrize(a)              # enforce palindrome
        conv = convolution_fft(a_sym, a_sym)
        mx = float(np.max(conv))

        # linearly increase β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights w ∝ exp(β·(conv – mx))
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of the surrogate w.r.t. a_sym
        w_rev = w_norm[::-1]
        rev = convolution_fft(a_sym, w_rev)            # length = 3n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]          # slice of length n

        # because a = softmax(z) → a_sym = S(a), we need the symmetric
        # part of the gradient for the chain rule
        grad_a_sym = symmetrize(grad_a)

        dot = np.dot(grad_a_sym, a)                    # ⟨grad_a_sym , a⟩
        grad_z = a * (grad_a_sym - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} surrogate={compute_score(a_sym):.6f}")

    # final vector (symmetrised and normalised)
    a = softmax(z)
    a_sym = symmetrize(a)
    total = a_sym.sum()
    if total == 0.0:
        return np.full_like(a_sym, 1.0 / n)
    return a_sym / total


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # enforce exact sum = 1 (numerical safety)
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    else:
        w = np.full_like(w, 1.0 / w.size)
    return w


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (used for polishing)
# ----------------------------------------------------------------------
def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.05,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate is kept on the simplex and forced to stay symmetric.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2 * shift(a, K) summed over all tight indices K
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)
        a = symmetrize(a)               # keep the palindrome property

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} surrogate={compute_score(a):.6f}")

    return a

# Linear‑programming refinement (optional, requires SciPy)
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that A @ g = conv(f, g).
    Column j of A is f shifted right by j places.
    """
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    return A


def lp_improve(a: np.ndarray, t: float = 0.3) -> np.ndarray:
    """
    One LP refinement step: solve
          maximise Σ g_i
          s.t.   a * g  ≤  M   (element‑wise)
                g ≥ 0
    where M = max(conv(a, a)).
    The feasible g is mixed back: a ← (1‑t)·a + t·ĝ,
    where ĝ is the normalised LP solution.
    Returns the new (symmetrised, normalised) vector.
    """
    if not _LP_AVAILABLE:
        return a

    n = a.size
    conv = convolution_fft(a, a)
    M = float(conv.max())

    # Build the constraint matrix once (cost O(n²) but acceptable for n≤4096)
    A = build_conv_matrix(a)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear program: maximise Σg  ⇔ minimise –Σg
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        return a

    g = res.x
    s = g.sum()
    if s == 0.0:
        return a
    g = g / s

    # convex combination, enforce symmetry and simplex
    a_new = (1.0 - t) * a + t * g
    a_new = symmetrize(a_new)
    a_new = project_to_simplex(a_new)
    return a_new

# Seed generators (diverse families)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function sequence with a low official score.
    The routine mixes diverse parametric seeds, a smooth‑max Adam phase,
    optional LP refinements and a final projected sub‑gradient polishing.
    It respects the provided wall‑clock budget and always returns a
    non‑empty, non‑negative list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously stored result
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose lengths (larger → more degrees of freedom)
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024]   # try biggest first; later ones are cheaper

    for n in lengths:
        if time.time() >= deadline - 6.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a modest pool of seeds
        # --------------------------------------------------------------
        seeds: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            generate_mixture_shape(n, 0.5),
        ]

        # random beta‑family seeds
        for _ in range(4):
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(generate_beta_shape(n, a, b, eps))

        # random power‑family seeds
        for _ in range(4):
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))

        # --------------------------------------------------------------
        # Keep a few most promising seeds according to the fast surrogate
        # --------------------------------------------------------------
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:5]]   # at most five per length

        # --------------------------------------------------------------
        # 1b. Allocate a slice of time for each seed
        # --------------------------------------------------------------
        per_seed_time = max(2.0, (deadline - time.time()) / (len(top_seeds) * 2))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 5.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – budget ≈{per_seed_time:.1f}s")
            seed_deadline = time.time() + per_seed_time

            # normalise and enforce symmetry before optimisation
            seed_vec = seed_vec / seed_vec.sum()
            seed_vec = symmetrize(seed_vec)

            # initialise Adam via log‑softmax (avoids division by zero)
            z0 = np.log(seed_vec + 1e-12)

            # ----------------------------------------------------------
            # 2. Smooth‑max Adam optimisation (with symmetry)
            # ----------------------------------------------------------
            a_opt = smooth_adam_optimize_sym(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=300.0,
                ramp_iters=15000,
                lr=0.07,
            )

            # ----------------------------------------------------------
            # 3. Optional LP refinement (a handful of passes)
            # ----------------------------------------------------------
            if time.time() < deadline - 5.0 and _LP_AVAILABLE:
                for lp_iter in range(3):
                    if time.time() >= deadline - 5.0:
                        break
                    a_opt = lp_improve(a_opt, t=0.3)

            # ----------------------------------------------------------
            # 4. Official evaluation
            # ----------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ----------------------------------------------------------
            # 5. Light exact polishing (if time permits)
            # ----------------------------------------------------------
            if time.time() < deadline - 5.0:
                polish_deadline = deadline - 2.0
                cand_arr = np.array(best_seq, dtype=float)
                cand_arr = subgradient_descent_exact(
                    cand_arr,
                    polish_deadline,
                    max_iter=60_000,
                    step0=0.12,
                )
                try:
                    val2 = evaluate_sequence(cand_arr.tolist())
                    print(f"   Polished score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, cand_arr.tolist()
                        print(f"   *** Updated best after polish: {best_val:.6f} ***")
                except Exception:
                    pass

    # ------------------------------------------------------------------
    # 6. Final global polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=150_000,
            step0=0.08,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 7. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
```
