def adam_optimize(init_a, deadline, max_iters=30000, base_lr=0.05, lam_min=0.5, lam_max=150.0, ramp_iters=5000, eval_interval=200):
    # ensure initial sum=1
    a0 = np.array(init_a, dtype=float)
    a0 = a0 / a0.sum()
    u = np.log(a0 + 1e-12)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a0.copy()
    best_score = compute_score(best_a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # Compute a from logits
        a = np.exp(u - np.max(u))
        a = a / a.sum()
        lam = lam_min + (lam_max - lam_min) * min(1.0, t / ramp_iters)
        # Obtain gradient and smooth score
        _, grad_a = smooth_score_and_grad(a, lam)
        # Convert to gradient wrt u
        inner = np.dot(a, grad_a)
        grad_u = a * (grad_a - inner)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr_t = base_lr * (np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t))
        u = u - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur_score = compute_score(a)  # actual max-based
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}]  iter {t:,}  new best {best_score:.6f}")

    # evaluate final a
    final_score = compute_score(best_a)
    if final_score < best_score:
        best_score = final_score
        best_a = best_a.copy()
    return best_a, best_score

"""
Adam‑Softmax optimizer for the step‑function “max‑autocorrelation” problem.

We minimise
    score(a) = 2·n·max(conv(a,a)) / (sum(a))²          (scale‑invariant)

by working on the probability simplex Σa = 1, a_i ≥ 0.
The algorithm consists of three stages:

1️⃣  Random restarts – a Dirichlet draw (with a random concentration α) gives
    a diverse initial point on the simplex.

2️⃣  Adam optimisation on the logits u (a = softmax(u)).
    A smooth surrogate of the max (log‑sum‑exp with a temperature λ) is used:
        f_λ(a) = 2·n· (log Σ exp(λ·conv(a,a)))/λ .
    Its gradient wrt a is
        ∇_a f_λ = 2·n·2·(a_rev ∗ w)   with   w_i ∝ exp(λ·conv_i) .
    By the chain rule we obtain the gradient w.r.t. the logits u,
    then perform Adam updates.  λ is increased linearly from a small value
    (≈0.5) to a large one (≈150) – a continuation method that sharpens the
    surrogate and drives the solution toward a true minimiser of the max.

3️⃣  A short stochastic hill‑climber that moves a fraction of the mass from
    one index to another and accepts the move whenever the exact (non‑smoothed)
    objective improves.

The routine keeps the best vector seen over all restarts, prints intermediate
scores and finally returns the best list of floats.  All helper functions are
defined at top level; no I/O aside from `print` is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real FFT, fast)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT.
    Result length is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective helpers (working with Σa = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def smooth_score_and_grad(a: np.ndarray, lam: float):
    """
    Returns the smoothed objective value 2·n·(log‑sum‑exp)/λ
    and its gradient w.r.t. a (including the factor 2·n).

    Parameters
    ----------
    a : ndarray, shape (n,)
        Probability vector (Σa = 1, a_i ≥ 0).
    lam : float
        Temperature for the softmax smoothing (λ > 0).

    Returns
    -------
    smooth_score : float
        2·n·smooth_max(conv(a,a)).
    grad_a : ndarray, shape (n,)
        Gradient of the smoothed objective w.r.t. a.
    """
    n = a.shape[0]

    # linear convolution a*a
    conv = convolution_fft(a, a)               # length 2n‑1
    maxc = np.max(conv)

    # softmax weights (numerically stable)
    exp_vals = np.exp(lam * (conv - maxc))      # ≤ 1
    w = exp_vals / np.sum(exp_vals)            # shape (2n‑1,)

    # smooth value = maxc + log(sum_exp)/λ
    smooth_val = maxc + np.log(np.sum(exp_vals)) / lam
    smooth_score = 2.0 * n * smooth_val

    # gradient of Σ w_i·conv_i w.r.t. a
    a_rev = a[::-1]
    g_full = convolution_fft(a_rev, w)         # length 3n‑2
    grad_a = 2.0 * n * g_full[n - 1:n - 1 + n]   # central n entries

    return smooth_score, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the logits (softmax parametrisation)
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    base_lr: float = 0.10,
    lam_min: float = 0.5,
    lam_max: float = 150.0,
    ramp_iters: int = 20_000,
    eval_interval: int = 250,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits u (a = softmax(u)) using a
    λ‑continuation of the smooth max objective.

    Returns the best vector (on the simplex) and its true score.
    """
    # initialise logits safely (add tiny epsilon to avoid log(0))
    a0 = np.asarray(init_a, dtype=float)
    a0 = np.maximum(a0, 1e-30)
    a0 /= a0.sum()
    u = np.log(a0)

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a0.copy()
    best_score = compute_score(best_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # ----- current probabilities -----
        u_max = np.max(u)
        a = np.exp(u - u_max)
        a /= a.sum()

        # ----- λ schedule (linear warm‑up) -----
        lam = lam_min + (lam_max - lam_min) * min(1.0, t / ramp_iters)

        # ----- smoothed objective & gradient w.r.t. a -----
        _, grad_a = smooth_score_and_grad(a, lam)

        # ----- convert to gradient w.r.t. logits (softmax Jacobian) -----
        inner = np.dot(a, grad_a)
        grad_u = a * (grad_a - inner)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr_t = base_lr * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
        u -= lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # ----- periodic exact evaluation -----
        if t % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}]  iter {t:,}  new best {best_score:.6f}")

    # final check in case the last a is best
    final_score = compute_score(best_a)
    if final_score < best_score:
        best_score = final_score
        best_a = best_a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps: int = 600_000):
    """
    Stochastic hill‑climber that moves a fraction of mass from one index
    to another and accepts the move only if the true objective improves.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    # ensure we are on the simplex
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_score = compute_score(best_a)

    # transfer fraction (scaled adaptively)
    step_scale = 0.12

    steps = 0
    while steps < max_steps and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose moving a fraction of mass from i to j
        delta = step_scale * best_a[i]
        if delta <= 0:
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # stay on the simplex (project tiny negatives to zero)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # success → enlarge step
        else:
            step_scale *= 0.98                         # failure → shrink step

        steps += 1

    return best_a.tolist(), best_score

# Main entry point required by the harness

def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine performs many random restarts, Adam optimisation on each
    (with a λ‑continuation), and finishes with a short hill‑climbing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Determine sequence length (use fallback if present)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 1024

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback (if a previous construction is supplied)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main optimisation loop – random restarts until time runs out
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline - 12.0:      # reserve ~12 s for final polish
        restart_counter += 1

        # draw a Dirichlet start with a random concentration α
        alpha = np.random.uniform(0.02, 3.0)
        a0 = np.random.dirichlet(np.full(n, alpha))

        # allocate a slice of the remaining budget to this restart
        remaining = deadline - time.time()
        sub_deadline = time.time() + max(6.0, remaining * 0.55)

        print(f"[{time.time():.1f}] Restart {restart_counter}, α={alpha:.3f}")

        refined_a, refined_score = adam_optimize(
            init_a=a0,
            deadline=sub_deadline,
            base_lr=0.12,
            lam_min=0.5,
            lam_max=180.0,
            ramp_iters=15_000,
            eval_interval=300,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"  → new global best after Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – hill‑climber (and a short Adam “finetune”)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print("\nFinal hill‑climb polishing")
        polished_seq, polished_val = hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=500_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

        # one short Adam refinement after hill‑climb
        if time.time() < deadline - 1.0:
            print("\nFinal short Adam refinement")
            refined_a, refined_score = adam_optimize(
                init_a=np.array(best_seq),
                deadline=deadline,
                max_iters=30_000,
                base_lr=0.08,
                lam_min=0.5,
                lam_max=200.0,
                ramp_iters=10_000,
                eval_interval=200,
            )
            if refined_score < best_val:
                best_val = refined_score
                best_seq = refined_a.tolist()
                print(f"Adam refinement improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # final sanity normalisation (Σa = 1)
    best_seq = np.array(best_seq, dtype=float)
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0:
        best_seq = np.full_like(best_seq, 1.0 / n)
    else:
        best_seq = best_seq / best_seq.sum()
    best_seq = best_seq.tolist()

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq
