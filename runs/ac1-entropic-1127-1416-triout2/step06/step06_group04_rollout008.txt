"""
LP‑Driven search for a step‑function that minimizes
      2·n·max(conv(a,a)) / (sum a)²

The algorithm alternates two phases:

1️⃣  Linear‑programming phase (the “LP‑step”) – for the current
    probability vector `f` (non‑negative, Σf=1) we compute the
    maximal total mass that can be added while keeping the
    self‑convolution bounded by the current worst value `M`.
    The LP is
        maximise   Σ b_j
        subject to 0 ≤ b ,  (f * b)[k] ≤ M   ∀k
    The optimizer returns a new direction `g = b / Σb`.  Mixing
    `f ← (1‑t)·f + t·g` (with a line‑search on `t`) never increases
    the worst convolution value, and typically lowers it.

2️⃣  Stochastic polishing – a cheap Gaussian walk on the simplex
    (projected after each perturbation) that accepts any downhill
    move.  This removes the tiny irregularities left after the LP
    phase.

Both phases respect the global time budget (default 1000 s).  The
candidate is always kept on the probability simplex, therefore the
official score equals the fast surrogate `2·n·max(conv)`.  The routine
guarantees a non‑empty return (the uniform 256‑vector) even in the
worst case.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = u - cssv / np.arange(1, n + 1)
    rho_idx = np.where(rho > 0)[0]
    if rho_idx.size == 0:
        theta = 0.0
    else:
        rho_max = rho_idx[-1]
        theta = cssv[rho_max] / (rho_max + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def build_conv_matrix_sparse(f: np.ndarray):
    """
    Sparse matrix representation of the linear map b → f * b
    (full linear convolution, length 2·n‑1).
    For each offset d = 0,…,n‑1 the constant value f[d] is placed on
    the d‑th sub‑diagonal.
    """
    from scipy.sparse import diags

    n = f.shape[0]
    # avoid exact zeros – they would make the LP unbounded
    eps = 1e-15
    f_clip = np.maximum(f, eps)

    # each diagonal (sub‑diagonal) has length n
    data = [np.full(n, f_clip[d]) for d in range(n)]
    offsets = np.arange(0, -n, -1)          # 0, -1, -2, …, -(n‑1)
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


# ----------------------------------------------------------------------
# Simple random walk for final polishing
# ----------------------------------------------------------------------
def random_walk_polish(f: np.ndarray, deadline: float) -> np.ndarray:
    """
    Cheap Gaussian perturbations on the simplex, accepting only
    improvements of the surrogate objective 2·n·max(conv).
    """
    n = f.shape[0]
    best = f.copy()
    best_val = 2.0 * n * np.max(np.convolve(best, best))
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        sigma = 0.015 / np.sqrt(it)          # shrinking step size
        cand = best + sigma * rng.standard_normal(n)
        cand = project_to_simplex(cand)
        cur_val = 2.0 * n * np.max(np.convolve(cand, cand))
        if cur_val < best_val:
            best_val = cur_val
            best = cand
        # very occasional progress print
        if it % 20000 == 0:
            print(f"    Random‑walk iteration {it:,d} best={best_val:.6f}")
    return best

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    iter_no = 0

        conv_f = np.convolve(f, f)

        # Build the constraint matrix A (sparse)
        A = build_conv_matrix_sparse(f)

        c = -np.ones(n)

        res = scipy.optimize.linprog(
            c,

            b_ub=b_ub,

        if not res.success:

            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            print(f"LP returned zero total mass at iteration {iter_no}.")
            break

        # New direction (still a probability vector)
        g = b / sum_b

        # ------------------------------------------------------------------
        # Line‑search mixing: find a t ∈ (0,1] with a genuine improvement.
        # ------------------------------------------------------------------
        improved = False
        t = 0.5                       # start with a moderate step
        while t > 1e-4:
            f_new = (1.0 - t) * f + t * g
            # re‑normalise (tiny drift may appear)
            f_new = f_new / f_new.sum()
            new_obj = 2.0 * n * np.max(np.convolve(f_new, f_new))
            if new_obj < cur_obj - 1e-12:
                # accept the update
                f = f_new
                cur_obj = new_obj
                improved = True
                break
            t *= 0.5

        if not improved:
            # No improvement with any reasonable step size ⇒ stop early.
            print(f"Iteration {iter_no}: mixing failed to improve objective.")
            break

        # keep global best
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            print(f"Iter {iter_no:3d}: new best surrogate = {best_obj:.6f}")

        # early exit if we already beat the target (≈1.503)
        if best_obj <= 1.503:
            print("Target reached – exiting LP loop.")
            break

    # --------------------------------------------------------------
    # 3.  Final stochastic polishing (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:

        best_f = random_walk_polish(best_f, deadline)
