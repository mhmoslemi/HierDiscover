"""
Propose a low‑autocorrelation step sequence.

The search consists of four phases:

1️⃣  **Diverse seeding** – generate many candidates from a few
    parametric families (arcsine, beta, power‑shape, mixtures,
    triangular bumps).  All candidates are normalised to sum = 1 and
    evaluated with a cheap surrogate objective
    ``score = 2·n·max(conv(a,a))`` (exact when Σa=1).

2️⃣  **Smooth‑max Adam optimisation** – starting from the best few seeds,
    minimise a smooth approximation of the max (log‑sum‑exp) by Adam on
    the soft‑max logits.  The temperature τ is increased from 0.5 to 50,
    turning the surrogate into the true max.

3️⃣  **Exact convex refinement** – run a long projected sub‑gradient descent
    on the genuine max‑convolution objective, followed by an
    exponentiated‑gradient polish (mirror‑descent with KL geometry).

4️⃣  **Local hill‑climbing** – random two‑coordinate mass transfers are
    performed on the current best vector; moves that improve the
    surrogate are kept (simulated‑annealing style acceptance is also
    possible if time permits).

All phases respect a global 1000 s budget and keep the globally best
vector (according to the official ``evaluate_sequence``) for the final
return.  The implementation is completely self‑contained and uses only
NumPy and the provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast exact surrogate for the official score.
    a is assumed to be non‑negative and normalised to Σa = 1,
    therefore the denominator of evaluate_sequence disappears and
    the score is ``2·n·max(conv(a,a))``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al. (2008) implementation.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators (used for seeding)

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family on {0,…,n‑1}: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    return seq / seq.sum()


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()

# Exact sub‑gradient descent (convex refinement)
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.12) -> np.ndarray:
    """
    Euclidean‑projected sub‑gradient descent on the true max‑convolution.
    ``initial_a`` must be normalised (Σa = 1).  The step size decays as
    step0 / sqrt(t).  Stops when the deadline is hit.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # tolerance to locate the indices attaining the maximum
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·(a * mask) convolved with reversed mask
        rev_mask = mask[::-1]
        conv_rev = convolution_fft(a, rev_mask)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / math.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 120_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Exponentiated‑gradient polishing (mirror‑descent w.r.t. KL).
    """
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev_mask = mask[::-1]
        conv_rev = convolution_fft(a, rev_mask)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (math.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def adam_optimize_smooth_max(a_init: np.ndarray,
                             deadline: float,
                             max_iters: int = 60_000,
                             tau_start: float = 0.5,
                             tau_end: float = 50.0,
                             base_lr: float = 0.35) -> np.ndarray:
    """
    Optimise the smooth log‑sum‑exp surrogate of ``max(conv)``.
    The optimisation variable is the soft‑max logits w (so Σa = 1 automatically).
    """
    n = a_init.shape[0]
    L = 2 * n - 1

    # initialise logits from a_init (add tiny jitter for stability)
    w = np.log(a_init + 1e-12)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / max_iters)

        # softmax to get current a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # convolution a*a
        conv = convolution_fft(a, a)

        # compute smooth‑max weights: softmax(τ·conv)
        scaled = conv * tau
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)   # stabilise
        sum_exp = exp_scaled.sum()
        weights = exp_scaled / sum_exp               # shape (L,)

        # gradient wrt a (2·conv of a with reversed weights)
        rev_w = weights[::-1]
        conv_rev = convolution_fft(a, rev_w)        # length n+L‑1
        grad_a = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # gradient wrt logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (learning rate decays as 1/√t)
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = base_lr / math.sqrt(it)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional logging with the exact surrogate
        if it % 5000 == 0:
            surrogate = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.2f} surrogate={surrogate:.6f}")

    # final softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    return a

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # keep numerical non‑negativity
            if a_new[i] < 0.0:
                continue
            new_score = compute_score(a_new)
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                successes += 1
                # occasional logging
                if successes % 200 == 0:
                    print(f"    hill‑climb improvements: {successes} (score={cur_score:.6f})")
        # final official evaluation
        try:
            val = evaluate_sequence(a.tolist())
            print(f"Hill‑climb final official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print("*** Updated global best after hill‑climbing! ***")
        except Exception as e:
            print(f"Evaluation failed after hill‑climbing: {e}")

    # ------------------------------------------------------------------
    # 5.  Guard – ensure we always return a non‑empty sequence
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 512
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
