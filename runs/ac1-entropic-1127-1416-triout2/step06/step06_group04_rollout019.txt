"""
Propose a step‐function (non‑negative sequence) that minimizes
    2·n·max(conv(a,a)) / (sum(a))**2

The algorithm proceeds in two phases:

1️⃣  **Exploratory sampling** – for several lengths (1024, 2048, 4096, …)
    we generate a large number of candidate probability vectors:
    * deterministic starters (uniform, arcsine, beta‑family grid,
      edge‑concentrated shapes),
    * random families (beta shapes, mixtures, Dirichlet samples, edge shapes).
    The fast surrogate score `2·n·max(conv)` (computed with FFT) is used
    and the best overall candidate is kept.

2️⃣  **Intensive polishing** – starting from the best sampled vector we
    apply a sequence of projected gradient optimisers:
        – flattening step (L2 distance of the self‑convolution to the flat
          level 1/(2n‑1)) using Adam,
        – smooth‑max step (log‑sum‑exp surrogate of the max) using Adam
          with a very high final temperature (λ≈3000),
        – exact sub‑gradient descent on the true max‑convolution,
        – stochastic Gaussian walk on the simplex (simulated‑annealing‑style).
    All stages respect the global deadline (default 1000 s).  The final
    official score is obtained with the provided `evaluate_sequence`
    routine.

If the search fails to produce any candidate a uniform 256‑vector is
returned as a safe fallback.
"""

import time
import numpy as np

# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (returns length len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (full length)."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)).
    The vector is assumed to be normalised (Σa = 1).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation – O(n log n).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Symmetric (or asymmetric) beta‑family density on {0,…,n‑1} (unnormalised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities (renormalised)."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) density."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_edge_shape(n: int, w_frac: float) -> np.ndarray:
    """Mass uniformly placed on the two edge blocks of relative width w_frac."""
    w = max(1, int(round(w_frac * n)))
    shape = np.zeros(n, dtype=float)
    shape[:w] = 1.0
    shape[-w:] = 1.0
    total = shape.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return shape / total


def generate_dirichlet_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """Random Dirichlet vector by sampling Gamma(α,1) with random α."""
    # choose a concentration parameter uniformly in (0.1,5)
    conc = rng.uniform(0.1, 5.0, size=n)
    gam = rng.gamma(conc, 1.0)
    total = gam.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return gam / total

def flatten_gradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Gradient of ‖conv(a,a) – avg‖₂² where avg = 1/(2n‑1) and Σa = 1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient: 4·(a ⊛ diff_rev)  (⊛ = convolution)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)          # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def adam_flatten(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.35,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the L₂ flattening loss.
    Returns the best vector (projected) and its fast surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        grad, loss = flatten_gradient(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 25000 == 0:
            print(f"    Adam‑flatten it={it:,d} loss={loss:.2e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth surrogate of the max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value B) where
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv‑max)).
    The gradient is derived for the full objective
        2·n·max(conv) (sum(a)=1 is enforced by projection).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights
    B = max_c + np.log(sum_exp) / lam

    # weighted sub‑gradient: 2·(a ⊛ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # since we keep Σa = 1 via projection, we only need the
    # gradient of the max term; the scaling factor 2·n is irrelevant
    # for the direction (the projection restores the scale).
    grad = weighted_grad
    return grad, B


def adam_max(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 400_000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_end: float = 3000.0,
    noise_every: int = 1000,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution surrogate.
    Returns the best vector (projected) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # quadratic temperature schedule
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam‑max it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} lam={lam:.1f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max; back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.30,
    max_iter: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    A simple back‑tracking line‑search guarantees non‑increase.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term (with double counting)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # back‑tracking line‑search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 60000 == 0:
            print(f"    Sub‑grad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (cheap Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = compute_score(a_prop)

        # accept if improvement, otherwise a tiny uphill chance
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 80000 == 0:
            print(f"    Random walk it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine performs an aggressive sampling phase followed by
    intensive gradient‑based polishing, always respecting the global
    1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Use previously known construction if available
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Massive exploratory sampling
    # --------------------------------------------------------------
    # candidate lengths – we start with 1024,2048,4096 and may add 8192 if time permits
    lengths = [1024, 2048, 4096]
    if time.time() < deadline - 200:   # reserve at least 200 s for polishing
        lengths.append(8192)

    global_best_score = float("inf")
    global_best_vec = None
    global_best_n = None

    for n in lengths:
        if time.time() >= deadline - 180:   # keep 180 s for later stages
            break
        print(f"\n=== Sampling length n = {n} ===")

        # ----- deterministic starters -----
        starters: List[np.ndarray] = [
            np.full(n, 1.0 / n),   # uniform
            arcsine_initial(n),    # arcsine
        ]

        # grid of symmetric beta exponents
        for alpha in np.linspace(0.05, 0.95, 19):
            starters.append(generate_beta_shape(n, alpha, alpha, eps=1e-6))

        # edge‑concentrated shapes
        for w_frac in [0.02, 0.05, 0.10, 0.15]:
            starters.append(generate_edge_shape(n, w_frac))

        # evaluate deterministic starters
        for vec in starters:
            sc = compute_score(vec)
            if sc < global_best_score:
                global_best_score = sc
                global_best_vec = vec.copy()
                global_best_n = n

        # ----- random sampling -----
        # aim for ~40 k candidates per length (adjusted by remaining time)
        max_samples = int(min(40000, (deadline - time.time()) * 25))
        for _ in range(max_samples):
            if time.time() >= deadline - 180:
                break
            r = rng.random()
            if r < 0.25:                     # Dirichlet
                vec = generate_dirichlet_shape(rng, n)
            elif r < 0.55:                   # simple beta shape
                alpha = rng.uniform(0.02, 3.5)
                beta = rng.uniform(0.02, 3.5)
                eps = rng.uniform(1e-8, max(0.01, 0.02 * n))
                vec = generate_beta_shape(n, alpha, beta, eps)
            elif r < 0.80:                   # mixture of two betas
                w = rng.random()
                a1 = rng.uniform(0.02, 3.5)
                b1 = rng.uniform(0.02, 3.5)
                e1 = rng.uniform(1e-8, max(0.01, 0.02 * n))
                a2 = rng.uniform(0.02, 3.5)
                b2 = rng.uniform(0.02, 3.5)
                e2 = rng.uniform(1e-8, max(0.01, 0.02 * n))
                vec = generate_beta_mixture(n, a1, b1, e1, a2, b2, e2, w)
            else:                            # edge shape with random width
                w_frac = rng.uniform(0.02, 0.3)
                vec = generate_edge_shape(n, w_frac)

            sc = compute_score(vec)
            if sc < global_best_score:
                global_best_score = sc
                global_best_vec = vec.copy()
                global_best_n = n

        print(f"  Best fast score for n={n} → {global_best_score:.6f}")

    # If we never found a valid vector (should not happen), fallback
    if global_best_vec is None:
        print("No candidate found during sampling – falling back to uniform 256‑vector.")
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        return best_seq

    # --------------------------------------------------------------
    # 2.  Intensive polishing of the globally best candidate
    # --------------------------------------------------------------
    a = global_best_vec.astype(float)
    n = a.shape[0]
    print(f"\n=== Polishing candidate of length {n} (fast score {global_best_score:.6f}) ===")

    # ---- flattening stage ----
    a, _ = adam_flatten(
        initial_a=a,
        time_deadline=deadline,
        max_iter=300_000,
        lr0=0.35,
    )

    # ---- smooth‑max Adam stage ----
    if time.time() < deadline - 5.0:
        a, _ = adam_max(
            initial_a=a,
            time_deadline=deadline,
            max_iter=400_000,
            lr0=0.12,
            lam_start=0.5,
            lam_end=3000.0,
            noise_every=1200,
            noise_scale=5e-5,
        )

    # ---- exact sub‑gradient descent ----
    if time.time() < deadline - 5.0:
        a, _ = subgradient_optimize(
            initial_a=a,
            time_deadline=deadline,
            step0=0.30,
            max_iter=300_000,
        )

    # ---- stochastic Gaussian walk ----
    if time.time() < deadline - 5.0:
        a, _ = random_local_search(
            initial_a=a,
            time_deadline=deadline,
            init_step=0.015,
            max_iter=300_000,
        )

    # ---- final official evaluation ----
    final_val = evaluate_sequence(a.tolist())
    print(f"\nFinal official score after polishing = {final_val:.6f}")

    if final_val < best_val:
        best_val = final_val
        best_seq = a.tolist()
        print(f"*** New global best (length {n}) = {best_val:.6f} ***")

    # --------------------------------------------------------------
    # 3.  Global polishing of the current best (if time remains)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 8.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # longer flattening
        a, _ = adam_flatten(
            initial_a=a,
            time_deadline=deadline,
            max_iter=400_000,
            lr0=0.38,
        )

        # stronger smooth‑max
        if time.time() < deadline - 4.0:
            a, _ = adam_max(
                initial_a=a,
                time_deadline=deadline,
                max_iter=500_000,
                lr0=0.10,
                lam_start=0.5,
                lam_end=4000.0,
                noise_every=1500,
                noise_scale=4e-5,
            )

        # sub‑gradient refinement
        if time.time() < deadline - 4.0:
            a, _ = subgradient_optimize(
                initial_a=a,
                time_deadline=deadline,
                step0=0.28,
                max_iter=400_000,
            )

        # final random walk
        if time.time() < deadline - 2.0:
            a, _ = random_local_search(
                initial_a=a,
                time_deadline=deadline,
                init_step=0.012,
                max_iter=400_000,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # 4.  Guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        print("No valid sequence produced – returning uniform 256‑vector.")
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
