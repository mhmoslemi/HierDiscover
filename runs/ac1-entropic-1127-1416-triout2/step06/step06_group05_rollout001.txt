"""
Propose a non‑negative step sequence with a low official score.

Algorithm
---------

1.  For a few vector lengths (1024, 2048) start from a few simple shapes:
    * uniform,
    * arcsine,
    * a random beta family shape.

2.  For each starter run a short *linear‑programming* phase.
    Let `f` be the current shape (∑f = 1) and `M = max(conv(f,f))`.
    We solve

        max  Σ b_j
        s.t. conv(f,b) ≤ M   (2 n‑1 linear constraints)
             b_j ≥ 0

    The solution `b` is the “most massive” vector that can be convolved
    with `f` without increasing the current peak `M`.  After scaling
    `g = b / Σb` the peak is reduced to at most `M/Σb`.  If the new
    shape `g` yields a smaller official value we accept it; otherwise we
    try a convex mix `f←(1‑t)f + t g` with a few trial values of `t`.

    The LP is solved with SciPy's HiGHS implementation using a sparse
    constraint matrix – the whole step is fast (≈ 0.5 s for n=2048).

3.  The LP loop is repeated (≤ 12 iterations or until no improvement)
    and therefore drives the max‑convolution value down rapidly.

4.  A brief *random‑walk polishing* (Gaussian perturbations on the
    simplex) is applied to escape possible plateaus.

5.  The best sequence among all lengths is evaluated with the official
    `evaluate_sequence`.  If nothing is found a uniform fallback is
    returned.

The whole routine respects the 1000 s budget, prints intermediate
scores and always returns a valid non‑negative list.
"""

import time
import numpy as np
from typing import List, Tuple

# Optional SciPy imports (LP solver + sparse matrices)
# ----------------------------------------------------------------------
try:

    from scipy.sparse import csr_matrix
except Exception:  # pragma: no cover
    linprog = None
    csr_matrix = None

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for any length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """Fast surrogate of the official objective (assumes Σa = 1)."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (beta families and arcsine)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density (used in the original baseline)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Linear‑programming helper (maximise Σb under conv(f,b) ≤ M)
# ----------------------------------------------------------------------


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σb_j   s.t.   conv(f,b) ≤ M,   b ≥ 0 .
    Returns the optimal b (not normalised) or None on failure.
    """
    if linprog is None:
        return None

    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # Build constraint matrix A (size (2n‑1) × n) in CSR format.
    # Row k corresponds to (f * b)_k = Σ_j f_{k‑j} b_j .
    # Only indices where 0 ≤ k‑j < n contribute.
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # j runs from j_start … j_end inclusive
        # value = f_{k‑j}
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])

    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=float)
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # maximise Σb  →  minimise -Σb
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 30},
        )
    except Exception:  # pragma: no cover
        return None

    if res.success:
        return np.maximum(res.x, 0.0)  # safeguard against tiny negatives
    return None


# ----------------------------------------------------------------------
# Simple random local walk (Gaussian perturbations on the simplex)
# ----------------------------------------------------------------------


def random_local_search(
    init_a: np.ndarray, deadline: float, max_iter: int = 80_000
) -> Tuple[np.ndarray, float]:
    """Cheap Gaussian walk; accepts improvements and occasional uphill moves."""
    a = init_a.copy()
    best = a.copy()
    best_sc = fast_score(a)
    n = a.size
    rng = np.random.default_rng()
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        # step size decays like 1/√it
        step = 0.02 / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        cand = a + noise
        cand = project_to_simplex(cand)
        sc = fast_score(cand)
        if sc < best_sc or (rng.random() < 1e-4 and sc < 10 * best_sc):
            a = cand
            if sc < best_sc:
                best_sc = sc
                best = cand.copy()
        if it % 20_000 == 0:
            print(f"    Random walk iter={it:,d} cur={sc:.6f} best={best_sc:.6f}")
    return best, best_sc

    # ------------------------------------------------------------------
    if not best_seq:
        # safe uniform fallback (length 256)
        best_seq = [1.0] * 256
        best_val = official_score(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
