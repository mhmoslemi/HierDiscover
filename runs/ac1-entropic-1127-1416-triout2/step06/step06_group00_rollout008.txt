"""
Propose a low‑score step‑function.

Algorithm Overview
------------------
1. **Initial seeds** – generate a diverse pool of candidate vectors:
   * random sparse supports,
   * exponentially decaying profiles (parameterised by a decay factor),
   * simple uniform vectors.
   All seeds are projected onto the probability simplex (∑aᵢ = 1,
   aᵢ ≥ 0).

2. **Smoothed optimisation** – minimise the smooth approximation of
   `max(conv(a,a))` using the Adam optimiser.
   The smoothed objective is the log‑sum‑exp:
       f_β(a) = (1/β)·log Σ exp(β·conv_i) .
   β is annealed from a modest value (≈5) to a large one (≈80),
   turning the smooth problem gradually into the original max‑problem.
   After every Adam step the iterate is projected back onto the simplex.

3. **Projected sub‑gradient refinement** – a classic sub‑gradient method
   with Nesterov‑style momentum is applied to the exact (non‑smooth)
   objective `max(conv)`.  The step size follows a 1/√t schedule.

4. **Stochastic hill‑climb polishing** – a cheap mass‑transfer / swap
   neighbourhood is explored; moves are accepted only if they improve
   the true score (the call to `evaluate_sequence`).  This step pushes
   the solution past remaining shallow local minima.

All stages respect the global 1000 s wall‑clock budget.  The routine
keeps the best exact score seen so far and returns the corresponding
sequence as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a ⊛ b using a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a⊛a).  This value is proportional to the true score,
    the proportionality factor being 1 / (Σa)² (=1 here).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        # uniform simplex point
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def seed_random_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random sparse vector with a random support size."""
    p = rng.uniform(0.05, 0.6)                 # density
    k = max(2, int(round(p * n)))             # support size
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def seed_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay profile a_i ∝ r^i (0<r<1), normalised."""
    # pick a decay factor r that gives a reasonably spread distribution
    # r close to 1 gives a near‑uniform vector; r close to 0 makes it
    # highly front‑loaded.  We sample on a log‑scale.
    log_r = rng.uniform(np.log(0.998), np.log(0.9999))
    r = np.exp(log_r)
    powers = np.power(r, np.arange(n, dtype=float))
    a = powers / powers.sum()
    return project_to_simplex(a)


def seed_uniform(n: int) -> np.ndarray:
    """Trivial uniform vector (used only as a fallback)."""
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed optimisation (log‑sum‑exp) using Adam
# ----------------------------------------------------------------------


def smooth_adam_optimize(start_a: np.ndarray,
                         time_deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 80.0,
                         lr: float = 0.6,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution using Adam.

    Parameters
    ----------
    start_a : np.ndarray
        Initial probability vector (sum = 1, non‑negative).
    time_deadline : float
        Absolute Unix time after which the method must stop.
    max_iters : int
        Upper bound on iterations.
    beta_start, beta_end : float
        Annealing schedule for the smoothness parameter.
    lr : float
        Base learning rate for Adam.
    eps : float
        Small constant for numerical stability.

    Returns
    -------
    best_a, best_score
        The best vector seen (still on the simplex) and its exact
        (non‑smooth) score as returned by ``compute_score``.
    """
    a = start_a.astype(float).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ----- β‑annealing -------------------------------------------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- smoothed objective -----------------------------------------
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                     # probability distribution

        # ----- gradient of f_β (cross‑correlation) -----------------------
        a_rev = a[::-1]
        # cross‑correlation a_rev * w (full conv), then take central n entries
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # ----- Adam update ------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        a = a_new

        # ----- keep best exact score occasionally -------------------------
        if it % 200 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    # final check
    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient refinement with momentum (Nesterov)
# ----------------------------------------------------------------------


def subgrad_momentum_optimize(init_a: np.ndarray,
                              time_deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.09,
                              momentum: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with Nesterov momentum on the exact
    max‑convolution objective.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    vel = np.zeros_like(a)                     # momentum buffer

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Nesterov‑style update
        step = step0 / np.sqrt(t)
        vel = momentum * vel - step * g
        a_new = a + vel
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if t % 500 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

        # small random jitter every 2000 steps to shake out of plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    # final verification
    cur = compute_score(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb polishing (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(sequence,
               time_deadline,
               max_steps: int = 500_000) -> tuple[list[float], float]:
    """
    Very cheap stochastic hill‑climb: randomly swap two entries or
    transfer a fraction of mass from one position to another.  Every
    candidate is projected onto the simplex and evaluated with the *true*
    objective (via ``evaluate_sequence``).
    """
    a = np.array(sequence, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())

    step_scale = 0.10   # initial proportion of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_val = evaluate_sequence(cand.tolist())

        if cur_val < best_val:
            best_val = cur_val
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_val
