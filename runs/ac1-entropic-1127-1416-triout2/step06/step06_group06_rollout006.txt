"""Hybrid optimizer with hard max gradient and annealed hill‑climb.

The algorithm works on the probability simplex (sum = 1).  It
1. performs several random Dirichlet restarts;
2. on each restart runs a smoothed exponentiated‑gradient phase,
   followed by a hard‑max sub‑gradient phase that directly follows the
   worst convolution entry;
3. finally refines the best solution with a simulated‑annealing hill‑climber.
The whole search respects the 1000 s budget and returns the best
sequence found.

All helper functions are top‑level; no I/O, no closures.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_score(a: np.ndarray) -> float:
    """Fast replica of evaluate_sequence for vectors with sum == 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# ----------------------------------------------------------------------
# Smoothed exponentiated gradient descent (identical to the previous code)
# ----------------------------------------------------------------------
def smooth_exp_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    lam_start: float = 1.0,
    lam_end: float = 30.0,
    eval_interval: int = 400,
) -> tuple[np.ndarray, float]:
    """Exponential‑gradient descent on a smoothed surrogate of max(conv)."""
    a = init_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # convolution a*a
        conv = convolution_fft(a, a)
        M = np.max(conv)
        # temperature schedule for softmax
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / 2_000_000)
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)
        g = 2.0 * g_full[n - 1 : n - 1 + n]
        eta = eta0 / np.sqrt(it)
        step = -eta * g
        step = np.clip(step, -50, 50)
        a = a * np.exp(step)
        a = project_to_simplex(a)
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] smooth iter {it:,} new best {best_score:.6f}")
    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

# Hard‑max sub‑gradient optimizer
# ----------------------------------------------------------------------
def hard_max_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.6,
    eval_interval: int = 300,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that follows the true worst convolution entry."""
    a = init_a.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        M = np.max(conv)
        # mask of active (max) entries – tolerance avoids floating noise
        tol = 1e-12
        mask = (conv >= M - tol).astype(float)
        # if many active entries, weight them uniformly
        w = mask / np.maximum(np.sum(mask), 1.0)
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)
        g = 2.0 * g_full[n - 1 : n - 1 + n]
        eta = eta0 / np.sqrt(it)
        step = -eta * g
        step = np.clip(step, -50, 50)
        a_new = a * np.exp(step)
        a_new = project_to_simplex(a_new)
        cur_score = compute_score(a_new)
        # simple monotone acceptance – keep the better of the two
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            a = a_new
            if it % eval_interval == 0:
                print(f"[{time.time():.1f}] hard_max iter {it:,} new best {best_score:.6f}")
        else:
            # if the new point is not better, reject it (keep old a)
            # still occasionally accept to escape plateaus – simple probabilistic rule
            if np.random.rand() < 0.001:  # rare random jump
                a = a_new
    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

# Simulated‑annealing hill‑climber
# ----------------------------------------------------------------------
def annealing_hill_climb(
    start_seq,
    deadline,
    max_steps: int = 1_000_000,
    init_temp: float = 0.02,
    final_temp: float = 1e-6,
) -> tuple[list[float], float]:
    """Stochastic hill‑climb with temperature‑controlled acceptance."""
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score
    steps = 0
    # exponential cooling schedule
    total_steps_allowed = max_steps
    while steps < total_steps_allowed and time.time() < deadline:
        # linear interpolation of temperature
        T = init_temp * ((final_temp / init_temp) ** (steps / total_steps_allowed))
        # propose a simple mass‑transfer move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a random fraction of the mass at i towards j
        frac = np.random.rand() * 0.2  # up to 20 % of a[i]
        delta = frac * a[i]
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # maintain simplex (the move already preserves sum, only need non‑negativity)
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)  # safety projection
        cand_score = compute_score(cand)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        else:
            # accept with probability exp(-(Δ)/T)
            prob = np.exp(-(cand_score - cur_score) / (T + 1e-12))
            if np.random.rand() < prob:
                a = cand
                cur_score = cand_score
        steps += 1
        # occasional larger random reshuffle to escape deep local minima
        if steps % 50000 == 0:
            # pick a small subset and randomise its distribution
            idx = np.random.choice(n, size=5, replace=False)
            sub = np.random.dirichlet(np.ones(5))
            cand = a.copy()
            total_sub_mass = cand[idx].sum()
            cand[idx] = total_sub_mass * sub
            cand = project_to_simplex(cand)
            cand_score = compute_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
        if steps % 200000 == 0:
            print(f"[{time.time():.1f}] hill‑climb step {steps:,} best {best_score:.6f}")
    return best_a.tolist(), best_score

    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # uniform point if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast objective (sum of coefficients is assumed to be 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent (same spirit as the previous code)
# ----------------------------------------------------------------------


def smooth_exp_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    lam_start: float = 1.0,
    lam_end: float = 30.0,
    eval_interval: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Smoothed exponentiated‑gradient descent on
        f(a) = max_k (a*a)[k]  (a lives on the simplex).

    A softmax over the convolution entries yields a smooth surrogate.
    The gradient of the surrogate is
        g = 2 * (a_rev  *  soft_weights)
    where the convolution is taken over the full length (3·n‑2) and the
    central n entries are kept.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # linear convolution a*a
        conv = convolution_fft(a, a)                # length 2n‑1
        M = np.max(conv)                           # current max value

        # softmax temperature (λ) slowly increases
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / 2_000_000)

        # compute softmax weights (stable version)
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)            # length 2n‑1, sums to 1

        # gradient = 2 * (a_rev * w)[n-1 : n-1+n]
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)         # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # step size
        eta = eta0 / np.sqrt(it)

        # multiplicative update with clipping for numeric safety
        step = -eta * g
        step = np.clip(step, -50, 50)              # avoid overflow in exp
        a = a * np.exp(step)

        # back onto the simplex
        a = project_to_simplex(a)

        # periodic evaluation of the true (non‑smoothed) score
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] smooth iter {it:,} new best {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient descent (directly follows the worst convolution)
# ----------------------------------------------------------------------


def hard_max_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.6,
    eval_interval: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that follows the true worst convolution entry.
    The update uses a multiplicative (exponentiated) step and projects
    back onto the simplex.  The method is monotone (keeps the best
    solution seen) but occasionally accepts a random jump to avoid
    stagnation.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = convolution_fft(a, a)
        M = np.max(conv)

        # mask of active (max) entries – tolerance avoids floating noise
        tol = 1e-12
        mask = (conv >= M - tol).astype(float)
        w = mask / max(np.sum(mask), 1.0)

        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        eta = eta0 / np.sqrt(it)
        step = -eta * g
        step = np.clip(step, -50, 50)

        a_new = a * np.exp(step)
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)

        # keep the better of the two points
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            a = a_new
            if it % eval_interval == 0:
                print(f"[{time.time():.1f}] hard_max iter {it:,} new best {best_score:.6f}")
        else:
            # rare random jump to escape plateaus
            if np.random.rand() < 0.001:
                a = a_new

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (final polishing)
# ----------------------------------------------------------------------


def annealing_hill_climb(
    start_seq,
    deadline,
    max_steps: int = 800_000,
    init_temp: float = 0.02,
    final_temp: float = 1e-6,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb with a temperature schedule.
    Moves transfer a random fraction of mass from one position to another.
    Worse moves are occasionally accepted according to exp(−Δ/T).
    """
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    steps = 0
    while steps < max_steps and time.time() < deadline:
        # exponential cooling
        T = init_temp * ((final_temp / init_temp) ** (steps / max_steps))

        # ---- basic mass‑transfer move ----
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move up to 20 % of the source mass
        frac = np.random.rand() * 0.2
        delta = frac * a[i]
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)   # keep numerical sanity

        cand_score = compute_score(cand)

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        else:
            # accept with probability exp(−Δ/T)
            prob = np.exp(-(cand_score - cur_score) / (T + 1e-12))
            if np.random.rand() < prob:
                a = cand
                cur_score = cand_score

        # ---- occasional larger reshuffle to break out of deep minima ----
        if steps % 50_000 == 0 and steps > 0:
            idx = np.random.choice(n, size=5, replace=False)
            sub = np.random.dirichlet(np.ones(5))
            total = a[idx].sum()
            cand = a.copy()
            cand[idx] = total * sub
            cand = project_to_simplex(cand)
            cand_score = compute_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()

        if steps % 200_000 == 0 and steps > 0:
            print(f"[{time.time():.1f}] hill‑climb step {steps:,} best {best_score:.6f}")

        steps += 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The procedure performs several random restarts, a smoothed exponentiated‑gradient
    phase, a hard‑max refinement, and finishes with an annealed hill‑climbing
    polishing phase.  The best sequence found before the time budget expires is
    returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Determine a sensible length.  Use the fallback length if available;
    # otherwise try a modestly larger size (up to 4096) which still fits
    # comfortably into the time budget.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2048
    # try a larger length if we have headroom
    n_big = min(4096, n * 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate a raw list with the official scorer.
    # ------------------------------------------------------------------
    def full_eval(seq):
        return evaluate_sequence(seq)

    # ------------------------------------------------------------------
    # Use an existing fallback, if present, as a baseline.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        fb_val = full_eval(fallback)
        print(f"Fallback score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val = fb_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random restarts with two‑stage optimisation.
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline - 15.0:               # keep ~15 s for polishing
        restart_counter += 1
        # choose a length: occasionally try the bigger size
        cur_n = n_big if np.random.rand() < 0.4 else n
        alpha = np.random.uniform(0.1, 5.0)
        a0 = np.random.dirichlet(np.full(cur_n, alpha))

        # allocate a slice of the remaining budget for this restart
        remaining = deadline - time.time()
        sub_deadline = time.time() + max(5.0, remaining * 0.3)

        print(f"[{time.time():.1f}] Restart {restart_counter}, n={cur_n}, α={alpha:.2f}")

        # ---- smoothed exponentiated‑gradient phase ----
        a_smooth, sc_smooth = smooth_exp_optimize(
            a0,
            deadline=sub_deadline,
            eta0=0.55,
            lam_start=0.8,
            lam_end=28.0,
            eval_interval=500,
        )

        # ---- hard‑max refinement (use what time is left after the smooth phase) ----
        if time.time() < deadline - 5.0:
            remaining2 = deadline - time.time()
            hard_deadline = time.time() + min(remaining2 * 0.6, 20.0)
            a_refined, sc_refined = hard_max_optimize(
                a_smooth,
                deadline=hard_deadline,
                eta0=0.6,
                eval_interval=400,
            )
        else:
            a_refined, sc_refined = a_smooth, sc_smooth

        # keep global best
        if sc_refined < best_val:
            best_val = sc_refined
            best_seq = a_refined.tolist()
            print(f"*** New global best {best_val:.6f} after restart {restart_counter}")

    # ------------------------------------------------------------------
    # Phase 2 – final annealed hill‑climb polishing.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n=== Final annealed hill‑climb ===")
        polished_seq, polished_val = annealing_hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=800_000,
            init_temp=0.02,
            final_temp=1e-6,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return value.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = full_eval(best_seq)

    final_score = full_eval(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq
