def hill_climb_peak_aware(a_initial, deadline, max_steps=50000):
    a = a_initial.copy()
    best = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    step_scale = 0.05
    while max_steps>0 and time.time() < deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # contribution vector
        idx = np.arange(n)
        j = k - idx
        contrib = np.zeros_like(a)
        mask = (j>=0) & (j < n)
        contrib[mask] = a[idx[mask]] * a[j[mask]]
        total = contrib.sum()
        if total == 0:
            # fallback random move
            i = np.random.randint(n)
        else:
            # sample i according to contrib
            probs = contrib / total
            i = np.random.choice(n, p=probs)
        # choose target j far from (k-i)
        # compute a set of indices that are not close to (k-i)
        avoid = (k - i)
        # distance threshold
        # Choose j random but ensure distance > n//4 or something.
        j_target = np.random.randint(n)
        # Could try few attempts to ensure distance large
        attempts = 0
        while attempts<5:
            j_target = np.random.randint(n)
            if abs(j_target - avoid) > n//4:
                break
            attempts+=1
        delta = step_scale * a[i]
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j_target] += delta
        # projection might not be needed as sum unchanged and nonnegative but ensure nonneg
        cand = np.maximum(cand, 0)
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand
            a = cand
            step_scale = min(step_scale*1.05, 0.3)
        else:
            # maybe accept occasionally
            if np.random.rand() < 0.01:  # small acceptance
                a = cand
            step_scale *= 0.97
        max_steps -= 1
    return best, best_sc

def adam_smooth_gd(a0, time_deadline, max_iter=5000, lr=0.5, beta1=0.9, beta2=0.999, eps_factor=0.2):
    a = a0.astype(float).copy()
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        eps = maxc * max(eps_factor * (0.99 ** (t - 1)), 0.01)
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)
        # gradient
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new

        # occasional jitter
        if t % 500 == 0:
            jitter = np.random.randn(n) * (lr * 0.001 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)
    return best_a, best_sc

def hill_climb_peak_aware(a_initial, time_deadline, max_steps=20000):
    a = a_initial.astype(float).copy()
    n = len(a)
    best = a.copy()
    best_sc = compute_score(a)
    step_scale = 0.08
    while max_steps>0 and time.time() < time_deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # contribution per index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]
        total = np.sum(contrib)
        if total > 0:
            probs = contrib / total
            i = np.random.choice(n, p=probs)
        else:
            i = np.random.randint(n)
        # pick target j far from mirror index
        target = np.random.randint(n)
        # ensure not same
        attempts = 0
        while attempts < 5 and (abs(target - (k - i)) < n // 4 or target == i):
            target = np.random.randint(n)
            attempts += 1
        delta = step_scale * a[i]
        if delta == 0:
            # pick another i?
            max_steps -= 1
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[target] += delta
        # ensure non-negative
        cand = np.maximum(cand, 0.0)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand
            a = cand

        else:
            # occasional acceptance
            if np.random.rand() < 0.02:
                a = cand
            step_scale *= 0.97
        max_steps -= 1
    return best, best_sc

"""
Multi‑phase optimizer for the max‑convolution objective.

The search consists of:
  1. Deterministic seeds (Hann, triangular, Dirichlet‑draws).
  2. Massive random sampling on several lengths (1024‑8192) with a
     probability‑simplex projection.
  3. Adam‑based gradient descent on a smooth soft‑max surrogate of
     the max‑convolution (exactly the same gradient as the baseline
     but with adaptive first‑/second‑moment steps).
  4. A peak‑aware hill‑climber that preferentially moves mass away
     from the indices contributing to the current maximal autocorrelation.
  5. Optional symmetry averaging and a final call to the official
     `evaluate_sequence`.  All steps respect the 1000 s budget and
     keep the sequence on the probability simplex (non‑negative,
     sum = 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (non‑negative, sum = 1).  Works for any non‑empty vector.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors already normalised
    to sum(a)=1.  Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple random generators
# ----------------------------------------------------------------------


def random_sparse_vector(n: int) -> np.ndarray:
    """
    Random probability vector of length n.
    A random subset of coordinates (density 10‑90 %) receives equal mass.
    """
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def random_dirichlet_vector(n: int, alpha: float) -> np.ndarray:
    """
    Draw a Dirichlet‑type vector (Gamma(alpha,1) i.i.d.) and normalise.
    Alpha controls sparsity (alpha<<1 → spiky, alpha>>1 → smooth).
    """
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def hann_window(n: int) -> np.ndarray:
    """Classic Hann (raised‑cosine) window, normalised to sum = 1."""
    w = 0.5 * (1 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    w /= w.sum()
    return w


def triangular_window(n: int) -> np.ndarray:
    """
    Symmetric discrete triangular window, normalised.
    The peak is at the centre (or two centre points for even n).
    """
    m = (n - 1) / 2.0
    i = np.arange(n)
    w = 1.0 - np.abs(i - m) / (m + 1.0)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Phase‑2: Adam optimiser on a smooth soft‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_gd(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5_000,
    lr: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_factor: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smoothed objective

        f_ε(a) = ε·log Σ_k exp( (a⊛a)_k / ε ),

    where ε shrinks with the iteration count (same schedule as the
    baseline).  The exact gradient is

        ∇f = 2·(w_rev ⊛ a),

    with w = softmax((a⊛a)/ε).  After each step the vector is projected
    onto the simplex.  Returns the best vector (lowest exact score) seen.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε never drops below 1 % of the current max
        eps = maxc * max(eps_factor * (0.99 ** (t - 1)), 0.01)

        # soft‑max weights
        exp_vals = np.exp((conv - maxc) / eps)          # stable shift
        w = exp_vals / np.sum(exp_vals)                # Σ w = 1

        # gradient  g = 2·(w_rev ⊛ a)
        w_rev = w[::-1]                                # length 2n‑1
        conv_rev = convolution_fft(a, w_rev)           # length 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny random jitter to escape plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (lr * 0.001 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase‑3: Peak‑aware hill‑climbing
# ----------------------------------------------------------------------


def hill_climb_peak_aware(
    a_initial: np.ndarray,
    time_deadline: float,
    max_steps: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber that focuses on the shift where the
    convolution attains its maximum.  It preferentially transfers
    mass away from indices that contribute most to that peak.
    """
    a = a_initial.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    step_scale = 0.08  # initial fraction of mass to move

    while max_steps > 0 and time.time() < time_deadline:
        # current convolution and its dominant shift
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # contributions of each index to the peak (a_i * a_{k-i})
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]
        total = contrib.sum()

        if total > 0:
            probs = contrib / total
            i = np.random.choice(n, p=probs)
        else:
            i = np.random.randint(n)

        # choose a destination index far enough from the “mirror” partner
        partner_i = k - i
        j = np.random.randint(n)
        tries = 0
        while tries < 5 and (abs(j - partner_i) < n // 4 or j == i):
            j = np.random.randint(n)
            tries += 1

        delta = step_scale * a[i]
        if delta <= 0.0:
            max_steps -= 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # enforce feasibility
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cand_sc = compute_score(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_a = cand
            a = cand
            step_scale = min(step_scale * 1.07, 0.30)   # become bolder
        else:
            # occasional uphill acceptance (tiny probability)
            if np.random.rand() < 0.015:
                a = cand
            step_scale *= 0.96                         # become more cautious

        max_steps -= 1

    return best_a, best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Replace a by the symmetric average (a + reverse(a))/2
    and re‑project onto the simplex.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

# Main entry point for the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the given time budget
    (≈ 1000 s) and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback using a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – deterministic seeds (Hann, triangular, Dirichlet draws)
    # ------------------------------------------------------------------
    deterministic_lengths = [1024, 2048, 4096, 8192]
    for n in deterministic_lengths:
        for maker in (
            hann_window,
            triangular_window,
            lambda n: random_dirichlet_vector(n, 0.5),
            lambda n: random_dirichlet_vector(n, 2.0),
        ):
            if time.time() >= deadline - 5.0:
                break
            a = maker(n)
            a = project_to_simplex(a)          # safety net
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[det] n={n}, shape={maker.__name__ if hasattr(maker, '__name__') else 'custom'}, score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – massive random sampling (maintain a small elite list)
    # ------------------------------------------------------------------
    elite_size = 5
    elite = []                               # list of (score, vector)

    while time.time() < deadline - 30.0:      # reserve ~30 s for later phases
        n = int(np.random.choice([1024, 2048, 4096, 8192]))
        if np.random.rand() < 0.55:
            a = random_sparse_vector(n)
        else:
            alpha = np.random.uniform(0.1, 5.0)
            a = random_dirichlet_vector(n, alpha)

        sc = compute_score(a)

        # keep the absolute best
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[rand] n={n}, score={sc:.6f}")

        # maintain the elite list
        if len(elite) < elite_size:
            elite.append((sc, a))
        else:
            worst_idx, worst_sc = max(enumerate(elite), key=lambda x: x[1][0])
            if sc < worst_sc:
                elite[worst_idx] = (sc, a)

    # ------------------------------------------------------------------
    # Phase 2 – Adam gradient refinement of the elite members
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0 and elite:
        per_candidate = remaining / len(elite)
        for idx, (sc, vec) in enumerate(elite):
            sub_deadline = time.time() + per_candidate - 2.0
            if time.time() >= sub_deadline:
                break
            a_ref, _ = adam_smooth_gd(
                a0=vec,
                time_deadline=sub_deadline,
                max_iter=5_000,
                lr=0.5,
                eps_factor=0.2,
            )
            ref_sc = compute_score(a_ref)
            if ref_sc < best_score:
                best_score = ref_sc
                best_seq = a_ref.tolist()
                print(f"[adam] elite {idx} improved to {ref_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – peak‑aware hill‑climbing on the current best vector
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_current = np.array(best_seq, dtype=float)
        hill_deadline = deadline - 2.0
        a_hill, hill_sc = hill_climb_peak_aware(
            a_initial=a_current,
            time_deadline=hill_deadline,
            max_steps=20_000,
        )
        if hill_sc < best_score:
            best_score = hill_sc
            best_seq = a_hill.tolist()
            print(f"[hill] improved to {hill_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final symmetry averaging (cheap, often beneficial)
    # ------------------------------------------------------------------
    a_sym = final_symmetry_average(np.array(best_seq, dtype=float))
    sym_sc = compute_score(a_sym)
    if sym_sc < best_score:
        best_score = sym_sc
        best_seq = a_sym.tolist()
        print(f"[sym] symmetry averaging improved to {sym_sc:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided `evaluate_sequence`)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
