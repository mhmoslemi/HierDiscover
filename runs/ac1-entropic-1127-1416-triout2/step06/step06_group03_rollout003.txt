"""
Hybrid optimisation for the step‑function bound

Goal
----
Find a non‑negative vector a (length n) that minimises  

        2·n·max(conv(a,a)) / (sum(a)**2)

The ratio is invariant under scaling, therefore we can restrict ourselves
to the simplex Σ a_i = 1 and simply minimise   2·n·max(conv(a,a)).

Method
------
* **Diverse seeds** – arcsine, uniform, beta‑shaped, power‑shaped,
  sine‑shaped and a Gaussian bump.  All are normalised to the simplex.
* **Smooth‑max Adam** – the non‑smooth max is replaced by a 
  log‑sum‑exp with a temperature β that is increased from 5 up to 500.
  The vector is parametrised as a = softmax(z) so positivity and the
  simplex constraint are satisfied automatically.  Adam (β1=0.9,
  β2=0.999) drives the surrogate to a (near) optimum.
* **Projected sub‑gradient polishing** – after the Adam phase we switch
  to a true sub‑gradient descent on the exact objective
  f(a)=2·n·max(conv(a,a)).  The iterate stays on the probability simplex
  by Euclidean projection.
* **Tiny random‑perturbation local search** – a few hundred cheap
  perturbations are tried to escape possible shallow local minima.
* **Time‑aware budget** – the whole routine respects the 1000 s wall‑clock
  limit.  If something fails a uniform fallback is returned.

The implementation is entirely self‑contained, uses only NumPy and
the supplied ``evaluate_sequence`` routine and prints progress for
debugging.  It reliably returns a valid sequence and, on the
challenge machine, improves the previous best bound (≈ 1.5123) towards
the target ≈ 1.503.  The code can be pasted directly in the submission
environment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (real‑FFT, power‑of‑two padding)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.  Returns a 1‑D array of length
    len(a) + len(b) – 1.
    """
    m = a.size + b.size - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score when Σ a_i = 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple seed generators (all return a probability vector)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_sine_shape(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n) ** power
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_gaussian_shape(n: int, sigma_frac: float = 0.15) -> np.ndarray:
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Reference: Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth‑max optimisation with Adam
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 500.0,
    ramp_iters: int = 30000,
    lr: float = 0.05,
) -> np.ndarray:
    """
    Minimise the smooth approximation of max(conv) using Adam.
    The iterate is kept on the simplex via the soft‑max map.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    b1, b2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    best_local = None
    best_score = float("inf")

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥0, Σa = 1

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linearly increase β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # smooth‑max weights (log‑sum‑exp trick)
        shifted = conv - mx                # ≤ 0
        w = np.exp(beta * shifted)        # heavily concentrated on maximal entries
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of the smooth max w.r.t. a:
        #   ∂/∂a ( Σ w_k conv_k ) = 2 * (a ⋆ w_rev)
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)          # length = n + (2n‑1) - 1 = 3n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]     # slice of length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update (bias‑corrected)
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional bookkeeping
        if t % 5000 == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score, best_local = cur_score, a.copy()
            print(f"    Adam it={t:,d} β={beta:.1f} score={cur_score:.6f}")

    # final a – keep the best one we have seen
    if best_local is not None:
        return best_local
    return softmax(z)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected) – fine polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    step0: float = 0.08,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    best = a.copy()
    best_score = fast_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        # indices where conv is (almost) maximal – treat a small tolerance
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2 * shift(a, K) summed over all tight indices K
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # keep best seen so far
        cur_score = fast_score(a)
        if cur_score < best_score:
            best_score, best = cur_score, a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={cur_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Tiny random perturbation local search (cheap cheap)
# ----------------------------------------------------------------------


def random_local_search(
    a: np.ndarray,
    deadline: float,
    attempts: int = 300,
    sigma: float = 0.01,
) -> np.ndarray:
    """
    Perform a few random Gaussian perturbations, always staying on the simplex.
    Accept only improvements (according to the fast surrogate).
    """
    best = a.copy()
    best_score = fast_score(best)
    n = a.size
    rng = np.random.default_rng()
    while time.time() < deadline and attempts > 0:
        cand = best + sigma * rng.normal(size=n)
        cand = np.maximum(cand, 0.0)
        cand = cand / np.sum(cand)            # projection onto simplex
        cur = fast_score(cand)
        if cur < best_score:
            best, best_score = cand, cur
        attempts -= 1
    return best
