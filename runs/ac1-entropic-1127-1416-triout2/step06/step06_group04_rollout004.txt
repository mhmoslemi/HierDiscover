"""
Propose a sequence of non‑negative heights minimising
    2·n·max(conv(a,a)) / (∑a)²
(the official `evaluate_sequence` score).

The algorithm follows several phases, each respecting the global
1000 s budget:

1️⃣  **Exploratory sampling** – random beta‑family / mixture shapes and a few
    deterministic starters (uniform, arcsine, exponential, symmetric beta).
    The best candidate (fast surrogate score) is kept.

2️⃣  **Flattening (L₂ loss)** – Adam optimiser minimises the L₂ distance of
    the self‑convolution to a perfectly flat level.  This reduces the peak
    of the convolution.

3️⃣  **Smooth‑max Adam** – minimise a log‑sum‑exp surrogate of the
    max‑convolution.  The temperature λ is increased (1 → ~1500) so the
    surrogate approaches the true L∞ norm.

4️⃣  **Exact sub‑gradient descent** – projected sub‑gradient descent on the
    true max‑convolution with back‑tracking line‑search.

5️⃣  **Stochastic local walk** – cheap Gaussian perturbations on the simplex,
    accepting improvements (and occasional tiny uphill moves) to escape
    shallow basins.

Each stage runs until its allocated portion of the remaining time expires,
so the whole search automatically adapts to the available budget.
If no candidate is found a safe uniform 256‑vector fallback is returned.

The code is self‑contained, uses only NumPy and the provided
`evaluate_sequence` function, and prints progress information.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)).
    Assumes a is already normalised to Σa = 1.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Duchi et al. (2008) – O(n log n).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """Symmetric beta‑family density on {0,…,n‑1} (unnormalised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta densities (renormalised)."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) density."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_exponential_shape(n: int, rate: float) -> np.ndarray:
    """Symmetric exponential decay from the ends."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Gradient of the flattening (L₂) loss
# ----------------------------------------------------------------------


def flatten_gradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Gradient of ||conv(a,a) – avg||₂² where avg = 1/(2n‑1) and Σa = 1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient = 4 * (a ⊛ diff_rev)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)          # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def adam_flatten(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 10_000_000,
    lr0: float = 0.30,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the L₂ flattening loss.
    Returns the best vector (projected) and its fast surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        grad, loss = flatten_gradient(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 20000 == 0:
            print(
                f"    Adam‑flatten it={it:,d} loss={loss:.3e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth surrogate of max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value B) where
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv‑max)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max on convolution entries
    B = max_c + np.log(sum_exp) / lam

    # Weighted sub‑gradient of the convolution:
    # ∂/∂a Σ_k w_k (a*a)_k = 2 * (a ⊛ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_max(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 10_000_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 1500.0,
    noise_every: int = 1000,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution surrogate.
    Returns the best vector (projected) and its fast surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam‑max it={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f} lam={lam:.1f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max; back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.28,
    max_iter: int = 10_000_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector (projected) and its fast surrogate score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # simple back‑tracking line‑search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad it={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (cheap Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 10_000_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    Returns the best vector encountered and its fast surrogate score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"    Random walk it={it:,d} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine mixes extensive shape sampling, a flattening stage,
    a high‑temperature smoothed‑max Adam run, exact sub‑gradient descent,
    and a final stochastic walk.  All phases obey the global time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use a previously computed construction if present (fallback)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose problem size (we focus on n = 2048)
    # ------------------------------------------------------------------
    n = 2048
    if best_seq:
        n = len(best_seq)          # honour the length of a known good candidate

    # Helper to normalise a vector to sum = 1
    def normalise(v: np.ndarray) -> np.ndarray:
        s = np.sum(v)
        if s <= 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    # Starting point for the intensive search
    if best_seq:
        a_start = normalise(np.array(best_seq, dtype=float))
    else:
        a_start = arcsine_initial(n)

    # ------------------------------------------------------------------
    # 2.  Exploratory sampling (≈10 % of the budget)
    # ------------------------------------------------------------------
    sampling_deadline = start_time + min(120.0, 0.12 * (deadline - start_time))
    best_candidate = a_start.copy()
    best_candidate_sc = compute_score(best_candidate)
    print(f"Initial fast score = {best_candidate_sc:.6f}")

    # deterministic starters
    starters = [
        np.full(n, 1.0 / n),                     # uniform
        arcsine_initial(n),                      # classic arcsine
        generate_exponential_shape(n, 0.001),
        generate_exponential_shape(n, 0.01),
        generate_exponential_shape(n, 0.1),
    ]
    for alpha in np.linspace(0.15, 0.85, 13):
        starters.append(generate_beta_shape(n, alpha, alpha, eps=1e-6))

    for st in starters:
        sc = compute_score(st)
        if sc < best_candidate_sc:
            best_candidate_sc = sc
            best_candidate = st.copy()

    # random beta / mixture sampling
    while time.time() < sampling_deadline:
        if rng.random() < 0.35:   # mixture branch
            w = rng.random()
            cand = generate_beta_mixture(
                n,
                rng.uniform(0.02, 3.5),
                rng.uniform(0.02, 3.5),
                rng.uniform(1e-8, max(0.01, 0.02 * n)),
                rng.uniform(0.02, 3.5),
                rng.uniform(0.02, 3.5),
                rng.uniform(1e-8, max(0.01, 0.02 * n)),
                w,
            )
        else:                     # single beta
            cand = generate_beta_shape(
                n,
                rng.uniform(0.02, 3.5),
                rng.uniform(0.02, 3.5),
                rng.uniform(1e-8, max(0.01, 0.02 * n)),
            )
        sc = compute_score(cand)
        if sc < best_candidate_sc:
            best_candidate_sc = sc
            best_candidate = cand.copy()
        # occasional progress report (every ~30 s)
        if int(time.time()) % 30 == 0:
            print(f"Sampling best fast score {best_candidate_sc:.6f}")

    print(f"Best fast score after sampling: {best_candidate_sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Flattening stage (L₂ loss) – ~25 % of remaining budget
    # ------------------------------------------------------------------
    flatten_deadline = time.time() + 0.25 * (deadline - time.time())
    print(f"Flattening stage until {flatten_deadline - start_time:.1f}s")
    a_flat, _ = adam_flatten(
        initial_a=best_candidate,
        time_deadline=flatten_deadline,
        max_iter=10_000_000,
        lr0=0.35,
    )

    # ------------------------------------------------------------------
    # 4.  Smooth‑max Adam stage – ~35 % of remaining budget
    # ------------------------------------------------------------------
    max_deadline = time.time() + 0.35 * (deadline - time.time())
    print(f"Smooth‑max Adam stage until {max_deadline - start_time:.1f}s")
    a_max, _ = adam_max(
        initial_a=a_flat,
        time_deadline=max_deadline,
        max_iter=10_000_000,
        lr0=0.12,
        lam_start=1.0,
        lam_end=1500.0,
        noise_every=1200,
        noise_scale=4e-5,
    )

    # ------------------------------------------------------------------
    # 5.  Exact sub‑gradient descent – ~20 % of remaining budget
    # ------------------------------------------------------------------
    sub_deadline = time.time() + 0.20 * (deadline - time.time())
    print(f"Sub‑gradient descent stage until {sub_deadline - start_time:.1f}s")
    a_sub, _ = subgradient_optimize(
        initial_a=a_max,
        time_deadline=sub_deadline,
        step0=0.28,
        max_iter=10_000_000,
    )

    # ------------------------------------------------------------------
    # 6.  Stochastic local walk – remaining time (≈10 % left)
    # ------------------------------------------------------------------
    walk_deadline = deadline - 2.0
    print(f"Random‑walk polishing until {walk_deadline - start_time:.1f}s")
    a_final, _ = random_local_search(
        initial_a=a_sub,
        time_deadline=walk_deadline,
        init_step=0.015,
        max_iter=10_000_000,
    )

    # ------------------------------------------------------------------
    # 7.  Official evaluation
    # ------------------------------------------------------------------
    final_seq = a_final.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"Final official score = {final_val:.6f}")

    if final_val < best_val:
        best_val = final_val
        best_seq = final_seq

    # ------------------------------------------------------------------
    # 8.  Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
