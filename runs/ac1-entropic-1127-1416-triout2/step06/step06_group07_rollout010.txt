"""
Propose a low‑score non‑negative step function.

Strategy
--------
1.  Several random Dirichlet restarts (α sampled widely) are run.
    Each restart is refined with a mirror‑descent style optimiser that
    minimises a soft‑max approximation of the max‑convolution.
    The temperature (β) is increased more aggressively (β_max=80) so that
    the optimiser focuses earlier on the true ℓ∞‑norm.
2.  The best vector found is fed to a simulated‑annealing hill‑climber.
    Moves consist of swapping two entries or transferring a fraction of
    mass from one coordinate to another; uphill moves are occasionally
    accepted according to a linear cooling schedule.  The step‑size is
    adapted on the fly.
3.  The final vector is returned (or a tiny uniform fallback if nothing
    was found).  All work respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real‑valued linear convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # the simplex is the uniform distribution in this degenerate case
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (when Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming a sums to 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Smooth optimisation (soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_optimize(
    a: np.ndarray,
    time_deadline: float,
    iters: int = 200_000,
    beta0: float = 0.5,
    beta_max: float = 80.0,
    eta0: float = 0.5,
    report_every: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth (log‑sum‑exp) approximation of the
    max‑convolution using an exponentiated‑gradient mirror descent.
    """
    n = a.shape[0]
    L = next_pow2(3 * n - 2)   # length sufficient for all FFTs used below

    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, iters + 1):
        if time.time() >= time_deadline - 0.5:
            break

        # ----- a*a ----------------------------------------------------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = np.max(conv)

        # ----- soft‑max weights ---------------------------------------
        beta = min(beta_max, beta0 * np.sqrt(t))
        # stabilise exponentials
        exp_shift = beta * (conv - max_c)
        np.clip(exp_shift, -700, 700, out=exp_shift)   # avoid overflow
        exp_terms = np.exp(exp_shift)
        w = exp_terms / np.sum(exp_terms)

        # ----- gradient = 2·(w ⋆ a_rev) -------------------------------
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        # slice that aligns with a
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- exponentiated (mirror) update -------------------------
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        # ----- occasional reporting ------------------------------------
        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  iter {t:8d} – new best score {cur_val:.6f}")

    # final check
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber
# ----------------------------------------------------------------------


def annealing_optimize(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Metropolis‑style annealing that moves probability mass between
    coordinates.  The temperature decays linearly from an initial value
    proportional to the starting score.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = fast_score(cur)

    best = cur.copy()
    best_score = cur_score

    # temperature schedule
    T0 = max(1e-6, cur_score * 0.2)  # start relatively warm
    final_T = 1e-12

    step_scale = 0.2  # initial fraction of mass moved in a transfer
    step = 0

    while time.time() < deadline and step < max_steps:
        # linear temperature decay
        T = max(final_T, T0 * (1.0 - step / max_steps))

        cand = cur.copy()
        if np.random.rand() < 0.5:
            # mass transfer between two positions
            i, j = np.random.randint(n, size=2)
            if i != j and cur[i] > 0:
                delta = step_scale * cur[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)
        else:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        # maintain feasibility
        cand = project_to_simplex(cand)
        cand_score = fast_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / T):
            cur = cand
            cur_score = cand_score
            # record best
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                # succeed → enlarge step_scale a little
                step_scale = min(step_scale * 1.02, 0.5)
            else:
                # failed move → shrink step_scale
                step_scale *= 0.99
        step += 1

    return best, best_score

    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Try a provided fallback (if any)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val = fb_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Choose a length (2048 works well)
    # --------------------------------------------------------------
    n = 2048
    if time.time() + 5 > deadline:
        n = 1024
    print(f"Using sequence length n = {n}")

    # --------------------------------------------------------------
    # 2️⃣  Random restarts + smooth optimisation
    # --------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    smooth_budget = remaining * 0.70          # ~70 % of what is left
    num_restarts = 6
    per_restart_budget = smooth_budget / max(1, num_restarts)

    for restart in range(num_restarts):
        if time.time() >= deadline - 10.0:
            break

        # random Dirichlet – concentration α encourages diversity
        alpha = np.random.uniform(0.1, 1.5)
        a0 = np.random.gamma(alpha, 1.0, size=n)
        a0 = a0 / a0.sum()

        init_score = fast_score(a0)
        if init_score < best_val:
            best_val = init_score
            best_seq = a0.tolist()
            print(f"Restart {restart}: initial score = {init_score:.6f}")

        # smooth optimisation within its time slice
        restart_deadline = min(deadline - 5.0, time.time() + per_restart_budget)
        refined_a, refined_score = smooth_optimize(
            a0,
            time_deadline=restart_deadline,
            iters=250_000,
            beta0=0.5,
            beta_max=80.0,
            eta0=0.5,
            report_every=30_000,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Restart {restart}: refined best score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final annealing polish (if we still have time)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nStarting annealing polishing")
        a_start = np.array(best_seq, dtype=float)
        polished_seq, polished_val = annealing_optimize(
            a_start,
            deadline=deadline - 1.0,
            max_steps=3_000_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Annealing improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Safeguard – tiny uniform vector if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 5️⃣  Report final score and return
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
