"""
Propose a non‑negative step sequence with a low official score.

The search consists of four phases

1. **Random sampling** – draw many beta‑family (including mixtures) and a few
   deterministic starters for several lengths.  The best few candidates of
   each length are kept.

2. **Local optimisation** – each kept candidate is refined by a long Adam
   run on a smooth log‑sum‑exp surrogate of the max‑convolution, followed
   by a continuation with L‑BFGS‑B (gradients are obtained analytically).
   Afterwards an exact sub‑gradient descent and a cheap Gaussian walk are
   applied.

3. **Global polishing** – the best sequence overall is subjected to a very
   long Adam run, a final L‑BFGS continuation with even larger temperature,
   a last sub‑gradient phase and a final random walk.

4. **Safety fallback** – if nothing is found a uniform 256‑vector is
   returned.

All stages respect the global 1000 s time budget and keep track of the
best sequence seen (according to the official ``evaluate_sequence``).
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for all n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator is 1 and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (beta families, mixtures, arcsine)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α-1}·(n-1-i+ε)^{β-1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density (the shape used in the original baseline)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (required for Adam/L‑BFGS)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the smooth log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, smooth_max) where grad is the gradient w.r.t. a of the
    official objective ``2·n·smooth_max``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                       # soft‑max weights
    B = max_c + np.log(sum_exp) / lam            # smooth approximation of the max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)                                # normally 1 after projection
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Adam optimiser (used for the bulk of the work)
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 400.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector seen (projected on the simplex) together with
    its fast surrogate score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (guarantees non‑increase of the true objective)
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees that the official objective never increases.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap Gaussian walk (stochastic local search)
# ----------------------------------------------------------------------


def random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Simple Gaussian walk on the simplex, accepting occasional uphill moves.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = fast_score(a_prop)

        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Helper for L‑BFGS (softmax re‑parameterisation)
# ----------------------------------------------------------------------


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (guarantees Σa = 1, a_i ≥ 0)."""
    e = np.exp(x - np.max(x))
    return e / e.sum()


def fun_and_grad(x: np.ndarray, lam: float) -> Tuple[float, np.ndarray]:
    """
    Objective and gradient for L‑BFGS.
    The optimisation variable is the log‑it vector `x`; the actual coefficients
    are obtained by a softmax, guaranteeing feasibility.
    The surrogate objective is 2·n·smooth_max(conv(a,a)).
    """
    a = softmax(x)
    grad_a, B = smooth_gradient(a, lam)          # grad_a is ∂/∂a of 2·n·B
    n = a.shape[0]
    obj = 2.0 * n * B

    # chain rule: ∂obj/∂x = a ∘ (grad_a - <grad_a,a>)
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    return obj, grad_x


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.

    The routine performs:
      * random sampling of beta‑family shapes for several lengths,
      * short Adam polishing of the best samples,
      * a continuation with L‑BFGS‑B on a smooth surrogate,
      * exact sub‑gradient descent and a final Gaussian walk,
      * a global polishing of the overall best vector.
    All steps respect the global 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate with the official evaluator.
    # ------------------------------------------------------------------
    def official_score(vec: np.ndarray) -> float:
        return evaluate_sequence(vec.tolist())

    # ------------------------------------------------------------------
    # 0.  Use a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = official_score(np.array(fallback))
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Exploratory sampling for several vector lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Sampling length n = {n} ===")

        # --------------------------------------------------------------
        # Deterministic starters
        # --------------------------------------------------------------
        starters = [
            np.full(n, 1.0 / n),   # uniform
            arcsine_shape(n),      # classic arcsine
        ]

        # --------------------------------------------------------------
        # Random beta‑family shapes
        # --------------------------------------------------------------
        candidates = starters.copy()
        max_samples = int(5_000 * (budget_s / 1000.0))
        for _ in range(max_samples):
            if time.time() >= deadline - 30.0:
                break

            if rng.random() < 0.4:   # mixture of two mirrored betas
                w = rng.random()
                alpha1 = rng.uniform(0.02, 1.0)
                beta1 = rng.uniform(0.5, 2.0)
                eps1 = rng.uniform(1e-9, 0.02 * n)
                # mirrored second component
                alpha2 = beta1
                beta2 = alpha1
                eps2 = eps1
                cand = generate_beta_mixture(
                    n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
                )
            else:                    # plain beta
                alpha = rng.uniform(0.02, 1.5)
                beta = rng.uniform(0.02, 1.5)
                eps = rng.uniform(1e-9, 0.02 * n)
                cand = generate_beta_shape(n, alpha, beta, eps)

            candidates.append(cand)

        # --------------------------------------------------------------
        # Keep the few best according to the fast surrogate
        # --------------------------------------------------------------
        scored = [(fast_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_local = [c for _, c in scored[:5]]

        # --------------------------------------------------------------
        # Polishing each of the local survivors
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_local):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Polishing survivor {idx + 1}/{len(top_local)} (n={n}) …")

            # ---- Adam phase -------------------------------------------------
            remaining = deadline - time.time()
            max_iter_adam = int(min(500_000, remaining * 35))
            cand_opt, _ = adam_optimize(
                init_a=cand,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.11,
                lam_start=1.0,
                lam_end=450.0,
                noise_every=1200,
                noise_scale=3e-5,
            )

            # ---- L‑BFGS continuation (log‑softmax parametrisation) -------
            try:
                from scipy.optimize import minimize
                # initialise logits
                x = np.log(cand_opt + 1e-12)
                for lam in [150, 350, 600, 900]:
                    if time.time() >= deadline - 2.0:
                        break
                    res = minimize(
                        fun=lambda xx: fun_and_grad(xx, lam),
                        x0=x,
                        jac=True,
                        method="L-BFGS-B",
                        options={"maxiter": 1800, "disp": False},
                    )
                    x = res.x
                cand_opt = softmax(x)
            except Exception:
                # SciPy not available – simply keep the Adam result
                pass

            # ---- Exact sub‑gradient descent ---------------------------------
            remaining = deadline - time.time()
            max_iter_sub = int(min(200_000, remaining * 25))
            cand_opt, _ = subgradient_optimize(
                init_a=cand_opt,
                deadline=deadline,
                step0=0.24,
                max_iter=max_iter_sub,
            )

            # ---- Final cheap random walk ------------------------------------
            remaining = deadline - time.time()
            max_iter_rw = int(min(200_000, remaining * 20))
            cand_opt, _ = random_local_search(
                init_a=cand_opt,
                deadline=deadline,
                init_step=0.013,
                max_iter=max_iter_rw,
            )

            # ---- Official evaluation -----------------------------------------
            val = official_score(cand_opt)
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = cand_opt.tolist()
                print(f"    *** New global best = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3. Global polishing of the current best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # ---- Very long Adam run (high temperature) -----------------------
        remaining = deadline - time.time()
        max_iter_adam = int(min(800_000, remaining * 40))
        a, _ = adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.09,
            lam_start=1.0,
            lam_end=800.0,
            noise_every=1300,
            noise_scale=2e-5,
        )

        # ---- L‑BFGS continuation with even larger λ -----------------------
        try:
            from scipy.optimize import minimize
            x = np.log(a + 1e-12)
            for lam in [300, 600, 1000]:
                if time.time() >= deadline - 2.0:
                    break
                res = minimize(
                    fun=lambda xx: fun_and_grad(xx, lam),
                    x0=x,
                    jac=True,
                    method="L-BFGS-B",
                    options={"maxiter": 2500, "disp": False},
                )
                x = res.x
            a = softmax(x)
        except Exception:
            pass

        # ---- Sub‑gradient polishing ---------------------------------------
        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 30))
        a, _ = subgradient_optimize(
            init_a=a,
            deadline=deadline,
            step0=0.22,
            max_iter=max_iter_sub,
        )

        # ---- Final random walk --------------------------------------------
        remaining = deadline - time.time()
        max_iter_rw = int(min(300_000, remaining * 45))
        a, _ = random_local_search(
            init_a=a,
            deadline=deadline,
            init_step=0.010,
            max_iter=max_iter_rw,
        )

        final_val = official_score(a)
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        print("No candidate produced – falling back to uniform vector.")
        best_seq = [1.0] * 256
        best_val = official_score(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
