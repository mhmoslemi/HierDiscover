```
def swap_local_search(a, deadline, max_iter=100000):
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = fast_score_vec(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while it < max_iter and time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
        # re-evaluate
        new_score = fast_score_vec(cur_a)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            # reject swap, revert
            cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
        it += 1
    return best_a
```

```
def subgradient_optimize(init_a, deadline, step0=0.3, max_iter=200000):
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score_vec(a)
    n = a.shape[0]
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))
        # gradient for active max term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # step size
        step = step0 / np.sqrt(it)
        # backtrack
        for _ in range(12):
            a_new = a - step * grad
            # project onto simplex
            a_new = project_to_simplex(a_new)
            # compute new max conv
            new_max = np.max(convolution_fft(a_new, a_new))
            if new_max < max_val - 1e-15:
                break
            step *= 0.5
        cur_sc = fast_score_vec(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_sc
```

The evaluation function  

\[
\text{score}(a)=\frac{2\,n\;\max\big((a*a)\big)}{\big(\sum a\big)^2}
\]

is *scale‑invariant*: multiplying the whole vector by a constant does not change the
score.  Consequently we may always work with vectors whose sum equals 1.
For a vector of length `n` the formal lower bound of the objective is  

\[
\frac{2n}{2n-1}\approx 1 ,
\]

obtained when the self‑convolution is perfectly flat.  Uniform vectors give a
score of 2, so the task is to lower the maximal entry of the convolution.

The programme below follows three ideas that together improve the previous
state‑of‑the‑art code:

1. **Explicit Beta‑family search** –  
   The family  
   \[
   a_i\propto (i+\varepsilon)^{\alpha-1}\,(n-1-i+\varepsilon)^{\beta-1},
   \qquad \varepsilon>0,
   \]
   contains the classic arcsine shape (α=β=½) and many more “U‑shaped’’
   distributions that push mass to the ends of the interval.  
   A modest grid over `α` and `β` (≈ 900 points) is evaluated with a fast
   FFT based convolution.  The best grid point is kept.

2. **Projected sub‑gradient descent on the exact max‑convolution** –  
   Given the current index `k` that attains the maximal convolution value,
   the sub‑gradient of the objective is  

   \[
   g_i = 
   \begin{cases}
        2\,a_{k-i} &\text{if }0\le k-i<n,\\[4pt]
        0           &\text{otherwise.}
   \end{cases}
   \]

   A decreasing step size `step = step0/√t` together with a back‑tracking line
   search guarantees that the true objective never increases.
   The projection onto the probability simplex keeps the vector non‑negative
   and normalised.

3. **Pure combinatorial hill‑climbing (random swaps)** –  
   After the continuous optimisation the algorithm performs a very cheap
   stochastic walk that only swaps two entries.  Because the sum is preserved,
   a swap is cheap to evaluate (`O(n log n)`) and can repeatedly flatten the
   convolution.  Only improving swaps are accepted, which still gives a solid
   local optimum.

All stages respect the 1000 s budget, keep the best vector seen so far, and
fall back to a uniform 256‑vector if nothing else is found.

The code is completely self‑contained; the only external function it uses is
`evaluate_sequence`, which is provided by the competition harness.

```python
"""
search for a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases

1.  exhaustive grid search over the Beta‑family a_i ∝ (i+ε)^{α‑1}(n‑1‑i+ε)^{β‑1}
    (α,β in a fine grid).  The best candidate is kept.

2.  projected sub‑gradient descent on the exact max‑convolution.
    The sub‑gradient for the active index k is
        g_i = 2·a_{k‑i}  (if 0≤k‑i<n, else 0).
    A decreasing stepsize and a back‑tracking line‑search guarantee that the
    true (official) objective never increases.

3.  a cheap stochastic walk that repeatedly swaps two entries of the vector.
    Swaps that lower the FFT‑computed max‑convolution are kept.
    The walk further flattens the convolution.

All work is done on vectors whose entries sum to one, because the objective
is scale‑invariant.  The final vector is evaluated with the official
`evaluate_sequence` function and returned.
"""

import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Linear algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for all n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (sum of a is assumed 1)."""
    return float(np.max(convolution_fft(a, a)))


def fast_score_vec(a: np.ndarray) -> float:
    """Fast surrogate of the official score (sum a = 1)."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x≥0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generation
# ----------------------------------------------------------------------


def beta_shape_log(n: int, alpha: float, beta: float, eps: float = 0.5) -> np.ndarray:
    """
    Beta‑family density a_i ∝ (i+eps)^{α‑1}·(n‑1‑i+eps)^{β‑1}.
    Computed in log‑space for numerical stability.
    The result is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    # avoid log(0) by eps>0
    log_a = (alpha - 1.0) * np.log(i + eps) + (beta - 1.0) * np.log(
        n - 1.0 - i + eps
    )
    # shift to avoid overflow/underflow
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    total = a.sum()
    if total == 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


# ----------------------------------------------------------------------
# Exact sub‑gradient descent
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 200_000,
) -> (np.ndarray, float):
    """
    Projected sub‑gradient descent on the true objective.
    Guarantees that the official score never increases.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score_vec(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # decreasing stepsize
        step = step0 / np.sqrt(it)

        # simple back‑tracking line search (max 12 attempts)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_max = np.max(convolution_fft(a_new, a_new))
            if new_max < max_val - 1e-15:
                break
            step *= 0.5

        cur_sc = fast_score_vec(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Random swap local search (cheap hill‑climbing)
# ----------------------------------------------------------------------


def swap_local_search(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
) -> np.ndarray:
    """
    Stochastic walk that repeatedly swaps two entries.
    Only swaps that strictly improve the max‑convolution are kept.
    """
    a = init_a.copy()
    cur_sc = fast_score_vec(a)
    best_a = a.copy()
    best_sc = cur_sc
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0

    while it < max_iter and time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # perform swap
        a[i], a[j] = a[j], a[i]
        new_sc = fast_score_vec(a)

        if new_sc < cur_sc:
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(

    # --------------------------------------------------------------

    def official_score(vec: np.ndarray) -> float:
        return evaluate_sequence(vec.tolist())

    n = 8192
    if deadline - time.time() < 200.0:   # not enough time → fall back to 2048

    print(f"\n=== Starting search, vector length = {n} ===")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    alphas = np.linspace(0.05, 2.0, 30)   # 30 values → 900 grid points

    # --------------------------------------------------------------
    if best_local is not None and time.time() < deadline - 5.0:

    # 6. Guard – guarantee a non‑empty return.
    # --------------------------------------------------------------
    if not best_seq:

        best_val = official_score(fallback)
        best_seq = fallback.tolist()
        print("Fallback to uniform 256‑vector, score =", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
```
