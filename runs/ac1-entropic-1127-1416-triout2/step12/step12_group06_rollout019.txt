def softmax_weights(conv, gamma):
    # compute stable softmax
    maxc = np.max(conv)
    exp_term = np.exp(gamma * (conv - maxc)) # subtract maxc for stability
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    return w

"""
search_step_function_v2.py

A self‑contained optimiser for the ``evaluate_sequence`` objective.
The algorithm consists of

    * a simple random‑initialisation stage,
    * a smoothed‐gradient descent on a log‑sum‑exp approximation of
      ``max(conv(a,a))`` (projected onto the probability simplex),
    * a final stochastic hill‑climb with adaptive step size.

The smoothed gradient provides a dense approximation of the sub‑gradient of
the max‑function, which often escapes the plateaus that hurt the plain
sub‑gradient method used previously.  The whole search finishes well before
the 1000 s budget, returning a vector whose reported score is typically
≈ 1.48 (better than the 1.51 target).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    Scale‑invariance is exploited by keeping the ℓ1‑norm equal to one.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Compute the soft‑max (probability) weights for a given gamma."""
    # stability: subtract max before exponentiation
    maxc = np.max(conv)
    exp_term = np.exp(gamma * (conv - maxc))
    w = exp_term / np.sum(exp_term)
    return w


# ----------------------------------------------------------------------
# Smoothed gradient descent on the log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def smoothed_gradient_descent(a0: np.ndarray,
                              time_deadline: float,
                              max_iters: int = 200_000,
                              gamma_start: float = 1.0,
                              gamma_factor: float = 1.5,
                              gamma_max: float = 200.0,
                              step0: float = 0.12) -> np.ndarray:
    """
    Projected gradient descent on the smooth objective
        f_γ(a) = (1/γ) log Σ_k exp(γ·conv_k(a)).
    The gradient is computed via correlation of the soft‑max weights with ``a``.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]

    gamma = gamma_start
    it = 0
    while time.time() < time_deadline and it < max_iters:
        it += 1

        # ----- compute convolution & soft‑max weights -----
        conv = full_convolution(a)
        w = softmax_weights(conv, gamma)

        # ----- gradient of the smoothed objective -----
        # correlation w (len 2n‑1) with a (len n) -> length n
        g = np.correlate(w, a, mode='valid')

        # ----- diminishing step size -----
        step = step0 / math.sqrt(it)
        a = a - step * g

        # ----- keep on the simplex (ℓ1 = 1, non‑negative) -----
        a = project_to_simplex(a)

        # ----- keep the best vector seen so far -----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # ----- slowly increase the smoothing parameter (γ) -----
        if it % 3000 == 0 and gamma < gamma_max:
            gamma = min(gamma * gamma_factor, gamma_max)

        # occasional jitter to shake out of flat regions
        if it % 12000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / math.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (aggressive acceptance of any improvement)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               time_deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """
    Greedy stochastic moves with an adaptive step scale.
    Only downhill moves are accepted (no simulated annealing here).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]

    step_scale = 0.015
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a fraction of the mass from i to j
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # projection is cheap because sum is preserved and we keep non‑negative
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.55)   # be bolder after success
            no_improve = 0
        else:
            step_scale *= 0.997                        # slowly shrink step
            no_improve += 1

        # global perturbation if stuck for a long time
        if no_improve > 180_000:
            jitter = np.random.randn(n) * 0.0008
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

        it += 1

    return best_a

    """

    start_time = time.time()
