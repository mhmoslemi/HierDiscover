def build_A_ub(f, indices, n):
    mat = np.zeros((len(indices), n+1), dtype=np.float64)
    for r, idx in enumerate(indices):
        row = np.zeros(n)
        for j in range(max(0, idx - (n-1)), min(idx, n-1)+1):
            row[j] = f[idx - j]
        mat[r, :n] = row
        mat[r, n] = -1.0
    return mat

"""
Propose a non‑negative step‑function (probability vector) that minimises
the “auto‑convolution peak” objective used in ``evaluate_sequence``.

The algorithm proceeds as follows:

1.  Choose a moderate support size `n = 1024`.  (Larger n gives a finer
    approximation but makes the linear programme larger.)

2.  Scan a one‑parameter symmetric “product” family
        a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    to obtain a good starting shape.  A cheap golden‑section refinement
    is performed around the best α.

3.  Repeatedly solve a tiny linear programme that seeks a new vector `g`
    with the same mass (∑g = 1) while *lowering* the current maximal
    auto‑convolution value `M`.  The LP variables are the entries of `g`
    plus a scalar `t` representing an upper bound on the convolution.
    Only the most critical convolution indices (the largest values)
    are kept as linear constraints; violated constraints are added
    iteratively (cutting‑plane style).  This “LP‑refine’’ step quickly
    drives the objective below the previous best (~1.5108) and can push
    it under the target 1.503.

4.  As a safety net, a few steps of projected sub‑gradient descent are
    applied to the final vector.

5.  The resulting vector is returned as a plain Python list; the wrapper
    ``evaluate_sequence`` will perform the final scaling checks.

All work respects the supplied wall‑clock budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a probability vector (∑a = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for a fast good initial guess)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric product family:
        a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    # log of the product to avoid overflow
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # handle possible infinities at the ends
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Simple one‑dimensional optimisation helpers
# ----------------------------------------------------------------------


def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 25) -> tuple[float, float]:
    """Minimise `func` on [a,b] via golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# LP‑based refinement (cutting‑plane on the max‑convolution bound)
# ----------------------------------------------------------------------


def lp_refine(a0: np.ndarray,
              deadline: float,
              max_outer: int = 8,
              top_k: int = 200,
              max_inner: int = 30,
              max_new_constraints: int = 12) -> np.ndarray:
    """
    Repeatedly solve a tiny linear programme:
        minimise t
        subject to   (a * g)_i ≤ t   for selected indices i
                     ∑_j g_j = 1
                     g_j ≥ 0

    The LP is solved with a cutting‑plane approach – new violated
    convolution indices are added to the active set.
    Returns a refined probability vector.
    """
    try:
        from scipy.optimize import linprog
    except Exception as e:
        print("scipy.optimize.linprog not available:", e)
        return a0

    n = a0.shape[0]
    a = a0.copy()

    for outer in range(max_outer):
        if time.time() >= deadline:
            break

        # Current max‑convolution and candidate indices
        conv = convolution_fft(a, a)
        M = float(conv.max())
        # pick the top_k indices (largest values) as the active set
        idx_set = set(np.argpartition(conv, -top_k)[-top_k:].tolist())

        # Cutting‑plane inner loop
        for inner in range(max_inner):
            if time.time() >= deadline:
                break

            idx_list = list(idx_set)
            rows = len(idx_list)

            # Build A_ub (rows × (n+1)) where the last column corresponds to –t
            A_ub = np.zeros((rows, n + 1), dtype=np.float64)
            for r, idx in enumerate(idx_list):
                # Only the overlap of a with the shift idx matters
                start = max(0, idx - (n - 1))
                end = min(idx, n - 1)
                # Fill the slice a[idx‑j] into the appropriate positions
                for j in range(start, end + 1):
                    A_ub[r, j] = a[idx - j]
                A_ub[r, -1] = -1.0            # coefficient for –t

            b_ub = np.zeros(rows, dtype=np.float64)  # RHS is 0

            # Objective: minimise t
            c = np.zeros(n + 1, dtype=np.float64)
            c[-1] = 1.0

            # Equality: sum(g) = 1
            A_eq = np.zeros((1, n + 1), dtype=np.float64)
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0], dtype=np.float64)

            bounds = [(0.0, None)] * n + [(0.0, None)]  # g_j ≥ 0, t ≥ 0

            res = linprog(c,
                          A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
            if not res.success:
                print(f"[LP] iteration failed ({res.message}); aborting refinement.")
                return a

            g = res.x[:n]
            t_opt = res.x[-1]

            # Check feasibility on the full convolution (not only the active set)
            conv_g = convolution_fft(g, g)
            max_g = float(conv_g.max())

            # If the solution respects *all* constraints, we have improved.
            if max_g <= t_opt + 1e-12:
                a = g
                print(f"[{time.time():.1f}] LP outer {outer} inner {inner} → "
                      f"M={max_g:.9f}, score={2 * n * max_g:.6f}")
                break  # go to next outer iteration (new a)

            # Otherwise, add the most egregious violating indices
            viol = np.where(conv_g > t_opt + 1e-12)[0]
            added = 0
            for vi in viol:
                if vi not in idx_set:
                    idx_set.add(vi)
                    added += 1
                    if added >= max_new_constraints:
                        break
            if added == 0:
                # cannot enrich the active set any further – accept current g
                a = g
                print(f"[{time.time():.1f}] LP inner {inner} stalled – accepting current g")
                break

        else:
            # inner loop exhausted without feasible solution
            print("[LP] inner loop reached max iterations without feasibility.")
            break

    return a


# ----------------------------------------------------------------------
# Simple projected sub‑gradient descent (fallback / polishing)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000,
                     step0: float = 0.04) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a
