"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
1.  Global exploration (≈200 s):
    * dense sweeps of several parametric families
      (power‑law, exponential, Beta, “inverse‑sqrt” and sine‑based shapes)
    * many Dirichlet draws with a wide range of concentration parameters.
    The K best vectors (by the exact objective 2·n·max(conv(a,a))) are kept.

2.  Exponentiated‑gradient polishing (≈600 s):
    * For each retained vector we run an EG optimiser that minimises
      max(conv(a,a)) on the probability simplex.
    * A large number of random restarts are performed while time permits.

3.  Final hill‑climbing (≈150 s):
    * Small mass‑transfer moves between random indices are accepted
      if they improve the objective (simple Metropolis‑style search).
    * Step size adapts automatically.

All three phases respect the global 1000 s wall‑clock budget.  The best
sequence found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for speed
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(n, 0.0)
    return a / s


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(n, 0.0)
    return a / s


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    Works for α,β > 0 (including values < 1 which give U‑shapes).
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(n, 0.0)
    return a / s


def generate_inverse_sqrt(n: int, gamma: float = 1.0) -> np.ndarray:
    """
    Symmetric “inverse‑sqrt” shape:
        a_i ∝ [(i+0.5)*(n‑i‑0.5)]^{-γ/2}.
    For γ>0 this is U‑shaped, concentrating mass near the ends.
    """
    i = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(i * (n - i))
    a = denom ** (-gamma)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(n, 0.0)
    return a / s


def generate_sin_power(n: int, power: float = -1.0) -> np.ndarray:
    """
    Sinusoidal family:
        a_i ∝ sin(π·(i+0.5)/n)^power.
    Negative powers emphasise the ends, positive powers the centre.
    """
    theta = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    base = np.sin(theta)
    # avoid zero division for large negative powers
    base = np.where(base == 0, 1e-12, base)
    a = base ** power
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(n, 0.0)
    return a / s


# ----------------------------------------------------------------------
# Baseline global search
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 12):
    """
    Sweeps several parametric families and draws many random Dirichlet samples.
    Returns up to ``max_keep`` best vectors (as numpy arrays) according to
    the exact objective.
    """
    best = []   # list of (score, vector)

    def consider(vec: np.ndarray):
        """Project to simplex, evaluate and possibly store."""
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- Power‑law ----------------------------------------------------
    for p in np.linspace(0.0, 3.0, 300):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # ---- Exponential --------------------------------------------------
    for lam in np.linspace(1e-6, 0.2, 300):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # ---- Beta (U‑shaped especially) ----------------------------------
    al_vals = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    be_vals = al_vals
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, alpha, beta))

    # ---- Inverse‑sqrt family ------------------------------------------
    for gamma in np.linspace(0.2, 3.0, 150):
        if time.time() >= deadline:
            break
        consider(generate_inverse_sqrt(n, gamma))

    # ---- Sin‑power family ---------------------------------------------
    for pw in np.linspace(-2.0, 2.0, 200):
        if time.time() >= deadline:
            break
        consider(generate_sin_power(n, pw))

    # ---- Random Dirichlet draws ---------------------------------------
    conc_levels = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per = 2000   # 9 × 2000 ≈ 18 000 draws – fast on modern hardware
    for conc in conc_levels:
        alpha = np.full(n, conc)
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(alpha)
            consider(vec)

    best.sort(key=lambda x: x[0])
    return [v for sc, v in best]


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing
# ----------------------------------------------------------------------


def eg_polish(a0: np.ndarray, deadline: float,
              eta0: float = 0.9, tol: float = 1e-12,
              max_iter: int = 10_000_000) -> np.ndarray:
    """
    EG minimisation of  max(conv(a,a)) on the simplex.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # active set (indices within a tiny relative tolerance of the maximum)
        eps = max_val * tol
        active = np.where(conv >= max_val - eps)[0]

        # gradient: g_i = 2 * sum_{k in active} a_{k-i}
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # decreasing step size
        eta = eta0 / np.sqrt(it)

        # EG multiplicative update
        a = a * np.exp(-eta * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional tiny jitter to escape flat regions
        if it % 800 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a


# ----------------------------------------------------------------------
# Simple hill‑climbing / local refinement
# ----------------------------------------------------------------------


def hill_climb(a0: np.ndarray, deadline: float,
               init_step: float = 0.02,
               min_step: float = 1e-6) -> np.ndarray:
    """
    Random pairwise mass‑transfer hill‑climb.
    A move transfers a fraction of mass from i to j; the move is kept
    if it improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    step = init_step

    while time.time() < deadline:
        # pick two distinct indices, i must have some mass
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step * a[i]
        if delta <= 0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # protect against negative rounding errors
        if cand[i] < 0.0:
            cand[i] = 0.0
        cand /= cand.sum()   # renormalise

        val = compute_score(cand)

        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            # success → slowly increase step (up to a safe ceiling)
            step = min(step * 1.03, 0.2)
        else:
            # failure → shrink step
            step = max(step * 0.97, min_step)

    return best_a


# ----------------------------------------------------------------------
# Main entry point for the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 20231128, budget_s: int = 1000, **_):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin for the harness

    # -------------------------------------------------
    # Problem size – 4096 gives a finer discretisation
    # -------------------------------------------------
    n = 4096

    best_seq = None
    best_val = float("inf")

    # -------------------------------------------------
    # Optional fallback from a previously known solution
    # -------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -------------------------------------------------
    # 1. Global baseline exploration
    # -------------------------------------------------
    baseline_deadline = deadline - 650.0   # leave ~650 s for polishing & hill‑climb
    if time.time() < baseline_deadline:
        candidates = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -------------------------------------------------
    # 2. Exponentiated‑gradient polishing (many restarts)
    # -------------------------------------------------
    polish_deadline = deadline - 200.0
    if best_seq is not None and time.time() < polish_deadline:
        # polish the current best
        a0 = np.array(best_seq, dtype=float)
        refined = eg_polish(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[eg‑polish‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # additional random restarts
        while time.time() < polish_deadline:
            # initialise from a fresh, possibly skewed Dirichlet draw
            conc = np.random.choice([0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = eg_polish(a_start, polish_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[eg‑polish‑restart] new best = {val:.6f}")

    # -------------------------------------------------
    # 3. Final hill‑climbing refinement
    # -------------------------------------------------
    hill_deadline = deadline - 10.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = hill_climb(a_start, hill_deadline)
        val = compute_score(final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -------------------------------------------------
    # 4. Fallback to a uniform distribution if nothing succeeded
    # -------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # 5. Final evaluation with the official function
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
