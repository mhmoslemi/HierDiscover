def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

       2·n·max(conv(a,a)).

def smooth_max_optim(a0: np.ndarray, deadline: float,
                     tau_start: float = 0.30,
                     tau_end: float = 1e-5,
                     tau_decay: float = 0.98,
                     step0: float = 0.20) -> np.ndarray:
    """Minimise a smooth approximation of max(conv) using an
    FFT‑based gradient and Euclidean projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    while time.time() < deadline:
        # anneal temperature for the soft‑max
        if t % 100 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # convolution a*a
        conv = full_convolution(a)            # length m
        # soft‑max weights
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()         # length m

        # gradient: 2 * (a ⋆ w)   (cross‑correlation)
        # we compute via FFT: g = 2 * conv_full[(m‑1)‑i]
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 1 - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]

        # extract the gradient for each coordinate
        idx = (m - 1) - np.arange(n)          # indices in conv_full
        grad = 2.0 * conv_full[idx]

        # step size (decreasing)
        step = step0 / np.sqrt(t + 1.0)
        a = a - step * grad
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 1500 == 0 and t > 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t + 1.0))
            a = a + jitter
            a = project_to_simplex(a)

        t += 1

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a
