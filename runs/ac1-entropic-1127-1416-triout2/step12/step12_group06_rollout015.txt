def exp_gradient_descent(a0, time_deadline, step0=0.5, tol=1e-12):
    a = a0.copy()
    best = a.copy()
    best_M = full_convolution(a).max()
    it = 0
    while time.time() < time_deadline:
        it += 1
        conv = full_convolution(a)
        M = conv.max()
        # find active set
        active = np.where(conv >= M - 1e-12)[0]  # allow tolerance
        # compute subgradient
        g = np.zeros_like(a)
        n = a.shape[0]
        for k in active:
            # g_i += 2 * a_{k-i}
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # exponential update
        eta = step0 / math.sqrt(it)
        # avoid overflow if eta*g large: compute factor = np.exp(-eta*g)
        # apply update: a_i = a_i * exp(-eta * g_i)
        multiplicative = np.exp(-eta * g)
        a = a * multiplicative
        # renormalize to sum 1
        sum_a = a.sum()
        if sum_a == 0:
            # reinitialize to uniform
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a /= sum_a
        # evaluate
        conv_new = full_convolution(a)
        M_new = conv_new.max()
        if M_new < best_M:
            best_M = M_new
            best = a.copy()
    return best

def mirror_adam(a0, time_deadline, eta0=0.5, beta1=0.9, beta2=0.999):
    a = a0.copy()
    m = np.zeros_like(a) # first moment
    v = np.zeros_like(a) # second moment
    t = 0
    best_a = a.copy()
    best_M = full_convolution(a).max()
    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]
        # gradient as before
        g = np.zeros_like(a)
        n = a.shape[0]
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        # Update moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # Compute step size
        step = eta0 / (np.sqrt(v_hat) + 1e-8)
        # Mirror descent update using exponentials:
        a = a * np.exp(-step * m_hat)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum
        # check best
        conv_new = full_convolution(a)
        M_new = conv_new.max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    return best_a
```

"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence.  The approach is:

1.  A broad parametric sweep over several families (power‑law,
    geometric, beta‑shape, mixed plateau+power) to obtain a decent
    starting vector.
2.  A mirror‑descent optimisation on the simplex (log‑parameterised
    a = softmax(z)).  The update uses the exact sub‑gradient of the
    current maximum convolution entry and an Adam‑like momentum
    schedule; this converges much faster than the simple Euclidean
    sub‑gradient used previously.
3.  A bias‑aware stochastic hill‑climb that moves mass from high‑value
    coordinates to low‑value ones.  The move size is adaptive.

All phases respect the 1000 s budget and keep track of the best
solution seen (according to compute_score, i.e. the value of the
objective when the vector is normalised to sum‑1).  The final
candidate is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def power_law(n: int, p: float, delta: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+delta)^(-p).  delta≥0 shifts the singularity away from i=0."""
    i = np.arange(1, n + 1, dtype=np.float64) + delta
    a = i ** (-p)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q**i  with 0<q<1."""
    if not (0.0 < q < 1.0):
        return None
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Plateau of length L then a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i**alpha * (1-x_i)**beta   (x_i = (i+0.5)/n)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# 1. Broad parametric sweep
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    """Search many parametric families and keep the best vector."""
    best_vec = None
    best_val = float('inf')

    # ----- power‑law (including shift) -----
    for delta in (0.0, 0.5, 1.0):
        for p in np.linspace(0.30, 0.90, 121):   # step 0.005
            if time.time() >= deadline:
                return best_vec
            a = power_law(n, p, delta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- geometric with extremely fine grid near 1 -----
    qs = np.concatenate((
        np.linspace(0.90, 0.99, 101),
        np.linspace(0.990, 0.9999, 200)
    ))
    for q in qs:
        if time.time() >= deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ----- mixed plateau + power‑law -----
    for L_frac in np.linspace(0.02, 0.30, 15):
        for p in np.linspace(0.30, 0.80, 26):
            if time.time() >= deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- beta‑shape (more extreme exponents) -----
    alphas = np.linspace(-0.95, 3.0, 24)
    betas = np.linspace(-0.95, 3.0, 24)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- random convex combos of a few bases -----
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    power05 = power_law(n, 0.5)
    geom98 = geometric(n, 0.98)
    bases = [uniform, power05, geom98]

    for _ in range(1500):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(bases))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, bases))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    print(f"[parametric sweep] best score = {best_val:.6f}")
    return best_vec


# ----------------------------------------------------------------------
# 2. Mirror‑descent optimisation (log‑parameterisation + Adam‑style)
# ----------------------------------------------------------------------
def mirror_descent(a0: np.ndarray, deadline: float,
                   eta0: float = 0.35,
                   beta1: float = 0.9,
                   beta2: float = 0.999) -> np.ndarray:
    """
    Optimises a on the simplex by working in the log‑space z = log(a)
    and applying an Adam‑style mirror‑descent update.
    """
    n = a0.shape[0]
    # initialise log‑variables safely
    z = np.log(np.maximum(a0, 1e-12))
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = a0.copy()
    best_M = full_convolution(a0).max()
    t = 0

    while time.time() < deadline:
        t += 1
        # obtain current a on the simplex
        a = np.exp(z)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        conv = full_convolution(a)
        M = conv.max()
        # active indices (within a tiny tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of the active quadratic forms
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # transform to gradient w.r.t. z (mirror map = KL)
        avg = np.dot(a, g)
        g_z = a * (g - avg)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta0 / (np.sqrt(v_hat) + 1e-8)

        # mirror‑descent update in the log‑space
        z -= step * m_hat

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.004 / math.sqrt(t))
            z += jitter

        # evaluate the new point for bookkeeping
        a_new = np.exp(z)
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= a_new_sum

        M_new = full_convolution(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()

    return best_a


# ----------------------------------------------------------------------
# 3. Biased stochastic hill‑climb
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """
    Greedy stochastic moves that preferentially take mass from large
    entries and give it to small ones.  Step‑size adapts to successes.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    fails = 0

    # pre‑compute index array once
    indices = np.arange(n, dtype=np.int64)

    while time.time() < deadline and it < max_iter:
        # sample source index proportionally to current mass (large entries more likely)
        i = np.random.choice(indices, p=a)
        # sample destination uniformly but avoid picking the source
        j = np.random.randint(n)
        if j == i:
            continue

        delta = step_scale * a[i]       # move a fraction of the source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep feasibility (non‑negative) via projection
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand                     # keep the improvement
            step_scale = min(step_scale * 1.07, 0.6)
            fails = 0
        else:
            step_scale *= 0.996
            fails += 1

        it += 1

        # global jitter after a long dry spell
        if fails > 180_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            fails = 0

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for harness bookkeeping
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Fallback (if a previously discovered vector is available)
    # ------------------------------------------------------------------
    best_seq, best_val = try_fallback()

    # ------------------------------------------------------------------
    # 1. Parametric sweep (≈ 250 s)
    # ------------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    n = 4096                     # a bit larger than the previous 2048
    if time.time() < sweep_deadline:
        a0 = parametric_sweep(n, sweep_deadline)
        if a0 is not None:
            val = compute_score(a0)
            print(f"[sweep] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a0.tolist()

    # ------------------------------------------------------------------
    # 2. Mirror‑descent polishing (≈ 400 s)
    # ------------------------------------------------------------------
    md_deadline = deadline - 150.0
    if best_seq is not None and time.time() < md_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_md = mirror_descent(a_start, md_deadline,
                              eta0=0.35, beta1=0.9, beta2=0.999)
        val = compute_score(a_md)
        print(f"[mirror‑descent] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_md.tolist()

    # ------------------------------------------------------------------
    # 3. Final stochastic hill‑climb (rest of the budget)
    # ------------------------------------------------------------------
    hc_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hc_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_hc = hill_climb(a_start, hc_deadline)
        val = compute_score(a_hc)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_hc.tolist()

    # ------------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final report using the official evaluate_sequence
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
