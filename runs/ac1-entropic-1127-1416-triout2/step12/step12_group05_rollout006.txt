"""
Improved search for a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm builds on the previous state‑of‑the‑art but adds several
new ideas:

1. **Richer analytic families** –  a convex mixture of the symmetric
   product family and the exponential‑U family (three free parameters:
   α, γ and mixture weight λ).  This gives a much more flexible shape
   than any single family.

2. **Focused parameter optimisation** –  a fast random‑grid + local
   hill‑climb on (α,γ,λ) using the exact objective.  The search runs
   while time permits and keeps the best vector found.

3. **Longer Adam on the smooth‑max surrogate** –  we run Adam with a
   higher final temperature (β≈2·10⁴) and a slightly larger learning
   rate, again limited only by the remaining time budget.

4. **Projected sub‑gradient descent** –  the same exact‑max descent as
   before, but with a step‑size schedule that adapts to the current
   iteration count.

5. **Final simulated‑annealing hill‑climb** –  a more aggressive SA
   schedule (higher initial temperature, slower cooling) to polish
   the solution.

All steps respect the 1000 s wall‑clock limit and make heavy use of
FFT‑based convolution (O(n log n)) for speed.  The routine returns
the best sequence (list of floats) found within the budget.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)^2).
    a is assumed non‑negative (but we guard against a zero sum).
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s

# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


def mixed_product_exp_family(n: int, alpha: float, gamma: float,
                            lam: float) -> np.ndarray:
    """
    Convex mixture of the product family and the exponential‑U family.
    lam=1 → pure product, lam=0 → pure exponential‑U.
    """
    prod = symmetric_product_family(n, alpha)
    expu = exp_u_family(n, gamma)
    w = lam * prod + (1.0 - lam) * expu
    w /= w.sum()
    return w

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 15.0,
                         beta_end: float = 20_000.0,
                         lr: float = 0.015,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    No forced symmetry unless ``sym_interval`` > 0.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise via log‑softmax (ensures positivity)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z) / np.exp(z).sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear schedule for the smooth‑max temperature β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z) / np.exp(z).sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 2500 == 0:
            a_tmp = np.exp(z) / np.exp(z).sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved → {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z) / np.exp(z).sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement → {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved → {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 6_000_000,
                  temp_factor: float = 1.2e-3,
                  step_scale_init: float = 0.15) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # higher initial temperature (proportional to current best)
    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                       # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.08, 0.5)   # reward progress
        else:
            step_scale *= 0.96

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def build_mixed_vector(n: int,
                       alpha: float,
                       gamma: float,
                       lam: float) -> np.ndarray:
    """
    Returns a normalised vector from the mixture
    lam·product(alpha) + (1‑lam)·expU(gamma).
    """
    vec = mixed_product_exp_family(n, alpha, gamma, lam)
    return vec


def random_param_search(n: int,

                        init_alpha: float = None,
                        init_gamma: float = None,

    Returns the best vector found and its score.

        sc = compute_score(cand)
        best_vec, best_score = cand, sc

    start = time.time()
    # Phase‑1: pure random sampling

        gamma = np.random.uniform(0.0, 6.0)

        if sc < best_score:

            best_vec = vec.copy()
            best_alpha, best_gamma, best_lam = alpha, gamma, lam

    # Phase‑2: simple hill‑climb on parameters (Gaussian proposals)
    if best_vec is None:  # fallback – use pure product family

            best_vec = vec.copy()
            best_alpha, best_gamma, best_lam = alpha, gamma, lam
            print(f"[{time.time():.1f}] param hill improved → {best_score:.6f}")

    return best_vec, best_score
