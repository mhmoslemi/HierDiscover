    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # baseline: optional fallback using existing height_sequence_1 defined globally.

    # Stage 1: LP-based shape optimization
    n_small = 512
    # initial shape parameters (α)
    init_alpha = 0.5
    f = symmetric_product_family(n_small, init_alpha)  # normalized sum 1
    best_f = f.copy()
    best_score_small = compute_score(f)
    t = 0.4
    decay = 0.85
    max_lp_iters = 12
    for it in range(max_lp_iters):
        if time.time() >= deadline - 30:
            break
        # compute M
        conv_f = convolution_fft(f, f)
        M = conv_f.max()
        # Build Toeplitz constraint matrix
        c_col = np.concatenate([f, np.zeros(n_small-1)])

            print(f"[{time.time():.1f}] LP iteration {it} failed, breaking")
            break
        g = res.x

        sumg = g.sum()

        f = (1 - t) * f + t * g

        f = f / f.sum()

"""
Hybrid LP + smooth‑max optimisation for the min‑max self‑convolution problem.

Stage 1 – a cheap 1‑D LP (SciPy) on a modest size (n≈512) builds a non‑uniform
distribution f that already reduces max (f∗f).  The LP maximises Σ g_j under the
linear constraints (f∗g)_k ≤ M where M = max(f∗f).  After a few convex‑combination
steps (t ≈ 0.4 → 0) we obtain a shape f_small.

Stage 2 – up‑sample f_small to the original work‑size (n=8192) by block‑repeat;
the shape is kept on the probability simplex (∑a_i=1).

Stage 3 – Adam optimiser on a smooth log‑sum‑exp surrogate of the max‑convolution
(β temperature scheduled from 5 to 8000).  This further flattens the autocorrelation
while staying in the feasible simplex.

Stage 4 – a short projected sub‑gradient descent and a light simulated‑annealing
hill‑climb polish the final vector.

All steps respect the 1000 s wall‑clock budget, fall back gracefully if SciPy is
unavailable, and finally return a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, scoring, projections)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a)/(sum(a)²).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max()) / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# ----------------------------------------------------------------------
# Analytic families (used for a quick coarse scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Guard against overflow for extreme α
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 5.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.012,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp (smooth‑max) surrogate of max(conv).
    No forced symmetry unless ``sym_interval`` > 0.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # log‑softmax initialisation guarantees positivity
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z) / np.exp(z).sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule for smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z) / np.exp(z).sum()

        # optional symmetrisation every ``sym_interval`` steps
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                # smooth‑max weights

        # gradient of smooth‑max w.r.t. a  (see derivation in the original code)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 4000 == 0:
            a_tmp = np.exp(z) / np.exp(z).sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z) / np.exp(z).sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     step0: float = 0.04) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000,
                  temp_factor: float = 1e-3,
                  step_scale_init: float = 0.15) -> Tuple[np.ndarray, float]:
    """
    Light simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based shape optimisation (small n)
# ----------------------------------------------------------------------


def lp_shape_optimisation(n: int,
                          init_alpha: float,
                          max_lp_iters: int,
                          t0: float,
                          decay: float,
                          deadline: float) -> Tuple[np.ndarray, float]:
    """
    Perform a few convex‑combination steps with a linear programme:
    maximise Σ g_j  s.t. (f∗g)_k ≤ M  (for all k)  and g≥0.
    Returns the best distribution found (sum = 1) and its small‑scale score.
    """
    # initialise with a product‑family (arcsine‑like) shape
    f = symmetric_product_family(n, init_alpha)

    best_f = f.copy()
    best_score = compute_score(f)

    t = t0
    for it in range(max_lp_iters):
        if time.time() >= deadline - 5.0:
            break

        conv_f = convolution_fft(f, f)
        M = conv_f.max()

        # build the Toeplitz matrix that maps g ↦ f∗g
        col = np.concatenate([f, np.zeros(n - 1)])
        row = np.concatenate([f[:1], np.zeros(n - 1)])
        try:
            from scipy.linalg import toeplitz
            A = toeplitz(col, row)            # shape (2n‑1, n)
        except Exception as e:
            print("SciPy toeplitz unavailable:", e)
            break

        # linear programme: maximise Σ g_j
        from scipy.optimize import linprog
        obj = -np.ones(n)                     # minimise –Σg  ≡ maximise Σg
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M)

        res = linprog(obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'disp': False, 'maxiter': 500})
        if not res.success:
            print(f"[{time.time():.1f}] LP iteration {it} failed – exiting LP phase")
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()                       # renormalise to Σ=1

        # convex combination and re‑normalise (numerical safety)
        f = (1.0 - t) * f + t * g
        f = f / f.sum()

        sc = compute_score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
            print(f"[{time.time():.1f}] LP iter {it} improved small‑score to {best_score:.6f}")

        t *= decay

    return best_f, best_score
