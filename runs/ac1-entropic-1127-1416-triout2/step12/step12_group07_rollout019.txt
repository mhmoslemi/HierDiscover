"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

The search consists of three stages:

1. **Initialisation** – start from a few simple analytic families
   (geometric, linear‑decreasing, Dirichlet‑sampled) and keep the best
   one as the seed.

2. **Simulated‑annealing (SA) refinement** – repeatedly move a random
   amount of mass between two positions, accept the move either if it
   improves the (smooth) objective ``2·n·max(conv)`` or with a
   temperature‑dependent probability.  The temperature is lowered
   linearly with the wall‑clock budget; the move size adapts to the
   current mass at the source index.

3. **Greedy hill‑climb polishing** – a fast stochastic hill‑climb that
   tries tiny mass‑transfers and only accepts improvements, to finish
   the optimisation.

All arithmetic respects the simplex constraint (non‑negative,
sum‑to‑one) via Euclidean projection onto the probability simplex.
The algorithm runs for the whole allotted budget (default 1000 s) and
returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length = 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a normalised vector
    (Σa == 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant; we therefore keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple families (used for initial seeds)
# ----------------------------------------------------------------------
def geometric(n: int, r: float) -> np.ndarray | None:
    """a_i = r**i  (geometric decay).  Returns a normalised vector."""
    if not (0.0 < r < 1.0):
        return None
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def linear_decreasing(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)  (simple decreasing shape, normalised)."""
    a = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a /= a.sum()
    return a


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1))."""
    a = rng.dirichlet(np.ones(n))
    return a / a.sum()


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float, max_iter: int = 500_000) -> np.ndarray:
    """Tiny random mass‑transfer moves, greedy acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    step_scale = 0.04
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a fraction of the source mass to the target
        delta = step_scale * a[i]                  # up to ~few % of source
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.08, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.987
            no_improve += 1

        if no_improve > 100_000:
            # shake to escape a flat plateau
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1

    return best_a

    """
    Search for a non‑negative coefficient sequence that minimises
    evaluate_sequence.  Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # 0. Optional fallback (use an existing construction if provided)
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Choose a good initial seed (few analytic families + random)
    # -----------------------------------------------------------------
    n = 2048
    candidates = []

    # (a) geometric decay (several r values)
    for r in (0.55, 0.70, 0.85, 0.93):
        vec = geometric(n, r)
        if vec is not None:
            candidates.append(vec)

    # (b) linear decreasing
    candidates.append(linear_decreasing(n))

    # (c) a few Dirichlet samples (coarse random start)
    for _ in range(3):
        candidates.append(random_dirichlet(n, rng))

    # pick the best of the few cheap candidates
    best_sc = float("inf")
    for vec in candidates:
        sc = compute_score(vec)
        if sc < best_sc:
            best_sc, best_a = sc, vec.copy()

    best_seq = best_a.tolist()
    best_val = compute_score(best_a)
    print(f"[initial] best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Simulated annealing refinement (main workhorse)
    # -----------------------------------------------------------------
    # keep a copy that stays on the simplex (sum=1)
    a = best_a.copy()
    cur_sc = best_val
    best_a_overall = a.copy()
    best_sc_overall = cur_sc

    # annealing schedule parameters
    T0 = 0.08            # start temperature
    T_final = 2e-5       # final temperature
    max_iters = 400_000

    it = 0
    while time.time() < deadline and it < max_iters:
        # linear temperature decay
        progress = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (1.0 - progress) + T_final * progress

        # propose a mass transfer move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # limit the transferred amount to a fraction of the source mass
        max_move = min(a[i], 0.12)                     # at most 12 % of source
        delta = rng.random() * max_move               # uniform in [0, max_move]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # maintain feasibility
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cand_sc = compute_score(cand)
        d_sc = cand_sc - cur_sc

        # accept if improvement or with Metropolis probability
        if d_sc < 0 or rng.random() < np.exp(-d_sc / max(T, 1e-12)):
            a = cand
            cur_sc = cand_sc
            if cand_sc < best_sc_overall:
                best_sc_overall, best_a_overall = cand_sc, cand.copy()

        # early exit if we have already reached the target
        if best_sc_overall < 1.503:
            break

        it += 1

    print(f"[SA] best score after annealing = {best_sc_overall:.6f}")

    # -----------------------------------------------------------------
    # 3. Final greedy hill‑climb polishing
    # -----------------------------------------------------------------
    if time.time() < deadline:
        final_vec = hill_climb(best_a_overall, deadline)
        final_sc = compute_score(final_vec)
        print(f"[hill‑climb] final (internal) score = {final_sc:.6f}")
        if final_sc < best_sc_overall:
            best_a_overall = final_vec
            best_sc_overall = final_sc

    # -----------------------------------------------------------------
    # 4. Return the best sequence (as plain Python list)
    # -----------------------------------------------------------------
    final_seq = best_a_overall.tolist()
    final_official = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Official score = {final_official:.6f}")
    return final_seq
