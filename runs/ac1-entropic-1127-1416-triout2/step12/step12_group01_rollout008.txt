def lp_improve(f, deadline, max_iter=30, alpha=0.1, tol=1e-8):
    n = f.shape[0]
    best_f = f.copy()
    best_M = compute_max_conv(f)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        M = best_M
        # build constraint matrix based on current f
        # Build sparse matrix A (2n-1 x n)
        rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
        cols = np.tile(np.arange(n), n)
        data = np.repeat(f, n)
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # binary?
        # Solve LP: maximize sum(g) s.t. A g <= M, g >= 0
        c = -np.ones(n)  # minimize -sum(g)
        b_ub = np.full(2*n-1, M, dtype=float)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            # fallback to no improvement and break
            break
        g = res.x
        # combine
        f_new = (1 - alpha) * f + alpha * g
        # ensure non-negative
        f_new = np.maximum(f_new, 0.0)
        # renormalize sum to 1
        total = f_new.sum()
        if total <= 0:
            f_new = np.full_like(f_new, 1.0 / n)
        else:
            f_new = f_new / total
        # compute new max conv
        M_new = compute_max_conv(f_new)
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_f = f_new
            f = f_new
        else:
            # If no improvement, reduce alpha maybe, or break?
            # For simplicity, break if not improving after few steps
            if alpha > 0.01:
                alpha *= 0.5
            else:
                break
    return best_f

def solve_lp_for_f(f, M):
    n = len(f)
    # build constraint matrix A
    A = build_constraint_matrix(f)
    # bounds for g: g >=0
    bounds = [(0, None)] * n
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M, dtype=float)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: zero vector
        return np.zeros_like(f)
    return res.x

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. Initialise a small pool of promising seeds (fallback solution,
   a few Dirichlet draws, or parametric families).

2. For each seed run a **linear‑programming polishing** loop:
   * Fix the current vector f (∑f=1) and compute M = max(f∗f).
   * Solve the LP   maximise   Σ g_j
          subject to  (f∗g)_k ≤ M   for all k=0…2n‑2,
                      g ≥ 0 .
     The constraint matrix is the Toeplitz convolution matrix of f.
   * Mix the LP solution g into f with a small step size α
     (convex combination, renormalise to the simplex) and keep the
     improvement if the new maximal autocorrelation decreases.
   * Decrease α when no improvement is observed; stop after a
     time limit or after a fixed number of iterations.

   The LP step tries to increase the total mass of the vector while
   keeping the cross‑convolution f∗g below the current peak M; the
   convex combination therefore reduces the peak after renormalisation.

3. After the LP polishing run a fast **exponentiated‑gradient (EG)**
   descent (the same scheme that was used previously) to further
   reduce the peak.

4. Finish with a short simulated‑annealing hill‑climb to escape possible
   local minima.

All work respects the 1000 s wall‑clock budget.  The final sequence
is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional imports – SciPy may be unavailable on a minimal installation.
# ----------------------------------------------------------------------
try:
    import scipy.sparse
    import scipy.optimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# Numerical utilities

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    # Numerical noise may produce tiny negative values – clip.
    return np.maximum(conv, 0.0)


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear autocorrelation of a."""
    return float(full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric families (useful for generating a few diverse seeds)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape:
       a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Linear‑programming polishing utilities (requires SciPy)
# ----------------------------------------------------------------------


def _build_conv_constraint_matrix(f: np.ndarray) -> "scipy.sparse.csr_matrix":
    """
    Build the (2n‑1)×n constraint matrix A such that
        (f∗g)_k = Σ_j A[k, j]·g_j .
    The entry A[k, j] = f_{k‑j} if 0 ≤ k‑j < n else 0.
    """
    n = f.shape[0]
    # row indices are i + j for all i,j in [0,n)
    rows = np.add.outer(np.arange(n, dtype=np.int64),
                        np.arange(n, dtype=np.int64)).ravel()
    cols = np.tile(np.arange(n, dtype=np.int64), n)
    data = np.repeat(f, n)  # each f_i appears n times, once per column
    m = 2 * n - 1
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n))


def _solve_lp_max_sum(given_f: np.ndarray, bound_M: float) -> np.ndarray:
    """
    Solve
        maximise   Σ g_j
        subject to A·g ≤ bound_M   (A obtained from given_f)
                   g ≥ 0 .
    Returns the optimal g (as a dense 1‑D NumPy array).  In the very
    unlikely case that the LP fails, a zero vector is returned.
    """
    n = given_f.shape[0]
    # Build sparse constraint matrix
    A = _build_conv_constraint_matrix(given_f)

    # Linear programming data
    c = -np.ones(n)                       # minimise –Σ g_j  ⇒ maximise Σ g_j
    b_ub = np.full(2 * n - 1, bound_M)    # right‑hand side for each lag
    bounds = [(0.0, None)] * n             # g_j ≥ 0

    # Solve using HiGHS (fast, supports sparse matrices)
    res = scipy.optimize.linprog(c,
                                 A_ub=A,
                                 b_ub=b_ub,
                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        # If the solver fails for any reason we fall back to a zero vector.
        return np.zeros_like(given_f)
    return res.x


def lp_polish(f0: np.ndarray, deadline: float,
              max_iter: int = 30,
              init_alpha: float = 0.2,
              alpha_decay: float = 0.5,
              verbose_prefix: str = "") -> np.ndarray:
    """
    Repeated LP‑based polishing of a probability vector.

    Parameters
    ----------
    f0 : np.ndarray
        Starting probability vector (sum ≈ 1, non‑negative).
    deadline : float
        Absolute time (seconds since epoch) after which the routine must stop.
    max_iter : int
        Upper bound on the number of LP iterations.
    init_alpha : float
        Initial mixing weight for the convex combination f←(1‑α)f+αg.
    alpha_decay : float
        Multiplicative factor applied when a step fails to improve.
    verbose_prefix : str
        Prefix added to all printed status lines (useful when several
        seeds are processed in parallel).

    Returns
    -------
    np.ndarray
        The best vector discovered during the run (still normalised to sum=1).
    """
    if not _SCIPY_AVAILABLE:  # pragma: no cover
        print(f"{verbose_prefix}SciPy unavailable – skipping LP polish.")
        return f0.copy()

    f = f0.copy()
    best_f = f.copy()
    best_M = compute_max_conv(f)

    alpha = init_alpha
    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # 1) solve LP for the current f
        g = _solve_lp_max_sum(f, best_M)

        # If the LP returns the zero vector there is nothing to gain.
        if np.allclose(g, 0):
            break

        # 2) try a convex combination
        f_candidate = (1.0 - alpha) * f + alpha * g
        f_candidate = np.maximum(f_candidate, 0.0)
        s = f_candidate.sum()
        if s <= 0.0:
            f_candidate = np.full_like(f_candidate, 1.0 / f_candidate.shape[0])
        else:
            f_candidate /= s

        M_candidate = compute_max_conv(f_candidate)

        # 3) accept if the peak decreased
        if M_candidate < best_M - 1e-12:
            f = f_candidate
            best_f = f_candidate.copy()
            best_M = M_candidate
            # be a little optimistic – try a slightly larger step next round
            alpha = min(alpha * 1.1, 0.5)
            print(f"{verbose_prefix}[lp‑iter {it}] improved M = {best_M:.6e}, α = {alpha:.4f}")
        else:
            # no improvement – shrink the step size
            alpha *= alpha_decay
            if alpha < 1e-4:
                print(f"{verbose_prefix}[lp‑iter {it}] α too small, stopping.")
                break

    return best_f

def eg_polish(a0: np.ndarray, deadline: float,
              eta0: float = 0.7, tol: float = 1e-10) -> np.ndarray:
    """
    Exponentiated‑gradient minimisation of max_k (a*a)_k on the simplex.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * tol
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: g_i = 2·∑_{k∈active} a_{k‑i}
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % 500 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

        if vec.sum() == 0:
            return
        vec = vec / vec.sum()
        seeds.append(vec.copy())

    for conc in (0.2, 0.5, 1.0, 2.0, 4.0):

            a = np.random.dirichlet(np.full(n, conc))
            try_add(a)

    # 2) Power‑law sweep (coarse)

        if time.time() - start > max_time:
            return seeds

            if time.time() - start > max_time:
                return seeds
            try_add(generate_beta(n, a_param, b_param))

    return seeds

    seed_pool = []
    if fallback is not None:
        seed_pool.append(best_seq.copy())
    if time.time() < deadline - 250.0:
        # generate a few more seeds (time‑budgeted)
        new_seeds = generate_seed_pool(n, max_time=seed_pool_time)
        seed_pool.extend(new_seeds)

    # Keep only the most promising few (by exact score)
    seed_pool.sort(key=lambda v: compute_score(v))
    seed_pool = seed_pool[:5]  # at most five seeds to stay within time budget

    # --------------------------------------------------------------
    # 2. LP‑based polishing for every seed
    # --------------------------------------------------------------
    lp_deadline = deadline - 150.0  # keep time for EG + anneal
    for idx, seed_vec in enumerate(seed_pool):
        if time.time() >= lp_deadline:
            break
        print(f"[lp‑stage] polishing seed {idx} (score {compute_score(seed_vec):.6f})")
        polished = lp_polish(seed_vec, lp_deadline,
                             max_iter=30,
                             init_alpha=0.2,
                             alpha_decay=0.5,
                             verbose_prefix=f"[seed{idx}] ")
        val = compute_score(polished)
        print(f"[lp‑stage] seed {idx} after LP score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = polished.copy()

    # --------------------------------------------------------------
    # 3. Exponentiated‑gradient polishing (fast descent)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        eg_deadline = deadline - 80.0
        print(f"[eg‑polish] starting EG polishing (current best {best_val:.6f})")
        refined = eg_polish(best_seq, eg_deadline)
        val = compute_score(refined)
        print(f"[eg‑polish] after EG score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined.copy()

    # --------------------------------------------------------------
    # 4. Final simulated‑annealing polish (short run)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        anneal_deadline = deadline - 5.0
        print(f"[anneal] final annealing (current best {best_val:.6f})")
        final = simulated_annealing(best_seq, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] after annealing score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = final.copy()

    # --------------------------------------------------------------
    # 5. Fallback to a uniform distribution if nothing succeeded
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_val = compute_score(best_seq)

    # --------------------------------------------------------------
    # 6. Report using the official evaluation function
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq.tolist()
