def full_convolution_fft(a):
    n = a.shape[0]
    m = 2*n - 1
    size = 1 << ((m-1).bit_length())  # next power of two >= m
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[:m]
    return conv
```

"""
A three‑stage optimizer for the non‑negative step‑function problem.

Stage 1 – wide parametric and random sampling.
          Generates candidates from offset‑power‑law, geometric and Dirichlet
          families (including coarse block‑shapes).  The ten best (lowest
          2·n·max(conv)) are kept.

Stage 2 – smooth‑gradient polishing.
          Uses a log‑sum‑exp (soft‑max) approximation of the max‑convolution.
          For a given smoothing parameter α the objective is
                fα(a) = 2·n·(1/α)·log Σ_k exp(α·conv_k(a)),
          whose gradient can be obtained via a single FFT‑based convolution.
          A few passes with increasing α (5→80) and a momentum step improve the
          candidates while always projecting onto the probability simplex.

Stage 3 – simulated‑annealing hill‑climb.
          Starting from the best polished vector, random mass‑transfer moves
          are proposed and accepted with a Metropolis rule (temperature slowly
          cools).  This allows occasional uphill steps to escape shallow local
          minima.

All stages respect the global 1000 s wall‑clock budget.  The final sequence
is returned as a plain Python list; its official score is obtained by calling
the provided ``evaluate_sequence`` routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx_i = 1, x_i ≥ 0."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a by real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` when Σa = 1:
           2·n·max(conv(a,a)).
       The function is scale‑invariant; we therefore keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------
def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    """a_i = (i+offset)^{-p}   (offset>0)."""
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, r: float) -> np.ndarray:
    """a_i = r^i  (0<r<1)."""
    if r <= 0 or r >= 1:
        return None
    a = r ** np.arange(n, dtype=np.float64)
    if a.sum() == 0:
        return None
    a /= a.sum()
    return a


def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    """Coarse piecewise‑constant shape: K random decreasing blocks."""
    blocks = rng.dirichlet(np.ones(K))
    blocks = np.sort(blocks)[::-1]                     # decreasing
    repeats = np.full(K, n // K, dtype=int)
    remainder = n - repeats.sum()
    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)[:n]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Stage 1 – broad sampling
# ----------------------------------------------------------------------
def generate_initial_candidates(n: int, deadline: float, keep: int = 10):
    """Return up to *keep* best vectors sampled from a variety of families."""
    best = []       # list of (score, vector), smallest score first

    def consider(vec):
        if vec is None or np.isnan(vec).any():
            return
        sc = compute_score(vec)
        if len(best) < keep:
            best.append((sc, vec))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec)
            best.sort(key=lambda x: x[0])

    rng = np.random.default_rng(987654)

    # ----- offset‑power‑law (wide offset range) -----
    offs = np.linspace(0.0, n * 0.6, 15)[1:]   # avoid offset = 0 (division)
    ps = np.linspace(0.2, 2.0, 25)
    for off in offs:
        for p in ps:
            if time.time() >= deadline:
                break
            vec = powerlaw_offset(n, p, off)
            consider(vec)
        if time.time() >= deadline:
            break

    # ----- geometric decay -----
    rs = np.linspace(0.30, 0.99, 30)
    for r in rs:
        if time.time() >= deadline:
            break
        vec = geometric(n, r)
        consider(vec)

    # ----- random Dirichlet (different concentration) -----
    alphas = [0.1, 0.3, 0.6, 1.0, 2.0]
    for a0 in alphas:
        for _ in range(150):
            if time.time() >= deadline:
                break
            vec = rng.dirichlet(np.full(n, a0))
            consider(vec)

    # ----- coarse block shapes -----
    K = max(5, n // 200)          # ~10‑30 blocks for n≈2000
    for _ in range(800):
        if time.time() >= deadline:
            break
        vec = block_shape(n, K, rng)
        consider(vec)

    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Stage 2 – smooth‑gradient polishing
# ----------------------------------------------------------------------
def smoothed_obj_and_grad(a: np.ndarray, alpha: float):
    """
    Returns a tuple (obj, grad) where
        obj  = 2·n·smax  with  smax = (1/α)·log Σ exp(α·conv)
        grad = ∇obj  (projected onto the simplex afterwards)
    The gradient is obtained via a single FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)                     # length 2n‑1
    max_c = conv.max()
    # stable soft‑max weights
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    w_norm = w / w_sum                                 # Σ w_norm = 1

    # cross‑correlation a ⋆ w_norm (equivalent to Σ_k w_k·2·a_{k‑j})
    w_rev = w_norm[::-1]                               # reverse for correlation
    m = 3 * n - 2
    size = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(w_rev, n=size)
    cross = np.fft.irfft(A * B, n=size)[:m]            # full linear conv
    raw_grad = 2.0 * cross[(2 * n - 2) - np.arange(n)]   # shape (n,)

    # smoothed objective (2·n·smax)
    smax = max_c + np.log(w_sum) / alpha
    obj = 2.0 * n * smax
    grad = 2.0 * n * raw_grad
    return obj, grad


def gradient_descent_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed max‑convolution.
    Uses a schedule of increasing smoothing parameters.
    """
    a = a0.copy()
    best = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    momentum = np.zeros_like(a)
    alphas = [5.0, 10.0, 20.0, 40.0, 80.0]
    it = 0

    while time.time() < deadline:
        for alpha in alphas:
            # a few inner steps per smoothing level
            for _ in range(8):
                if time.time() >= deadline:
                    break
                obj_s, grad = smoothed_obj_and_grad(a, alpha)

                # basic adaptive step size (inverse of gradient norm)
                step = 0.15 / (np.linalg.norm(grad) + 1e-12)

                # Nesterov‑style momentum update
                momentum = 0.85 * momentum - step * grad
                cand = a + momentum
                cand = project_to_simplex(cand)

                # exact score for bookkeeping (cheap compared to FFTs)
                cand_val = compute_score(cand)
                if cand_val < best_val:
                    best_val, best = cand_val, cand.copy()

                # accept if the smoothed objective improves,
                # otherwise fall back to a plain gradient step.
                if obj_s < 2.0 * n * (np.log(np.exp(alpha * (full_convolution_fft(a).max() - full_convolution_fft(a).max()))).sum() / alpha):
                    a = cand
                else:
                    cand2 = a - step * grad
                    a = project_to_simplex(cand2)

                it += 1
                if it > 4000:        # safety stop
                    break
            if time.time() >= deadline or it > 4000:
                break
        if it > 4000:
            break
    return best


# ----------------------------------------------------------------------
# Stage 3 – simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def annealing_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Stochastic hill‑climb with a simple temperature schedule.
    Randomly moves a fraction of mass from one index to another.
    Occasionally accepts uphill moves based on Metropolis probability.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = compute_score(a)
    cur_val = best_val

    rng = np.random.default_rng(112233)

    T = 0.020                     # initial temperature
    decay = 0.9995                # exponential cooling
    update = 500                  # temperature update frequency
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 8 % of the source mass (scaled by a[i])
        delta = rng.random() * 0.08 * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cand_val = compute_score(cand)
        dE = cand_val - cur_val

        # Metropolis acceptance
        if dE < 0 or rng.random() < np.exp(-dE / T):
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val, best = cand_val, cand.copy()

        it += 1
        if it % update == 0:
            T *= decay
            # slowly shrink the typical move size as we cool
            if T < 1e-5:
                T = 1e-5

    return best

            refined = gradient_descent_refine(vec, grad_deadline)
            val = compute_score(refined)
            print(f"[grad‑refine {idx}] score = {val:.6f}")
            if val < best_polished_val:
                best_polished_val, best_polished = val, refined

        if best_polished is not None:
            if best_polished_val < best_val:
                best_val = best_polished_val
                best_seq = best_polished.tolist()

    # ------------------------------------------------------------------
    # 3. Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline - 5.0
    if best_seq is not None and time.time() < final_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = annealing_hill_climb(a_start, final_deadline)
        final_val = compute_score(a_final)
        print(f"[annealing] final exact score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_final.tolist()

    # ------------------------------------------------------------------
    # 4. Safeguard – uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5. Official evaluation (includes security wrappers)
    # ------------------------------------------------------------------
    official_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported (official) score = {official_score:.6f}")
    return best_seq
