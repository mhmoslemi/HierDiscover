Propose a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

Algorithm
---------
1.  Choose a moderate length (n = 2048) – large enough to obtain a
    fine discretisation but small enough for a dense linear programme.
2.  Initialise a with the best symmetric product‑family vector found by a
    cheap coarse sweep over the exponent α.
3.  **LP improvement step** – repeatedly
        • compute M = max(a*a)
        • solve   maximise   Σ b_i
          subject  b ≥ 0
                    a ⊛ b ≤ M   (full a‑convolution with b)
          (a linear programme, solved with SciPy’s HiGHS)
        • renormalise b to a probability vector g
        • mix a ← (1‑t)·a + t·g   (t decays with the progress)
    The LP forces the new vector to spread mass while keeping the
    cross‑convolution below the current peak, which provably reduces the
    objective unless a is already optimal.
4.  Refine the best vector with three smooth‑optimisers:
        – Adam on a log‑sum‑exp surrogate of the max‑convolution,
        – projected sub‑gradient descent on the exact max‑objective,
        – a simulated‑annealing hill‑climb that moves mass between entries.
5.  Return the best sequence found (a plain ``list[float]``).

All steps respect the 1000 s wall‑clock budget.  If SciPy is unavailable
the LP stage is skipped and the algorithm falls back to the smoother‑only
pipeline.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# basic utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT.
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)^2).
    a is assumed non‑negative.
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# ----------------------------------------------------------------------
# analytic families (used for a fast seed)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Discrete beta‑type density."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Optimisers that work on the smooth surrogate or the exact max
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 800_000,
                         beta_start: float = 10.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.012,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    The iterate is kept in log‑space (soft‑max) to guarantee positivity.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)                     # logits
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z) / np.exp(z).sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a /= a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 2000 == 0:
            a_tmp = np.exp(z)
            a_tmp /= a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final evaluation
    a_fin = np.exp(z)
    a_fin /= a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact L∞‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient at the maximizing index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 8_000_000,
                  temp_factor: float = 1e-3,
                  step_scale_init: float = 0.2) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 15) -> Tuple[float, float]:
    """Golden‑section search for a unimodal 1‑D function."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Helper for the LP stage
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n‑1, n)) such that
    A @ b = a ⊛ b   (full convolution).
    Row k contains a[i] where i ranges over indices that contribute to
    (a⊛b)[k].
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a[i]
    return A

# Main search driver

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time‑budget.
    Returns the best sequence as a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback to a previously known sequence
    # ------------------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. choose problem size
    # ------------------------------------------------------------------
    n = 2048  # moderate size – good trade‑off for the LP stage

    # ------------------------------------------------------------------
    # 2. quick coarse sweep of the symmetric product family for a seed
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse sweep for seed vector (n={n})")
    alphas = np.linspace(0.0, 6.0, 31)
    seed_a = None
    for alpha in alphas:
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            seed_a = vec
            print(f"[{time.time():.1f}] new seed α={alpha:.3f} → {best_score:.6f}")

    if seed_a is None:
        seed_a = np.full(n, 1.0 / n, dtype=np.float64)

    a = seed_a.copy()

    # ------------------------------------------------------------------
    # 3. LP improvement loop (requires SciPy)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
        have_linprog = True
    except Exception:
        have_linprog = False
        print("SciPy linprog not available – skipping LP stage.")

    if have_linprog:
        print(f"[{time.time():.1f}] entering LP improvement loop (n={n})")
        t_mix = 0.2
        lp_iter = 0
        while time.time() < deadline - 30.0:
            lp_iter += 1
            M = convolution_fft(a, a).max()
            # build the linear‑inequality matrix A @ b ≤ M
            A = build_conv_matrix(a)
            b_ub = np.full(A.shape[0], M, dtype=np.float64)

            # solve maximize Σ b_i  <=> minimize -Σ b_i
            res = linprog(c=-np.ones(n),
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': max(1, int(deadline - time.time()))})
            if not res.success:
                print(f"[{time.time():.1f}] LP failed: {res.message}")
                break

            b = res.x
            sum_b = b.sum()
            if sum_b < 1e-12:
                print(f"[{time.time():.1f}] LP returned near‑zero sum, aborting.")
                break

            g = b / sum_b
            # convex combination and re‑normalisation
            a = (1.0 - t_mix) * a + t_mix * g
            a = a / a.sum()

            sc = compute_score(a)
            if sc < best_score - 1e-6:
                best_score = sc
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] LP iter {lp_iter} improved to {best_score:.6f}")
                # shrink mixing factor slowly to fine‑tune
                t_mix = max(t_mix * 0.92, 0.01)
            else:
                # modestly reduce t_mix if no progress
                t_mix = max(t_mix * 0.99, 0.005)

    # ------------------------------------------------------------------
    # 4. Adam (smooth‑max) refinement – no forced symmetry
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Adam refinement (smooth max)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=a_start,
            deadline=deadline - 150.0,
            max_iters=800_000,
            beta_start=10.0,
            beta_end=12000.0,
            lr=0.010,
            sym_interval=0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent on the exact max‑objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline - 60.0,
            max_iters=2_500_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑gradient best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb (higher temperature)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=8_000_000,
            temp_factor=1e-3,
            step_scale_init=0.2,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. safety fallback in the unlikely event that nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
