def subgradient_expgrad(a0, deadline, eta0=0.5):

    a = a0.copy()
    best = a
    best_val = compute_score(a)
    t=0
    while time.time() < deadline:
        conv = full_convolution(a)
        max_val = conv.max()
        # identify a few indexes where conv close to max (within tolerance)
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]
        # compute gradient
        grad = np.zeros_like(a)
        idx_array = np.arange(a.shape[0])
        for k in active:
            # j = k - idx for each i; require j in [0,n-1]
            j = k - idx_array
            mask = (j >= 0) & (j < a.shape[0])
            grad[mask] += 2 * a[j[mask]]
        # update with exponentiated gradient
        eta = eta0 / np.sqrt(t+1)
        a = a * np.exp(-eta * grad)
        # renorm to sum=1
        a_sum = a.sum()
        if a_sum == 0 or not np.isfinite(a_sum):
            # fallback uniform
            a = np.full_like(a, 1.0/a.shape[0])
        else:
            a = a / a_sum
        # optionally jitter: every so often random move small
        if t % 2000 == 0 and t>0:
            jitter = np.random.randn(a.shape[0]) * (eta0 * 0.005 / np.sqrt(t+1))
            a = a + jitter
            # project onto simplex (non-negative)
            a = np.maximum(a, 0)

        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
        t+=1
    return best

def subgradient_expgrad(a0, deadline, eta0=0.5, jitter_interval=1000):
    a = a0.copy()
    best = a.copy()
    best_score = compute_score(a)
    t=0
    while time.time() < deadline:
        conv = full_convolution(a)
        max_val = conv.max()
        # get indices where conv close to max
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]
        # compute subgradient
        grad = np.zeros_like(a)
        idx = np.arange(a.shape[0])
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < a.shape[0])
            grad[mask] += 2 * a[j[mask]]
        # update with exponentiated gradient
        eta = eta0 / np.sqrt(t + 1)
        # small step to avoid overflow
        a = a * np.exp(-eta * grad)
        # renorm
        sum_a = a.sum()
        if sum_a <= 0 or not np.isfinite(sum_a):
            a = np.full_like(a, 1.0/a.shape[0])
        else:
            a = a / sum_a
        # optional jitter
        if t % jitter_interval == 0 and t > 0:
            jitter = np.random.randn(a.shape[0]) * (eta0 * 0.01 / np.sqrt(t+1))
            a = a + jitter
            a = np.maximum(a, 0)
            a = project_to_simplex(a)  # ensure sum=1
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        t += 1
    return best

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
1. **Initial pool** – a diverse set of candidates is generated from
   several parametric families (power‑law, exponential, beta, Gaussian,
   sine‑power) and many random Dirichlet draws with different
   concentration parameters.  The best few (by the exact objective
   ``2·n·max(conv)``) are kept.

2. **Genetic optimisation** – a simple GA maintains a population of
   size ``pop_sz``.  Offspring are created by crossover (weighted average)
   of two parents followed by a Gaussian mutation; the child is projected
   onto the probability simplex.  If it improves upon the worst element
   of the population it replaces it.  Random immigrants are injected
   periodically to keep diversity.

3. **Sub‑gradient polishing** – the best vector found by the GA is
   refined with an exponentiated‑gradient / sub‑gradient method that
   directly minimises the exact ``max(conv)``.  The gradient is obtained
   via an FFT‑based convolution with the active constraint mask, and the
   update is performed in the simplex using multiplicative (mirror‑descent)
   steps.  A small amount of jitter is added occasionally.

4. **Final report** – the refined vector is evaluated with the official
   ``evaluate_sequence`` function and returned as a plain list of floats.

All phases respect the 1000 s wall‑clock budget.  The implementation is
fully self‑contained and uses only ``numpy`` and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, v.size + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.size
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1)."""
    n = a.size
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a normalised vector)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 1.0 / n)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def random_dirichlet(n: int, conc: float) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc,…,conc)."""
    return np.random.dirichlet(np.full(n, conc))


# ----------------------------------------------------------------------
# Genetic algorithm utilities
# ----------------------------------------------------------------------


def crossover(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Weighted average of two parents, then re‑normalise."""
    alpha = np.random.rand()
    child = alpha * a + (1.0 - alpha) * b
    s = child.sum()
    if s <= 0 or not np.isfinite(s):
        child = np.full_like(child, 1.0 / child.size)
    else:
        child = child / s
    return child


def mutate(a: np.ndarray, sigma: float) -> np.ndarray:
    """Add Gaussian noise and project back onto the simplex."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s <= 0 or not np.isfinite(s):
        b = np.full_like(b, 1.0 / b.size)
    else:
        b = b / s
    return b


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exponentiated‑gradient)
# ----------------------------------------------------------------------


def subgradient_expgrad(a0: np.ndarray, deadline: float,
                       eta0: float = 0.6,
                       jitter_interval: int = 1500) -> np.ndarray:
    """
    Projected exponentiated‑gradient minimiser for  max(conv(a,a)).
    Maintains the simplex by multiplicative updates.
    """
    a = a0.copy()
    best = a.copy()
    best_val = compute_score(a)
    n = a.size
    m = 2 * n - 1

    t = 0
    while time.time() < deadline:
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-12
        active_mask = conv >= max_val - eps
        # binary weight vector for active constraints
        w = active_mask.astype(float)          # length m
        rev_w = w[::-1]

        # FFT‑based cross‑correlation to obtain the sub‑gradient
        size = 1 << ((n + rev_w.size - 1 - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.size - 1]

        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]            # shape (n,)

        eta = eta0 / np.sqrt(t + 1.0)
        # multiplicative update (mirror descent)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional jitter to escape plateaux
        if t % jitter_interval == 0 and t > 0:
            jitter = np.random.randn(n) * (eta0 * 0.01 / np.sqrt(t + 1.0))
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best = cur_val, a.copy()

        t += 1

    return best

def ga_optimize(initial_pop, deadline, pop_sz: int = 30,
                concs: list = None, n: int = 2048):
    """
    Simple steady‑state GA that keeps the best ``pop_sz`` individuals.
    Returns the best vector found before ``deadline``.
    """
    if concs is None:
        concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]

    # keep only the top ``pop_sz`` candidates
    population = sorted(initial_pop, key=lambda x: x[0])[:pop_sz]
    best_score, best_vec = population[0]

    iteration = 0
    while time.time() < deadline:
        # parent selection (uniform)
        i, j = np.random.choice(len(population), 2, replace=False)
        parent1 = population[i][1]
        parent2 = population[j][1]

        # generate child
        child = crossover(parent1, parent2)
        # mutation magnitude decays with time
        elapsed = time.time() - (deadline - (deadline - start_time))
        sigma = 0.08 * (1.0 - (time.time() - start_time) / (deadline - start_time))
        child = mutate(child, sigma)

        child_score = compute_score(child)

        # replace worst if better
        if child_score < population[-1][0]:
            population[-1] = (child_score, child)
            population.sort(key=lambda x: x[0])
            if child_score < best_score:
                best_score, best_vec = child_score, child.copy()

        # inject a random immigrant every 200 iterations
        if iteration % 200 == 0:
            conc = np.random.choice(concs)
            immigrant = random_dirichlet(n, conc)
            imm_score = compute_score(immigrant)
            if imm_score < population[-1][0]:
                population[-1] = (imm_score, immigrant)
                population.sort(key=lambda x: x[0])
                if imm_score < best_score:
                    best_score, best_vec = imm_score, immigrant.copy()

        iteration += 1

    return best_vec, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    n = 2048                     # length of the step‑function
    pop_sz = 30                  # GA population size
    initial_candidates = []      # (score, vector)

    # -----------------------------------------------------------------
    # 0.  Fallback (if a previously discovered construction is present)
    # -----------------------------------------------------------------
    try:
        fallback = np.array(height_sequence_1, dtype=float)  # type: ignore
        fallback = np.maximum(fallback, 0.0)
        fallback = fallback / fallback.sum()
        fallback_score = compute_score(fallback)
        initial_candidates.append((fallback_score, fallback))
        print(f"[fallback] score = {fallback_score:.6f}")
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1.  Generate a diverse initial pool
    # -----------------------------------------------------------------
    # a) parametric sweeps (light enough to stay within time)
    for p in np.linspace(0.2, 2.0, 21):
        if time.time() >= deadline:
            break
        vec = generate_powerlaw(n, p)
        initial_candidates.append((compute_score(vec), vec))

    for lam in np.linspace(0.001, 0.15, 20):
        if time.time() >= deadline:
            break
        vec = generate_exponential(n, lam)
        initial_candidates.append((compute_score(vec), vec))

    for alpha in np.linspace(0.2, 3.0, 12):
        if time.time() >= deadline:
            break
        for beta in np.linspace(0.2, 3.0, 12):
            vec = generate_beta(n, alpha, beta)
            initial_candidates.append((compute_score(vec), vec))

    for sigma in np.linspace(0.2 * n, 0.5 * n, 12):
        if time.time() >= deadline:
            break
        vec = generate_gaussian(n, sigma)
        initial_candidates.append((compute_score(vec), vec))

    for p in np.linspace(0.2, 5.0, 12):
        if time.time() >= deadline:
            break
        vec = generate_sine_power(n, p)
        initial_candidates.append((compute_score(vec), vec))

    # b) many Dirichlet draws with varied concentrations
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_conc = 200
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            vec = random_dirichlet(n, conc)
            initial_candidates.append((compute_score(vec), vec))

    # Keep only the best ``pop_sz`` candidates for the GA
    initial_candidates.sort(key=lambda x: x[0])
    init_pop = initial_candidates[:pop_sz]

    # -----------------------------------------------------------------
    # 2.  Genetic optimisation (major chunk of the budget)
    # -----------------------------------------------------------------
    ga_deadline = deadline - 20.0   # reserve ~20 s for polishing
    best_vec, best_score = ga_optimize(init_pop, ga_deadline,
                                       pop_sz=pop_sz, n=n)
    print(f"[GA] best score after evolution = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3.  Sub‑gradient polishing (final refinement)
    # -----------------------------------------------------------------
    polish_deadline = deadline
    refined = subgradient_expgrad(best_vec, polish_deadline,
                                 eta0=0.6, jitter_interval=1200)
    refined_score = compute_score(refined)
    print(f"[polish] refined score = {refined_score:.6f}")

    # -----------------------------------------------------------------
    # 4.  Final official evaluation & return
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(refined.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return refined.tolist()
