def smooth_max_grad(a, gamma):
    conv = convolution_fft(a, a)
    max_val = conv.max()
    w = np.exp(gamma * (conv - max_val))  # avoid overflow
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum
    # gradient = 2 * convolution_fft(w[::-1], a)  # full conv
    grad_full = convolution_fft(w[::-1], a)  # length = 3n-2
    start = 2 * a.shape[0] - 2  # 2n-2
    grad = 2.0 * grad_full[start:start + a.shape[0]]
    return grad

    a = np.array(start_a, dtype=np.float64)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    while t < max_iters:
        if time.time() >= deadline:
            break
        t += 1
        conv = convolution_fft(a, a)
        # compute softmax weights of conv (approx max)
        max_c = conv.max()
        exp_vals = np.exp(gamma * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        # compute gradient: g = 2 * conv(w[::-1] with a) slice
        w_rev = w[::-1]
        grad_full = convolution_fft(w_rev, a)  # length = 3n-2

        v = beta2 * v + (1 - beta2) * (grad * grad)

        a_new = np.maximum(a_new, 0.0)

"""
Search for a non‑negative step‑function that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

The approach mixes three ideas:

1.  A quick sweep of analytic families (product, power‑law, exponential‑U,
    Gaussian, cosine windows) to obtain a decent starting point.
2.  A low‑dimensional block representation tuned with differential evolution
    (DE).  The block parameters are exponentiated and normalised, giving a
    flexible, symmetric shape while keeping the optimisation cheap.
3.  A smooth‑max gradient descent (Adam optimiser) that uses a soft‑max
    approximation of the `max` and a correct quadratic sub‑gradient (the
    diagonal terms are weighted by 2).  After the Adam phase a short simulated
    annealing hill‑climb finishes the search.

All helpers are defined at module level – no closures – and the total
wall‑clock budget (default 1000 s) is respected.
"""

import time
import math
import numpy as np
from scipy.optimize import differential_evolution, linprog

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}  (U‑shaped, ends heavy)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-β}  (monotone, heavy left side)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """w_i ∝ exp(-γ·dist(i, centre)).  Heavy ends, exponential decay."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian: w_i ∝ exp(-γ·(i‑mid)²)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, window_type: str) -> np.ndarray:
    """Standard cosine windows (Hann, Hamming, Blackman)."""
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (heavy left side)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped start."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Low‑dimensional block optimisation (DE)
# ----------------------------------------------------------------------


def de_optimize(n: int, block_cnt: int, deadline: float):
    """DE on a piecewise‑constant representation (block_cnt blocks)."""
    block_len = n // block_cnt
    bounds = [(0.0, 1.0)] * block_cnt

    # we keep the objective tiny – only the max‑convolution matters
    def obj(params):
        params = np.asarray(params, dtype=np.float64)
        full = np.repeat(params, block_len)[:n]
        s = full.sum()
        if s == 0.0:
            return np.inf
        a = full / s
        conv = convolution_fft(a, a)
        return float(2.0 * n * conv.max())

    # adapt the number of generations to the remaining time
    approx_one_gen = 0.005  # rough seconds per generation on this machine
    max_generations = max(5, int((deadline - time.time()) / (approx_one_gen * 15)))
    result = differential_evolution(
        obj,
        bounds,
        strategy="best1bin",
        maxiter=max_generations,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=None,
        polish=False,
        init="latinhypercube",
        disp=False,
    )
    best_params = np.asarray(result.x, dtype=np.float64)
    best_full = np.repeat(best_params, block_len)[:n]
    best_a = best_full / best_full.sum()
    return best_a, compute_score(best_a)


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (Adam)
# ----------------------------------------------------------------------


def adam_optimize(start_a: np.ndarray,
                  gamma: float,
                  deadline: float,
                  lr0: float = 0.1,
                  max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max approximation of max(conv)."""
    n = start_a.shape[0]
    a = start_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    while t < max_iters and time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)

        # soft‑max weights (stable)
        max_c = conv.max()
        exp_vals = np.exp(gamma * (conv - max_c))
        w = exp_vals / exp_vals.sum()

        # gradient = 2 * (w reversed) * a   (full convolution, slice the centre)
        w_rev = w[::-1]
        grad_full = convolution_fft(w_rev, a)          # length = 3n‑2
        start = 2 * n - 2
        grad = 2.0 * grad_full[start:start + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam improved to {sc:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc

# Simple stochastic hill‑climb (simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_sc = compute_score(cur_a)

    best_a = cur_a.copy()
    best_sc = cur_sc

    T0 = best_sc * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        i, j = np.random.randint(n, size=2)
        if np.random.rand() < 0.5:               # swap two entries
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            delta = step_scale * cand[i]
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_sc = compute_score(cand)

        delta_sc = cand_sc - cur_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / (T + 1e-18)):
            cur_a, cur_sc = cand, cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_sc

    if fb is not None:

            fb_vec = [float(x) for x in fb]
            fb_sc = evaluate_sequence(fb_vec)
            if fb_sc < best_score:
                best_score = fb_sc
                best_seq = np.asarray(fb_vec, dtype=np.float64)
                print(f"[{time.time() - start_time:.1f}] fallback score = {fb_sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Determine problem size
    # ------------------------------------------------------------------
    n = best_seq.shape[0] if best_seq is not None and best_seq.shape[0] >= 1024 else 4096
    print(f"[{time.time() - start_time:.1f}] using length n={n}")

    # ------------------------------------------------------------------
    # 2. Quick scan of analytic families (cheap but diverse)
    # ------------------------------------------------------------------
    # Uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    sc = compute_score(uniform)
    if sc < best_score:
        best_score = sc
        best_seq = uniform.copy()
        print(f"[{time.time() - start_time:.1f}] uniform score = {sc:.6f}")

    # Product family
    alphas = np.concatenate([np.linspace(0.0, 5.0, 31), np.linspace(5.1, 200.0, 30)])
    for alpha in alphas:
        if time.time() >= deadline - 250.0:
            break
        cand = product_family(n, alpha)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.1f}] product α={alpha:.2f} score = {sc:.6f}")

    # Left‑heavy power family
    betas = np.linspace(0.0, 30.0, 31)
    for beta in betas:
        if time.time() >= deadline - 250.0:
            break
        cand = left_heavy_power_family(n, beta)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.1f}] left‑heavy β={beta:.2f} score = {sc:.6f}")

    # Exponential‑U family
    gammas = np.linspace(0.0, 30.0, 31)
    for gamma in gammas:
        if time.time() >= deadline - 250.0:
            break
        cand = exp_u_family(n, gamma)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.1f}] exp‑U γ={gamma:.2f} score = {sc:.6f}")

    # Gaussian family
    gauss_gammas = np.logspace(-4, 2, 30)
    for gamma in gauss_gammas:
        if time.time() >= deadline - 250.0:
            break
        cand = gaussian_family(n, gamma)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.1f}] gaussian γ={gamma:.5g} score = {sc:.6f}")

    # Cosine windows
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 250.0:
            break
        cand = window_family(n, win)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.1f}] window {win} score = {sc:.6f}")

    # Random starts (few, just to diversify)
    for _ in range(60):
        if time.time() >= deadline - 250.0:
            break
        if np.random.rand() < 0.5:
            cand = random_monotone_start(n)
        else:
            cand = random_u_start(n)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.1f}] random start score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 3. DE on a piecewise‑constant parametrisation (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        block_cnt = 64
        print(f"[{time.time() - start_time:.1f}] starting DE (blocks={block_cnt})")
        de_a, de_sc = de_optimize(n, block_cnt, deadline)
        if de_sc < best_score:
            best_score = de_sc
            best_seq = de_a.copy()
            print(f"[{time.time() - start_time:.1f}] DE improved to {de_sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max Adam optimisation (the main fine‑tuning phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        # start from the current best
        start_a = best_seq.copy()
        # gamma controls the softness of the max; we anneal it a little
        gammas = [20.0, 40.0, 80.0]
        for gamma in gammas:
            if time.time() >= deadline - 20.0:
                break
            print(f"[{time.time() - start_time:.1f}] Adam with γ={gamma:.1f}")
            adam_a, adam_sc = adam_optimize(start_a, gamma, deadline,
                                            lr0=0.12, max_iters=400_000)
            if adam_sc < best_score:
                best_score = adam_sc
                best_seq = adam_a.copy()
                start_a = adam_a.copy()
                print(f"[{time.time() - start_time:.1f}] Adam improved to {adam_sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Final short stochastic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (SA)")
        hc_a, hc_sc = hill_climb_sa(best_seq, deadline, max_steps=800_000)
        if hc_sc < best_score:
            best_score = hc_sc
            best_seq = hc_a.copy()
            print(f"[{time.time() - start_time:.1f}] hill‑climb improved to {hc_sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
