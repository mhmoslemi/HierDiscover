"""
propose_candidate
=================

A multi‑stage search for a non‑negative step‑function (a list of heights)
minimising

        2·n·max(convolve(a,a)) / (sum(a)²)

under the simplex constraint.  

The algorithm works as follows

1. Choose a moderate problem size (default n = 4096, larger if a very long
   fallback sequence is supplied).  Larger n gives a better theoretical bound
   while still allowing many optimisation steps within the 1000 s budget.

2. Initialise from a *rich* library of analytic families
   (arcsine‑type, left‑heavy power, exponential, mixtures with a uniform
   component) and from a few random starts.  The best of these is kept as the
   warm‑start.

3. **Adam on a smooth‑max surrogate** – minimise  
   fβ(a) = (1/β)·log Σk exp(β·(a*a)k)  with a schedule that drives β
   from a modest value up to 15000.  The parametrisation a = softmax(z) keeps
   the iterate on the probability simplex automatically.

4. **Projected sub‑gradient descent** on the true max‑convolution objective.
   This cheap iteration drives the worst‑convolution entry down further.

5. **Symmetrisation** – the problem is invariant under reversal, so the
   average of a and its reverse is tried.

6. **Simulated‑annealing hill‑climb** – a final random‑walk on the simplex
   with an exponential cooling schedule, accepting occasional uphill moves.

All phases respect the global wall‑clock budget (the harness passes the
budget in seconds).  The best sequence found is returned as a plain Python
list; its score is printed for logging.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # imported for completeness – not used here


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the evaluator (up to the irrelevant global scale).
    The vector `a` is assumed to be non‑negative and normalised to sum = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    α = 0.5 gives the discrete arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{-β} (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(‑γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_family(n: int, alpha: float, lam: float) -> np.ndarray:
    """Mixture (1‑λ)·arcsine(α) + λ·uniform."""
    base = symmetric_product_family(n, alpha)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = (1.0 - lam) * base + lam * uniform
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing shape (sorted power of a uniform sample)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑type shape."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

# Optimisers
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 4.0,
                         beta_end: float = 15000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth‑max surrogate
          fβ(a) = (1/β)·log Σk exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).  β is
    increased logarithmically from `beta_start` to `beta_end`.
    """
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise in log‑space

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (log‑linear)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max over the β‑scaled convolution values (stable)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # shape (2n‑1,)

        # gradient of the smoothed max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length n + (2n‑1) – 1 = 3n‑2, we need the middle n entries
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # chain rule through softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            cur_a = softmax(z)
            sc = compute_score(cur_a)
            if sc < best_score:
                best_score = sc
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    final_a = softmax(z)
    final_sc = compute_score(final_a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = final_a.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Very cheap per iteration, so many steps can be afforded.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry (k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]        # diagonal term appears only once

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad improved to {best_score:.6f}")

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# High‑level driver

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects a wall‑clock
    budget (default 1000 s) and returns the best list of heights it has
    discovered.
    """
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0                # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  If the harness supplies a fallback candidate, evaluate it.
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]    # type: ignore
        fb_score = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose problem size.
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # We aim for a size that balances quality and speed.
    n = max(4096, fallback_len)               # 4096 is a comfortable default
    print(f"[{time.time():.1f}] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Build a diverse set of analytic starters and keep the best.
    # ------------------------------------------------------------------
    starters = []

    # arcsine‑type and uniform mixtures
    alphas = np.linspace(0.35, 0.70, 8)        # focus around the classic 0.5
    lams   = np.linspace(0.0, 0.30, 5)        # uniform mixing proportion
    for alpha in alphas:
        for lam in lams:
            vec = mixed_family(n, alpha, lam)
            starters.append(vec)

    # left‑heavy power families
    betas = np.linspace(0.4, 2.0, 9)
    for beta in betas:
        starters.append(left_heavy_power_family(n, beta))

    # pure exponential distance families
    gammas = np.linspace(0.0, 10.0, 9)
    for gamma in gammas:
        starters.append(exp_u_family(n, gamma))

    # random monotone shapes
    for _ in range(4):
        starters.append(random_monotone_start(n))
        starters.append(random_u_start(n))

    # If a fallback exists, up‑sample it (simple repeat) to the chosen n
    if fallback_len > 0 and fallback_len < n:
        repeat = n // fallback_len
        up = np.repeat(np.array(best_seq, dtype=np.float64), repeat)
        # possibly pad a few extra zeros if needed
        if up.size < n:
            up = np.concatenate([up, np.zeros(n - up.size)])
        up = up[:n]
        up = up / up.sum()
        starters.append(up)

    # Evaluate starters
    for vec in starters:
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
    print(f"[{time.time():.1f}] best after analytic scan = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Normalise the current best candidate.
    # ------------------------------------------------------------------
    a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 4.  Time budgeting for the remaining phases.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0.0:
        final_seq = a.tolist()
        print(f"[{time.time():.1f}] time exhausted – returning current best")
        return final_seq

    # Split the remaining budget (roughly) among the three main phases.
    adam_frac = 0.35
    subg_frac = 0.35
    sa_frac   = 0.30

    adam_deadline = time.time() + adam_frac * remaining
    subg_deadline = time.time() + subg_frac * remaining
    sa_deadline   = time.time() + sa_frac * remaining

    # ------------------------------------------------------------------
    # 5.  Adam optimisation of the smooth‑max surrogate.
    # ------------------------------------------------------------------
    if time.time() < adam_deadline:
        print(f"[{time.time():.1f}] Adam phase (≈{adam_deadline - time.time():.1f}s)")
        a_adam, sc_adam = adam_smooth_optimize(
            a,
            adam_deadline,
            max_iters=200_000,
            beta_start=4.0,
            beta_end=15000.0,
            lr=0.018,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            a = a_adam
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Projected sub‑gradient descent (exact max‑convolution).
    # ------------------------------------------------------------------
    if time.time() < subg_deadline:
        print(f"[{time.time():.1f}] Sub‑gradient phase (≈{subg_deadline - time.time():.1f}s)")
        a_sub, sc_sub = subgrad_optimize(
            a,
            subg_deadline,
            max_iters=300_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            a = a_sub
            print(f"[{time.time():.1f}] Sub‑gradient improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Optional symmetrisation (the problem is symmetric).
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] Symmetrisation improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Final hill‑climb (simulated annealing).
    # ------------------------------------------------------------------
    if time.time() < sa_deadline:
        print(f"[{time.time():.1f}] Hill‑climb (SA) phase (≈{sa_deadline - time.time():.1f}s)")
        a_sa, sc_sa = hill_climb_sa(
            a,
            sa_deadline,
            max_steps=200_000,
        )
        if sc_sa < best_score:
            best_score = sc_sa
            a = a_sa
            print(f"[{time.time():.1f}] Hill‑climb improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final evaluation using the supplied `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
