def f_and_grad(z, beta):
    a = softmax(z)
    conv = convolution_fft(a, a)  # shape 2n-1
    # compute f_beta = (1/beta) * log(sum(exp(beta * conv)))
    maxc = conv.max()
    # stable
    exp_beta = np.exp(beta * (conv - maxc))
    sum_exp = exp_beta.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta
    # compute weights
    w = exp_beta / sum_exp
    # gradient w.r.t a: g_a = 2 * conv_rev[len(a)-1:len(a)-1+len(a)] where conv_rev = convolution_fft(a[::-1], w)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)
    # get gradient size n
    # conv_rev length = n + len(w) -1 = maybe 2n-1; Actually len(a_rev)=n, len(w)=2n-1, conv_rev length = n + (2n-1) - 1 = 3n-2.
    # But original code used conv_rev[len(a)-1:len(a)-1+len(a)] to get gradient of shape n.
    grad_a = 2.0 * conv_rev[n-1:n-1+n]
    # chain rule: grad_z = a * (grad_a - np.dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return f, grad_z

"""
Improved step‑function search.

The driver combines a quick analytic scan of the symmetric “product”
family, a few random starts and a high‑precision refinement with
L‑BFGS‑B on a smooth‑max surrogate of the true objective.
The surrogate is

    f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k),

whose gradient is obtained analytically.  By gradually increasing the
temperature β (50 → 200 → 1 000 → 5 000) we approach the true
max‑convolution while the second‑order method converges quickly.
The whole search respects the 1000 s wall‑clock budget,
tries several problem sizes (1024 … 8192) and several random
initialisations, and finally returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS callbacks (required – no closures)
# ----------------------------------------------------------------------
GLOBAL_BETA = 1.0
GLOBAL_DEADLINE = float("inf")
GLOBAL_BEST_SOFT = float("inf")
GLOBAL_BEST_SEQ = None


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the evaluator (sum of a is assumed 1).
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    α = 0.5 gives the discrete arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing shape, normalised to sum 1."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Exponentially decaying shape centred in the middle."""
    gamma = np.random.uniform(0.0, 12.0)
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------


def lbfgs_obj_and_grad(z: np.ndarray):
    """
    Objective + gradient for L‑BFGS‑B.
    Uses the global variable GLOBAL_BETA.
    Returns tuple (f, grad_z).
    """
    a = softmax(z)
    conv = convolution_fft(a, a)
    maxc = conv.max()
    beta = GLOBAL_BETA

    # smooth‑max value
    exp_beta = np.exp(beta * (conv - maxc))
    sum_exp = exp_beta.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # weights for the soft‑max
    w = exp_beta / sum_exp

    # gradient w.r.t. a  (2 * (a_rev * w) convolution, sliced)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)
    n = a.shape[0]
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

    # keep a global record of the best soft‑max value seen
    global GLOBAL_BEST_SOFT, GLOBAL_BEST_SEQ
    if f < GLOBAL_BEST_SOFT:
        GLOBAL_BEST_SOFT = f
        GLOBAL_BEST_SEQ = a.copy()

    return f, grad_z


def lbfgs_callback(_):
    """Abort the optimisation if we run out of time."""
    if time.time() >= GLOBAL_DEADLINE:
        raise KeyboardInterrupt


def refine_lbfgsb(a_start: np.ndarray,
                  beta_schedule,
                  max_iter_per_stage: int,
                  deadline: float) -> np.ndarray:
    """
    Refine `a_start` with a sequence of L‑BFGS‑B runs.
    `beta_schedule` is an iterable of β values (increasing).
    Each stage runs at most `max_iter_per_stage` iterations.
    """
    global GLOBAL_BETA, GLOBAL_DEADLINE, GLOBAL_BEST_SOFT, GLOBAL_BEST_SEQ
    a = a_start.copy()
    for beta in beta_schedule:
        GLOBAL_BETA = beta
        GLOBAL_DEADLINE = deadline
        GLOBAL_BEST_SOFT = float("inf")
        GLOBAL_BEST_SEQ = None

        z0 = np.log(a + 1e-12)

        # L‑BFGS‑B options – we keep them very light
        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(
                lbfgs_obj_and_grad,
                z0,
                method="L-BFGS-B",
                jac=True,
                callback=lbfgs_callback,
                options=opts,
            )
        except KeyboardInterrupt:
            # timed out – fall back to the best we saw inside callback
            if GLOBAL_BEST_SEQ is not None:
                a = GLOBAL_BEST_SEQ
            break

        # use the optimiser’s latest point (or the best we tracked)
        if GLOBAL_BEST_SEQ is not None and compute_score(GLOBAL_BEST_SEQ) < compute_score(res.x):
            a = GLOBAL_BEST_SEQ
        else:
            a = softmax(res.x)

        # keep the iterate normalised (softmax already does)
        # optional tiny symmetrisation to stay in the symmetric subspace
        a = (a + a[::-1]) * 0.5
        a /= a.sum()
        if time.time() >= deadline:
            break

    return a


# ----------------------------------------------------------------------
# Small simulated‑annealing hill‑climb (optional final polish)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 500_000) -> np.ndarray:
    """Very light SA that only moves mass between two positions."""
    n = start_a.size
    a = start_a.copy()
    cur_score = compute_score(a)
    best_a, best_score = a.copy(), cur_score

    T0 = cur_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        cand = a.copy()
        i, j = np.random.randint(n, size=2)
        if i != j:
            delta = step_scale * a[i]
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0.0:
                cand[i] = 0.0
        # re‑normalise
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        score = compute_score(cand)
        if score < cur_score or np.random.rand() < np.exp(-(score - cur_score) / (T + 1e-18)):
            a, cur_score = cand, score
            if score < best_score:
                best_a, best_score = cand.copy(), score
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97
        T = T0 * (1.0 - step / max_steps)
    return best_a
