def exponentiated_gradient_optimize(start_a, deadline, max_iters=500_000, eta_start=0.5, eta_end=0.005, sym_interval=1000, eps=1e-30):
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute conv
        conv = convolution_fft(a, a)
        maxc = conv.max()
        max_idx = int(np.argmax(conv))
        # compute subgradient g
        g = np.zeros_like(a)
        # valid i values compute as: i range [max(0, max_idx - (n-1)) .. min(n-1, max_idx)]
        i_min = max(0, max_idx - (a.shape[0] - 1))
        i_max = min(a.shape[0] - 1, max_idx)
        # for i in range(i_min, i_max+1):
        #   j = max_idx - i
        #   g[i] = 2*a[j]
        # This vectorized:
        idx_i = np.arange(i_min, i_max+1, dtype=np.int64)
        idx_j = max_idx - idx_i
        g[idx_i] = 2.0 * a[idx_j]  # using a array

        # Form eta schedule:
        eta = eta_start * (eta_end / eta_start) ** (t / max_iters)

        # EG update:
        a = a * np.exp(-eta * g)
        # maintain positivity:
        a = np.maximum(a, eps)  # avoid zero
        # renormalize:
        sum_a = a.sum()
        a /= sum_a

        # symmetrize at intervals
        if sym_interval and (t % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()  # re-normalize after sym

        # compute score after sym/per some frequency
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f} at iter {t}")

    # Return best_a
    return best_a, best_score

"""Propose a non‑negative step function minimizing the auto‑convolution peak.

The search proceeds in several phases:

1.  A quick scan of a few analytically‑parameterised families
    (product, power‑law, exponential‑U, mixtures, asymmetric product,
    product+U) on a modest size `n = 16384`.  The best parameter of each
    family is recorded.

2.  Local refinement of the most promising families with SciPy
    (Nelder‑Mead, golden‑section) when SciPy is available.

3.  A dedicated mirror‑descent optimiser on the simplex:
    exponentiated‑gradient (EG) updates using the exact sub‑gradient
    of the supremum of the convolution (the index where the auto‑
    convolution attains its maximum).  The learning‑rate is annealed
    from 0.2 to 0.001 and the iterate is periodically symmetrised
    (a ← (a+a[::-1])/2) to keep the distribution symmetric.
    Random jitter is injected every 50 k iterations to avoid stagnation.

4.  A final Adam optimisation on a smooth‑max surrogate (as in the
    previous state‑of‑the‑art) to polish the solution.

All phases respect a global wall‑clock budget of ~1000 s; the best
vector found (as a plain list of floats) is returned.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1] if np.any(cond) else 0
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric product family:
        a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Guard against overflow at the ends
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def product_expU_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    """Product family multiplied by an exponential‑U factor."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod - gamma * dist)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def asymmetric_product_family(n: int, beta_left: float, beta_right: float) -> np.ndarray:
    """Two‑parameter product family: a_i ∝ (i+½)^{‑β₁}(n‑½‑i)^{‑β₂}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta_left * np.log(i + 0.5) - beta_right * np.log(n - 0.5 - i))
    w /= w.sum()
    return w

def adam_smooth_optimize(start_a: np.ndarray,

                         beta_start: float = 10.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.015,
                         sym_interval: int = 2000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate of max(conv)."""
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth‑max of the convolution
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score

def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 600_000,
                eta_start: float = 0.2,
                eta_end: float = 0.001,
                sym_interval: int = 2000,
                jitter_interval: int = 50_000,
                jitter_scale: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with KL‑divergence (exponentiated‑gradient) on the exact
    max‑convolution objective.  The sub‑gradient of max(conv) is taken at the
    index where the convolution attains its maximum.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    eps = 1e-14

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- exact convolution & max index ---------------------------------
        conv = convolution_fft(a, a)
        maxc = conv.max()
        k = int(conv.argmax())

        # --- sub‑gradient of the supremum (single active index) -----------
        # g_i = 2 * a_{k-i} wherever the index is valid, else 0
        g = np.zeros_like(a)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        if i_min <= i_max:
            idx_i = np.arange(i_min, i_max + 1, dtype=np.int64)
            idx_j = k - idx_i
            g[idx_i] = 2.0 * a[idx_j]

        # --- learning‑rate schedule ---------------------------------------
        t_frac = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** t_frac

        # --- exponentiated‑gradient update ---------------------------------
        a = a * np.exp(-eta * g + eps)          # eps prevents 0⋅log0 later
        a = np.maximum(a, eps)                 # keep strictly positive
        a /= a.sum()

        # --- optional symmetrisation --------------------------------------
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a /= a.sum()

        # --- occasional jitter to escape plateaus -------------------------
        if jitter_interval and (it % jitter_interval == 0):
            a += np.random.normal(scale=jitter_scale, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

        # --- bookkeeping ---------------------------------------------------
        if it % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG iter {it} improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
        print(f"[{time.time():.1f}] EG final improvement to {best_score:.6f}")

    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> tuple[np.ndarray, float]:
    """Long random walk on the simplex with a tiny temperature schedule."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 20) -> tuple[float, float]:
    """Minimise `func` on [a,b] using golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

    sc = compute_score(vec)
    if sc < best_score:
        best_score = sc
        best_seq = vec.tolist()
        print(f"[{time.time():.1f}] new best (uniform) = {best_score:.6f}")

    # ----- symmetric product family ------------------------------------
    best_alpha = None
    best_alpha_sc = float('inf')
    for alpha in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.3f}) = {best_score:.6f}")
        if sc < best_alpha_sc:
            best_alpha_sc = sc
            best_alpha = alpha
    family_records['product'] = (best_alpha, best_alpha_sc)

    # ----- left‑heavy power law ----------------------------------------
    best_beta = None
    best_beta_sc = float('inf')
    for beta in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")
        if sc < best_beta_sc:
            best_beta_sc = sc
            best_beta = beta
    family_records['leftpower'] = (best_beta, best_beta_sc)

    # ----- exponential U shape -----------------------------------------
    best_gamma = None
    best_gamma_sc = float('inf')
    for gamma in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")
        if sc < best_gamma_sc:
            best_gamma_sc = sc
            best_gamma = gamma
    family_records['expU'] = (best_gamma, best_gamma_sc)

    # ----- mixture uniform + U -----------------------------------------
    best_mix = None
    best_mix_sc = float('inf')
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 8.0, 17):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")
            if sc < best_mix_sc:
                best_mix_sc = sc
                best_mix = (lam, gamma)
    family_records['mix'] = (best_mix, best_mix_sc)

    # ----- asymmetric product (two‑parameter grid) --------------------
    best_asym = None
    best_asym_sc = float('inf')
    beta_grid = np.linspace(0.0, 15.0, 13)
    for b1 in beta_grid:
        for b2 in beta_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = asymmetric_product_family(n, b1, b2)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (asym β1={b1:.2f},β2={b2:.2f}) = {best_score:.6f}")
            if sc < best_asym_sc:
                best_asym_sc = sc
                best_asym = (b1, b2)
    family_records['asym'] = (best_asym, best_asym_sc)

    # ----- product + exponential U (two‑parameter grid) ---------------
    best_pe = None
    best_pe_sc = float('inf')
    a_grid = np.linspace(0.0, 15.0, 13)
    g_grid = np.linspace(0.0, 8.0, 9)
    for a in a_grid:
        for g in g_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = product_expU_family(n, a, g)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (prod‑exp α={a:.2f},γ={g:.2f}) = {best_score:.6f}")
            if sc < best_pe_sc:
                best_pe_sc = sc
                best_pe = (a, g)
    family_records['prodexpU'] = (best_pe, best_pe_sc)

    # ------------------------------------------------------------------
    # 3. Local refinement of promising families (SciPy if available)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import minimize
        scipy_available = True
    except Exception:
        scipy_available = False

    if scipy_available and time.time() < deadline - 20.0:
        # product family 1‑D golden‑section refinement (already coarse, tighten)
        if family_records.get('product')[0] is not None:
            α0 = family_records['product'][0]
            lo = max(0.0, α0 - 0.8)
            hi = min(15.0, α0 + 0.8)

            def score_alpha(a):
                return compute_score(symmetric_product_family(n, a))

            α_opt, α_sc = golden_section_optimize(score_alpha, lo, hi, max_iter=25)
            vec = symmetric_product_family(n, α_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined product‑α={α_opt:.6f} → {best_score:.6f}")

        # mixture refinement (2‑D Nelder‑Mead)
        if family_records.get('mix')[0] is not None:
            lam0, gam0 = family_records['mix'][0]

            def score_mix(p):
                lam = np.clip(p[0], 0.0, 1.0)
                gam = np.clip(p[1], 0.0, 8.0)
                return compute_score(mixed_uniform_u_family(n, lam, gam))

            res = minimize(score_mix,
                           x0=np.array([lam0, gam0]),
                           method='Nelder-Mead',
                           options={'maxiter': 200, 'xatol': 1e-5, 'fatol': 1e-5})
            lam_opt, gam_opt = np.clip(res.x, [0.0, 0.0], [1.0, 8.0])
            vec = mixed_uniform_u_family(n, lam_opt, gam_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined mix (λ={lam_opt:.6f},γ={gam_opt:.6f}) → {best_score:.6f}")

        # product‑expU refinement (2‑D Nelder‑Mead)
        if family_records.get('prodexpU')[0] is not None:
            a0, g0 = family_records['prodexpU'][0]

            def score_pe(p):
                aa = np.clip(p[0], 0.0, 15.0)
                gg = np.clip(p[1], 0.0, 8.0)
                return compute_score(product_expU_family(n, aa, gg))

            res = minimize(score_pe,
                           x0=np.array([a0, g0]),
                           method='Nelder-Mead',
                           options={'maxiter': 250, 'xatol': 1e-5, 'fatol': 1e-5})
            a_opt, g_opt = np.clip(res.x, [0.0, 0.0], [15.0, 8.0])
            vec = product_expU_family(n, a_opt, g_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined prod‑exp (α={a_opt:.6f},γ={g_opt:.6f}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. First Adam (smooth‑max) pass (helps to escape shallow local minima)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] first Adam refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 150.0,
            max_iters=800_000,
            beta_start=10.0,
            beta_end=9000.0,
            lr=0.014,
            sym_interval=1500,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after first Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient optimisation on the exact max‑norm
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] EG optimisation start")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = eg_optimize(
            a_start,
            deadline - 100.0,
            max_iters=500_000,
            eta_start=0.2,
            eta_end=0.001,
            sym_interval=2500,
            jitter_interval=40_000,
            jitter_scale=5e-5,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after EG best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Second Adam pass (higher β, finer temperature)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] second Adam pass")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = adam_smooth_optimize(
            a_start,
            deadline - 30.0,
            max_iters=350_000,
            beta_start=50.0,
            beta_end=13000.0,
            lr=0.012,
            sym_interval=1000,
        )
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.tolist()
            print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Explicit symmetrisation test (may improve marginally)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final hill‑climb (simulated annealing) for polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=3_500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Fallback to a safe uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final evaluation through the official wrapper (includes safety checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
