def gradient_descent_smooth(a0, time_deadline, alpha_start=10, alpha_end=200, iter_max=...):
    a = a0.copy()
    best = a.copy()
    best_score = compute_score(best)
    t=0
    while time.time() < time_deadline:
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iters) # linear schedule; but we need max_iters predetermined maybe based on time.

        # compute conv
        conv = full_convolution(a)
        # compute softmax weights
        phi = alpha * conv
        # subtract max for numerical stability
        phi_max = phi.max()
        exp_phi = np.exp(phi - phi_max)  # stable
        w = exp_phi / exp_phi.sum()
        # compute gradient
        a_rev = a[::-1]
        conv_rev = full_convolution(a_rev, w)  # returns length 3n-2
        grad = 2.0 * conv_rev[n-1:2*n-1]
        # momentum: maintain v (velocity)
        # step size schedule: eta = eta0 / sqrt(t+1)
        eta = eta0 / np.sqrt(t + 1)
        # update with negative gradient
        a_new = a - eta * grad
        a_new = project_to_simplex(a_new)
        # optional jitter if stuck?
        # Evaluate periodically e.g., every 100 iterations
        if t % 50 == 0:
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best = a_new.copy()
        a = a_new
        t += 1
        if t > max_iters: break

    return best

"""
Efficient search for a low‑autocorrelation non‑negative step function.

The algorithm consists of three stages:
1.  Initialise a candidate (fallback, interpolated best, or random Dirichlet).
2.  Smooth‑max Adam descent:
    – The objective max(conv(a,a)) is approximated by a smooth log‑sum‑exp.
    – Gradients are obtained analytically via FFT‑based convolutions.
    – Adam with a linearly rising “sharpness’’ parameter (α) drives the
      sequence toward a flat autocorrelation.
3.  Stochastic hill‑climb on the simplex to polish the result.

All operations stay on the probability simplex (∑a = 1, a_i ≥ 0) so the
evaluation function is scale‑invariant.  The code respects the 1000 s
budget by checking the deadline frequently and falls back to a uniform
vector if nothing is found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolutions (FFT based)
# ----------------------------------------------------------------------
def conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def conv_generic(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary real vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum normalised to 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact evaluate_sequence score when sum(a)=1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser for the smooth‑max surrogate
# ----------------------------------------------------------------------
def adam_smooth_descent(
    a0: np.ndarray,
    deadline: float,
    alpha_start: float = 10.0,
    alpha_end: float = 200.0,
    eta: float = 0.04,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimiser on the smooth approximation
        L(a) = (1/α)·log Σ_k exp(α·conv_k)

    Returns the best vector seen during the run (projected onto simplex).
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Adam state
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        # linearly increase sharpness α
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max(1, int(0.5 * (deadline - time.time()))))
        # convolution a*a
        conv = conv_self(a)                # length m
        # soft‑max weights
        phi = alpha * conv
        phi_max = phi.max()
        exp_phi = np.exp(phi - phi_max)    # stabilise
        w = exp_phi / exp_phi.sum()        # shape (m,)

        # gradient of smooth max: 2·(a_rev * w) restricted to n entries
        a_rev = a[::-1]
        conv_rev = conv_generic(a_rev, w)  # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : 2 * n - 1]

        # Adam update
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation (cheap compared to full run)
        if t % 120 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[adam] iter {t:6d}, exact score {val:.6f}")

            # early stop if we already beat the target
            if best_val < 1.503:
                print("[adam] target reached, exiting early")
                break

    return best_a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb on the simplex
# ----------------------------------------------------------------------
def hill_climb(
    a_start: np.ndarray,
    deadline: float,
    max_iter: int = 1_200_000,
) -> np.ndarray:
    """Greedy stochastic mass‑transfer moves with adaptive step scale."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = conv_self(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = conv_self(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.04, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1

        if no_improve > 180_000:
            # jump out of a plateau
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Helper – up‑sample a shorter vector to a longer length
# ----------------------------------------------------------------------
def upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger size, followed by simplex projection."""
    old_n = seq.shape[0]
    if new_n == old_n:
        return seq.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n)
    interp = np.interp(x_new, x_old, seq)
    return project_to_simplex(interp)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  The strategy is:

    * start from a known good construction (or a uniform vector),
    * run a smooth‑max Adam optimiser on the full sequence,
    * finish with a stochastic hill‑climb for polishing,
    * respect the time budget (≈ 1000 s) and return a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 0.  Load fallback if available
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Initialise a candidate (uniform if nothing else)
    # --------------------------------------------------------------
    if best_seq is None:
        n0 = 2048
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)
        print(f"[init] uniform baseline score = {best_val:.6f}")

    # work with numpy array, ensure sum = 1
    a_cur = np.array(best_seq, dtype=float)
    a_cur = project_to_simplex(a_cur)

    # --------------------------------------------------------------
    # 2.  Optionally enlarge the dimension (more freedom → lower bound)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    target_n = a_cur.shape[0]
    if remaining > 250.0 and target_n < 4096:
        target_n = 4096
    if target_n != a_cur.shape[0]:
        a_cur = upscale(a_cur, target_n)
        print(f"[resize] up‑scaled to n = {target_n}")

    # --------------------------------------------------------------
    # 3.  Run a few Adam‑descent restarts (smooth‑max)
    # --------------------------------------------------------------
    gd_deadline = deadline - 60.0          # reserve ~1 min for final hill‑climb
    n = a_cur.shape[0]
    print(f"[stage‑1] Adam smooth‑max start, n = {n}")

    # keep the best found during this phase
    best_a = a_cur.copy()
    best_score = compute_score(best_a)

    # one warm start from the current best
    a_opt = adam_smooth_descent(best_a, gd_deadline)
    score_opt = compute_score(a_opt)
    if score_opt < best_score:
        best_score, best_a = score_opt, a_opt.copy()
        print(f"[stage‑1] improved to {score_opt:.6f}")

    # additional random restarts if time permits
    while time.time() < gd_deadline - 5.0:
        # random Dirichlet‑like start
        rand_vec = np.random.rand(n)
        rand_vec = project_to_simplex(rand_vec)
        a_opt = adam_smooth_descent(rand_vec, gd_deadline)
        score_opt = compute_score(a_opt)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[stage‑1‑rnd] new best {score_opt:.6f}")
            # if we already beat the required threshold we can stop early
            if best_score < 1.503:
                break

    # --------------------------------------------------------------
    # 4.  Final stochastic hill‑climb (fine‑grained polishing)
    # --------------------------------------------------------------
    hill_deadline = deadline - 5.0
    print("[stage‑2] hill‑climb start")
    a_final = hill_climb(best_a, hill_deadline)
    final_score = compute_score(a_final)
    if final_score < best_score:
        best_score, best_a = final_score, a_final.copy()
        print(f"[stage‑2] improved to {final_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Return the best vector discovered (as plain Python list)
    # --------------------------------------------------------------
    best_seq = best_a.tolist()
    # final official evaluation (may clip, enforce sum≥0.01 etc.)
    reported = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {reported:.6f}")
    return best_seq
