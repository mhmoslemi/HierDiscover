"""
Non‑negative step‑function optimiser.

The core idea is a **linear‑programming expansion**:  
starting from a feasible vector f (L1‑norm = 1) we compute  
M = max(conv(f,f)).  The constraints conv(f,g) ≤ M are linear in g,
so we can maximise ‖g‖₁ subject to them.  After normalising g back to
L1‑norm = 1 we either accept g (if it improves the true objective) or
accept a convex combination (t = ½) of f and g.  Repeating this step
while time permits reduces the maximal autocorrelation.

The best vector obtained from the LP stage is then polished by the
previously successful components:
* smooth‑max gradient descent (a soft‑max approximation of the max‑norm),
* projected sub‑gradient polishing,
* a short simulated‑annealing hill‑climb.

All phases respect the 1000 s wall‑clock budget and the final sequence
is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Pre‑computed index pattern for the convolution matrix
# ----------------------------------------------------------------------


def _conv_matrix_pattern(n: int):
    """
    Return three 1‑D arrays (row, col, idx) describing the non‑zero pattern
    of the Toeplitz matrix representing the linear map g ↦ conv(f,g).
    For a concrete f the data vector is simply f[idx].
    """
    total_nnz = n * n                     # total number of non‑zeros
    row = np.empty(total_nnz, dtype=np.int32)
    col = np.empty(total_nnz, dtype=np.int32)

    pos = 0
    for k in range(2 * n - 1):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        length = end - start + 1
        row[pos:pos + length] = k
        col[pos:pos + length] = np.arange(start, end + 1, dtype=np.int32)
        pos += length

    idx = row - col                      # index into f for each non‑zero
    return row, col, idx


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def smooth_max_optim(a0: np.ndarray, deadline: float,
                     tau_start: float = 0.30,
                     tau_end: float = 1e-5,
                     tau_decay: float = 0.98,
                     step0: float = 0.20) -> np.ndarray:
    """Minimise a smooth approximation of max(conv) using an
    FFT‑based gradient and Euclidean projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    while time.time() < deadline:
        if t % 100 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)            # length 2n‑1
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()         # soft‑max weights

        # gradient = 2 * (a ⋆ w)  (cross‑correlation)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]

        idx = (2 * n - 2) - np.arange(n)      # positions of the needed slice
        grad = 2.0 * conv_full[idx]

        step = step0 / np.sqrt(t + 1.0)
        a = a - step * grad
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 1500 == 0 and t > 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t + 1.0))
            a = a + jitter
            a = project_to_simplex(a)

        t += 1

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx_range = np.arange(n)
        for k in active:
            j = k - idx_range
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (stochastic hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

        M = conv_f.max()

        # linear programme: maximise 1ᵀg  ⇔ minimise –1ᵀg

        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

            new_score = compute_score(new_vec)

                print(f"[LP {lp_iter}] combo improvement: {best_score:.6f}")
