"""
Propose a low‑score step‑function using:
    * a quick parametric sweep for good seeds,
    * Adam optimisation on a soft‑max (log‑sum‑exp) surrogate of
      max(conv(a,a)), with projection onto the probability simplex,
    * a final stochastic hill‑climb.
The code works for any length; we use n≈4096 (more degrees of freedom
than the previous 2048) while keeping per‑iteration cost modest by using
FFT‑based convolutions.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, all entries ≥ 0)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full(n, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based (full) convolutions
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())      # next power of two ≥ m
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[:m]
    return conv


def full_convolution_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two arbitrary 1‑D arrays using FFT."""
    n, m = a.shape[0], b.shape[0]
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a normalised vector (Σa = 1):
#   2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors (used in the quick parametric sweep)
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Smoothed surrogate of max(conv) and its gradient.
# We use the log‑sum‑exp (soft‑max) with a temperature α.
# ----------------------------------------------------------------------
def smooth_gradient(a: np.ndarray, alpha: float):
    """
    Returns the gradient of   f_α(a) = (1/α)·log( Σ_k exp(α·(a*a)_k) )
    with respect to a.  The function is smooth for any α>0.
    """
    conv = full_convolution(a)
    cmax = conv.max()
    # soft‑max weights (stable)
    exp_term = np.exp(alpha * (conv - cmax))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.full_like(exp_term, 1.0 / exp_term.size)
    else:
        weights = exp_term / w_sum

    # gradient = 2 · (a ⋆ weights)   (⋆ = cross‑correlation)
    # implement via convolution with reversed weights
    w_rev = weights[::-1]
    conv_a_wrev = full_convolution_general(a, w_rev)
    m = len(weights)                      # = 2·n‑1
    g = 2.0 * conv_a_wrev[m - 1 : m - 1 - len(a) : -1]
    return g, cmax  # cmax is the (unsmoothed) max for monitoring

# Adam optimiser on the smoothed objective.
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray, deadline: float,
                  max_iter: int = 200_000,
                  lr: float = 0.1,
                  alpha_start: float = 10.0,
                  alpha_end: float = 2000.0):
    """
    Projected Adam on the soft‑max surrogate.
    Returns the best vector (in terms of the exact score) found during the run.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # linear schedule for the soft‑max temperature
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)

        grad, cur_max = smooth_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # occasional exact evaluation to catch improvements
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[Adam] iter {t:,}  α={alpha:.1f}  score={sc:.6f}")

    # final check
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score, best_a = final_sc, a.copy()
        print(f"[Adam] final  score={final_sc:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (aggressive variant)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  The routine combines a quick parametric
    scan, an Adam optimiser on a smoothed max‑convolution objective,
    and a final stochastic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float('inf')

    # -----------------------------------------------------------------
    # 0. Fallback – use a previously discovered construction if present
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        sc = compute_score(np.array(fallback, dtype=float))
        print(f"[fallback] score = {sc:.6f}")
        if sc < best_score:
            best_score, best_seq = sc, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Quick parametric sweep to obtain a solid seed
    # -----------------------------------------------------------------
    n = 4096
    if time.time() < deadline:
        vec, sc = parametric_sweep_once(n, deadline)
        if vec is not None and sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[sweep] best score = {sc:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam optimisation (major chunk of the remaining budget)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        a0 = np.array(best_seq, dtype=float)
        # allocate a generous number of iterations based on remaining time
        remaining = deadline - time.time()
        max_iter = int(remaining * 200)          # rough 200 iters / sec
        max_iter = max(max_iter, 30_000)        # ensure at least modest work
        print(f"[Adam] starting with max_iter={max_iter:,}")
        a_opt = adam_optimize(a0, deadline, max_iter=max_iter)
        sc = compute_score(a_opt)
        if sc < best_score:
            best_score, best_seq = sc, a_opt.tolist()
            print(f"[Adam] improved score = {sc:.6f}")

        # occasional random restarts while time permits
        while time.time() < deadline - 20.0:
            a0 = np.random.rand(n)
            a0 /= a0.sum()
            remain = deadline - time.time()
            iters = int(remain * 150)
            iters = max(iters, 5_000)
            a_restart = adam_optimize(a0, deadline, max_iter=iters,
                                      lr=0.12, alpha_start=5.0, alpha_end=1500.0)
            sc = compute_score(a_restart)
            if sc < best_score:
                best_score, best_seq = sc, a_restart.tolist()
                print(f"[Adam‑restart] new best score {sc:.6f}")

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, deadline)
        sc = compute_score(a_final)
        if sc < best_score:
            best_score, best_seq = sc, a_final.tolist()
            print(f"[hill‑climb] final best score = {sc:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing else succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.full(n, 1.0 / n, dtype=float))

    # -----------------------------------------------------------------
    # Final report using the official evaluate_sequence (includes clamping)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
