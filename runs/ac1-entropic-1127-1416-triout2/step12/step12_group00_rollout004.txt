    shift = conv.max()

    # Update tau gradually (maybe every 50 steps)
    if t % 50 == 0 and tau > tau_end:
        tau *= tau_decay
        tau = max(tau, tau_end)

    # compute softmax weights w

    exp_vals = np.exp((conv - shift) / tau)

    w = exp_vals / exp_vals.sum()

    # compute gradient

    rev_w = w[::-1]

    size = 1 << ((n + len(rev_w) - 1).bit_length())   # n + m -1

    A = np.fft.rfft(a, n=size)

    B = np.fft.rfft(rev_w, n=size)

    conv_full = np.fft.irfft(A * B, n=size)

    idx = (len(conv) - 1) - np.arange(n)

    grad = 2.0 * conv_full[idx]

    # Adam update

    t += 1

    m = beta1 * m + (1 - beta1) * grad

    m_hat = m / (1 - beta1 ** t)

    a = a - step

    a = project_to_simplex(a)

    # track best

    cur_val = compute_score(a)

    if cur_val < best_val: update best

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
1. **Broad parametric sweep** – generate candidates from several
   families (power‑law, exponential, beta, Gaussian, sine‑power) and a
   large set of random Dirichlet draws.  The best few are retained.

2. **Adam on a smooth‐max surrogate** – starting from each retained
   candidate we minimise a temperature‑annealed soft‑max
   approximation of the max‑convolution using the Adam optimiser.
   The gradient is obtained with one FFT‑based convolution per
   iteration and the iterate is projected onto the probability simplex.

3. **Projected sub‑gradient polishing** – a short sub‑gradient
   descent on the exact objective (the max of the convolution) removes
   the remaining bias of the smooth surrogate.

All phases respect a 1000 s wall‑clock budget; the best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())     # power of two ≥ m
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (assuming sum(a) == 1):
        2 * n * max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=float)) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (discrete Beta shape)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i‑mid)²/(2σ²))."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Adam optimiser for the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 0.30,
                  tau_end: float = 5e-7,
                  tau_decay: float = 0.995,
                  lr: float = 0.15,
                  beta1: float = 0.90,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """
    Minimise the smooth‑max approximation of max(conv(a,a))
    using Adam.  ``a0`` is assumed to be on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    best_a = a.copy()
    best_val = compute_score(a)

    # Adam state
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    t = 0
    tau = tau_start

    while time.time() < deadline:
        # anneal the temperature
        if t % 30 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # convolution a*a
        conv = full_convolution(a)
        shift = conv.max()

        # soft‑max weights
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient of the smooth‑max (2 * a ★ w)
        rev_w = w[::-1]
        size = 1 << ((3 * n - 3).bit_length())   # >= n + m - 1
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)

        idx = (m - 1) - np.arange(n)          # indices producing the gradient
        grad = 2.0 * conv_full[idx]

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)

        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = 2.0 * n * shift          # because sum(a)==1
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to inject perturbations
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (lr * 0.004 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exact max‑convolution)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient accumulation
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t + 1.0)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a
