"""
Proposal: asymmetric decreasing exponential / power‐law step functions.

The search proceeds in three phases (all respecting a 1000 s wall‑clock budget).

1️⃣  Analytic families – quick grid scans over:
      • uniform baseline,
      • left‑heavy power law a_i ∝ (i+½)^‑β  (β up to 30),
      • monotone exponential a_i ∝ exp(‑α·i) (α up to 5).
   The best parameters are refined with a 1‑D golden‑section search.

2️⃣  Adam smooth‑max optimisation – starting from the best analytic vector.
    The parameters are kept as log‑probabilities (soft‑max) so that the sum
    constraint is always satisfied.  No symmetrisation is applied (asymmetry
    helps to push the peak of the convolution away from the centre).  The
    temperature β for the smooth‑max is annealed from 10 to 10 000.

3️⃣  Projected sub‑gradient descent (exact max‑objective) and a short
    simulated‑annealing hill‑climb for final polishing.

The routine returns a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) for a probability vector (∑a = 1).
    Smaller is better.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # fallback to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def uniform_family(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def decreasing_exponential_family(n: int, alpha: float) -> np.ndarray:
    """
    Exponential decay: a_i ∝ exp(‑α·i), i = 0,…,n‑1.
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-alpha * i)
    w_sum = w.sum()
    # when alpha is 0 the vector is uniform; avoid division by zero
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w /= w_sum
    return w


def mixed_uniform_exponential_family(n: int, lam: float, alpha: float) -> np.ndarray:
    """
    Convex mixture of a uniform vector and a decreasing exponential.
    lam ∈ [0,1] is the weight on the uniform part.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    exp_vec = decreasing_exponential_family(n, alpha)
    w = lam * uniform + (1.0 - lam) * exp_vec
    w /= w.sum()
    return w

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 900_000,
                         beta_start: float = 10.0,
                         beta_end: float = 10_000.0,
                         lr: float = 0.018,
                         sym_interval: int = 0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    start_a : np.ndarray, sum(start_a) == 1
    deadline : float, Unix timestamp after which optimisation must stop.
    sym_interval : if >0 the iterate is symmetrised every ``sym_interval``
                   steps; set to 0 to keep the vector asymmetric.

    Returns
    -------
    best_a, best_score
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise log‑probabilities
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # optional symmetrisation (disabled by default)
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smoothed max via log‑sum‑exp
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()          # weights that approximate the arg‑max

        # gradient of the smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # convert to gradient w.r.t. logits (softmax parametrisation)
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 6000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.042) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of (a*a)[k] w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_500_000) -> tuple[np.ndarray, float]:
    """
    Very simple simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.13

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.6:               # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        else:                                    # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 18) -> tuple[float, float]:
    """
    Minimises ``func`` on [a,b] using golden‑section search.
    Returns (best_parameter, best_value).
    """
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

        beta_opt, beta_sc = golden_section_optimize(score_beta, lo, hi)
        vec = left_heavy_power_family(n, beta_opt)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] refined β={beta_opt:.6f} → {best_score:.6f}")

    # refine exponential α
    if best_alpha is not None and time.time() < deadline - 20.0:
        lo = max(0.0, best_alpha - 0.5)
        hi = best_alpha + 0.5

        def score_alpha(a):
            return compute_score(decreasing_exponential_family(n, a))

        alpha_opt, alpha_sc = golden_section_optimize(score_alpha, lo, hi)
        vec = decreasing_exponential_family(n, alpha_opt)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] refined α={alpha_opt:.6f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. First Adam optimisation (asymmetric – no symmetrisation)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] starting Adam optimisation")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 120.0,
            max_iters=800_000,
            beta_start=10.0,
            beta_end=10_000.0,
            lr=0.018,
            sym_interval=0,          # keep asymmetry
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Projected sub‑gradient descent (exact max)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 80.0,
            max_iters=1_500_000,
            step0=0.042,
        )
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Optional final hill‑climb (simulated annealing)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_500_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Safety net – fallback to uniform if necessary
    # -----------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq))

    # final check with the official wrapper (handles clipping etc.)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
