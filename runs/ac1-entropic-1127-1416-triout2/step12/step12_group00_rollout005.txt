"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
1. **Broad exploration** – a richer family of parametric shapes
   (power‑law, exponential, beta, Gaussian, sinusoidal, Hann/Hamming,
   triangular, linear ramps) plus many random sparse supports.
   The best few candidates are retained.

2. **Multi‑restart smoothed‑max descent** – each retained candidate is
   refined by a soft‑max (log‑sum‑exp) approximation of the max‑convolution.
   The gradient is computed by a single FFT and the iterate is projected
   onto the probability simplex.  Several restarts with different
   hyper‑parameters increase the chance of escaping shallow minima.

3. **Intensive sub‑gradient polishing** – a classic projected sub‑gradient
   method (active‑set) is run for a longer period to tighten the active
   constraints.

4. **Stochastic hill‑climb** – a final simulated‑annealing phase gives a
   last opportunity to improve the solution.

All phases respect the 1000 s wall‑clock budget; the best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for efficient FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector with sum=1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_hann(n: int) -> np.ndarray:
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    return a / a.sum()


def generate_hamming(n: int) -> np.ndarray:
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    return a / a.sum()


def generate_blackman(n: int) -> np.ndarray:
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * np.arange(n) / (n - 1))
    )
    return a / a.sum()


def generate_linear_ramp(n: int, up: bool = True) -> np.ndarray:
    if up:
        a = np.arange(1, n + 1, dtype=float)
    else:
        a = np.arange(n, 0, -1, dtype=float)
    return a / a.sum()


def generate_triangular(n: int) -> np.ndarray:
    a = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    return a / a.sum()


def generate_random_sparse(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on `m` distinct random indices."""
    if m > n:
        m = n
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Baseline global search
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 12,
    random_draws: int = 5000,
) -> list[np.ndarray]:
    """
    Generate a diverse set of candidates and keep the best ``max_keep``.
    """
    best = []  # list of (score, vector)

    def maybe_keep(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    rng = np.random.default_rng(0)

    # 1) Power‑law sweep
    for p in np.linspace(0.05, 2.5, 81):
        if time.time() >= deadline:
            break
        maybe_keep(generate_powerlaw(n, p))

    # 2) Exponential sweep
    for lam in np.linspace(0.0005, 0.3, 200):
        if time.time() >= deadline:
            break
        maybe_keep(generate_exponential(n, lam))

    # 3) Beta sweep (dense grid, including heavy‑tailed shapes)
    al_vals = np.concatenate(
        (np.logspace(-1.2, -0.1, 12), np.linspace(0.2, 4.0, 20), np.logspace(0.2, 0.9, 10))
    )
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            maybe_keep(generate_beta(n, alpha, beta))

    # 4) Gaussian sweep
    for sigma in np.linspace(0.05 * n, 0.45 * n, 30):
        if time.time() >= deadline:
            break
        maybe_keep(generate_gaussian(n, sigma))

    # 5) Sine‑power sweep
    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        maybe_keep(generate_sine_power(n, p))

    # 6) Window families
    maybe_keep(generate_hann(n))
    maybe_keep(generate_hamming(n))
    maybe_keep(generate_blackman(n))
    maybe_keep(generate_linear_ramp(n, up=True))
    maybe_keep(generate_linear_ramp(n, up=False))
    maybe_keep(generate_triangular(n))

    # 7) Random sparse supports
    for m in (40, 50, 60, 70, 80, 90):
        draws = random_draws // len([40, 50, 60, 70, 80, 90])
        for _ in range(draws):
            if time.time() >= deadline:
                break
            maybe_keep(generate_random_sparse(n, m, rng))

    # 8) Pure random Dirichlet draws
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    for conc in concs:
        for _ in range(random_draws // len(concs)):
            if time.time() >= deadline:
                break
            a = rng.dirichlet([conc] * n)
            maybe_keep(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (log‑sum‑exp)
# ----------------------------------------------------------------------


def smooth_max_optim(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.5,
    tau_end: float = 1e-6,
    tau_decay: float = 0.99,
    step0: float = 0.25,
) -> np.ndarray:
    """Minimise a smooth approximation of max(conv) using FFT‑based gradient."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    while time.time() < deadline:
        if t % 120 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)          # length m
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()       # soft‑max weights

        # gradient = 2 * (a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]

        idx = (m - 1) - np.arange(n)        # map back to original coordinates
        grad = 2.0 * conv_full[idx]

        step = step0 / np.sqrt(t + 1.0)
        a -= step * grad
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape flat regions
        if t % 2000 == 0 and t > 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t + 1.0))
            a += jitter
            a = project_to_simplex(a)

        t += 1

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing
# ----------------------------------------------------------------------


def subgradient_polish(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.35,
) -> np.ndarray:
    """Classic projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # accumulate sub‑gradient from all active indices
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

# Simulated annealing (stochastic hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 1e-3,
    final_temp: float = 1e-6,
    cooling_factor: float = 0.9993,
    step_scale_init: float = 0.06,
    step_scale_min: float = 1e-5,
) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a
