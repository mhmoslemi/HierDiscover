"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

The search consists of the following phases (all respect the global 1000 s budget):

0.  Use a fallback sequence if one is supplied.
1.  Scan a variety of analytic families (product, power‑law, exponential,
    Gaussian, cosine windows) and many random Dirichlet‑like starts.
2.  Perform a continuation optimisation of a *smooth* max‑objective.
    The max is replaced by a log‑sum‑exp with a gradually increasing
    temperature λ.  Optimisation is done on the soft‑max logits with Adam,
    while staying on the probability simplex.
3.  Cheap projected sub‑gradient descent on the true max‑objective.
4.  A few LP‑based refinement steps (the classic “mix with a better
    partner” iteration).
5.  Final stochastic hill‑climb (simulated‑annealing style).

The best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast for repeated use)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped, ends heavy: w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone, heavy left side: w_i ∝ (i+0.5)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian: w_i ∝ exp(-γ·(i‑mid)²)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, window_type: str) -> np.ndarray:
    """Standard cosine windows (Hann, Hamming, Blackman)."""
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing (left‑heavy) vector."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped start (exponential decay from centre)."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed max optimisation (continuation + Adam on logits)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(a: np.ndarray, lam: float):
    """
    Returns the smoothed max value
        (1/lam)·log∑exp(lam·conv)
    and the gradient w.r.t. the *logits* y (where a = softmax(y)).
    """
    # a already sums to 1 and is non‑negative
    conv = convolution_fft(a, a)                     # length 2n‑1
    c = lam * conv
    c_max = c.max()
    exp_c = np.exp(c - c_max)                       # stability
    p = exp_c / exp_c.sum()                         # softmax weights over the convolution
    # gradient w.r.t. a : g = 2 * (p ⋆ a_rev) restricted to n entries
    a_rev = a[::-1]
    conv_grad_full = convolution_fft(p, a_rev)      # length 3n‑2
    n = a.shape[0]
    grad_a = 2.0 * conv_grad_full[n - 1 : 2 * n - 1]  # slice of length n
    # transform to gradient w.r.t. logits y (softmax Jacobian)
    s = np.dot(a, grad_a)                           # scalar ⟨a,∇_a f⟩
    grad_y = a * (grad_a - s)
    # smoothed max value
    smooth_max = (np.log(exp_c.sum()) + c_max) / lam
    return smooth_max, grad_y


def smoothed_optimize(start_a: np.ndarray,
                      deadline: float,
                      lam_schedule,
                      max_iters_per_lam: int = 8000):
    """
    Continuation optimiser: gradually increase λ (temperature) while running Adam
    on the logits y of the soft‑max parameterisation a = softmax(y).
    Returns the best vector found (probability vector) and its true score.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    y = np.log(a + 1e-16)                 # initialise logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 1
    best_a = a.copy()
    best_score = compute_score(a)

    for lam in lam_schedule:
        # heuristic base step size that shrinks with √λ
        step_base = 0.06 / math.sqrt(lam)
        for _ in range(max_iters_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad_y = smooth_obj_and_grad(a, lam)
            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y -= step_base * m_hat / (np.sqrt(v_hat) + eps)
            a = softmax_logits(y)          # back to simplex
            if t % 2000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time():.1f}] smoothed (λ={lam:.1f}) improved to {best_score:.6f}"
                    )
            t += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheap long tail)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method on the true max‑objective (200 % faster than EG for long tails).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (the “state‑of‑the‑art” mixing step)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,   g ≥ 0.
    Return the convex mixture (1‑t)·f + t·ĝ where ĝ = g/Σg.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A such that (A @ g) = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)            # maximise Σg → minimise −Σg
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # HiGHS (fast) linear programme
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new

# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves (classical SA).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
