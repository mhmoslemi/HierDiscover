def project_onto_simplex(v):
    # Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def softmax_weights(conv, beta):
    maxc = conv.max()
    # stable
    exp_term = np.exp(beta * (conv - maxc))
    sum_exp = exp_term.sum()
    if sum_exp == 0 or not np.isfinite(sum_exp):
        # fallback to uniform
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exp_term / sum_exp

def compute_smooth_gradient(a, beta):
    conv = full_convolution_fft(a)
    w = softmax_weights(conv, beta)  # length 2n-1
    # compute gradient: g = 2 * sum_k w_k * a_{k-i}
    # Use convolution with reversed w.
    w_rev = w[::-1]
    # conv_rev = np.convolve(a, w_rev)
    # use FFT for speed
    n = len(a)
    m = len(w_rev)
    L = n + m - 1
    n_fft = 1 << ((L - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    W = np.fft.rfft(w_rev, n=n_fft)
    conv_rev = np.fft.irfft(A * W, n=n_fft)[:L]
    idx = 2 * n - 2 - np.arange(n)
    g = 2.0 * conv_rev[idx]
    # Return gradient and max conv for evaluation
    return g, conv.max()

def adam_update(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    # ensure positivity
    a_new = np.maximum(a_new, 0.0)
    # project onto simplex
    a_proj = project_onto_simplex(a_new)
    return a_proj, m, v

def adam_polish(a0, deadline, beta_start=10.0, beta_factor=1.02, lr_start=0.5, lr_decay=0.9995, max_iters=100000):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    beta = beta_start
    lr = lr_start

    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max = compute_smooth_gradient(a, beta)
        # step
        a, m, v = adam_update(a, grad, m, v, it, lr)
        # evaluate real objective
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # schedule updates
        beta = beta * beta_factor
        lr = lr * lr_decay
        # occasional jitter to escape local minima
        if it % 500 == 0:
            # jitter small random multiplicative noise
            jitter = np.random.randn(n) * (0.01 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()
    return best_a

def compute_smooth_gradient(a, beta):
    n = len(a)
    # conv = a*a
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    # compute rfft of a.
    A = np.fft.rfft(a, n=n_fft)
    # conv = irfft(A * A)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]
    maxc = conv.max()
    # weight
    # compute w = softmax(beta*(conv - maxc))
    # Use stable exp.
    scaled = beta * (conv - maxc)
    # to avoid overflow, clamp scaled maybe between -700 and 700.
    # Use np.exp(np.clip(scaled, -700, 700))
    exp_scaled = np.exp(np.clip(scaled, -700, 700))
    sum_exp = exp_scaled.sum()
    if sum_exp == 0 or not np.isfinite(sum_exp):
        w = np.full_like(conv, 1.0 / conv.shape[0])
    else:
        w = exp_scaled / sum_exp

    # compute gradient: g = 2 * (a * w_rev) conv at appropriate indices.
    w_rev = w[::-1]
    # compute rfft of w_rev at same length n_fft (need same size for convolution)
    W = np.fft.rfft(w_rev, n=n_fft)
    conv_rev = np.fft.irfft(A * W, n=n_fft)[:(n + m - 1)]  # length = n + m -1 = n + 2n-1 - 1 = 3n-2
    # extract g elements
    idx = 2 * n - 2 - np.arange(n)
    g = 2.0 * conv_rev[idx.astype(int)]
    return g, maxc

def project_onto_simplex(v):
    # Works for arbitrary dimension n
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def compute_smooth_gradient(a, beta):
    n = len(a)
    # Compute conv
    conv = np.convolve(a, a)  # full length 2n-1
    maxc = conv.max()
    # Softmax weights
    scaled = beta * (conv - maxc)
    exp_scaled = np.exp(np.clip(scaled, -700, 700))
    sum_exp = exp_scaled.sum()
    if sum_exp == 0 or not np.isfinite(sum_exp):
        w = np.full_like(conv, 1.0 / conv.shape[0])
    else:
        w = exp_scaled / sum_exp

    # Compute gradient: g_i = 2 * sum_j w_{i+j} * a_j
    # Equivalent to correlation of a with w; we can use np.convolve with reversed w.
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)  # full length n + (2n-1) -1 = 3n-2
    idx = 2 * n - 2 - np.arange(n)  # indices descending
    g = 2.0 * conv_rev[idx]
    return g, maxc

def project_onto_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum x = 1}."""
    # Duchi et al. algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Ensure sum is exactly 1 (numerical fix)
    if w.sum() > 0:
        w = w / w.sum()
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def generate_powerlaw(n, p):
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total

def generate_exponential(n, lam):
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total

def generate_beta(n, alpha, beta):
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total

def generate_triangular(n):
    a = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    return a / a.sum()

def generate_cosine_shape(n):
    # cosine positive half-cycle
    theta = np.pi * (np.arange(n) - (n - 1) / 2) / n
    a = np.cos(theta)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a

def generate_gaussian(n, sigma):
    i = np.arange(n)
    mu = (n - 1) / 2.0
    a = np.exp(- ((i - mu) ** 2) / (2.0 * sigma ** 2))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total

def baseline_search(n, deadline, max_keep=8):
    best = []  # list of (score, vector)

    def try_candidate(vec):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep – dense grid
    for p in np.linspace(0.1, 2.0, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep – dense grid
    for lam in np.linspace(0.0001, 0.05, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep – medium resolution
    al_vals = np.linspace(0.2, 5.0, 25)
    be_vals = np.linspace(0.2, 5.0, 25)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Triangular shape (single)
    if time.time() < deadline:
        try_candidate(generate_triangular(n))

    # 5) Cosine shape (single)
    if time.time() < deadline:
        try_candidate(generate_cosine_shape(n))

    # 6) Gaussian sweep – several sigma values
    for sigma_frac in np.linspace(0.05, 0.5, 30):
        if time.time() >= deadline:
            break
        sigma = sigma_frac * n
        try_candidate(generate_gaussian(n, sigma))

    # 7) random Dirichlet draws with several concentrations
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 800  # reduce a bit for time
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

def adam_update(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    a_proj = project_onto_simplex(a_new)
    return a_proj, m, v

def adam_polish(a0, deadline):
    """Smooth Adam‑based polishing on the simplex."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    # initial objective
    _, conv_max = compute_smooth_gradient(a, beta=10.0)
    best_val = 2.0 * n * conv_max
    best_a = a.copy()
    it = 0
    beta = 10.0
    lr = 0.5
    while time.time() < deadline:

        if cur_val < best_val:

            best_a = a.copy()
        a, m, v = adam_update(a, grad, m, v, it, lr)
        # schedule
        beta = min(beta * 1.02, 200.0)

            a = a * np.exp(jitter)
            a = a / a.sum()
    return best_a

def simulated_annealing(a_start, deadline,
                        init_temp=1e-3,
                        final_temp=1e-6,
                        cooling_factor=0.9995,
                        step_scale_init=0.03,
                        step_scale_min=1e-6):
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1.  Global exploration – generate candidates from several parametric families
    (power‑law, exponential, beta, triangular, cosine, Gaussian) and a large
    pool of random Dirichlet draws.  The top‑k candidates (by the exact
    objective 2·n·max(conv(a,a))) are kept.

2.  Smooth Adam‑based polishing – minimise a smoothed version of the
    max‑convolution using a soft‑max weighting (temperature β).  The
    gradient of the smooth objective is
        g_i = 2·∑_k w_k·a_{k‑i},
    where w = softmax(β·conv).  Adam updates are performed on the simplex
    (multiplicative positivity + Euclidean projection).  β is increased
    gradually so the smooth problem approaches the true max‑problem.

3.  Optional random‑restart polishing – while time permits short Adam runs
    are launched from random Dirichlet starts.

4.  Final simulated‑annealing hill‑climb for fine‑grained local improvement.

All helper functions are defined at top level (no closures) and the search
stops well before the 1000 s wall‑clock limit, returning the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length) using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    if w.sum() > 0:
        w = w / w.sum()
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Parametric families for the global sweep
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_triangular(n: int) -> np.ndarray:
    a = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    return a / a.sum()


def generate_cosine_shape(n: int) -> np.ndarray:
    theta = np.pi * (np.arange(n) - (n - 1) / 2) / n
    a = np.cos(theta)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    mu = (n - 1) / 2.0
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Global baseline search
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Return up to ``max_keep`` best vectors found by a quick global sweep."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep
    for p in np.linspace(0.1, 2.0, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(0.0001, 0.05, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep
    al_vals = np.linspace(0.2, 5.0, 25)
    be_vals = np.linspace(0.2, 5.0, 25)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) simple deterministic shapes
    if time.time() < deadline:
        try_candidate(generate_triangular(n))
    if time.time() < deadline:
        try_candidate(generate_cosine_shape(n))

    # 5) Gaussian sweep
    for frac in np.linspace(0.05, 0.5, 30):
        if time.time() >= deadline:
            break
        sigma = frac * n
        try_candidate(generate_gaussian(n, sigma))

    # 6) random Dirichlet draws
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 800
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smoothing utilities for Adam polishing
# ----------------------------------------------------------------------


def compute_smooth_gradient(a: np.ndarray, beta: float):
    """
    Returns (gradient, max_conv) of the smoothed max of the auto‑convolution.
    The smoothing uses a soft‑max with temperature ``beta``.
    """
    n = a.shape[0]
    # full auto‑convolution
    conv = np.convolve(a, a)                     # length 2n‑1
    maxc = conv.max()

    # softmax weights (stable)
    scaled = beta * (conv - maxc)
    exp_scaled = np.exp(np.clip(scaled, -700, 700))
    sum_exp = exp_scaled.sum()
    if sum_exp == 0 or not np.isfinite(sum_exp):
        w = np.full_like(conv, 1.0 / conv.shape[0])
    else:
        w = exp_scaled / sum_exp

    # gradient g_i = 2 * sum_j w_{i+j} * a_j
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)            # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)              # map to required entries
    g = 2.0 * conv_rev[idx]
    return g, maxc


def adam_update(a, grad, m, v, t, lr,
                beta1=0.9, beta2=0.999, eps=1e-8):
    """One step of Adam on the simplex (projected after the additive update)."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    a_proj = project_onto_simplex(a_new)
    return a_proj, m, v


def adam_polish(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Smooth Adam‑based polishing of a simplex‑constrained vector.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # initial objective
    _, conv_max = compute_smooth_gradient(a, beta=10.0)
    best_val = 2.0 * n * conv_max
    best_a = a.copy()

    it = 0
    beta = 10.0      # start temperature
    lr = 0.5         # initial learning rate

    while time.time() < deadline:
        it += 1
        grad, conv_max = compute_smooth_gradient(a, beta)
        cur_val = 2.0 * n * conv_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        a, m, v = adam_update(a, grad, m, v, it, lr)

        # schedule: raise temperature, decay learning rate
        beta = min(beta * 1.02, 200.0)
        lr = lr * 0.9995

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = np.random.randn(n) * (0.01 / np.sqrt(it + 1))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a

def eg_polish(a0: np.ndarray, deadline: float,
              eta0: float = 0.7, tol: float = 1e-10) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * tol
        active = np.where(conv >= max_val - eps)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        if it % 500 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()
    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

    # 1. Global baseline exploration
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 600.0  # reserve ~10 min for polishing + anneal
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam‑based polishing (primary + short random restarts)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 150.0
    if best_seq is not None and time.time() < polish_deadline:
        remaining = polish_deadline - time.time()
        # primary long run (≈60 % of remaining time)
        primary_time = remaining * 0.6
        primary_deadline = time.time() + primary_time
        a0 = np.array(best_seq, dtype=float)
        refined = adam_polish(a0, primary_deadline)
        val = compute_score(refined)
        print(f"[adam‑polish‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # short restarts (≈30 s each) while time permits
        while time.time() < polish_deadline:
            time_left = polish_deadline - time.time()
            if time_left < 30.0:
                break
            slice_deadline = time.time() + min(30.0, time_left)
            conc = np.random.choice([0.2, 0.5, 1.0, 2.0, 4.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = adam_polish(a_start, slice_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[adam‑polish‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback to the uniform distribution if everything else fails
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
