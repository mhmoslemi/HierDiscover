"""
Hybrid multi‑start optimizer for the auto‑convolution peak problem.

The algorithm proceeds as follows:

1.  Obtain a starting shape – the previously discovered sequence
    (global ``height_sequence_1``) – and upsample it to a larger
    support size (default 8192).  Several random families (U‑shape,
    power‑law, sparse spikes) are also generated.

2.  Every candidate is normalised to a probability vector (sum = 1) and
    its exact score is computed with ``compute_score`` (the same
    objective used by ``evaluate_sequence``).

3.  The best candidate is refined by a sequence of optimisation stages,
    each respecting the remaining wall‑clock budget:
      * Adam optimisation on a smooth‑max surrogate of the peak
        (log‑sum‑exp), with an annealing schedule for the temperature.
      * Projected sub‑gradient descent on the true max‑objective.
      * Optional short L‑BFGS‑B refinement if SciPy is available.
      * A final simulated‑annealing hill‑climb to polish the result.

4.  After each stage the current best vector is symmetrised
    (a↦(a+a[::-1])/2) and re‑projected onto the probability simplex.
    Small random perturbations are occasionally added to escape
    shallow local minima.

The routine stops before the supplied budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a probability vector (∑a = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for random starts)
# ----------------------------------------------------------------------


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse start: ~5 % non‑zero entries, uniform mass on them."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sort a powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


# ----------------------------------------------------------------------
# Helper for up‑sampling a known sequence
# ----------------------------------------------------------------------


def upsample_seq(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation up‑sampling to length ``target_n``."""
    if seq.size == target_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.size)
    x_new = np.linspace(0.0, 1.0, target_n)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 10.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.015,
                         sym_interval: int = 1000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).

    Returns the best vector found and its exact (non‑smooth) score.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # map gradient into log‑space (softmax parametrisation)
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent for the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (lightweight polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Simple random walk on the simplex with a decaying temperature.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Optional L‑BFGS‑B refinement (requires SciPy)
# ----------------------------------------------------------------------


def lbfgsb_refine(start_a: np.ndarray,
                  deadline: float,
                  beta: float = 2000.0) -> tuple[np.ndarray, float]:
    """
    One‑shot L‑BFGS‑B on the smooth‑max with a fixed temperature.
    Returns refined vector + exact score.
    """
    try:
        from scipy.optimize import minimize
    except Exception:
        return start_a, compute_score(start_a)

    n = start_a.shape[0]
    eps = 1e-12
    z0 = np.log(start_a + eps)

    def smooth_obj(z):
        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()
        # smooth max value
        val = maxc + np.log(w.sum()) / beta
        # gradient wrt a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a
        return float(val), grad_z.astype(np.float64)

    res = minimize(fun=lambda z: smooth_obj(z)[0],
                   x0=z0,
                   jac=lambda z: smooth_obj(z)[1],
                   method='L-BFGS-B',
                   options={'maxiter': 120, 'ftol': 1e-12, 'gtol': 1e-12})

    a_opt = softmax(res.x)
    a_opt = (a_opt + a_opt[::-1]) * 0.5
    a_opt = project_to_simplex(a_opt)
    sc = compute_score(a_opt)
    return a_opt, sc


# ----------------------------------------------------------------------
# Main driver – required entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects a wall‑clock budget and returns a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0. Determine target support size (power of two, FFT‑friendly)
    # ------------------------------------------------------------------
    target_n = 8192  # reasonable compromise between expressivity & speed

    # ------------------------------------------------------------------
    # 1. Load a previously found construction, upsample it
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=np.float64)
        base = base / base.sum()
        candidate = upsample_seq(base, target_n)
    else:
        candidate = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    candidate = project_to_simplex(candidate)
    best_a = candidate
    best_score = compute_score(best_a)
    print(f"[{time.time():.1f}] Initial up‑sampled score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Generate a handful of random starts (including analytic families)
    # ------------------------------------------------------------------
    random_starts = [
        random_u_start(target_n),
        random_sparse_start(target_n),
        random_monotone_start(target_n),
        exp_u_family(target_n, gamma=np.random.uniform(0.0, 6.0)),
        left_heavy_power_family(target_n, beta=np.random.uniform(0.0, 12.0)),
    ]

    for rs in random_starts:
        rs = project_to_simplex(rs)
        sc = compute_score(rs)
        if sc < best_score:
            best_score = sc
            best_a = rs.copy()
            print(f"[{time.time():.1f}] Random start improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – iteratively refine the current best
    # ------------------------------------------------------------------
    # The loop tries several optimisation passes while time permits.
    while time.time() < deadline - 30.0:
        # ---- a) Adam smooth‑max refinement ----
        print(f"[{time.time():.1f}] Adam refinement (remaining {deadline - time.time():.1f}s)")
        a_adam, sc_adam = adam_smooth_optimize(
            best_a,
            deadline - 10.0,
            max_iters=500_000,
            beta_start=12.0,
            beta_end=10000.0,
            lr=0.013,
            sym_interval=800,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_a = a_adam
            print(f"[{time.time():.1f}] New best after Adam = {best_score:.6f}")

        # ---- b) Sub‑gradient descent on the true max ----
        print(f"[{time.time():.1f}] Sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            best_a,
            deadline - 8.0,
            max_iters=1_200_000,
            step0=0.04,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_a = a_sub
            print(f"[{time.time():.1f}] New best after sub‑grad = {best_score:.6f}")

        # ---- c) Optional L‑BFGS‑B refinement (if SciPy present) ----
        if time.time() < deadline - 5.0:
            try:
                import scipy  # noqa: F401
                print(f"[{time.time():.1f}] L‑BFGS‑B refinement")
                a_lbfgs, sc_lbfgs = lbfgsb_refine(best_a, deadline - 4.0, beta=3000.0)
                if sc_lbfgs < best_score:
                    best_score = sc_lbfgs
                    best_a = a_lbfgs
                    print(f"[{time.time():.1f}] New best after L‑BFGS‑B = {best_score:.6f}")
            except Exception:
                pass

        # ---- d) Small random perturbation + brief Adam to escape local trap ----
        if time.time() < deadline - 12.0:
            eps = 0.02
            pert = best_a + eps * np.random.randn(target_n)
            pert = np.clip(pert, 0.0, None)
            pert = project_to_simplex(pert)
            a_pert, sc_pert = adam_smooth_optimize(
                pert,
                deadline - 6.0,
                max_iters=150_000,
                beta_start=15.0,
                beta_end=8000.0,
                lr=0.011,
                sym_interval=500,
            )
            if sc_pert < best_score:
                best_score = sc_pert
                best_a = a_pert
                print(f"[{time.time():.1f}] Perturb‑Adam improvement to {best_score:.6f}")

        # If no improvement in the last full cycle, break early
        # (heuristic – we keep trying until time runs out)
        # The loop will naturally stop when the deadline is near.

    # ------------------------------------------------------------------
    # 4. Final symmetrisation and polishing
    # ------------------------------------------------------------------
    best_a = (best_a + best_a[::-1]) * 0.5
    best_a = project_to_simplex(best_a)

    # Final short hill‑climb
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] Final hill‑climb")
        a_final, sc_final = hill_climb_sa(best_a, deadline, max_steps=800_000)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final
            print(f"[{time.time():.1f}] Hill‑climb improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence (as plain Python list)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_a.tolist()
