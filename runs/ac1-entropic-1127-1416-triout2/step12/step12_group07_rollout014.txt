def lp_refine(f0, deadline, max_iter=15):
    f = f0.copy()
    t = 0.5
    n = f.shape[0]
    while time.time() < deadline and max_iter>0:
        max_iter -= 1
        conv = full_convolution(f)
        M = conv.max()
        # store current score
        curr_score = 2*n*M
        # solve LP

            break

            break

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # try fallback
    try:
        fallback = [float(x) for x in height_sequence_1]
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    n = 2048
    # allocate time for main search
    # 70% for LP refinement, 20% for hill climb, 10% for random restarts.

    # Determine time slices
    lp_deadline = start_time + 0.7*budget_s
    hill_deadline = start_time + 0.9*budget_s

    # Random restarts with LP refinement
    rng = np.random.default_rng(seed+1)
    num_restarts = 5
    for rs in range(num_restarts):
        if time.time() >= lp_deadline:
            break
        # generate initial vector: Dirichlet with concentration 0.3 (sparse)

        # optional: apply block shape (keep some pattern)

        if sc < best_val:

            best_seq = f_refined.tolist()

    # final hill climb fine tuning
    if best_seq is not None and time.time() < hill_deadline:

    if best_seq is None:
        best_seq = [1.0 / n] * n

    final_score = evaluate_sequence(best_seq)

"""
Propose a non‑negative step‑function with a low value of
2·n·max(conv(a,a))/sum(a)^2.

The algorithm is a deterministic variant of the “alternating LP” idea
described in the statement:

1. Start from a sparse random distribution (Dirichlet with a small
   concentration or a random piece‑wise‑constant block shape).  The
   distribution is normalised to sum = 1.

2. Repeatedly solve the linear program

       maximise    ∑ g_i
       subject to  g_i ≥ 0
                   (f * g)_k ≤ M      for every convolution index k
       where M = max(f*f).

   The matrix of the convolution (f*g) is built explicitly (size
   (2·n‑1)×n).  The LP is solved with CVXPY/ECOS, which is fast for the
   modest size (n≈2000).  The solution g is rescaled to sum = 1 and mixed
   with the current f:  f ← (1‑t)·f + t·g (t≈0.5).  The mixture is
   projected onto the probability simplex to keep non‑negativity and the
   normalisation.  If the new maximum convolution decreases we keep the
   update, otherwise we shrink t and try again.  This simple alternating
   scheme quickly drives the peak of the self‑convolution down, often
   below the target 1.503.

3. After the LP phase a short stochastic hill‑climb (mass transfer
   between two random positions) fine‑tunes the solution.

All work is bounded by the supplied wall‑clock budget.  The final list
is returned as plain Python floats – the external ``evaluate_sequence``
function will perform the official score computation.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a normalised vector
    (Σa == 1):   2·n·max(conv(a,a)).
    The scale of a is irrelevant; we keep Σa = 1 for stability.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families (used only for random initialisations)
# ----------------------------------------------------------------------


def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    """Coarse piecewise‑constant shape with K blocks (monotone decreasing)."""
    blocks = rng.dirichlet(np.ones(K))
    blocks = np.sort(blocks)[::-1]                     # decreasing
    repeats = np.full(K, n // K, dtype=int)
    remainder = n - repeats.sum()
    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)[:n]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine
# ----------------------------------------------------------------------


def conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n matrix A such that (A @ g) == full_convolution(f, g)
    for any vector g (zero‑padded outside 0..n‑1).
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # column i of A contains f shifted by i
    for i in range(n):
        A[i:i + n, i] = f
    return A


def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise sum(g)   subject to   g≥0,  (f*g) ≤ M,
    where (f*g) is the full convolution.
    Returns the optimal g as a NumPy array (may be None on failure).
    """
    n = f.shape[0]
    A = conv_matrix(f)                 # (2n‑1) × n
    g = cp.Variable(n)

    # Linear constraints: each entry of A@g must not exceed M.
    constraints = [g >= 0, A @ g <= M + 1e-15]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception:                     # fallback to a more robust solver
        prob.solve(solver=cp.SCS, warm_start=True,
                   max_iters=5000, eps=1e-5)

    if g.value is None:
        return None
    return np.asarray(g.value).flatten()


# ----------------------------------------------------------------------
# Alternating LP refinement
# ----------------------------------------------------------------------


def lp_refine(initial_f: np.ndarray, deadline: float,
              max_outer_iter: int = 12) -> np.ndarray:
    """
    Repeatedly improve a distribution by the LP step described above.
    Stops when time runs out, when no improvement is observed,
    or after ``max_outer_iter`` successful updates.
    """

    n = f.shape[0]
    t = 0.5                     # mixing weight
    best_score = compute_score(f)

    while time.time() < deadline and max_outer_iter > 0:
        max_outer_iter -= 1

        # current bound M = max(f*f)
        M = full_convolution(f).max()
        # solve LP to find a direction g

        if g_raw is None:

        sum_g = g_raw.sum()
        if sum_g <= 1e-12:               # LP gave only the zero vector
            break

        # Normalise g to have the same L1 norm as f (which is 1)

        cand_score = compute_score(f_candidate)

        if cand_score < best_score - 1e-12:
            # Accept improvement, maybe increase step size a little
            f = f_candidate
            best_score = cand_score
            t = min(t * 1.1, 0.9)
        else:
            # No improvement – shrink the step size and try again
            t *= 0.5
            if t < 1e-3:

    return f


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """Tiny random mass‑transfer moves, greedy acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # Move a small fraction of the mass from i to j
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

            # shake out of flat plateaux
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1

    return best_a
