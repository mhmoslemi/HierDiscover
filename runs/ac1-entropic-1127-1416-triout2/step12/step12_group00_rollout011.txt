def subgradient_optimize(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = compute_score(a)
    t=0
    while time.time() < deadline:
        # compute convolution
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]
        # compute subgradient
        grad = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            # contributions: for each i, j such that i + j = k
            # Where j = k - i
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]
        # step size: diminishing
        step = 0.2 / np.sqrt(t + 1)
        a -= step * grad
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
        t += 1
        # occasional jitter
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)
    return best, best_val

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
1. **Exploration** – generate candidates from a rich collection of
   parametric families (power‑law, symmetric power‑law, exponential,
   beta, Gaussian, sine‑power) and a large pool of random Dirichlet draws.
   Keep the best few (by the exact objective
   `2·n·max(conv)` – the scaling of ``evaluate_sequence`` for sum=1).

2. **Deterministic polishing** – for every promising candidate run a
   projected sub‑gradient descent on the true ``max(conv)`` objective.
   The sub‑gradient of ``max(conv)`` is obtained from the active
   convolution indices and the iteration uses a simple 1/√t step‑size.
   This phase gets the lion’s share of the time‑budget.

3. **Stochastic hill‑climb** – a long simulated‑annealing pass
   (Metropolis moves that re‑allocate a small amount of mass between
   two coordinates) finishes the search and helps to escape any
   remaining shallow minima.

All vectors are kept on the probability simplex (sum = 1, non‑negative);
the scaling does not affect the final score because ``evaluate_sequence``
uses the homogeneous ratio `max(conv)/(sum^2)`.  The programme respects
the 1000 s wall‑clock limit and prints intermediate scores.

The final result is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Zero vector projects to the uniform distribution.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for efficient FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used inside the optimiser.
    The input is assumed to be a probability vector (sum = 1).
    Returns ``2·n·max(conv(a,a))``.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families for the exploration phase
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_sym_powerlaw(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(1, n + 1, dtype=float)
    rev = np.arange(n, 0, dtype=float)
    a = i ** (-p) + rev ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(- (i‑mid)² / (2σ²))."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total

    """Generate many candidates, keep the ``top_k`` with lowest score."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < top_k:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Power‑law and symmetric power‑law
    for p in np.linspace(0.05, 4.0, 81):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))
        if time.time() >= deadline:
            break
        try_candidate(generate_sym_powerlaw(n, p))

    # 2) Exponential
    for lam in np.linspace(0.0001, 0.2, 150):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta (including very U‑shaped cases)
    al_vals = np.concatenate(
        (np.linspace(0.05, 0.5, 9), np.linspace(0.6, 3.0, 13))
    )
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Gaussian sweep
    for sigma in np.linspace(0.05 * n, 0.5 * n, 20):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # 5) Sine‑power sweep
    for p in np.linspace(0.1, 8.0, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # 6) Random Dirichlet draws with many concentrations
    concs = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_conc = 2000 // len(concs)
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    # Return just the vectors, ordered by score
    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (the heavy workhorse)
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the true max‑convolution objective.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of max(conv) – sum of contributions from active offsets
        grad = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]

        # decreasing step size
        step = 0.20 / np.sqrt(t)
        a -= step * grad
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional small random jitter to escape plateaus
        if t % 4000 == 0:
            jitter = np.random.randn(n) * (0.006 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (fine‑tuning)
# ----------------------------------------------------------------------


def annealing_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Metropolis annealing that moves a small amount of mass between two coordinates."""
    a = a0.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = 1e-2          # initial temperature
    final_T = 1e-7    # final temperature
    cooling = 0.9999
    step_scale = 0.02

    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)

        if a[i] <= 0.0:
            continue
        delta = step_scale * a[i]          # move this fraction from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        dval = val - cur_val

        # Metropolis acceptance
        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-15)):
            a = cand
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

            # a bit more aggressive when we improve
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            step_scale = max(step_scale * 0.995, 1e-6)

        T = max(final_T, T * cooling)

        # occasional global jitter
        if np.random.rand() < 0.001:
            a += np.random.randn(n) * (step_scale * 0.02)
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin so the harness does not cut us off
    deadline = start_time + budget_s - 5.0

    # Length of the step‑function; the value of ``n`` directly appears
    # in the evaluation formula.
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad baseline exploration (≈ 200 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 600.0  # оставляем ~ 600 s для polishing+anneal
    if time.time() < baseline_deadline:
        candidates = baseline_explore(n, baseline_deadline, top_k=6)
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Heavy sub‑gradient polishing (≈ 600 s)
    # -----------------------------------------------------------------
    if best_seq is not None:
        polish_deadline = deadline - 200.0
        # we will run polishing on several seeds:
        seeds = [np.array(best_seq)]
        # a few random perturbations of the current best
        for _ in range(3):
            pert = project_to_simplex(
                np.maximum(seeds[0] + 0.02 * np.random.randn(n), 0.0)
            )
            seeds.append(pert)

        polished_best = seeds[0].copy()
        polished_val = best_val

        for idx, start_vec in enumerate(seeds):
            if time.time() >= polish_deadline:
                break
            local_deadline = polish_deadline - (time.time() - start_time) - 1.0
            if local_deadline <= time.time():
                break
            vec_opt, val_opt = subgradient_optimize(start_vec, local_deadline)
            print(f"[polish #{idx}] score = {val_opt:.6f}")
            if val_opt < polished_val:
                polished_val, polished_best = val_opt, vec_opt.copy()

        best_val, best_seq = polished_val, polished_best.tolist()
        print(f"[polish] best after polishing = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Simulated annealing fine‑tuning (remaining time)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_start = np.array(best_seq, dtype=float)
        vec_final, val_final = annealing_optimize(a_start, deadline)
        print(f"[anneal] final score = {val_final:.6f}")
        if val_final < best_val:
            best_val, best_seq = val_final, vec_final.tolist()

    # -----------------------------------------------------------------
    # 4. Safety net – uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Final official score report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
