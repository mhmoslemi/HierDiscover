def smooth_opt(a0, time_budget, start_time):
    # a0 is numpy array normalized sum 1.
    best_a = a0
    best_val = compute_score(a0)

    # Convert to logits
    eps = 1e-12
    z = np.log(a0 + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05   # Learning rate for logits

    alpha = 10.0
    alpha_max = 5000.0
    alpha_factor = 1.005  # increase each iteration

    it = 0
    while time.time() < time_budget:
        it += 1

        # recompute a
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / np.sum(exp_z)

        # compute conv
        conv = full_convolution(a)
        max_conv = conv.max()

        # compute weights
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # compute gradient w.r.t a via cross-correlation
        w_rev = w[::-1]
        conv_w = full_convolution(a, w_rev)
        idx = 2 * n - 2 - np.arange(n)
        g = 2.0 * conv_w[idx]

        # compute to logits gradient
        avg = np.dot(g, a)
        grad_z = a * (g - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # gradient descent step
        z = z - step

        # schedule alpha
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # evaluate and record best
        if it % 10 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                # print progress
                print(f"[smooth_opt] it={it}, score={cur_val:.6f}, alpha={alpha:.1f}")

        # stop early if improvement stall? No.

    return best_a, best_val
```

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1.  Global exploration – a quick sweep of a few parametric families and a
    large pool of Dirichlet draws.  The top‑k (by the exact objective
    2·n·max(conv)) are retained.

2.  Smoothed exponentiated‑gradient (EG) polishing – we minimise a smooth
    approximation of the max‑convolution:
        L(a) = softmax_α(conv(a,a))   (α starts small and is increased)
    The EG update uses the gradient of the weighted sum
        g_i = 2·∑_k w_k·a_{k‑i},   w = softmax_α(conv).
    A decreasing step‑size η=η₀/√t and occasional random jitter help
    escape plateaux.

3.  Optional L‑BFGS‑style (Adam) fine‑tuning on the logits (softmax
    parametrisation) – a short run for the current best vector.

4.  Final simulated‑annealing hill‑climb for a few seconds.

All stages respect the 1000 s wall‑clock budget and keep the best
sequence found.  The returned list satisfies the requirements of
``evaluate_sequence`` (non‑negative, sum > 0.01, values ≤ 1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum == 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used only for the cheap global sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Global exploration (cheap baseline)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 6):
    """Return up to ``max_keep`` best vectors found by a quick sweep."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s                     # project onto the simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep – coarse grid
    for p in np.linspace(0.1, 2.0, 80):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep – coarse grid
    for lam in np.linspace(0.0001, 0.05, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep – modest resolution
    al_vals = np.linspace(0.2, 5.0, 12)
    be_vals = np.linspace(0.2, 5.0, 12)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) random Dirichlet draws with several concentration values
    concs = [0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 800   # ≈ 4800 draws total – still cheap
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed EG polishing (soft‑max approximation of the max)
# ----------------------------------------------------------------------


def weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute gradient g_i = 2·∑_k w_k·a_{k‑i}
    using a single FFT‑based convolution.
    """
    n = a.shape[0]
    # reverse w (needed for the cross‑correlation)
    w_rev = w[::-1]
    conv = full_convolution(a, w_rev)          # length 3·n‑2
    idx = 2 * n - 2 - np.arange(n)            # map to correct positions
    g = 2.0 * conv[idx]
    return g


def smooth_eg_optimize(a0: np.ndarray, deadline: float,
                       eta0: float = 0.8,
                       alpha_start: float = 5.0,
                       alpha_factor: float = 1.01,
                       alpha_max: float = 2000.0,
                       tol: float = 1e-12) -> np.ndarray:
    """
    Smoothed exponentiated‑gradient optimiser.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    alpha = alpha_start

    while time.time() < deadline:
        it += 1

        # --- convolution of a with itself ---------------------------------
        conv = full_convolution(a)
        max_c = conv.max()

        # --- soft‑max weights over the convolution entries -----------------
        # shift by max_c for numerical stability
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # --- gradient of the weighted sum --------------------------------
        g = weighted_gradient(a, w)

        # --- EG multiplicative update (project back to simplex) ----------
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # renormalise, protect against under‑flow
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # --- record best --------------------------------------------------
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[smooth‑EG] it={it:,}, score={cur_val:.6f}, alpha={alpha:.1f}")

        # --- temperature (α) schedule ------------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # --- occasional jitter to escape plateaux -------------------------
        if it % 800 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.002 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a

def adam_finetune(a_start: np.ndarray, deadline: float,
                  lr: float = 0.12,
                  alpha_start: float = 20.0,
                  alpha_factor: float = 1.008,
                  alpha_max: float = 5000.0) -> np.ndarray:
    """
    Adam optimiser on the logit parametrisation of the simplex.
    Uses the same soft‑max weighted loss as ``smooth_eg_optimize``.
    """
    n = a_start.shape[0]
    # initialise logits (avoid log(0))
    eps = 1e-24
    z = np.log(a_start + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_a = a_start.copy()
    best_val = compute_score(best_a)

    it = 0
    alpha = alpha_start

    while time.time() < deadline:
        it += 1
        # softmax (stable)
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # convolution and soft‑max weights
        conv = full_convolution(a)
        max_c = conv.max()
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient wrt a
        g_a = weighted_gradient(a, w)

        # convert to gradient wrt logits
        avg = np.dot(g_a, a)
        grad_z = a * (g_a - avg)

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z - step

        # temperature schedule
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # evaluate occasionally
        if it % 15 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[adam] it={it:,}, score={cur_val:.6f}, alpha={alpha:.1f}")

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Very short Metropolis‑style hill‑climb."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0.0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

                                    alpha_factor=1.010,
                                    alpha_max=4000.0)
            val = compute_score(refined)
            print(f"[adam‑finetune] refined score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()

    # ------------------------------------------------------------------
    # 3. Final simulated‑annealing polish (≈ the last 100 s)
    # ------------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # ------------------------------------------------------------------
    # 4. Fallback to a uniform distribution if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5. Report the official score before returning
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
