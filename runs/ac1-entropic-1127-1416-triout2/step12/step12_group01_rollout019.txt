def adam_optimize(start_a, deadline):
    n = len(start_a)
    eps_vec = 1e-12
    u = np.log(start_a + 1e-12)   # initial log-space
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9; beta2 = 0.999; epsilon = 1e-8
    t = 0
    best_a = start_a.copy()
    best_score = compute_score(best_a)
    while time.time()<deadline:
        t+=1
        # compute current a
        max_u = np.max(u)  # for numeric stability
        exp_u = np.exp(u - max_u)   # shift
        a = exp_u / np.sum(exp_u)
        # compute conv
        conv = full_convolution(a)
        max_val = np.max(conv)
        # active indices
        eps = max_val * 1e-12 (maybe relative tolerance)
        active = np.where(conv >= max_val - eps)[0]
        # compute subgradient g
        # same as before
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            i_idx = np.arange(i_start, i_end + 1)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]
        # compute gradient w.r.t u: g_u = a * (g - np.dot(g, a))
        g_u = a * (g - np.dot(g, a))
        # Adam update
        m = beta1 * m + (1 - beta1) * g_u
        v = beta2 * v + (1 - beta2) * (g_u ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        eta = 0.05 / (np.sqrt(t) + 1)  # maybe schedule
        u -= eta * m_hat / (np.sqrt(v_hat) + epsilon)
        # Optionally add small random noise occasionally to escape
        if t % 2000 == 0:
            # jitter as random normal scaled by 0.01
            u += np.random.randn(n) * 0.001
        # compute new score
        a = np.exp(u - np.max(u))
        a = a / np.sum(a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
        # maybe break if stuck for many iterations

"""
Non‑negative step‑function optimiser for evaluate_sequence

This implementation builds on the previous exponential‑gradient
approach but replaces it with a mirror‑descent optimiser based on
the Adam algorithm (soft‑max parametrisation).  The search consists
of three stages:

1. Global exploration – dense sweeps of power‑law, exponential and
   beta families (including extreme α,β values) together with a large
   pool of random Dirichlet draws.  The best few candidates (by the
   exact objective 2·n·max(conv)) are retained.

2. Adam polishing – each retained candidate is refined by minimising
   the convex‑concave objective max_k (a*a)_k using the sub‑gradient
   of the active lag(s) and a soft‑max (mirror) parametrisation
   a_i = exp(u_i)/∑_j exp(u_j).  Adam’s adaptive step‑sizes
   converge faster than the plain exponentiated‑gradient.

3. Random‑restart Adam – while time permits we launch additional
   Adam runs from fresh Dirichlet starts to escape possible local
   minima.

All timings are measured against the supplied wall‑clock budget.
The routine returns the best sequence found (as a plain Python list)
which is directly fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape:
       a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total

def baseline_search(n: int, deadline: float, max_keep: int = 10):
    """Sweep families and many random Dirichlet samples.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s                     # project onto the simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep – dense log‑grid (covers very flat to peaked)
    for p in np.linspace(0.05, 5.0, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep – wide range
    for lam in np.linspace(1e-5, 0.2, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep – explore extreme U‑shapes and asymmetric shapes
    al_vals = np.concatenate([np.geomspace(0.01, 0.2, 12),
                              np.linspace(0.2, 3.0, 20)])
    be_vals = np.concatenate([np.geomspace(0.01, 0.2, 12),
                              np.linspace(0.2, 3.0, 20)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) massive random Dirichlet draws with several concentration values
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1500   # ≈ 10 500 draws (fast on typical hardware)
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam polishing (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def adam_polish(a0: np.ndarray, deadline: float,
                lr: float = 0.05,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                tol: float = 1e-12) -> np.ndarray:
    """
    Adam‑based optimisation of   max_k (a*a)_k   on the simplex.
    The variables are kept in the soft‑max domain ``u`` (a_i = exp(u_i)/∑exp(u)).
    Returns the best vector observed before the deadline.
    """
    n = a0.shape[0]

    # initialise u from a0 (avoid log(0))
    u = np.log(a0 + 1e-20)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # soft‑max projection
        max_u = np.max(u)                 # for numerical stability
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # compute convolution and active set
        conv = full_convolution(a)
        max_val = conv.max()
        # identify indices whose value is within a tiny relative tolerance
        active = np.where(conv >= max_val - max_val * tol)[0]

        # sub‑gradient of the max‑convolution (w.r.t. a)
        g = np.zeros_like(a)
        # loop over the (usually) tiny active set
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # convert to gradient w.r.t. u (mirror‑descent formula)
        g_u = a * (g - np.dot(g, a))

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)   # gentle decay with iterations
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random jitter to escape plateaux (no bias)
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (lr * 0.001)
            u += jitter

        # evaluate current a (cheap – we already have it)
        cur_val = float(2.0 * n * max_val)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # early exit if progress stalls (optional)
        if t % 5000 == 0:
            # recompute a to avoid drift
            max_u = np.max(u)
            exp_shift = np.exp(u - max_u)
            a = exp_shift / exp_shift.sum()
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

    # ensure we return a valid probability vector
    if best_a is None:
        best_a = a0.copy()
    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing (optional final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep on simplex
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 3000 == 0:
            # global jitter
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

The algorithm works in three phases:

1. **Global sweep** – dense scans of three parametric families
   (power‑law, exponential, beta) with far‑reaching parameter ranges
   plus many random Dirichlet draws.  The ten best candidates (by the
   exact objective ``2·n·max(conv)``) are retained.

2. **Adam polishing** – each retained candidate is refined by minimising
   ``max_k (a*a)_k`` on the simplex.  The variables are kept in a
   soft‑max parametrisation ``a_i = exp(u_i) / Σ_j exp(u_j)`` and an
   Adam optimiser (mirror‑descent) is applied to the sub‑gradient of the
   active lag(s).  Adaptive step‑sizes converge faster than the plain
   exponentiated‑gradient used previously.

3. **Random‑restart polishing & annealing** – while time permits new
   Adam runs are launched from fresh Dirichlet starts; a short simulated‑
   annealing pass is applied to the current best vector to escape possible
   local minima.

All stages respect the 1000 s wall‑clock budget.  The routine returns the
best sequence found as a plain list of floats, ready for the official
``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power‑of‑two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape:
       a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration (baseline)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 10):
    """Sweeps families and a large pool of Dirichlet draws, keeping the
    best ``max_keep`` vectors (as numpy arrays)."""
    best = []   # [(score, vector), ...]

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s                     # project onto the simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep – wide log‑grid
    for p in np.linspace(0.05, 5.0, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep – broad range
    for lam in np.linspace(1e-5, 0.2, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep – extreme U‑shapes + asymmetric shapes
    al_vals = np.concatenate([np.geomspace(0.01, 0.2, 12),
                              np.linspace(0.2, 3.0, 20)])
    be_vals = np.concatenate([np.geomspace(0.01, 0.2, 12),
                              np.linspace(0.2, 3.0, 20)])
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) massive random Dirichlet draws
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1500   # ≈ 10 500 draws – fast on modern CPUs
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam polishing (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def adam_polish(a0: np.ndarray, deadline: float,
                lr: float = 0.05,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                tol: float = 1e-12) -> np.ndarray:
    """
    Adam‑based optimisation of   max_k (a*a)_k   on the simplex.
    The variables are stored as ``u`` with ``a = softmax(u)``.
    Returns the best vector observed before the deadline.
    """
    n = a0.shape[0]
    # initialise in log‑space (avoid log(0))
    u = np.log(a0 + 1e-20)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # ---- soft‑max projection ----
        max_u = np.max(u)                 # numerical stability
        exp_shift = np.exp(u - max_u)
        a = exp_shift / exp_shift.sum()

        # ---- convolution and active set ----
        conv = full_convolution(a)
        max_val = conv.max()
        active = np.where(conv >= max_val - max_val * tol)[0]

        # ---- sub‑gradient w.r.t. a ----
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # ---- mirror‑descent gradient (soft‑max Jacobian) ----
        g_u = a * (g - np.dot(g, a))

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(t) + 1.0)   # gentle decay
        u -= step * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of plateaux
        if t % 2000 == 0:
            u += np.random.randn(n) * (lr * 0.001)

        # ---- keep the best solution seen so far ----
        cur_val = float(2.0 * n * max_val)   # identical to compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # (optional) recompute a every few thousand iterations to avoid drift
        if t % 5000 == 0:
            max_u = np.max(u)
            exp_shift = np.exp(u - max_u)
            a = exp_shift / exp_shift.sum()
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()

    return best_a if best_a is not None else a0


# ----------------------------------------------------------------------
# Light simulated‑annealing (optional final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a /= a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # problem size – 2048 offers a good trade‑off
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previous run
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 300.0   # reserve most of the budget for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=10)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam polishing (primary + random restarts)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        # polish the current best
        a0 = np.array(best_seq, dtype=float)
        refined = adam_polish(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[adam‑polish‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # extra random restarts while time permits
        while time.time() < polish_deadline:
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = adam_polish(a_start, polish_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[adam‑polish‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback to uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
