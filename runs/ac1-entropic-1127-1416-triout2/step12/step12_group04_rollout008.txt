"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

using a multi‑phase scheme:
1.  Wide analytic scan (uniform, product, power, Gaussian, windows, random).
2.  Smoothing‑gradient descent (soft‑max approximation of the max) with an
    exponential β‑schedule and Euclidean projection onto the probability simplex.
3.  LP‑based refinement, now with an optimal mixing weight obtained by a
    cheap ternary search on the exact quadratic form of the convolution.
4.  Symmetrisation (the optimum is symmetric) and a final simulated‑annealing
    hill‑climb.

The algorithm respects the 1000 s wall‑clock budget and returns a plain list
ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (returns length a+b‑1)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑0.5‑i))^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone left‑heavy: w_i ∝ (i+0.5)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, window_type: str) -> np.ndarray:
    """Standard cosine windows."""
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing (heavy left)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped start."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

# Smoothed max gradient descent
# ----------------------------------------------------------------------


def smooth_grad_descent(
    start_a: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    step0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using
    a soft‑max with a β–schedule, Euclidean projection onto the simplex
    after each step. Returns the best vector seen.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    phase_start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # β schedule (exponential increase)
        frac = (time.time() - phase_start) / (deadline - phase_start)
        beta = beta_start * ((beta_end / beta_start) ** frac)

        # convolution and soft‑max weights
        p = convolution_fft(a, a)                 # length 2n‑1
        p_max = p.max()
        w_raw = np.exp(beta * (p - p_max))        # shift for stability
        w = w_raw / w_raw.sum()                   # normalised weights

        # gradient of Σ w_i p_i  (smooth max)
        w_rev = w[::-1]
        conv = convolution_fft(w_rev, a)          # full length 3n‑2
        idx = 2 * n - 2 - np.arange(n)            # positions we need
        grad = 2.0 * conv[idx] - w[2 * np.arange(n)] * a

        # step size (inverse‑sqrt schedule)
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = project_to_simplex(a)

        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] smooth‑GD improved to {sc:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP refinement with optimal mixing weight (ternary search)
# ----------------------------------------------------------------------


def optimal_mix_t(conv_f: np.ndarray, conv_g: np.ndarray, conv_fg: np.ndarray) -> float:
    """
    Find t∈[0,1] minimising max_i ((1‑t)²·conv_f[i] + 2t(1‑t)·conv_fg[i] + t²·conv_g[i]).
    The function is convex → ternary search works.
    """
    lo, hi = 0.0, 1.0
    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        mix1 = (1 - t1) ** 2 * conv_f + 2 * t1 * (1 - t1) * conv_fg + t1 ** 2 * conv_g
        mix2 = (1 - t2) ** 2 * conv_f + 2 * t2 * (1 - t2) * conv_fg + t2 ** 2 * conv_g
        if mix1.max() < mix2.max():
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0


def lp_refine_step_opt(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0, then mix f with the normalised solution using the
    optimal mixing weight t* (found by ternary search).
    Returns the new vector (projected onto the simplex).
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A where (A @ g) = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)  # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # HiGHS LP solver (fast)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f
    g = g / sum_g

    # Pre‑compute convolutions once
    conv_f = convolution_fft(f, f)
    conv_g = convolution_fft(g, g)
    conv_fg = convolution_fft(f, g)

    # Optimal mixing weight
    t_opt = optimal_mix_t(conv_f, conv_g, conv_fg)

    f_new = (1.0 - t_opt) * f + t_opt * g
    f_new = project_to_simplex(f_new)
    return f_new

# Stochastic hill‑climb (simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
) -> tuple[np.ndarray, float]:
    """Simple SA on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

        # small random perturbation to avoid possible stagnation
        pert = a * np.exp(0.02 * np.random.randn(n))
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()
        a_opt, sc_opt = smooth_grad_descent(
            pert,
            deadline - 150.0,
            beta_start=5.0,
            beta_end=5000.0,
            step0=0.6,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] after smooth‑GD best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  LP refinement loop with optimal mixing weight
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time() - start_time:.1f}] starting LP refinement")
        lp_iters = 0
        while time.time() < deadline - 60.0 and lp_iters < 30:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step_opt(a, M)
            sc = compute_score(a_new)
            if sc < best_score - 1e-7:
                best_score = sc
                a = a_new
                print(f"[{time.time() - start_time:.1f}] LP iter {lp_iters} improved to {best_score:.6f}")
            else:
                # if no improvement, we keep trying – could still be useful later
                pass
            lp_iters += 1

    # ------------------------------------------------------------------
    # 6.  Symmetrisation (the optimum is symmetric)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time() - start_time:.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Return result
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
