"""
Propose a step‑function (non‑negative vector) that minimizes the
autocorrelation peak evaluated by ``evaluate_sequence``.

Approach
--------
0.  Helper utilities: FFT convolution, simplex projection, soft‑max.
1.  Baseline families (uniform, product, power‑law, exponential‑U, …)
    are explored on a moderate size (n = 8192).  For each family a rapid
    global optimisation is performed with SciPy's differential evolution.
2.  The best candidate becomes the start point for a smooth‑max Adam
    optimiser (no forced symmetrisation – asymmetry is useful for lowering
    the peak).  The temperature β is annealed from a modest value to a
    large one, effectively approximating the true max‑operator.
3.  Exact projected sub‑gradient descent is applied as a final polishing
    step (step‑size ∝ 1/√t).
4.  A cheap greedy local‑search (mass transfer between two coordinates)
    tries to shave off the remaining few hundredths.
5.  The final vector is projected onto the probability simplex (∑a = 1)
    and returned as a plain Python list.  Throughout the run the remaining
    wall‑clock time is respected; the algorithm aborts early if the budget
    is exhausted.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the optimiser.
    a is assumed to be a probability vector (∑a = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def uniform_family(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # avoid overflow for very large alpha
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def product_expU_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod - gamma * dist)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def asymmetric_product_family(n: int, beta_left: float, beta_right: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta_left * np.log(i + 0.5) - beta_right * np.log(n - 0.5 - i))
    w /= w.sum()
    return w


def exponential_decay_family(n: int, lam: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         beta_start: float = 5.0,
                         beta_end: float = 20000.0,
                         lr: float = 0.012,
                         sym_interval: int = 0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    Returns the best vector found (probability simplex) and its exact score.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)          # log‑softmax parameters

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        a = softmax(z)

        # occasional symmetrisation (disabled by default)
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 4000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of conv[k] w.r.t. a (see analysis)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def greedy_local_search(start_a: np.ndarray,
                        deadline: float,
                        max_trials: int = 300_000,
                        init_eps: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Cheap greedy mass‑transfer hill‑climb.
    Randomly moves a small fraction of mass from one index to another
    and keeps the change if the objective improves.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_score = compute_score(a)
    eps = init_eps

    for trial in range(1, max_trials + 1):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = eps * a[i]
        # ensure we do not make a[i] negative
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_score:
            a = cand
            best_score = sc
            # when we improve, allow a slightly larger step size
            eps = min(eps * 1.04, 0.1)
        else:
            # slowly shrink step size on failures
            eps *= 0.998

        if trial % 5000 == 0:
            print(f"[{time.time():.1f}] greedy search trial {trial}, best={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Global search over analytic families (using SciPy DE if available)
# ----------------------------------------------------------------------


def optimise_family_de(func, bounds, max_iter=12, seed=0):
    """
    Wrapper around scipy.optimize.differential_evolution.
    Returns (best_vector, best_score, best_params).
    """
    try:
        from scipy.optimize import differential_evolution
    except Exception:
        return None, float("inf"), None

    # objective for DE – we minimise the score directly
    def de_obj(p):
        vec = func(*p)
        return compute_score(vec)

    result = differential_evolution(de_obj,
                                    bounds=bounds,
                                    maxiter=max_iter,
                                    popsize=15,
                                    polish=True,
                                    seed=seed,
                                    updating='deferred',
                                    disp=False)
    best_params = result.x
    best_vec = func(*best_params)
    best_sc = compute_score(best_vec)
    return best_vec, best_sc, best_params

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best vector found (plain Python list).
    The routine respects the wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # Choose problem size
    # -----------------------------------------------------------
    n = 8192   # power of two – fast FFTs and enough degrees of freedom

    # -----------------------------------------------------------
    # Baseline uniform
    # -----------------------------------------------------------
    vec = uniform_family(n)
    sc = compute_score(vec)
    best_seq, best_score = vec.tolist(), sc
    print(f"[{time.time():.1f}] baseline uniform score = {best_score:.6f}")

    # -----------------------------------------------------------
    # Helper to update best
    # -----------------------------------------------------------
    def consider(v, sc):
        global best_seq, best_score
        if sc < best_score:
            best_score = sc
            best_seq = v.tolist()
            print(f"[{time.time():.1f}] NEW BEST = {best_score:.6f}")

    # -----------------------------------------------------------
    # Coarse families + DE refinement
    # -----------------------------------------------------------
    families = [
        ("product", symmetric_product_family,
         [(0.0, 12.0)]),                                          # α
        ("left_power", left_heavy_power_family,
         [(0.0, 12.0)]),                                          # β
        ("exp_U", exp_u_family,
         [(0.0, 6.0)]),                                           # γ
        ("mix_U", mixed_uniform_u_family,
         [(0.0, 1.0), (0.0, 6.0)]),                               # λ,γ
        ("asym_prod", asymmetric_product_family,
         [(0.0, 12.0), (0.0, 12.0)]),                             # βL,βR
        ("prod_expU", product_expU_family,
         [(0.0, 12.0), (0.0, 6.0)]),                              # α,γ
        ("exp_decay", exponential_decay_family,
         [(0.0, 6.0)])                                            # λ
    ]

    # allocate some time for DE runs
    de_time_budget = deadline - time.time() - 250.0   # leave ~250s for later phases
    per_family_budget = max(5.0, de_time_budget / len(families))
    print(f"[{time.time():.1f}] Starting DE optimisation, approx {per_family_budget:.1f}s per family")

    for name, func, bounds in families:
        if time.time() >= deadline - 250.0:
            break
        # quick coarse sampling (5 points per dimension) just to get a feel
        # (not strictly necessary but inexpensive)
        # then run DE
        print(f"[{time.time():.1f}] DE on family {name}")
        vec_opt, sc_opt, _ = optimise_family_de(
            lambda *p: func(n, *p) if isinstance(p, (list, tuple, np.ndarray)) else func(n, p),
            bounds,
            max_iter=12,
            seed=seed + hash(name) % 1000
        )
        if vec_opt is not None:
            consider(vec_opt, sc_opt)

    # -----------------------------------------------------------
    # First Adam (smooth‑max) optimisation – no forced symmetrisation
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] Adam optimisation (phase 1)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 150.0,
            max_iters=800_000,
            beta_start=5.0,
            beta_end=20000.0,
            lr=0.012,
            sym_interval=0   # keep asymmetry
        )
        consider(a_opt, sc_opt)

    # -----------------------------------------------------------
    # Projected sub‑gradient descent (exact max)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] Sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 60.0,
            max_iters=2_500_000,
            step0=0.04
        )
        consider(a_sub, sc_sub)

    # -----------------------------------------------------------
    # Greedy local mass‑transfer hill‑climb
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] Greedy local search")
        a_loc, sc_loc = greedy_local_search(
            np.array(best_seq, dtype=np.float64),
            deadline - 5.0,
            max_trials=200_000,
            init_eps=0.02
        )
        consider(a_loc, sc_loc)

    # -----------------------------------------------------------
    # Final symmetrisation test (optional – sometimes helps)
    # -----------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] Symmetrisation improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # Final official evaluation (includes all security checks)
    # -----------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
