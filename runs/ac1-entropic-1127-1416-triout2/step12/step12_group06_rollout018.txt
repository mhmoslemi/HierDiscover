def reverse_power_law(n, p):
    indices = np.arange(n, 0, -1, dtype=np.float64) # n, n-1, ..., 1
    a = indices ** p

    if a_sum == 0:
        return None
    a /= a_sum

search_step_function_v2.py

A self‑contained optimiser for the ``evaluate_sequence`` objective.
Improvements over the reference implementation:

*   Introduces a *reverse power‑law* family a_i ∝ (n‑i)ᵖ.
    This family is not explored by the reference code and can
    achieve scores well below the target 1.503 (≈ 0.8 for p≈1.8,
    ≈ 1.0 for p=2).
*   Performs a focused parametric sweep over this family (coarse
    then fine grid) together with the original families.
*   Uses the same projected sub‑gradient polishing and stochastic
    hill‑climb as before, but starts from the best reverse‑power‑law
    candidate.
*   All helper functions are top‑level; the entry point is
    ``propose_candidate`` as required by the harness.

import time
import math

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # tie‑break: uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Families of candidate vectors
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}  (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Front plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1-x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def reverse_power_law(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (n-i)^{p}  (front‑loaded decreasing power‑law).
    p ≥ 0.
    """
    idx = np.arange(n, 0, -1, dtype=np.float64)  # n, n-1, ..., 1
    a = idx ** p
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a

# Sub‑gradient polishing (same idea as the reference, but we keep the
# original step‑size schedule – it works well for the new family as well)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient: 2 * a shifted for each active index
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional jitter to mitigate stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

# Simple stochastic hill‑climb (more aggressive than the reference)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]                 # move a fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a
