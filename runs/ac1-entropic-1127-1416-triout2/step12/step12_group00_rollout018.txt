def solve_lp_for_f(f, M, bound=1000.0):
    n = len(f)
    rows = 2*n -1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        # Compute list of j
        # for each j, coefficient f[k-j]
        # using slice
        # May assign via vectorized
        j_range = np.arange(j_min, j_max+1)
        A[k, j_range] = f[k - j_range]
    # define bounds for x: (0, bound)
    bounds = [(0, bound) for _ in range(n)]
    c = -np.ones(n)  # maximize sum => minimize -sum
    b_ub = np.full(rows, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    return res

"""
non‑negative step‑function optimiser (≈ 1000 s)

The search proceeds in three stages:

1. **Broad generation** – a fast sweep of several parametric families
   (power‑law, exponential, beta, Gaussian, sine) together with many
   random Dirichlet draws.  The top‑k candidates (lowest score) are kept.

2. **Linear‑programming refinement** – for each kept candidate we solve
   a small LP:
        maximise   Σ g_j
        subject to 0 ≤ g_j ≤ 1000,
                    (f * g)_k ≤ M    for all k,
   where * denotes ordinary convolution, f is the current vector and
   M = max(f*f).  The solution g can place a lot of mass on coordinates
   where f is tiny, therefore after normalising g the new vector has a
   strictly smaller maximum convolution.  We repeat this step a few
   times (or until no improvement).

3. **Smooth‑max polishing** – an Adam‑type gradient descent on a
   log‑sum‑exp surrogate for the max‑convolution, with Euclidean projection
   onto the probability simplex after each step.  This finishes the
   optimisation and yields the final sequence.

All phases respect the global time‑budget; the best sequence found is
returned as a plain list of floats.
"""
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when sum(a)=1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Parametric families (used for the initial broad search)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    tot = a.sum()
    if tot == 0:
        return np.full(n, 0.0)
    return a / tot


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    tot = a.sum()
    if tot == 0:
        return np.full(n, 0.0)
    return a / tot


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    tot = a.sum()
    if tot == 0:
        return np.full(n, 0.0)
    return a / tot


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    tot = a.sum()
    if tot == 0:
        return np.full(n, 0.0)
    return a / tot


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    tot = a.sum()
    if tot == 0:
        return np.full(n, 0.0)
    return a / tot

# ----------------------------------------------------------------------


def baseline_search(
    n: int, deadline: float, max_keep: int = 6, random_draws: int = 3000
):
    best = []  # (score, vector)

    def consider(v: np.ndarray):
        v = np.maximum(v, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        if len(best) < max_keep:
            best.append((sc, v.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, v.copy())
            best.sort(key=lambda x: x[0])

    # power‑law
    for p in np.linspace(0.1, 2.0, 101):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # exponential
    for lam in np.linspace(0.0001, 0.2, 200):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # beta
    al_vals = np.linspace(0.2, 4.0, 31)
    be_vals = np.linspace(0.2, 4.0, 31)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    # Gaussian
    for sigma in np.linspace(0.1 * n, 0.5 * n, 25):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # sine‑power
    for p in np.linspace(0.1, 6.0, 31):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # random Dirichlet (wide concentration range, incl. very sparse)
    concs = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    for conc in concs:
        draws = random_draws // len(concs)
        for _ in range(draws):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            consider(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the (2·n‑1) × n matrix A such that (f * g) = A·g.
    Row k contains f_{k‑j} at column j whenever the index is inside [0,n‑1].
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        if j_min > j_max:
            continue
        idx = np.arange(j_min, j_max + 1)
        A[k, idx] = f[k - idx]
    return A


def lp_refine(start_vec: np.ndarray, deadline: float, max_iters: int = 8):
    """
    Repeatedly solve the LP described in the paper, always keeping the
    best feasible vector (lowest max‑convolution).  Returns the best vector
    found before ``deadline``.
    """
    a = start_vec.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = full_convolution(a).max()
        A = build_conv_matrix(a)
        rows = A.shape[0]
        b = np.full(rows, M, dtype=np.float64)

        # maximise Σ g_j  ⇔ minimise –Σ g_j
        c = -np.ones(a.shape[0], dtype=np.float64)
        bounds = [(0.0, 1000.0) for _ in range(a.shape[0])]

        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": max(1, int(deadline - time.time()))},
            )
        except Exception as e:  # pragma: no cover
            print(f"[LP {it}] linprog crashed: {e}")
            break

        if not res.success:
            print(f"[LP {it}] linprog did not succeed (status={res.status})")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            print(f"[LP {it}] trivial solution (sum≈0)")
            break

        # Normalise g to have unit sum (the same normalisation used by evaluate_sequence)
        g_norm = g / sum_g
        g_norm = project_to_simplex(g_norm)

        # 1‑step replacement – the theory guarantees a non‑increase of the bound
        a_candidate = g_norm.copy()
        cand_val = compute_score(a_candidate)

        print(
            f"[LP {it}] M={M:.6e}, sum_g={sum_g:.3f}, score={cand_val:.6f}"
        )

        if cand_val < best_val - 1e-9:
            best_val, best_a = cand_val, a_candidate.copy()
            a = a_candidate  # continue from the improved point
        else:
            # no improvement; stop early
            break

    return best_a


# ----------------------------------------------------------------------
# Adam‑based smooth‑max optimiser (final polishing)
# ----------------------------------------------------------------------


def smooth_max_adam(a0: np.ndarray, deadline: float,
                    tau_start: float = 0.2,
                    tau_end: float = 1e-5,
                    tau_decay: float = 0.985,
                    step0: float = 0.12) -> np.ndarray:
    """
    Minimise a smooth approximation of max(conv) with an Adam optimiser.
    The smooth objective is
          τ·log Σ exp( conv / τ )
    where τ is slowly annealed.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    best_a = a.copy()
    best_val = compute_score(a)

    # Adam moments
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tau = tau_start

    while time.time() < deadline:
        t += 1
        if t % 150 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)           # length m
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()        # soft‑max weights

        # gradient of the smooth max (2·a ⋆ w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * full[idx]

        # Adam update
        m1 = beta1 * m1 + (1.0 - beta1) * grad
        m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)
        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)
        step = step0 * m1_hat / (np.sqrt(m2_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     n: int = 2048,
                     **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` under the given wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1.  Broad baseline exploration (≈ 200 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 600.0
    if time.time() < baseline_deadline:
        cand_vecs = baseline_search(
            n, baseline_deadline, max_keep=6, random_draws=3000
        )
        for v in cand_vecs:
            val = compute_score(v)
            if val < best_val:
                best_val, best_seq = val, v.tolist()
        print(f"[baseline] best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2.  Linear‑programming refinement (≈ 300 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        lp_deadline = time.time() + 300.0
        a_start = np.array(best_seq, dtype=float)
        a_refined = lp_refine(a_start, lp_deadline, max_iters=6)
        val = compute_score(a_refined)
        print(f"[LP‑refine] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 3.  Adam smooth‑max polishing (≈ remaining time)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        polish_deadline = deadline - 10.0
        a0 = np.array(best_seq, dtype=float)
        a_final = smooth_max_adam(a0, polish_deadline)
        val = compute_score(a_final)
        print(f"[polish] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 4.  Safeguard – fall back to a uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5.  Report using the official evaluator
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
