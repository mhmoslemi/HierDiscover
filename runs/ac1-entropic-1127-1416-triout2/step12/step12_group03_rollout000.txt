"""
Propose a step‑function sequence that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

using a multi‑restart pipeline:

1.  Load a fallback if available.
2.  Choose a modest length (2048) – fast enough for many iterations.
3.  Scan a handful of analytic families (arcsine‑type, left‑heavy,
    exponential, mixed) and keep the best as a start.
4.  Refine the start with three complementary optimisers, each run
    for a time‑slice that respects the global 1000 s budget:
       • Exponentiated‑gradient (mirror descent) on the true max‑convolution.
       • L‑BFGS on a smooth‑max surrogate (β‑homotopy from 4 → 5000).
       • Projected sub‑gradient descent on the exact max‑convolution.
5.  Optional symmetrisation (the problem is invariant under reversal).
6.  A final simulated‑annealing hill‑climb for fine local improvements.
7.  Return the best sequence found (as a plain Python list).

All heavy work is done with FFT‑based convolutions; the code never
creates closures or lambdas and respects the single‑CPU constraint.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize  # L‑BFGS optimiser


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution via real FFT.
    Result length = len(a) + len(b) - 1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (with sum(a)=1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families used for quick initialisation
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    α=0.5 gives the discrete arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Heavy on the left side: a_i ∝ (i+½)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mix of uniform and exponential‑U."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing shape (sorted powers of uniform draws)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Pure exponential‑U with random decay."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

def eg_optimize(start_a: np.ndarray, deadline: float,
                max_iters: int = 800_000, eta0: float = 0.6):
    """Exponentiated‑gradient (mirror‑descent) on the true max‑convolution."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]  # diagonal term appears once

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray, deadline: float,
                    max_iters: int = 5_000_000, step0: float = 0.07):
    """Projected sub‑gradient descent on the exact max‑convolution."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 2_000_000):
    """Simple simulated‑annealing hill‑climb on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max surrogate + L‑BFGS optimiser
# ----------------------------------------------------------------------


class SmoothMax:
    """Smooth‑max surrogate f_β(a) = (1/β)·log Σ exp(β·(a*a)_k)."""

    def __init__(self, beta: float, n: int):
        self.beta = beta
        self.n = n

    def f(self, z: np.ndarray) -> float:
        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # scale to avoid overflow
        exp_w = np.exp(self.beta * (conv - maxc))
        sum_exp = exp_w.sum()
        # smooth max value; the constant maxc is added back
        return maxc + (np.log(sum_exp) / self.beta)

    def grad(self, z: np.ndarray) -> np.ndarray:
        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(self.beta * (conv - maxc))
        sum_exp = exp_w.sum()
        w = exp_w / sum_exp                      # softmax weights over convolution entries
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)     # length = 2n‑1
        grad_a = 2.0 * conv_rev[self.n - 1:self.n - 1 + self.n]
        dot_ag = a.dot(grad_a)
        # Jacobian of softmax
        return a * grad_a - dot_ag * a


def lbfgs_optimize(start_a: np.ndarray, deadline: float,
                   beta_schedule=None, maxiter_per_stage=200):
    """
    L‑BFGS minimisation of the smooth‑max surrogate.
    The schedule runs several β values (from gentle to very sharp).
    """
    if beta_schedule is None:
        beta_schedule = [4, 20, 100, 500, 2000, 5000]

    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise softmax parameters

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        obj = SmoothMax(beta, n)
        # L‑BFGS on the softmax parametrisation
        res = minimize(obj.f, z, method='L-BFGS-B', jac=obj.grad,
                       options={'maxiter': maxiter_per_stage,
                                'disp': False,
                                'ftol': 1e-12})
        z = res.x
        a = softmax(z)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] L‑BFGS (β={beta}) improved to {best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    `evaluate_sequence` function.  Returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0            # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load a fallback candidate if supplied by the harness.
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose problem size (a modest 2048 works well for many iterations).
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    if fallback_len >= 2048:
        n = fallback_len
    else:
        n = 2048
    print(f"[{time.time():.1f}] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Quick scan of analytic families.
    # ------------------------------------------------------------------
    families = []

    # symmetric product (arcsine‑type)
    alphas = np.linspace(0.25, 1.2, 20)
    families.append(('sym', alphas, lambda a: symmetric_product_family(n, a)))

    # left‑heavy power
    betas = np.linspace(0.3, 2.0, 18)
    families.append(('left', betas, lambda b: left_heavy_power_family(n, b)))

    # exponential‑U
    gammas = np.linspace(0.0, 10.0, 22)
    families.append(('exp', gammas, lambda g: exp_u_family(n, g)))

    # mixed uniform + exponential‑U
    lams = np.linspace(0.0, 1.0, 11)
    gammas2 = np.linspace(0.0, 12.0, 13)
    families.append(('mix',
                     [(lam, gam) for lam in lams for gam in gammas2],
                     lambda p: mixed_uniform_u_family(n, p[0], p[1])))

    # Random monotone start (single draw)
    families.append(('rand_monotone', [None], lambda _: random_monotone_start(n)))

    # Random exponential‑U start
    families.append(('rand_exp', [None], lambda _: random_u_start(n)))

    # Evaluate each candidate quickly and keep the best.
    for name, params, maker in families:
        for p in params:
            if time.time() >= deadline - 5.0:
                break
            if p is None:
                vec = maker(None)
            else:
                vec = maker(p)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best ({name}) ⇒ {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Normalise the working vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)
    print(f"[{time.time():.1f}] after scan best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Time allocation for the remaining phases.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    eg_time   = min(180.0, remaining * 0.20)
    lb_time   = min(300.0, remaining * 0.30)
    sub_time  = min(300.0, remaining * 0.30)
    sa_time   = max(0.0, remaining - (eg_time + lb_time + sub_time))

    # ------------------------------------------------------------------
    # 5.  Exponentiated‑gradient phase.
    # ------------------------------------------------------------------
    if time.time() + eg_time < deadline:
        print(f"[{time.time():.1f}] EG phase (~{eg_time:.1f}s)")
        a, sc = eg_optimize(a, time.time() + eg_time,
                            max_iters=900_000, eta0=0.6)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] EG improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  L‑BFGS on smooth‑max surrogate.
    # ------------------------------------------------------------------
    if time.time() + lb_time < deadline:
        print(f"[{time.time():.1f}] L‑BFGS phase (~{lb_time:.1f}s)")
        a, sc = lbfgs_optimize(a, time.time() + lb_time,
                               beta_schedule=[4, 20, 100, 500, 2000, 5000],
                               maxiter_per_stage=250)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] L‑BFGS improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Projected sub‑gradient descent (many cheap steps).
    # ------------------------------------------------------------------
    if time.time() + sub_time < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient phase (~{sub_time:.1f}s)")
        a, sc = subgrad_optimize(a, time.time() + sub_time,
                                 max_iters=7_000_000, step0=0.07)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] Sub‑gradient improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Optional symmetrisation (the problem is reversal‑invariant).
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        best_seq = a.tolist()
        print(f"[{time.time():.1f}] Symmetrisation improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    if time.time() + sa_time < deadline:
        print(f"[{time.time():.1f}] Hill‑climb (SA) phase (~{sa_time:.1f}s)")
        a, sc = hill_climb_sa(a, time.time() + sa_time,
                              max_steps=2_500_000)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] Hill‑climb improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10.  Final evaluation using the provided `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
