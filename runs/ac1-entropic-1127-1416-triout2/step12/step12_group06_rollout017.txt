        i, j = np.random.randint(n, size=2)

"""
search_step_function_v2.py

A multi‑stage optimiser for the ``evaluate_sequence`` objective.

Stages
------
1.  Very broad parametric sweep (power‑law, geometric, beta,
    mixed plateau+power‑law and a more flexible plateau‑power family).
2.  Smooth‑max gradient descent (softmax approximation of the
    max‑convolution) with projection onto the probability simplex.
3.  Simulated‑annealing hill‑climb for final polishing.

The algorithm keeps several timing checkpoints so that the whole
search finishes comfortably within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two (real) vectors using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The vector is assumed to be already non‑negative and sum‑normalised.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}   (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{alpha} (1-x_i)^{beta},
    where x_i = (i+0.5)/n.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """
    First L = floor(L_frac * n) entries are equal to the first entry of the
    power‑law tail, the remaining entries follow that tail.
    """
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def plateau_power_custom(n: int, L_frac: float, p: float, r: float) -> np.ndarray:
    """
    More flexible plateau‑power shape.
    L = floor(L_frac * n) leading entries have height `c`,
    the remaining entries follow a power‑law tail with exponent `p`
    and a scaling factor `d`.  The ratio `r = c / d` is a free parameter.
    """
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    # unscaled tail (starts with 1)
    tail_unscaled = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    sum_tail = tail_unscaled.sum()
    # solve for scaling factor d so that sum(c*L + d*sum_tail) = 1
    # with c = r * d
    d = 1.0 / (L * r + sum_tail)
    c = r * d
    head = np.full(L, c, dtype=np.float64)
    tail = d * tail_unscaled
    a = np.concatenate([head, tail])
    a /= a.sum()                     # safety – numerical errors
    return a


# ----------------------------------------------------------------------
# Parametric sweep – tries a wide variety of families and keeps the best.
# ----------------------------------------------------------------------
def parametric_sweep(n: int, time_deadline: float) -> np.ndarray:
    """Search a large set of parametric families and return the best vector."""
    best_vec = None
    best_val = float('inf')

    # ----- power‑law (coarse) -----
    for p in np.linspace(0.10, 2.00, 191):
        if time.time() >= time_deadline:
            return best_vec
        a = power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ----- power‑law (fine around promising region) -----
    if best_vec is not None:
        for p in np.linspace(0.30, 1.20, 451):
            if time.time() >= time_deadline:
                return best_vec
            a = power_law(n, p)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- geometric -----
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 91),
            np.linspace(0.990, 0.9999, 110))):
        if time.time() >= time_deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ----- mixed plateau + power‑law -----
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= time_deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- flexible plateau‑power (extra degree of freedom) -----
    for L_frac in np.linspace(0.01, 0.50, 25):
        for p in np.linspace(0.20, 1.20, 35):
            for log_r in np.linspace(-1.0, 1.0, 21):   # r in [0.1,10]
                if time.time() >= time_deadline:
                    return best_vec
                r = 10.0 ** log_r
                a = plateau_power_custom(n, L_frac, p, r)
                if a is None:
                    continue
                val = compute_score(a)
                if val < best_val:
                    best_val, best_vec = val, a

    # ----- beta‑shape -----
    alphas = np.linspace(-0.8, 3.0, 20)
    betas = np.linspace(-0.8, 3.0, 20)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= time_deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ----- random convex combos of three simple shapes -----
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    power = power_law(n, 0.5)
    geom = geometric(n, 0.98)

    basic = [uniform, power, geom]

    for _ in range(2500):
        if time.time() >= time_deadline:
            break
        coeffs = np.random.rand(3)
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basic))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (softmax approximation of max)
# ----------------------------------------------------------------------
def smooth_max_descent(a0: np.ndarray, time_deadline: float,
                       lambda_start: float = 5.0,
                       lambda_end: float = 30.0,
                       steps_target: int = 2000) -> np.ndarray:
    """
    Minimise a smooth approximation of max(conv(a,a)).
    Uses a softmax with a temperature that is gradually lowered.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]
    t = 0
    while time.time() < time_deadline and t < steps_target:
        # linearly increase the softmax sharpness
        lam = lambda_start + (lambda_end - lambda_start) * (t / steps_target)

        # convolution a*a
        conv = full_convolution(a)
        m = conv.max()
        # stable softmax weights
        exp_term = np.exp(lam * (conv - m))
        w = exp_term / exp_term.sum()

        # gradient of the smooth max: g = 2 * (a * reverse(w))
        rev_w = w[::-1]
        grad_full = full_convolution_two(a, rev_w)                      # length 3n-2
        # extract the n‑dimensional gradient (reverse the slice as derived)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n][::-1]

        # normalise step size (makes the update comparable across dimensions)
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        step = 0.12 / (norm + 1e-12)          # 0.12 is a modest base step

        a_candidate = a - step * grad
        a_candidate = project_to_simplex(a_candidate)

        cur_val = compute_score(a_candidate)
        if cur_val < best_val:
            best_val, best_a = cur_val, a_candidate.copy()

        a = a_candidate
        t += 1

        # occasional tiny jitter to avoid exact stagnation
        if t % 300 == 0:
            jitter = np.random.randn(n) * (0.001 / math.sqrt(t + 1))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb_sa(a_start: np.ndarray, time_deadline: float,
                  max_iter: int = 800_000) -> np.ndarray:
    """
    Greedy stochastic moves with a simple temperature schedule.
    Allows occasional uphill moves to escape local minima.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    n = a.shape[0]

    # initial temperature – a fraction of the current score
    T0 = 0.02 * best_score
    it = 0
    while time.time() < time_deadline and it < max_iter:
        # linear cooling
        T = T0 * (1.0 - it / max_iter)

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a small fraction of mass from i to j
        delta = 0.015 * a[i]               # 1.5 % of source entry
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        new_score = compute_score(cand)
        dE = new_score - cur_score
        accept = False
        if dE < 0:
            accept = True
        else:
            # Metropolis criterion
            if T > 0 and np.random.rand() < math.exp(-dE / T):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, cand.copy()

        it += 1

        # occasional global jitter after long stagnation
        if it % 150_000 == 0:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            cur_score = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback to a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad parametric sweep (≈ 250 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    n = 2048
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Smooth‑max gradient descent (majority of remaining time)
    # -----------------------------------------------------------------
    grad_deadline = deadline - 150.0
    if best_seq is not None and time.time() < grad_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = smooth_max_descent(a0, grad_deadline,
                                        lambda_start=5.0,
                                        lambda_end=30.0,
                                        steps_target=2500)
        val = compute_score(a_polished)
        print(f"[smooth‑max] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb_sa(a_start, hill_deadline, max_iter=900_000)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation and reporting
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
