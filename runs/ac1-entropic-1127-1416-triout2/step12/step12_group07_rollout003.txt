def full_convolution_fft(a):
    n = len(a)
    m = 2*n-1
    nfft = 1 << ((m-1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    C = A * A
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv

def cross_corr_fft(a, w):
    # compute scalar cross-correlation of a (len n) with w (len 2n-1)
    n = len(a)
    m = len(w)  # 2n-1
    L = n + m - 1

def cross_corr_fft(a, w):
    n = a.shape[0]
    m = w.shape[0]  # 2n-1
    L = n + m - 1
    nfft = 1 << ((L - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    W_rev = np.fft.rfft(w[::-1], n=nfft)
    C = A * W_rev
    conv_full = np.fft.irfft(C, n=nfft)  # length L
    start = n - 1
    return conv_full[start:start + n]

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm works in three stages:

1. **Broad parametric sweep** – generate many candidate families
   (power‑law, geometric, beta‑polynomial, cosine‑power,
   double‑sided power‑law, piecewise‑constant blocks, triangular,
   U‑shaped).  Keep the best few (by the true objective
   ``2·n·max(conv)``).

2. **Smooth‑max Adam optimisation** – starting from each top candidate,
   run a projected Adam optimiser on a smooth approximation of the
   objective, ``smooth_max(conv) = (1/β)·log Σ exp(β·conv)``.
   The smoothing parameter β is increased during the run so that the
   optimiser first sees a smooth landscape and finally focuses on the
   true max.  After each step the iterate is Euclidean‑projected onto the
   probability simplex (Σa=1, a_i ≥ 0).  The best point seen (according to
   the exact objective) is retained.

3. **Final greedy mass‑transfer hill‑climb** – a short stochastic
   refinement that moves a small fraction of probability mass between two
   random coordinates and accepts only improving moves.  This step
   exploits the remaining wall‑clock budget.

The whole search respects the supplied time budget (default 1000 s) and
returns a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (length 2·len(a)‑1) using a real FFT.
    """
    n = a.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    C = A * A
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def cross_corr_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ w)[i] = Σ_t a[t]·w[t+i]  for i=0,…,n‑1.
    w must have length 2·n‑1 (as produced by ``full_convolution_fft``).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    L = n + m - 1
    nfft = _next_pow2(L)
    A = np.fft.rfft(a, n=nfft)
    W_rev = np.fft.rfft(w[::-1], n=nfft)
    C = A * W_rev
    conv_full = np.fft.irfft(C, n=nfft)
    start = n - 1
    return conv_full[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a normalised vector Σa = 1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, beta: float) -> np.ndarray:
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    a /= a.sum()
    return a


def cosine_power(n: int, expo: float) -> np.ndarray:
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    blocks = rng.dirichlet(np.ones(K))
    blocks = np.sort(blocks)[::-1]
    repeats = np.full(K, n // K, dtype=int)
    remainder = n - repeats.sum()
    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)[:n]
    a /= a.sum()
    return a


def double_powerlaw(n: int, p: float, offset: float) -> np.ndarray:
    """
    Symmetric sum of two power‑law tails:
        a_i ∝ (i+off)^(-p) + (n‑1‑i+off)^(-p)
    """
    idx = np.arange(n, dtype=np.float64) + offset
    rev = (n - 1 - np.arange(n, dtype=np.float64)) + offset
    a = idx ** (-p) + rev ** (-p)
    a /= a.sum()
    return a


def triangular(n: int) -> np.ndarray:
    """Symmetric triangle: 1,2,…,⌈n/2⌉,…,2,1."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(a, n + 1 - a)
    a /= a.sum()
    return a


def u_shape(n: int) -> np.ndarray:
    """U‑shaped (large at the ends, small in the centre)."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a = np.maximum(a, n + 1 - a)
    a = a.astype(np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Global parametric sweep (keeps top‑K candidates)
# ----------------------------------------------------------------------
def sweep_candidates(n: int, deadline: float, top_k: int = 12) -> list[tuple[float, np.ndarray]]:
    """Explore families (including double‑power‑law) and random blocks."""
    best = []                     # list of (score, vector), sorted ascending

    def consider(vec: np.ndarray):
        if vec is None or np.isnan(vec).any():
            return
        sc = compute_score(vec)
        if len(best) < top_k:
            best.append((sc, vec))
            best.sort(key=lambda x: x[0])
        else:
            if sc < best[-1][0]:
                best[-1] = (sc, vec)
                best.sort(key=lambda x: x[0])

    rng = np.random.default_rng(123456)

    # ----- power‑law with offset -----
    for p in np.linspace(0.2, 2.0, 45):
        if time.time() >= deadline:
            break
        for off in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]:
            if off >= n:
                continue
            consider(powerlaw_offset(n, p, off))

    # ----- geometric decay -----
    for r in np.linspace(0.3, 0.999, 80):
        if time.time() >= deadline:
            break
        consider(geometric(n, r))

    # ----- beta‑shape (polynomial tail) -----
    for beta in np.linspace(0.2, 40.0, 50):
        if time.time() >= deadline:
            break
        consider(beta_shape(n, beta))

    # ----- cosine‑power -----
    for e in np.linspace(0.1, 8.0, 40):
        if time.time() >= deadline:
            break
        consider(cosine_power(n, e))

    # ----- double‑sided power‑law -----
    for p in np.linspace(0.2, 2.0, 30):
        if time.time() >= deadline:
            break
        for off in [0.5, 1.0, 2.0, 5.0, 10.0]:
            consider(double_powerlaw(n, p, off))

    # ----- deterministic shapes (triangle & U) -----
    consider(triangular(n))
    consider(u_shape(n))

    # ----- random block shapes (coarse search) -----
    trials = 3000
    for K in [10, 20, 30, 50, 80]:
        for _ in range(trials // 5):
            if time.time() >= deadline:
                break
            consider(block_shape(n, K, rng))

    return best


# ----------------------------------------------------------------------
# Smooth‑max Adam refinement
# ----------------------------------------------------------------------
def refine_adam(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smoothed max of the convolution.
    Returns the best point found and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # smoothing schedule
    beta = 5.0                       # initial inverse temperature
    beta_inc = 1.02
    beta_max = 200.0

    # learning‑rate schedule
    lr = 0.5
    lr_decay = 0.9

    # bookkeeping
    best_a = a.copy()
    conv = full_convolution_fft(a)
    best_sc = float(2.0 * n * conv.max())
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1

        # current convolution
        conv = full_convolution_fft(a)
        maxc = conv.max()
        sc = float(2.0 * n * maxc)

        # keep best exact point
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # soft‑max weights (smooth approximation of the max)
        # we work in the log‑domain to avoid overflow
        exps = np.exp(beta * (conv - maxc))
        w = exps / exps.sum()

        # gradient of the smooth max: 2·(a ⋆ w)
        g = 2.0 * cross_corr_fft(a, w)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # schedule adjustments
        if beta < beta_max:
            beta = min(beta_max, beta * beta_inc)

        if it % 200 == 0:
            lr *= lr_decay

        # occasional random jitter to escape flat plateaus
        if no_improve > 3000:
            jitter = np.random.randn(n) * (0.001 / np.sqrt(it))
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

    return best_a, best_sc


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (tiny final polish)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float, max_iter: int = 400_000) -> tuple[np.ndarray, float]:
    a = a_start.copy()
    n = a.shape[0]
    conv = full_convolution_fft(a)
    best_sc = float(2.0 * n * conv.max())
    best_a = a.copy()
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:
            cand[i] = 0.0
        cand = project_to_simplex(cand)

        conv = full_convolution_fft(cand)
        sc = float(2.0 * n * conv.max())
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.987
            no_improve += 1

        if no_improve > 80_000:
            jitter = np.random.randn(n) * 0.0008
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1

    return best_a, best_sc
