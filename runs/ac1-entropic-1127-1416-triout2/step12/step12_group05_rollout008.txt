        # if we have been idle for a while, inject a tiny random perturbation
        if time.time() - last_improve > 30.0:
            noise = np.random.rand(n) * 1e-3
            a = normalize(a + noise)
            step_scale = 0.25
            last_improve = time.time()

    # ------------------------------------------------------------------
    # 5. final Adam refinement on the smooth‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] Starting final Adam refinement")
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=best_a,
            deadline=deadline,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=8000.0,
            lr=0.010,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] Adam final gave {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)   # includes all security checks
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
