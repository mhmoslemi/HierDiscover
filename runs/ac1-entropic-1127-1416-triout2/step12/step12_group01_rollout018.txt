def project_simplex(v):
    # v is array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def projected_gradient_descent(a0, deadline, eta0=0.01, T0=0.01, Tmin=1e-8):
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_val = compute_score(a)
    T = T0
    it = 0
    while time.time() < deadline and T > Tmin:
        it += 1
        conv = full_convolution(a)
        maxc = conv.max()
        # compute softmax weights
        exp_arg = (conv - maxc) / T
        # subtract max to avoid overflow (already sub)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            break
        w = w / w_sum
        # compute gradient g = 2*n*(cor + a)
        w_rev = w[::-1]
        rev = np.convolve(a, w_rev)
        cor = rev[(2*n-2) - np.arange(n)]
        grad = 2.0 * n * (cor + a)
        # step size adapt
        eta = eta0 / np.sqrt(it)
        # gradient descent: a_new = a - eta * grad
        a_new = a - eta * grad
        # project
        a_new = project_onto_simplex(a_new)
        cur_val = compute_score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best = a_new.copy()
        a = a_new
        # temperature cooling
        T *= 0.98
    return best
```

"""
Optimiser for ``evaluate_sequence``

Overview
--------
* Use a moderate problem size (n≈1024) – fast enough to allow many
  iterations while still giving a strong bound.
* Global exploration: sweep several parametric families (power‑law,
  exponential, beta with extreme parameters, Gaussian, sine‑power) and
  a large pool of Dirichlet draws.  The top few candidates (by the exact
  objective 2·n·max(conv)) are retained.
* Fine polishing: minimise a smooth approximation of the max‑convolution
  via projected gradient descent on the log‑sum‑exp surrogate.
  The gradient of the surrogate can be computed with two FFT‑based
  convolutions per iteration and the iterate is projected back onto the
  probability simplex.
* Optional extra polish with the original exponentiated‑gradient (EG)
  scheme and a short simulated‑annealing run for final tweaks.
* All steps respect a 1000 s wall‑clock budget; timing checks are used
  throughout the search.

The returned list of floats is a feasible non‑negative sequence whose
score (as reported by ``evaluate_sequence``) should be below the previous
best 1.510445 and aim at the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using real FFT (full length)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return full_convolution_fft(a, a)


# ----------------------------------------------------------------------
# Objective (exact) – assumes sum(a)=1
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (onto {x≥0, Σx=1})
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1          # zero‑based index of last positive
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all return a normalized probability vector)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (p>0)."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta_ext(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete beta shape with arbitrary positive α,β:
        a_i ∝ i^{α‑1}·(n‑i+1)^{β‑1}
    (uses logs for numerical stability when α or β < 1).
    """
    i = np.arange(1, n + 1, dtype=float)
    log_a = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    # shift to avoid overflow
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """a_i ∝ exp(-½·((i‑μ)/(σ))²) with σ = sigma_frac·n."""
    x = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    sigma = max(sigma_frac * n, 1e-12)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^{p}  (p>0)."""
    i = np.arange(n) + 0.5
    a = np.sin(np.pi * i / n) ** p
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total

def baseline_search(n: int, deadline: float, max_keep: int = 12):
    """
    Sweep the parametric families and many random Dirichlet draws.
    Returns up to ``max_keep`` best normalized vectors.
    """
    best = []   # list of (score, vector)

    def consider(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law
    for p in np.linspace(0.01, 3.0, 250):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))

    # 2) exponential
    for lam in np.linspace(1e-5, 0.3, 250):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # 3) beta (including extreme α,β < 1)
    al_vals = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    be_vals = al_vals
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta_ext(n, alpha, beta))

    # 4) Gaussian
    for sigma_frac in np.linspace(0.05, 0.30, 30):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma_frac))

    # 5) sine‑power
    for p in np.linspace(0.5, 5.0, 30):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # 6) random Dirichlet draws
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per = 600          # ≈ 4 800 draws total (fast)
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(alpha_vec)
            consider(vec)

    return [vec for _, vec in sorted(best, key=lambda x: x[0])]


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def smooth_gradient_descent(a0: np.ndarray, deadline: float,
                           eta0: float = 0.08,
                           T0_factor: float = 0.5,
                           Tmin: float = 1e-8) -> np.ndarray:
    """
    Minimise  f_T(a)=2·n·T·log Σ_k exp((a*a)_k / T)   (T→0 → max).
    Uses projected (additive) gradient descent with an exponentially
    decreasing temperature schedule.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    # initial temperature relative to the current max‑conv
    cur_conv = full_convolution(a)
    T = max(T0_factor * cur_conv.max(), Tmin)

    it = 0
    while time.time() < deadline and T > Tmin:
        it += 1

        # 1) convolution of a with itself
        conv = full_convolution(a)                     # length 2n‑1

        # 2) soft‑max weights w_k = exp(conv_k/T) / Σ
        maxc = conv.max()
        # subtract maxc for numerical stability (still negative or zero)
        w = np.exp((conv - maxc) / T)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            break
        w /= w_sum

        # 3) gradient: g = 2·n·(cor + a) where cor_i = Σ_k w_k·a_{k‑i}
        w_rev = w[::-1]
        rev = full_convolution_fft(a, w_rev)          # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)               # pick the required slice
        cor = rev[idx]

        grad = 2.0 * n * (cor + a)

        # 4) step size (decreasing)
        eta = eta0 / np.sqrt(it)

        # 5) projected additive update
        a_new = a - eta * grad
        a_new = project_onto_simplex(a_new)

        # 6) best tracking
        cur_val = compute_score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()

        a = a_new

        # 7) temperature cooling
        T = max(Tmin, T * 0.985)

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.0005 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a

# Original exponentiated‑gradient polish (kept as a fallback)
# ----------------------------------------------------------------------


def eg_polish(a0: np.ndarray, deadline: float,
              eta0: float = 0.7, tol: float = 1e-10) -> np.ndarray:
    """Sub‑gradient EG optimiser on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * tol
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        for k in active:
            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)
            i_idx = np.arange(i0, i1 + 1)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        a = a / a.sum()

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if it % 500 == 0:
            jitter = np.random.randn(n) * (eta0 * 1e-3 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a

# Light simulated‑annealing (final escape from local minima)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 1e-6,
                        cooling: float = 0.9993,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 5e-7) -> np.ndarray:
    """Metropolis hill‑climb (accepting uphill moves) on the exact objective."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling)

        if it % 3500 == 0:
            # occasional global jitter
            jitter = np.random.randn(n) * (step_scale * 0.015)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 1024 provides a good speed / quality trade‑off
    # -----------------------------------------------------------------
    n = 1024

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previous solution (if available)
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 300.0   # reserve ≈5 min for polishing + anneal
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smoothed gradient‑descent polishing (many restarts)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        # primary polish of the current best
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_gradient_descent(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth‑GD‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # extra random restarts while time permits
        while time.time() < polish_deadline:
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_gradient_descent(a_start, polish_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Optional fallback to the original EG polish (covers any gaps)
    # -----------------------------------------------------------------
    eg_deadline = deadline - 60.0
    if best_seq is not None and time.time() < eg_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = eg_polish(a0, eg_deadline)
        val = compute_score(refined)
        print(f"[eg‑polish‑fallback] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # -----------------------------------------------------------------
    # 4. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 5. Fallback to a uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # -----------------------------------------------------------------
    # 6. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
