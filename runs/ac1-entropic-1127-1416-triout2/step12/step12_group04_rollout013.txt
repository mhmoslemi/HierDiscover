def smoothed_gradient_descent(a, deadline, max_iters=200000, init_lr=0.2, init_lambda=10.0, final_lambda=200.0):
    n = a.shape[0]; m=2*n-1.
    a = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    beta1=0.9; beta2=0.999; eps=1e-8
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute conv
        conv = convolution_fft(a, a)  # size 2n-1
        lam = init_lambda + (final_lambda - init_lambda) * (t / max_iters)
        # compute w = softmax(lam*conv)
        # we need for stability: subtract max.
        exp_arg = lam * conv
        max_exp = np.max(exp_arg)
        w = np.exp(exp_arg - max_exp)
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # gradient
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # size 3n-2
        # diag term
        diag_idx = 2*np.arange(n)
        diag_vals = w[diag_idx] * a
        grad = 2.0 * conv_rev[n-1:n-1+n] - diag_vals

        # Adam update
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * (grad * grad)
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        step = init_lr / np.sqrt(t)  # decaying lr
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = project_to_simplex(a)  # ensures sum=1, nonneg

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] GD improved to {sc:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def solve_lp_max_sum(f, M):
    n = len(f)
    m = 2*n -1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    c = -np.ones(n)
    bounds = [(0.0,None)]*n
    b = np.full(m, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'time_limit': 15})
    if not res.success:
        return None
    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return None
    return g / sum_g

"""
Propose a non‑negative step sequence that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

The algorithm consists of four phases:

1.  A quick analytic scan (uniform, product, power and random starts)
    to obtain a reasonable initial vector.
2.  Smoothed‑max gradient descent (Adam) on the exact objective.
    The “max” is approximated by a soft‑max with an increasing temperature
    λ, which yields gradients that take **all** convolution entries into account.
3.  A linear‑programming refinement step that solves the classic
    “mix with a new extreme point” LP.  The mixing weight t is chosen by a
    cheap 1‑D line‑search that directly minimises the resulting max‑convolution.
4.  A final stochastic hill‑climb (simulated‑annealing style) for fine tuning.

All phases respect the global 1000 s wall‑clock budget.  The routine returns
a plain Python list ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution via real FFT.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-0.5-i))^{-α} – U‑shaped, heavy ends."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-β} – monotone, heavy left side."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """w_i ∝ exp(-γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """w_i ∝ exp(-γ·(i‑mid)²) – discrete Gaussian."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (heavy left side)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped (product‑like) start."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (Adam)
# ----------------------------------------------------------------------


def smoothed_gradient_descent(start_a: np.ndarray,
                               deadline: float,
                               max_iters: int = 200_000,
                               init_lr: float = 0.2,
                               init_lambda: float = 8.0,
                               final_lambda: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max objective.
    Returns the best vector encountered and its hard score.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # 1) convolution a*a
        conv = convolution_fft(a, a)                 # length 2n‑1
        # 2) temperature schedule
        lam = init_lambda + (final_lambda - init_lambda) * (t / max_iters)
        # 3) soft‑max weights
        exp_arg = lam * conv
        exp_arg -= np.max(exp_arg)                    # numerical safety
        w = np.exp(exp_arg)
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # 4) gradient of the smoothed max
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)         # length 3n‑2
        diag_idx = 2 * np.arange(n)                  # positions 0,2,4,...
        grad = 2.0 * conv_rev[n - 1:n - 1 + n] - w[diag_idx] * a

        # 5) Adam update
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * (grad * grad)
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        step = init_lr / math.sqrt(t)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # 6) project onto simplex
        a = project_to_simplex(a)

        # occasional hard evaluation
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] GD improved to {sc:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (state‑of‑the‑art mixing)
# ----------------------------------------------------------------------


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0.
    Returns the normalised solution g / Σg or None on failure.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (size m × n)  A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)        # maximise Σg  → minimise −Σg
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # HiGHS is fast and respects a time limit (seconds)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 12})

    if not res.success:
        return None

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return None
    return g / sum_g


def best_mix_t(f: np.ndarray, g: np.ndarray,
               deadline: float, max_iters: int = 30) -> tuple[float, float]:
    """
    1‑D line search for the mixing weight t∈[0,1] that minimises
    max_k ((1−t)f + t g) * ((1−t)f + t g).
    Returns (t_opt, hard_score).
    """
    n = f.shape[0]
    conv_f = convolution_fft(f, f)
    conv_g = convolution_fft(g, g)
    conv_fg = convolution_fft(f, g)

    def max_for_t(t: float) -> float:
        # quadratic combination of the three convolutions
        combo = (1 - t) ** 2 * conv_f + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_g
        return float(combo.max())

    lo, hi = 0.0, 1.0
    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        m1 = max_for_t(t1)
        m2 = max_for_t(t2)
        if m1 < m2:
            hi = t2
        else:
            lo = t1
    t_opt = (lo + hi) / 2.0
    best_val = max_for_t(t_opt)
    best_score = 2.0 * n * best_val
    return t_opt, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Simple SA that swaps or transfers mass between two positions.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # (optional) use the supplied fallback sequence
    # ------------------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time() - start_time:.1f}] fallback score = {fb:.6f}")
        best_seq = [float(x) for x in height_sequence_1]               # type: ignore
        best_score = fb
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a problem size
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # use at least 2048 entries; larger N gives a richer search space
    n = max(2048, fallback_len)
    print(f"[{time.time() - start_time:.1f}] using n = {n}")

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (normalises internally)
    # ------------------------------------------------------------------
    def try_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best = {sc:.6f}")

    # ------------------------------------------------------------------
    # 0. Very cheap baseline
    # ------------------------------------------------------------------
    try_candidate(np.full(n, 1.0 / n, dtype=np.float64))

    # ------------------------------------------------------------------
    # 1. Analytic families + random starts
    # ------------------------------------------------------------------
    # product family (U‑shaped)
    alphas = np.concatenate([np.linspace(0.0, 5.0, 51),
                             np.linspace(5.5, 30.0, 25)])
    for a in alphas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(product_family(n, a))

    # left‑heavy power family
    betas = np.linspace(0.0, 20.0, 41)
    for b in betas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(left_heavy_power_family(n, b))

    # random starts
    for _ in range(120):
        if time.time() >= deadline - 250.0:
            break
        if np.random.rand() < 0.5:
            try_candidate(random_monotone_start(n))
        else:
            try_candidate(random_u_start(n))

    # ------------------------------------------------------------------
    # 2. Normalise the current best to a probability vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.asarray(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 3. Smoothed‑max gradient descent (Adam)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        print(f"[{time.time() - start_time:.1f}] starting smoothed GD")
        a, sc = smoothed_gradient_descent(a, deadline - 150.0)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] after GD best = {sc:.6f}")

    # ------------------------------------------------------------------
    # 4. LP‑based refinement with line‑search on the mixing weight
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print(f"[{time.time() - start_time:.1f}] starting LP refinement")
        lp_iters = 0
        while time.time() < deadline - 30.0 and lp_iters < 15:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            g_norm = solve_lp_max_sum(a, M)
            if g_norm is None:
                break
            t_opt, new_score = best_mix_t(a, g_norm, deadline)
            if new_score < best_score - 1e-9:
                best_score = new_score
                a = (1 - t_opt) * a + t_opt * g_norm
                a = project_to_simplex(a)
                print(f"[{time.time() - start_time:.1f}] LP step {lp_iters} t={t_opt:.3f} → {new_score:.6f}")
            else:
                # If no improvement, shrink future mixing weight range implicitly
                pass
            lp_iters += 1

    # ------------------------------------------------------------------
    # 5. Symmetrisation (the problem is symmetric)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time() - start_time:.1f}] symmetrisation improved to {sc_sym:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (SA)")
        a, sc = hill_climb_sa(a, deadline, max_steps=500_000)
        if sc < best_score:
            best_score = sc
            print(f"[{time.time() - start_time:.1f}] hill‑climb improved to {sc:.6f}")

    # ------------------------------------------------------------------
    # 7. Return the best vector found (as plain Python floats)
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished search – final score = {final_score:.6f}")
    return final_seq
