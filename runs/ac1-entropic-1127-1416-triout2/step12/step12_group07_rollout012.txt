assistantfinal**Solution Overview**

The script searches for a non‑negative step‑function (a list of heights) that minimises  

```
evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²)
```  

(the value is scale‑invariant, therefore we keep `sum(a)=1`).  

The algorithm consists of four stages:

1. **Broad parametric & random block search** – many families (`power‑law`,
   `geometric`, `beta`, `cosine‑power`, `exponential`, plus random piece‑wise
   constant blocks) are sampled on several values of `n` (`1024, 2048, 4096`).
   The `top_k` best candidates are kept.

2. **Smoothed gradient descent** – the non‑smooth “max” is approximated by a
   log‑sum‑exp with a temperature `λ`.  For a fixed `λ` the gradient can be
   written as a convolution, which is evaluated with FFTs.  Starting from the
   best candidate we run a continuation on `λ` (5 → 10 → 20 → … → 1024) with
   a diminishing step size and Euclidean projection onto the probability
   simplex.

3. **Fast stochastic hill‑climb** – from the polished vector we perform
   millions of tiny mass‑transfer moves (`a[i] → a[j]`).  Moves that improve
   the exact objective are always accepted; otherwise the step size is
   slowly reduced.  Periodic random jitter helps to escape plateaus.

4. **Fallback & final report** – if every stage fails a uniform vector is
   returned.  The official score (with the security wrapper) is printed.

All stages respect the global wall‑clock budget (default 1000 s).  
All helper functions are top‑level, there are no closures or lambdas, and the
code only uses `numpy` (no file or network I/O).

```python
"""
Search for a non‑negative step‑function that minimises
`evaluate_sequence`.  The pipeline:

1. Parametric sweep over many families (power‑law, geometric, beta,
   cosine‑power, exponential, random block shapes) for three lengths
   (1024, 2048, 4096).  The best `top_k` candidates (by the exact score)
   are retained.

2. Smoothed gradient descent using a log‑sum‑exp surrogate for `max`,
   with analytical gradient expressed via FFT‑based convolutions.
   The temperature `lam` is increased (continuation) while the step size
   is reduced; each iterate is projected onto the simplex.

3. Greedy stochastic hill‑climb that moves a small amount of mass between
   two random indices, always accepting improvements and slowly adapting
   the step size.

4. If nothing succeeds a uniform vector is returned.  The final score is
   obtained by the provided `evaluate_sequence` (which includes the
   security checks).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b using an FFT (real)."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1 - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)[: n + m - 1]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for the auto‑convolution a*a."""
    return full_convolution_fft(a, a)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a normalised vector
    (Σa == 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant; we therefore keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate shapes (all return a normalised vector)
# ----------------------------------------------------------------------
def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray | None:
    """a_i = (i+offset)^(-p) ; offset>0 to avoid division by zero."""
    if offset <= 0:
        return None
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    s = a.sum()
    if s == 0:
        return None
    return a / s


def geometric(n: int, r: float) -> np.ndarray | None:
    """a_i = r**i (geometric decay, r in (0,1))."""
    if not (0.0 < r < 1.0):
        return None
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    return a / s


def beta_shape(n: int, beta: float) -> np.ndarray | None:
    """a_i = (1 - i/(n-1))**beta  (monotonically decreasing)."""
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    s = a.sum()
    if s == 0:
        return None
    return a / s


def cosine_power(n: int, expo: float) -> np.ndarray | None:
    """a_i = cos(π i / (2 n))**expo (decreasing cosine shape)."""
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return None
    return a / s


def exponential(n: int, lam: float) -> np.ndarray | None:
    """a_i = exp(-lam * i) ; lam>0."""
    if lam <= 0:
        return None
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    if s == 0:
        return None
    return a / s


def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    """
    Coarse piecewise‑constant shape: generate K positive values,
    sort them decreasingly, repeat each value to fill the n entries.
    """
    blocks = rng.dirichlet(np.ones(K))
    blocks = np.sort(blocks)[::-1]                     # decreasing
    repeats = np.full(K, n // K, dtype=int)
    remainder = n - repeats.sum()
    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)[:n]
    return a / a.sum()

# Global parametric sweep (keeps top‑k candidates)
# ----------------------------------------------------------------------
def sweep_candidates(n: int, deadline: float, top_k: int = 20) -> list[tuple[float, np.ndarray]]:
    """Explore families + random block shapes; return list of (score, vector)."""
    best: list[tuple[float, np.ndarray]] = []

    def consider(vec: np.ndarray | None):
        if vec is None or np.isnan(vec).any():
            return
        sc = compute_score(vec)
        if len(best) < top_k:
            best.append((sc, vec))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec)
            best.sort(key=lambda x: x[0])

    rng = np.random.default_rng(8675309)

    # ----- power‑law with offset -----
    for p in np.linspace(0.1, 2.5, 70):
        if time.time() >= deadline:
            break
        for off in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]:
            if time.time() >= deadline:
                break
            consider(powerlaw_offset(n, p, off))

    # ----- geometric decay -----
    for r in np.linspace(0.55, 0.9999, 150):
        if time.time() >= deadline:
            break
        consider(geometric(n, r))

    # ----- beta‑shape (polynomial tail) -----
    for beta in np.linspace(0.5, 30.0, 60):
        if time.time() >= deadline:
            break
        consider(beta_shape(n, beta))

    # ----- cosine‑power -----
    for e in np.linspace(0.2, 12.0, 70):
        if time.time() >= deadline:
            break
        consider(cosine_power(n, e))

    # ----- exponential decay -----
    for lam in np.linspace(0.0005, 0.02, 70):
        if time.time() >= deadline:
            break
        consider(exponential(n, lam))

    # ----- random block shapes (coarse search) -----
    K = max(20, n // 100)          # a few dozen blocks
    trials = 5000
    for _ in range(trials):
        if time.time() >= deadline:
            break
        consider(block_shape(n, K, rng))

    return best


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def smooth_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed objective
        L_λ(a) = (1/λ) log Σ exp(λ·conv(a,a)).
    The temperature λ is increased geometrically (continuation).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    # continuation schedule
    lam = 5.0
    while time.time() < deadline and lam <= 2048.0:
        # number of inner iterations for the current λ
        inner = 250
        for it in range(inner):
            if time.time() >= deadline:
                break

            # full auto‑convolution
            conv = full_convolution(a)                     # length 2n‑1
            scaled = lam * conv
            # stable softmax
            max_scaled = np.max(scaled)
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # gradient of Σ w_k·conv_k  (see analysis)
            w_rev = w[::-1]
            conv_rev = full_convolution_fft(a, w_rev)     # length n + (2n‑1)‑1 = 3n‑2
            # pick the slice that corresponds to Σ_i a_i w_{i+j}
            idx = (len(w_rev) - 1) - np.arange(n)
            g = 2.0 * conv_rev[idx]

            # step size (decreases with λ and iteration number)
            step = 0.3 / (np.sqrt(it + 1) * lam)

            # projected update
            cand = a - step * g
            cand = project_to_simplex(cand)

            sc = compute_score(cand)
            if sc < best_sc:
                best_sc, best_a = sc, cand.copy()
                a = cand
            else:
                # occasional acceptance of non‑improving moves (helps escape)
                if np.random.rand() < 0.005:
                    a = cand
                # otherwise keep the current point
        lam *= 2.0                      # increase temperature
    return best_a


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Tiny random mass‑transfer moves, greedy acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    step_scale = 0.07          # fraction of the donor mass
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]                # move a fraction of mass
        if delta <= 0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.08, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        if no_improve > 150_000:
            # shake the current point to leave a flat plateau
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine respects a wall‑clock budget
    (`budget_s` seconds) and prints progress information.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # safety margin

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (if a previous good construction is present)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]    # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global parametric + random block sweep (several lengths)
    # -----------------------------------------------------------------
    lengths = [1024, 2048, 4096]                # three different resolutions
    sweep_deadline = deadline - 500.0           # reserve ~500 s for refinements
    top_candidates = []

    for n in lengths:
        if time.time() >= sweep_deadline:
            break
        print(f"[sweep] exploring length n = {n}")
        cand = sweep_candidates(n, sweep_deadline, top_k=12)
        top_candidates.extend(cand)

    if top_candidates:
        # keep the globally best candidate from the sweep
        top_candidates.sort(key=lambda x: x[0])
        best_sc, best_vec = top_candidates[0]
        print(f"[global sweep] best score = {best_sc:.6f} (n={len(best_vec)})")
        if best_sc < best_val:
            best_val, best_seq = best_sc, best_vec.tolist()

    # -----------------------------------------------------------------
    # 2. Smoothed gradient descent (continuation on λ)
    # -----------------------------------------------------------------
    sub_deadline = deadline - 30.0
    if best_seq is not None and time.time() < sub_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[smooth descent] starting")
        a_polished = smooth_descent(a0, sub_deadline)
        val = compute_score(a_polished)
        print(f"[smooth descent] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[hill‑climb] starting")
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – fall back to a uniform vector if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 2048
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)
        print("[fallback uniform] score =", best_val)

    # -----------------------------------------------------------------
    # 5. Official evaluation (includes the security wrappers)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print("\nSearch finished. Reported (official) score = {:.6f}".format(final_score))
    return best_seq
```
