def gradient_descent_smoothed(a_initial, deadline):
   a = a_initial.copy()
   best_a = a.copy()
   best_score = compute_score(a)
   beta = 5.0
   step_lr = 0.5 # initial step size (relative to 1)
   t = 0
   while time.time() < deadline:
      t += 1
      conv = full_convolution(a)
      # smoothing
      max_c = conv.max()
      # Use beta schedule: increase slowly
      if t % 1000 == 0 and beta < 200:
          beta *= 1.2
      # compute softmax weights
      # shift
      shifted = conv - max_c
      # exp(beta * shifted) may underflow, clamp
      exp_beta = np.exp(beta * shifted)
      w = exp_beta / exp_beta.sum()
      # gradient
      g = weighted_gradient(a, w)  # shape n
      # step size schedule
      lr = step_lr / np.sqrt(t)   # decaying
      # update
      a_candidate = a - lr * g
      # project onto simplex
      a_candidate = project_to_simplex(a_candidate)
      # evaluate
      new_score = compute_score(a_candidate)
      if new_score < best_score:
          best_score = new_score
          best_a = a_candidate.copy()
          a = a_candidate
          # maybe increase step size a bit
          step_lr = min(step_lr * 1.05, 2.0)
      else:
          # if not improved, maybe accept with probability small
          # Simulated annealing temperature that decreases
          T = max(0.01, (deadline - time.time()) / (deadline - start_time) * 0.2)  # small
          if np.random.rand() < np.exp(-(new_score - best_score) / T):
              a = a_candidate
          # else keep a
          step_lr *= 0.99

   return best_a

"""
search_step_function_improved.py

A fast optimiser for the ``evaluate_sequence`` objective.
Main ideas
-----------
*   Use the best known construction (``height_sequence_1``) as a seed.
*   Refine it with a *smooth‑max* projected gradient descent.
    The max of the convolution is approximated by a soft‑max
    (log‑sum‑exp) with an increasing temperature (beta).  This yields
    a smooth gradient that drives all large convolution entries down.
*   After the gradient phase perform a short aggressive stochastic
    hill‑climb (mass transfers between two positions) to polish the
    result.
*   All vectors are kept on the probability simplex (∑a = 1, a_i ≥ 0)
    so the objective `2·n·max(conv)` is scale‑invariant and matches the
    official ``evaluate_sequence`` score.

The routine respects the 1000 s time budget, prints intermediate
progress and finally returns a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x | x_i ≥ 0, sum x_i = 1}."""
    if np.all(v <= 0):
        # If everything is non‑positive the uniform distribution is the
        # projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (linear, full length) – FFT based
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, b: np.ndarray | None = None) -> np.ndarray:
    """
    Linear (full) convolution of a with b using a real FFT.
    If b is None the function returns a * a.
    Length of the result is len(a) + len(b) - 1.
    """
    if b is None:
        b = a
    n = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())          # next power of two ≥ n
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:n]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep ∑a = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)          # a * a
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient of a soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------
def weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  sum_k w_k * conv_k(a,a)  w.r.t. a.
    The formula reduces to  2 * (w * a_rev)[i + n‑1]  for i = 0..n‑1,
    where a_rev is a reversed copy of a.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_full = full_convolution(w, a_rev)          # length = (2n‑1)+n‑1 = 3n‑2
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]          # shape (n,)
    return grad


# ----------------------------------------------------------------------
# Smoothed projected gradient descent on the max‑convolution
# ----------------------------------------------------------------------
def gradient_descent_smooth(a0: np.ndarray, deadline: float,
                            start_time: float) -> np.ndarray:
    """
    Refine a vector by minimising a smooth approximation of
    ``max(conv(a,a))``.  The smoothness parameter ``beta`` grows
    slowly, turning the soft‑max into the true max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    n = a.shape[0]

    # hyper‑parameters
    step0 = 0.6          # initial step size (relative to 1)
    beta = 5.0           # start of soft‑max temperature
    t = 0

    while time.time() < deadline:
        t += 1
        # current convolution and max
        conv = full_convolution(a)
        max_conv = conv.max()

        # slowly increase the sharpness of the soft‑max
        if t % 800 == 0 and beta < 200.0:
            beta *= 1.12

        # soft‑max weights, shifted to avoid overflow
        shifted = conv - max_conv                 # ≤ 0
        exp_beta = np.exp(beta * shifted)        # may under‑flow → zero is fine
        w = exp_beta / exp_beta.sum()

        # gradient of the smooth objective
        g = weighted_gradient(a, w)

        # learning rate – classic 1/√t decay
        step = step0 / math.sqrt(t)

        # projected gradient step
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        # evaluate
        val = compute_score(a_new)

        # keep the best ever seen
        if val < best_val:
            best_val, best_a = val, a_new.copy()
            a = a_new                             # continue from the improvement
            # become bolder after a win
            step0 = min(step0 * 1.05, 1.5)
        else:
            # no improvement – shrink step a little and occasionally jitter
            step0 *= 0.993
            if t % 2000 == 0:
                jitter = np.random.randn(n) * (step0 * 0.003)
                a = project_to_simplex(a + jitter)

    return best_a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (aggressive mass‑transfer moves)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 600_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.025
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]                 # move a fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.7)
            no_improve = 0
        else:
            step_scale *= 0.996
            no_improve += 1

        it += 1
        if no_improve > 120_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1.  Normalise the seed vector (ensure sum = 1)
    # -----------------------------------------------------------------
    if best_seq is not None:
        a_seed = np.array(best_seq, dtype=float)
        a_seed /= a_seed.sum()
    else:
        # uniform fallback (unlikely to happen)
        n = 2048
        a_seed = np.full(n, 1.0 / n, dtype=float)

    # -----------------------------------------------------------------
    # 2.  Gradient descent with a smooth max‑convolution
    # -----------------------------------------------------------------
    gd_deadline = deadline - 300.0          # reserve time for hill‑climb
    if time.time() < gd_deadline:
        a_refined = gradient_descent_smooth(a_seed, gd_deadline, start_time)
        val = compute_score(a_refined)
        print(f"[gradient] refined fast score = {val:.6f}")
        if val * (sum(a_refined) ** 2) < best_val:   # scores are comparable
            best_val = evaluate_sequence(a_refined.tolist())
            best_seq = a_refined.tolist()
            print(f"[gradient] official score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3.  Final stochastic hill‑climb for polishing
    # -----------------------------------------------------------------
    hill_deadline = deadline
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final fast score = {val:.6f}")
        final_score = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] official score = {final_score:.6f}")
        if final_score < best_val:
            best_val, best_seq = final_score, a_final.tolist()

    # -----------------------------------------------------------------
    # 4.  Safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5.  Final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
