Define function project_to_simplex(v):

def project_to_simplex(v):
    # projection onto simplex {x >=0, sum = 1}
    # Based on algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1.   Global exploration – a few thousand random Dirichlet draws
     (different concentration parameters).  The best few candidates
     are kept.

2.   Soft‑Exponential‑Gradient (soft‑EG) polishing.
     * The true objective is   f(a)=max_k (a*a)_k .
     * We minimise a smooth surrogate
           φ_λ(a)=log‑sum‑exp( λ·conv(a) ) / λ
       (λ grows from 5 to ≈300).  Its gradient is
           ∇φ = 2·( a ⋆ w ),   w_i ∝ exp(λ·conv_i) .
       * The EG update  a_i ← a_i·exp(−η·∇φ_i)  is performed,
         followed by Euclidean projection onto the simplex.
       * η decays like 1/√t, λ grows slowly, and occasional
         jitter‑perturbations help escape plateaux.
       * Many independent restarts are run while time permits;
         the best vector seen globally is retained.

3.   Final short simulated‑annealing hill‑climb to polish the
     best vector from step 2.

The whole search respects a 1000 s wall‑clock budget, leaving a
few seconds margin for the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0,  ∑_i x_i = 1 }.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest ρ satisfying u_ρ > (cumulative_sum_ρ - 1)/ρ
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become zero – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise (can drift slightly)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def correlation_via_fft(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation (a ⋆ w) of lengths n and m=2·n‑1.
    Returned vector contains n entries g_j = Σ_i a_i·w_{i+j}.
    """
    A = np.fft.fft(a, n=L)
    W = np.fft.fft(w, n=L)
    corr_full = np.fft.ifft(A * np.conj(W)).real
    start = len(w) - 1               # index of zero lag
    return corr_full[start:start + a.shape[0]]

# Search primitives
# ----------------------------------------------------------------------


def random_dirichlet_vectors(n: int, deadline: float, max_vectors: int = 3000):
    """
    Generate up to ``max_vectors`` random Dirichlet draws with a
    selection of concentration parameters.  Returns a list of numpy
    arrays (already normalised to sum‑one).
    """
    candidates = []
    concs = [0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = max_vectors // len(concs)

    for conc in concs:
        alpha = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                return candidates
            vec = np.random.dirichlet(alpha)
            candidates.append(vec)

    return candidates


def soft_eg_optimize(a0: np.ndarray,
                     deadline: float,
                     L: int,
                     max_iter: int = 1500,
                     eta0: float = 0.6,
                     lambda_start: float = 5.0,
                     lambda_target: float = 300.0) -> np.ndarray:
    """
    Soft‑Exponential‑Gradient minimisation of the max‑convolution.
    Works on the simplex (a already sums to one).
    Returns the best vector encountered before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    lam = lambda_start
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # --- convolution (linear) via FFT ---------------------------------
        A = np.fft.fft(a, n=L)
        conv_full = np.fft.ifft(A * A).real
        conv = conv_full[: 2 * n - 1]

        # --- smooth weights (soft‑max) -----------------------------------
        max_c = conv.max()
        shifted = conv - max_c                # non‑positive, avoids overflow
        exp_shift = np.exp(lam * shifted)
        w = exp_shift / exp_shift.sum()       # Σ w = 1, concentrates near max

        # --- gradient: g = 2·(a ⋆ w) ------------------------------------
        g = 2.0 * correlation_via_fft(a, w, L)

        # --- EG update with decaying step size --------------------------
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        a = project_to_simplex(a)

        # --- evaluate true objective ------------------------------------
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # --- increase smoothing parameter gradually ----------------------
        if lam < lambda_target:
            lam = min(lambda_target, lam * 1.02)

        # --- occasional random jitter to escape flat regions ------------
        if it % 500 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = project_to_simplex(a)

    return best_a


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """
    Small Metropolis‑style hill‑climb (identical to the one used before)
    but started from the best vector found by the soft‑EG phase.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1

        # pick two (different) indices
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin for the external harness
    deadline = start_time + budget_s - 5.0

    # problem size – a trade‑off between expressiveness and speed
    n = 2048

    # pre‑compute the FFT length used everywhere (≥3·n‑2)
    L = 1 << ((3 * n - 2 - 1).bit_length())

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. optional fallback from a previous run
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            cand = [float(x) for x in fallback[:n]]
            val = evaluate_sequence(cand)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. global random Dirichlet exploration (≈ 100 s)
    # -----------------------------------------------------------------
    explore_deadline = deadline - 800.0  # leave ~800 s for polishing + anneal
    if time.time() < explore_deadline:
        print("[exploration] sampling random Dirichlet vectors …")
        dirs = random_dirichlet_vectors(
            n, explore_deadline, max_vectors=2000
        )
        for vec in dirs:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
        print(f"[exploration] best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Soft‑EG polishing (majority of the budget)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 30.0   # keep a few seconds for final anneal
    while time.time() < polish_deadline:
        # decide a start point – 50 % chance to restart randomly,
        # otherwise continue polishing the current best.
        if best_seq is not None and np.random.rand() < 0.5:
            a0 = np.array(best_seq, dtype=float)
        else:
            # random Dirichlet with a random concentration in [0.1,5]
            conc = np.random.uniform(0.1, 5.0)
            a0 = np.random.dirichlet(np.full(n, conc))
        # run a short soft‑EG optimisation from that start
        local_deadline = min(polish_deadline,
                             time.time() + 12.0)  # ≈12 s per run
        refined = soft_eg_optimize(a0, local_deadline, L,
                                   max_iter=1200,
                                   eta0=0.6,
                                   lambda_start=5.0,
                                   lambda_target=250.0)
        refined_val = compute_score(refined)
        if refined_val < best_val:
            best_val, best_seq = refined_val, refined.tolist()
            print(f"[soft‑EG] new best = {best_val:.6f} "
                  f"at t={time.time() - start_time:.1f}s")

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (≈ 20 s)
    # -----------------------------------------------------------------
    final_deadline = deadline
    if best_seq is not None and time.time() < final_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, final_deadline)
        final_val = compute_score(final)
        if final_val < best_val:
            best_val, best_seq = final_val, final.tolist()
            print(f"[anneal] final best = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. fallback to uniform distribution (should never happen)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
