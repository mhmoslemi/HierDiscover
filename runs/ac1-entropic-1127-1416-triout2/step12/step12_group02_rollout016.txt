def product_family(n, alpha):

    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w_log = -alpha * logprod
    w_log -= np.max(w_log)
    w = np.exp(w_log)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full(n, 1.0 / n)
    else:
        w /= w_sum
    return w

def lp_refine(f, num_iters, t, deadline):
   best_f = f.copy()
   best_score = compute_score(best_f)
   for i in range(num_iters):
       if time.time() >= deadline - 0.5: break
       conv = convolution_fft(f, f)
       M = conv.max()
       g = solve_lp(f, M)
       if g is None:
           break
       # normalize g
       g = g / g.sum()
       # mix
       f = (1-t) * f + t * g
       # project onto simplex to correct any numeric drift
       f = project_to_simplex(f)
       sc = compute_score(f)
       if sc < best_score:
           best_score = sc
           best_f = f.copy()
           print(f'[{time.time():.1f}] LP iteration {i+1} improved score {best_score:.6f}')
   return best_f, best_score

"""
Propose a step‑function sequence (non‑negative heights) that minimizes

    2·n·max(convolution) / (sum a)²

The algorithm combines three ideas:

1. **Analytic start** – a symmetric “product” U‑shape obtained from a
   simple closed‑form formula; this already beats the uniform baseline.

2. **Linear‑programming refinement** – given a current candidate `f`,
   we solve the LP

        maximise   Σ g_i
        subject to  f * g ≤ M   (M = max(f*f)),
                    g_i ≥ 0 ,

   which moves mass towards entries where `f` is small without increasing the
   worst convolution value.  The solution `g` is mixed with `f`
   (convex combination) and the process is repeated a few times.
   The LP is tiny (≈2 n constraints, n variables) for n≈1 000 and solved
   with SciPy’s HiGHS interior‑point method.

3. **Final polishing** – a short projected‑sub‑gradient descent on the
   exact max‑convolution objective, starting from the best LP‑result.

All steps respect the 1000 s wall‑clock budget; the best sequence found
is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used as initial guesses)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric product family: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    # work in log‑space to avoid under‑/overflow
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w_log = -alpha * logprod
    w_log -= np.max(w_log)            # stabilise exponentiation
    w = np.exp(w_log)
    s = w.sum()
    if s == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w /= s
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    w /= w.sum()
    return w


def dist_power_family(n: int, beta: float) -> np.ndarray:
    """U‑shape based on distance to the nearer endpoint."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i) + 0.5
    w = d ** (-beta)
    w /= w.sum()
    return w


def sum_dist_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} + (n‑½‑i)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta) + (n - 0.5 - i) ** (-beta)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “alternating” step)
# ----------------------------------------------------------------------


def _build_lp_matrix(f: np.ndarray):
    """
    Build the (2n‑1) × n sparse matrix A with
        (A·g)[k] = Σ_i f_i g_{k-i}
    i.e. the convolution f * g written as a linear map in g.
    """
    from scipy.sparse import csr_matrix

    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # each constraint k corresponds to a sliding window of size up to n
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over valid positions of f that contribute to this k
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(k - i)          # index of g
            data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_step(f: np.ndarray, M: float):
    """
    Solve:
        maximise   Σ g_i
        subject to  f * g ≤ M   (component‑wise)
                    g_i ≥ 0
    Returns the optimal g (as a dense ndarray) or None on failure.
    """
    from scipy.optimize import linprog

    n = f.shape[0]
    A = _build_lp_matrix(f)

    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum → minimise -sum
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        return None
    g = res.x
    # normalise to a probability vector (scaling does not affect the final objective)
    ss = g.sum()
    if ss == 0.0:
        return None
    g /= ss
    return g


def lp_refine(f0: np.ndarray, iters: int, mix: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve a distribution by the LP step described above.
    *mix* ∈ (0,1) is the convex‑combination weight applied after each LP solve.
    Returns the best vector seen and its exact score.
    """
    best_f = f0.copy()
    best_score = compute_score(best_f)

    f = f0.copy()
    for it in range(1, iters + 1):
        if time.time() >= deadline - 0.5:
            break

        # current peak of f*f
        cur_conv = convolution_fft(f, f)
        M = cur_conv.max()

        g = solve_lp_step(f, M)
        if g is None:
            break

        # convex mixing and projection onto the simplex
        f = (1.0 - mix) * f + mix * g
        f = project_to_simplex(f)

        sc = compute_score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
            print(f"[{time.time():.1f}] LP iter {it} improved score = {best_score:.6f}")

    return best_f, best_score


# ----------------------------------------------------------------------
# Simple projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgrad_descent(start_a: np.ndarray, deadline: float,
                    max_iters: int = 300_000,
                    step0: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent for the exact max‑convolution objective.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline - 0.5:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of max‑convolution (sub‑gradient)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad iter {t} improved score = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score
