"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Broad parametric sweep** (≈ 300 s).  
   Several families of monotone decreasing vectors are explored:
   power‑law, geometric, mixed plateau‑power‑law, beta‑shape,
   stretched exponential, and power‑exponential.  The best
   candidate (lowest exact score) is retained.

2. **Smooth‐gradient polishing** (≈ 500 s).  
   The nonsmooth objective ``max(conv(a,a))`` is approximated by
   a log‑sum‑exp (soft‑max) with temperature ε.  The gradient of
   the smooth surrogate is obtained analytically by a single FFT‑based
   convolution, and projected gradient descent on the probability
   simplex is performed with a diminishing step‐size.  ε is annealed
   slowly, allowing the method to escape shallow local minima.

3. **Aggressive stochastic hill‑climb** (remaining time).  
   Random mass transfers between two coordinates are accepted only
   if they improve the exact max‑convolution; otherwise the step size
   is gradually shrunk.  Periodic jitter helps to break plateaus.

All phases respect the 1000 s budget, fall back to a uniform vector
if nothing is found, and finally evaluate the sequence with the
provided ``evaluate_sequence`` function.
"""

import math

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    N = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=N)
    C = A * A
    conv = np.fft.irfft(C, n=N)[:m]
    return conv


def full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two arbitrary real vectors via FFT."""
    L = x.shape[0] + y.shape[0] - 1
    N = 1 << ((L - 1).bit_length())
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:L]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
       2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def stretched_exponential(n: int, c: float, beta: float) -> np.ndarray:
    """a_i ∝ exp(-c·i^beta), i=0,…,n-1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-c * (i ** beta))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def power_exponential(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·q^{i}, 0<q<1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * (q ** (i - 1))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
# Parametric sweep – very broad exploration
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> np.ndarray:
    best_vec = None
    best_val = float('inf')

    # Power‑law
    for p in np.linspace(0.10, 2.00, 191):
        if time.time() >= deadline:
            return best_vec
        a = power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # Geometric (dense near 1)
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 91),
            np.linspace(0.990, 0.9999, 110))):
        if time.time() >= deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # Mixed plateau‑power
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # Beta‑shape
    alphas = np.linspace(-0.8, 3.0, 20)
    betas = np.linspace(-0.8, 3.0, 20)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # Stretched exponential
    for c in np.linspace(0.0001, 5.0, 30):
        for beta in np.linspace(0.1, 2.0, 20):
            if time.time() >= deadline:
                return best_vec
            a = stretched_exponential(n, c, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # Power‑exponential
    for p in np.linspace(0.10, 2.0, 30):
        for q in np.linspace(0.90, 0.9999, 80):
            if time.time() >= deadline:
                return best_vec
            a = power_exponential(n, p, q)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # Random convex combinations of three basic shapes
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    geom = geometric(n, 0.98)
    power = power_law(n, 0.5)
    basic = [uniform, geom, power]
    for _ in range(2000):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(3)
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basic))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Smooth‑gradient polishing (log‑sum‑exp surrogate of the max)
# ----------------------------------------------------------------------
def smooth_descent(a0: np.ndarray, deadline: float,
                   step0: float = 0.12,
                   epsilon0: float = None) -> np.ndarray:
    """Projected gradient descent on a smooth approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    conv = full_convolution(a)
    best_M = conv.max()
    t = 0
    epsilon = epsilon0 if epsilon0 is not None else best_M
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()

        # anneal temperature slowly
        if t % 2000 == 0:
            epsilon = max(epsilon * 0.95, 1e-10)

        # weights for the smooth max
        w = np.exp(conv / epsilon)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient of the smooth max
        a_rev = a[::-1]
        g_full = full_convolution_two(a_rev, w)      # length 3n‑2
        g = 2.0 * g_full[n - 1: n - 1 + n]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasionally add a tiny jitter to avoid stagnation
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.001 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (aggressive local improvement)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.02
    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)
        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1
        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0
    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Fallback – use previously discovered construction if present
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Parametric sweep (≈ 300 s)
    # -----------------------------------------------------------------
    n = 4096                # a larger canvas often yields a better constant
    sweep_deadline = deadline - 500.0
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Smooth‑gradient polishing (≈ 400 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 80.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = smooth_descent(a0, polish_deadline, step0=0.12)
        val = compute_score(a_polished)
        print(f"[smooth‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Fallback to uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluate_sequence
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
