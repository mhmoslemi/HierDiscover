"""
solve_step_function.py

A second‑generation optimiser for the ``evaluate_sequence`` objective.
Key ideas
---------

* Explore a richer set of parametric families, in particular **U‑shaped**
  (heavy at the ends) distributions such as
  - symmetric power‑law on the product (i+1)*(n-i)
  - sum of two power‑law tails
  - Beta‑distributions with negative exponents (more mass near the edges).

* Keep the best few candidates and **refine them with a smooth‑max
  gradient descent**.  The non‑smooth max is replaced by the
  log‑sum‑exp (soft‑max) with an annealed temperature β.  The gradient
  of the soft‑max is a weighted sum of the convolution gradients,
  which can be evaluated efficiently via FFT‑based correlation.

* The optimisation is performed on the probability simplex
  (∑a = 1, a ≥ 0) using an Euclidean projection after each update.
  Adam‑style adaptive step sizes provide fast convergence.

* After the smooth‑max refinement a short stochastic hill‑climb
  (mass transfers between two coordinates) is applied to polish the
  solution.

The whole pipeline respects the 1000 s time budget and returns a
list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Full correlation of vector a (length n) with vector w (length m).
    Returns an array of length n+m-1 whose central n entries are
    the gradient contribution needed for the soft‑max.
    """
    n = a.shape[0]
    m = w.shape[0]
    L = 1 << ((n + m - 1 - 1).bit_length())      # next power of two ≥ n+m-1
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(w[::-1], n=L)                # reverse w for correlation
    C = A * B
    corr = np.fft.irfft(C, n=L)[: n + m - 1]
    return corr


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    This coincides with ``evaluate_sequence`` for sum(a)=1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def double_power_tail(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} + (n-i)^{-p} (heavy at both ends)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p) + (n - i + 1) ** (-p)
    a /= a.sum()
    return a


def symmetric_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^{-p} – very heavy at the borders."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = n - i + 1.0
    a = (i * j) ** (-p)
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i ∝ x_i^{alpha} (1-x_i)^{beta},  x_i = (i+0.5)/n .
    Allows symmetric or skewed shapes; negative exponents put mass near
    the boundaries.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (annealed β)
# ----------------------------------------------------------------------
def smooth_max_refine(a0: np.ndarray,
                      deadline: float,
                      max_iters: int = 8000,
                      beta_start: float = 5.0,
                      beta_end: float = 2000.0,
                      lr: float = 0.02) -> np.ndarray:
    """
    Projected Adam descent on a smooth approximation of max(conv(a,a)).
    The soft‑max temperature β is increased linearly (in log‑space)
    from ``beta_start`` to ``beta_end``.
    """
    a = a0.copy()
    a = project_to_simplex(a)               # ensure Σa = 1
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    n = a.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # annealed temperature
        beta = beta_start * (beta_end / beta_start) ** (it / max_iters)

        # convolution of the current vector with itself
        conv = full_convolution(a)
        max_val = conv.max()

        # soft‑max weights (numerically stable)
        w_exp = np.exp(beta * (conv - max_val))
        w = w_exp / w_exp.sum()                     # length 2n‑1, sums to 1

        # gradient of the smooth‑max = 2 * correlation(a, w)
        corr = full_correlation(a, w)                # length 3n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n]          # central n entries

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # occasional tiny jitter to avoid stagnation
        if it % 2000 == 0:
            jitter = np.random.randn(n) * (lr * 0.003)
            a += jitter
            a = project_to_simplex(a)

    return a

# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_iter: int = 300_000) -> np.ndarray:
    """Greedy stochastic mass‑transfer moves, projected onto the simplex."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.02
    no_improve = 0
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]          # move a fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 120_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  The algorithm consists of:
        1. a broad parametric sweep over several families,
        2. smooth‑max Adam refinement of the best few candidates,
        3. a final stochastic hill‑climb.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0.  Optional fallback using a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1.  Parametric sweep over several lengths
    # -----------------------------------------------------------------
    candidate_ns = [1024, 2048, 4096]   # a few sizes – each costly, so we limit count
    for n in candidate_ns:
        if time.time() >= deadline:
            break
        print(f"\n--- Exploring length n={n} ---")

        candidates = []

        # uniform
        candidates.append(('uniform', uniform(n)))

        # classic decreasing power‑law
        for p in np.linspace(0.15, 2.5, 24):
            candidates.append((f'power_{p:.2f}', power_law(n, p)))

        # double‑tail power‑law (heavy at both ends)
        for p in np.linspace(0.3, 2.5, 23):
            candidates.append((f'double_tail_{p:.2f}', double_power_tail(n, p)))

        # symmetric product‑power (very strong edge weight)
        for p in np.linspace(0.5, 3.0, 16):
            candidates.append((f'sym_prod_{p:.2f}', symmetric_product_power(n, p)))

        # beta shapes with negative exponents (edges)
        alphas = np.linspace(-1.5, -0.2, 9)
        betas  = np.linspace(-1.5, -0.2, 9)
        for a_exp in alphas:
            for b_exp in betas:
                a_vec = beta_shape(n, a_exp, b_exp)
                if a_vec is not None:
                    candidates.append((f'beta_{a_exp:.2f}_{b_exp:.2f}', a_vec))

        # -----------------------------------------------------------------
        # Evaluate all candidates, keep the best few for polishing
        # -----------------------------------------------------------------
        evaluated = []
        for name, vec in candidates:
            val = compute_score(vec)
            evaluated.append((val, vec))
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        evaluated.sort(key=lambda x: x[0])
        top_k = 5
        top_vectors = [vec for (_, vec) in evaluated[:top_k]]
        print(f"[sweep n={n}] best score so far = {best_val:.6f}")

        # -----------------------------------------------------------------
        # 2. Smooth‑max refinement of the top candidates
        # -----------------------------------------------------------------
        for idx, a0 in enumerate(top_vectors):
            if time.time() >= deadline:
                break
            print(f"[refine n={n} candidate {idx+1}/{top_k}]")
            a_ref = smooth_max_refine(a0, deadline,
                                      max_iters=9000,
                                      beta_start=8.0,
                                      beta_end=2500.0,
                                      lr=0.025)
            score = compute_score(a_ref)
            if score < best_val:
                best_val, best_seq = score, a_ref.tolist()
                print(f"  >>> new best after refinement: {score:.6f}")

        # -----------------------------------------------------------------
        # 3. Final hill‑climb starting from the current best (if any)
        # -----------------------------------------------------------------
        if best_seq is not None and time.time() < deadline:
            print(f"[hill‑climb n={n}] starting hill‑climb")
            a_start = np.array(best_seq, dtype=float)
            a_hc = hill_climb(a_start, deadline, max_iter=250_000)
            score = compute_score(a_hc)
            if score < best_val:
                best_val, best_seq = score, a_hc.tolist()
                print(f"  >>> new best after hill‑climb: {score:.6f}")

    # -----------------------------------------------------------------
    # 4. Safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluator
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported official score = {final_score:.6f}")
    return best_seq
