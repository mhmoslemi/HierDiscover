def eg_optimize(start_a, deadline, max_iters=500000, eta0=1.0):
    n = len(start_a)
    a = start_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a,a)
        k = int(np.argmax(conv))
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]
        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improvement: {best_sc:.6f}")
    return best_a, best_sc
```

"""
Improved search for a step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in several phases:

1. Load a fallback sequence if the harness provides one.
2. Choose a problem size n (4096 by default, larger if fallback is long).
3. Scan a handful of analytically‑motivated families:
     – symmetric product  a_i ∝ ((i+½)(n‑½‑i))^{‑α}
     – arcsine × exponential   a_i ∝ ((i+½)(n‑½‑i))^{‑α}·exp(‑γ·|i−centre|/centre)
     – left‑heavy power   a_i ∝ (i+½)^{‑β}
   The best candidate of this cheap scan is kept.
4. Refine the candidate with three optimisation phases that respect the
   simplex constraint:
     • Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
     • Adam on a smooth‑max (log‑sum‑exp) surrogate, slowly increasing β.
     • Projected sub‑gradient descent on the original max‑convolution.
5. Optional symmetrisation (the problem is invariant under reversal).
6. A short simulated‑annealing hill‑climb for any remaining local improvements.
7. Return the best sequence (as a plain list).  All phases share the global
   wall‑clock budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the evaluator (up to a scale that cancels
    because the vector is normalised to sum = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the inexpensive initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{‑α}.
    α = 0.5 gives the discrete arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def arcsine_exp_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    """
    Product of the symmetric‑product law and a gentle exponential decay
    from the centre.  The distance is normalised to [0, 1].
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    centre = (n - 1) / 2.0
    dist_norm = np.abs(i - centre) / centre
    w = w * np.exp(-gamma * dist_norm)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing sequence (used for random restarts)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers (all respect the simplex constraint)
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 600_000,
                eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improvement → {best_sc:.6f}")

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 4.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.022) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    with β growing from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise from a (softmax parametrisation)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (log‑linear)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # chain rule through softmax
        dot = np.dot(a, grad_a)
        grad_z = a * grad_a - dot * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_sc:
                best_sc = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improvement → {best_sc:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_sc:
        best_sc = sc_fin
        best_a = a_fin.copy()
    return best_a, best_sc


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 4_000_000,
                     step0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the exact max‑convolution.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad improvement → {best_sc:.6f}")

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing style random walk on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_sc = compute_score(cur_a)

    best_a = cur_a.copy()
    best_sc = cur_sc

    T0 = best_sc * 1e-4
    T = T0
    scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_sc = compute_score(cand)

        delta = cand_sc - cur_sc
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a, cur_sc = cand, cand_sc
            if cur_sc < best_sc:
                best_sc, best_a = cur_sc, cur_a.copy()
                scale = min(scale * 1.07, 0.5)
        else:
            scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_sc

        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (β={beta:.2f}) → {best_score:.6f}")

    # d) a few random monotone starts (useful for escaping local minima)
    for _ in range(3):
        if time.time() >= deadline - 5.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] random start improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Normalise the working vector (simplex, sum = 1).
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)
    print(f"[{time.time():.1f}] after initial scan best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Slice the remaining budget for the refinement phases.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    eg_time   = 0.20 * remaining
    adam_time = 0.20 * remaining
    subg_time = 0.35 * remaining
    sa_time   = remaining - (eg_time + adam_time + subg_time)

    # ------------------------------------------------------------------
    # 5.  Exponentiated‑gradient phase.
    # ------------------------------------------------------------------
    if time.time() + eg_time < deadline:
        print(f"[{time.time():.1f}] EG phase (≈{eg_time:.1f}s)")
        a_eg, sc_eg = eg_optimize(a, time.time() + eg_time,
                                  max_iters=800_000,
                                  eta0=1.0)
        if sc_eg < best_score:
            best_score, a = sc_eg, a_eg
            print(f"[{time.time():.1f}] EG improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Adam optimisation of the smooth‑max surrogate.
    # ------------------------------------------------------------------
    if time.time() + adam_time < deadline:
        print(f"[{time.time():.1f}] Adam phase (≈{adam_time:.1f}s)")
        a_adam, sc_adam = adam_smooth_optimize(a,
                                                time.time() + adam_time,
                                                max_iters=500_000,
                                                beta_start=4.0,
                                                beta_end=8000.0,
                                                lr=0.022)
        if sc_adam < best_score:
            best_score, a = sc_adam, a_adam
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Long projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() + subg_time < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient phase (≈{subg_time:.1f}s)")
        a_sub, sc_sub = subgrad_optimize(a,
                                         time.time() + subg_time,
                                         max_iters=6_000_000,
                                         step0=0.15)
        if sc_sub < best_score:
            best_score, a = sc_sub, a_sub
            print(f"[{time.time():.1f}] Sub‑gradient improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Optional symmetrisation (the problem is symmetric under reversal).
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score, a = sc_sym, a_sym
        print(f"[{time.time():.1f}] Symmetrisation improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    if time.time() + sa_time < deadline:
        print(f"[{time.time():.1f}] Hill‑climb (SA) phase (≈{sa_time:.1f}s)")
        a_sa, sc_sa = hill_climb_sa(a,
                                    time.time() + sa_time,
                                    max_steps=2_000_000)
        if sc_sa < best_score:
            best_score, a = sc_sa, a_sa
            print(f"[{time.time():.1f}] Hill‑climb improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10.  Final evaluation using the external harness.
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return final_seq
