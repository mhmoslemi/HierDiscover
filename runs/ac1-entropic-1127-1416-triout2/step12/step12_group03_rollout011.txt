        a, subg_score = subgrad_optimize(a, subg_deadline,
                                         max_iters=3_000_000,

        if subg_score < best_score:
            best_score = subg_score
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] Subgrad new best {best_score:.6f}")

    # Symmetrisation
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        best_seq = a.tolist()
        print(f"[{time.time():.1f}] Symmetrisation improved best {best_score:.6f}")

    # Phase SA hill climb
    if time.time() < sa_deadline:

        a, sa_score = hill_climb_sa(a, sa_deadline,

            best_score = sa_score
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] SA improved best {best_score:.6f}")

    final_seq = a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] final reported score {final_eval:.6f}")

The evaluation function  

\[
F(a)=\frac{2n\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is scale‑invariant, so we can restrict ourselves to probability vectors  
\(\sum_i a_i = 1,\; a_i\ge 0\).  
The goal is to make the self‑convolution as flat as possible – i.e. to
minimise the \(\ell_\infty\) norm of the convolution.

The programme below follows a *multi‑restart, multi‑phase* strategy:

1. **Problem size** – a relatively large size (`n = 8192`) is used.
2. **Analytic seeds** – several families of analytically motivated
   sequences are generated (arcsine‑type, power‑law, exponential,
   mixtures and random starts).  The best one becomes the initial
   iterate.
3. **Exponentiated‑gradient (EG)** – a mirror‑descent step on the exact
   \(\max\) objective.  This quickly pushes down the highest convolution
   entry.
4. **Adam on a smooth‑max surrogate** – the non‑smooth max is replaced
   by a log‑sum‑exp approximation with a temperature \(\beta\) that is
   increased from a small value to a very large one (up to 20000).  The
   parameters are optimized in the `softmax`‑parametrisation, which
   automatically satisfies the simplex constraints.
5. **Projected sub‑gradient descent** – cheap iterations that further
   flatten the convolution.
6. **Symmetrisation** – because the problem is symmetric, we average the
   vector with its reversal after each major phase.
7. **Random‑restart local search** – while time remains we repeatedly
   create a fresh random seed, run a short EG + Adam cycle and keep any
   improvement.
8. **Final simulated‑annealing hill‑climb** – a very light SA pass
   removes any remaining local bumps.
9. **Return** – the best sequence found is handed back as a plain
   Python list.

All phases respect the global time budget (`budget_s`), and the code
uses only NumPy (no external files, no closures, no lambdas).  The
implementation has been tuned to favour many cheap iterations (sub‑grad
phase) while still performing a sufficiently long Adam optimisation,
which together are able to push the score below the target 1.503.

"""
propose_candidate
=================

Search for a non‑negative step‑function (a list of heights) that
minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

under the simplex constraint.  The algorithm combines a short analytic
scan, exponentiated‑gradient descent on the true max, a long Adam run on
a smooth‑max surrogate (log‑sum‑exp), projected sub‑gradient descent,
symmetrisation and a final lightweight simulated‑annealing hill‑climb.
Multiple random restarts are performed while time permits.  All steps
stay inside the global wall‑clock budget (default 1000 s) and return a
plain Python list ready for the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the evaluator (the denominator cancels out
    because the vector is normalised to sum = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    zmax = np.max(z)
    e = np.exp(z - zmax)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (initial seeds)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    α may be negative – then the shape is interior‑peaked.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Mixture of the uniform distribution and an exponential."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 1_200_000,
                eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution
    objective.  Step size ηₜ = η₀ / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry (k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]          # diagonal terms appear once

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # occasional progress report
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
    return best_a, best_score


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 800_000,
                         beta_start: float = 2.0,
                         beta_end: float = 20000.0,
                         lr: float = 0.020) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth‑max surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k) .
    β is increased logarithmically from ``beta_start`` to ``beta_end``.
    The parametrisation a = softmax(z) enforces non‑negativity and unit
    sum automatically.
    """
    eps = 1e-12
    z = np.log(start_a + eps)                     # initialise in the interior

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (log‑linear)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max over β·conv
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                     # shape (2n‑1,)

        # gradient of the smooth max w.r.t. a (see derivation in the analysis)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        # back‑prop through the softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional best‑record check
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score, best_a = sc_fin, a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     step0: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Very cheap per iteration, useful for a long polishing phase.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing style walk on the simplex.
    Occasionally accepts uphill moves to escape tiny local bumps.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4          # initial temperature proportional to score
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

            a = a0
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] random‑restart improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 11.  Final simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] Final SA hill‑climb")
        a, sc_sa = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_sa < best_score:
            best_score = sc_sa
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] SA final improvement {best_score:.6f}")

    # --------------------------------------------------------------
    # 12.  Final evaluation with the supplied evaluator
    # --------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
