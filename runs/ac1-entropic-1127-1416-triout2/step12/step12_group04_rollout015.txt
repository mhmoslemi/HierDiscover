1. Choose a moderately large length n (default 5000) and start from a
   symmetric “U‑shaped” analytic family (product_family with α≈0.5).
2. Re‑parameterise the probabilities by a soft‑max (a_i = exp(z_i)/∑exp(z))
   so the positivity and sum‑to‑one constraints are always satisfied.
3. Optimise a smooth approximation of the max‑convolution:
        f_β(a) = (1/β)·log(∑_k exp(β·(a*a)_k)).
   Its gradient can be expressed by a convolution with the weight vector
   w_k ∝ exp(β·(a*a)_k).  The gradient w.r.t. the logits z is computed by
   the chain rule (soft‑max Jacobian).
4. Perform a continuation/annealing on β: start with small β (smooth)
   and gradually increase it (β = 0.5 → 1 → 2 → 5 → 10 → 20 → 50), using
   L‑BFGS‑B on each stage (no constraints needed because of the soft‑max).
5. After the continuation, symmetrise the solution (the problem is symmetric)
   and return the resulting list of heights.

All heavy work is done with FFT‑based convolutions (O(n log n) per gradient)
and the whole search respects the 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities (FFT based convolution, soft‑max, analytic families)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum
    return w


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_obj(z: np.ndarray, n: int, beta: float):
    """
    Smoothed objective f_β and its gradient w.r.t. logits z.

    Returns (obj, grad_z) where grad_z has the same shape as z.
    """
    # a = softmax(z) satisfies a_i >= 0 and sum a_i = 1
    a = softmax(z)

    # full self‑convolution of a (length 2n‑1)
    conv = convolution_fft(a, a)

    # max entry (used for numerical stability)
    maxc = conv.max()

    # weights w_k = exp(β·conv_k) / Σ exp(β·conv)
    shift = conv - maxc          # shift to avoid overflow
    exp_vals = np.exp(beta * shift)
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp       # length 2n‑1, non‑negative, sums to 1

    # gradient of f_β w.r.t. a : g = Σ_k w_k ∇(a*a)_k
    # ∇(a*a)_k = 2 * shift of a (see derivation).  Efficiently:
    a_rev = a[::-1]                         # reverse a
    conv_wa = convolution_fft(w, a_rev)    # length (2n‑1)+n‑1 = 3n‑2
    # slice to obtain exactly n components:
    grad_a = 2.0 * conv_wa[n - 1 : n - 1 + n]   # shape (n,)

    # chain rule through soft‑max: ∂f/∂z_i = a_i (g_i - <g,a>)
    dot = np.dot(grad_a, a)                 # scalar <g,a>
    grad_z = a * (grad_a - dot)             # shape (n,)

    # f_β = maxc + (1/β)·log Σ exp(β·(conv - maxc))
    obj = maxc + (np.log(sum_exp) / beta)

    return obj, grad_z


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The implementation follows a continuation L‑BFGS optimisation on a
    smooth approximation of the max‑convolution objective.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 0.  Use a provided fallback if it exists and is decent
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}s] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem size (a few thousand entries give a good trade‑off)
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    n = max(4000, fallback_len)   # at least 4000 entries
    print(f"[{time.time() - start_time:.1f}s] using n = {n}")

    # ------------------------------------------------------------------
    # 2. Initial analytic scan (very cheap)
    # ------------------------------------------------------------------
    # try the symmetric product family around α ≈ 0.5 (the arcsine‑like shape)
    alphas = [0.3, 0.4, 0.5, 0.6, 0.7]
    for alpha in alphas:
        if time.time() >= deadline - 10.0:
            break
        a0 = product_family(n, alpha)
        sc = compute_score(a0)
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
            print(f"[{time.time() - start_time:.1f}s] init α={alpha:.2f} score={sc:.6f}")

    # If no good analytic start, fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3. Convert to logits for soft‑max parametrisation
    # ------------------------------------------------------------------
    a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a /= a_sum
    # logits: z_i = log(a_i)  (soft‑max of these reproduces a)
    z = np.log(a + 1e-30)

    # ------------------------------------------------------------------
    # 4. Continuation optimisation (increase β progressively)
    # ------------------------------------------------------------------
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]   # smoother → sharper
    for beta in betas:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        # allocate a modest number of L‑BFGS iterations; each iteration ≈ O(n log n)
        max_iter = int(min(300, max(30, remaining * 5)))  # keep total time in check
        print(f"[{time.time() - start_time:.1f}s] β={beta:.1f} → max_iter={max_iter}")

        # minimize the smooth objective; ``smooth_obj`` returns (f, grad)
        res = minimize(
            fun=smooth_obj,
            x0=z,
            args=(n, beta),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
        z = res.x
        # quick progress report
        a = softmax(z)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}s] improved score after β={beta:.1f} → {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final symmetrisation (the objective is symmetric)
    # ------------------------------------------------------------------
    a = softmax(z)               # ensure exact simplex
    a = (a + a[::-1]) * 0.5      # average with its mirror
    a = a / a.sum()              # re‑normalise (should stay 1)
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_seq = a.copy()
        print(f"[{time.time() - start_time:.1f}s] after symmetrisation score = {final_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Return result (plain Python list for the harness)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    reported = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}s] finished – reported score = {reported:.6f}")
    return final_seq
