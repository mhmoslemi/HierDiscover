"""
search_step_function.py

Optimisation pipeline for ``evaluate_sequence``.
The algorithm consists of three stages:

1.  A very broad parametric sweep (including new families such as the
    arcsine shape and double‑power‑law) to obtain a good initial vector.
2.  An exponentiated‑gradient descent on a smooth approximation of the
    max‑convolution objective.  The smoothing parameter β is increased
    gradually, while the step‑size is annealed.  The update
    a_i ← a_i·exp(−η·∇) preserves non‑negativity and the simplex
    constraint after normalisation.
3.  An optional final polishing using the classic sub‑gradient method
    (as in the reference implementation) and a short stochastic hill‑climb.

All stages respect the overall 1000 s time budget.  The code works for any
length n; we use n=2048 (the length of the reference construction).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v_i ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution (real inputs)
# ----------------------------------------------------------------------
def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution of two real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    N = 1 << ((n - 1).bit_length())          # next power of two ≥ n
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    conv = np.fft.irfft(Z, N)[:n]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (full convolution)."""
    return fft_convolve(a, a)


# ----------------------------------------------------------------------
# Objective for a normalised vector (Σa = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for Σa = 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate vectors (new + original)
# ----------------------------------------------------------------------
def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return uniform_shape(n)
    a /= s
    return a


def geometric_symmetric(n: int, q: float) -> np.ndarray:
    """Symmetric geometric: a_i ∝ q^i + q^{n-1-i}."""
    i = np.arange(n, dtype=np.float64)
    a = q ** i + q ** (n - 1 - i)
    a /= a.sum()
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def arcsine_shape(n: int) -> np.ndarray:
    """a_i ∝ 1 / sqrt(x (1-x))   (Beta(½,½) distribution)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = 1.0 / (np.pi * np.sqrt(x * (1.0 - x)))
    a /= a.sum()
    return a


def double_power_law(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p) + (n + 1 - idx) ** (-p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def smoothmax_and_grad(a: np.ndarray, beta: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and its gradient
    with respect to a.  The gradient corresponds to the (soft) sub‑gradient
    of the max, i.e. g = Σ_k w_k·∂conv_k/∂a  + a  where
    w_k = exp(β·conv_k) / Σ_j exp(β·conv_j) (softmax weights).
    """
    n = a.shape[0]
    conv = full_convolution(a)                 # length 2n‑1
    maxc = conv.max()
    # stabilise exponentiation
    exp_term = np.exp(beta * (conv - maxc))
    Z = exp_term.sum()
    w = exp_term / Z                           # softmax over convolution entries

    # gradient part g1_j = Σ_k w_k·a_{k-j}
    w_rev = w[::-1]
    corr = fft_convolve(a, w_rev)              # length 3n‑2
    # extract indices 2n‑2‑j  (0 ≤ j < n)
    g1 = corr[2 * n - 2 - np.arange(n)]

    grad = g1 + a                               # full gradient of smooth max
    smooth_val = maxc + np.log(Z) / beta        # smooth approximation of the true max
    return smooth_val, grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (EG) on the smooth max objective
# ----------------------------------------------------------------------
def exponential_gradient_descent(a0: np.ndarray,
                                 deadline: float,
                                 max_iter: int = 200_000,
                                 eta0: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Run EG with a gradually increasing smoothing parameter β.
    Returns the best vector found and its (exact) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    n = a.shape[0]

    while time.time() < deadline and t < max_iter:
        # schedule for the smoothness (start smooth, finish close to true max)
        beta = min(5000.0, 1.0 + 0.5 * t)        # linear increase, capped
        _, grad = smoothmax_and_grad(a, beta)

        # step size (standard 1/√t schedule)
        eta = eta0 / math.sqrt(t + 1.0)

        # multiplicative EG update – clip to avoid overflow/underflow
        delta = -eta * grad
        delta = np.clip(delta, -20.0, 20.0)
        a = a * np.exp(delta)

        # renormalise to the simplex (positivity already guaranteed)
        s = a.sum()
        if s == 0.0:
            a = uniform_shape(n)
        else:
            a /= s

        # occasional exact evaluation (cheap compared to the smooth step)
        if t % 64 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()

        t += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Classic sub‑gradient polishing (the reference method, kept as a
# final “polish” because it is cheap and often yields a small extra gain)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray,
                       deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (as a last‑ditch refinement)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M, best_a, a = M_cand, cand.copy(), cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Expanded parametric sweep (adds the new families)
# ----------------------------------------------------------------------
def parametric_sweep_extended(n: int, deadline: float) -> np.ndarray | None:
    best_vec = None
    best_val = float('inf')

    # 1. Power‑law
    for p in np.linspace(0.05, 2.5, 250):
        if time.time() >= deadline:
            return best_vec
        a = power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # 2. Double power‑law (symmetric edges)
    for p in np.linspace(0.05, 1.5, 120):
        if time.time() >= deadline:
            return best_vec
        a = double_power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # 3. Geometric symmetric
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 81),
            np.linspace(0.990, 0.9999, 120))):
        if time.time() >= deadline:
            return best_vec
        a = geometric_symmetric(n, q)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # 4. Mixed plateau + power‑law
    for L_frac in np.linspace(0.02, 0.30, 15):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 5. Beta‑shape grid
    alphas = np.linspace(-0.5, 3.0, 15)
    betas = np.linspace(-0.5, 3.0, 15)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 6. Arcsine shape (new)
    a = arcsine_shape(n)
    val = compute_score(a)
    if val < best_val:
        best_val, best_vec = val, a

    # 7. Random convex combinations of a few basic primitives
    uniform = uniform_shape(n)
    pow05 = power_law(n, 0.5)
    geom98 = geometric_symmetric(n, 0.98)
    basics = [uniform, pow05, geom98]
    for _ in range(2000):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(len(basics))
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basics))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # 0. optional fallback (previous best known construction)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. broad parametric sweep (≈ 300 s)
    # ------------------------------------------------------------------
    sweep_deadline = deadline - 400.0
    n = 2048
    if time.time() < sweep_deadline:
        vec = parametric_sweep_extended(n, sweep_deadline)
        if vec is not None:
            s = compute_score(vec)
            print(f"[parametric sweep] best exact score = {s:.6f}")
            if s < best_val:
                best_val, best_seq = s, vec.tolist()

    # ------------------------------------------------------------------
    # 2. Exponentiated‑gradient descent (main optimisation phase)
    # ------------------------------------------------------------------
    eg_deadline = deadline - 120.0
    if best_seq is not None and time.time() < eg_deadline:
        a0 = np.array(best_seq, dtype=float)
        # run a few independent restarts – each gets a share of the remaining time
        remaining = eg_deadline - time.time()
        for run in range(3):
            sub_deadline = time.time() + remaining / (3 - run)
            a_opt, val_opt = exponential_gradient_descent(
                a0,
                sub_deadline,
                max_iter=150_000,
                eta0=200.0
            )
            print(f"[EG descent {run}] best exact score = {val_opt:.6f}")
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.tolist()
                a0 = a_opt   # seed the next run with the improved vector

    # ------------------------------------------------------------------
    # 3. Sub‑gradient polishing (quick final refinement)
    # ------------------------------------------------------------------
    polish_deadline = deadline - 30.0
    if best_seq is not None and time.time() < polish_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a_start, polish_deadline, step0=0.12)
        val = compute_score(a_polished)
        print(f"[sub‑grad] polished exact score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climb (last few seconds)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final exact score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final reporting using the official evaluate_sequence
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
