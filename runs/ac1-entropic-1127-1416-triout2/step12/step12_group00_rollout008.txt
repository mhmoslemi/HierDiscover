"""
non‑negative step‑function optimiser for ``evaluate_sequence``.

The algorithm consists of the following stages (all respecting a
1000 s wall‑clock budget).

1. **Exploration** – draw many random Dirichlet vectors with several
   concentration parameters, generate a few parametric families
   (power‑law, exponential, beta, Gaussian, sine‑power) and keep the
   best few (including their monotone‑decreasing versions).

2. **Adam‑based smooth‑max descent** – minimise a smooth
   log‑sum‑exp approximation of the peak of the full convolution.
   The gradient is obtained with a single FFT.  The optimiser uses the
   Adam update rule, an annealing temperature ``τ`` and a simple
   non‑negative simplex projection (clip then renormalise).  The run is
   repeated from several starting points.

3. **Projected sub‑gradient polishing** – classic sub‑gradient steps
   on the exact objective (the true maximum of the convolution).  A
   decreasing step‑size and occasional jitter help to escape plateaus.

4. **Final refinement** – a short simulated‑annealing hill‑climb and
   a deterministic “sort‑descending” re‑ordering (largest entries are
   placed opposite the smallest ones, which by the rearrangement
   inequality minimises the central autocorrelation).

The best sequence found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to unit sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fall back to the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[:m]
    return conv


def exact_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return project_to_simplex(a)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    return project_to_simplex(a)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    a[~np.isfinite(a)] = 0.0
    return project_to_simplex(a)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return project_to_simplex(a)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    return project_to_simplex(a)

# Exploration (baseline)
# ----------------------------------------------------------------------


def try_update_best(vec: np.ndarray, best, max_keep: int):
    """Update the list of best candidates (sorted by score)."""
    if not np.isfinite(vec).all():
        return
    sc = exact_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def baseline_exploration(n: int, deadline: float, max_keep: int = 8):
    """Generate a diverse pool of candidates and keep the best few."""
    best = []  # list of (score, vector)

    # 1) parametric families (coarse sweeps)
    for p in np.linspace(0.1, 2.0, 81):
        if time.time() >= deadline:
            break
        try_update_best(generate_powerlaw(n, p), best, max_keep)

    for lam in np.linspace(0.0005, 0.25, 120):
        if time.time() >= deadline:
            break
        try_update_best(generate_exponential(n, lam), best, max_keep)

    al_vals = np.linspace(0.2, 4.0, 27)
    be_vals = np.linspace(0.2, 4.0, 27)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_update_best(generate_beta(n, alpha, beta), best, max_keep)

    for sigma in np.linspace(0.1 * n, 0.5 * n, 20):
        if time.time() >= deadline:
            break
        try_update_best(generate_gaussian(n, sigma), best, max_keep)

    for p in np.linspace(0.1, 6.0, 25):
        if time.time() >= deadline:
            break
        try_update_best(generate_sine_power(n, p), best, max_keep)

    # 2) random Dirichlet draws (different concentrations)
    concentrations = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_c = 2500 // len(concentrations)
    for conc in concentrations:
        for _ in range(draws_per_c):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_update_best(a, best, max_keep)

            # also keep a monotone‑decreasing version (often better)
            a_sorted = np.sort(a)[::-1]
            try_update_best(a_sorted, best, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

# Smooth‑max optimiser (Adam)
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.40,
    tau_end: float = 1e-7,
    tau_decay: float = 0.96,
    lr: float = 0.12,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """Adam optimiser for the smooth log‑sum‑exp approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    m_exp = np.zeros_like(a)
    v_exp = np.zeros_like(a)

    best_a = a.copy()
    best_val = exact_score(a)

    t = 0
    tau = tau_start

    while time.time() < deadline:
        # anneal temperature
        if t % 80 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # full convolution a*a
        conv = full_convolution(a)
        shift = conv.max()
        # soft‑max weights
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()               # length = 2n‑1

        # gradient of the smooth max: 2 * (a * w_rev) (cross‑correlation)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        corr = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        # Adam update
        t += 1
        m_exp = beta1 * m_exp + (1 - beta1) * grad
        v_exp = beta2 * v_exp + (1 - beta2) * (grad ** 2)
        m_hat = m_exp / (1 - beta1 ** t)
        v_hat = v_exp / (1 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto the feasible set (non‑negative simplex)
        a = project_to_simplex(a)

        # occasional jitter to shake out of plateaus
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr * 0.004 / np.sqrt(t + 1.0))
            a = a + jitter
            a = project_to_simplex(a)

        cur_val = exact_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a

def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.35):
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = exact_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # active set (within 1e‑9 relative tolerance)
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient accumulation
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = exact_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (final stochastic climb)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float):
    """Low‑temperature Metropolis hill‑climb."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = exact_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = 1e-3
    final_T = 1e-6
    cooling = 0.9995
    step_scale = 0.07

    it = 0
    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0.0:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        val = exact_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
            step_scale = min(step_scale * 1.02, 0.5)
        else:
            step_scale = max(step_scale * 0.98, 1e-5)

        T = max(final_T, T * cooling)

        # tiny perturbation once in a while
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.02)
            a = a + jitter
            a = project_to_simplex(a)
            cur_val = exact_score(a)

    return best_a


# ----------------------------------------------------------------------
# Final helper: deterministic monotone re‑ordering
# ----------------------------------------------------------------------


def monotone_descending(vec: np.ndarray) -> np.ndarray:
    """Return the decreasingly sorted version of ``vec`` (preserves sum)."""
    return np.sort(vec)[::-1] / np.sort(vec)[::-1].sum()

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 28957, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # a few seconds safety margin

    # ------------------------------------------------------------------
    # Determine problem size (use fallback length if available)
    # ------------------------------------------------------------------
    n = 2048
    fallback = None
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        n = len(fallback)
    except Exception:
        pass

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback (previous best known construction)
    # ------------------------------------------------------------------
    if fallback is not None:
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback

    # ------------------------------------------------------------------
    # 1.  Broad exploration (≈ 130 s)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 130.0
    if time.time() < explore_deadline:
        cand_vectors = baseline_exploration(n, explore_deadline, max_keep=10)
        for vec in cand_vectors:
            val = exact_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[exploration] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Smooth‑max Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    # Allocate most of the remaining budget to this stage
    remaining = deadline - time.time()
    # leave ≈ 250 s for polishing/annealing
    grad_budget = max(0.0, remaining - 250.0)
    if grad_budget > 30.0 and best_seq is not None:
        print("[smooth‑max] start multi‑restart Adam optimisation")
        grad_deadline = time.time() + grad_budget
        # Prepare a pool of starting points: current best + random Dirichlet draws
        start_pool = []
        best_vec = np.array(best_seq, dtype=float)
        start_pool.append(project_to_simplex(best_vec))
        start_pool.append(np.full(n, 1.0 / n))
        start_pool.append(monotone_descending(best_vec))
        for _ in range(4):
            rand = np.random.dirichlet([0.5] * n)
            start_pool.append(project_to_simplex(rand))
        # run Adam from each start while time permits
        for idx, init in enumerate(start_pool):
            if time.time() >= grad_deadline:
                break
            local_deadline = min(grad_deadline, time.time() + 40.0)  # ≤40 s per run
            a_opt = smooth_max_adam(init, local_deadline)
            val = exact_score(a_opt)
            print(f"[smooth‑max #{idx}] refined score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
        print(f"[smooth‑max] best after this phase = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Sub‑gradient polishing (focus on active constraints)
    # ------------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_ref = subgradient_polish(a0, polish_deadline)
        val = exact_score(a_ref)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_ref.tolist()

    # ------------------------------------------------------------------
    # 4.  Simulated annealing (final stochastic hill‑climb)
    # ------------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = exact_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # 5.  Deterministic monotone re‑ordering (often improves the bound)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_mono = monotone_descending(np.array(best_seq, dtype=float))
        val = exact_score(a_mono)
        print(f"[monotone] score after sorting = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_mono.tolist()

    # ------------------------------------------------------------------
    # Safeguard – fall back to uniform if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the function we must minimise)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
