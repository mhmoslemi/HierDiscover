def project_simplex(v):
    """Euclidean projection onto simplex {x>=0, sum=1}."""
    if v.sum() == 1 and np.all(v >=0):
        return v
    # algorithm by Wang & Carreira-Perpinan
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1.  **Global sampling** – a short (≈30 s) sweep of three parametric families
    (power‑law, exponential, beta) and many random Dirichlet draws.
    The best sample is symmetrised and projected onto the simplex.

2.  **Smoothed Adam optimisation** – minimise a smooth surrogate of the
    true objective

        fβ(a) = max(conv(a,a)) + (1/β)·log Σ exp(β·conv(a,a))

    via Adam with an ever‑increasing smoothing parameter β.
    Gradients are obtained analytically and evaluated with FFT‑based
    convolutions.  After each update we project back onto the simplex
    and enforce symmetry (aᵢ = a_{n‑1‑i}), which is provably optimal for this
    problem.

3.  **Exact EG polishing** – a few thousand exponentiated‑gradient steps
    on the original max‑convolution objective, again using FFT for the
    sub‑gradient (the active set is the set of indices attaining the
    maximum).  This fine‑tunes the solution.

4.  **Optional simulated‑annealing** – a short Metropolis hill‑climb to
    escape possible plateaus.

All stages respect a global 1000 s wall‑clock budget (a 5 s safety margin
is kept).  The best sequence found is returned as a plain Python list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, O(N log N))
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Convolution of two real vectors using a real FFT.
    Returns the full linear convolution (len = len(x)+len(y)-1).
    """
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    c = np.fft.irfft(X * Y, n=n_fft)
    return c[:n]


# ----------------------------------------------------------------------
# Scoring utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # if already feasible – fast path
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (produce probability vectors)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return project_simplex(a)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    return project_simplex(a)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    return project_simplex(a)


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed surrogate of the max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  lr: float = 0.05,
                  beta_start: float = 5.0,
                  beta_max: float = 200.0,
                  beta_inc_iter: int = 2000) -> np.ndarray:
    """
    Adam minimisation of the smoothed objective

        fβ(a) = max(conv) + (1/β)·log Σ exp(β·conv)

    The iterates stay on the simplex and are kept exactly symmetric:
    aᵢ = a_{n‑1‑i}.
    """
    a = a0.copy()
    a = project_simplex(a)
    a = (a + a[::-1]) / 2.0               # enforce symmetry

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    it = 0
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        it += 1

        # ----- smoothed objective & softmax weights -----
        conv = full_convolution(a)
        max_conv = conv.max()

        # robust softmax (subtract max to avoid overflow)
        exps = np.exp(beta * (conv - max_conv))
        sum_exps = exps.sum()
        if sum_exps == 0 or not np.isfinite(sum_exps):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = exps / sum_exps

        # ----- gradient of the smoothed objective -----
        # g_i = 2 * Σ_k w_k a_{k-i}
        a_rev = a[::-1]
        N = 1 << ((len(w) + len(a_rev) - 1 - 1).bit_length())
        W = np.fft.rfft(w, n=N)
        A_rev = np.fft.rfft(a_rev, n=N)
        conv_w_arev = np.fft.irfft(W * A_rev, n=N)
        start = len(a) - 1
        g = 2.0 * conv_w_arev[start:start + len(a)]

        # ----- Adam update -----
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step

        # ----- projection + symmetrisation -----
        a = project_simplex(a)
        a = (a + a[::-1]) / 2.0

        # ----- keep best (actual objective) -----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # ----- increase smoothing parameter gradually -----
        if it % beta_inc_iter == 0:
            beta = min(beta * 1.5, beta_max)

    return best_a


# ----------------------------------------------------------------------
# Exact EG polishing on the true max‑convolution objective
# ----------------------------------------------------------------------


def eg_fine_polish(a0: np.ndarray, deadline: float,
                   eta0: float = 0.5,
                   tol: float = 1e-12,
                   max_iters: int = 200_000) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the exact objective
    f(a) = max(conv(a,a)).
    Uses the active set (indices attaining the current maximum) to form the
    true sub‑gradient.  The iterates remain on the simplex and stay symmetric.
    """
    a = a0.copy()
    a = project_simplex(a)
    a = (a + a[::-1]) / 2.0

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = full_convolution(a)
        max_conv = conv.max()
        # active set (allow a tiny relative tolerance)
        eps = max_conv * tol
        active = (conv >= max_conv - eps).astype(float)

        # gradient = 2 * Σ_{k active} a_{k-i}
        a_rev = a[::-1]
        N = 1 << ((len(active) + len(a_rev) - 1 - 1).bit_length())
        ACT = np.fft.rfft(active, n=N)
        A_rev = np.fft.rfft(a_rev, n=N)
        conv_act_arev = np.fft.irfft(ACT * A_rev, n=N)
        start = len(a) - 1
        g = 2.0 * conv_act_arev[start:start + len(a)]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        a = project_simplex(a)
        a = (a + a[::-1]) / 2.0

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing (optional final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """
    Simple Metropolis hill‑climb that occasionally accepts worsening moves.
    Operates on the simplex and respects symmetry.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # choose two distinct indices (both must have non‑zero mass)
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j or a[j] <= 0.0:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand = np.maximum(cand, 0.0)
        cand = project_simplex(cand)
        cand = (cand + cand[::-1]) / 2.0

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional global jitter to keep things moving
        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = project_simplex(a)
            a = (a + a[::-1]) / 2.0
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted wall‑clock time.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between expressivity and speed
    # ------------------------------------------------------------------
    n = 4096

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback from a previously stored solution (if any)
    # ------------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fb_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb_seq
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Global baseline exploration (≈30 s)
    # ------------------------------------------------------------------
    baseline_deadline = start_time + 30.0
    candidates = []

    # power‑law sweep
    for p in np.linspace(0.2, 1.5, 30):
        if time.time() >= baseline_deadline:
            break
        candidates.append(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(0.001, 0.03, 30):
        if time.time() >= baseline_deadline:
            break
        candidates.append(generate_exponential(n, lam))

    # beta sweep (coarse)
    al_vals = np.linspace(0.5, 3.0, 8)
    be_vals = np.linspace(0.5, 3.0, 8)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= baseline_deadline:
                break
            candidates.append(generate_beta(n, alpha, beta))

    # massive random Dirichlet draws with varied concentration
    while time.time() < baseline_deadline:
        conc = np.random.choice([0.2, 0.5, 1.0, 2.0, 4.0])
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # evaluate candidates, keep the best
    for vec in candidates:
        vec = (vec + vec[::-1]) / 2.0            # enforce symmetry
        vec = project_simplex(vec)
        score = compute_score(vec)
        if score < best_val:
            best_val = score
            best_seq = vec.copy()
    print(f"[baseline] best known score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation (major chunk of the remaining budget)
    # ------------------------------------------------------------------
    adam_deadline = deadline - 150.0   # keep ~2 min for EG + annealing
    if time.time() < adam_deadline:
        print(f"[adam] optimisation until t ≈ {adam_deadline - start_time:.1f}s")
        refined = adam_optimize(best_seq, adam_deadline)
        refined = (refined + refined[::-1]) / 2.0
        refined = project_simplex(refined)
        refined_score = compute_score(refined)
        print(f"[adam] refined score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val, best_seq = refined_score, refined.copy()

    # ------------------------------------------------------------------
    # 3. Exact EG polishing (≈2 min)
    # ------------------------------------------------------------------
    eg_deadline = deadline - 30.0
    if time.time() < eg_deadline:
        print(f"[eg] polishing until t ≈ {eg_deadline - start_time:.1f}s")
        polished = eg_fine_polish(best_seq, eg_deadline)
        polished_score = compute_score(polished)
        print(f"[eg] polished score = {polished_score:.6f}")
        if polished_score < best_val:
            best_val, best_seq = polished_score, polished.copy()

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing polish (last ~25 s)
    # ------------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if time.time() < anneal_deadline:
        print(f"[anneal] final pass until t ≈ {anneal_deadline - start_time:.1f}s")
        final = simulated_annealing(best_seq, anneal_deadline)
        final_score = compute_score(final)
        print(f"[anneal] final score = {final_score:.6f}")
        if final_score < best_val:
            best_val, best_seq = final_score, final.copy()

    # ------------------------------------------------------------------
    # 5. Fallback to uniform distribution if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 6. Report via the official evaluation function
    # ------------------------------------------------------------------
    reported = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {reported:.6f}")
    return best_seq.tolist()
