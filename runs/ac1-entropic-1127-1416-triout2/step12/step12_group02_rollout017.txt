def build_constraints(f, active_idx):
    n = len(f)
    rows = []
    cols = []
    data = []
    for row_i, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i values range
        i_vals = np.arange(i_start, i_end+1)
        j_vals = k - i_vals
        rows.extend([row_i] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())
    # create coo then csr
    A = coo_matrix((data, (rows, cols)), shape=(len(active_idx), n), dtype=np.float64).tocsr()
    return A

def lp_optimize(f, M, active_idx):
    n = len(f)

"""Improved step‑function search.

This implementation follows a multi‑stage strategy:

1. **Coarse analytic families** – a quick scan of a few 1‑D families
   (product, exponential‑U, mixed) on a reasonably large support.
2. **Fine 1‑D optimisation** – golden‑section search for the product family.
3. **Active‑set linear‑programming improvement** – given the current vector
   `f`, solve a linear program that maximises the ℓ₁‑mass of a new vector `g`
   subject to the active convolution constraints `f∗g ≤ M`.  Blend `g` into
   `f` with a small step size, symmetrise and re‑project onto the simplex.
   Only the constraints that are near‑tight (within 0.01 % of the current
   maximum) are kept, keeping the LP tiny.
4. **Smooth‑max Adam optimisation** – minimise the smooth approximation of
   the true max‑objective using a log‑softmax parametrisation.
5. **Projected sub‑gradient descent** on the exact max‑objective.
6. **Final simulated‑annealing hill‑climb** for polishing.

The algorithm adapts the support size (`n`) to the remaining time
(8192 or 16384) and respects the 1000 s wall‑clock budget.  The best
sequence found (a probability vector, i.e. sum = 1, non‑negative) is
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric product family: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Guard against overflow / underflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Linear‑programming active‑set improvement
# ----------------------------------------------------------------------


def build_active_constraint_matrix(f: np.ndarray, active_idx: np.ndarray):
    """Return a sparse (len(active_idx) × n) matrix representing
    (f ∗ g)[k] ≤ M for the active shifts k."""
    import scipy.sparse as sp

    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals                     # positions of g
        rows.extend([row] * i_vals.size)
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())
    A = sp.coo_matrix((data, (rows, cols)), shape=(len(active_idx), n), dtype=np.float64).tocsr()
    return A


def lp_step(f: np.ndarray, M: float, active_idx: np.ndarray):
    """Solve   max 1ᵀg   s.t.   f∗g ≤ M   on the active indices.
    Returns the optimal g (projected to the simplex) or None on failure."""
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception:
        return None

    A = build_active_constraint_matrix(f, active_idx)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)          # minimise –sum(g)  ⇔ maximise sum(g)
    bounds = [(0.0, None)] * f.shape[0]

    # HiGHS works with sparse A_ub directly
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = project_to_simplex(res.x)
    return g


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (same as the previous code)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 10.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.015,
                         sym_interval: int = 1500) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate of max(conv)."""
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # symmetrisation
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (same as previous code)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent for the exact max‑objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (same as previous code)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a simple cooling schedule."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Golden‑section optimiser (for 1‑D families)
# ----------------------------------------------------------------------


def golden_section_optimize(func, a: float, b: float, max_iter: int = 20) -> tuple[float, float]:
    """Minimise `func` on [a,b] using golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a, c, fc = c, d, fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b, d, fd = d, c, fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns a plain Python list.
    """
    # optional linear‑programming facilities
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
        _lp_available = True
    except Exception:
        _lp_available = False

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # Choose support size (larger n gives a finer shape but costs more)
    # -----------------------------------------------------------
    # Use the larger size if we have enough time left after the quick scan.
    provisional_n = 16384
    if deadline - time.time() < 300:          # not enough time for a huge FFT loop
        provisional_n = 8192
    n = provisional_n
    print(f"[{time.time():.1f}] Using n = {n}")

    # -----------------------------------------------------------
    # 0.  If a fallback from previous run is supplied, try it.
    # -----------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Coarse scan of a few analytic families (fast)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] Coarse family scan (n={n})")

    # Uniform baseline (mostly for sanity)
    vec = np.full(n, 1.0 / n, dtype=np.float64)
    sc = compute_score(vec)
    if sc < best_score:
        best_score = sc
        best_seq = vec.tolist()
        print(f"[{time.time():.1f}] new best (uniform) = {best_score:.6f}")

    # ---- Symmetric product family ------------------------------------------------
    best_alpha = None
    best_alpha_sc = float('inf')
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 120.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (product α={alpha:.3f}) = {best_score:.6f}")
        if sc < best_alpha_sc:
            best_alpha_sc = sc
            best_alpha = alpha
    # -----------------------------------------------------------
    # 2. Fine golden‑section refinement of the product family
    # -----------------------------------------------------------
    if best_alpha is not None:
        lo = max(0.0, best_alpha - 0.5)
        hi = min(12.0, best_alpha + 0.5)

        def score_alpha(a):
            return compute_score(symmetric_product_family(n, a))

        α_opt, α_sc = golden_section_optimize(score_alpha, lo, hi, max_iter=25)
        vec = symmetric_product_family(n, α_opt)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] refined product α={α_opt:.6f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Exponential‑U family (quick scan)
    # -----------------------------------------------------------
    best_gamma = None
    best_gamma_sc = float('inf')
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 120.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")
        if sc < best_gamma_sc:
            best_gamma_sc = sc
            best_gamma = gamma
    # -----------------------------------------------------------
    # 4. Mixed uniform + U (tiny 2‑D grid)
    # -----------------------------------------------------------
    best_mix = None
    best_mix_sc = float('inf')
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 120.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")
            if sc < best_mix_sc:
                best_mix_sc = sc
                best_mix = (lam, gamma)

    # -----------------------------------------------------------
    # 5. Active‑set linear‑programming improvement (if scipy is present)
    # -----------------------------------------------------------
    if _lp_available and best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] entering LP‑based refinement")
        f = np.array(best_seq, dtype=np.float64)
        lp_iters = 0
        while time.time() < deadline - 30.0 and lp_iters < 20:
            conv_f = convolution_fft(f, f)
            M = conv_f.max()
            # Choose constraints that are within 0.01 % of the maximum.
            tol = max(1e-12, M * 1e-4)
            active_idx = np.where(conv_f >= M - tol)[0]
            # If too many active constraints, keep only the largest few.
            if active_idx.size > 30:
                # keep the top‑30 values
                topk = np.argsort(conv_f)[-30:]
                active_idx = np.sort(topk)

            g = lp_step(f, M, active_idx)
            if g is None:
                break

            # blend with a modest step size; if no improvement, shrink the step.
            improved = False
            step = 0.3
            for _ in range(5):
                f_new = (1.0 - step) * f + step * g
                f_new = (f_new + f_new[::-1]) * 0.5
                f_new = project_to_simplex(f_new)
                sc_new = compute_score(f_new)
                if sc_new < best_score - 1e-9:
                    best_score = sc_new
                    best_seq = f_new.tolist()
                    f = f_new
                    improved = True
                    print(f"[{time.time():.1f}] LP iter {lp_iters+1} improved to {best_score:.6f}")
                    break
                step *= 0.5
            if not improved:
                # no progress – stop early
                break
            lp_iters += 1

    # -----------------------------------------------------------
    # 6. First Adam (smooth‑max) refinement
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] first Adam refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 120.0,
            max_iters=400_000,
            beta_start=10.0,
            beta_end=8000.0,
            lr=0.015,
            sym_interval=2000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after first Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Projected sub‑gradient descent (exact max)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 80.0,
            max_iters=1_500_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑gradient best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 8. Second Adam pass (more aggressive β)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] second Adam pass")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = adam_smooth_optimize(
            a_start,
            deadline - 40.0,
            max_iters=200_000,
            beta_start=50.0,
            beta_end=12000.0,
            lr=0.012,
            sym_interval=1500,
        )
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.tolist()
            print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 9. Explicit symmetrisation (sometimes helps)
    # -----------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 10. Final hill‑climb (simulated annealing)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 11. Fallback to uniform if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
