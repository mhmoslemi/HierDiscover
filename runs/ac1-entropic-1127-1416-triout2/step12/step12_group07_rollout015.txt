```
def adam_refine(a0, deadline):
    # parameters
    a = a0.copy()
    best = a.copy()
    best_score = compute_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta_temp = 5.0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_conv = conv.max()
        # smooth max using current beta_temp
        shift = conv - max_conv
        w = np.exp(beta_temp * shift)
        p = w / w.sum()
        # gradient: 2 * correlate(p, a)
        grad = 2.0 * np.correlate(p, a, mode='valid')
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.1 / (np.sqrt(v_hat) + eps) * m_hat
        cand = a - step
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
        # acceptance: always move to cand? we could keep cand if improvement else maybe revert?
        if cand_score <= compute_score(a) or np.random.rand() < 0.001:
            a = cand
        # increase beta_temp gradually
        beta_temp = min(beta_temp * 1.02, 200.0)
        # occasional restart jitter if stuck
        if t % 500 == 0:
            # jitter
            jitter = np.random.randn(a.shape[0]) * 0.001
            a = project_to_simplex(a + jitter)
    return best
```

```
def smooth_max_descent(a0, deadline, max_iter=2000):
    a = a0.copy()
    best = a.copy()
    best_score = compute_score(a)
    t = 0
    beta = 5.0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_base = 0.5  # base step size scaling
    
    while time.time() < deadline and t < max_iter:
        t += 1
        # compute smooth max
        conv = full_convolution(a)
        max_conv = conv.max()
        # compute softmax weights
        w = np.exp(beta * (conv - max_conv))  # max shift for stability.
        p = w / w.sum()
        # gradient
        grad = 2.0 * np.correlate(p, a, mode='valid')
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step_base * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
        # accept candidate if improvement else maybe keep old
        if cand_score <= compute_score(a):
            a = cand
        # adjust beta (increase)
        beta = min(beta * 1.02, 200.0)
        # occasional jitter
        if t % 500 == 0:
            jitter = np.random.randn(a.shape[0]) * 0.001
            a = project_to_simplex(a + jitter)
    return best
```

```
def hill_climb_anneal(a_start, deadline, max_iter=400000):
    a = a_start.copy()
    cur_score = compute_score(a)
    best = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    step_scale = 0.05
    T0 = 0.01
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        if delta == 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / (T0 * (1 - it / max_iter))):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                step_scale = min(step_scale * 1.05, 0.6)  # increase step after success
        else:
            step_scale *= 0.99
    return best
```

```python
"""
Optimising a non‑negative step function to minimise
evaluate_sequence() = 2·n·max(conv(a,a)) / (sum(a)²).

Approach
--------
1. **Broad exploratory sweep** – generate many candidate shapes:
   power‑law, geometric, beta‑shape, cosine‑power, Gamma,
   log‑normal, logistic, random Dirichlet draws and sparse
   uniform spikes.  Keep the best few (lowest true score).

2. **Projected smooth‑max gradient descent** – minimise a
   smooth approximation of the L∞‑norm:
   `smooth_max(conv) = (1/β)·log Σ exp(β·conv)`.
   The gradient can be expressed as a correlation between the
   soft‑max weights and the current vector.  An Adam‑style
   update with Euclidean projection onto the probability simplex
   is performed while gradually increasing β (sharpening the
   approximation).

3. **Simulated‑annealing hill‑climb** – final fine‑tuning by
   moving a fraction of mass between two random indices.
   Accept improvements and, with a slowly decreasing temperature,
   also occasional worsening moves to escape plateaus.

The whole pipeline respects a 1000 s wall‑clock budget and returns
the best sequence found (as a plain Python list).  The routine
relies only on numpy and the supplied ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        rho = 0
    else:
        rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of appropriate size."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence when Σa == 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Parametric families

def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a

def geometric(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a

def beta_shape(n: int, beta: float) -> np.ndarray:
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a

def cosine_power(n: int, expo: float) -> np.ndarray:
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a

def gamma_shape(n: int, k: float, theta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (k - 1.0)) * np.exp(-i / theta)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a

def lognormal_shape(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.exp(- (np.log(i) - mu) ** 2.0 / (2.0 * sigma ** 2.0))
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a

def logistic_decay(n: int, mu: float, s: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - mu) / s))
    a = np.maximum(a, 0.0)
    ssum = a.sum()
    if ssum == 0:
        return None

# Smooth‑max gradient descent (Adam style) with projection
# ----------------------------------------------------------------------
def smooth_max_descent(a0: np.ndarray, deadline: float, max_iter: int = 5000) -> np.ndarray:
    a = a0.copy()

    n = a.shape[0]
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_base = 0.5
    beta = 5.0                     # temperature for the soft‑max
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1
        conv = full_convolution(a)
        max_conv = conv.max()

        # soft‑max weights (numerically stable)
        w = np.exp(beta * (conv - max_conv))
        p = w / w.sum()

        # gradient = 2 * correlation(p, a)
        grad = 2.0 * np.correlate(p, a, mode='valid')

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step_base * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()

        # Accept if it improves the true objective, otherwise keep old a
        if cand_score <= compute_score(a):
            a = cand

        # slowly sharpen the approximation
        beta = min(beta * 1.02, 250.0)

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(a + jitter)

    return best

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        rho = 0
    else:
        rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of appropriate size."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence when Σa == 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def geometric(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, beta: float) -> np.ndarray:
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def cosine_power(n: int, expo: float) -> np.ndarray:
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def gamma_shape(n: int, k: float, theta: float) -> np.ndarray:

    a = (i ** (k - 1.0)) * np.exp(-i / theta)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def lognormal_shape(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.exp(- (np.log(i) - mu) ** 2.0 / (2.0 * sigma ** 2.0))
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a

    a = np.maximum(a, 0.0)
    ssum = a.sum()
    if ssum == 0:
        return None
    a /= ssum
    return a

        a_final = hill_climb_anneal(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – uniform distribution if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Official evaluation (includes the security wrappers)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported (official) score = {final_score:.6f}")
    return best_seq
```
