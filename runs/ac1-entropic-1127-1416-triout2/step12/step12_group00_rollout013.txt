def normalize_simplex(v):
    v = np.maximum(v, 0)
    s = v.sum()
    if s == 0:
        return np.full_like(v, 1.0/len(v))
    return v / s

def adam_optimize(a0, deadline, tau_start=0.3, tau_end=1e-6, tau_decay=0.98,
                  lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    n = len(a)
    m = 2*n-1
    # initial tau
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        t += 1
        # compute convolution
        conv = full_convolution(a)   # length m
        # smoothmax weights
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()
        # gradient via FFT cross-correlation
        rev_w = w[::-1]
        size = 1 << ((n + len(rev_w) - 1 - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + len(rev_w) - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * (grad * grad)
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = normalize_simplex(a)

        # track best using true score
        if t % 200 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        # anneal tau occasionally
        if t % 1000 == 0:
            tau = max(tau_end, tau * tau_decay)
        # occasional small jitter
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (lr * 0.001 / np.sqrt(t))
            a = a + jitter
            a = normalize_simplex(a)

    return best_a

"""
non‑negative step‑function optimiser for ``evaluate_sequence``

The optimiser works in three stages:

1. **Broad seeding** – a large, diverse pool of initial vectors is built
   from several parametric families (beta, power‑law, Gaussian, sine‑power,
   Hann) and from random Dirichlet draws.  All vectors are projected to the
   probability simplex (sum = 1, non‑negative).

2. **Adam‑based smooth‑max descent** – each seed is refined by minimising a
   smooth approximation of the max‑convolution:
   fτ(a)=τ·log∑exp(conv(a)/τ).  The gradient is obtained by a single FFT‑based
   cross‑correlation, and the Adam optimiser is used together with an
   exponential annealing of τ.  The iterates are re‑normalised after every
   step.  The best vector over all runs is kept.

3. **Final polishing** – a classic sub‑gradient method (active‑set) and a
   short simulated‑annealing hill‑climb are applied to the best candidate
   found in the second stage.

The whole pipeline respects the 1000 s wall‑clock budget and prints the
progress and intermediate scores.  The returned list is a plain Python list
of non‑negative floats (already scaled – the evaluation function is scale
invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def normalize_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / total


def next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n‑1) using a real FFT.
    The result is exact up to floating‑point noise.
    """
    n = a.shape[0]
    m = 2 * n - 1
    size = next_pow2(m)
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector whose sum is 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return normalize_simplex(a)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    return normalize_simplex(a)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    # replace infinities / NaNs that can appear for extreme parameters
    a[~np.isfinite(a)] = 0.0
    return normalize_simplex(a)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return normalize_simplex(a)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    return normalize_simplex(a)


def generate_hann(n: int) -> np.ndarray:
    # a_i ∝ sin²(π·i/(n‑1)) which is the classic Hann window
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * i / (n - 1)) ** 2
    return normalize_simplex(a)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-6,
    tau_decay: float = 0.995,
    lr: float = 0.25,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max approximation of max(conv).
    The iterate is kept on the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    size_fft = next_pow2(3 * n - 2)          # enough for a * w_rev
    tau = tau_start

    # Adam state
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----------- convolution of a with itself (for the smooth‑max) ----------
        A = np.fft.rfft(a, n=size_fft)                 # forward FFT once per iter
        conv_full = np.fft.irfft(A * A, n=size_fft)[:m]  # a * a
        max_conv = conv_full.max()

        # ----------- soft‑max weights -----------------------------------------
        shift = max_conv
        w_exp = np.exp((conv_full - shift) / tau)
        w = w_exp / w_exp.sum()                       # length m

        # ----------- gradient of the smooth‑max --------------------------------
        rev_w = w[::-1]                               # length m
        B = np.fft.rfft(rev_w, n=size_fft)
        cross = np.fft.irfft(A * B, n=size_fft)       # length n+m‑1
        idx = (m - 1) - np.arange(n)                  # map to gradient indices
        grad = 2.0 * cross[idx]

        # ----------- Adam update -----------------------------------------------
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m_vec / (1.0 - beta1_pow)
        v_hat = v_vec / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = normalize_simplex(a)

        # ----------- bookkeeping -------------------------------------------------
        if it % 250 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[adam] iter {it}, score {cur_val:.6f}")

        # ----------- temperature schedule ---------------------------------------
        if it % 500 == 0:
            tau = max(tau_end, tau * tau_decay)

        # ----------- occasional jitter to escape plateaus -----------------------
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (lr * 0.002 / np.sqrt(it))
            a += jitter
            a = normalize_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as used previously)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # accumulate sub‑gradient from all active constraints
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = normalize_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = normalize_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (optional final escape)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """Simple Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = normalize_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = normalize_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    n = 2048
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Build a diverse pool of seeds
    # -----------------------------------------------------------------
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n))

    # beta family – very “U” shaped as well as mildly skewed
    for alpha in [0.05, 0.1, 0.2, 0.5, 0.8]:
        for beta in [0.05, 0.1, 0.2, 0.5, 0.8]:
            seeds.append(generate_beta(n, alpha, beta))

    # sine‑power – concentrates mass near the centre for small p, near the edges for large p
    for p in [0.3, 1.0, 3.0, 5.0]:
        seeds.append(generate_sine_power(n, p))

    # power‑law – gentle decay
    for p in [0.1, 0.3, 0.6, 1.0]:
        seeds.append(generate_powerlaw(n, p))

    # Gaussian windows of varying width
    for sigma in [0.15 * n, 0.30 * n, 0.45 * n]:
        seeds.append(generate_gaussian(n, sigma))

    # Hann window (standard smooth window)
    seeds.append(generate_hann(n))

    # random Dirichlet draws with a range of concentrations
    for conc in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(4):
            seeds.append(np.random.dirichlet([conc] * n))

    # add the fallback (if any) to the pool
    if best_seq is not None:
        seeds.append(np.array(best_seq, dtype=float))

    # -----------------------------------------------------------------
    # 2. Adam optimisation on each seed (time‑sliced)
    # -----------------------------------------------------------------
    # give ~80 % of the remaining budget to this stage
    opt_deadline = start_time + budget_s * 0.80
    per_seed_budget = 30.0          # seconds we allow for a single run (adjustable)

    for idx, seed_vec in enumerate(seeds):
        if time.time() >= opt_deadline:
            break
        # ensure feasibility
        a0 = normalize_simplex(seed_vec)

        # budget for this particular run
        remaining = opt_deadline - time.time()
        this_deadline = time.time() + min(per_seed_budget, remaining - 5.0)
        if this_deadline <= time.time():
            break

        refined = adam_optimize(a0, this_deadline)
        cur_val = compute_score(refined)

        if cur_val < best_val:
            best_val, best_seq = cur_val, refined.tolist()
            print(f"[adam seed {idx}] improved score = {cur_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (focus on active constraints)
    # -----------------------------------------------------------------
    polish_deadline = start_time + budget_s * 0.92
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, polish_deadline)
        cur_val = compute_score(a_refined)
        print(f"[polish] refined score = {cur_val:.6f}")
        if cur_val < best_val:
            best_val, best_seq = cur_val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing – final stochastic escape
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 2.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        final_val = compute_score(a_final)
        print(f"[anneal] final score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a_final.tolist()

    # -----------------------------------------------------------------
    # 5. Safeguard – fall back to a uniform vector if nothing better was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
