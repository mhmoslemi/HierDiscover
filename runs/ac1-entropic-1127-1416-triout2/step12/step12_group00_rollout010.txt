def smooth_exp_optim(a0, deadline, tau_start=0.2, tau_end=1e-7, tau_decay=0.95, eta0=0.5):
    a = a0.copy()
    best = a.copy()
    best_val = compute_score(a)
    tau = tau_start
    t = 0
    while time.time() < deadline:
        if t % 200 == 0:
            tau = max(tau_end, tau * tau_decay)
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()
        # gradient = 2 * (a ⋆ w_rev)
        rev_w = w[::-1]
        size = 1 << ((2*a.shape[0] - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[:2*a.shape[0]-1]
        # gradient for a[i] is 2 * conv_full[n-1 - i]
        idx = (2*a.shape[0] - 2) - np.arange(a.shape[0])
        grad = 2.0 * conv_full[idx]
        # multiplicative update
        eta = eta0 / np.sqrt(t + 1)
        a = a * np.exp(-eta * grad)
        # renormalize to sum 1
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / a_sum
        # evaluate
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
        t += 1
    return best

"""
Hybrid optimiser for the “step‑function” challenge.

The score of a sequence a (length n) is
    2·n·max_k Σ_{i+j=k} a_i a_j   (the sum of a is normalised to 1).

The algorithm builds many initial candidates (parametric families,
random Dirichlet draws, sparse vectors), keeps the best few,
and refines them with three stages:

1. **Exponentiated‑gradient soft‑max descent** – minimise a smooth
   approximation  τ·log Σ exp(conv/τ) using the exact gradient obtained
   from an FFT‑based cross‑correlation.  The update is multiplicative
   (mirror descent w.r.t. KL) and the vector is kept on the probability
   simplex.

2. **Projected sub‑gradient polishing** – classic sub‑gradient steps on
   the true max‑convolution objective with an adaptive stepsize.

3. **Simulated‑annealing hill‑climb** – stochastic pairwise mass swaps
   that accept worsening moves with a temperature schedule, giving a
   final escape from shallow local minima.

Time is split roughly 25 % / 45 % / 20 % / 10 % of the total budget.
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
        2·n·max(conv(a,a)).
    The vector is assumed to lie on the simplex.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a normalised vector)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 1.0 / n)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_geometric(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i  (0<r<1)"""
    a = r ** np.arange(n, dtype=float)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return a / total


def generate_sparse(n: int, k: int, concentration: float = 0.5) -> np.ndarray:
    """k random positions get Dirichlet‑distributed mass, others zero."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    raw = np.random.dirichlet([concentration] * len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = raw
    return a  # already sums to 1

# Baseline exploration (wide variety of families + random draws)
# ----------------------------------------------------------------------


def baseline_search(
    n: int, deadline: float, max_keep: int = 12, random_draws: int = 5000
):
    """Explore many parametric families and random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Power‑law sweep
    for p in np.linspace(0.05, 3.0, 121):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) Exponential sweep
    for lam in np.linspace(1e-4, 0.5, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta sweep (grid)
    al_vals = np.linspace(0.2, 6.0, 35)
    be_vals = np.linspace(0.2, 6.0, 35)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Gaussian sweep
    for sigma in np.linspace(0.05 * n, 0.7 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # 5) Sine‑power sweep
    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # 6) Geometric sweep
    for r in np.linspace(0.1, 0.99, 40):
        if time.time() >= deadline:
            break
        try_candidate(generate_geometric(n, r))

    # 7) Random sparse vectors (different sparsities)
    for k in [5, 10, 20, 40, 80, 160]:
        for _ in range(random_draws // 60):
            if time.time() >= deadline:
                break
            try_candidate(generate_sparse(n, k, concentration=0.2))

    # 8) Pure Dirichlet draws (various concentrations)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    per_conc = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_exp_optim(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-7,
    tau_decay: float = 0.95,
    eta0: float = 0.8,
):
    """Mirror‑descent on τ·log Σ exp(conv/τ) with KL‑projection.
    Returns the best vector seen before the deadline."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    tau = tau_start
    t = 0
    while time.time() < deadline:
        # anneal the temperature
        if t % 120 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # evaluate convolution
        conv = full_convolution(a)               # length 2n‑1
        shift = conv.max()
        # soft‑max weights
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()            # shape (2n‑1,)

        # gradient of the smooth‑max (2·a ⋆ w_rev)
        rev_w = w[::-1]
        size = 1 << ((2 * n - 1 - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: 2 * n - 1]

        idx = (2 * n - 2) - np.arange(n)         # positions of ∂/∂a_i in conv_full
        grad = 2.0 * conv_full[idx]

        # multiplicative (exponential) update – mirror descent on the simplex
        eta = eta0 / np.sqrt(t + 1.0)
        # Clip gradients to avoid overflow
        grad_clipped = np.clip(grad, -50.0, 50.0)
        mult = np.exp(-eta * grad_clipped)
        a = a * mult
        # Re‑normalise to stay on the simplex
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # record best
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        t += 1

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (as in the original code)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # build sub‑gradient from all active indices
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.006 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (final stochastic hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9997,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 8000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.012)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a few seconds safety margin
    deadline = start_time + budget_s - 5.0

    # Choose a size that balances flexibility and speed.
    # Larger n gives more degrees of freedom; 4096 works comfortably
    # within the 1000 s budget on typical hardware.
    n = 4096 if budget_s >= 800 else 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. (optional) simple fallback – uniform distribution
    # -----------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    val = compute_score(uniform)
    best_seq = uniform.tolist()
    best_val = val
    print(f"[fallback] uniform score = {val:.6f}")

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 20‑30 % of budget)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline, start_time + budget_s * 0.25)
    if time.time() < baseline_deadline:
        prints = []
        candidates = baseline_search(
            n, baseline_deadline, max_keep=12, random_draws=5000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
            prints.append(f"{val:.6f}")
        print(f"[baseline] found {len(candidates)} candidates, best = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Exponentiated‑gradient soft‑max optimisation (≈ 40 % budget)
    # -----------------------------------------------------------------
    if best_seq is not None:
        grad_deadline = min(deadline - 120.0, start_time + budget_s * 0.70)
        top_vectors = []

        # start from the current best and a few diversifiers
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        top_vectors.append(best_vec)

        top_vectors.append(uniform)                 # uniform
        for _ in range(3):
            top_vectors.append(project_to_simplex(np.random.rand(n)))

        refined_best = best_vec.copy()
        refined_val = best_val

        for idx, start_vec in enumerate(top_vectors):
            if time.time() >= grad_deadline:
                break
            local_deadline = grad_deadline - (time.time() - start_time) - 1.0
            if local_deadline <= time.time():
                break
            a_ref = smooth_exp_optim(start_vec, local_deadline)
            val = compute_score(a_ref)
            print(f"[exp‑opt {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_ref.copy()

        best_val, best_seq = refined_val, refined_best.tolist()
        print(f"[exp‑opt] best after exponentiated‑gradient = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 20 % budget)
    # -----------------------------------------------------------------
    polish_deadline = min(deadline - 60.0, start_time + budget_s * 0.90)
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_refined)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Final simulated annealing (last ≈10 % of budget)
    # -----------------------------------------------------------------
    anneal_deadline = deadline
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if somehow nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
