"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The routine proceeds in several stages within the 1000 s wall‑clock budget:

1.  Coarse scan of several analytic families (product, left‑right power,
    Gaussian, Kaiser, exponential‑U) on a modest grid.
2.  Random dense sampling of the same families (≈3000 trials).
3.  Low‑dimensional parametrisation of the log‑density by a truncated Fourier
    basis and Adam optimisation of the smooth‑max surrogate of the max‑convolution.
    The basis size is small (≈17 parameters) which allows many optimisation
    steps even for n≈8192.
4.  Projected sub‑gradient descent on the exact max‑objective (few thousand iters).
5.  A final simulated‑annealing hill‑climb on the simplex to polish the result.

All steps respect the time budget; the best sequence found is returned as a
plain ``list[float]``.  The implementation uses only NumPy (and SciPy for the
Kaiser window) and performs all convolutions via FFT for speed.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_right_power_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑α}·(n‑½‑i)^{‑β}, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-alpha * np.log(i + 0.5) - beta * np.log(n - 0.5 - i))
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def gaussian_family(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle. sigma_frac ∈ (0,0.5] gives σ = sigma_frac·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = max(sigma_frac, 1e-6) * n
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return normalize(w)


def kaiser_family(n: int, beta: float) -> np.ndarray:
    """Kaiser window (SciPy implementation)."""
    from scipy.special import i0

    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    # argument of the sqrt is 1 - ((i-alpha)/alpha)^2
    arg = 1.0 - ((i - alpha) / alpha) ** 2
    arg[arg < 0.0] = 0.0
    w = i0(beta * np.sqrt(arg)) / i0(beta)
    return normalize(w)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_param_optimize(B: np.ndarray,
                        start_c: np.ndarray,
                        deadline: float,
                        max_iters: int = 500_000,
                        beta_start: float = 5.0,
                        beta_end: float = 5000.0,
                        lr: float = 0.02) -> Tuple[np.ndarray, float, np.ndarray]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv) using a low‑dimensional
    Fourier basis B (n×d).  Returns the best vector a, its score and the best
    coefficient vector c.
    """
    n, d = B.shape
    c = start_c.copy()
    m = np.zeros_like(c)
    v = np.zeros_like(c)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    # ----- bookkeeping -----
    best_a = None
    best_score = float("inf")
    best_c = c.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule for smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # forward pass
        z = B @ c                      # (n,)
        a = np.exp(z)
        a_sum = a.sum()
        a = a / a_sum                  # normalise to lie on simplex

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max wrt a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max layer (a = exp(z)/sum)
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # gradient wrt coefficient vector c
        grad_c = B.T @ grad_z

        # Adam update on c
        m = beta1 * m + (1.0 - beta1) * grad_c
        v = beta2 * v + (1.0 - beta2) * (grad_c ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        c -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 3000 == 0:
            a_tmp = np.exp(B @ c)
            a_tmp /= a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                best_c = c.copy()
                print(f"[{time.time():.1f}] Adam param improved to {best_score:.6f}")

    # final evaluation
    a_fin = np.exp(B @ c)
    a_fin /= a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        best_c = c.copy()
        print(f"[{time.time():.1f}] Adam param final improvement to {best_score:.6f}")

    return best_a, best_score, best_c


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000,
                     step0: float = 0.04) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        # Euclidean projection onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 6_000_000,
                  temp_factor: float = 1e-3,
                  step_scale_init: float = 0.2) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # project to simplex
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            cand = np.full_like(cand, 1.0 / n)

        cand_score = compute_score(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
