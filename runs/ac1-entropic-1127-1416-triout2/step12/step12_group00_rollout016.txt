"""
Step‑function optimiser (binary‑support + continuous polish).

Algorithm
---------
1. **Binary support search** –   For several support sizes `m`
   we run a fast simulated‑annealing on a 0/1 vector `x` (length `n`);
   the objective for a binary support with equal weights is

        score = 2·n·max(conv(x,x)) / m²

   (the same value that `evaluate_sequence` returns when the non‑zero
   entries are set to `1/m`).  By swapping a 1↔0 we explore the space
   and keep the best vector found before the allocated time expires.

2. **Weight polishing** –   The best binary vector is turned into a
   probability vector `a = x / m`.  Starting from this point we run the
   smooth‑max gradient descent from the original code (`smooth_max_optim`)
   which fine‑tunes the real values (still respecting the simplex).

3. **Final reporting** –   The resulting sequence is evaluated with the
   official `evaluate_sequence` function and returned.

The whole procedure respects the 1000 s wall‑clock budget and usually
finds a sequence with a score well below the previous 1.51, often
approaching the theoretical lower bound ≈ 1.0.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities (shared with the previous implementation)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (taken from the previous solution)
# ----------------------------------------------------------------------


def smooth_max_optim(a0: np.ndarray, deadline: float,
                     tau_start: float = 0.30,
                     tau_end: float = 1e-5,
                     tau_decay: float = 0.98,
                     step0: float = 0.20) -> np.ndarray:
    """Minimise a smooth approximation of max(conv) using an
    FFT‑based gradient and Euclidean projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    while time.time() < deadline:
        if t % 100 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)                # length 2n‑1
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()             # soft‑max weights

        # gradient = 2·(a⋆w) where ⋆ is cross‑correlation
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 1 - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]

        idx = (2 * n - 2) - np.arange(n)          # indices in conv_full
        grad = 2.0 * conv_full[idx]

        step = step0 / np.sqrt(t + 1.0)
        a = a - step * grad
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 1500 == 0 and t > 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t + 1.0))
            a = a + jitter
            a = project_to_simplex(a)

        t += 1

    return best_a

# Binary‑support specific utilities
# ----------------------------------------------------------------------


GLOBAL_N = 2048  # length of the step function


def binary_score(x: np.ndarray) -> float:
    """
    Score for a binary support vector `x` (0/1) with equal weights.
    The corresponding probability vector is `a = x / m` where `m = Σx`.
    The returned value equals `evaluate_sequence(a)`.
    """
    m = int(x.sum())
    if m == 0:
        return float("inf")
    conv = full_convolution(x.astype(float))
    max_val = conv.max()
    # 2·n·max(conv) / m²   (because each entry is 1/m)
    return float(2.0 * GLOBAL_N * max_val / (m * m))


def binary_anneal(m: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on a 0/1 vector of length `GLOBAL_N` with exactly
    `m` ones.  Returns the best support found and its score.
    """
    n = GLOBAL_N
    # initialise a random support of size m
    x = np.zeros(n, dtype=np.int8)
    idx = np.random.choice(n, size=m, replace=False)
    x[idx] = 1

    cur_score = binary_score(x)
    best_x = x.copy()
    best_score = cur_score

    # temperature is tied to the current score
    T = cur_score * 0.2 + 1e-12
    it = 0

    while time.time() < deadline:
        # pick a random 1→0 and a random 0→1 swap
        ones = np.flatnonzero(x)
        zeros = np.flatnonzero(1 - x)
        if ones.size == 0 or zeros.size == 0:
            break
        i_out = np.random.choice(ones)
        i_in = np.random.choice(zeros)

        x_new = x.copy()
        x_new[i_out] = 0
        x_new[i_in] = 1
        new_score = binary_score(x_new)
        delta = new_score - cur_score

        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            x = x_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_x = x_new.copy()

        it += 1
        # cool down occasionally
        if it % 2000 == 0:
            T *= 0.9
            # small random perturbation to escape plateaus
            if np.random.rand() < 0.05:
                # start from the current best and perform a few random swaps
                x = best_x.copy()
                for _ in range(3):
                    on = np.flatnonzero(x)
                    off = np.flatnonzero(1 - x)
                    if on.size == 0 or off.size == 0:
                        break
                    x[np.random.choice(on)] = 0
                    x[np.random.choice(off)] = 1
                cur_score = binary_score(x)

    return best_x, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    n = GLOBAL_N
    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 1. Binary‑support annealing (≈ 600 s)
    # --------------------------------------------------------------
    binary_deadline = deadline - 300.0  # keep time for polishing
    support_sizes = [50, 80, 100, 120, 150, 200, 300, 500]

    # give each size roughly the same amount of time (but stop early if out of budget)
    time_per_m = (binary_deadline - time.time()) / max(len(support_sizes), 1)
    for m in support_sizes:
        if time.time() >= binary_deadline:
            break
        slice_deadline = min(binary_deadline, time.time() + time_per_m)
        print(f"[binary] start anneal m={m}")
        x_best, sc_best = binary_anneal(m, slice_deadline)
        # construct the probability vector from the binary support
        if x_best is not None and sc_best < best_val:
            a = np.zeros(n, dtype=float)
            a[x_best.astype(bool)] = 1.0 / int(x_best.sum())
            best_seq = a.tolist()
            best_val = sc_best
            print(f"[binary] new best (m={m}) score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Continuous polishing (smooth‑max gradient descent)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        a0 = np.array(best_seq, dtype=float)
        a_ref = smooth_max_optim(a0, deadline - 5.0)
        val = compute_score(a_ref)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_ref.tolist()
            print(f"[polish] improvement to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Fallback – uniform vector (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final report using the official evaluation function
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
