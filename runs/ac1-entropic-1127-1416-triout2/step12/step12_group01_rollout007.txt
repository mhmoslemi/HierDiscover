def convolution_fft(x, y):
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m-1).bit_length())
    X = np.fft.rfft(x, n_fft)
    Y = np.fft.rfft(y, n_fft)
    Z = X * Y
    return np.fft.irfft(Z, n_fft)[:m]

def smooth_eg_polish(a0, deadline, eta0=0.5, p_start=0.5,
                     p_end=15.0, p_growth=1.001):
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    it=0
    p = p_start
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)  # length 2n-1
        # compute weight: conv**p (with epsilon to avoid zeros)
        eps = 1e-15
        weight = np.maximum(conv, eps) ** p
        weight_sum = weight.sum()
        if weight_sum == 0:
            weight = np.full_like(conv, 1.0 / conv.size)
        else:
            weight = weight / weight_sum
        # gradient
        a_rev = a[::-1]
        grad_full = convolution_fft(weight, a_rev)
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]
        # step size
        eta = eta0 / np.sqrt(it)
        # EG update
        a = a * np.exp(-eta * grad)
        # re‑normalise to simplex
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        # record best
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # increase p slowly to focus on max
        if p < p_end:

        # occasional jitter to avoid stagnation
        if it % 1000 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()
    return best

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Global sweep** – try a handful of parametric families
   (power‑law, exponential, beta) and many random Dirichlet draws.
   Keep the best few candidates (projected onto the simplex).

2. **Smooth exponentiated‑gradient (EG) polishing** – minimise a
   smooth surrogate of the L∞‑norm of the auto‑convolution.
   The surrogate weight for lag *k* is  w_k ∝ (conv_k)^p,
   where the exponent *p* is increased gradually from a
   smoothing value (≈0.5) up to a focussed one (≈15).  The EG
   update  a_i ← a_i·exp(−η·g_i)  (followed by simplex
   normalisation) respects the non‑negativity and sum‑to‑1
   constraints.  The gradient g is the weighted sum of the
   contributions of each lag and is obtained efficiently by an
   FFT‑based convolution.

3. **Random restarts** – while time permits, restart the EG polish
   from fresh Dirichlet samples and keep any improvement.

4. **Final simulated‑annealing pass** – a short Metropolis hill‑climb
   to escape possible plateaus.

The whole routine respects the 1000 s wall‑clock budget and returns
the best sequence found (a plain list of floats).  All numbers are
non‑negative and the sequence is automatically scaled to satisfy the
security checks in ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary real vectors via FFT."""
    m = len(x) + len(y) - 1
    n_fft = _next_pow2(m)
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective (exact, cheap because we work on the simplex)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a probability vector
    (sum(a)=1, a_i≥0).  Faster than calling ``evaluate_sequence``.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the cheap global sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration – cheap baseline
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Sweep families + many Dirichlet draws, keep the top ``max_keep``."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law grid
    for p in np.linspace(0.1, 2.0, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential grid
    for lam in np.linspace(0.0001, 0.05, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta grid (coarse)
    al_vals = np.linspace(0.2, 5.0, 25)
    be_vals = np.linspace(0.2, 5.0, 25)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # massive Dirichlet draws
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1200
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth EG polishing – minimise a soft‑max surrogate of the peak
# ----------------------------------------------------------------------


def smooth_eg_polish(a0: np.ndarray, deadline: float,
                     eta0: float = 0.5,
                     p_start: float = 0.5,
                     p_end: float = 15.0,
                     p_growth: float = 1.001) -> np.ndarray:
    """
    Multiplicative (exponentiated‑gradient) descent on the
    surrogate   f_p(a) = (∑_k (conv_k)^p)^{1/p} .
    ``p`` is increased gradually from ``p_start`` to ``p_end``.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    p = p_start
    eps = 1e-15  # avoid exact zeros when raising to a power

    while time.time() < deadline:
        it += 1

        # linear convolution a*a
        conv = full_convolution(a)

        # smooth weights w_k ∝ (conv_k)^p
        w = np.maximum(conv, eps) ** p
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # gradient = 2 * sum_k w_k * a_{k-i}
        a_rev = a[::-1]
        grad_full = convolution_fft(w, a_rev)          # length = 3n‑2
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]

        # step‑size (standard EG schedule)
        eta = eta0 / np.sqrt(it)

        # EG multiplicative update (ensures a_i ≥ 0)
        a = a * np.exp(-eta * grad)

        # simplex projection (renormalisation)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasionally report progress
            if it % 500 == 0:
                print(f"[smooth‑EG] iter {it}, score {cur_val:.6f}")

        # increase the exponent p slowly – the surrogate becomes
        # closer to the true max as p grows.
        if p < p_end:
            p = min(p_end, p * p_growth)

        # small random jitter every few hundred iterations to escape flats
        if it % 1000 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a

# Light simulated‑annealing polish (as in the original code)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback to a uniform distribution if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
