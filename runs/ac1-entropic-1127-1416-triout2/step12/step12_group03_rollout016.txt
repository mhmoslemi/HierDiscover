def upsample(a_small, n_large):
    # use numpy interpolation
    x_small = np.linspace(0,1,len(a_small))
    x_large = np.linspace(0,1,n_large)
    return np.interp(x_large, x_small, a_small)

# -*- coding: utf-8 -*-
"""
Propose candidate for step function optimization.

The algorithm proceeds in several stages:
1. Load a fallback sequence if available.
2. Choose a working length n (defaults to 2048) and generate a pool of
   diverse initial candidates (analytic families, random Dirichlet,
   monotone shapes).
3. Run a lightweight evolutionary optimisation (crossover + mutation)
   on the simplex for the remaining time slice, keeping the best
   candidate.
4. Refine the best candidate with three complementary gradient‑based
   methods: exponentiated‑gradient mirror descent on the true max‑convolution,
   Adam on a smooth‑max surrogate, and a long projected sub‑gradient
   descent.
5. Optionally symmetrise and finish with a short simulated‑annealing
   hill‑climb.
6. Return the sequence as a plain Python list; the harness will call
   evaluate_sequence on it.
"""

import time
import math
import numpy as np

# Basic utilities
def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a shape that sums to 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def softmax(z: np.ndarray) -> np.ndarray:
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# Analytic families ---------------------------------------------------------

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑½‑i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a

def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a

def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a

def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]          # decreasing
    a /= a.sum()
    return a

def random_dirichlet_start(n: int) -> np.ndarray:
    a = np.random.rand(n)
    a /= a.sum()
    return a

# Optimisers ---------------------------------------------------------------

def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 600_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror‑descent) on the true max‑convolution."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate of the max‑convolution."""
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing hill‑climb on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def upsample_to_length(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation of a_small to length n_large, preserving total mass."""
    x_small = np.linspace(0.0, 1.0, a_small.size)
    x_large = np.linspace(0.0, 1.0, n_large)
    interpolated = np.interp(x_large, x_small, a_small)
    interpolated = np.maximum(interpolated, 0.0)
    if interpolated.sum() == 0.0:
        interpolated = np.full_like(interpolated, 1.0 / n_large)
    else:
        interpolated = interpolated / interpolated.sum()
    return interpolated

"""
Propose candidate for the step‑function minimisation problem.

The algorithm consists of:

1. Load a fallback candidate if one is supplied by the harness.
2. Work with a fixed moderately large length (default 2048) that allows
   many cheap iterations.
3. Build a small, diverse population of analytic and random shapes and
   evaluate them.
4. Run a lightweight evolutionary optimisation (crossover + mutation)
   on the simplex while keeping the population symmetric.
5. Refine the best candidate with three complementary gradient‑based
   methods:
   – Exponentiated‑gradient mirror descent on the true max‑convolution.
   – Adam optimiser on a smooth‑max surrogate (log‑sum‑exp of the
     convolution entries).
   – Long projected sub‑gradient descent on the exact max‑convolution.
6. Symmetrise the current best shape, then perform a short simulated‑
   annealing hill‑climb for any remaining improvements.
7. Return the final sequence as a plain list; the external harness will
   call `evaluate_sequence` on it.

All phases respect the global wall‑clock budget (default 1000 s) and
use only NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two that is ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the evaluator (up to the global scaling
    that cancels out).  The vector `a` should be non‑negative and
    normalised to sum = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the initial population)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    The classic arcsine distribution corresponds to α = 0.5.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Power‑law decay from the left edge."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay away from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex combination of uniform and exponential shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """A decreasing random shape (generated by sorting powered uniforms)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]          # decreasing
    a /= a.sum()
    return a


def random_dirichlet_start(n: int) -> np.ndarray:
    """Simple random point on the simplex (Dirichlet(1,…,1))."""
    a = np.random.rand(n)
    a /= a.sum()
    return a

def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 600_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the true max‑convolution
    objective.  The step size follows η_t = η0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry (k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]  # diagonal term appears only once

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth‑max surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Very cheap per iteration, allowing many steps.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Very simple simulated‑annealing style random walk on the simplex.
    Occasionally accepts uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def upsample_to_length(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linear interpolation from a_small to length n_large, followed by
    re‑normalisation to sum = 1.
    """
    x_small = np.linspace(0.0, 1.0, a_small.size)
    x_large = np.linspace(0.0, 1.0, n_large)
    interpolated = np.interp(x_large, x_small, a_small)
    interpolated = np.maximum(interpolated, 0.0)
    if interpolated.sum() == 0.0:
        interpolated = np.full_like(interpolated, 1.0 / n_large)
    else:
        interpolated = interpolated / interpolated.sum()
    return interpolated


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    `evaluate_sequence` objective.  The method combines a tiny evolutionary
    population, several gradient‑based refinements and a final simulated‑
    annealing hill‑climb, while respecting the global wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Reserve a small safety margin for the harness.
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Load a fallback candidate if supplied.
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = np.array(fb_vec, dtype=np.float64)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a working size.
    # ------------------------------------------------------------------
    n = 2048
    if best_seq is not None and best_seq.size > n:
        n = best_seq.size
    print(f"[{time.time():.1f}] working length n = {n}")

    # ------------------------------------------------------------------
    # 2. Initialise a small, diverse population.
    # ------------------------------------------------------------------
    population = []
    scores = []

    # Analytic families (all symmetric by construction)
    population.append(symmetric_product_family(n, 0.5))   # classic arcsine
    population.append(symmetric_product_family(n, 0.8))
    population.append(symmetric_product_family(n, 0.3))
    population.append(left_heavy_power_family(n, 0.8))
    population.append(left_heavy_power_family(n, 1.2))
    population.append(exp_u_family(n, 0.5))
    population.append(exp_u_family(n, 2.0))
    population.append(mixed_uniform_u_family(n, 0.6, 0.8))
    population.append(random_monotone_start(n))
    population.append(random_dirichlet_start(n))

    if best_seq is not None:
        seq = best_seq.copy()
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0.0:
            seq = np.full_like(seq, 1.0 / seq.shape[0])
        else:
            seq = seq / seq.sum()
        population.append(seq)

    # Evaluate initial scores and keep the best
    for a in population:
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.shape[0])
        sc = compute_score(a)
        scores.append(sc)

    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]
    print(f"[{time.time():.1f}] initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Evolutionary optimisation (crossover + mutation) on the simplex.
    # ------------------------------------------------------------------
    pop_size = len(population)
    evo_deadline = deadline - 30.0   # keep a buffer for the fine tuning steps
    iters = 0
    while time.time() < evo_deadline:
        i, j = np.random.randint(pop_size, size=2)
        a = population[i]
        b = population[j]
        alpha = np.random.rand()
        child = alpha * a + (1.0 - alpha) * b

        # Occasional small Gaussian mutation
        if np.random.rand() < 0.25:
            noise = np.random.normal(scale=0.02, size=child.shape)
            child = child + noise
            child = np.maximum(child, 0.0)

        # Enforce symmetry (the optimal solution is expected to be symmetric)
        child = (child + child[::-1]) * 0.5
        child = project_to_simplex(child)

        child_score = compute_score(child)

        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score

        if child_score < best_score:
            best_score = child_score
            best_seq = child.copy()
            print(f"[{time.time():.1f}] EA improved to {best_score:.6f}")

        iters += 1
        if iters % 2000 == 0:
            print(f"[{time.time():.1f}] EA iteration {iters}, current best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Split the remaining budget among gradient‑based refinements.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    eg_time   = 0.25 * remaining
    adam_time = 0.25 * remaining
    subg_time = 0.30 * remaining
    sa_time   = remaining - (eg_time + adam_time + subg_time)

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient phase.
    # ------------------------------------------------------------------
    if time.time() + eg_time < deadline:
        print(f"[{time.time():.1f}] EG phase (~{eg_time:.1f}s)")
        a_eg, sc_eg = eg_optimize(best_seq,
                                  time.time() + eg_time,
                                  max_iters=500_000,
                                  eta0=0.8)
        if sc_eg < best_score:
            best_score = sc_eg
            best_seq = a_eg
            print(f"[{time.time():.1f}] EG improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Adam on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    if time.time() + adam_time < deadline:
        print(f"[{time.time():.1f}] Adam phase (~{adam_time:.1f}s)")
        a_adam, sc_adam = adam_smooth_optimize(best_seq,
                                                time.time() + adam_time,
                                                max_iters=300_000,
                                                beta_start=2.0,
                                                beta_end=8000.0,
                                                lr=0.022)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Long projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() + subg_time < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient phase (~{subg_time:.1f}s)")
        a_sub, sc_sub = subgrad_optimize(best_seq,
                                          time.time() + subg_time,
                                          max_iters=2_000_000,
                                          step0=0.04)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub
            print(f"[{time.time():.1f}] Sub‑gradient improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Symmetrisation (the problem is invariant under reversal).
    # ------------------------------------------------------------------
    a_sym = (best_seq + best_seq[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        best_seq = a_sym
        print(f"[{time.time():.1f}] Symmetrisation improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Final simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    if time.time() + sa_time < deadline:
        print(f"[{time.time():.1f}] Hill‑climb (SA) phase (~{sa_time:.1f}s)")
        a_sa, sc_sa = hill_climb_sa(best_seq,
                                    time.time() + sa_time,
                                    max_steps=1_500_000)
        if sc_sa < best_score:
            best_score = sc_sa
            best_seq = a_sa
            print(f"[{time.time():.1f}] Hill‑climb improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Final evaluation via the provided harness.
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
