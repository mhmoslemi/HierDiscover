def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length = 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    """
    Coarse piecewise‑constant shape: generate K positive values, sort them
    decreasingly to enforce monotonicity, repeat each value to fill the
    n entries (last block may be slightly larger).  Normalise to sum = 1.
    """

    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)
    a = a[:n]                                          # safety trim
    a /= a.sum()
    return a
