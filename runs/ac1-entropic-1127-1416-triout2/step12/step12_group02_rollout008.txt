"""
Robust multi‑stage optimiser for the auto‑convolution peak problem.

The routine follows these phases (all respecting a 1000 s wall clock):

1. **Coarse scan of simple analytic families** (product, power‑law, U‑shape,
   mixture, asymmetric product, product + U, cosine‑U).  The best parameter
   of each family is recorded.

2. **Local refinement** of the most promising families using SciPy’s
   ``Nelder‑Mead`` (2‑D) or golden‑section (1‑D) optimisation.

3. **Multiple Adam passes** on the smooth‑max surrogate of the max‑convolution.
   Several random restarts are performed; the best run is kept.

4. **Exact projected sub‑gradient descent** on the true max‑objective to
   polish the solution.

5. **Linear interpolation** of the current best vector with the uniform
   distribution (and a few simple smoothings) – a cheap way to flatten
   the convolution peak.

6. **Final simulated‑annealing hill‑climb** for an extra polishing step.

All vectors are kept on the probability simplex (non‑negative, sum = 1)
using a Euclidean projection.  The algorithm is deliberately
parameter‑light; most hyper‑parameters are chosen to give a good
trade‑off between exploration and speed.

The function ``propose_candidate`` returns the best sequence found as a
plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic full convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) for a probability vector (∑a = 1).
    The function is cheap because it uses the FFT based convolution.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families of non‑negative step functions
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric U‑shape."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def asymmetric_product_family(n: int, beta_left: float, beta_right: float) -> np.ndarray:
    """Two‑parameter product family: a_i ∝ (i+½)^{‑β₁}(n‑½‑i)^{‑β₂}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta_left * np.log(i + 0.5) - beta_right * np.log(n - 0.5 - i))
    w /= w.sum()
    return w


def product_expU_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    """Product family multiplied by an exponential‑U factor."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod - gamma * dist)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def cosine_U_family(n: int, kappa: float) -> np.ndarray:
    """
    Cosine‑U shape:
        a_i ∝ (1 + cos(pi * i/(n‑1)))^kappa,
    which peaks at both ends and is low in the centre.
    """
    i = np.arange(n, dtype=np.float64)
    w = (1.0 + np.cos(np.pi * i / (n - 1))) ** kappa
    w_sum = w.sum()
    if w_sum == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / w_sum

def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse start: ~5 % non‑zero entries, uniform mass on them."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sort a powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (soft‑max surrogate of the max‑conv)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 10.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.015,
                         sym_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    start_a : np.ndarray (probability vector)
        Initial point.
    deadline : float
        Unix timestamp after which optimisation must stop.
    max_iters : int
        Upper bound on the number of Adam iterations.
    beta_start / beta_end : float
        Annealing schedule for the smooth‑max temperature.
    lr : float
        Base learning rate.
    sym_interval : int
        Periodic symmetrisation (a ← (a+a[::-1])/2) to keep the distribution
        essentially symmetric; set to 0 to disable.
    Returns
    -------
    best_a, best_score
        Best vector found and its exact (non‑smooth) score.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear annealing of the temperature
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth‑max surrogate
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # chain rule through the soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update on the pre‑softmax logits
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 4_000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent for the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of max(conv) w.r.t a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3_000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a tiny temperature schedule.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a small chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Golden‑section search (1‑D refinement)
# ----------------------------------------------------------------------


def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 15) -> tuple[float, float]:
    """Minimises `func` on [a,b] using golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

# Main driver required by the platform

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns a plain list.
    """
    # optional SciPy import for refinement stages
    try:
        from scipy.optimize import minimize
        scipy_available = True
    except Exception:
        scipy_available = False

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback: use the previous best if available (global variable)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose working length (power of two for fast FFT)
    # ------------------------------------------------------------------
    n = 8192

    # ------------------------------------------------------------------
    # 2. Coarse scan of analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse scan (n={n})")
    family_records = {}

    # uniform baseline
    uniform_vec = np.full(n, 1.0 / n, dtype=np.float64)
    sc = compute_score(uniform_vec)
    if sc < best_score:
        best_score = sc
        best_seq = uniform_vec.tolist()
        print(f"[{time.time():.1f}] new best (uniform) = {best_score:.6f}")

    # -------- product (symmetric) --------
    best_alpha, best_alpha_sc = None, float('inf')
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.3f}) = {best_score:.6f}")
        if sc < best_alpha_sc:
            best_alpha_sc = sc
            best_alpha = alpha
    family_records['product'] = (best_alpha, best_alpha_sc)

    # -------- left‑heavy power law --------
    best_beta, best_beta_sc = None, float('inf')
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")
        if sc < best_beta_sc:
            best_beta_sc = sc
            best_beta = beta
    family_records['leftpower'] = (best_beta, best_beta_sc)

    # -------- exponential U shape --------
    best_gamma, best_gamma_sc = None, float('inf')
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")
        if sc < best_gamma_sc:
            best_gamma_sc = sc
            best_gamma = gamma
    family_records['expU'] = (best_gamma, best_gamma_sc)

    # -------- mixed uniform + U shape --------
    best_mix, best_mix_sc = None, float('inf')
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")
            if sc < best_mix_sc:
                best_mix_sc = sc
                best_mix = (lam, gamma)
    family_records['mix'] = (best_mix, best_mix_sc)

    # -------- asymmetric product (2‑parameter) --------
    best_asym, best_asym_sc = None, float('inf')
    beta_grid = np.linspace(0.0, 12.0, 13)
    for b1 in beta_grid:
        for b2 in beta_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = asymmetric_product_family(n, b1, b2)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (asym β1={b1:.2f},β2={b2:.2f}) = {best_score:.6f}")
            if sc < best_asym_sc:
                best_asym_sc = sc
                best_asym = (b1, b2)
    family_records['asym'] = (best_asym, best_asym_sc)

    # -------- product + exponential U (2‑parameter) --------
    best_pe, best_pe_sc = None, float('inf')
    a_grid = np.linspace(0.0, 12.0, 13)
    g_grid = np.linspace(0.0, 6.0, 7)
    for a in a_grid:
        for g in g_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = product_expU_family(n, a, g)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (prod‑exp α={a:.2f},γ={g:.2f}) = {best_score:.6f}")
            if sc < best_pe_sc:
                best_pe_sc = sc
                best_pe = (a, g)
    family_records['prodexpU'] = (best_pe, best_pe_sc)

    # -------- cosine U shape --------
    best_kappa, best_kappa_sc = None, float('inf')
    for kappa in np.linspace(0.0, 12.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = cosine_U_family(n, kappa)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (cos‑κ={kappa:.3f}) = {best_score:.6f}")
        if sc < best_kappa_sc:
            best_kappa_sc = sc
            best_kappa = kappa
    family_records['cosineU'] = (best_kappa, best_kappa_sc)

    # ------------------------------------------------------------------
    # 3. Local refinement of promising families (SciPy if available)
    # ------------------------------------------------------------------
    if scipy_available and time.time() < deadline - 20.0:
        # ---- symmetric product (1‑D) refinement ----
        if family_records.get('product')[0] is not None:
            α0 = family_records['product'][0]
            lo = max(0.0, α0 - 0.6)
            hi = min(12.0, α0 + 0.6)

            def score_alpha(a):
                return compute_score(symmetric_product_family(n, a))

            α_opt, α_sc = golden_section_optimize(score_alpha, lo, hi, max_iter=25)
            vec = symmetric_product_family(n, α_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined product‑α={α_opt:.6f} → {best_score:.6f}")

        # ---- mixture (lam,γ) refinement ----
        if family_records.get('mix')[0] is not None:
            lam0, gam0 = family_records['mix'][0]

            def score_mix(p):
                lam = np.clip(p[0], 0.0, 1.0)
                gam = np.clip(p[1], 0.0, 6.0)
                return compute_score(mixed_uniform_u_family(n, lam, gam))

            res = minimize(score_mix,
                           x0=np.array([lam0, gam0]),
                           method='Nelder-Mead',
                           options={'maxiter': 150, 'xatol': 1e-4, 'fatol': 1e-4})
            lam_opt, gam_opt = np.clip(res.x, [0.0, 0.0], [1.0, 6.0])
            vec = mixed_uniform_u_family(n, lam_opt, gam_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined mix (λ={lam_opt:.6f},γ={gam_opt:.6f}) → {best_score:.6f}")

        # ---- product+expU refinement ----
        if family_records.get('prodexpU')[0] is not None:
            a0, g0 = family_records['prodexpU'][0]

            def score_pe(p):
                a = np.clip(p[0], 0.0, 12.0)
                g = np.clip(p[1], 0.0, 6.0)
                return compute_score(product_expU_family(n, a, g))

            res = minimize(score_pe,
                           x0=np.array([a0, g0]),
                           method='Nelder-Mead',
                           options={'maxiter': 200, 'xatol': 1e-4, 'fatol': 1e-4})
            a_opt, g_opt = np.clip(res.x, [0.0, 0.0], [12.0, 6.0])
            vec = product_expU_family(n, a_opt, g_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined prod‑expU (α={a_opt:.6f},γ={g_opt:.6f}) → {best_score:.6f}")

        # ---- asymmetric product refinement (2‑D) ----
        if family_records.get('asym')[0] is not None:
            b1_0, b2_0 = family_records['asym'][0]

            def score_asym(p):
                b1 = np.clip(p[0], 0.0, 12.0)
                b2 = np.clip(p[1], 0.0, 12.0)
                return compute_score(asymmetric_product_family(n, b1, b2))

            res = minimize(score_asym,
                           x0=np.array([b1_0, b2_0]),
                           method='Nelder-Mead',
                           options={'maxiter': 180, 'xatol': 1e-4, 'fatol': 1e-4})
            b1_opt, b2_opt = np.clip(res.x, 0.0, 12.0)
            vec = asymmetric_product_family(n, b1_opt, b2_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined asym (β1={b1_opt:.6f},β2={b2_opt:.6f}) → {best_score:.6f}")

        # ---- cosine‑U refinement (1‑D) ----
        if family_records.get('cosineU')[0] is not None:
            k0 = family_records['cosineU'][0]

            def score_cos(k):
                return compute_score(cosine_U_family(n, k))

            k_opt, k_sc = golden_section_optimize(score_cos,
                                                  max(0.0, k0 - 0.6),
                                                  min(12.0, k0 + 0.6),
                                                  max_iter=25)
            vec = cosine_U_family(n, k_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined cosine‑κ={k_opt:.6f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. First Adam (smooth‑max) optimisation – multiple restarts
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] multi‑restart Adam optimisation")
        # keep a copy of the current best as the first start
        starts = [np.array(best_seq, dtype=np.float64)]

        # add a few random restarts drawn from diverse families
        while len(starts) < 4 and time.time() < deadline - 250.0:
            choice = np.random.choice(['product', 'asym', 'cos', 'rand'])
            if choice == 'product':
                a0 = symmetric_product_family(n, np.random.uniform(0.0, 12.0))
            elif choice == 'asym':
                a0 = asymmetric_product_family(n,
                                                np.random.uniform(0.0, 12.0),
                                                np.random.uniform(0.0, 12.0))
            elif choice == 'cos':
                a0 = cosine_U_family(n, np.random.uniform(0.0, 12.0))
            else:
                a0 = random_monotone_start(n)
            starts.append(a0)

        best_adam_a = None
        best_adam_score = float('inf')
        for idx, a_start in enumerate(starts):
            if time.time() >= deadline - 150.0:
                break
            print(f"[{time.time():.1f}] Adam run {idx + 1}/{len(starts)}")
            a_opt, sc_opt = adam_smooth_optimize(
                a_start,
                deadline - 120.0,
                max_iters=350_000,
                beta_start=10.0,
                beta_end=8000.0,
                lr=0.015,
                sym_interval=2000,
            )
            if sc_opt < best_adam_score:
                best_adam_score = sc_opt
                best_adam_a = a_opt.copy()
                print(f"[{time.time():.1f}] run {idx + 1} best = {best_adam_score:.6f}")

        if best_adam_a is not None and best_adam_score < best_score:
            best_score = best_adam_score
            best_seq = best_adam_a.tolist()
            print(f"[{time.time():.1f}] after Adam passes best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 130.0:
        print(f"[{time.time():.1f}] sub‑gradient descent (fine tuning)")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=2_500_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Linear interpolation with uniform (cheap flattening)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] linear mixture with uniform")
        a_best = np.array(best_seq, dtype=np.float64)
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        for lam in np.linspace(0.0, 1.0, 21):
            cand = lam * a_best + (1 - lam) * uniform
            cand = project_to_simplex(cand)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.tolist()
                print(f"[{time.time():.1f}] mixture λ={lam:.2f} improves to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Small Gaussian smoothing (optional)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15.0:
        print(f"[{time.time():.1f}] Gaussian smoothing trials")
        a_best = np.array(best_seq, dtype=np.float64)
        # try a few kernel widths
        for sigma in (0.8, 1.2, 1.6):
            if time.time() >= deadline - 12.0:
                break
            # build symmetric Gaussian kernel
            x = np.arange(-n + 1, n)
            kernel = np.exp(-0.5 * (x / sigma) ** 2)
            kernel /= kernel.sum()
            smoothed = np.convolve(a_best, kernel, mode='same')
            smoothed = project_to_simplex(smoothed)
            sc = compute_score(smoothed)
            if sc < best_score:
                best_score = sc
                best_seq = smoothed.tolist()
                print(f"[{time.time():.1f}] sigma={sigma:.2f} smoothing improves to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=3_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Safety fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes safety checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
