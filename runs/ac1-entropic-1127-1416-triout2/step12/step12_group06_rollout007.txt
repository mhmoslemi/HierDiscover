"""
Propose a candidate step‑function by direct smooth‑max gradient descent.
The algorithm:
    * choose a moderate length (default 4096)
    * initialise many random feasible vectors (Dirichlet, power‑law, beta‑shape,
      interpolation of the provided fallback)
    * for each start run a projected gradient descent on a differentiable surrogate
      of the max‑convolution (soft‑max with a temperature ε that anneals to 0).
      The gradient is computed efficiently via FFT‑based convolution.
    * keep the best vector (lowest true evaluate_sequence score) and finally
      return it.
The code respects the 1000 s budget and uses only numpy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # if all entries non‑positive, the projection is the uniform vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Return the linear (full) convolution of two real vectors using a real FFT.
    Length of the result is len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())   # next power of two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:n]
    return conv


def full_convolution_self(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a convolution."""
    return full_convolution_fft(a, a)


def compute_score_normed(a: np.ndarray) -> float:
    """
    Compute the evaluation score for a vector that is already normalised
    to have sum = 1.  The exact value is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric initialisers
# ----------------------------------------------------------------------
def power_law_vec(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a

# Gradient descent on a smooth approximation of the max‑convolution.
# ----------------------------------------------------------------------
def smooth_max_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               eta0: float = 0.25,
                               eps0: float = 0.10,
                               eta_decay: float = 0.9995,
                               eps_decay: float = 0.9997,
                               eps_min: float = 1e-5) -> np.ndarray:
    """
    Projected gradient descent on the soft‑max surrogate:
        f_ε(a) = ε·log Σ_k exp( conv(a,a)_k / ε )
    The gradient can be expressed via a convolution with the softmax weights.
    The method keeps the iterates on the simplex.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution_self(a).max()
    eta = eta0
    eps = eps0
    it = 0

    while time.time() < deadline:
        # 1) compute convolution and the soft‑max weights
        conv = full_convolution_self(a)                     # length 2n‑1
        M = conv.max()

        # soft‑max (numerically stable)
        # subtract M to avoid overflow; divide by eps
        exp_terms = np.exp((conv - M) / eps)
        w = exp_terms / exp_terms.sum()                     # length 2n‑1, sums to 1

        # 2) gradient = 2 * Σ_j a_j w_{i+j}
        # Achieved via convolution of a with reversed w
        w_rev = w[::-1]
        grad_full = full_convolution_fft(a, w_rev)        # length 3n‑2
        # index = 2n‑2‑i extracts Σ_j a_j w_{i+j}
        idx = 2 * n - 2
        gradient = 2.0 * grad_full[idx - np.arange(n)]

        # 3) projected gradient step
        a = a - eta * gradient
        a = project_to_simplex(a)

        # 4) keep the best solution seen so far
        conv_new = full_convolution_self(a)
        M_new = conv_new.max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
            # a successful step -> allow a little larger step next time
            eta = min(eta * 1.03, 1.0)
        else:
            # no improvement -> shrink step
            eta *= 0.995

        # 5) annealing schedules
        eps = max(eps_min, eps * eps_decay)
        eta = max(1e-6, eta * eta_decay)

        it += 1
        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (eta * 0.02)
            a = a + jitter
            a = project_to_simplex(a)

    return best_a

Propose a candidate step‑function by smooth‑max gradient descent.

The algorithm works as follows:

1.  Choose a moderate length n (default 4096).  A longer sequence gives a
    lower theoretical bound (≈ 1.0) while still being cheap enough for FFT‑based
    operations.

2.  Build a pool of reasonable initial vectors (uniform, power‑law,
    geometric, several beta‑shapes, a few random Dirichlet draws and an
    interpolated fallback if provided).

3.  For many random restarts run a projected gradient descent on a
    differentiable surrogate of the objective:

        f_ε(a) = ε·log ∑_k exp(conv(a,a)_k / ε)      (soft‑max of the convolution)

    The gradient can be expressed through a convolution with the soft‑max
    weights and is computed efficiently with FFTs.  After each step the
    iterate is projected onto the probability simplex to keep the coefficients
    non‑negative and summing to one.

4.  Keep the best vector (lowest true score = 2·n·max(conv(a,a))).

5.  Perform a brief polishing phase using the classic sub‑gradient method
    from the reference implementation to squeeze a little more performance.

6.  Return the best sequence as a plain Python list.  The returned vector
    is already normalised, so ``evaluate_sequence`` can be called directly.

The whole search respects the 1000 s budget; the heavy lifting is done with
NumPy and FFTs, which are fast enough to allow thousands of iterations.

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of two real vectors using a real FFT.
    Length of the result is len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:n]
    return conv


def full_convolution_self(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a convolution."""
    return full_convolution_fft(a, a)


def compute_score_normed(a: np.ndarray) -> float:
    """
    Compute the objective for a vector already normalised to sum = 1.
    The exact value returned by `evaluate_sequence` is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution_self(a)
    return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # 0. Optional fallback to the provided construction (if any)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback = np.array(fallback, dtype=float)
        fallback = project_to_simplex(fallback)
        fallback_score = evaluate_sequence(fallback.tolist())
        print(f"[fallback] score = {fallback_score:.6f}")
        if fallback_score < best_score:
            best_score = fallback_score
            best_seq = fallback.tolist()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a working length.
    #    4096 gives a theoretical lower bound of ≈ 1.0 and is still fast.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 2. Build a pool of initial vectors.
    # ------------------------------------------------------------------
    init_pool = []

    # uniform baseline
    init_pool.append(np.full(n, 1.0 / n, dtype=float))

    # several power‑law exponents
    for p in (0.3, 0.5, 0.7, 0.9, 1.1):
        init_pool.append(power_law_vec(n, p))

    # geometric ratios close to 1
    for q in (0.97, 0.985, 0.995):
        init_pool.append(geometric_vec(n, q))

    # a handful of beta‑shapes covering symmetric and skewed cases
    for alpha, beta in ((0.2, 0.2), (0.5, 0.5), (1.0, 1.0),
                        (2.0, 0.5), (0.5, 2.0)):
        init_pool.append(beta_vec(n, alpha, beta))

    # few random Dirichlet draws
    for _ in range(8):
        init_pool.append(random_simplex_vec(n))

    # interpolate the fallback solution to the current length, if available
    try:
        old = np.array([float(x) for x in height_sequence_1], dtype=float)
        L_old = old.shape[0]
        x_old = np.linspace(0, 1, L_old, endpoint=False)
        x_new = np.linspace(0, 1, n, endpoint=False)
        interp = np.interp(x_new, x_old, old)
        interp = project_to_simplex(interp)
        init_pool.append(interp)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – many random restarts with smooth‑max GD.
    # ------------------------------------------------------------------
    polishing_reserve = 30.0  # seconds reserved for the final polishing phase
    while time.time() < deadline - polishing_reserve:
        remaining = deadline - polishing_reserve - time.time()
        # each restart gets at most 40 s (or a fraction of what is left)
        run_time = min(40.0, remaining * 0.6)
        run_deadline = time.time() + run_time

        # select a random initial vector and add a tiny jitter
        a0 = init_pool[np.random.randint(len(init_pool))].copy()
        a0 = project_to_simplex(a0 + np.random.randn(n) * 1e-5)

        a_opt = smooth_max_gradient_descent(
            a0,
            deadline=run_deadline,
            eta0=0.25,
            eps0=0.12,
            eta_decay=0.9995,
            eps_decay=0.9998,
            eps_min=1e-6,
        )

        cur_score = compute_score_normed(a_opt)
        print(f"[run] candidate normed score = {cur_score:.6f}")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.tolist()
            print(f"   ** new best (normed) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final polishing using the classic sub‑gradient method
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_best = np.array(best_seq, dtype=float)
        polish_deadline = deadline
        a_polished = subgradient_polish_local(a_best, polish_deadline)
        polished_score = compute_score_normed(a_polished)
        print(f"[polish] polished normed score = {polished_score:.6f}")
        if polished_score < best_score:
            best_score = polished_score
            best_seq = a_polished.tolist()

    # ------------------------------------------------------------------
    # 5. Safeguard – fall back to the uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6. Final official evaluation and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
