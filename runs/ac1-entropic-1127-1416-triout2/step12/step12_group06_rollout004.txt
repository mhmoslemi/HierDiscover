"""
Search for a non‑negative step‑function that minimises the
``evaluate_sequence`` objective.

Algorithm
---------

1. **Broad parametric sweep** – explore several analytic families
   (power‑law, geometric, beta‑shape, mixed plateau+power) and keep the
   best vector (normalised to sum = 1).  This provides a good start.

2. **Smooth‑max Adam optimisation** – minimise a
   ``soft‑max`` approximation of ``max(conv(a,a))``.
   The soft‑max is ``(1/α)·log Σ exp(α·conv)``.
   The gradient of the soft‑max can be written as a convolution, so each
   iteration requires only two FFTs (O(n log n)).
   The optimisation runs with an Adam‑style adaptive step and a
   continuation schedule that slowly increases the smoothing parameter
   ``α`` from 1 to 30, turning the objective into the exact maximum at
   the end.  After each update the vector is projected onto the probability
   simplex.

3. **Fine stochastic hill‑climb** – a cheap greedy local search that
   moves a small fraction of mass between two random positions, keeping
   the best vector found.

The whole pipeline respects the 1000 s budget, prints intermediate
scores and finally returns the best sequence (as a plain Python list)
that was found.

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper returning the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear full convolution of two arbitrary vectors via FFT."""
    m = x.shape[0] + y.shape[0] - 1
    n_fft = _next_pow2(m)
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv

# ----------------------------------------------------------------------
# Scoring helpers (normalised to sum = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
       2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a

def geometric(n: int, q: float) -> np.ndarray:
    if q <= 0.0 or q >= 1.0:
        return None
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a

def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a

def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a

# Parametric sweep (broad but still cheap)
# ----------------------------------------------------------------------
def parametric_sweep(n: int, time_deadline: float) -> np.ndarray:
    best_vec = None
    best_val = float('inf')
    # -------- power‑law ----------
    for p in np.linspace(0.10, 2.00, 191):
        if time.time() >= time_deadline:
            return best_vec
        a = power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # -------- geometric ----------
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 91),
            np.linspace(0.990, 0.9999, 110))):
        if time.time() >= time_deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # -------- mixed plateau + power ----------
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= time_deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # -------- beta shape ----------
    alphas = np.linspace(-0.8, 3.0, 20)
    betas  = np.linspace(-0.8, 3.0, 20)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= time_deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)

def adam_optimize(a0: np.ndarray, time_deadline: float,
                  max_iter: int = 200_000,
                  step_size: float = 0.02,
                  alpha_max: float = 30.0) -> np.ndarray:
    """
    Optimise the objective 2·n·max(conv(a,a)) using a soft‑max
    approximation of the max and Adam updates.  The sequence ``a`` is kept
    on the probability simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_obj = float('inf')
    alpha = 1.0

        t += 1

        # ----- current convolution -----
        conv = full_convolution(a)             # length 2n-1

        # ----- smooth max (soft‑max) -----
        maxc = conv.max()
        # shift for numerical stability
        exp_shift = np.exp(alpha * (conv - maxc))
        w = exp_shift / exp_shift.sum()        # softmax weights, sum to 1

        # ----- gradient of soft‑max -----
        a_rev = a[::-1]
        conv_w_rev = full_convolution_fft(w, a_rev)          # length 3n-2
        g = 2.0 * conv_w_rev[n-1:n-1+n]                      # shape (n,)

        # ----- objective (soft max) -----
        soft_max = maxc + np.log(exp_shift.sum()) / alpha
        obj = 2.0 * n * soft_max

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if t % 1000 == 0:
                print(f"[adam] iter {t}, soft‑obj {obj:.6f}")

        # ----- Adam update (gradient of final objective) -----
        grad = 2.0 * n * g          # chain rule: d(2n*softmax)/da = 2n * dg/da
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a = a - step_size * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a = project_to_simplex(a)

        # ----- continuation on smoothing parameter -----
        if alpha < alpha_max:
            alpha = 1.0 + (alpha_max - 1.0) * (t / max_iter)
            if alpha > alpha_max:
                alpha = alpha_max

    return best_a

"""
Search for a non‑negative step‑function that minimises the
``evaluate_sequence`` objective.

Algorithm
---------

1. **Broad parametric sweep** – explore several analytic families
   (power‑law, geometric, beta‑shape, mixed plateau+power) and keep the
   best vector (normalised to sum = 1).  This provides a good start.

2. **Smooth‑max Adam optimisation** – minimise a
   ``soft‑max`` approximation of ``max(conv(a,a))``.
   The soft‑max is ``(1/α)·log Σ exp(α·conv)``.
   Its gradient can be written as a convolution, so each iteration
   needs only two FFTs (O(n log n)).  Adam‑style adaptive steps are used,
   with a continuation schedule that slowly raises the smoothing
   parameter α from 1 to 30, turning the objective into the exact maximum
   at the end.  After each update the vector is projected onto the
   probability simplex.

3. **Fine stochastic hill‑climb** – a cheap greedy local search that
   moves a small fraction of mass between two random positions,
   keeping the best vector found.

The whole pipeline respects the 1000 s budget, prints intermediate
scores and finally returns the best sequence (as a plain Python list)
found during the run.
"""

import time

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper returning the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear full convolution of two arbitrary vectors via FFT."""
    m = x.shape[0] + y.shape[0] - 1
    n_fft = _next_pow2(m)
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
       2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    if q <= 0.0 or q >= 1.0:
        return None
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a
