def linear_convolution(x, y):
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

The algorithm consists of three stages

1. **Global sampling** – a fast sweep of a few parametric families plus many
   random Dirichlet draws (the candidates are projected onto the simplex).

2. **Fast exponentiated‑gradient polishing** – the objective 2·n·max(a*a)
   is a scale‑invariant piece‑wise smooth function.  At each iteration we
   (a) compute the full convolution a*a, (b) find the active constraints
   (indices where the maximum is attained up to a tiny tolerance) and
   (c) obtain the sub‑gradient by a *single* FFT‑based convolution (instead
   of the O(n²) loops used previously).  The update
       a_i ← a_i·exp(‑η·g_i)
   respects the non‑negativity and is followed by a renormalisation onto the
   probability simplex.  A decreasing step size η=η₀/√t and occasional jitter
   keep the iterates moving.

   Using the FFT gradient makes it cheap to run thousands of iterations and
   to try many random restarts, which yields a much lower peak of the
   self‑convolution.

3. **(Optional) short simulated‑annealing refinement** – a few Metropolis
   moves are applied to the best vector found in order to escape possible
   plateaux.

The whole search respects the 1000 s wall‑clock budget; the majority of the
time is spent on the polishing stage.  The final sequence is returned as a
plain Python list of floats and is evaluated with the official
``evaluate_sequence`` function before reporting the score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------


def linear_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    m = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (self‑convolution)."""
    return linear_convolution(a, a)


# ----------------------------------------------------------------------
# Objective (exact, works on vectors that already sum to 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used only for the quick global sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^‑p."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, α: float, β: float) -> np.ndarray:
    """Discrete beta shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, α - 1.0)
    right = np.power(n - i + 1.0, β - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Global exploration (keeps only the best few candidates)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Generate a small pool of promising vectors."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep
    for p in np.linspace(0.1, 2.0, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(0.0001, 0.03, 150):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep (coarse)
    al_vals = np.linspace(0.2, 4.0, 12)
    be_vals = np.linspace(0.2, 4.0, 12)
    for α in al_vals:
        for β in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, α, β))

    # 4) random Dirichlet draws with a few concentrations
    concs = [0.3, 0.6, 1.0, 2.0, 4.0]
    draws_per_conc = 800                # ≈ 4 000 draws total
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Fast exponentiated‑gradient polishing (FFT gradient)
# ----------------------------------------------------------------------


def eg_polish_fft(a0: np.ndarray, deadline: float,
                  eta0: float = 0.7, tol: float = 1e-10) -> np.ndarray:
    """
    Projected exponentiated‑gradient minimisation of max(a*a) on the simplex.
    Gradient is obtained via a single FFT‑based convolution of the active‑set
    indicator with the reversed vector.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # 1) compute convolution and identify active constraints
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * tol
        active = (conv >= max_val - eps).astype(float)          # length 2n‑1

        # 2) sub‑gradient = 2 * correlation(active, a)
        a_rev = a[::-1]
        corr = linear_convolution(active, a_rev)               # length 3n‑2
        start = n - 1
        g = 2.0 * corr[start:start + n]

        # 3) EG update + simplex projection
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing (optional final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # Choose a size that gives a good trade‑off between expressiveness and speed.
    # 4096 provides roughly twice the degrees of freedom of the original 2048.
    n = 4096

    best_seq = None
    best_val = float("inf")

    # -------------------------------------------------------------
    # 0. Use a fallback if the notebook already contains a solution
    # -------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -------------------------------------------------------------
    # 1. Global exploration (quick sweep)
    # -------------------------------------------------------------
    baseline_deadline = deadline - 300.0   # keep ~5 min for polishing + anneal
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=8)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best score = {best_val:.6f}")

    # -------------------------------------------------------------
    # 2. Main polishing with fast EG (many random restarts)
    # -------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = eg_polish_fft(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[eg‑polish‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # additional random restarts while we have time
        while time.time() < polish_deadline:
            conc = np.random.choice([0.3, 0.6, 1.0, 2.0, 4.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = eg_polish_fft(a_start, polish_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[eg‑polish‑restart] new best = {val:.6f}")

    # -------------------------------------------------------------
    # 3. Very short simulated‑annealing fine‑tune
    # -------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -------------------------------------------------------------
    # 4. Fallback to a uniform distribution if everything failed
    # -------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
