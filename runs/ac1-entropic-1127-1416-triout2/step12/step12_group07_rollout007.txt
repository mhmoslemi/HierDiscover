"""
Propose a non‑negative step‑function that (approximately) minimizes
the evaluation function used in the competition.

Algorithm
---------
1. **Broad exploration** – generate many candidate shapes from several
   parametric families (power‑law, geometric, exponential, cosine‑power,
   symmetric power‑law) and from random Dirichlet/block constructions.
   Keep the best few (by the exact objective).

2. **Smooth‑max gradient descent** – starting from each of the top
   candidates we minimise a smooth approximation of
   ``max(conv(a,a))``:
        f_α(a)= (1/α)·log Σ_k exp(α·conv_k(a)).
   Its gradient is 2·(w * a) where w are the soft‑max weights.
   We use an exponential schedule for ``α`` and a simple momentum
   update, projecting back onto the probability simplex after every step.

3. **Simulated‑annealing hill‑climb** – a final stochastic refinement.
   Random mass‑transfer moves are accepted with a Metropolis probability
   that gradually cools, allowing occasional uphill steps to escape
   plateaux.

The whole search respects a 1000 s wall‑clock budget and returns the
best sequence found (as a plain ``list`` of floats).  The code contains
numerous diagnostic ``print`` statements to trace progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    n = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())       # next power of two
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:n]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper: full convolution a*a."""
    return fft_convolve(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` for a normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    """a_i = (i+offset)^(-p)   (offset>0 to avoid division by zero)."""
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        return None
    a /= a_sum
    return a


def geometric(n: int, r: float) -> np.ndarray:
    """a_i = r^i   (geometric decay)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def exponential_decay(n: int, tau: float) -> np.ndarray:
    """a_i = exp(-i/tau)   (continuous exponential decay)."""
    a = np.exp(-np.arange(n, dtype=np.float64) / tau)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def cosine_power(n: int, expo: float) -> np.ndarray:
    """a_i = cos(π i / (2 n))^expo  (decreasing cosine shape)."""
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, beta: float) -> np.ndarray:
    """a_i = (1 - i/(n-1))^beta   (monotonically decreasing)."""
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def sym_powerlaw(n: int, p: float, offset: float) -> np.ndarray:
    """Symmetric decreasing shape: sum of left and right power‑law tails."""
    idx = np.arange(n, dtype=np.float64) + offset
    left = idx ** (-p)
    right = left[::-1]
    a = left + right
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def sym_geometric(n: int, r: float) -> np.ndarray:
    """Symmetric geometric decay: r^i + r^(n-1-i)."""
    base = r ** np.arange(n, dtype=np.float64)
    a = base + base[::-1]
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    """Coarse piece‑wise‑constant decreasing shape with K blocks."""
    blocks = rng.dirichlet(np.ones(K))
    blocks = np.sort(blocks)[::-1]
    repeats = np.full(K, n // K, dtype=int)
    remainder = n - repeats.sum()
    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)[:n]
    a /= a.sum()
    return a


def random_decreasing_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random decreasing vector by sorting a Dirichlet sample."""
    a = rng.dirichlet(np.ones(n))
    a = np.sort(a)[::-1]
    a /= a.sum()
    return a

    for p in np.linspace(0.10, 4.0, 48):

# Smooth‑max gradient descent refinement
# ----------------------------------------------------------------------
def smooth_max_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected gradient descent on a smooth approximation of max(conv)."""
    a = a0.copy()
    best = a.copy()
    best_sc = compute_score(best)
    n = a.shape[0]

    # hyper‑parameters
    step0 = 0.7
    beta = 0.9                 # momentum
    alpha = 0.8                # start for smooth‑max
    alpha_inc = 1.3
    max_alpha = 80.0
    vel = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution
        conv = full_convolution(a)
        max_c = conv.max()
        # smooth max weights
        exp_term = np.exp(alpha * (conv - max_c))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp                   # length 2n‑1

        # gradient = 2 * (w * a) convolution (first n entries)
        grad_full = fft_convolve(w, a)           # length 3n‑2
        grad = 2.0 * grad_full[:n]

        # momentum update
        vel = beta * vel - (step0 / np.sqrt(it)) * grad

        # candidate step
        cand = a + vel
        cand = project_to_simplex(cand)

        # evaluate true score
        sc_cand = compute_score(cand)
        if sc_cand < best_sc:
            best_sc = sc_cand
            best = cand.copy()
            # optional: print progress
            if it % 100 == 0:
                print(f"[smooth‑grad] iter {it}, new best score = {best_sc:.6f}")

        # accept if improvement, else keep current a
        if sc_cand < compute_score(a):
            a = cand

        # gradually tighten the smooth approximation
        if it % 250 == 0 and alpha < max_alpha:
            alpha = min(alpha * alpha_inc, max_alpha)

    return best

# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb_sa(a_start: np.ndarray, deadline: float, max_iter: int = 500_000) -> np.ndarray:
    """Greedy stochastic hill‑climb with Metropolis acceptance."""
    a = a_start.copy()
    best = a.copy()
    best_sc = compute_score(best)
    n = a.shape[0]

    rng = np.random.default_rng(987654321)

    # annealing schedule
    T0 = 0.03 * best_sc               # start temperature proportional to score
    T = T0
    decay = 0.99997

    step_scale = 0.06
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a fraction of mass from i to j
        delta = step_scale * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc_cand = compute_score(cand)
        delta_sc = sc_cand - best_sc

        if delta_sc < 0 or rng.random() < np.exp(-delta_sc / max(T, 1e-12)):
            a = cand
            if sc_cand < best_sc:
                best_sc, best = sc_cand, cand.copy()
                # log improvement occasionally
                if it % 1000 == 0:
                    print(f"[SA] iter {it}, best score = {best_sc:.6f}")
        # temperature decay
        T *= decay
        # adapt step scale a bit
        if it % 2000 == 0 and T < T0 * 0.1:
            step_scale = max(step_scale * 0.95, 0.001)

        it += 1

    return best
