"""
Proposes a step function with near-uniform self‑convolution.
The approach
1. initialise a symmetric arcsine‑type distribution;
2. minimise the L2 deviation of the convolution from the constant
   C = 1/(2·n‑1) using Adam on the soft‑max parametrisation
   (this flattens the whole autocorrelation, not only the peak);
3. finish with a cheap projected sub‑gradient descent that directly
   lowers the current maximum value;
4. return the best sequence found before the time budget expires.
Both phases respect the global wall‑clock budget (default 1000 s).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by the grader (sum of a is assumed = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (initial guesses)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    α = 0.5 gives the discrete arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_uniform_optimize(start_a: np.ndarray,
                          deadline: float,
                          lr: float = 0.08,
                          max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the squared‑error objective
        F(a) = Σ_k ( (a*a)_k - C )²   with C = 1/(2n‑1) .
    The variable is the log‑it vector `z` (softmax parametrisation)
    guaranteeing non‑negativity and the simplex constraint.
    """
    n = start_a.shape[0]
    C = 1.0 / (2 * n - 1)

    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        a = softmax(z)

        # convolution a*a  (length 2n-1)
        conv = convolution_fft(a, a)

        # error w.r.t. the uniform target
        e = conv - C

        # cross‑correlation a ⋆ e  → gradient = 4·(a ⋆ e)
        rev_e = e[::-1]
        corr = convolution_fft(a, rev_e)
        start_idx = len(e) - 1
        grad_a = 4.0 * corr[start_idx:start_idx + n]

        # map gradient to the soft‑max logits
        dot_ag = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot_ag)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional progress report
        if it % 5000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam‑uniform improved to {best_score:.6f}")

    # final evaluation
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     step0: float = 0.07,
                     max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Cheap projected sub‑gradient descent on the exact max‑convolution
    objective.  Works well as a final polishing phase.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # active‑set sub‑gradient (identical to the one used in EG)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(it)
        a = a - step * g
        # Project back onto the probability simplex
        # Use Euclidean projection (the same routine as in the reference code)
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1, dtype=a.dtype)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            theta = 0.0
        else:
            rho = np.where(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
        a = np.maximum(a - theta, 0.0)

        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

    """

    start_time = time.time()

    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load a fallback candidate if one is supplied by the harness.
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    best_score = compute_score(init_a)

            init_a,

            lr=0.08,

        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.tolist()
            print(f"[{time.time():.1f}] Adam‑uniform improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Short projected sub‑gradient polishing.
    # ------------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=np.float64)
    if time.time() + subg_time < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient polishing (≈{subg_time:.1f}s)")
        a_sub, sc_sub = subgrad_optimize(
            cur_a,

            step0=0.07,

            best_seq = a_sub.tolist()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

is invariant under a positive scaling, therefore we may (and do) keep the
vector on the probability simplex \(\sum a_i = 1,\ a_i\ge 0\).
When the self‑convolution is perfectly flat,
\((a*a)_k = \frac{1}{2n-1}\) for all \(k\), the objective reaches its
theoretical lower bound  

\[
E_{\min}= \frac{2n}{2n-1}\;.
\]

The classic *arcsine* distribution produces a nearly uniform convolution,
but it is not optimal.  The key idea of the new search is to **flatten the
whole convolution**, not only its maximum:

*   minimise the squared‑error  
    \[
    F(a)=\sum_k\bigl((a*a)_k-\frac1{2n-1}\bigr)^2 .
    \]

*   Optimise \(F\) with Adam on the *soft‑max* parametrisation  
    \(a_i=\operatorname{softmax}_i(z)\); this guarantees the simplex
    constraints without an explicit projection.

*   After the “flat‑convolution’’ phase perform a cheap projected
    sub‑gradient descent that directly reduces the current maximum – a
    final polishing step.

All phases respect the global time budget (default 1000 s).  The
algorithm works for any length; we use a reasonably large \(n=8192\)
(which gives a bound only 0.00006 away from 1) while staying well within
memory limits.

The code below follows exactly this plan and returns a plain Python list
that can be fed to `evaluate_sequence`.  It prints intermediate scores so
that progress can be monitored.

```python
"""
Proposes a step function with near‑uniform self‑convolution.

Algorithm
---------
1.  Initialise a symmetric arcsine‑type distribution.
2.  Adam optimisation of the L2 deviation of the convolution from the
    constant C = 1/(2·n‑1).  The variables are logits z with a = softmax(z),
    guaranteeing non‑negativity and ∑a = 1.
3.  A short projected sub‑gradient descent that directly lowers the
    current maximal convolution value (polishing).
4.  Return the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the grader (the sum of a is assumed = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (initial guesses)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    α = 0.5 gives the discrete arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_uniform_optimize(start_a: np.ndarray,
                          deadline: float,
                          lr: float = 0.08,
                          max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the squared‑error objective
        F(a) = Σ_k ( (a*a)_k - C )²,   C = 1/(2n‑1).

    The variable is the log‑it vector `z` (soft‑max parametrisation) so that
    a = softmax(z) automatically satisfies a_i ≥ 0 and Σ a_i = 1.
    """
    n = start_a.shape[0]
    C = 1.0 / (2 * n - 1)

    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        a = softmax(z)

        # a * a  (length 2n‑1)
        conv = convolution_fft(a, a)

        # deviation from the uniform target
        e = conv - C

        # gradient of   Σ e_k²   w.r.t. a : 4 * (a ⋆ e)
        rev_e = e[::-1]
        corr = convolution_fft(a, rev_e)
        start_idx = len(e) - 1               # position of lag 0
        grad_a = 4.0 * corr[start_idx:start_idx + n]

        # map to gradient w.r.t. logits (soft‑max Jacobian)
        dot_ag = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot_ag)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional reporting
        if it % 5000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam‑uniform improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     step0: float = 0.07,
                     max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Cheap projected sub‑gradient descent on the exact max‑convolution
    objective.  Used as a polishing step after the Adam phase.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(it)
        a = a - step * g

        # Euclidean projection onto the simplex
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1, dtype=a.dtype)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            theta = 0.0
        else:
            rho = np.where(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
        a = np.maximum(a - theta, 0.0)

        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function observes the global time budget and
    returns a plain Python list ready for evaluation.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the harness.
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Load a fallback candidate if supplied.
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose the problem size.
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # Prefer a fairly large n but stay well inside memory limits.
    n = max(8192, fallback_len) if fallback_len < 4096 else fallback_len
    print(f"[{time.time():.1f}] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2. Initialise with a symmetric arcsine‑type distribution.
    # ------------------------------------------------------------------
    # A tiny sweep over a few α values (seconds only) to get a good start.
    init_a = None
    init_score = float("inf")
    if time.time() < deadline - 20.0:
        for alpha in (0.5, 0.45, 0.55, 0.4, 0.6):
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < init_score:
                init_score = sc
                init_a = vec
                print(f"[{time.time():.1f}] init α={alpha:.2f} ⇒ {sc:.6f}")

    if init_a is None:
        init_a = np.full(n, 1.0 / n, dtype=np.float64)

    init_a = np.maximum(init_a, 0.0)
    init_a = init_a / init_a.sum()
    best_seq = init_a.tolist()
    best_score = compute_score(init_a)
    print(f"[{time.time():.1f}] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Time allocation.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_time = 0.75 * remaining          # main flattening phase
    subg_time = remaining - adam_time      # polishing phase

    # ------------------------------------------------------------------
    # 4. Adam optimisation that flattens the whole convolution.
    # ------------------------------------------------------------------
    if time.time() + adam_time < deadline:
        print(f"[{time.time():.1f}] Adam‑uniform phase (≈{adam_time:.1f}s)")
        a_adam, sc_adam = adam_uniform_optimize(
            init_a,
            deadline=time.time() + adam_time,
            lr=0.08,
            max_iters=5_000_000,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.tolist()
            print(f"[{time.time():.1f}] Adam‑uniform improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Short projected sub‑gradient polishing.
    # ------------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=np.float64)
    if time.time() + subg_time < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient polishing (≈{subg_time:.1f}s)")
        a_sub, sc_sub = subgrad_optimize(
            cur_a,
            deadline=time.time() + subg_time,
            step0=0.07,
            max_iters=5_000_000,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] Polishing improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation via the provided `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = best_seq
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
```
