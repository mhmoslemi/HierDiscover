Search for a non‑negative step‑function minimising

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of several fast phases that respect a global wall‑clock
budget (default 1000 s):

1.  Very wide scan of analytic families (U‑shaped, Gaussian, cosine‑power,
    exponential‑U, left‑heavy power, window functions) on a moderate problem
    size (n≈2048).  The best vector is kept.

2.  Smooth‑max Adam optimisation.  The non‑smooth “max” is replaced by a
    temperature‑controlled soft‑max, allowing a well‑behaved gradient.
    The scheme uses the exact gradient of the smoothed objective via FFT
    convolutions and Euclidean projection onto the probability simplex.

3.  (Optional) Upsampling to double the length and a short Adam refinement.
    Splitting each mass into two equal halves often reduces the peak
    autocorrelation more than the increase of the factor 2·n.

4.  LP‑based mixing (the classic state‑of‑the‑art step), applied only when the
    length is ≤2048 to keep the linear program reasonably sized.

5.  Symmetrisation – the problem is symmetric, so averaging with the reversed
    vector frequently improves the bound.

6.  A brief simulated‑annealing hill‑climb for the remaining budget.

The final sequence is returned as a plain Python list; it is already
non‑negative and normalised (∑a=1), so ``evaluate_sequence`` can be called
directly on it.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using real FFT (fast)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: w_i ∝ ((i+0.5)*(n‑0.5‑i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone left‑heavy: w_i ∝ (i+0.5)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay from the centre (U‑shaped)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, window_type: str) -> np.ndarray:
    """Standard cosine windows (Hann, Hamming, Blackman)."""
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def cosine_power_family(n: int, p: float) -> np.ndarray:
    """Symmetric cosine‑power window: (0.5‑0.5·cos)^p."""
    i = np.arange(n, dtype=np.float64)
    w = (0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))) ** p
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing (heavy‑left) vector."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped start via exponential‑U family."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    beta_init: float = 0.2,
    beta_mult: float = 1.001,
    beta_max: float = 50.0,
    eval_interval: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed objective
        fβ(a) = (1/β)·log Σ exp(β·conv_k)

    The gradient of fβ can be written as a convolution of the soft‑max
    weights with the reversed vector, which makes each iteration O(n log n).
    """
    n = start_a.shape[0]
    a = start_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # state for the smooth‑max temperature
    beta = beta_init

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- compute soft‑max weights -----
        conv = convolution_fft(a, a)                     # length 2n‑1
        c = beta * conv
        c_max = np.max(c)                                # for numerical stability
        w = np.exp(c - c_max)
        w_sum = w.sum()
        if w_sum == 0.0:  # degenerate – fall back to uniform weights
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ----- gradient of the smoothed max -----
        # g_i = 2 * Σ_k w_k * a_{k-i}  (indices out of range are ignored)
        grad_full = convolution_fft(w, a[::-1])          # length 2n‑1
        g = 2.0 * grad_full[n - 1 : n - 1 + n]           # slice of length n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (math.sqrt(t))                      # simple decay
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # keep feasible: non‑negative and sum‑to‑one
        a = project_to_simplex(a)

        # anneal the temperature slowly
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mult)

        # periodic evaluation of the true (non‑smoothed) objective
        if t % eval_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (classical mixing step)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0, then mix f←(1‑t)f + t·g_norm.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (m × n) such that A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)  # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 20},
    )
    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f
    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
) -> tuple[np.ndarray, float]:
    """Simple SA with occasional swaps and mass transfers."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def upsample_vector(v: np.ndarray, factor: int = 2) -> np.ndarray:
    """Duplicate each entry `factor` times and divide the mass accordingly."""
    if factor <= 1:
        return v.copy()
    v_rep = np.repeat(v, factor)
    v_rep = v_rep / float(factor)  # keep total mass unchanged
    # re‑normalise to guard against floating‑point drift
    return v_rep / v_rep.sum()


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The function respects the global time budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback candidate supplied by the harness
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose a base problem size (moderate for LP‑refinement)
    # ------------------------------------------------------------------
    # If we already have a decent fallback with >1500 entries we keep it,
    # otherwise we start with n = 2048 (a sweet spot between quality and speed).
    fallback_len = len(best_seq) if best_seq is not None else 0
    base_n = fallback_len if fallback_len >= 1500 else 2048
    print(f"[{time.time() - start_time:.1f}] using base length n={base_n}")

    # ------------------------------------------------------------------
    # Helper to test a candidate (normalised internally)
    # ------------------------------------------------------------------
    def try_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Very wide analytic scan (families + random starts)
    # ------------------------------------------------------------------
    try_candidate(np.full(base_n, 1.0 / base_n, dtype=np.float64))  # uniform baseline

    # Product family
    alphas = np.concatenate([np.linspace(0.0, 5.0, 51), np.linspace(5.1, 200.0, 40)])
    for alpha in alphas:
        if time.time() >= deadline - 250:
            break
        try_candidate(product_family(base_n, alpha))

    # Left‑heavy power family
    betas = np.linspace(0.0, 30.0, 61)
    for beta in betas:
        if time.time() >= deadline - 250:
            break
        try_candidate(left_heavy_power_family(base_n, beta))

    # Exponential‑U family
    gammas = np.linspace(0.0, 20.0, 41)
    for gamma in gammas:
        if time.time() >= deadline - 250:
            break
        try_candidate(exp_u_family(base_n, gamma))

    # Gaussian family
    gauss_gammas = np.logspace(-4, 2, 40)
    for gamma in gauss_gammas:
        if time.time() >= deadline - 250:
            break
        try_candidate(gaussian_family(base_n, gamma))

    # Cosine‑power windows (symmetric)
    for p in np.linspace(0.1, 8.0, 35):
        if time.time() >= deadline - 250:
            break
        try_candidate(cosine_power_family(base_n, p))

    # Standard cosine windows
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 250:
            break
        try_candidate(window_family(base_n, win))

    # Random starts
    for _ in range(200):
        if time.time() >= deadline - 250:
            break
        if np.random.rand() < 0.5:
            try_candidate(random_monotone_start(base_n))
        else:
            try_candidate(random_u_start(base_n))

    # ------------------------------------------------------------------
    # 3.  Normalise the current best (fallback to uniform if nothing)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(base_n, 1.0 / base_n, dtype=np.float64)
    else:
        a = np.asarray(best_seq, dtype=np.float64)
    a = project_to_simplex(a)
    best_seq = a.copy()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 4.  Adam optimisation on the smooth‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        print(f"[{time.time() - start_time:.1f}] starting Adam optimisation")
        a_opt, sc_opt = adam_optimize(
            a,
            deadline - 150.0,
            max_iters=180_000,
            lr=0.06,
            beta_init=0.2,
            beta_max=40.0,
            eval_interval=4000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Optional up‑sampling to larger length and a short Adam pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0 and a.shape[0] <= 2048:
        print(f"[{time.time() - start_time:.1f}] up‑sampling to double length")
        a_big = upsample_vector(a, factor=2)          # length 2·base_n
        # a short Adam run on the larger problem
        a_big_opt, sc_big_opt = adam_optimize(
            a_big,
            deadline - 70.0,
            max_iters=80_000,
            lr=0.05,
            beta_init=0.3,
            beta_max=30.0,
            eval_interval=2500,
        )
        if sc_big_opt < best_score:
            best_score = sc_big_opt
            a = a_big_opt
            print(f"[{time.time() - start_time:.1f}] after up‑sample Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  LP‑based refinement (only for manageable sizes)
    # ------------------------------------------------------------------
    if a.shape[0] <= 2048 and time.time() < deadline - 70.0:
        print(f"[{time.time() - start_time:.1f}] starting LP refinement loop")
        mix_weight = 0.18
        lp_iters = 0
        while time.time() < deadline - 30.0 and lp_iters < 25:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc = compute_score(a_new)
            if sc < best_score - 1e-7:
                best_score = sc
                a = a_new
                print(f"[{time.time() - start_time:.1f}] LP step {lp_iters} improved to {best_score:.6f}")
            else:
                mix_weight = max(0.04, mix_weight * 0.85)
            lp_iters += 1

    # ------------------------------------------------------------------
    # 7.  Symmetrisation (the problem is symmetric)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time() - start_time:.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Final stochastic hill‑climb (SA) for the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=1_200_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(
        f"\n[{time.time() - start_time:.1f}] finished search – final reported score = {final_score:.6f}"
    )
    return final_seq
