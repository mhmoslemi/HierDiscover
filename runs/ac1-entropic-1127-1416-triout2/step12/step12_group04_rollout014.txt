"""
Propose a non‑negative step function that lowers the evaluation score.
Algorithm
---------

1. Choose a relatively large length (default 8192) – a larger n allows the
   convolution peak to approach the theoretical lower bound 1.

2. Initialise the vector with a handful of analytic families
   (U‑shaped product, left‑heavy power, exponential‑U, Gaussian) and a few
   random starts.  The best of these (measured by the exact objective)
   becomes the starting point.

3. Smooth‑max gradient descent.
   *   Use a soft‑max (log‑sum‑exp) to approximate max(conv(a,a)).
   *   The gradient of `softmax(conv)` w.r.t. a can be written as
       grad = conv(w, a)[:n] + conv(w[::-1], a)[:n],
       where w are the soft‑max weights (a probability vector of length 2n‑1).
   *   Perform projected gradient descent on the simplex with a decreasing
       step size and slowly increase the soft‑max temperature (β) so that the
       optimisation first flattens the whole convolution and later concentrates
       on the true maximum.

4. Refine the best vector with a tiny linear‑programming step that
   only enforces the currently active convolution constraints
   (those whose value is within 1 % of the peak).  This cheap LP
   pushes mass into “quiet’’ positions without building the full
   O(n²) constraint matrix.

5. Final short stochastic hill‑climb (simulated‑annealing style)
   to escape any remaining local minima.

All phases respect the 1000 s wall‑clock budget and the final
vector is returned as a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective when sum(a)=1:
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logw = -alpha * (np.log(i + 0.5) + np.log(n - 0.5 - i))
    logw -= np.max(logw)                     # avoid overflow
    w = np.exp(logw)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone left‑heavy: w_i ∝ (i+0.5)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    logw = -beta * np.log(i + 0.5)
    logw -= np.max(logw)
    w = np.exp(logw)
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Exponential U‑shape: w_i ∝ exp(-γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logw = -gamma * dist
    logw -= np.max(logw)
    w = np.exp(logw)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian: w_i ∝ exp(-γ·(i‑mid)²)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    logw = -gamma * ((i - mid) ** 2)
    logw -= np.max(logw)
    w = np.exp(logw)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Heavy‑left random decreasing vector."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped start (exponential distance)."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(start_a: np.ndarray,
                       deadline: float,
                       max_iters: int = 300_000,
                       beta_start: float = 0.5,
                       beta_factor: float = 1.001,
                       beta_max: float = 1500.0,
                       step0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed max:
        f(a) = (1/β)·log‑sum‑exp(β·conv(a,a))

    Returns the best vector encountered.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    beta = beta_start
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # (1) full convolution a*a
        conv = convolution_fft(a, a)            # length 2n‑1

        # (2) soft‑max weights
        scaled = beta * (conv - conv.max())
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # (3) gradient = conv(w,a)[:n] + conv(w[::-1],a)[:n]
        grad1 = convolution_fft(w, a)[:n]
        grad2 = convolution_fft(w[::-1], a)[:n]
        grad = grad1 + grad2

        # (4) step size (simple 1/√t schedule)
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = project_to_simplex(a)

        # occasional exact evaluation
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:.1f}] smooth descent iter {it} → {best_score:.6f}")

        # increase smoothing temperature
        beta = min(beta * beta_factor, beta_max)

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap LP refinement (only active constraints)
# ----------------------------------------------------------------------


def lp_refine_active(f: np.ndarray,
                     M: float,
                     mix_weight: float = 0.25,
                     rel_thresh: float = 0.99) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M   for a subset of indices k
    where (f*f)_k is within ``rel_thresh`` of the current maximum M.
    Return the convex mixture (1‑t)·f + t·g_norm, projected onto the simplex.
    """
    n = f.size
    conv = convolution_fft(f, f)
    # indices that are “active’’ (close to the peak)
    active = np.where(conv >= rel_thresh * M)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=int)

    # Build the linear constraint matrix A (|active| × n)
    p = active.size
    A = np.zeros((p, n), dtype=np.float64)
    for row, k in enumerate(active):
        # for each feasible j, coefficient is f_{k‑j}
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # indices j = lo..hi
        j_range = np.arange(lo, hi + 1, dtype=int)
        A[row, j_range] = f[k - j_range]

    c = -np.ones(n, dtype=np.float64)      # maximise Σ g_j  → minimise −Σ g_j
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(p, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 5})
    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f
    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Simple hill‑climb (simulated‑annealing) – final polishing
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 1e-4
    T = T0
    scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                scale = min(scale * 1.07, 0.5)
        else:
            scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a fairly large problem size (ignore any provided fallback)
    # ------------------------------------------------------------------
    n = 8192
    print(f"[{time.time() - start_time:.1f}] problem size n = {n}")

    # ------------------------------------------------------------------
    # Helper to keep the best vector seen so far
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    def consider(vec: np.ndarray):
        nonlocal best_a, best_score
        a = np.maximum(vec, 0.0)
        s = a.sum()
        if s == 0.0:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best init score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 1. Scan a few analytic families (very modest sweep)
    # ------------------------------------------------------------------
    consider(np.full(n, 1.0 / n, dtype=np.float64))       # uniform

    # product family – a handful of alphas (U‑shaped)
    for alpha in (0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0):
        if time.time() >= deadline - 200:
            break
        consider(product_family(n, alpha))

    # left‑heavy power family
    for beta in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        if time.time() >= deadline - 200:
            break
        consider(left_heavy_power_family(n, beta))

    # exponential‑U family
    for gamma in (0.2, 0.5, 1.0, 2.0, 4.0):
        if time.time() >= deadline - 200:
            break
        consider(exp_u_family(n, gamma))

    # Gaussian family (wide to narrow)
    for gamma in (1e-4, 1e-3, 5e-3, 1e-2, 2e-2, 5e-2, 1e-1):
        if time.time() >= deadline - 200:
            break
        consider(gaussian_family(n, gamma))

    # Random starts (provides diversity)
    for _ in range(30):
        if time.time() >= deadline - 200:
            break
        if np.random.rand() < 0.5:
            consider(random_monotone_start(n))
        else:
            consider(random_u_start(n))

    # ------------------------------------------------------------------
    # If nothing survived the scan (unlikely), fall back to uniform
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. Smooth‑max gradient descent (major part of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 100.0:
        print(f"[{time.time() - start_time:.1f}] start smooth‑max descent")
        gd_deadline = time.time() + remaining * 0.70
        best_a, best_score = smooth_max_descent(
            best_a,
            gd_deadline,
            max_iters=400_000,
            beta_start=0.5,
            beta_factor=1.001,
            beta_max=1500.0,
            step0=0.9,
        )
        print(f"[{time.time() - start_time:.1f}] after descent best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Light LP refinement using only the active constraints
    # ------------------------------------------------------------------
    while time.time() < deadline - 40.0:
        conv = convolution_fft(best_a, best_a)
        M = float(conv.max())
        new_a = lp_refine_active(best_a, M, mix_weight=0.25, rel_thresh=0.99)
        new_score = compute_score(new_a)
        if new_score < best_score - 1e-7:
            best_score = new_score
            best_a = new_a
            print(f"[{time.time() - start_time:.1f}] LP refinement improved to {best_score:.6f}")
        else:
            # if no improvement, reduce mixing weight and try again
            break

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing hill‑climb (short tail)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (SA)")
        sa_deadline = deadline
        best_a, best_score = hill_climb_sa(best_a, sa_deadline, max_steps=800_000)
        print(f"[{time.time() - start_time:.1f}] after SA best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return a plain Python list (evaluate_sequence will clamp/normalise)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_reported = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished – final reported score = {final_reported:.6f}")
    return final_seq
```

This implementation uses a larger problem size, a smooth‑max gradient descent that moves mass away from the central peak, a cheap LP refinement focused on the currently active convolution constraints, and a short simulated‑annealing style hill‑climb to polish the final result. All phases respect the 1000 s wall‑clock limit and return a list ready for `evaluate_sequence`.
