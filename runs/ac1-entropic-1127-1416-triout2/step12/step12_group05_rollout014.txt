"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm combines three ideas:
1. A small‑scale linear‑programming (LP) phase that drives the
   convolution towards a flat (near‑constant) shape.  For a short
   vector (n≈1024) we solve
        max ∑ g_i   s.t.   (f * g)_k ≤ M   for all k,
   where M = max(f*f).  The LP yields an extreme point g that,
   after rescaling, reduces the maximal autocorrelation.
2. The LP solution is interpolated to a larger length (n≈16384) and
   refined by a long Adam optimisation of a smooth‑max surrogate
   of the maximum convolution.  A cheap symmetrisation step
   (mirroring the vector every few iterations) keeps the shape
   essentially even, which is known to improve flatness.
3. A final projected sub‑gradient descent and a simulated‑annealing
   hill‑climb finish the optimisation, polishing any remaining
   peaks.

All steps respect the 1000 s wall‑clock budget; intermediate best
vectors are kept and the final one is returned as a plain list of
floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (O(N log N)).
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) / (sum(a)^2).
    a is assumed non‑negative.
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 5.0,
                         beta_end: float = 20000.0,
                         lr: float = 0.015,
                         sym_interval: int = 10) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    The vector is periodically symmetrised (mirrored) to keep it
    essentially even.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise via log‑softmax (ensures positivity)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z) / np.exp(z).sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule for smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z) / np.exp(z).sum()

        # optional symmetrisation
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true evaluation
        if it % 2000 == 0:
            a_tmp = np.exp(z) / np.exp(z).sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z) / np.exp(z).sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 6_000_000,
                  temp_factor: float = 2e-3,
                  step_scale_init: float = 0.15) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Small‑scale LP phase (n≈1024)
# ----------------------------------------------------------------------


def lp_refine_small(n_small: int,
                    iters: int,
                    deadline: float,
                    t_mix: float = 0.2) -> np.ndarray:
    """
    Starting from a uniform vector (sum = sqrt(2·n_small)),
    repeatedly solve the LP that maximises Σ g_i under
    (f * g)_k ≤ M, rescales g to the same ℓ1‑norm,
    and mixes it into f.
    Returns the refined vector (still with sum = sqrt(2·n_small)).
    """
    # uniform initialization with the right ℓ1‑norm
    f = np.full(n_small, math.sqrt(2 * n_small) / n_small)

    rows = 2 * n_small - 1

    for it in range(iters):
        if time.time() >= deadline:
            break

        conv_f = convolution_fft(f, f)
        M = conv_f.max()

        # Build matrix A where A[k,i] = f[i] if i contributes to conv index k
        A = np.zeros((rows, n_small), dtype=np.float64)
        for i in range(n_small):
            A[i:i + n_small, i] = f[i]

        # Linear programme: maximize sum(g)  <=> minimize -sum(g)
        c = -np.ones(n_small)
        b = np.full(rows, M, dtype=np.float64)

        # Solve with HiGHS (fast and robust)
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            print(f"[{time.time():.1f}] LP iteration {it} failed – skipping")
            break

        g = res.x
        s = g.sum()
        if s == 0:
            break
        # Rescale g to the same ℓ1‑norm as f
        g = g * (math.sqrt(2 * n_small) / s)

        # Convex mix
        f = (1.0 - t_mix) * f + t_mix * g

    # At the end we return a *probability* vector (sum = 1) for the
    # large‑scale phase – the exact scaling does not matter because the
    # final objective is scale‑invariant.
    return normalize(f)


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation metric.  Returns the best sequence (plain list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Optional fallback (previous best if available)
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Small‑scale LP refinement (n≈1024)
    # -----------------------------------------------------------
    n_small = 1024
    print(f"[{time.time():.1f}] LP refinement on n={n_small}")
    f_small = lp_refine_small(n_small=n_small,
                              iters=6,
                              deadline=deadline - 200.0,
                              t_mix=0.2)

    # -----------------------------------------------------------
    # 2. Upsample to a larger working size (n≈16384)
    # -----------------------------------------------------------
    n_big = 16384
    print(f"[{time.time():.1f}] up‑sampling to n={n_big}")
    x_small = np.linspace(0, 1, n_small)
    x_big = np.linspace(0, 1, n_big)
    a_big = np.interp(x_big, x_small, f_small)
    a_big = normalize(a_big)                     # ensure sum = 1

    # -----------------------------------------------------------
    # 3. Adam optimisation with periodic symmetrisation
    # -----------------------------------------------------------
    if time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Adam refinement (symmetrised)")
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=a_big,
            deadline=deadline - 150.0,
            max_iters=500_000,
            beta_start=5.0,
            beta_end=20000.0,
            lr=0.015,
            sym_interval=10,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Projected sub‑gradient descent (exact max)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline - 60.0,
            max_iters=1_500_000,
            step0=0.05,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Final simulated‑annealing hill‑climb
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=6_000_000,
            temp_factor=2e-3,
            step_scale_init=0.15,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Fallback to a safe uniform vector if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
