def upsample(seq, n):
    seq = np.asarray(seq, dtype=float)
    old_n = len(seq)
    if old_n == n:
        return seq.copy()
    # sample using linear interpolation
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    return np.interp(new_x, old_x, seq)
```

"""
Propose a step‑function sequence that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm proceeds in several stages within the 1000 s wall‑clock budget:

1.  Initialise from the previous construction `height_sequence_1` (if available)
    by up‑sampling to a working length `n=8192`.
2.  Scan a handful of analytic families (product‑power, left‑heavy power,
    exponential‑U, beta) on a coarse grid and keep the best.
3.  Perform a dense random search over the same families (≈3000 samples).
4.  Refine the best candidate with several (symmetric) Adam runs on a
    smooth‑max surrogate of the max‑convolution.
5.  Run projected sub‑gradient descent on the exact max‑objective.
6.  Finish with a simulated‑annealing hill‑climb that uses both single‑entry
    swaps and block‑mass transfers.
7.  As a last touch, symmetrise and smooth the final vector and return it.

All steps respect the time budget, fall back to a uniform vector if needed,
and print intermediate scores for monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)^2).  Assumes a ≥ 0."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


def upsample(seq: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to length n."""
    seq = np.asarray(seq, dtype=float).ravel()
    old_n = seq.shape[0]
    if old_n == n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, n)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------
# Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 5.0,
                         beta_end: float = 20000.0,
                         lr: float = 0.018,
                         sym_interval: int = 10) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate of max(conv)."""
    n = start_a.shape[0]
    eps = 1e-12
    # initialise via log‑softmax (ensures positivity)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = normalize(best_a)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # Temperature schedule for smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = normalize(a)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = normalize(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        mx = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - mx))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        grad_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * grad_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 3000 == 0:
            a_tmp = normalize(np.exp(z))
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final evaluation
    a_fin = normalize(np.exp(z))
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.045) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    n = start_a.shape[0]
    a = normalize(start_a)
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = normalize(a)                     # projection onto simplex

        # enforce symmetry occasionally (helps stability)
        if t % 5000 == 0:
            a = (a + a[::-1]) * 0.5
            a = normalize(a)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 4_000_000,
                  temp_factor: float = 2e-3,
                  step_scale_init: float = 0.2) -> tuple[np.ndarray, float]:
    """Simulated‑annealing hill‑climb with swap & block‑transfer moves."""
    n = start_a.size
    cur_a = normalize(start_a.copy())
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # block‑mass transfer
            # choose source block
            Lmax = min(100, n // 10)
            l = np.random.randint(1, Lmax + 1)
            i = np.random.randint(0, n - l + 1)
            # choose destination block (may overlap)
            j = np.random.randint(0, n - l + 1)
            if i != j:
                src = cand[i:i + l]
                src_sum = src.sum()
                if src_sum > 0:
                    delta = step_scale * src_sum
                    # remove fraction from source uniformly
                    src_factor = max(0.0, (src_sum - delta) / src_sum)
                    cand[i:i + l] = src * src_factor
                    # add equally to destination
                    cand[j:j + l] += delta / l

        cand = normalize(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """

    Returns the best sequence found as a plain list of floats.

    np.random.seed(seed)

    deadline = start_time + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # Helper to test a candidate and update the current best
    # ------------------------------------------------------------------
    def try_candidate(arr: np.ndarray):
        nonlocal best_seq, best_score
        arr = np.maximum(arr, 0.0)
        arr = normalize(arr)
        sc = compute_score(arr)
        if sc < best_score:
            best_score = sc
            best_seq = arr.tolist()
            print(f"[{time.time():.1f}] new best = {best_score:.6f}")
        return sc

    # 0. Optional fallback to previous construction (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        fb_seq = np.asarray(height_sequence_1, dtype=float)
        fb_seq = upsample(fb_seq, 8192)
        fb_sc = try_candidate(fb_seq)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Working length
    # ------------------------------------------------------------------
    n = 8192

    # ------------------------------------------------------------------
    # 1. Coarse scan of analytic families (extended ranges)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse scan (n={n})")
    # symmetric product family
    for alpha in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        try_candidate(vec)
        # a few random right‑shifts
        for _ in range(2):
            d = np.random.randint(0, n // 20 + 1)
            shifted = np.concatenate([vec[d:], np.zeros(d)])
            try_candidate(shifted)

    # left‑heavy power family
    for beta in np.linspace(0.0, 15.0, 61):

        vec = left_heavy_power_family(n, beta)
        try_candidate(vec)

        vec = exp_u_family(n, gamma)
        try_candidate(vec)

    # beta family (moderate exponents)
    for _ in range(30):
        if time.time() >= deadline - 30.0:
            break
        p = np.random.uniform(0.05, 4.0)
        q = np.random.uniform(0.05, 4.0)
        vec = beta_family(n, p, q)
        try_candidate(vec)

    # ------------------------------------------------------------------
    # 2. Random dense search over families
    # ------------------------------------------------------------------
    trials = 0
    max_trials = 3000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(5)
        if choice == 0:  # product family
            alpha = np.random.uniform(0.0, 15.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 1:  # beta family
            p = np.random.uniform(0.05, 5.0)
            q = np.random.uniform(0.05, 5.0)
            vec = beta_family(n, p, q)
        elif choice == 2:  # left‑heavy power
            beta = np.random.uniform(0.0, 15.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 3:  # exponential‑U

        else:  # mixture uniform + product

    # final official evaluation (includes security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
