assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function `a = (a0,…,a_{n‑1})` that minimises  

\[
F(a)=\frac{2n\;\max (a*a)}{(\sum_i a_i)^2},
\]

where `*` denotes the ordinary (aperiodic) convolution.
The optimisation is performed inside the allowed 1000 s wall‑clock budget.

Key ideas
---------

1. **Large problem size** – the theoretical lower bound `2·n/(2·n‑1)→1` is approached only for large `n`.  
   We therefore aim for `n = 16384` (≈ 16 k) which still keeps the final `np.convolve` fast enough.

2. **Analytic initial families** – a short sweep over a few well‑known families  
   (symmetric product, exponential tails, power‑law tails, mixtures) supplies an excellent
   starting point.  The sweep is cheap compared with the later iterative phase.

3. **Exponentiated‑gradient (mirror descent)** – using the sub‑gradient of the *active* convolution entry
   we perform many cheap updates of the form  

   \[
   a_i \leftarrow a_i\exp(-\eta\,g_i),\qquad
   a\gets a/\sum a,
   \]

   with a diminishing step size `η = η0/√t`.  This phase quickly lowers the worst
   convolution value.

4. **Smooth‑max surrogate optimisation (Adam)** – the non‑smooth `max` is replaced by  

   \[
   f_{\beta}(a)=\frac1\beta\log\!\Bigl(\sum_k e^{\beta\,(a*a)_k}\Bigr),
   \]

   and Adam is run on the parametrisation `a = softmax(z)`.  
   The temperature `β` is increased from a modest value (`4`) up to `5000`
   (continuation).

5. **Optional symmetrisation** – the problem is invariant under reversal,
   therefore the current iterate is mirrored and projected back onto the simplex.

6. **Final stochastic hill‑climb** – a brief simulated‑annealing walk
   (swap or mass‑transfer moves) attempts to escape any remaining local plateau.

All phases respect the global deadline; intermediate improvements are printed.
The final list of heights is returned and evaluated with the supplied `evaluate_sequence`.

#!/usr/bin/env python
"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(convolve(a,a)) / (sum(a)²)

by a large‑scale search (n≈16 k).  The algorithm consists of:

1.  A quick sweep over several analytically‑motivated families
    (symmetric product, exponential tails, power‑law tails, mixed uniform)
    to obtain a good initial shape.

2.  Exponentiated‑gradient (mirror descent) on the exact max‑convolution
    objective (sub‑gradient of the active entry).  The step size decays as
    ηₜ = η₀ / √t.

3.  Adam optimisation of the smooth‑max surrogate
    f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k) with a continuation
    schedule β∈[4, 5000].

4.  Optional symmetrisation (a←½(a+a⁻¹)).

5.  A short simulated‑annealing hill‑climb for final polishing.

All phases stop as soon as the overall 1000 s budget (minus a small safety
margin) is exhausted.  The returned list is ready for ``evaluate_sequence``.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two that is ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution using the real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the evaluator (with sum(a)=1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Parametric families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑½‑i))^{-α}  (α=0.5 → discrete arcsine)."""
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{-β}   (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """
    Convex mixture of uniform and exponential‑tail shapes:
    a = λ·Uniform + (1‑λ)·exp_u(gamma).
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random monotone decreasing vector (used for diversification)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑tail start."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 800_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Sub‑gradient is taken at a (any) maximising index k.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of the active entry (k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]       # diagonal appears only once

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved → {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate f_β(a) with a continuation schedule
    for β.  The optimisation variable is the log‑domain vector z with
    a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (log‑linear)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable soft‑max over the β‑scaled conv values
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a :  ∂f/∂a = 2 * (w ⋆ a_rev)  (restricted to length n)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # chain rule through softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot_ag)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved → {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Very simple simulated‑annealing walk on the simplex.
    Moves are either a swap of two entries or a fractional mass transfer.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4          # initial temperature
    T = T0
    step_scale = 0.12                # typical transferred fraction

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    # ------------------------------------------------------------------
    # 8.  Final simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        print(f"[{time.time():.1f}] SA phase (≈{remaining:.1f}s)")
        a_sa, sc_sa = hill_climb_sa(a, time.time() + remaining,
                                    max_steps=int(remaining * 2000))
        if sc_sa < best_score:
            best_score = sc_sa
            a = a_sa
            print(f"[{time.time():.1f}] SA improved best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final evaluation with the supplied `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return final_seq
