def gradient_max_conv(a):
   n = a.size
   conv = convolution_fft(a, a)
   k = np.argmax(conv)
   # gradient: g_i = 2 * a_{k - i} if index within range
   g = np.zeros_like(a)
   idx = np.arange(n)
   j = k - idx
   mask = (j >= 0) & (j < n)
   g[mask] = 2.0 * a[j[mask]]
   return g, conv.max()

"""
Propose a high‑quality step‑function (non‑negative sequence) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

where conv is the aperiodic convolution.  The search proceeds as follows:

1. Choose a fairly large support size n (default 8192, or a longer
   fallback if the harness supplies one).
2. Build a pool of analytic starters (arcsine‐type, power‑law,
   exponential, random monotone) and keep the best according to the
   exact objective (sum normalised to 1).
3. Refine the best starter with three complementary optimisation phases,
   all respecting the remaining wall‑clock budget:
   * Exponentiated‑gradient (mirror descent) on the *exact* max‑convolution.
   * Adam optimiser on a smooth‑max surrogate  fβ(a)= (1/β)·log Σ exp(β·(a*a)).
   * Projected sub‑gradient descent on the true max‑convolution.
4. Symmetrise the candidate (the problem is invariant under reversal) and
   perform a short simulated‑annealing hill‑climb for any remaining
   improvements.
5. Return the final sequence as a plain list; its sum is 1, so the
   evaluator’s scaling cancels out.

All helper functions are top‑level, there are no closures or lambdas,
and the implementation stays well within the 1000 s budget on a
single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised (sum = 1) sequence:
    2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (starting points)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑½‑i))^{-α} (arcsine is α=0.5)."""
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{-β} (mass concentrates on the left)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing shape (non‑increasing from left to right)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the active max‑convolution entry
# ----------------------------------------------------------------------


def max_conv_gradient(a: np.ndarray):
    """
    Returns (g, max_val) where g_i = 2·a_{k−i} for the index k where
    (a*a)_k attains its maximum.  This is the exact sub‑gradient of the
    max‑convolution functional.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    k = int(np.argmax(conv))
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    return g, float(conv[k])


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 1_000_000,
                eta0: float = 1.0):
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Step‑size η_t = η0 / √t.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        g, _ = max_conv_gradient(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a /= s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG iter {t} → {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 4.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.02):
    """
    Adam minimisation of the smooth‑max surrogate

        fβ(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    β is increased log‑linearly from β_start to β_end.
    The parametrisation a = softmax(z) guarantees feasibility.
    """
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise logits

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule β in log‑space
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights over convolution entries
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed max w.r.t. a:
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # chain rule through the softmax parametrisation
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam iter {it} → {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.1):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Simple and cheap; we can run many iterations.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        g, _ = max_conv_gradient(a)
        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad iter {t} → {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000):
    """
    Very simple simulated‑annealing walk on the simplex.
    Occasionally accepts uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided `evaluate_sequence` function.  The routine respects the
    global time budget (≈ budget_s seconds) and returns a plain Python
    list ready for evaluation.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # small safety margin

    # ------------------------------------------------------------------
    # 0.  Load a fallback candidate if the harness supplies one.
    # ------------------------------------------------------------------
    fallback_seq = None
    try:
        fallback_seq = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback_seq)
        print(f"[{time.time():.1f}] fallback score = {fb_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose the problem size.
    # ------------------------------------------------------------------
    base_n = 8192
    if fallback_seq is not None and len(fallback_seq) > base_n:
        n = len(fallback_seq)
    else:
        n = base_n
    print(f"[{time.time():.1f}] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Build a pool of analytic starters and keep the best.
    # ------------------------------------------------------------------
    candidates = []

    # arcsine‑type sweep
    for alpha in np.linspace(0.3, 0.9, 13):
        candidates.append(symmetric_product_family(n, alpha))

    # left‑heavy power law
    for beta in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(left_heavy_power_family(n, beta))

    # exponential decays
    for gamma in [0.4, 0.8, 1.2, 2.0]:
        candidates.append(exp_u_family(n, gamma))

    # a few random monotone starts
    for _ in range(3):
        candidates.append(random_monotone_start(n))

    # include fallback if available (resampled if lengths differ)
    if fallback_seq is not None:
        fb = np.array(fallback_seq, dtype=np.float64)
        if fb.size != n:
            # simple linear interpolation to the target size
            fb = np.interp(np.linspace(0, fb.size - 1, n),
                           np.arange(fb.size), fb)
        fb = np.maximum(fb, 0.0)
        fb = fb / fb.sum()
        candidates.append(fb)

    best_a = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates, 1):
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
        print(f"[{time.time():.1f}] init #{idx} score = {sc:.6f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3.  Time allocation for the remaining phases.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        final = best_a.tolist()
        print(f"[{time.time():.1f}] time exhausted early – returning best")
        return final

    eg_time   = remaining * 0.30
    adam_time = remaining * 0.25
    subg_time = remaining * 0.30
    sa_time   = remaining - (eg_time + adam_time + subg_time)

    # ------------------------------------------------------------------
    # 4.  Exponentiated‑gradient (mirror descent).
    # ------------------------------------------------------------------
    if time.time() + eg_time < deadline:
        print(f"[{time.time():.1f}] EG phase (~{eg_time:.1f}s)")
        a_eg, sc_eg = eg_optimize(best_a,
                                  time.time() + eg_time,
                                  max_iters=2_000_000,
                                  eta0=1.0)
        if sc_eg < best_score:
            best_score = sc_eg
            best_a = a_eg
            print(f"[{time.time():.1f}] EG improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Adam optimisation of the smooth‑max surrogate.
    # ------------------------------------------------------------------
    if time.time() + adam_time < deadline:
        print(f"[{time.time():.1f}] Adam phase (~{adam_time:.1f}s)")
        a_adam, sc_adam = adam_smooth_optimize(best_a,
                                                time.time() + adam_time,
                                                max_iters=500_000,
                                                beta_start=4.0,
                                                beta_end=12000.0,
                                                lr=0.02)
        if sc_adam < best_score:
            best_score = sc_adam
            best_a = a_adam
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Projected sub‑gradient descent (many cheap steps).
    # ------------------------------------------------------------------
    if time.time() + subg_time < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient phase (~{subg_time:.1f}s)")
        a_sub, sc_sub = subgrad_optimize(best_a,
                                         time.time() + subg_time,
                                         max_iters=6_000_000,
                                         step0=0.12)
        if sc_sub < best_score:
            best_score = sc_sub
            best_a = a_sub
            print(f"[{time.time():.1f}] Sub‑gradient improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Optional symmetrisation (the problem is invariant under reversal).
    # ------------------------------------------------------------------
    a_sym = (best_a + best_a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        best_a = a_sym
        print(f"[{time.time():.1f}] Symmetrisation improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Final simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    if time.time() + sa_time < deadline:
        print(f"[{time.time():.1f}] SA hill‑climb (~{sa_time:.1f}s)")
        a_sa, sc_sa = hill_climb_sa(best_a,
                                    time.time() + sa_time,
                                    max_steps=2_000_000)
        if sc_sa < best_score:
            best_score = sc_sa
            best_a = a_sa
            print(f"[{time.time():.1f}] SA improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final evaluation with the provided harness.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return final_seq
