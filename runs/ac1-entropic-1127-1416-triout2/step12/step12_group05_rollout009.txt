def triangular_center_family(n, width_frac):
    # width_frac from 0 to 1, defines half-width of triangle
    half_width = int(max(1, width_frac * n / 2))
    i = np.arange(n)

"""
Propose an improved step‑function using an expanded family search,
mirror‑descent on the simplex (exponentiated‑gradient) and a final
simulated‑annealing climb.

The algorithm
--------------
0. Optional fallback to the previously best construction (height_sequence_1).
1. Work with a relatively fine discretisation (n = 8192).
2. Systematically scan several analytic families – symmetric centre‑heavy,
   Gaussian, triangular, exponential decay (both sides) and the earlier
   product / power / beta families – on a modest grid; keep the best
   candidate.
3. Perform a dense random search (≈8000 evaluations) over the same
   families plus mixtures; allow random zero‑padding shifts that break
   symmetry.
4. Refine the currently best one‑parameter family with a golden‑section
   line‑search over a narrow interval.
5. Run projected exponentiated‑gradient descent (mirror descent w.r.t.
   the KL‑divergence) on the exact max‑convolution objective.  The update
   uses the sub‑gradient at the index where the convolution attains its
   maximum and a diminishing stepsize ηₜ = η₀/√t.
6. Finish with a longer simulated‑annealing hill‑climb (higher initial
   temperature, slower cooling) that moves mass between entries.
7. Return the best sequence (as a plain list) after a final call to
   evaluate_sequence (which applies all security checks).

All steps respect the supplied wall‑clock budget (default 1000 s).  The
search explores new centre‑heavy families that were not used in the
previous implementation, and uses a KL‑mirror‑descent optimiser that
often finds lower peaks than plain Euclidean sub‑gradient descent.
"""

# Core utilities (FFT convolution, scoring, projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s

# Analytic families

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def symmetric_center_power_family(n: int, alpha: float) -> np.ndarray:
    """Centre‑heavy: a_i ∝ [(i+½)(n‑½‑i)]^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    # guard against overflow/underflow
    expo = np.clip(alpha * logprod, -700, 700)
    w = np.exp(expo)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def right_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (n‑½‑i)^{‑β} (mirror of left‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return w / w.sum()


def gaussian_center_family(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred on the middle of the interval."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def triangular_center_family(n: int, width_frac: float) -> np.ndarray:
    """
    Symmetric triangular bump.
    width_frac ∈ (0,1] controls half‑width relative to n.
    """
    half_width = int(max(1, width_frac * n / 2.0))
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center)
    w = np.maximum(0.0, (half_width - dist) / half_width)
    return w / w.sum()


def exp_decay_family(n: int, gamma: float, left: bool = True) -> np.ndarray:
    """Exponential decay from one side (U‑shaped if left=False)."""
    i = np.arange(n, dtype=np.float64)
    if left:
        w = np.exp(-gamma * i)
    else:
        w = np.exp(-gamma * (n - 1 - i))
    return w / w.sum()


def mixed_uniform_gaussian_family(n: int, lam: float, sigma: float) -> np.ndarray:
    """Convex mixture of uniform and centred Gaussian."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    gaussian = gaussian_center_family(n, sigma)
    w = lam * uniform + (1.0 - lam) * gaussian
    return w / w.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    # avoid overflow for extreme p,q
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / s
    return w

def exp_gradient_descent(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using exponentiated gradients.
    Uses the exact sub‑gradient of the max‑convolution at each step.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # exponentiated‑gradient step with diminishing stepsize
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        # re‑normalise to the simplex (KL‑projection is simply scaling)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # occasional true‑objective evaluation
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] exp‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[{time.time():.1f}] exp‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 12_000_000,
                  temp_factor: float = 1.2e-3,
                  step_scale_init: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # higher initial temperature proportional to current best score
    T0 = cur_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        # two possible moves: swap or transfer a fraction of mass
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 12) -> Tuple[float, float]:
    """Golden‑section search for the minimum of func on [lo,hi]."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

    # 6. Final simulated‑annealing hill‑climb (high temperature)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=12_000_000,
            temp_factor=1.2e-3,
            step_scale_init=0.25,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Fallback to a safe uniform vector if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
