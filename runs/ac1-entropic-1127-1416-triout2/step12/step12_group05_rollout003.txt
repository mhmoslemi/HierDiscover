"""
Evolutionary block‑wise optimisation for the max‑convolution objective.
The algorithm works on a coarse block representation of the step
function, performs a (μ,λ) evolution‑strategy on the simplex (log‑space
softmax), and finishes with a projected sub‑gradient refinement.
It is designed to be quick enough to run within the 1000 s wall‑clock
limit and to explore shapes that are not covered by the analytic
families used previously.
"""
import time
import math
import numpy as np
from typing import List, Tuple

def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    return np.fft.irfft(C, nfft)[:m]

def compute_score(a: np.ndarray) -> float:
    s = np.sum(a)
    if s < 1e-12:
        return float('inf')
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto probability simplex (non‑negative, sums to 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def block_to_vector(blocks: np.ndarray, repeats: int) -> np.ndarray:
    """Expand a block representation to a full step‑function vector."""
    return np.repeat(blocks, repeats)

def upsample_sequence(seq: np.ndarray, n_target: int) -> np.ndarray:
    """Linearly up‑sample a sequence to length n_target and renormalise."""
    if seq.size == n_target:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.size)
    x_new = np.linspace(0.0, 1.0, n_target)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total <= 0.0:
        new_seq = np.full_like(new_seq, 1.0 / n_target)
    else:
        new_seq /= total
    return new_seq

def block_es_optimize(start_blocks: np.ndarray,
                      repeats: int,
                      deadline: float,
                      lam: int = 20,
                      sigma_init: float = 0.3,
                      sigma_min: float = 1e-5,
                      sigma_max: float = 2.0) -> Tuple[np.ndarray, float]:
    """Simple (1,λ) evolution strategy on the simplex (log‑softmax params)."""
    K = start_blocks.size
    # log‑space centre
    log_mu = np.log(start_blocks + 1e-12)
    sigma = sigma_init
    best_blocks = start_blocks.copy()
    best_score = compute_score(block_to_vector(best_blocks, repeats))
    gen = 0

    while time.time() < deadline - 1.0:
        gen += 1
        # generate λ offspring
        cand_blocks = None
        cand_score = np.inf
        for _ in range(lam):
            # sample Gaussian perturbation in log‑space
            log_candidate = log_mu + sigma * np.random.randn(K)
            # stable soft‑max to obtain simplex‑feasible block vector
            exp_shift = np.exp(log_candidate - np.max(log_candidate))
            blocks = exp_shift / exp_shift.sum()
            a_vec = block_to_vector(blocks, repeats)
            sc = compute_score(a_vec)
            if sc < cand_score:
                cand_score = sc
                cand_blocks = blocks

        # decide on acceptance
        if cand_score < best_score:
            best_score = cand_score
            best_blocks = cand_blocks
            log_mu = np.log(best_blocks + 1e-12)
            sigma = min(sigma * 1.05, sigma_max)
            print(f"[{time.time():.1f}] ES gen {gen}: improved to {best_score:.6f}")
        else:
            sigma = max(sigma * 0.95, sigma_min)

    return best_blocks, best_score

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> List[float]:
    """

    ``evaluate_sequence``.  The routine builds a block‑wise
    representation, runs a simple evolution‑strategy on the simplex
    (log‑softmax parametrisation) and finishes with a projected
    sub‑gradient refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 0. Initialise a reasonable starting point.
    # ------------------------------------------------------------------
    n_target = 8192                     # length used for the optimisation
    # try to use the previously discovered sequence if it exists
    try:
        init_seq = np.array([float(x) for x in height_sequence_1], dtype=np.float64)
        init_seq = init_seq / init_seq.sum()
    except Exception:
        init_seq = np.full(256, 1.0 / 256, dtype=np.float64)   # fallback uniform

    # up‑sample / reshape to the optimisation length
    init_seq = upsample_sequence(init_seq, n_target)

    # choose a block size that divides n_target – 256 blocks → 32 items each
    K = 256
    repeats = n_target // K
    blocks = init_seq.reshape(K, repeats).mean(axis=1)

    # ------------------------------------------------------------------
    # 1. Block‑wise evolution‑strategy search.
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] starting block ES (n={n_target}, K={K})")
    best_blocks, best_block_score = block_es_optimize(
        start_blocks=blocks,
        repeats=repeats,
        deadline=deadline - 200.0,   # reserve time for final refinements
        lam=20,

"""
Evolutionary block‑wise optimiser for the max‑convolution objective.

The algorithm works on a coarse block representation of the step
function, runs a simple (1,λ) evolution‑strategy in log‑softmax
coordinates (all candidates stay on the probability simplex), and
finishes with a projected sub‑gradient refinement followed by a short
simulated‑annealing hill‑climb.  It is designed to stay well within the
1000 s wall‑clock budget while exploring shapes that are not covered by
the analytic families used previously.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)^2).
    a is assumed non‑negative (the sum is never zero for our uses).
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Block representation helpers
# ----------------------------------------------------------------------


def block_to_vector(blocks: np.ndarray, repeats: int) -> np.ndarray:
    """
    Expand a block representation (total mass per block) to a full
    step‑function vector.  The result always sums to 1.
    """
    # each block mass is spread uniformly over its `repeats` entries
    return np.repeat(blocks, repeats) / repeats


def upsample_sequence(seq: np.ndarray, n_target: int) -> np.ndarray:
    """Linearly up‑sample a sequence to length `n_target` and renormalise."""
    if seq.size == n_target:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.size)
    x_new = np.linspace(0.0, 1.0, n_target)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total <= 0.0:
        new_seq = np.full_like(new_seq, 1.0 / n_target)
    else:
        new_seq /= total
    return new_seq


# ----------------------------------------------------------------------
# Evolution‑Strategy on blocks
# ----------------------------------------------------------------------


def block_es_optimize(start_blocks: np.ndarray,
                      repeats: int,
                      deadline: float,
                      lam: int = 20,
                      sigma_init: float = 0.3,
                      sigma_min: float = 1e-5,
                      sigma_max: float = 2.0) -> Tuple[np.ndarray, float]:
    """
    Simple (1,λ) evolution‑strategy on the simplex.
    Parameters
    ----------
    start_blocks : np.ndarray
        Initial total mass per block (already sums to 1).
    repeats : int
        How many fine‑grained entries each block expands to.
    deadline : float
        Absolute time at which the optimisation must stop.
    lam : int
        Offspring size per generation.
    sigma_init : float
        Initial standard deviation of the Gaussian mutation (log‑space).
    Returns
    -------
    (best_blocks, best_score) : tuple
        The best block vector found and its corresponding objective value.
    """
    K = start_blocks.size
    log_mu = np.log(start_blocks + 1e-12)      # centre in log‑space
    sigma = sigma_init
    best_blocks = start_blocks.copy()
    best_score = compute_score(block_to_vector(best_blocks, repeats))
    gen = 0

    while time.time() < deadline - 1.0:
        gen += 1
        # generate λ offspring
        cand_blocks = None
        cand_score = np.inf
        for _ in range(lam):
            log_cand = log_mu + sigma * np.random.randn(K)
            # stable soft‑max → simplex
            exp_shift = np.exp(log_cand - np.max(log_cand))
            blocks = exp_shift / exp_shift.sum()
            a_vec = block_to_vector(blocks, repeats)
            sc = compute_score(a_vec)
            if sc < cand_score:
                cand_score = sc
                cand_blocks = blocks

        # accept if improvement, otherwise shrink step‑size
        if cand_score < best_score:
            best_score = cand_score
            best_blocks = cand_blocks
            log_mu = np.log(best_blocks + 1e-12)
            sigma = min(sigma * 1.05, sigma_max)
            print(f"[{time.time():.1f}] ES gen {gen}: improved to {best_score:.6f}")
        else:
            sigma = max(sigma * 0.95, sigma_min)

    return best_blocks, best_score


# ----------------------------------------------------------------------
# Fine‑grained optimisation helpers
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000,
                  temp_factor: float = 1e-3,
                  step_scale_init: float = 0.2) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
