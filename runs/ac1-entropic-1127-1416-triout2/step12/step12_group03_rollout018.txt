"""
Propose a step‑function (non‑negative sequence) that minimizes

        2·n·max(convolve(a,a)) / (sum(a)²)

under the simplex constraint sum(a)=1 (the ratio is scale‑invariant).

The search proceeds in several stages:

1.  Choose a reasonably large problem size (default n=8192) to bring the
    theoretical lower bound 2n/(2n‑1) close to 1.
2.  Generate a pool of starting points:
    * Analytic families (arcsine‑type, power‑law, exponential decay).
    * Random Dirichlet draws with varying concentration.
    * The optional fallback sequence supplied by the harness.
3.  Pick the best start (lowest exact score using FFT‑based convolution).
4.  **Adam on a smooth‑max surrogate** – minimise
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    while driving β from ≈ 0.5 to a large value (≈ 2·10⁴).  The parameters
    (learning‑rate, β‑schedule) are chosen for robustness on long vectors.
5.  **Projected sub‑gradient descent** on the true max‑convolution objective
    to polish the solution.
6.  A cheap symmetrisation test (the problem is invariant under reversal).
7.  A final short simulated‑annealing hill‑climb for any remaining local
    improvements.
8.  Return the best sequence found as a plain Python list.

All heavy work uses FFT‑based convolution (O(n log n)) and respects the
global 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a (assumed non‑negative).  The vector is *not*
    required to be normalised – the formula is scale‑invariant.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    maxb = float(conv.max())
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    return 2.0 * n * maxb / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    zmax = np.max(z)
    e = np.exp(z - zmax)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (starting points)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑½‑i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{-beta}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ · dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_dirichlet_start(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a random point on the simplex (Gamma/Dirichlet)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 0.5,
                         beta_end: float = 20_000.0,
                         lr: float = 0.025) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth‑max surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the softmax parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (geometric progression)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)

        # soft‑max over β‑scaled convolution values
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # chain rule through softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.08) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing hill‑climb on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    `evaluate_sequence` function.  The implementation respects the total
    wall‑clock budget and returns a plain list ready for evaluation.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from the harness
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose a problem size (large enough to approach the bound)
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    n = max(8192, fallback_len)       # target length
    print(f"[{time.time():.1f}] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Build a pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # analytic families (coarse scan)
    alphas = np.linspace(0.2, 1.5, 13)
    for a in alphas:
        candidates.append(symmetric_product_family(n, a))
    betas = np.linspace(0.3, 2.0, 7)
    for b in betas:
        candidates.append(left_heavy_power_family(n, b))
    gammas = np.linspace(0.0, 12.0, 6)
    for g in gammas:
        candidates.append(exp_u_family(n, g))

    # random Dirichlet draws
    for _ in range(6):
        conc = np.random.uniform(0.3, 2.0)
        candidates.append(random_dirichlet_start(n, conc))

    # incorporate fallback if present (padded/truncated)
    if best_seq is not None:
        fb_arr = np.array(best_seq, dtype=np.float64)
        if fb_arr.size < n:
            padded = np.concatenate([fb_arr, np.zeros(n - fb_arr.size)])
        else:
            padded = fb_arr[:n]
        if padded.sum() == 0.0:
            padded = np.full(n, 1.0 / n)
        else:
            padded = padded / padded.sum()
        candidates.append(padded)

    # ------------------------------------------------------------------
    # 3.  Pick the best start (exact score)
    # ------------------------------------------------------------------
    for a in candidates:
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] initial candidate improved to {best_score:.6f}")

    # Normalise the working vector
    a = np.array(best_seq, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 4.  Time allocation for the remaining phases
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_time = remaining * 0.55
    subg_time = remaining * 0.30
    sa_time = max(0.0, remaining - (adam_time + subg_time))

    # ------------------------------------------------------------------
    # 5.  Adam on the smooth‑max surrogate
    # ------------------------------------------------------------------
    if time.time() + adam_time < deadline:
        print(f"[{time.time():.1f}] Adam phase (≈{adam_time:.1f}s)")
        a_adam, sc_adam = adam_smooth_optimize(
            a,
            time.time() + adam_time,
            max_iters=500_000,
            beta_start=0.5,
            beta_end=20_000.0,
            lr=0.025,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            a = a_adam
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Projected sub‑gradient descent (cheaper per iteration)
    # ------------------------------------------------------------------
    if time.time() + subg_time < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient phase (≈{subg_time:.1f}s)")
        a_sub, sc_sub = subgrad_optimize(
            a,
            time.time() + subg_time,
            max_iters=3_000_000,
            step0=0.08,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            a = a_sub
            print(f"[{time.time():.1f}] Sub‑gradient improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Optional symmetrisation
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] Symmetrisation improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() + sa_time < deadline and sa_time > 5.0:
        print(f"[{time.time():.1f}] Hill‑climb (SA) phase (≈{sa_time:.1f}s)")
        a_sa, sc_sa = hill_climb_sa(
            a,
            time.time() + sa_time,
            max_steps=2_000_000,
        )
        if sc_sa < best_score:
            best_score = sc_sa
            a = a_sa
            print(f"[{time.time():.1f}] Hill‑climb improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final evaluation via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
