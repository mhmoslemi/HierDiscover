"""
search_step_function.py

Search for a non‑negative step‑function minimising ``evaluate_sequence``.
The algorithm proceeds as follows:

0.  If a previous solution (``height_sequence_1``) is available, try it first.
1.  Choose a relatively large support size (default ``n = 16384``).
2.  Scan several analytic families (product, power‑law, exponential‑U,
    mixed uniform+U, product+U, asymmetric product and a new “two‑block”
    family).  The best parameters of each family are remembered.
3.  Refine the most promising families with SciPy’s local optimiser
    (Nelder‑Mead or bounded golden‑section) when SciPy is present.
4.  Run a smooth‑max Adam optimiser (soft‑max parametrisation, β annealed
    from a modest to a very large value) while periodically symmetrising
    the vector and projecting onto the probability simplex.
5.  Perform a long projected sub‑gradient descent on the true
    ``max(conv)`` objective.
6.  Execute a few short random re‑starts of sub‑gradient descent from
    random sparse or U‑shaped starts.
7.  Apply an explicit symmetrisation test.
8.  Finish with a simulated‑annealing hill‑climb for final polishing.
9.  Return the best sequence found as a plain ``list[float]``.
All steps respect the ``budget_s`` wall‑clock limit (default 1000 s) and
periodically check the deadline.
"""

import time
import math

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) for a probability vector (∑a = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric product family:
        a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Guard against overflow for alpha=0 (logprod can be large)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def product_expU_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    """Product family multiplied by an exponential‑U factor."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod - gamma * dist)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def asymmetric_product_family(n: int, beta_left: float, beta_right: float) -> np.ndarray:
    """Two‑parameter product family: a_i ∝ (i+½)^{‑β₁}(n‑½‑i)^{‑β₂}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta_left * np.log(i + 0.5) - beta_right * np.log(n - 0.5 - i))
    w /= w.sum()
    return w


def two_block_family(n: int, r: float, p: float) -> np.ndarray:
    """
    Two‑block (U‑shaped) family.
    * r  ∈ (0, 0.5) fraction of indices belonging to each end block.
    * p  > 0   relative mass allocated to the end blocks vs the middle.
    The vector is piecewise constant on the end blocks and the middle.
    """
    if r <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if r >= 0.5:
        r = 0.5
    k = max(1, int(round(r * n)))          # number of entries per side
    end_len = 2 * k
    mid_len = n - end_len
    # total mass for ends vs middle
    mass_end = p / (p + 1.0)
    mass_mid = 1.0 / (p + 1.0)
    a = np.zeros(n, dtype=np.float64)
    if end_len > 0:
        a[:k] = mass_end / end_len
        a[-k:] = mass_end / end_len
    if mid_len > 0:
        a[k:n - k] = mass_mid / mid_len
    # numerical safety – re‑normalise
    a /= a.sum()
    return a

def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse start: ~5 % non‑zero entries, uniform mass on them."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sort a powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 20.0,
                         beta_end: float = 15000.0,
                         lr: float = 0.01,
                         sym_interval: int = 1500) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    start_a : np.ndarray
        Initial probability vector (sum = 1, non‑negative).
    deadline : float
        Unix timestamp after which optimisation must stop.
    max_iters : int
        Upper bound on the number of Adam iterations.
    beta_start / beta_end : float
        Annealing schedule for the smooth‑max temperature.
    lr : float
        Base learning rate.
    sym_interval : int
        Every ``sym_interval`` steps the iterate is symmetrised and projected.

    Returns
    -------
    best_a, best_score
        Best vector found (as a numpy array) and its exact score.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)          # work in log‑space (softmax)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear schedule for temperature in log‑space
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a (derived analytically)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score

# Projected sub‑gradient descent on the true max‑objective

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent for the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a decreasing temperature schedule.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Golden‑section optimisation (1‑D)
# ----------------------------------------------------------------------


def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 20) -> tuple[float, float]:
    """
    Minimises ``func`` on [a,b] using golden‑section search.
    Returns the best parameter and its function value.
    """
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    return (c, fc) if fc < fd else (d, fd)

    except Exception:
        scipy_available = False

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback to a previously found construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a working length
    # ------------------------------------------------------------------
    # The length determines the granularity of the shape.  A power‑of‑two
    # makes FFTs cheap.
    n = 16384
    print(f"[{time.time():.1f}] using support size n = {n}")

    # ------------------------------------------------------------------
    # 2. Coarse scan of analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse scan of analytic families")
    family_records = {}

    # uniform baseline
    vec = np.full(n, 1.0 / n, dtype=np.float64)
    sc = compute_score(vec)
    if sc < best_score:
        best_score = sc
        best_seq = vec.tolist()
        print(f"[{time.time():.1f}] new best (uniform) = {best_score:.6f}")

    # ---- Symmetric product family (α) ----
    best_alpha = None
    best_alpha_sc = float('inf')
    for alpha in np.linspace(0.0, 30.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.3f}) = {best_score:.6f}")
        if sc < best_alpha_sc:
            best_alpha_sc, best_alpha = sc, alpha
    family_records['product'] = (best_alpha, best_alpha_sc)

    # ---- Left‑heavy power law (β) ----
    best_beta = None
    best_beta_sc = float('inf')
    for beta in np.linspace(0.0, 30.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")
        if sc < best_beta_sc:
            best_beta_sc, best_beta = sc, beta
    family_records['leftpower'] = (best_beta, best_beta_sc)

    # ---- Exponential U‑shape (γ) ----
    best_gamma = None
    best_gamma_sc = float('inf')
    for gamma in np.linspace(0.0, 12.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")
        if sc < best_gamma_sc:
            best_gamma_sc, best_gamma = sc, gamma
    family_records['expU'] = (best_gamma, best_gamma_sc)

    # ---- Mixed uniform + U (λ,γ) ----
    best_mix = None
    best_mix_sc = float('inf')
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 12.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")
            if sc < best_mix_sc:
                best_mix_sc, best_mix = sc, (lam, gamma)
    family_records['mix'] = (best_mix, best_mix_sc)

    # ---- Asymmetric product (β₁,β₂) ----
    best_asym = None
    best_asym_sc = float('inf')
    beta_grid = np.linspace(0.0, 15.0, 9)
    for b1 in beta_grid:
        for b2 in beta_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = asymmetric_product_family(n, b1, b2)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (asym β1={b1:.2f},β2={b2:.2f}) = {best_score:.6f}")
            if sc < best_asym_sc:
                best_asym_sc, best_asym = sc, (b1, b2)
    family_records['asym'] = (best_asym, best_asym_sc)

    # ---- Product + Exponential U (α,γ) ----
    best_pe = None
    best_pe_sc = float('inf')
    a_grid = np.linspace(0.0, 30.0, 13)
    g_grid = np.linspace(0.0, 12.0, 7)
    for a in a_grid:
        for g in g_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = product_expU_family(n, a, g)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (prod‑exp α={a:.2f},γ={g:.2f}) = {best_score:.6f}")
            if sc < best_pe_sc:
                best_pe_sc, best_pe = sc, (a, g)
    family_records['prodexpU'] = (best_pe, best_pe_sc)

    # ---- Two‑block family (r,p) ----
    best_twob = None
    best_twob_sc = float('inf')
    r_vals = np.linspace(0.02, 0.30, 15)          # fraction of ends
    p_vals = np.logspace(-1, 1, 15)              # relative mass ratio
    for r in r_vals:
        for p in p_vals:
            if time.time() >= deadline - 30.0:
                break
            vec = two_block_family(n, r, p)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (two‑block r={r:.3f},p={p:.3f}) = {best_score:.6f}")
            if sc < best_twob_sc:
                best_twob_sc, best_twob = sc, (r, p)
    family_records['twoblock'] = (best_twob, best_twob_sc)

    # ------------------------------------------------------------------
    # 3. Local refinement of promising families (SciPy if available)
    # ------------------------------------------------------------------
    if scipy_available and time.time() < deadline - 25.0:
        print(f"[{time.time():.1f}] local refinement with SciPy")

        # product family 1‑D refinement
        if family_records.get('product')[0] is not None:
            a0 = family_records['product'][0]
            lo = max(0.0, a0 - 1.0)
            hi = min(30.0, a0 + 1.0)

            def score_alpha(a):
                return compute_score(symmetric_product_family(n, a))

            a_opt, a_sc = golden_section_optimize(score_alpha, lo, hi, max_iter=30)
            vec = symmetric_product_family(n, a_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] refined product α={a_opt:.6f} → {best_score:.6f}")

        # mixture refinement (2‑D)
        if family_records.get('mix')[0] is not None:
            lam0, gam0 = family_records['mix'][0]

            def score_mix(p):
                lam = np.clip(p[0], 0.0, 1.0)
                gam = np.clip(p[1], 0.0, 12.0)
                return compute_score(mixed_uniform_u_family(n, lam, gam))

            res = minimize(score_mix,
                           x0=np.array([lam0, gam0]),
                           method='Nelder-Mead',
                           options={'maxiter': 200, 'xatol': 1e-4, 'fatol': 1e-4})
            lam_opt, gam_opt = np.clip(res.x, [0.0, 0.0], [1.0, 12.0])
            vec = mixed_uniform_u_family(n, lam_opt, gam_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] refined mix (λ={lam_opt:.6f},γ={gam_opt:.6f}) → {best_score:.6f}")

        # product‑expU refinement (2‑D)
        if family_records.get('prodexpU')[0] is not None:
            a0, g0 = family_records['prodexpU'][0]

            def score_pe(p):
                a = np.clip(p[0], 0.0, 30.0)
                g = np.clip(p[1], 0.0, 12.0)
                return compute_score(product_expU_family(n, a, g))

            res = minimize(score_pe,
                           x0=np.array([a0, g0]),
                           method='Nelder-Mead',
                           options={'maxiter': 250, 'xatol': 1e-4, 'fatol': 1e-4})
            a_opt, g_opt = np.clip(res.x, [0.0, 0.0], [30.0, 12.0])
            vec = product_expU_family(n, a_opt, g_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] refined prod‑exp (α={a_opt:.6f},γ={g_opt:.6f}) → {best_score:.6f}")

        # two‑block refinement (2‑D): we use a log‑transform for p to keep positivity
        if family_records.get('twoblock')[0] is not None:
            r0, p0 = family_records['twoblock'][0]

            def score_twob(x):
                r = np.clip(x[0], 0.001, 0.499)        # keep within (0,0.5)
                logp = x[1]
                p = math.exp(logp)
                return compute_score(two_block_family(n, r, p))

            # initial log‑p
            x0 = np.array([r0, math.log(max(p0, 1e-6))])
            res = minimize(score_twob,
                           x0=x0,
                           method='Nelder-Mead',
                           options={'maxiter': 200, 'xatol': 1e-4, 'fatol': 1e-4})
            r_opt, logp_opt = res.x
            p_opt = math.exp(logp_opt)
            vec = two_block_family(n, r_opt, p_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] refined two‑block (r={r_opt:.6f},p={p_opt:.6f}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. First Adam (smooth‑max) refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] first Adam refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 150.0,
            max_iters=500_000,
            beta_start=20.0,
            beta_end=15000.0,
            lr=0.012,
            sym_interval=1500,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after first Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 130.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 50.0,
            max_iters=2_500_000,
            step0=0.08,
        )
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑gradient best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Additional random‑restart sub‑gradient passes (short)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        num_restarts = 3
        print(f"[{time.time():.1f}] random‑restart sub‑gradient passes ({num_restarts} restarts)")
        for r_id in range(num_restarts):
            if time.time() >= deadline - 5.0:
                break
            # pick a random start – either sparse or U‑shaped
            if np.random.rand() < 0.5:
                a0 = random_sparse_start(n)
            else:
                a0 = random_u_start(n)
            a_tmp, sc_tmp = subgrad_optimize(
                a0,
                deadline - 5.0,
                max_iters=300_000,
                step0=0.08,
            )
            if sc_tmp < best_score:
                best_score, best_seq = sc_tmp, a_tmp.tolist()
                print(f"[{time.time():.1f}] restart {r_id+1} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Second Adam pass (higher β, tighter symmetry)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] second Adam pass")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = adam_smooth_optimize(
            a_start,
            deadline - 30.0,
            max_iters=300_000,
            beta_start=50.0,
            beta_end=20000.0,
            lr=0.008,
            sym_interval=1000,
        )
        if sc_opt2 < best_score:
            best_score, best_seq = sc_opt2, a_opt2.tolist()
            print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Explicit symmetrisation test
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score, best_seq = sc_sym, a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Final hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=3_000_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Fallback – simple uniform if nothing else succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final verification via the official wrapper (includes safety checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
