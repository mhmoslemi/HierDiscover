        # sample without replacement
        indices = np.random.choice(n, s, replace=False)
        # create binary support
        v = np.zeros(n, dtype=np.float64)

        # conv of v (ordered pairs)
        conv_counts = full_convolution(v)  # using same FFT function but maybe we need exact integer counts; but using float fine.
        max_counts = conv_counts.max()
        # compute objective: 2*n*max_counts/(s*s)
        score = 2.0 * n * max_counts / (s * s)
        if score < best_val:
            # construct a sequence with uniform weight across support
            seq = np.zeros(n, dtype=np.float64)
            seq[indices] = 1.0 / s
            best_val = score
            best_seq = seq.tolist()
            print(f"[random subset] new best score {best_val:.6f} with size {s}")

    # Stage 2: local search improvement for best_seq if not None
    if best_seq is not None:
        # cast to array
        best_arr = np.array(best_seq)
        # get support
        support = np.where(best_arr > 0)[0].tolist()
        # Start local search: swaps.
        local_deadline = deadline - 60
        while time.time() < local_deadline:
            s = len(support)
            # choose i in support, j not in support
            i = np.random.choice(support)
            j = np.random.randint(n)
            while best_arr[j] > 0:  # ensure j not in support
                j = np.random.randint(n)
            # propose new support
            new_support = support.copy()
            idx_i = new_support.index(i)
            new_support[idx_i] = j
            # compute new score
            v = np.zeros(n, dtype=np.float64)
            v[new_support] = 1.0
            conv_counts = full_convolution(v)
            max_counts = conv_counts.max()
            new_score = 2.0 * n * max_counts / (s * s)
            # accept if improvement or maybe probabilistic?
            if new_score < best_val * (1 + 1e-4):  # slight tolerance? Actually we want improvement.
                # accept move
                support = new_support
                best_arr = np.zeros_like(best_arr)
                best_arr[new_support] = 1.0 / s
                best_val = new_score
                best_seq = best_arr.tolist()
                print(f"[local swap] improved score {best_val:.6f}")

    # Stage 3: EG polish on best_seq
    if best_seq is not None:
        # run EG on this candidate for remaining time
        polish_deadline = deadline
        refined = eg_polish(np.array(best_seq, dtype=np.float64), polish_deadline)

            best_val=refined_score
            best_seq=refined.tolist()

    # Stage 4: fallback to uniform if none
    if best_seq is None:

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinal reported score = {final_score:.6f}")
    return best_seq

def convolve_vectors(u, v):
    # real FFT of appropriate length.
    m = len(u) + len(v) - 1
    n_fft = 1 << ((m-1).bit_length())
    U = np.fft.rfft(u, n=n_fft)

    C = U * V
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def smooth_max_objective(a, beta):
    conv = full_convolution(a)
    max_val = conv.max()
    # compute smooth approximation value
    # use log-sum-exp
    # shift by max to avoid overflow
    shifted = conv - max_val
    expbeta = np.exp(beta * shifted)
    sumexp = expbeta.sum()
    sm_val = max_val + np.log(sumexp) / beta
    # compute gradient as described: w = exp(beta * conv) / sumexp
    w = np.exp(beta * conv) / np.exp(beta * max_val) / sumexp = expbeta / sumexp
    # compute gradient g = 2 * cross-correlation(a, w)
    w_rev = w[::-1]
    d = convolve_vectors(a, w_rev)
    g = 2.0 * d[(2*n-2) - np.arange(n)]
    return sm_val, g

def project_simplex(v):
    # returns projection onto simplex sum(v)=1, v>=0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    # Implements Duchi et al.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho

    if rho_arr.size == 0:

        theta = 0.0

def adam_optimize(a0, deadline, eta=0.1, beta1=0.9, beta2=0.999,
                  eps=1e-8, beta_start=0.5, beta_max=50.0,
                  beta_factor=1.001):
    a = a0.copy()
    n = a.shape[0]
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = compute_score(a)
    # initial beta
    beta = beta_start
    while time.time() < deadline:
        t += 1
        # gradually increase beta.
        beta = min(beta_max, beta * beta_factor) if t > 1 else beta_start
        # compute smooth max and gradient
        smooth_val, g = smooth_max_and_grad(a, beta)

        # Adam update
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * (g * g)

"""
Smooth‐max + Adam optimiser for ``evaluate_sequence``.

The algorithm repeatedly draws random initial vectors (Dirichlet or sparse
uniform on a random support) and runs a projected Adam optimiser that
minimises a smooth approximation of the Chebyshev norm
    f(a) = max_k (a*a)_k
by the log‑sum‑exp trick.  The gradient is obtained analytically via a
cross‑correlation computed with FFTs, and after each Adam step the
vector is projected onto the probability simplex.  The best sequence
found before the deadline is returned.

The optimiser works entirely in the simplex (non‑negative, sum = 1) and
never calls any external optimisation library, so it complies with the
harness requirements.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.  ``a`` must be a 1‑D array.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def convolve_vectors(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Length of the result is len(u)+len(v)‑1.
    """
    m = u.shape[0] + v.shape[0] - 1
    n_fft = 1 << ((m - 1).bit_length())
    U = np.fft.rfft(u, n=n_fft)
    V = np.fft.rfft(v, n=n_fft)
    C = U * V
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Exact objective (used for bookkeeping and final report)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth approximation of max(conv) and its gradient
# ----------------------------------------------------------------------
def smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns (smooth_max, gradient) where

        smooth_max = (1/beta) * log( sum_k exp(beta * (a*a)_k) ).

    The gradient of the smooth max w.r.t. ``a`` is
        g_i = 2 * sum_j a_j * w_{i+j},
    where w_k = exp(beta * conv_k) / sum_l exp(beta * conv_l).
    """
    n = a.shape[0]

    # linear convolution a*a
    conv = full_convolution(a)           # length 2n-1
    max_val = conv.max()

    # log‑sum‑exp (shifted for stability)
    shifted = conv - max_val
    exp_beta = np.exp(beta * shifted)
    sumexp = exp_beta.sum()
    smooth_val = max_val + np.log(sumexp) / beta

    # softmax weights
    w = exp_beta / sumexp                # length 2n-1, sums to 1

    # gradient via cross‑correlation g_i = 2 * Σ_j a_j w_{i+j}
    w_rev = w[::-1]
    d = convolve_vectors(a, w_rev)       # length 3n‑2
    # pick the slice (2n‑2)‑i for i = 0…n‑1
    g = 2.0 * d[(2 * n - 2) - np.arange(n)]

    return smooth_val, g


# ----------------------------------------------------------------------
# Adam optimiser on the simplex with an increasing smoothing parameter
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  eta: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  beta_start: float = 0.5,
                  beta_max: float = 60.0,
                  beta_factor: float = 1.001) -> np.ndarray:
    """
    Projected Adam minimiser for the smooth max objective.
    ``a0`` must be a probability vector (non‑negative, sum = 1).
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    beta = beta_start

    while time.time() < deadline:
        t += 1
        # increase smoothing slowly
        beta = min(beta_max, beta * beta_factor) if t > 1 else beta_start

        # smooth max and its gradient
        _, grad = smooth_max_and_grad(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adam step
        a -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # back to the simplex
        a = project_simplex(a)

        # exact objective (cheap, uses FFT)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter {t:6d}  beta {beta:6.3f}  best {best_val:.6f}")

    return best_a

# ----------------------------------------------------------------------

    """

    ``evaluate_sequence`` within the allocated time budget.
    The method mixes random Dirichlet starts, occasional sparse starts,
    and a projected Adam optimiser that minimises a smooth approximation
    of the max‑convolution.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    n = 2048
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Try the previously stored solution if present
    # ------------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            val = evaluate_sequence(fallback)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = [float(x) for x in fallback]
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Random restarts with Adam optimisation
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:          # reserve ~30 s for final polish
        # ----- initialise a random start --------------------------------
        # 30 % of the time we use a sparse uniform start to encourage
        # very sparse solutions (which are often beneficial for max‑conv).
        if np.random.rand() < 0.3:
            s = np.random.randint(45, 120)      # support size
            idx = np.random.choice(n, s, replace=False)
            a0 = np.zeros(n, dtype=float)

            conc = 10 ** np.random.uniform(-1.0, 0.7)   # roughly 0.1 … 5
            a0 = np.random.dirichlet(np.full(n, conc))

        allocated = min(18.0,
                        deadline - time.time() - 30.0)  # never consume the reserve
        sub_deadline = time.time() + allocated
        a_opt = adam_optimize(a0, sub_deadline)

        # ----- update the global best ------------------------------------
        cur_val = compute_score(a_opt)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_opt.tolist()
            print(f"[restart] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Final refinement of the best candidate (if any)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_start = np.array(best_seq, dtype=float)
        a_refined = adam_optimize(a_start, deadline)
        final_val = compute_score(a_refined)
        print(f"[final‑adam] refined score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()

    # ------------------------------------------------------------------
    # 3. Fallback to a uniform distribution if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 4. Report using the official evaluation function
    # ------------------------------------------------------------------
    reported = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {reported:.6f}")
    return best_seq
