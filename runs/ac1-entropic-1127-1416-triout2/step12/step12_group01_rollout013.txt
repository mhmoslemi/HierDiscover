def smooth_optimize(a0, time_deadline, init_tau, min_tau, decay_rate):
    # a0 is initial probability vector.
    x = np.log(a0 + 1e-30)   # log-probabilities maybe plus small epsilon
    # Adam parameters: m, v, beta1, beta2, epsilon.
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    t = 0
    best_a = a0.copy()
    best_val = compute_score(a0)
    tau = init_tau
    while time.time() < time_deadline:
        t += 1
        # compute a = softmax(x)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)   # stable
        a = exp_x / exp_x.sum()
        conv = full_convolution(a)
        # smooth objective
        if tau > 0:
            # compute softmax of conv / tau
            # for numerical stability subtract max
            max_c = np.max(conv)
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            w = w / w_sum
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # gradient w.r.t a
        a_rev = a[::-1]
        # convolve a_rev with w using FFT
        grad_full = full_convolution(a_rev, w)   # we need function that can convolve two arrays
        # slice appropriate indices
        start = len(a) - 1
        grad_a = 2.0 * grad_full[start:start+len(a)]

        # map grad to x
        grad_x = (grad_a - np.dot(grad_a, a)) * a

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = 0.01 / (np.sqrt(t) + 1)   # or constant small lr?
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # optionally enforce lower bound on tau (anneal)
        tau = max(min_tau, tau * decay_rate)

        # compute exact objective for best tracking
        if t % 100 == 0:
            a_curr = a
            val = compute_score(a_curr)
            if val < best_val:
                best_val = val
                best_a = a_curr.copy()
                print(f"[smooth-opt] iter {t}, val {val:.6f}, tau {tau:.6e}")

    return best_a

    A = np.fft.rfft(a, n=n_fft)

    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]

def smooth_optimize(a0, deadline, init_lr=0.05, init_tau_factor=0.2, min_tau=1e-7, decay=0.9995):
    a = a0.copy()
    n = a.shape[0]
    # initialize x = log(a) (add epsilon)
    eps = 1e-30
    x = np.log(a + eps)
    # Adam parameters
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t = 0
    # initial tau
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())
    best = a.copy()
    best_val = compute_score(a)
    while time.time() < deadline:
        t += 1
        # recompute a = softmax(x)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)  # stable
        a = exp_x / exp_x.sum()
        # compute conv
        conv = full_convolution(a)
        max_c = conv.max()
        # compute softmax weights w over conv (smooth approximation)
        # at small tau (approach max), we might fallback to hard max for numeric stability:
        if tau < 1e-12:
            # w is one-hot at argmax
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # compute stable softmax
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient wrt a
        a_rev = a[::-1]
        grad_full = full_convolution_two(a_rev, w)   # size len(a_rev)+len(w)-1
        start = len(a) - 1
        grad_a = 2.0 * grad_full[start:start + n]    # slice length n

        # gradient wrt x (softmax transformation)
        grad_x = (grad_a - np.dot(grad_a, a)) * a

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = init_lr / (np.sqrt(t) + 1.0)  # schedule
        x = x - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # decay tau
        tau = max(min_tau, tau * decay)

        # occasional exact evaluation
        if t % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best = a.copy()
                print(f"[smooth_opt] iter {t}, val {val:.6f}, tau {tau:.2e}")

    return best

"""
Search for a non‑negative step‑function that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Global exploration** – sweep three classical families
   (power‑law, exponential, beta) **and a new Gaussian family**
   together with many random Dirichlet samples.  Keep the best
   few candidates.

2. **Smoothed mirror‑descent** – minimise a smooth version of the
   max‑convolution objective using an Adam optimiser on the
   *soft‑max* parametrisation a = softmax(x).  The smoothness
   parameter τ is gradually annealed to 0, turning the smooth
   objective into the true max.  This produces a much more
   accurate descent than the plain exponentiated‑gradient used
   before.

3. **Final annealing** – a short simulated‑annealing pass to escape
   possible tiny local plateaux.

The whole search respects the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – FFT‑based linear convolution (any two arrays)
# ----------------------------------------------------------------------


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of a and b (real FFT)."""
    m = a.size + b.size - 1
    n_fft = 1 << ((m - 1).bit_length())         # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (used for the objective)."""
    return _linear_convolution(a, a)


# ----------------------------------------------------------------------
# Exact objective (no clipping – all entries already non‑negative)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle, a_i ∝ exp(-½·((i‑µ)/σ)²)."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration (keeps up to `max_keep` best vectors)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Sweep the families and many Dirichlet draws."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s                     # project onto the simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # ---- 1) power‑law -------------------------------------------------
    for p in np.linspace(0.1, 2.0, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # ---- 2) exponential ------------------------------------------------
    for lam in np.linspace(0.0001, 0.05, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # ---- 3) beta -------------------------------------------------------
    al_vals = np.linspace(0.2, 5.0, 25)
    be_vals = np.linspace(0.2, 5.0, 25)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # ---- 4) Gaussian ----------------------------------------------------
    sigma_vals = np.linspace(n / 16.0, n / 2.0, 30)   # from narrow to wide
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # ---- 5) Dirichlet draws --------------------------------------------
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1200
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smoothed optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    init_lr: float = 0.12,
                    init_tau_factor: float = 10.0,
                    min_tau: float = 1e-8,
                    tau_decay: float = 0.9995) -> np.ndarray:
    """
    Minimise  max_k (a*a)_k  using a smooth approximation.
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise x = log a (adding a tiny epsilon for safety)
    eps = 1e-30
    x = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initial temperature τ (smoothness)
    conv = full_convolution(a)
    tau = max(min_tau, init_tau_factor * conv.max())

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max mapping ----------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution of the current a (exact, for the soft‑max) ----------
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- smooth max weights ----------
        if tau < 1e-12:                     # essentially hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # stable soft‑max over the convolution values
            w = np.exp((conv - max_c) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # ----- gradient w.r.t. a (g_i = 2·∑_k w_k·a_{k-i}) ----------
        a_rev = a[::-1]
        grad_full = _linear_convolution(a_rev, w)   # length = 3n‑2
        start = n - 1
        grad_a = 2.0 * grad_full[start:start + n]

        # ----- chain rule for the soft‑max (x‑space) ----------
        # dF/dx_i = a_i·(g_i - <g,a>)
        dot = np.dot(grad_a, a)
        grad_x = (grad_a - dot) * a

        # ----- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = init_lr / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal the temperature ----------
        tau = max(min_tau, tau * tau_decay)

        # ----- periodic exact evaluation ----------
        if it % 250 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[smooth_opt] iter {it:6d}  score {val:.6f}  τ {tau:.2e}")

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (optional final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 is a good trade‑off between expressivity
    # and speed.
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously saved solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb = [float(x) for x in fallback]
            val = evaluate_sequence(fb)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fb
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 180 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 300.0   # reserve the rest for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=8)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
                print(f"[baseline] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smoothed Adam optimisation with several restarts (≈ 700 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        # one pass starting from the current best
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_optimize(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth‑opt‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # further restarts while time permits
        while time.time() < polish_deadline:
            remaining = polish_deadline - time.time()
            sub_deadline = time.time() + min(30.0, remaining)
            # random Dirichlet start (vary concentration)
            conc = np.random.choice([0.1, 0.2, 0.4, 0.7, 1.0, 2.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_optimize(a_start, sub_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑opt‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report the official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
