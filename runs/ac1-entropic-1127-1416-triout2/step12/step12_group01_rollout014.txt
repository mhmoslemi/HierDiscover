def local_search(a, conv, deadline):
    n = len(a)
    best_a = a.copy()
    best_max = conv.max()
    cur_max = best_max
    T = 1e-3
    T_min = 1e-6
    cooling = 0.9995
    step_scale = 0.05
    step_decay = 0.9999
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # random pick i with mass
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        if delta < 1e-12:
            continue

        # backup
        conv_old = conv.copy()
        a_i_old, a_j_old = a[i], a[j]

        # apply -delta to i
        conv[i:i+n] -= 2.0 * delta * a
        idx2i = 2*i
        if idx2i < 2*n - 1:
            conv[idx2i] += delta*delta
        a[i] = a_i_old - delta

        # apply +delta to j
        conv[j:j+n] += 2.0 * delta * a
        idx2j = 2*j
        if idx2j < 2*n - 1:
            conv[idx2j] += delta*delta
        a[j] = a_j_old + delta

        new_max = conv.max()
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_max - cur_max) / max(T, 1e-15)):
                accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
            # adapt step_scale up
            step_scale = min(step_scale * 1.01, 0.5)
        else:
            # revert
            conv[:] = conv_old
            a[i] = a_i_old
            a[j] = a_j_old

        # annealing schedule
        T = max(T_min, T * cooling)
        step_scale = max(step_scale * step_decay, 1e-6)

        if iteration % 10000 == 0:
            # occasional jitter: multiply by random factor and renormalize
            jitter = np.random.randn(n) * (step_scale * 0.1)
            a = a * np.exp(jitter)
            a = a / a.sum()
            conv = full_convolution(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
            # reset step_scale maybe
            step_scale = max(step_scale, 0.01)

    return best_a, best_max

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1. Global exploration – a dense sweep of three parametric families
   (power‑law, exponential, beta) plus a large pool of random Dirichlet,
   normal‑square and log‑normal draws.  The best few candidates (by the exact
   objective 2·n·max(conv(a,a))) are kept.

2. Exponentiated‑gradient (EG) polishing – the same sub‑gradient method as
   in the reference implementation, but with many random restarts.

3. Incremental simulated‑annealing – a fast hill‑climber that updates the
   full convolution incrementally when moving a small amount of mass from
   one index to another.  This allows millions of cheap moves within the
   remaining time budget and usually pushes the score a little further
   below the EG optimum.

The whole search respects the 1000 s wall‑clock limit, keeping a safety
margin of a few seconds for the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return float(2.0 * n * full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric families for the global sweep
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape:
       a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration (baseline)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Sweep families and many random draws, keep the ``max_keep`` best.
    Returns a list of numpy arrays (the candidates)."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep
    for p in np.linspace(0.1, 2.0, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(0.0001, 0.05, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep – moderate resolution
    al_vals = np.linspace(0.2, 5.0, 25)
    be_vals = np.linspace(0.2, 5.0, 25)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) random Dirichlet draws
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 800
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            try_candidate(np.random.dirichlet(alpha_vec))

    # 5) random normal‑square draws
    for _ in range(2000):
        if time.time() >= deadline:
            break
        rs = np.random.randn(n) ** 2
        try_candidate(rs)

    # 6) random log‑normal draws
    for _ in range(2000):
        if time.time() >= deadline:
            break
        rs = np.exp(np.random.randn(n))
        try_candidate(rs)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (sub‑gradient descent on max‑conv)
# ----------------------------------------------------------------------


def eg_polish(a0: np.ndarray, deadline: float,
              eta0: float = 0.7, tol: float = 1e-10) -> np.ndarray:
    """Exponentiated‑gradient minimisation of  max_k (a*a)_k on the simplex."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * tol
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: g_i = 2·∑_{k∈active} a_{k-i}
        # implement with convolution of a and an indicator of active indices
        # generate an indicator vector of length 2·n‑1
        ind = np.zeros_like(conv)
        ind[active] = 1.0
        # shift the indicator to align with a (reverse order)
        # The sub‑gradient equals 2 * (a * ind[::-1])_{n-1 : n-1+n}
        g = 2.0 * np.convolve(a, ind[::-1], mode='full')[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % 500 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a


# ----------------------------------------------------------------------
# Incremental simulated‑annealing / hill‑climbing
# ----------------------------------------------------------------------


def incremental_anneal(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """Fast ancilla that updates the convolution incrementally while moving
    mass between positions.  Returns the best vector seen."""
    a = a_start.copy()
    n = a.shape[0]
    conv = full_convolution(a)

    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    # annealing schedule
    T = 1e-3
    T_min = 1e-6
    cooling = 0.9995
    step_scale = 0.5          # start with fairly large relative steps
    step_decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        # pick source index with some mass
        i = np.random.randint(n)
        if a[i] <= 1e-15:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]          # amount to move
        if delta <= 0:
            continue

        # backing up current state (cheap copy)
        conv_old = conv.copy()
        ai_old, aj_old = a[i], a[j]

        # ---- apply -delta to i -------------------------------------------------
        conv[i:i + n] -= 2.0 * delta * a               # use old a for shift
        idx2i = 2 * i
        if idx2i < 2 * n - 1:
            conv[idx2i] += delta * delta
        a[i] = ai_old - delta

        # ---- apply +delta to j -------------------------------------------------
        conv[j:j + n] += 2.0 * delta * a               # a now contains updated i
        idx2j = 2 * j
        if idx2j < 2 * n - 1:
            conv[idx2j] += delta * delta
        a[j] = aj_old + delta

        new_max = conv.max()
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_max - cur_max) / max(T, 1e-15)):
                accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
            # gently increase step magnitude when we are moving downhill
            step_scale = min(step_scale * 1.01, 0.7)
        else:
            # revert everything
            conv[:] = conv_old
            a[i] = ai_old
            a[j] = aj_old

        # schedule updates
        T = max(T_min, T * cooling)
        step_scale = max(step_scale * step_decay, 1e-6)

        # occasional global jitter to shake out of plateaus
        if it % 20000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.1)
            a = a * np.exp(jitter)
            a = a / a.sum()
            conv = full_convolution(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # problem size – 2048 gives a good trade‑off between resolution and speed
    n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Fallback from previously stored sequence (if any)
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈120 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 600.0   # reserve the rest for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=8)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Exponentiated‑gradient polishing (≈250 s)
    # -----------------------------------------------------------------
    eg_deadline = deadline - 300.0
    if best_seq is not None and time.time() < eg_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = eg_polish(a0, eg_deadline)
        val = compute_score(refined)
        print(f"[eg‑polish‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # extra random restarts while time permits
        while time.time() < eg_deadline:
            conc = np.random.choice([0.2, 0.5, 1.0, 2.0, 4.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = eg_polish(a_start, eg_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[eg‑polish‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Incremental simulated‑annealing polish (≈600 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final_a = incremental_anneal(a_start, anneal_deadline)
        val = compute_score(final_a)
        print(f"[incremental‑anneal] final candidate score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final_a.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback – uniform distribution if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Final official evaluation and report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
