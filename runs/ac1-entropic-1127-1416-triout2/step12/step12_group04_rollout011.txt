"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

using a multi‑phase search:

1.  Very wide analytic scan (product, power, symmetric power,
    Gaussian, exponential‑U, cosine windows and random starts) with a
    fine grid around the best parameters.
2.  Smooth optimisation of a *log‑softmax* parametrisation with an
    Adam optimiser and a gradually increasing “max‑softening’’ λ
    (soft‑max approximation of the max‑convolution).  This gives a
    high‑quality descent direction for the true objective.
3.  Exact exponentiated‑gradient (mirror‑descent) with the *correct*
    sub‑gradient of the max‑convolution (previous code used an
    approximation).
4.  A short stochastic hill‑climb (swap / mass‑transfer moves) as a
    final polishing step.

All phases respect the global wall‑clock budget (default 1000 s) and
the best sequence found is returned as a plain Python list ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the initial wide scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}  (U‑shaped, ends heavy)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else np.full(n, 1.0 / n)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-β}  (monotone, heavy left side)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else np.full(n, 1.0 / n)


def symmetric_power_family(n: int, beta: float) -> np.ndarray:
    """Symmetric heavy‑ends shape obtained by adding a left‑heavy and its reverse."""
    left = left_heavy_power_family(n, beta)
    w = left + left[::-1]
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else np.full(n, 1.0 / n)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """w_i ∝ exp(-γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else np.full(n, 1.0 / n)


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """w_i ∝ exp(-γ·(i‑mid)²)  (discrete Gaussian)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else np.full(n, 1.0 / n)


def window_family(n: int, window_type: str) -> np.ndarray:
    """Standard cosine windows (Hann, Hamming, Blackman)."""
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else np.full(n, 1.0 / n)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (heavy left side)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped (product‑like) start."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smooth optimisation (soft‑max of the convolution, Adam on logits)
# ----------------------------------------------------------------------


def smooth_optimize(start_a: np.ndarray,
                    start_time: float,
                    deadline: float,
                    max_iters: int = 20000,
                    lam_start: float = 0.5,
                    lam_end: float = 80.0,
                    lr0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
          F_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
    The logits are kept in the log‑softmax parametrisation to enforce
    the simplex constraint.
    """
    n = start_a.shape[0]
    # initialise logits (log‑softmax) from the starting probability vector
    logits = np.log(start_a + 1e-12)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current probabilities
        a = softmax(logits)

        # convolution a*a
        conv = convolution_fft(a, a)          # length 2n‑1

        # current λ (exponential increase)
        lam = lam_start * (lam_end / lam_start) ** (t / max_iters)

        # soft‑max weights over the convolution entries
        lc = lam * conv
        max_lc = np.max(lc)
        w = np.exp(lc - max_lc)               # non‑negative, sum‑stable
        prob = w / w.sum()                    # Σ_k prob_k = 1

        # ---- gradient w.r.t. a (exact for the smoothed objective) ----
        # sliding sum of prob over windows of length n
        cs = np.concatenate([[0.0], np.cumsum(prob)])      # length 2n
        window_sum = cs[n:] - cs[:-n]                       # length n

        # cross‑term h_i = Σ_j prob_{i+j} a_j  (convolution of prob with reversed a)
        h_full = convolution_fft(prob, a[::-1])            # length 3n‑2
        h = h_full[n - 1:n - 1 + n]                         # slice of length n

        # gradient of the smoothed objective w.r.t. a
        g_a = a * window_sum + h

        # transform gradient to the logit space (softmax Jacobian)
        avg = np.dot(a, g_a)                # scalar = Σ_i a_i g_i
        grad_logits = a * (g_a - avg)       # shape (n,)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (decrease like 1/√t, similar to EG)
        lr = lr0 / math.sqrt(t)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional true‑objective evaluation
        if t % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.1f}] smooth iter {t} → {best_score:.6f}")

    # final check
    final_a = softmax(logits)
    final_score = compute_score(final_a)
    if final_score < best_score:
        best_score = final_score
        best_a = final_a.copy()
    return best_a, best_score

# Exact exponentiated‑gradient (mirror descent) with correct sub‑gradient
# ----------------------------------------------------------------------


def eg_optimize_correct(start_a: np.ndarray,
                        start_time: float,
                        deadline: float,
                        max_iters: int = 180_000,
                        eta0: float = 0.55) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Uses the exact sub‑gradient:
        g_i = a_i + a_{k*‑i}   if   0 ≤ k*‑i < n,   else 0
    where k* is the index attaining the maximal convolution entry.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)

        # g_i = a_i + a_{k-i} for valid pairs
        g = np.zeros_like(a)
        g[mask] = a[idx[mask]] + a[j[mask]]

        # multiplicative update (exponentiated gradient)
        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.1f}] EG correct iter {t} → {best_score:.6f}")

    # final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray,
                  start_time: float,
                  deadline: float,
                  max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing style local search.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 2e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:           # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # project back onto simplex (simple scaling, works because only tiny perturbations)
        s = cand.sum()
        if s > 0:
            cand = cand / s
        else:
            cand = np.full_like(cand, 1.0 / n)

        cand_score = compute_score(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The function respects the global time budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so the harness does not cut us off abruptly
    deadline = start_time + budget_s - 3.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback (if the runner provides a known sequence)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}] fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose problem size (larger gives a bigger design space)
    # ------------------------------------------------------------------
    n = 4096
    print(f"[{time.time() - start_time:.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (normalised internally)
    # ------------------------------------------------------------------
    def evaluate_and_update(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Very wide analytic scan
    # ------------------------------------------------------------------
    # uniform baseline
    evaluate_and_update(np.full(n, 1.0 / n, dtype=np.float64))

    # product family – coarse scan, then fine around the best α
    best_alpha = None
    best_alpha_score = float("inf")
    alphas_coarse = np.linspace(0.0, 30.0, 61)  # step 0.5
    for alpha in alphas_coarse:
        if time.time() >= deadline - 250:
            break
        a = product_family(n, alpha)
        sc = compute_score(a)
        if sc < best_alpha_score:
            best_alpha_score = sc
            best_alpha = alpha
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] product α={alpha:.3f} → {sc:.6f}")

    # fine scan around best α (if any)
    if best_alpha is not None:
        alphas_fine = np.linspace(max(0.0, best_alpha - 2.0),
                                  best_alpha + 2.0, 41)  # step ≈0.1
        for alpha in alphas_fine:
            if time.time() >= deadline - 250:
                break
            evaluate_and_update(product_family(n, alpha))

    # symmetric power family (heavy ends, parameter β)
    betas = np.linspace(0.0, 30.0, 61)
    for beta in betas:
        if time.time() >= deadline - 250:
            break
        evaluate_and_update(symmetric_power_family(n, beta))

    # left‑heavy power (asymmetric) – still useful as a seed
    for beta in betas:
        if time.time() >= deadline - 250:
            break
        evaluate_and_update(left_heavy_power_family(n, beta))

    # exponential‑U family
    gammas = np.linspace(0.0, 20.0, 41)
    for gamma in gammas:
        if time.time() >= deadline - 250:
            break
        evaluate_and_update(exp_u_family(n, gamma))

    # Gaussian family
    gauss_gammas = np.logspace(-4, 2, 40)
    for gamma in gauss_gammas:
        if time.time() >= deadline - 250:
            break
        evaluate_and_update(gaussian_family(n, gamma))

    # cosine windows
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 250:
            break
        evaluate_and_update(window_family(n, win))

    # random starts to inject diversity
    for _ in range(200):
        if time.time() >= deadline - 250:
            break
        if np.random.rand() < 0.5:
            evaluate_and_update(random_monotone_start(n))
        else:
            evaluate_and_update(random_u_start(n))

    # ------------------------------------------------------------------
    # 3.  Ensure we have a valid starting vector (fallback to uniform)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.asarray(best_seq, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 4.  Smooth Adam optimisation (soft‑max surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print(f"[{time.time() - start_time:.1f}] starting smooth Adam optimisation")
        a_smooth, sc_smooth = smooth_optimize(
            a,
            start_time,
            deadline - 180.0,
            max_iters=25000,
            lam_start=0.5,
            lam_end=80.0,
            lr0=0.07,
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            a = a_smooth
            print(f"[{time.time() - start_time:.1f}] after smooth → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Exact exponentiated‑gradient (mirror descent) phase
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time() - start_time:.1f}] starting exact EG optimisation")
        a_eg, sc_eg = eg_optimize_correct(
            a,
            start_time,
            deadline - 50.0,
            max_iters=150_000,
            eta0=0.55,
        )
        if sc_eg < best_score:
            best_score = sc_eg
            a = a_eg
            print(f"[{time.time() - start_time:.1f}] after EG → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final short stochastic hill‑climb (simulated‑annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (SA)")
        a_sa, sc_sa = hill_climb_sa(
            a,
            start_time,
            deadline,
            max_steps=600_000,
        )
        if sc_sa < best_score:
            best_score = sc_sa
            a = a_sa
            print(f"[{time.time() - start_time:.1f}] after SA → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Return the best sequence found (as a plain list)
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished – reported score = {final_score:.6f}")
    return final_seq
