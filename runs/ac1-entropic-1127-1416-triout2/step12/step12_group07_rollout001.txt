def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT; length = 2*len(a)-1."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` for a normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that
          (A @ g)[k]  ==  (f * g)[k]   (full convolution)
    for all k = 0 … 2n‑2.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = lil_matrix((m, n), dtype=np.float64)

    # column j = f shifted down by j positions
    for j in range(n):
        start = j
        end = start + n
        A[start:end, j] = f  # slice assignment copies the whole f array
    return A.tocsr()


def solve_lp_for_g(f: np.ndarray, M: float, time_limit: float = 30.0) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   sum(g_i)
        subject to conv(f, g) <= M   (componentwise)
                  g_i >= 0
    Returns the raw solution (may not sum to 1).  Returns None on failure.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)               # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # minimise -sum(g)  -> maximise sum(g)

    bounds = [(0, None)] * n
    # the HiGHS solver accepts a simple time limit via options (seconds)
    # we use it to keep each LP cheap
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit, 'dual_feasibility_tolerance': 1e-9})
    except Exception as e:
        print("[LP] Linprog exception:", e)
        return None

    if not res.success:
        # The LP may be infeasible only if numeric issues arise – treat as failure
        print("[LP] Solver failed:", res.message)
        return None

    g = np.maximum(res.x, 0.0)   # cut tiny negatives
    return g


# ----------------------------------------------------------------------
# LP‑based refinement
# ----------------------------------------------------------------------
def lp_refine(initial_f: np.ndarray, deadline: float, max_outer_iter: int = 8) -> np.ndarray:
    """
    Repeatedly improve the vector using the LP reshaping described above.
    """
    f = initial_f.copy()
    best_f = f.copy()
    best_score = compute_score(f)

    outer = 0
    while time.time() < deadline and outer < max_outer_iter:
        outer += 1
        conv_f = full_convolution(f)
        M = conv_f.max()

        # Solve LP to obtain a more spread vector g
        g_raw = solve_lp_for_g(f, M, time_limit=15.0)
        if g_raw is None:
            print(f"[LP‑refine] Iter {outer}: LP failed, aborting.")
            break

        # Normalise g to the simplex (sum = 1)
        if g_raw.sum() == 0:
            print(f"[LP‑refine] Iter {outer}: g sums to zero, abort.")
            break
        g = g_raw / g_raw.sum()

        # Try a few convex combinations of f and g
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            cand = (1.0 - t) * f + t * g
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_f = sc, cand.copy()
                f = cand
                improved = True
                print(f"[LP‑refine] Iter {outer}, blend {t:.2f} → score {sc:.6f}")
                break
        if not improved:
            # Fall back to pure g if it is already better than f
            sc_g = compute_score(g)
            if sc_g < best_score:
                best_score, best_f = sc_g, g.copy()
                f = g
                print(f"[LP‑refine] Iter {outer}, pure g improves → score {sc_g:.6f}")
                continue
            # No improvement – apply a tiny random jitter to escape possible plateau
            jitter = np.random.randn(*f.shape) * 1e-4
            f = project_to_simplex(f + jitter)
            # recompute score (mostly unchanged) and continue
            print(f"[LP‑refine] Iter {outer}, no improvement – jitter applied.")
    return best_f


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (final fine‑tuning)
# ----------------------------------------------------------------------
def hill_climb(start: np.ndarray, deadline: float, max_iter: int = 400_000) -> np.ndarray:
    a = start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        if no_improve > 120_000:
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1
    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Obtain a fallback candidate if available
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. If no fallback, start from a simple uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2048
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)
        print(f"[uniform] official score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Convert to a NumPy array (already normalised to sum≈1)
    # ------------------------------------------------------------------
    n = len(best_seq)
    init_vec = np.array(best_seq, dtype=float)
    # Normalise explicitly to avoid tiny deviations introduced by evaluate_sequence
    init_vec = init_vec / init_vec.sum()

    # ------------------------------------------------------------------
    # 3. LP‑based iterative reshaping (takes the bulk of the time budget)
    # ------------------------------------------------------------------
    lp_deadline = deadline - 120.0          # reserve ~2 min for final polishing
    if time.time() < lp_deadline:
        print("[stage‑1] LP‑based refinement start")
        refined = lp_refine(init_vec, lp_deadline, max_outer_iter=8)
        val = compute_score(refined)
        print(f"[stage‑1] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climb for tiny adjustments
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if time.time() < hill_deadline:
        print("[stage‑2] hill‑climb polishing start")
        start_vec = np.array(best_seq, dtype=float)
        polished = hill_climb(start_vec, hill_deadline, max_iter=500_000)
        val = compute_score(polished)
        print(f"[stage‑2] final polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, polished.tolist()

    # ------------------------------------------------------------------
    # 5. Safety net – fall back to uniform if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2048
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6. Official evaluation (including the security wrappers)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported (official) score = {final_score:.6f}")
    return best_seq
