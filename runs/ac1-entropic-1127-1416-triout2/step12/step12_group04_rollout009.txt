def smoothed_gradient_descent(start_a, deadline, max_iters=200_000, step0=0.2, eps0=0.02, eps_decay=0.999):

    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smoothing parameter
        eps = max(eps0 * (eps_decay ** t), 1e-6)
        # compute softmax weights
        # but shift by max to avoid overflow
        w = np.exp((conv - maxc) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w_norm = w / w_sum

        # gradient
        # convolution of w_norm (size 2n-1) with reversed a
        conv_full = convolution_fft(w_norm, a[::-1])
        g = conv_full[len(a)-1 : len(a)-1 + len(a)]  # length n
        diag_term = w_norm[2*np.arange(len(a))] * a
        grad = 2.0 * g - diag_term

        # step size
        step = step0 / np.sqrt(t)
        a = a - step * grad
        # project onto simplex
        a = project_to_simplex(a)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Smoothed GD improved to {sc:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score

"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

Algorithm outline
-----------------
1. **Fallback** – use the provided global `height_sequence_1` if available.
2. **Problem size** – pick a moderately large `n` (4096) unless a long
   fallback already exists.
3. **Analytic families** – evaluate many classic windows (U‑shaped,
   Gaussian, cosine …) together with a *parameterised product family*
   `((i+½)(n‑½‑i)+δ)^{-α}` and random mixtures with a uniform component.
   The best vector so far is kept.
4. **Exponentiated‑gradient (mirror descent)** – fast sub‑gradient
   optimisation on the exact max‑convolution objective.
5. **Smoothed gradient descent** – minimise a differentiable surrogate
   `ε·log Σ exp(conv/ε)` using the explicit gradient
   `∇ = 2·(w*a_rev) – diag(w)·a` (where `w` are the soft‑max weights of the
   convolution).  ε is decreased during the run (continuation).
6. **Projected sub‑gradient descent** – cheap long‑tail refinement.
7. **LP‑based mixing** – classic state‑of‑the‑art refinement step.
8. **Symmetrisation** – the problem is symmetric, so averaging with the
   reversed vector cannot hurt.
9. **Stochastic hill‑climb (simulated annealing)** – final escape from
   shallow local minima.
All phases respect the global wall‑clock budget (default 1000 s) and the
best vector found is returned as a plain Python list ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}  (U‑shaped, ends heavy)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def product_family_offset(n: int, alpha: float, offset: float) -> np.ndarray:
    """
    Same as product_family but with an additive offset inside the
    denominator to smooth the singularities at the ends:
        w_i ∝ ( (i+0.5)*(n-0.5-i) + offset )^{-α}
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 0.5 - i)
    w = np.power(prod + offset, -alpha, where=(prod + offset) > 0)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-β}  (monotone, heavy left side)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """w_i ∝ exp(-γ·(i‑mid)²)  (discrete Gaussian)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, window_type: str) -> np.ndarray:
    """Standard cosine windows (Hann, Hamming, Blackman)."""
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (heavy left side)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped (product‑like) start."""
    gamma = np.random.uniform(0.0, 12.0)
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimisation
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 300_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient on the exact max‑convolution objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]  # diagonal term weight 1

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {sc:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def smoothed_gradient_descent(start_a: np.ndarray,
                              deadline: float,
                              max_iters: int = 200_000,
                              step0: float = 0.20,
                              eps0: float = 0.03,
                              eps_decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        f_ε(a) = ε·log Σ_k exp( (a*a)_k / ε )
    which converges to the true max‑objective as ε→0.
    Uses the explicit gradient derived from the soft‑max weights.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = conv.max()
        eps = max(eps0 * (eps_decay ** t), 1e-8)

        # soft‑max weights (stable by subtraction of max)
        w = np.exp((conv - maxc) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w_norm = w / w_sum

        # gradient = 2 * (w_norm * a_rev)_valid - diag(w_norm) * a
        conv_full = convolution_fft(w_norm, a[::-1])          # length 3n‑2
        g = conv_full[n - 1:n - 1 + n]                       # size n
        diag_term = w_norm[2 * np.arange(n)] * a
        grad = 2.0 * g - diag_term

        step = step0 / math.sqrt(t)
        a = a - step * grad
        a = project_to_simplex(a)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Smoothed GD improved to {sc:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheap long tail)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Standard projected sub‑gradient descent on the true max‑objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (the “state‑of‑the‑art” iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0, then return the convex mixture
        f_new = (1−t)·f + t·g_norm,
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A such that A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)      # maximise Σ g → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Determine problem size
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    n = fallback_len if fallback_len >= 1024 else 4096
    print(f"[{time.time() - start_time:.1f}] using length n={n}")

    # ------------------------------------------------------------------
    # Helper to test a candidate vector (normalised internally)
    # ------------------------------------------------------------------
    def try_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total <= 0.0:
            return
        a = a / total
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Analytic families – broad sweep
    # ------------------------------------------------------------------
    # Uniform baseline
    try_candidate(np.full(n, 1.0 / n, dtype=np.float64))

    # Product family – coarse grid
    for alpha in np.linspace(0.0, 5.0, 71):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(product_family(n, alpha))

    # Product family with a small offset (smooth the singularities)
    for offset_factor in np.logspace(-6, -3, 8):
        if time.time() >= deadline - 250.0:
            break
        offset = offset_factor * (n ** 2)   # absolute offset
        for alpha in np.linspace(0.1, 4.0, 41):
            if time.time() >= deadline - 250.0:
                break
            try_candidate(product_family_offset(n, alpha, offset))

    # Left‑heavy power family
    for beta in np.linspace(0.0, 30.0, 61):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(left_heavy_power_family(n, beta))

    # Gaussian family
    for gamma in np.logspace(-4, 2, 38):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(gaussian_family(n, gamma))

    # Cosine windows
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(window_family(n, win))

    # Random monotone / U‑shape starts
    for _ in range(200):
        if time.time() >= deadline - 250.0:
            break
        if np.random.rand() < 0.5:
            try_candidate(random_monotone_start(n))
        else:
            try_candidate(random_u_start(n))

    # ------------------------------------------------------------------
    # 3. Random mixtures of product family + uniform (covers offset ideas)
    # ------------------------------------------------------------------
    uniform_vec = np.full(n, 1.0 / n, dtype=np.float64)
    for _ in range(300):
        if time.time() >= deadline - 250.0:
            break
        alpha = np.random.uniform(0.2, 4.0)
        lam = np.random.uniform(0.0, 0.4)                # mixing weight
        offset = np.random.lognormal(mean=-12, sigma=2.0) * (n ** 2)
        # ensure offset is not astronomically large
        offset = min(offset, 0.5 * (n ** 2))
        base = product_family_offset(n, alpha, offset)
        cand = (1.0 - lam) * base + lam * uniform_vec
        try_candidate(cand)

    # ------------------------------------------------------------------
    # 4. Initialise optimisation from the best vector so far
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.asarray(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient (mirror descent) phase
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print(f"[{time.time() - start_time:.1f}] starting EG optimisation")
        a_opt, sc_opt = eg_optimize(
            a,
            deadline - 180.0,
            max_iters=500_000,
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] after EG best = {best_score:.6f}")

        # Perturbed restart
        if time.time() < deadline - 180.0:
            perturb = a * np.exp(0.02 * np.random.randn(n))
            perturb = np.maximum(perturb, 0.0)
            perturb /= perturb.sum()
            a_opt2, sc_opt2 = eg_optimize(
                perturb,
                deadline - 120.0,
                max_iters=300_000,
                eta0=0.6,
            )
            if sc_opt2 < best_score:
                best_score = sc_opt2
                a = a_opt2
                print(f"[{time.time() - start_time:.1f}] after perturbed EG best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Smoothed gradient descent (continuation on ε)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        print(f"[{time.time() - start_time:.1f}] starting smoothed GD")
        a_opt, sc_opt = smoothed_gradient_descent(
            a,
            deadline - 70.0,
            max_iters=300_000,
            step0=0.18,
            eps0=0.04,
            eps_decay=0.9997,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] after smoothed GD best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Projected sub‑gradient descent (long cheap tail)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time() - start_time:.1f}] starting sub‑gradient descent")
        a_opt, sc_opt = subgrad_optimize(
            a,
            deadline - 30.0,
            max_iters=3_000_000,
            step0=0.07,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. LP‑based refinement loop (multiple mixes)
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print(f"[{time.time() - start_time:.1f}] starting LP refinement loop")
        mix_weight = 0.2
        lp_iters = 0
        while time.time() < deadline - 20.0 and lp_iters < 30:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc = compute_score(a_new)
            if sc < best_score - 1e-8:
                best_score = sc
                a = a_new
                print(f"[{time.time() - start_time:.1f}] LP step {lp_iters} improved to {best_score:.6f}")
            else:
                mix_weight = max(0.05, mix_weight * 0.85)
            lp_iters += 1

    # ------------------------------------------------------------------
    # 9. Symmetrisation (the problem is symmetric)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time() - start_time:.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 11. Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
