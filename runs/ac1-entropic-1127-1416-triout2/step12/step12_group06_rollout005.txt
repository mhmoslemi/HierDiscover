def correlation(a, w):
    n = a.shape[0]
    m = w.shape[0]  # = 2*n-1
    L = 1 << ((n + m - 1 - 1).bit_length())
    # Actually compute next power of two of n+m-1:
    L = 1 << ((n+m-1).bit_length())
    A = np.fft.rfft(a, n=L)
    W_rev = np.fft.rfft(w[::-1], n=L)
    conv = np.fft.irfft(A * W_rev, n=L)  # length L
    # conv length n+m-1, but we only need first n+m-1 values.
    # conv_full = conv[:n+m-1]
    # correlation[j] = conv_full[m-1 - j] for j=0..n-1
    start = m - 1
    # slice needed: indices start, start-1, ..., start-(n-1)
    # Equivalent to conv[start : start - n : -1]
    corr = conv[start : start - n : -1]
    return corr

def projected_gradient_descent(a0, deadline):

def full_convolution(a):
    n = a.shape[0];
    m = 2*n-1;
    size = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[:m]
    return conv

def hill_climb(a_start, deadline, max_iter=400_000):
    a = a_start.copy()
    best = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.02
    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # Transfer a fraction of mass from i to j
        delta = step_scale * a[i]  # move proportion
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.03, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.997
            no_improve += 1
        it += 1
        if no_improve > 100_000:
            # random restart near best
            a = best + np.random.randn(n) * 0.001

assistantfinalBelow is a self‑contained optimiser that improves on the previous implementation.  
It starts from the known construction (or a simple power‑law if none is
available), performs a quick parametric sweep, then refines the sequence
with a **smooth‑max projected gradient descent** (gradually sharpening the
approximation to the true max‑convolution) and finally finishes with a
stochastic hill‑climb. All steps respect the non‑negativity and simplex
constraints required by `evaluate_sequence`.

```python
"""
Propose a step‑function sequence that minimises the evaluate_sequence objective.

Algorithm overview
------------------
1. Initialise from a known construction (height_sequence_1) or from a
   power‑law shape.
2. Very brief parametric sweep over power‑law exponents to obtain a decent
   starting point.
3. Refine the vector by a projected gradient descent on a smooth approximation
   of   max(conv(a,a)).   The smoothness parameter α is increased slowly so the
   optimiser first sees a gentle landscape and then focuses on the true max.
   The gradient of the smoothed objective is 2·(a ⊛ w) where w are the
   soft‑max weights, which we compute efficiently via FFT‑based correlation.
4. Polish the result with a stochastic hill‑climb that swaps small amounts of
   mass between two entries while staying on the probability simplex.
5. Return the best sequence found (as a plain Python list of floats).

The optimiser runs within the 1000 s budget, respects all security checks
of evaluate_sequence and logs its progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ w)[j] = Σ_i a_i * w_{i+j}.
    a has length n, w has length 2n‑1 (the full convolution of a with itself).
    Returned array has length n.
    """
    n = a.shape[0]
    m = w.shape[0]          # = 2n‑1
    L = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    W_rev = np.fft.rfft(w[::-1], n=L)
    conv_full = np.fft.irfft(A * W_rev, n=L)[:n + m - 1]
    start = m - 1
    # slice start, start‑1, …, start‑(n‑1)
    corr = conv_full[start : start - n : -1]
    return corr


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1): 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families (used for a fast initial sweep)
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient‑descent refinement on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def smooth_max_gradient_descent(a0: np.ndarray,
                                deadline: float,
                                step0: float = 0.20,
                                alpha_start: float = 20.0,
                                alpha_factor: float = 1.05) -> np.ndarray:
    """
    Projected gradient descent on f(a) = softmax_α(max(conv(a,a))).
    The smoothing parameter α is increased geometrically as the optimisation
    proceeds, allowing coarse adjustments early and finer refinements later.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    alpha = alpha_start

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_c = conv.max()

        # soft‑max weights (numerically stable by subtracting max_c)
        w_raw = np.exp(alpha * (conv - max_c))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_raw / w_sum

        # gradient = 2·(a ⋆ w)
        grad = 2.0 * correlation(a, w)

        # normalised step size
        norm = np.linalg.norm(grad) + 1e-16
        step = step0 / math.sqrt(t) / norm
        a -= step * grad
        a = project_to_simplex(a)

        # occasional exact evaluation
        if t % 200 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                # accelerate smoothing when we make progress
                alpha = min(alpha * alpha_factor, 500.0)

        # slowly increase the smoothness parameter even if no improvement
        if t % 3000 == 0 and alpha < 500.0:
            alpha *= alpha_factor

        # tiny random jitter every few thousand iterations to escape plateaus
        if t % 10000 == 0:
            jitter = np.random.randn(a.shape[0]) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (fine polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_iter: int = 600_000) -> np.ndarray:
    """
    Greedy stochastic moves that transfer a small fraction of mass from one
    coordinate to another, always projecting back onto the simplex.
    """
    a = a_start.copy()
    best = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.04, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            # escape long flat region
            a = best + np.random.randn(n) * 0.001
            a = project_to_simplex(a)
            no_improve = 0

    return best

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Load a fallback construction if it is present
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Very quick parametric sweep (power‑law only, ≈5–10 s)
    # ------------------------------------------------------------------
    sweep_deadline = deadline - 650.0   # reserve most of the budget for refinement
    n = 2048
    if time.time() < sweep_deadline:
        best_local_val = float('inf')
        best_local_vec = None
        for p in np.linspace(0.15, 2.00, 191):   # step ≈0.01
            if time.time() >= sweep_deadline:
                break
            a = power_law(n, p)
            val = compute_score(a)
            if val < best_local_val:

        if best_local_vec is not None:
            # evaluate with the official function for fairness
            val_official = evaluate_sequence(best_local_vec.tolist())
            print(f"[parametric sweep] best official score = {val_official:.6f}")
            if val_official < best_val:
                best_val, best_seq = val_official, best_local_vec.tolist()

    # ------------------------------------------------------------------
    # 2. If we still have no sequence, fall back to a simple power‑law
    # ------------------------------------------------------------------
    if best_seq is None:
        a0 = power_law(n, p=0.5)
        best_seq = a0.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[default init] score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Smooth‑max projected gradient descent (core optimiser)
    # ------------------------------------------------------------------
    gd_deadline = deadline - 150.0   # leave ~150 s for the final hill‑climb
    if time.time() < gd_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_refined = smooth_max_gradient_descent(a_start, gd_deadline)
        seq = a_refined.tolist()
        val = evaluate_sequence(seq)
        print(f"[smooth‑grad] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, seq

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climb polish
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        seq = a_final.tolist()
        val = evaluate_sequence(seq)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, seq

    # ------------------------------------------------------------------
    # 5. Safeguard – guarantee a valid return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6. Final report (using the official evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
```
