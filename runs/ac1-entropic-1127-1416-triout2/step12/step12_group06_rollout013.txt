"""
Improved optimiser for ``evaluate_sequence``.

The optimiser works in three stages:

0.  A broad parametric sweep (power‑law, geometric, beta‑shape,
    mixed plateau + power‑law, two‑level) to obtain a good starting
    vector.

1.  A smooth‑max gradient descent (Adam optimiser) that minimises a
    differentiable surrogate
        smax_α(conv(a)) = (1/α)·log Σ_k exp(α·conv_k)
    of the true objective 2·n·max(conv).  The surrogate is gradually
    sharpened (α grows from 5 to 30) so that the iterates first move
    mass away from the large central convolution peak and then focus
    on the true max.  After every iteration the iterate is projected
    onto the probability simplex (non‑negative, sum = 1).

2.  A final stochastic hill‑climb (pairwise mass transfers) for a few
    seconds to polish the solution.

All phases respect the global time budget (≈ 1000 s).  The best vector
found is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray, b: np.ndarray = None) -> np.ndarray:
    """Return full linear convolution of a and b using real FFT.
    If b is None, returns a * a."""
    if b is None:
        b = a
    n = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:n]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return full_convolution_fft(a)


def correlation_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_j w_{i+j} * a_j   (valid correlation) for i = 0..n-1.
    Implemented via FFT convolution with the reversed weight vector.
    """
    w_rev = w[::-1]
    conv_full = full_convolution_fft(a, w_rev)          # length = 3n-2
    start = a.shape[0] - 1                               # n-1
    slice_ = conv_full[start:start + a.shape[0]]        # indices n-1 .. 2n-2
    return slice_[::-1]                                 # reverse to obtain g_i


# ----------------------------------------------------------------------
# Exact objective (used for reporting and final evaluation)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}, 0<q<1."""
    if not (0.0 < q < 1.0):
        return None
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """First L entries equal to the max of the tail, the tail follows a power‑law."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α}(1‑x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def two_level(n: int, frac: float, high: float) -> np.ndarray:
    """Two‑level vector: first ‘frac·n’ entries = high, rest = low (chosen so Σ=1)."""
    L = max(1, int(frac * n))
    low = (1.0 - high * L) / (n - L)
    if low < 0:
        return None
    a = np.full(n, low, dtype=np.float64)
    a[:L] = high
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Broad parametric sweep
# ----------------------------------------------------------------------
def parametric_sweep(n: int, time_deadline: float) -> np.ndarray:
    """Explore many families, keep the best vector found."""
    best_vec = None
    best_val = float('inf')

    # power‑law sweep
    for p in np.linspace(0.10, 5.00, 246):  # step ≈0.02
        if time.time() >= time_deadline:
            return best_vec
        a = power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # geometric sweep (very fine near 1)
    # log‑spaced for extremely close to 1
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 101),
            np.linspace(0.990, 0.9999, 150),
            1.0 - np.logspace(-5, -1, 100)[::-1]  # values 1‑1e‑5 … 1‑0.1
    )):
        if time.time() >= time_deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # mixed plateau + power‑law
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.10, 1.20, 56):
            if time.time() >= time_deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # beta‑shape sweep (broader exponent range)
    alphas = np.linspace(-2.0, 5.0, 31)
    betas = np.linspace(-2.0, 5.0, 31)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= time_deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # random two‑level shapes
    for _ in range(1500):
        if time.time() >= time_deadline:
            break
        frac = np.random.rand() * 0.5 + 0.05          # 0.05 … 0.55
        high = np.random.rand() * 4.0 / n + 1.0 / n   # modestly higher than uniform
        a = two_level(n, frac, high)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # random convex combos of three seeds
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    power = power_law(n, 0.5)
    geom = geometric(n, 0.98)
    basic = [uniform, power, geom]
    for _ in range(2000):
        if time.time() >= time_deadline:
            break
        coeffs = np.random.rand(3)
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basic))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (Adam + simplex projection)
# ----------------------------------------------------------------------
def smooth_max_gradient_descent(a0: np.ndarray,
                               time_deadline: float,
                               alpha_start: float = 5.0,
                               alpha_end: float = 30.0,
                               lr: float = 0.05,
                               beta1: float = 0.9,
                               beta2: float = 0.999,
                               eps: float = 1e-8) -> np.ndarray:
    """Projected Adam optimiser on the smoothed max‑convolution objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    n = a.shape[0]

    # schedule for alpha: linear growth over first 20000 iterations
    max_iters_for_alpha = 20000

    while time.time() < time_deadline:
        t += 1

        # --- convolution of the current iterate ---
        conv = full_convolution(a)                     # length 2n‑1

        # --- smoothed max (soft‑max) weights ---
        # increase α slowly so that early steps see a smooth landscape
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters_for_alpha)
        scaled = alpha * conv
        # numerical stability: subtract the max before exponentiation
        shift = np.max(scaled)
        exp_vals = np.exp(scaled - shift)
        w = exp_vals / exp_vals.sum()                 # shape (2n‑1,)

        # --- gradient of the surrogate (g_i = 2 Σ_j w_{i+j} a_j) ---
        g = 2.0 * correlation_with_weights(a, w)       # length n

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # --- occasional exact‑score check ---
        if t % 250 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # accelerate learning a little when we improve
                lr = min(lr * 1.03, 0.2)

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (pairwise mass transfer)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               time_deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so the harness can finish bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback (if a hand‑crafted solution is present)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    n = 2048  # length of the sequence we optimise

    # -----------------------------------------------------------------
    # 1. Parametric sweep (≈ 200 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 600.0
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Smooth‑max gradient descent (≈ 600 s)
    # -----------------------------------------------------------------
    grad_deadline = deadline - 200.0
    if time.time() < grad_deadline:
        # start from the current best; if none, fall back to uniform
        if best_seq is not None:
            a0 = np.array(best_seq, dtype=float)
        else:
            a0 = np.full(n, 1.0 / n, dtype=float)

        # run a few random restarts within the remaining time slice
        best_a = a0
        best_score = compute_score(best_a)
        restarts = 3
        for r in range(restarts):
            if time.time() >= grad_deadline:
                break
            # random Dirichlet start for diversity
            a_start = np.random.rand(n)
            a_start = project_to_simplex(a_start)
            a_opt = smooth_max_gradient_descent(a_start, grad_deadline)
            cur_score = compute_score(a_opt)
            print(f"[smooth GD restart {r+1}] score = {cur_score:.6f}")
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_opt

        # also try a short refinement from the parametric best (if any)
        if best_seq is not None and time.time() < grad_deadline:
            a_refine = smooth_max_gradient_descent(np.array(best_seq, dtype=float), grad_deadline)
            cur_score = compute_score(a_refine)
            print(f"[smooth GD refinement] score = {cur_score:.6f}")
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_refine

        if best_score < best_val:
            best_val, best_seq = best_score, best_a.tolist()
            print(f"[smooth GD] overall best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb (≈ last 200 s)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final smooth score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – uniform if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Final official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
