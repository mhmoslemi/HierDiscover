def proj_simplex(v):
    # Project onto simplex (sum=1, v>=0)
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max i s.t. u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_maxconv(a):
    n = a.shape[0]
    conv = full_convolution(a)
    max_val = conv.max()
    eps = max_val * 1e-12  # tolerance
    active = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    # compute contributions
    for k in active:
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # i runs from i_start..i_end
        # j = k - i
        # g_i += 2 * a_j
        # So for vectorized:
        i_idx = np.arange(i_start, i_end+1)
        j_idx = k - i_idx
        g[i_idx] += 2 * a[j_idx]
    return g, max_val

def adam_polish(a0, deadline, alpha=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_val = compute_score(a)

    # Optionally random jitter to avoid stagnation.

    while time.time() < deadline:
        t += 1
        g, max_val = subgradient_maxconv(a)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        delta = alpha * m_hat / (np.sqrt(v_hat) + eps)
        # step opposite direction
        a_new = a - delta
        # project onto simplex
        a_new = proj_simplex(a_new)
        a = a_new

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # maybe increase alpha? not needed.

        # occasionally add noise to escape plateaus
        if t % 500 == 0:
            noise = np.random.randn(n) * 1e-3
            a = proj_simplex(a + noise)

    return best_a

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1.  **Seed generation** – a diverse pool of candidate vectors:
    * uniform, power‑law, exponential, symmetric beta (coarse sweep) and
      an “arcsine‑like” ({1/√(i·(n‑i))}) shape.
    * several thousand random Dirichlet draws with varying concentration.

2.  **Fine symmetric‑beta sweep** – a dense 1‑D search over the
    exponent 𝛼=𝛽∈[0.05,2] (≈300 points).  The best vector found is added
    to the seed pool.

3.  **Projected Adam polishing** – for each promising seed we run a
    mirror‑descent style optimiser on the objective
    f(a)=maxₖ(a∗a)ₖ, keeping a on the probability simplex.
    The sub‑gradient of the max‑convolution is
        g_i = 2·∑_{k∈A} a_{k-i},
    where A is the (tiny) active set of indices attaining the maximum.
    Adam’s adaptive steps accelerate convergence; the iterate is
    projected onto the simplex after every update.

4.  **Optional short simulated‑annealing clean‑up** – a Metropolis
    hill‑climb from the best polished vector.

The whole search respects a wall‑clock budget (default 1000 s).  The
function returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / score utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parameterised families (used for seed creation)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape:
       a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – auxiliary for the optimiser
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise (might be off by 1e‑15)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def subgradient_maxconv(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Return (g, m) where
        g_i = 2·∑_{k∈A} a_{k-i},
    A = {k : (a*a)_k is (approximately) maximal},
    and m = max_k (a*a)_k.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    m = conv.max()
    # active set – tolerance relative to the maximum
    eps = m * 1e-12
    active = np.where(conv >= m - eps)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g, m


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------


def adam_polish(a0: np.ndarray, deadline: float,
                alpha: float = 0.05,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8) -> np.ndarray:
    """Projected Adam minimisation of maxₖ(a∗a)ₖ."""
    a = a0.copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        t += 1
        g, _ = subgradient_maxconv(a)

        # Adam moments
        m1 = beta1 * m1 + (1.0 - beta1) * g
        m2 = beta2 * m2 + (1.0 - beta2) * (g * g)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        step = alpha * m1_hat / (np.sqrt(m2_hat) + eps)

        # move opposite to the gradient
        a = proj_simplex(a - step)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to shake out of plateaux
        if t % 400 == 0:
            jitter = np.random.randn(n) * (alpha * 0.001 / np.sqrt(t))
            a = proj_simplex(a + jitter)

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (final polish, optional)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.02,
                        step_scale_min: float = 1e-7) -> np.ndarray:
    """Metropolis hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional global jitter
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = proj_simplex(a * np.exp(jitter))
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Seed creation helpers
# ----------------------------------------------------------------------


def add_candidate(vec: np.ndarray, pool: list):
    """Normalise `vec` onto the simplex, compute its score and store."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return
    vec = vec / s
    score = compute_score(vec)
    pool.append((score, vec))
