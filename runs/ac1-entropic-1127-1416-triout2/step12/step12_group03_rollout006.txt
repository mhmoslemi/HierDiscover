"""
search_step_function.py

Find a non‑negative step‑function (a list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

under the simplex constraint.  The algorithm combines:

* a fast analytic scan of a few one‑parameter families (arcsine‑like,
  left‑heavy power, exponential decay),
* a long mirror‑descent / exponentiated‑gradient phase on the true
  max‑convolution objective,
* Adam optimisation of a smooth‑max (log‑sum‑exp) surrogate with a
  slowly increasing temperature,
* a fine‑grained projected sub‑gradient descent,
* a short simulated‑annealing hill‑climb.

All phases respect the global 1000 s wall‑clock budget and
periodically keep the best sequence found.  The final sequence is
returned as a plain Python list ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution a*b using real FFT.
    Length = len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = rfft(a, n=nfft)
    B = rfft(b, n=nfft)
    C = A * B
    conv = irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` when the vector is
    normalised to sum = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def arcsine_family(n: int, alpha: float = 0.5) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    α=0.5 gives the classic arcsine shape.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Power‑law decreasing towards the right."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_decay_family(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Mixture of uniform and exponential decay."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    dec = exp_decay_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * dec
    return a / a.sum()


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 600_000,
                eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution
    objective.  The step size η_t = η0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]        # diagonal term appears only once

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if t % 8000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best = sc, a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_sc:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best = sc, a.copy()
    return best, best_sc


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 2.0,
                         beta_end: float = 20000.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate
         f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    with a logarithmic schedule for β.
    """
    eps = 1e-12
    # work in log‑softmax coordinates to stay on the simplex
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_sc = compute_score(start_a)
    best_a = start_a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (log‑linear)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)                     # simplex point
        conv = convolution_fft(a, a)
        cmax = conv.max()
        # numerically stable soft‑max over β·conv
        exp_w = np.exp(beta * (conv - cmax))
        w = exp_w / exp_w.sum()            # weights for the smooth max

        # gradient of smooth max w.r.t. a (including diagonal correction)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]
        # diagonal correction: subtract a_i * w_{2i}
        diag_idx = 2 * np.arange(len(a))
        grad -= a * w[diag_idx]

        # chain rule through softmax
        dot_ag = np.dot(a, grad)
        grad_z = a * grad - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_sc:
                best_sc, best_a = sc, a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_sc:.6f}")

    # final evaluation
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_sc:
        best_sc, best_a = sc_fin, a_fin.copy()
    return best_a, best_sc


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Very cheap per iteration, allowing many steps.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best = sc, a.copy()
                print(f"[{time.time():.1f}] Sub‑grad improved to {best_sc:.6f}")

    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best = sc, a.copy()
    return best, best_sc


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb on the simplex.
    Occasionally accepts uphill moves.
    """
    n = start_a.size
    cur = start_a.copy()
    cur_sc = compute_score(cur)

    best = cur.copy()
    best_sc = cur_sc

    T0 = cur_sc * 1e-4
    T = T0
    scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur.copy()
        # either swap or transfer a fraction of mass
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = scale * cur[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_sc = compute_score(cand)

        delta = cand_sc - cur_sc
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur, cur_sc = cand, cand_sc
            if cur_sc < best_sc:
                best_sc, best = cur_sc, cur.copy()
                scale = min(scale * 1.07, 0.5)
        else:
            scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best, best_sc

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided `evaluate_sequence` function.  The routine respects the
    global wall‑clock budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use a fallback candidate if the harness provides one.
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score, best_seq = fb_sc, fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose problem size.
    # ------------------------------------------------------------------
    target_n = 4096               # a good trade‑off between quality & speed
    if best_seq is not None and len(best_seq) > target_n:
        n = len(best_seq)
    else:
        n = target_n
    print(f"[{time.time():.1f}] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Quick analytic scan over a few families.
    # ------------------------------------------------------------------
    families = []

    # arcsine‑type with a few exponents
    alphas = np.linspace(0.35, 0.85, 11)
    for a in alphas:
        families.append(arcsine_family(n, a))

    # left‑heavy power
    betas = np.linspace(0.8, 2.5, 9)
    for b in betas:
        families.append(left_heavy_power_family(n, b))

    # exponential decay
    gammas = np.linspace(0.0, 12.0, 13)
    for g in gammas:
        families.append(exp_decay_family(n, g))

    # mixed uniform / decay
    for lam in [0.2, 0.5, 0.8]:
        for g in [2.0, 5.0, 9.0]:
            families.append(mixed_family(n, lam, g))

    # evaluate candidates quickly
    for cand in families:
        if time.time() >= deadline - 5.0:
            break
        cand_sc = compute_score(cand)
        if cand_sc < best_score:
            best_score = cand_sc
            best_seq = cand.tolist()
            print(f"[{time.time():.1f}] early scan improved ⇒ {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Normalise the working vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)
    print(f"[{time.time():.1f}] after scan best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Divide remaining time among optimisation phases.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    eg_time   = 0.20 * remaining
    adam_time = 0.40 * remaining
    subg_time = 0.30 * remaining
    sa_time   = remaining - (eg_time + adam_time + subg_time)

    # ------------------------------------------------------------------
    # 5.  Exponentiated‑gradient phase.
    # ------------------------------------------------------------------
    if time.time() + eg_time < deadline:
        print(f"[{time.time():.1f}] EG phase (≈{eg_time:.1f}s)")
        a, sc = eg_optimize(a, time.time() + eg_time,
                            max_iters=400_000,
                            eta0=0.7)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] EG improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Adam optimisation of the smooth‑max surrogate.
    # ------------------------------------------------------------------
    if time.time() + adam_time < deadline:
        print(f"[{time.time():.1f}] Adam phase (≈{adam_time:.1f}s)")
        a, sc = adam_smooth_optimize(a,
                                      time.time() + adam_time,
                                      max_iters=600_000,
                                      beta_start=2.0,
                                      beta_end=20000.0,
                                      lr=0.02)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Projected sub‑gradient descent (fine‑tuning).
    # ------------------------------------------------------------------
    if time.time() + subg_time < deadline:

            print(f"[{time.time():.1f}] Sub‑gradient improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Optional symmetrisation (the problem is symmetric).
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)

        print(f"[{time.time():.1f}] Symmetrisation improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final simulated‑annealing hill‑climb.

    if time.time() + sa_time < deadline:

    # ------------------------------------------------------------------
    # 10.  Final evaluation with the provided `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = best_seq
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
