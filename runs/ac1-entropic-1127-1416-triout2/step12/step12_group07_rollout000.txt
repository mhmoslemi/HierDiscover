"""
search for a step‑function (non‑negative list) that minimises
evaluate_sequence.  The strategy is:

1.  **Broad stochastic sweep** – generate many random candidates.
    *   Uniform‑mass on a random support of size k (k≈fraction·n).
    *   Dirichlet draws with small concentration α (sparse vectors).
    *   A few classic parametric families (power‑law, geometric, …) for safety.
    The best few candidates are kept.

2.  **Projected sub‑gradient refinement** – starting from the best
    candidate, perform sub‑gradient descent on the nonsmooth objective
    `max(conv(a,a))` (equivalently `2·n·max(conv)`).  A tiny line‑search
    over several step sizes and simple momentum accelerate convergence.

3.  **Stochastic hill‑climb** – final fine‑tuning by moving a small
    amount of mass between two random indices, greedy acceptance,
    with an adaptive step‑size and occasional random “jitter” to escape
    plateaus.

The whole procedure respects a wall‑clock budget (default 1000 s) and
returns the best sequence found.  The internal score is computed with
`compute_score`, which is scale‑invariant (vectors are kept normalised
to sum = 1).  The final official score is obtained via
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σv_i = 1, v_i ≥ 0."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Length = 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a normalised vector
    (Σa == 1):
        2·n·max(conv(a,a))
    Scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, beta: float) -> np.ndarray:
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    a /= a.sum()
    return a


def cosine_power(n: int, expo: float) -> np.ndarray:
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def random_support_vector(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k uniformly chosen positions receive equal mass (1/k)."""
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a


def random_dirichlet_vector(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet(α,…,α) – yields a sparse vector when α<1."""
    a = rng.dirichlet(np.full(n, alpha))
    return a

def _maybe_insert(best: list, item: tuple, maxlen: int):
    """Insert (score, vec) into the sorted best list if it improves."""
    if len(best) < maxlen:
        best.append(item)
        best.sort(key=lambda x: x[0])
    else:
        if item[0] < best[-1][0]:
            best[-1] = item
            best.sort(key=lambda x: x[0])

# Sub‑gradient refinement
# ----------------------------------------------------------------------
def subgrad_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv) with a tiny line‑search."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    vel = np.zeros_like(a)

    # a modest set of step‑size candidates (geometric)
    step_candidates = np.array([0.02, 0.05, 0.10, 0.18, 0.30])

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient = sum_{k∈active} 2 * a shifted by k
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # momentum term
        vel = 0.85 * vel - (0.1 / np.sqrt(t)) * g

        improved = False
        for step in step_candidates:
            cand = a + step * vel
            cand = project_to_simplex(cand)
            sc = compute_score(cand)
            if sc < best_sc:
                best_sc, best_a = sc, cand.copy()
                a = cand
                improved = True
                break

        if not improved:
            # occasional jitter to escape tiny plateaus
            if t % 250 == 0:
                jitter = np.random.randn(n) * (0.02 / np.sqrt(t))
                a = project_to_simplex(a + jitter)
            else:
                cand = a - (0.05 / np.sqrt(t)) * g
                a = project_to_simplex(cand)

    return best_a


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float, max_iter: int = 800_000) -> np.ndarray:
    """Tiny random mass‑transfer moves, greedy acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]               # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        # if stuck for a while, inject a tiny random jitter
        if no_improve > 120_000:
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1

    return best_a

        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad stochastic sweep
    # -----------------------------------------------------------------
    n = 2048
    sweep_deadline = deadline - 600.0   # reserve ~10 minutes for refinements
    top_k = 12
    best_candidates = []   # list of (score, vector)

    rng = np.random.default_rng(0xC0FFEE)

    def consider(vec: np.ndarray):
        if vec is None or np.isnan(vec).any():
            return
        sc = compute_score(vec)
        _maybe_insert(best_candidates, (sc, vec), top_k)

    # ---- classic families (kept low‑cost) ----
    # power‑law with offset
    for p in np.linspace(0.35, 1.20, 20):
        if time.time() >= sweep_deadline:
            break
        for off in [0.5, 1.0, 2.0, 5.0, 10.0]:
            if off >= n:
                continue
            vec = powerlaw_offset(n, p, off)
            consider(vec)

    # geometric
    for r in np.linspace(0.60, 0.99, 30):
        if time.time() >= sweep_deadline:
            break
        vec = geometric(n, r)
        consider(vec)

    # beta shape
    for beta in np.linspace(0.5, 12.0, 20):
        if time.time() >= sweep_deadline:
            break
        vec = beta_shape(n, beta)
        consider(vec)

    # cosine power
    for e in np.linspace(0.3, 5.0, 20):
        if time.time() >= sweep_deadline:
            break
        vec = cosine_power(n, e)
        consider(vec)

    # ---- random support vectors (the main source of improvement) ----
    fractions = [0.30, 0.45, 0.60, 0.75, 0.90]   # fraction of positions that carry mass
    while time.time() < sweep_deadline:
        for frac in fractions:
            if time.time() >= sweep_deadline:
                break
            k = max(2, int(frac * n))
            vec = random_support_vector(n, k, rng)
            consider(vec)
        # ---- Dirichlet draws (sparse) ----
        for alpha in [0.01, 0.03, 0.07, 0.15, 0.30]:
            if time.time() >= sweep_deadline:
                break
            vec = random_dirichlet_vector(n, alpha, rng)
            consider(vec)

    # keep the best candidate from the sweep
    if best_candidates:
        best_sc0, best_vec0 = best_candidates[0]
        print(f"[global sweep] best score = {best_sc0:.6f}")
        if best_sc0 < best_val:
            best_val, best_seq = best_sc0, best_vec0.tolist()

    # -----------------------------------------------------------------
    # 2. Sub‑gradient polishing
    # -----------------------------------------------------------------
    sub_deadline = deadline - 200.0
    if best_seq is not None and time.time() < sub_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgrad_refine(a0, sub_deadline)
        val = compute_score(a_polished)
        print(f"[sub‑grad polish] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – fall back to a uniform vector if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Official evaluation (includes security wrapper)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported (official) score = {final_score:.6f}")
    return best_seq
