"""
Propose a non‑negative step–function that (almost) minimizes

        2·n·max(convolve(a,a)) / (sum(a)²)

by:
  1. a very fast scan of the symmetric “U‑shaped” product family
     (a_i ∝ ((i+½)(n‑½‑i))^{-α}) to get a good starting point;
  2. a smooth‑max (soft‑max) gradient descent on the exact objective,
     using an entropy‑mirror (exponentiated‑gradient) update;
  3. a short symmetrisation step (the problem is symmetric);
  4. a final stochastic hill‑climb (simulated‑annealing) for polishing.
All phases respect the global 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric U‑shaped family:
        w_i ∝ ((i+½)(n‑½‑i))^{-α}.
    """
    i = np.arange(n, dtype=np.float64)
    # log of the denominator to avoid overflow/underflow
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Heavy‑left random decreasing vector (used as a fallback start)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (entropy mirror descent)
# ----------------------------------------------------------------------


def softmax_gd(start_a: np.ndarray,
               deadline: float,
               max_iters: int = 300_000,
               eta0: float = 0.5,
               lambda0: float = 10.0,
               lambda_inc: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed objective
        f_λ(a) = (1/λ)·log Σ_j exp(λ·conv_j)
    using an exponentiated‑gradient update.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    lam = lambda0
    t = 0
    while t < max_iters:
        t += 1
        if time.time() >= deadline:
            break

        # convolution a*a
        conv = convolution_fft(a, a)          # length 2n‑1

        # soft‑max weights (numerically stable)
        conv_max = conv.max()
        shifted = conv - conv_max
        exp_vals = np.exp(lam * shifted)
        w = exp_vals / exp_vals.sum()         # Σ w = 1, length 2n‑1

        # gradient of Σ w_j·conv_j w.r.t. a:
        #   g = 2· (a * w)   (first n entries of the convolution)
        grad_full = convolution_fft(a, w)      # length n+2n‑2 = 3n‑2
        grad = 2.0 * grad_full[:n]            # keep only the needed slice

        # exponentiated‑gradient step
        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * grad)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # increase smoothing temperature slowly
        lam *= (1.0 + lambda_inc)

        # occasional reporting
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:.1f}s] soft‑max GD improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> List[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The function respects the global time budget
    and returns a plain Python list of floats.
    """
    global height_sequence_1  # may be supplied by the harness
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback candidate supplied by the harness
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}s] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem size (power of two, at least 2048)
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # If we already have a long fallback we keep its length,
    # otherwise we target a larger size for better discretisation.
    n = max(2048, fallback_len)
    # round up to next power of two for cheap FFTs
    n = 1 << (n - 1).bit_length()
    print(f"[{time.time() - start_time:.1f}s] using length n = {n}")

    # ------------------------------------------------------------------
    # Helper to test a candidate (normalised internally)
    # ------------------------------------------------------------------
    def try_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}s] new best score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Very fast scan of the symmetric product family
    # ------------------------------------------------------------------
    # coarse grid
    alphas = np.linspace(0.1, 3.0, 151)   # step ≈0.02
    for alpha in alphas:
        if time.time() >= deadline - 200.0:
            break
        try_candidate(product_family(n, alpha))

    # fine grid around the current best α (if we have one)
    if best_seq is not None:
        # recover α from best_seq by a simple 1‑D line search is cheap;
        # we just reuse the current best vector as the start for later phases.
        pass

    # ------------------------------------------------------------------
    # 3. Add a few random monotone restarts (cheap)
    # ------------------------------------------------------------------
    for _ in range(30):
        if time.time() >= deadline - 200.0:
            break
        try_candidate(random_monotone_start(n))

    # ------------------------------------------------------------------
    # 4. Initialise the main optimisation vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)      # uniform fallback
    else:
        a = best_seq.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 5. Smooth‑max (soft‑max) mirror‑descent optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time() - start_time:.1f}s] starting smooth‑max GD")
        a_opt, sc_opt = softmax_gd(
            a,
            deadline - 60.0,           # leave some time for the final anneal
            max_iters=400_000,
            eta0=0.6,
            lambda0=20.0,
            lambda_inc=0.001,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}s] after GD best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation (the problem is symmetric)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time() - start_time:.1f}s] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final stochastic hill‑climb (simulated‑annealing)

    # ------------------------------------------------------------------
