"""
Improved search for a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm
-------------
0.  Grab the historic best (height_sequence_1) if available and
    normalise it to a probability vector.
1.  Generate a collection of diverse starting vectors:
    * a few analytic shapes (product‑α, exp‑U, mixed)
    * several random Dirichlet draws.
2.  For each start run a short Adam optimisation on the smooth‑max
    surrogate (≈150 k iterations).  Keep the best outcome.
3.  Refine the current best with a long projected sub‑gradient descent
    (≈1 M iterations) on the exact max‑objective.
4.  Finish with a *guided* hill‑climb that always attacks the shift
    contributing the current maximum autocorrelation.  Mass is moved
    from the most harmful pair to a distant location, the simplex
    constraint being satisfied automatically.
5.  Return the best sequence seen (as a plain list of floats).

All phases respect a global wall‑clock deadline (1000 s) and
periodically print timing / objective information.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)**2).
    a is assumed non‑negative.
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# ----------------------------------------------------------------------
# Analytic families (parameterised)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow for very large α
    if np.any(np.isinf(w)):
        w = np.where(np.isinf(w), np.finfo(np.float64).max, w)
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_product_u_family(n: int, alpha: float, gamma: float, lam: float) -> np.ndarray:
    """
    Convex mixture: lam·product(α) + (1‑lam)·exp‑U(γ).
    lam∈[0,1].
    """
    prod = symmetric_product_family(n, alpha)
    u = exp_u_family(n, gamma)
    w = lam * prod + (1.0 - lam) * u
    return normalize(w)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 5.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.01) -> Tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp smooth‑max of the convolution.
    Returns the best vector seen (still a probability simplex).
    """
    n = start_a.shape[0]
    eps = 1e-15
    # initialise via log‑softmax (ensures positivity)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = normalize(np.exp(z))
    best_score = compute_score(best_a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # temperature schedule for smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = normalize(np.exp(z))

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # smooth‑max weights

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 5000 == 0:
            a_tmp = normalize(np.exp(z))
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam iter {it:,} improved to {best_score:.6f}")

    # final check
    a_fin = normalize(np.exp(z))
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.04) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = np.maximum(a, 0.0)
        a /= a.sum()                     # simplex projection (fast)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad iter {t:,} improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def guided_hill_climb(start_a: np.ndarray,
                      deadline: float,
                      max_iters: int = 800_000,
                      step_factor: float = 0.15) -> Tuple[np.ndarray, float]:
    """
    Targeted hill‑climb that repeatedly attacks the shift
    responsible for the current maximal autocorrelation.
    """
    n = start_a.size
    a = start_a.copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    it = 0
    # temperature‑like decay for the move size
    move_factor = step_factor

    while it < max_iters and time.time() < deadline:
        it += 1

        conv = convolution_fft(a, a)
        k_max = int(np.argmax(conv))

        # contributions to the dominating shift
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        total_contrib = contrib.sum()
        if total_contrib == 0.0:
            # fallback to uniform random move
            probs = np.full(n, 1.0 / n)
        else:
            probs = contrib / total_contrib

        # pick source index biased by contribution
        src = np.random.choice(n, p=probs)

        # choose a distant destination (avoid src and its partner)
        partner = k_max - src
        dst = np.random.randint(n)
        # enforce a minimal distance to reduce re‑creation of the same product
        while (dst == src) or (dst == partner) or (abs(dst - src) < n // 8):
            dst = np.random.randint(n)

        # amount to move – a modest fraction of src's mass
        delta = move_factor * a[src]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # simplex is automatically respected (sum unchanged, positivity kept)
        if cand[src] < 0.0:
            # clip and renormalise safely
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()

        cand_score = compute_score(cand)

        # accept only if we improve (simple greedy policy)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                # increase move size a bit when we make progress
                move_factor = min(move_factor * 1.07, 0.3)
        else:
            # slowly shrink the move size on failures
            move_factor = max(move_factor * 0.97, 1e-4)

        if it % 20000 == 0:
            print(f"[{time.time():.1f}] guided‑hill iter {it:,} best={best_score:.6f}")

    # final check
    final_sc = compute_score(best_a)
    if final_sc < best_score:
        best_score, best_a = final_sc, best_a.copy()
    return best_a, best_score

def random_dirichlet_start(n: int, concentration: float = 0.5) -> np.ndarray:
    """
    Draw a random point from a Dirichlet distribution with all
    parameters equal to `concentration` (the smaller, the sparser).
    """
    draws = np.random.gamma(concentration, 1.0, size=n)
    return draws / draws.sum()

    # 5. Return result as plain list (evaluate_sequence will re‑apply clipping)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] search finished – reported score = {final_score:.6f}")
    return best_seq.tolist()
