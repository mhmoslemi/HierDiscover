"""
Improved step‑function search.

The algorithm builds a large symmetric distribution (default length 8192)
starting from the analytically motivated arcsine shape.
It then refines the shape by a sequence of optimisation phases:

* 1️⃣  Scan a one‑parameter family a_i ∝ ((i+½)(n‑½‑i))^{-α} to get a better
    initial point.
* 2️⃣  Adam optimisation of a smooth‑max surrogate  f_β(a) = (1/β)·log Σ
    exp(β·(a*a)_k)  with a linear schedule for β.
* 3️⃣  Projected sub‑gradient descent on the exact max‑convolution objective.
* 4️⃣  A short exponentiated‑gradient (mirror‑descent) fine‑tune.
* 5️⃣  Symmetrisation (the problem is invariant under reversal).
* 6️⃣  Random restarts of the EG optimiser to escape local minima.
* 7️⃣  A cheap simulated‑annealing hill‑climb for final polishing.
* 8️⃣  Linear mixing of the best solution with the uniform distribution
    (the objective is scale‑invariant, so this can occasionally lower the
    maximum convolution).

All phases respect the global wall‑clock budget (default 1000 s).  The code
uses only NumPy and the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (for sum(a)=1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    α=0.5 gives the discrete arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 300_000,
                eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the true max‑convolution.
    Step size η_t = η0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]          # diagonal appears once

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # periodic logging
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[EG] t={t:,} score={best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
    return best_a, best_score


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 1.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth‑max surrogate:
        f_β(a) = (1/β)·log Σ exp(β·(a*a)_k)
    The parameter β is increased exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (log‑linear)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable softmax over scaled conv values
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # chain rule through softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # periodic logging
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[Adam] it={it:,} score={best_score:.6f}")

    # final evaluation
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score, best_a = sc_fin, a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[Sub‑grad] t={t:,} score={best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Very simple simulated‑annealing walk on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                scale = min(scale * 1.07, 0.5)
        else:
            scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine respects the time
    budget, builds a symmetric distribution of length 8192, and applies
    several optimisation phases (Adam → sub‑gradient → EG → SA → mixing).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # 0. Optional fallback – keep it only if it is already longer than
    #    our target length; otherwise we ignore it.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[fallback] score={fb:.6f}")
        if len(fb_vec) >= 8192:
            best_seq = fb_vec
            best_score = fb
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose working size.
    # ------------------------------------------------------------------
    n = 8192                     # a size that gives a fairly low theoretical bound
    print(f"[setup] working length n={n}")

    # ------------------------------------------------------------------
    # 2. Initialise with arcsine (α=0.5) and a quick α‑scan.
    # ------------------------------------------------------------------
    a = symmetric_product_family(n, 0.5)
    best_a = a.copy()
    best_score = compute_score(best_a)

    # a modest scan around the arcsine exponent
    for alpha in np.linspace(0.35, 0.65, 13):
        if time.time() >= deadline - 30.0:
            break
        cand = symmetric_product_family(n, alpha)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[scan] α={alpha:.3f} → score={best_score:.6f}")

    a = best_a.copy()
    print(f"[init] best after scan = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Allocate time slices for optimisation phases.
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # Heuristic split – adaptively reduced if time runs low.
    adam_frac = 0.30
    subg_frac = 0.30
    eg_frac   = 0.15
    sa_frac   = 0.15
    rnd_frac  = 0.10   # small slot for random EG restarts

    # ------------------------------------------------------------------
    # 4. Adam optimisation of smooth max.
    # ------------------------------------------------------------------
    adam_dead = time.time() + adam_frac * remaining
    if time.time() < adam_dead:
        print("[phase] Adam smooth‑max optimisation")
        a, sc = adam_smooth_optimize(a, adam_dead,
                                     max_iters=250_000,
                                     beta_start=1.0,
                                     beta_end=5000.0,
                                     lr=0.03)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[Adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent.
    # ------------------------------------------------------------------
    subg_dead = time.time() + subg_frac * remaining
    if time.time() < subg_dead:
        print("[phase] Sub‑gradient descent")
        a, sc = subgrad_optimize(a, subg_dead,
                                 max_iters=800_000,
                                 step0=0.05)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[Sub‑grad] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Short EG fine‑tune.
    # ------------------------------------------------------------------
    eg_dead = time.time() + eg_frac * remaining
    if time.time() < eg_dead:
        print("[phase] Exponentiated‑gradient fine‑tune")
        a, sc = eg_optimize(a, eg_dead,
                            max_iters=200_000,
                            eta0=0.5)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[EG] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Enforce symmetry (invariant under reversal) and re‑project.
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score, best_a = sc_sym, a_sym.copy()
        print(f"[sym] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Random EG restarts – try to escape possible local minima.
    # ------------------------------------------------------------------
    rnd_dead = time.time() + rnd_frac * remaining
    while time.time() < rnd_dead:
        # create a noisy perturbation of the current best
        noise = np.random.randn(n) * 0.02
        pert = best_a + noise
        pert = np.maximum(pert, 0.0)
        pert = pert / pert.sum()
        # brief EG run
        eg_end = time.time() + 5.0               # 5 s per restart
        a_restart, sc_restart = eg_optimize(pert, eg_end,
                                             max_iters=30_000,
                                             eta0=0.6)
        if sc_restart < best_score:
            best_score, best_a = sc_restart, a_restart.copy()
            print(f"[restart] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Hill‑climb (simulated annealing) for final polishing.
    # ------------------------------------------------------------------
    sa_dead = time.time() + sa_frac * remaining
    if time.time() < sa_dead:
        print("[phase] Simulated‑annealing hill‑climb")
        a, sc = hill_climb_sa(best_a, sa_dead, max_steps=600_000)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[SA] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Linear mixing with the uniform distribution – sometimes lowers M.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for lam in np.linspace(0.0, 1.0, 11):
        cand = lam * best_a + (1.0 - lam) * uniform
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[mix] λ={lam:.2f} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 11. Final evaluation with the provided evaluator.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[final] reported score = {final_score:.6f}")
    return final_seq
