        d = cand_score - cur_score

            cur_score = compute_score(a)

"""
Optimization routine for the step‑function scoring problem.

The algorithm mixes several ideas:

1. **Broad parametric sweep** – explores many analytic families
   (power‑law, geometric, plateau+power, beta, sine, raised‑cosine,
   polynomial, exponential·power).  The best vector of the sweep is used
   as a warm‑start.

2. **Smoothed max‑gradient descent** – replaces the non‑smooth
   ``max(conv)`` by a smooth log‑sum‑exp approximation and performs
   Adam optimisation on the probability simplex (projection onto the
   simplex after each step).  This equalises the high convolution peaks.

3. **Simulated‑annealing hill‑climb** – a final stochastic phase that
   accepts worsening moves with a temperature schedule, allowing the
   search to escape local minima.

The whole procedure respects the 1000 s budget, returning a list of
non‑negative heights (sum ≈ 1) that aims to minimise
``evaluate_sequence``.  Intermediate scores are printed for monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a via a real FFT.
    Result length = 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two arbitrary 1‑D arrays using an FFT."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Return the exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    Scale‑invariant, so we keep the L1‑norm equal to one.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def sine_window(n: int, power: float = 1.0) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    a /= a.sum()
    return a


def raised_cosine(n: int, power: float = 1.0) -> np.ndarray:
    a = (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1))) * 0.5
    a = a ** power
    a /= a.sum()
    return a


def polynomial_decay(n: int, p: float = 1.0) -> np.ndarray:
    a = ((n - np.arange(n, dtype=np.float64)) ** p)
    a /= a.sum()
    return a


def exp_power(n: int, q: float, p: float) -> np.ndarray:
    """a_i ∝ q**i * (i+1)**(-p)  (q in (0,1))."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = (q ** np.arange(n, dtype=np.float64)) * (idx ** (-p))
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Parametric sweep
# ----------------------------------------------------------------------
def parametric_sweep(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Explore many analytic families and return the best vector found."""
    best_vec = None
    best_val = float('inf')

    # 1) Power‑law
    for p in np.linspace(0.05, 2.5, 241):
        if time.time() >= deadline:
            return best_vec, best_val
        a = power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # 2) Geometric (ratios close to 1)
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 91),
            np.linspace(0.990, 0.9999, 110))):
        if time.time() >= deadline:
            return best_vec, best_val
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # 3) Mixed plateau + power‑law
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= deadline:
                return best_vec, best_val
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 4) Beta‑shape (wider exponent range)
    alphas = np.linspace(-3.0, 3.0, 31)
    betas = np.linspace(-3.0, 3.0, 31)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec, best_val
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 5) Sine window
    for power in np.linspace(0.5, 3.0, 26):
        if time.time() >= deadline:
            return best_vec, best_val
        a = sine_window(n, power)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # 6) Raised‑cosine (Hann/Hamming‑like)
    for power in np.linspace(0.5, 3.0, 26):
        if time.time() >= deadline:
            return best_vec, best_val
        a = raised_cosine(n, power)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # 7) Polynomial decay
    for p in np.linspace(0.2, 3.0, 29):
        if time.time() >= deadline:
            return best_vec, best_val
        a = polynomial_decay(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # 8) Exponential·power family
    for q in np.linspace(0.90, 0.99, 91):
        for p in np.linspace(0.2, 2.0, 37):
            if time.time() >= deadline:
                return best_vec, best_val
            a = exp_power(n, q, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # 9) Random convex mixes of three basic shapes
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    power05 = power_law(n, 0.5)
    geom98 = geometric(n, 0.98)
    basic = [uniform, power05, geom98]
    for _ in range(2000):
        if time.time() >= deadline:
            break
        coeffs = np.random.rand(3)
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basic))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec, best_val


# ----------------------------------------------------------------------
# Smoothed max – Adam optimisation on the simplex
# ----------------------------------------------------------------------
def smooth_descent(a0: np.ndarray, deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam optimisation of a smooth log‑sum‑exp approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12  # base learning rate
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = full_convolution(a)
        cmax = conv.max()

        # smoothing parameter γ grows slowly (more emphasis on the true max)
        gamma = min(250.0, 5.0 + t * 0.005)

        # soft‑max weights (numerically safe)
        exp_vals = np.exp(gamma * (conv - cmax))
        w = exp_vals / exp_vals.sum()               # shape: (2n-1,)

        # gradient = 2 * sum_k w_k * shift(a, k)
        w_rev = w[::-1]                              # reversed weight vector
        conv_grad = fft_convolve(a, w_rev)           # length = 3n-2
        idx = (len(w) - 1) - np.arange(n)            # 2n-2 , … , n-1
        g = 2.0 * conv_grad[idx]

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # occasional jitter to escape flat regions
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (lr * 0.004 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def simulated_annealing(a0: np.ndarray, deadline: float,
                        max_iter: int = 800_000) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climb with a temperature schedule."""
    a = a0.copy()
    cur_score = compute_score(a)
    best_a, best_score = a.copy(), cur_score
    n = a.shape[0]

    start = time.time()
    it = 0
    while time.time() < deadline and it < max_iter:
        # linear temperature decay
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(1e-4, 0.07 * (1.0 - fraction))

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # Move a random fraction of mass from i to j
        max_move = 0.25 * a[i]          # up to 25 % of the source entry
        delta = max_move * np.random.rand()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        diff = cand_score - cur_score
        if diff < 0 or np.random.rand() < math.exp(-diff / T):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()

        # occasional global perturbation if stuck
        if it % 120_000 == 0 and it > 0:
            jitter = np.random.randn(n) * 0.001
            cand = best_a + jitter
            cand = project_to_simplex(cand)
            a = cand
            cur_score = compute_score(a)

        it += 1

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid overshooting the harness deadline
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback (previous best construction if present)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad parametric sweep (≈ 30 % of the budget)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    n = 2048
    if time.time() < sweep_deadline:
        vec, val = parametric_sweep(n, sweep_deadline)
        if vec is not None:
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, vec.tolist()

    # -----------------------------------------------------------------
    # 2. Smoothed‑max Adam polishing (≈ 40 % of the budget)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 100.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_opt, val = smooth_descent(a0, polish_deadline)
        print(f"[smooth descent] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()

    # -----------------------------------------------------------------
    # 3. Simulated‑annealing hill‑climb (≈ remaining time)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final, val = simulated_annealing(a_start, deadline)
        print(f"[simulated anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – if everything failed, fall back to a uniform vector
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
