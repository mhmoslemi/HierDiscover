    support size (n = 4096).  The best parameter of each family is recorded.

2.  Generate a few random monotone starts and refine each of them with a
    medium‑length Adam optimiser that works on a smooth‑max approximation
    of the true objective.  Symmetrisation is disabled – the optimum is
    typically asymmetric.

3.  Take the overall best candidate and run a long Adam optimisation with a
    high temperature‑annealing schedule (β up to 4·10⁴) and a small learning
    rate.  This produces a very accurate approximation of the true optimum.

4.  Perform projected momentum sub‑gradient descent on the exact max‑objective
    (gradient taken at the current worst convolution index).  The momentum
    term accelerates convergence and often yields a modest improvement.

5.  Finish with a simulated‑annealing hill‑climb that makes random swaps or
    mass transfers while accepting occasional uphill moves.  The temperature
    decays linearly to zero.

6.  The best vector found is finally evaluated with the official
    ``evaluate_sequence`` wrapper and returned as a plain Python list.

All helper functions are defined at top level; no closures or lambda
expressions are used.  The search respects the 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1).

    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    The implementation follows the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric product family: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # handle possible infinities at the boundaries (when alpha large)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def product_expU_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    """Product family multiplied by an exponential‑U factor."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod - gamma * dist)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def asymmetric_product_family(n: int, beta_left: float, beta_right: float) -> np.ndarray:
    """Two‑parameter product family: a_i ∝ (i+½)^{‑β₁}(n‑½‑i)^{‑β₂}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta_left * np.log(i + 0.5) - beta_right * np.log(n - 0.5 - i))
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Random initialisers (fallback)
# ----------------------------------------------------------------------


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sort a powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser
# ----------------------------------------------------------------------


def adam_optimize(start_a: np.ndarray,
                  deadline: float,
                  max_iters: int = 400_000,
                  beta_start: float = 10.0,
                  beta_end: float = 20_000.0,
                  lr: float = 0.012,
                  sym_interval: int = 0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    start_a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    deadline : float
        Unix timestamp after which the optimisation must stop.
    max_iters : int
        Upper bound on the number of Adam iterations.
    beta_start / beta_end : float
        Exponential annealing schedule for the smooth‑max temperature.
    lr : float
        Base learning rate.
    sym_interval : int
        If >0, symmetrise the iterate every `sym_interval` steps
        (a ← (a+a[::-1])/2) and re‑project onto the simplex.
        Use 0 to disable symmetrisation.

    Returns
    -------
    best_a, best_score
        Best vector found and its exact score (computed with compute_score).
    """
    n = start_a.shape[0]
    eps = 1e-12

    # work in log‑space to keep the simplex constraint via soft‑max
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal the smooth‑max temperature
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # optional symmetrisation (helps for some families but may hurt others)
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        # full convolution of a with itself
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth‑max surrogate w.r.t a
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)[len(a) - 1:len(a) - 1 + n]

        # convert to gradient w.r.t. logits z
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation of the true objective
        if it % 5000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection and evaluation
    a_fin = softmax(z)
    if sym_interval:
        a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected momentum sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_momentum_optimize(start_a: np.ndarray,
                              deadline: float,
                              max_iters: int = 2_000_000,
                              step0: float = 0.045,
                              momentum: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with a momentum term on the exact max‑objective.

    Parameters
    ----------
    start_a : np.ndarray
        Probability vector (sum = 1).
    deadline : float
        Unix timestamp after which optimisation must stop.
    max_iters : int
        Upper bound on iterations.
    step0 : float
        Initial step size (decayed as step0 / t**0.7).
    momentum : float
        Momentum coefficient (0 ≤ momentum < 1).

    Returns
    -------
    best_a, best_score
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    v = np.zeros_like(a)  # momentum buffer

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the worst convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # decaying step size (exponent 0.7 works well in practice)
        step = step0 / (t ** 0.7)

        v = momentum * v - step * g
        a = project_to_simplex(a + v)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad‑mom improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb on the simplex.

    Moves consist of either swapping two entries or transferring a fraction
    of the mass of one entry to another.  Acceptance follows the classic
    Metropolis rule with a linearly decreasing temperature.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4  # initial temperature (tiny)
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a chunk of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        # linear temperature schedule
        T = T0 * (1.0 - step / max_steps)

        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # gradually increase step size when we make progress
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# Golden‑section refinement of a single‑parameter family
# ----------------------------------------------------------------------


def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 20) -> tuple[float, float]:
    """
    Minimises `func` on [a,b] using golden‑section search.
    Returns the best parameter and its function value.
    """
    phi = (1.0 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns a plain Python list.
    """
    # optional SciPy import (used only for local refinement)
    try:
        from scipy.optimize import minimize
        scipy_available = True
    except Exception:
        scipy_available = False

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback to a previously found solution if it exists
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a working support size (larger n gives finer discretisation)
    # ------------------------------------------------------------------
    n = 4096   # power of two, still cheap with FFT

    # ------------------------------------------------------------------
    # 2. Coarse scan of analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse family scan (n={n})")
    # baseline uniform
    vec = np.full(n, 1.0 / n, dtype=np.float64)
    sc = compute_score(vec)
    if sc < best_score:
        best_score = sc
        best_seq = vec.tolist()
        print(f"[{time.time():.1f}] new best (uniform) = {best_score:.6f}")

    family_records = {}

    # ---- symmetric product family ----
    best_alpha, best_alpha_sc = None, float('inf')
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.3f}) = {best_score:.6f}")
        if sc < best_alpha_sc:
            best_alpha_sc = sc
            best_alpha = alpha
    family_records['product'] = (best_alpha, best_alpha_sc)

    # ---- left‑heavy power family ----
    best_beta, best_beta_sc = None, float('inf')
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")
        if sc < best_beta_sc:
            best_beta_sc = sc
            best_beta = beta
    family_records['leftpower'] = (best_beta, best_beta_sc)

    # ---- exponential U shape ----
    best_gamma, best_gamma_sc = None, float('inf')
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")
        if sc < best_gamma_sc:
            best_gamma_sc = sc
            best_gamma = gamma
    family_records['expU'] = (best_gamma, best_gamma_sc)

    # ---- mixed uniform + U shape ----
    best_mix, best_mix_sc = None, float('inf')
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")
            if sc < best_mix_sc:
                best_mix_sc = sc
                best_mix = (lam, gamma)
    family_records['mix'] = (best_mix, best_mix_sc)

    # ---- asymmetric product family (2‑parameter) ----
    best_asym, best_asym_sc = None, float('inf')
    beta_grid = np.linspace(0.0, 12.0, 13)
    for b1 in beta_grid:
        for b2 in beta_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = asymmetric_product_family(n, b1, b2)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (asym β1={b1:.2f},β2={b2:.2f}) = {best_score:.6f}")
            if sc < best_asym_sc:
                best_asym_sc = sc
                best_asym = (b1, b2)
    family_records['asym'] = (best_asym, best_asym_sc)

    # ---- product + exponential U family (2‑parameter) ----
    best_pe, best_pe_sc = None, float('inf')
    a_grid = np.linspace(0.0, 12.0, 13)
    g_grid = np.linspace(0.0, 6.0, 7)
    for a in a_grid:
        for g in g_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = product_expU_family(n, a, g)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (prod‑exp α={a:.2f},γ={g:.2f}) = {best_score:.6f}")
            if sc < best_pe_sc:
                best_pe_sc = sc
                best_pe = (a, g)
    family_records['prodexpU'] = (best_pe, best_pe_sc)

    # ------------------------------------------------------------------
    # 3. Local refinement of promising families (SciPy if available)
    # ------------------------------------------------------------------
    if scipy_available and time.time() < deadline - 20.0:
        # Refine symmetric product family with golden‑section (already coarse)
        if family_records.get('product')[0] is not None:
            a0 = family_records['product'][0]
            lo = max(0.0, a0 - 0.5)
            hi = min(12.0, a0 + 0.5)

            def score_alpha(p):
                return compute_score(symmetric_product_family(n, p))

            a_opt, a_sc = golden_section_optimize(score_alpha, lo, hi, max_iter=25)
            vec = symmetric_product_family(n, a_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined product‑α={a_opt:.6f} → {best_score:.6f}")

        # Refine the asymmetric product family (2‑D) with Nelder‑Mead
        if family_records.get('asym')[0] is not None:
            b1_0, b2_0 = family_records['asym'][0]

            def score_asym(p):
                b1 = np.clip(p[0], 0.0, 12.0)
                b2 = np.clip(p[1], 0.0, 12.0)
                return compute_score(asymmetric_product_family(n, b1, b2))

            res = minimize(score_asym,
                           x0=np.array([b1_0, b2_0]),
                           method='Nelder-Mead',
                           options={'maxiter': 200,
                                    'xatol': 1e-4,
                                    'fatol': 1e-4})
            b1_opt, b2_opt = np.clip(res.x, [0.0, 0.0], [12.0, 12.0])
            vec = asymmetric_product_family(n, b1_opt, b2_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined asym (β1={b1_opt:.6f},β2={b2_opt:.6f}) → {best_score:.6f}")

        # Refine product‑expU family (2‑D)
        if family_records.get('prodexpU')[0] is not None:
            a0, g0 = family_records['prodexpU'][0]

            def score_pe(p):
                a = np.clip(p[0], 0.0, 12.0)
                g = np.clip(p[1], 0.0, 6.0)
                return compute_score(product_expU_family(n, a, g))

            res = minimize(score_pe,
                           x0=np.array([a0, g0]),
                           method='Nelder-Mead',
                           options={'maxiter': 250,
                                    'xatol': 1e-4,
                                    'fatol': 1e-4})
            a_opt, g_opt = np.clip(res.x, [0.0, 0.0], [12.0, 6.0])
            vec = product_expU_family(n, a_opt, g_opt)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined prod‑exp (α={a_opt:.6f},γ={g_opt:.6f}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Short Adam refinements for random monotone starts
    # ------------------------------------------------------------------
    n_random = 6
    for r in range(n_random):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono {r}) = {best_score:.6f}")

        # short Adam (no symmetrisation) to polish the random start
        a_start = np.array(vec, dtype=np.float64)
        a_opt, sc_opt = adam_optimize(a_start,
                                      deadline - 5.0,
                                      max_iters=120_000,
                                      beta_start=10.0,
                                      beta_end=8_000.0,
                                      lr=0.015,
                                      sym_interval=0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] rand‑mono {r} Adam improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Long Adam optimisation on the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] long Adam optimisation (high β, no sym)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_optimize(a_start,
                                      deadline - 150.0,
                                      max_iters=800_000,
                                      beta_start=20.0,
                                      beta_end=40_000.0,
                                      lr=0.011,
                                      sym_interval=0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after long Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Projected momentum sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] momentum sub‑gradient descent")
        a_sub, sc_sub = subgrad_momentum_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 40.0,
            max_iters=2_500_000,
            step0=0.045,
            momentum=0.9,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Second Adam pass (even higher β) – keep asymmetry
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] second Adam pass (very high β)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = adam_optimize(
            a_start,
            deadline - 30.0,
            max_iters=400_000,
            beta_start=50.0,
            beta_end=80_000.0,
            lr=0.010,
            sym_interval=0,
        )
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.tolist()
            print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Explicit symmetrisation test (optional – keep if beneficial)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Final hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final simulated‑annealing hill‑climb")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=5_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Fallback to a safe uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final evaluation using the official wrapper (includes all safety checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
