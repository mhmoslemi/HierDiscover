"""
Propose a non‑negative step‑function that (heuristically) minimizes
``evaluate_sequence``.  The algorithm is simple but robust:

1. **Fallback** – if a previous good construction ``height_sequence_1``
   exists we start from it.

2. **Random Dirichlet exploration** – many short random vectors drawn from
   a Dirichlet distribution (often very sparse) are tested.  Because the
   objective is scale‑invariant we keep the vectors on the probability
   simplex.

3. **Hill‑climbing refinement** – from the best vector found we perform a
   greedy stochastic hill‑climb: a small amount of mass is transferred
   between two random positions, the move is accepted only if it improves
   the score.  The step‑size adapts automatically.

All phases respect a global wall‑clock limit (default 1000 s).  The
returned list contains plain Python ``float`` objects and satisfies the
non‑negativity requirements of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project vector ``v`` onto the simplex {x | x_i >= 0, sum(x) = 1}.
    Implementation follows the standard O(n log n) algorithm.
    """
    if np.all(v <= 0):
        # all non‑positive → return the centre of the simplex
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    # Find the last index where the adjusted value would be positive
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (exact, not FFT based – used only for scoring)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Exact linear convolution a*a (full length = 2*len(a)-1)."""
    return np.convolve(a, a)


def evaluate(a: np.ndarray) -> float:
    """
    Wrapper around the external ``evaluate_sequence`` that works with
    ``numpy`` arrays.  The conversion to a Python list is required by the
    security checks inside ``evaluate_sequence``.
    """
    # ``evaluate_sequence`` is assumed to be already in the global scope.
    return evaluate_sequence(a.tolist())


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (greedy with adaptive step size)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """
    Perform a greedy hill‑climb on the simplex.  At each step a fraction of
    mass is moved from a randomly chosen source index to a target index.
    The move size is automatically increased when improvements are found
    and decreased otherwise.
    """
    a = a_start.copy()
    best = a.copy()
    best_val = evaluate(best)
    n = a.shape[0]

    # initial step fraction (relative to the source value)
    step_frac = 0.05
    it = 0
    stagnation = 0

    while time.time() < deadline and it < max_iter:
        # pick two distinct positions
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # amount of mass transferred from i to j
        delta = step_frac * a[i]
        if delta == 0.0:            # nothing to move – skip
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep feasibility (non‑negative) by simple clipping; the simplex
        # projection restores the exact sum‑to‑1 constraint.
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cand_val = evaluate(cand)
        if cand_val < best_val:
            best_val, best, a = cand_val, cand.copy(), cand
            # successful move → be a little more aggressive
            step_frac = min(step_frac * 1.08, 0.60)
            stagnation = 0
        else:
            # no improvement → become more conservative
            step_frac *= 0.985
            stagnation += 1
            a = cand  # still walk to the new point (helps escape flat plateaus)

        # occasional “shake” if we are stuck for a long time
        if stagnation > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best + jitter)
            stagnation = 0

        it += 1

    return best

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of heights that minimise ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the outer harness
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # 0.  Use a previously known good construction if available
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        # ``height_sequence_1`` is expected to be a global variable containing
        # a list of floats from a prior run.
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)

        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Random Dirichlet exploration (very cheap, many samples)
    # --------------------------------------------------------------
    n = 2048                     # default length (same as previous runs)
    rng = np.random.default_rng(seed + 1)

    # we use a fairly sparse Dirichlet (small concentration) because it
    # creates a wide variety of shapes while still being fast to generate.
    alpha_small = 0.1
    # inner loop respects the remaining time budget
    while time.time() < deadline - 200.0:    # reserve ~200 s for refinement
        # draw a Dirichlet vector and normalise to the simplex
        vec = rng.dirichlet(np.full(n, alpha_small))
        # evaluate – if something is dramatically better we keep it
        val = evaluate_sequence(vec.tolist())
        if val < best_val:
            best_val, best_seq = val, vec.tolist()
            print(f"[random] new best = {val:.6f}")

    # --------------------------------------------------------------
    # 2. Hill‑climbing refinement from the best candidate
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a0 = np.array(best_seq, dtype=float)
        # a short projection onto the simplex (just in case of tiny drift)
        a0 = project_to_simplex(a0)
        refined = hill_climb(a0, deadline)
        refined_val = evaluate_sequence(refined.tolist())
        print(f"[hill‑climb] refined score = {refined_val:.6f}")

        if refined_val < best_val:
            best_val, best_seq = refined_val, refined.tolist()

    # --------------------------------------------------------------
    # 3. Fallback – uniform vector (the theoretical optimum for the
    #    exact objective, i.e. score = 2) if everything else failed.
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)
        print("[fallback‑uniform] score =", best_val)

    # Final official evaluation (includes the security wrappers)
    final_score = evaluate_sequence(best_seq)
    print("\nSearch finished. Final reported score =", f"{final_score:.6f}")
    return best_seq
