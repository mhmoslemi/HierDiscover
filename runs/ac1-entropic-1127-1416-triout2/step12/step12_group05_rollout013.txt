def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5

    n = 16384

    best_score = float('inf')
    best_seq = None

    # Pre-evaluate baseline families (product family alpha approx 0.5)
    alphas = np.linspace(0.1, 3.0, 20) # focus region near 0.5 but maybe optimum
    for alpha in alphas:
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score: update

    # Random restarts of Adam:
    restarts = 0
    max_restarts = 8
    while time.time() < deadline - 100 and restarts < max_restarts:
        # random init via mixture of families
        choice = np.random.randint(5)
        if choice == 0:
            alpha = np.random.uniform(0.0, 3.0)
            a0 = symmetric_product_family(n, alpha)
        elif choice == 1:
            beta = np.random.uniform(0.0, 6.0)
            a0 = left_heavy_power_family(n, beta)
        elif choice == 2:
            gamma = np.random.uniform(0.0, 4.0)
            a0 = exp_u_family(n, gamma)
        elif choice == 3:
            p = np.random.uniform(0.1, 2.0)
            q = np.random.uniform(0.1, 2.0)
            a0 = beta_family(n, p, q)
        else:
            lam = np.random.rand()
            alpha = np.random.uniform(0.0, 3.0)
            a0 = lam * np.full(n, 1.0/n) + (1 - lam) * symmetric_product_family(n, alpha)
            a0 = normalize(a0)

        # Optionally apply random shift (zero padding left)

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a, a)) / (sum(a)^2)

The algorithm is a multi‑stage global‑local search:

1.  Initialise with a collection of analytic families (product‑power,
    left‑heavy power, exponential‑U, beta‑type, uniform‑U mixture) on a
    moderate grid and keep the best candidate.
2.  Run several fresh restarts of a smooth‑max Adam optimiser
    (log‑sum‑exp surrogate of the max‑convolution).  The optimiser works
    on the simplex via a log‑softmax parametrisation.
3.  Refine the current best with a cheap “coordinate‑descent” that moves
    mass from the indices contributing most to the current convolution
    peak towards the smallest entries.  This uses only O(n) work per
    iteration and aggressively flattens the convolution.
4.  Apply a final simulated‑annealing hill‑climb on the exact objective
    to escape any remaining local traps.
5.  Return the best sequence found (as plain ``list[float]``).

All steps respect the provided time budget (default 1000 s).  The code
uses only NumPy and the already‑imported ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# Core convolution utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution via real FFT.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)^2).
    a is assumed non‑negative; the function guards against a near‑zero sum.
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s

# Analytic families (all produce a sum‑1 vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} (the “product‑power” family)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} (left‑heavy power law)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Discrete Beta‑type: a_i ∝ (i+½)^{p‑1} (n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w

def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 400_000,
    beta_start: float = 5.0,
    beta_end: float = 8000.0,
    lr: float = 0.012,
    sym_interval: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    No forced symmetry unless ``sym_interval`` > 0.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # log‑softmax parametrisation (guarantees positivity, sum=1)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    a_cur = np.exp(z) / np.exp(z).sum()
    best_a = a_cur.copy()
    best_score = compute_score(a_cur)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule for the smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z) / np.exp(z).sum()

        if sym_interval and (it % sym_interval == 0):
            # optional symmetrisation
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights (softmax over the convolution entries)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective check
        if it % 3000 == 0:
            a_tmp = np.exp(z) / np.exp(z).sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final evaluation
    a_fin = np.exp(z) / np.exp(z).sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score

def coord_descent_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 800_000,
    step_factor: float = 0.12,
) -> Tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of mass from the index that contributes most
    to the current convolution peak towards the smallest coefficient.
    This is O(n) per iteration and aggressively flattens the conv.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # pre‑compute a few useful arrays
    idx = np.arange(n)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # contribution of each a_i to conv[k]
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[j[mask]]

        i_high = int(np.argmax(contrib))
        # pick a low‑mass index – preferably the global minimum
        i_low = int(np.argmin(a))
        if i_low == i_high:
            # a different low index if they coincide
            i_low = (i_low + 1) % n

        delta = step_factor * a[i_high]
        if delta <= 0.0:
            step_factor *= 0.9
            continue

        a[i_high] -= delta
        a[i_low] += delta

        # keep feasibility
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] coord‑descent improved to {best_score:.6f}")
                # gradually tighten the step as we improve
                step_factor = max(step_factor * 0.92, 5e-3)

    return best_a, best_score

def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 6_000_000,
    temp_factor: float = 1e-3,
    step_scale_init: float = 0.22,
) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the exact objective.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

        ("left‑heavy", lambda b: left_heavy_power_family(n, b)),
        ("exp‑U", lambda c: exp_u_family(n, c)),
        ("beta", lambda d: beta_family(n, d[0], d[1])),
        ("mix‑U", lambda e: mixed_uniform_u_family(n, e[0], e[1])),
    ]

    # narrow but dense grids around the empirically good region
    for name, maker in families:
        if name == "product":
            alphas = np.linspace(0.0, 3.0, 31)
            for a in alphas:
                if time.time() >= deadline - 30.0:
                    break
                vec = maker(a)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best ({name} α={a:.3f}) = {best_score:.6f}")
        elif name == "left‑heavy":
            betas = np.linspace(0.0, 6.0, 31)
            for b in betas:
                if time.time() >= deadline - 30.0:
                    break
                vec = maker(b)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best ({name} β={b:.3f}) = {best_score:.6f}")
        elif name == "exp‑U":
            gammas = np.linspace(0.0, 4.0, 21)
            for g in gammas:
                if time.time() >= deadline - 30.0:
                    break
                vec = maker(g)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best ({name} γ={g:.3f}) = {best_score:.6f}")
        elif name == "beta":
            for _ in range(200):
                if time.time() >= deadline - 30.0:
                    break
                p = np.random.uniform(0.2, 3.0)
                q = np.random.uniform(0.2, 3.0)
                vec = maker((p, q))
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best (beta p={p:.2f},q={q:.2f}) = {best_score:.6f}")
        else:  # mix‑U
            for _ in range(150):
                if time.time() >= deadline - 30.0:
                    break
                lam = np.random.rand()
                gamma = np.random.uniform(0.0, 4.0)
                vec = maker((lam, gamma))
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Random dense search over the same families (≈1200 trials).
    # ------------------------------------------------------------------
    trials = 0
    max_trials = 1200
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(5)
        if choice == 0:          # product
            a = np.random.uniform(0.0, 3.0)
            vec = symmetric_product_family(n, a)
        elif choice == 1:        # left‑heavy
            b = np.random.uniform(0.0, 6.0)
            vec = left_heavy_power_family(n, b)
        elif choice == 2:        # exp‑U
            g = np.random.uniform(0.0, 4.0)
            vec = exp_u_family(n, g)
        elif choice == 3:        # beta
            p = np.random.uniform(0.2, 3.0)
            q = np.random.uniform(0.2, 3.0)
            vec = beta_family(n, p, q)
        else:                    # mixture uniform‑U
            lam = np.random.rand()
            gamma = np.random.uniform(0.0, 4.0)
            vec = mixed_uniform_u_family(n, lam, gamma)

        # occasional random right‑shift
        if np.random.rand() < 0.25:
            d = np.random.randint(0, n // 20 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam refinements (several fresh restarts).
    # ------------------------------------------------------------------
    restarts = 0
    max_restarts = 6
    while time.time() < deadline - 150.0 and restarts < max_restarts:
        restarts += 1
        # initialise from a random perturbation of the current best
        base = np.array(best_seq, dtype=np.float64) if best_seq is not None else np.full(n, 1.0 / n)
        noise = np.random.normal(scale=0.02, size=n)
        a0 = base + noise
        a0 = normalize(a0)

        print(f"[{time.time():.1f}] Adam restart {restarts}")

        a_opt, sc_opt = adam_smooth_optimize(
            start_a=a0,
            deadline=deadline - 80.0,
            max_iters=350_000,
            beta_start=5.0,
            beta_end=8000.0,
            lr=0.012,
            sym_interval=0,
        )

        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] Adam restart {restarts} gave new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Aggressive coordinate‑descent flattening.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] starting coordinate‑descent")
        a_cd, sc_cd = coord_descent_optimize(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline - 30.0,
            max_iters=600_000,
            step_factor=0.15,
        )
        if sc_cd < best_score:
            best_score, best_seq = sc_cd, a_cd.tolist()
            print(f"[{time.time():.1f}] coord‑descent improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=5_000_000,
            temp_factor=2e-3,
            step_scale_init=0.22,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Fallback to a uniform vector if nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 8.  Final official evaluation (includes all security checks).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
