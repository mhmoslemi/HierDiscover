def smooth_obj_and_grad(z, beta, n):
    # stable softmax for a
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / exp_z.sum()
    conv = convolution_fft(a, a)
    maxc = conv.max()
    # stable softmax for conv
    exp_w = np.exp(beta * (conv - maxc))
    w = exp_w / exp_w.sum()
    # gradient w.r.t a
    grad_a = 2.0 * convolution_fft(a[::-1], w)[n-1:n-1+n]
    dot_ag = np.dot(a, grad_a)

"""
Propose a sequence of non‑negative heights that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where the convolution is aperiodic.  The algorithm is a multi‑stage
search:

1.  Coarse scan of a single‑parameter “product’’ family
    a_i ∝ [(i+½)(n‑½‑i)]^{‑α}.  The best α gives a good starting point.
2.  Gradient‑based refinement of the exact smooth‑max surrogate using
    L‑BFGS‑B (SciPy).  β (the smooth‑max temperature) is increased in
    stages, the variables are the log‑softmax parameters *z* (so that
    positivity and the simplex constraint are always satisfied.
3.  A few random restarts of the L‑BFGS phase to escape local minima.
4.  Projected sub‑gradient descent on the true max‑objective (few
    hundred thousand steps – cheap because the vector length is moderate).
5.  Final simulated‑annealing hill‑climb on the simplex to polish the
    solution.
All steps respect the given time budget (default 1000 s).  The function
returns the best sequence found as a plain ``list[float]``.
"""

import time
import math
from typing import List, Tuple

import numpy as np
from scipy.optimize import minimize  # type: ignore


# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²) (sum(a) is assumed >0)."""
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# ----------------------------------------------------------------------
# Analytic families (used for the coarse scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    # log‑product is safe because i+0.5>0
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Smooth‑max surrogate for L‑BFGS
# ----------------------------------------------------------------------


def smooth_obj_and_grad(z: np.ndarray, beta: float, n: int) -> Tuple[float, np.ndarray]:
    """
    Smooth approximation of the objective and its gradient w.r.t. the
    log‑softmax variables *z*.
    Returns (obj, grad_z) where obj = 2·n·smooth_max(conv) and grad_z
    is the Euclidean gradient of obj.
    """
    # a = softmax(z) – always positive and sums to 1
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / exp_z.sum()

    # convolution of a with itself
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # stable soft‑max over the convolution values
    exp_w = np.exp(beta * (conv - maxc))
    w = exp_w / exp_w.sum()

    # gradient of the smooth‑max w.r.t. a
    grad_a = 2.0 * convolution_fft(a[::-1], w)[n - 1:n - 1 + n]

    # back‑prop through the soft‑max (a = softmax(z))
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - a * dot_ag

    # smooth‑max value (log‑sum‑exp trick)
    approx_max = maxc + np.log(exp_w.sum()) / beta
    obj = 2.0 * n * approx_max
    grad_z = 2.0 * n * grad_z
    return obj, grad_z


def l_bfgs_optimize(
    z0: np.ndarray,
    beta: float,
    n: int,
    max_iter: int,
    deadline: float,
) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smooth‑max surrogate.
    Returns the final *z* (log‑softmax parameters).
    """

    def fun(z):
        return smooth_obj_and_grad(z, beta, n)

    # SciPy's `minimize` stops after `max_iter` calls to the objective.
    res = minimize(
        fun,
        z0,
        method="L-BFGS-B",
        jac=True,
        options={"maxiter": max_iter, "disp": False},
    )
    # The optimiser may finish early if the time budget is exhausted – we
    # simply return whatever it produced.
    return res.x


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    step0: float = 0.045,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 4_000_000,
    temp_factor: float = 1e-3,
    step_scale_init: float = 0.2,
) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling schedule
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        if np.random.rand() < 0.5:
            # pure product
            alpha = np.random.uniform(0.0, 12.0)
            vec = symmetric_product_family(n, alpha)
        else:
            # mixture of two product families
            alpha1 = np.random.uniform(0.0, 12.0)
            alpha2 = np.random.uniform(0.0, 12.0)
            lam = np.random.rand()
            vec = lam * symmetric_product_family(n, alpha1) + (1 - lam) * symmetric_product_family(
                n, alpha2
            )
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  L‑BFGS refinement on a smooth‑max surrogate.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] starting L‑BFGS refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        # initialise log‑softmax parameters
        z = np.log(a_start + 1e-30)

        # Increasing sequence of β (smooth‑max temperature)
        beta_schedule = [50.0, 200.0, 800.0, 3000.0, 8000.0, 20000.0]
        for beta in beta_schedule:
            if time.time() >= deadline - 30.0:
                break
            max_iter_stage = 150
            z = l_bfgs_optimize(z, beta, n, max_iter_stage, deadline)
            a = np.exp(z - np.max(z))
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved to {best_score:.6f}")

        # a few random restarts
        for restart in range(3):
            if time.time() >= deadline - 30.0:
                break
            rand_vec = np.random.rand(n)
            rand_vec = rand_vec / rand_vec.sum()
            z_rand = np.log(rand_vec + 1e-30)
            for beta in [200.0, 800.0, 3000.0]:
                if time.time() >= deadline - 30.0:
                    break
                z_rand = l_bfgs_optimize(z_rand, beta, n, 120, deadline)
            a = np.exp(z_rand - np.max(z_rand))
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] L‑BFGS restart {restart} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Projected sub‑gradient descent (exact max objective).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline - 20.0,
            max_iters=300_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final simulated‑annealing hill‑climb (polish).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=4_000_000,
            temp_factor=5e-4,
            step_scale_init=0.2,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Fallback to a safe uniform vector if nothing succeeded.
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 8.  Final official evaluation and return.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
