"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

with a 1000 s wall‑clock budget.

Strategy
--------
* Use a modest length (n≈4096) for the cheap global search.
* Scan several analytic families (product‑power, left‑heavy power,
  exponential‑U, beta, symmetric‑U‑power) on a random grid,
  optionally applying a random right‑shift (zero‑padding).
* Keep the best candidate and up‑sample it to a larger size
  (n≈8192) for a high‑precision refinement.
* Refine the up‑sampled vector with two gradient‑based phases:
  – Adam optimiser on a smooth‑max surrogate of the max‑convolution.
  – Projected sub‑gradient descent on the exact max‑convolution.
* Finish with a short simulated‑annealing hill‑climb.
* All phases respect the global deadline; the current best vector
  is always returned.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities (fast FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)^2) .
    a is assumed non‑negative and non‑zero sum.
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Discrete beta‑type density on the grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


def symmetric_power_family(n: int, beta: float) -> np.ndarray:
    """Symmetric U‑shape with power‑law decay on both sides."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-beta * np.log(i + 0.5))
    right = np.exp(-beta * np.log(n - 0.5 - i))
    w = left + right
    w /= w.sum()
    return w


def symmetric_power_mix_family(n: int, lam: float, beta: float) -> np.ndarray:
    """Mixture of uniform and the symmetric power‑law family."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    power = symmetric_power_family(n, beta)
    w = lam * uniform + (1.0 - lam) * power
    w /= w.sum()
    return w

# Optimisers (same as in the reference implementation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 10.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.012,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of max(conv).
    If ``sym_interval``>0 the iterate is symmetrised every that many steps.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)           # log‑softmax parametrisation
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z) / np.exp(z).sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule for the smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z) / np.exp(z).sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 2000 == 0:
            a_tmp = np.exp(z) / np.exp(z).sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = np.exp(z) / np.exp(z).sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 8_000_000,
                  temp_factor: float = 1e-3,
                  step_scale_init: float = 0.2) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to up‑sample a small vector to a larger length
# ----------------------------------------------------------------------


def upscale_sequence(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to target_n, then renormalise."""
    src_n = seq.shape[0]
    if src_n == target_n:
        return seq.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, target_n)
    up = np.interp(x_tgt, x_src, seq)
    return normalize(up)


# ----------------------------------------------------------------------
# Main search driver
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000,
                     seed: int = 12345,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence minimising
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found (plain list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # a few seconds safety margin

    best_seq: List[float] = []
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Optional fallback (previous best if available)
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Fast global random search on a moderate size
    # -----------------------------------------------------------
    n_small = 4096
    families = [
        ("product", lambda: symmetric_product_family(n_small,
                                                     np.random.uniform(0.0, 30.0))),
        ("left_power", lambda: left_heavy_power_family(n_small,
                                                       np.random.uniform(0.0, 20.0))),
        ("exp_u", lambda: exp_u_family(n_small,
                                       np.random.uniform(0.0, 12.0))),
        ("beta", lambda: beta_family(n_small,
                                     np.random.uniform(0.1, 6.0),
                                     np.random.uniform(0.1, 6.0))),
        ("sym_power", lambda: symmetric_power_family(n_small,
                                                     np.random.uniform(0.0, 20.0))),
        ("sym_power_mix", lambda: symmetric_power_mix_family(
            n_small,
            np.random.rand(),
            np.random.uniform(0.0, 20.0))),
        ("mix_unif_u", lambda: mixed_uniform_u_family(
            n_small,
            np.random.rand(),
            np.random.uniform(0.0, 12.0))),
    ]

    trials = 0
    max_trials = 12_000  # generous, loop will stop earlier by deadline
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        name, gen = families[np.random.randint(len(families))]
        vec = gen()
        # occasional right‑shift (zero‑padding) to break symmetry
        if np.random.rand() < 0.25:
            d = np.random.randint(0, n_small // 20 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] trial #{trials} ({name}) improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 2. Short local refinement of the best candidate (still small n)
    # -----------------------------------------------------------
    if best_seq and time.time() < deadline - 120.0:
        # a quick Adam run on the small vector to shake off obvious
        # irregularities before up‑sampling
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=a_start,
            deadline=deadline - 110.0,
            max_iters=200_000,
            beta_start=5.0,
            beta_end=4000.0,
            lr=0.014,
            sym_interval=0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] post‑small‑Adam improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Upscale to a large size for high‑precision refinement
    # -----------------------------------------------------------
    n_large = 8192
    if best_seq and time.time() < deadline - 80.0:
        a_big = upscale_sequence(np.array(best_seq, dtype=np.float64), n_large)
        sc_big = compute_score(a_big)
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.tolist()
            print(f"[{time.time():.1f}] after up‑scale score = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Adam smooth‑max optimisation (full resolution)
    # -----------------------------------------------------------
    if best_seq and time.time() < deadline - 40.0:
        a_start = np.array(best_seq, dtype=np.float64)
        # allow a bit more iterations than the reference code
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=a_start,
            deadline=deadline - 30.0,
            max_iters=1_000_000,
            beta_start=10.0,
            beta_end=12000.0,
            lr=0.010,
            sym_interval=2000,          # gentle symmetrisation helps stability
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] Adam (large) improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Projected sub‑gradient descent (exact max‑conv)
    # -----------------------------------------------------------
    if best_seq and time.time() < deadline - 15.0:
        a_sub, sc_sub = subgrad_optimize(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline - 10.0,
            max_iters=2_500_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] Sub‑grad final improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Final short simulated‑annealing hill‑climb
    # -----------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=6_000_000,
            temp_factor=1e-3,
            step_scale_init=0.2,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA final improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Fallback to a plain uniform vector if everything failed
    # -----------------------------------------------------------
    if not best_seq:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)   # includes all security checks
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return best_seq
