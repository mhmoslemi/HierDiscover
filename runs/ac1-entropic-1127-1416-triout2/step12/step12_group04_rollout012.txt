def smoothed_gradient_descent(start_a, deadline, beta_schedule, iters_per_beta=5000, eta0=0.2):
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    for idx, beta in enumerate(beta_schedule):
        for t in range(1, iters_per_beta+1):
            if time.time() >= deadline:
                return best_a, best_score

            conv = convolution_fft(a, a)  # length 2n-1
            maxc = conv.max()

"""
search for a step‑function (list of non‑negative heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

by a multi‑phase strategy:

0. try a fallback sequence if it exists.
1. exhaustive scan of analytic families (including a new “center‑heavy”
   family that the previous runner missed).
2. exponentiated‑gradient (mirror‑descent) on the true max‑objective.
3. smoothed‑gradient descent on a log‑sum‑exp approximation of the
   max (helps to escape plateaus).
4. many LP‑based refinement steps with a decaying mixing weight.
5. a final stochastic hill‑climb that perturbs the log‑weights.
All phases respect the global wall‑clock budget (default 1000 s)
and the best vector found is returned (normalised to sum = 1).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast for many calls)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation (a⋆w) using FFT. Result length = len(a)+len(w)-1."""
    m = a.size + w.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    C = A * np.conj(W)
    corr = np.fft.irfft(C, n=nfft)[:m]
    return corr


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: w_i ∝ ((i+0.5)*(n‑0.5‑i))^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def center_heavy_family(n: int, alpha: float) -> np.ndarray:
    """Opposite of product_family: w_i ∝ ((i+0.5)*(n‑0.5‑i))^{α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing: w_i ∝ (i+0.5)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential: w_i ∝ exp(‑γ·dist(i,centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian (centre heavy)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, window_type: str) -> np.ndarray:
    """Standard cosine windows (Hann, Hamming, Blackman)."""
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Heavy left‑side random start."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped start."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 200_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation on the true max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / n)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def smoothed_gradient_descent(start_a: np.ndarray,
                              deadline: float,
                              beta_schedule: list[float],
                              iters_per_beta: int = 2_000,
                              eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the log‑sum‑exp surrogate of the max.
    Uses exponentiated‑gradient style updates for non‑negativity.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in beta_schedule:
        for t in range(1, iters_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_score

            conv = convolution_fft(a, a)
            maxc = conv.max()
            # probabilities proportional to exp(beta*conv)
            w = np.exp(beta * (conv - maxc))          # stabilised
            pw = w / w.sum()                          # length 2n‑1

            # gradient of the smooth objective:
            #   g = 2 * (a ⋆ pw)   (cross‑correlation)
            corr = correlation_fft(a, pw)              # length n+2n‑2
            m_pw = pw.shape[0]
            g = 2.0 * corr[m_pw - 1: m_pw - 1 + a.shape[0]]

            # step size decays with sqrt(iter) and with beta
            eta = eta0 / (math.sqrt(t) * (beta + 1e-12))
            a = a * np.exp(-eta * g)
            s = a.sum()
            a = a / s if s != 0 else np.full_like(a, 1.0 / a.shape[0])

            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        print(f"[{time.time():.1f}] beta {beta:.1f} done, best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (the classic “mixing” step)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.1) -> np.ndarray:
    """
    Solve  max Σ g_j   s.t.   (f * g)_k ≤ M,   g ≥ 0.
    Return the convex mixture (1‑t)·f + t·g_norm  projected to the simplex.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (size m×n) s.t. A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)                # maximise Σ g  <=> minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # HiGHS is fast; give it a modest time budget per solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 12})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Stochastic hill‑climb (Gaussian log‑weight perturbations)
# ----------------------------------------------------------------------


def hill_climb_gaussian(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex in log‑space; accepts every improvement
    and occasionally a worsening move (simulated‑annealing style).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    eps = 1e-12
    # annealing schedule (temperature decays linearly)
    T0 = cur_score * 5e-4
    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        T = T0 * (1.0 - step / max_steps)

        # propose by perturbing log‑weights
        loga = np.log(cur_a + eps)
        cand_log = loga + np.random.randn(n) * 0.015
        cand = np.exp(cand_log)
        cand = cand / cand.sum()

        cand_score = compute_score(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # slowly increase step size when we improve
                # (implemented by a tiny increase of the noise amplitude)
        # optional: every 5000 steps print progress
        if step % 5000 == 0:
            print(f"[{time.time():.1f}] hill‑climb best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns a plain Python list (the vector is automatically normalised to sum = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback (often a decent starting point)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem size
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # 2048 is a good compromise between speed and quality
    n = fallback_len if fallback_len >= 1024 else 2048
    print(f"[{time.time() - start_time:.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # Helper to test a candidate (automatically normalises)
    # ------------------------------------------------------------------
    def test_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            return
        a /= s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Analytic scan (wide but cheap)
    # ------------------------------------------------------------------
    # uniform baseline
    test_candidate(np.full(n, 1.0 / n, dtype=np.float64))

    # product (U‑shaped) family
    alphas = np.concatenate([np.linspace(0.0, 5.0, 41), np.linspace(5.1, 150.0, 30)])
    for α in alphas:
        if time.time() >= deadline - 400.0:
            break
        test_candidate(product_family(n, α))

    # centre‑heavy family (new)
    alphas_c = np.linspace(0.0, 4.0, 21)
    for α in alphas_c:
        if time.time() >= deadline - 400.0:
            break
        test_candidate(center_heavy_family(n, α))

    # left‑heavy power family
    betas = np.linspace(0.0, 30.0, 31)
    for β in betas:
        if time.time() >= deadline - 400.0:
            break
        test_candidate(left_heavy_power_family(n, β))

    # exponential‑U family
    gammas = np.linspace(0.0, 20.0, 41)
    for γ in gammas:
        if time.time() >= deadline - 400.0:
            break
        test_candidate(exp_u_family(n, γ))

    # Gaussian family
    gauss_gammas = np.logspace(-4, 2, 40)
    for γ in gauss_gammas:
        if time.time() >= deadline - 400.0:
            break
        test_candidate(gaussian_family(n, γ))

    # Cosine windows
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 400.0:
            break
        test_candidate(window_family(n, win))

    # Random starts
    for _ in range(200):
        if time.time() >= deadline - 400.0:
            break
        if np.random.rand() < 0.5:
            test_candidate(random_monotone_start(n))
        else:
            test_candidate(random_u_start(n))

    # ------------------------------------------------------------------
    # 3. Normalise the best vector we have so far
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.asarray(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient optimisation (phase 2)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print(f"[{time.time() - start_time:.1f}] Starting EG optimisation")
        a, sc = eg_optimize(a, deadline - 200.0, max_iters=250_000, eta0=0.6)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] after EG best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Smoothed‑gradient descent (phase 3)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        print(f"[{time.time() - start_time:.1f}] Starting smoothed‑gradient descent")
        beta_sched = [1.0, 2.5, 5.0, 10.0, 20.0, 40.0]
        a, sc = smoothed_gradient_descent(a, deadline - 120.0,
                                          beta_schedule=beta_sched,
                                          iters_per_beta=1500,
                                          eta0=0.2)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] after smooth‑GD best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Intensive LP‑based refinement (phase 4)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time() - start_time:.1f}] Starting LP refinement loop")
        mix = 0.12
        no_improve = 0
        while time.time() < deadline - 30.0 and no_improve < 30:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix)
            sc = compute_score(a_new)
            if sc < best_score - 1e-8:
                best_score = sc
                a = a_new
                no_improve = 0
                mix = max(0.03, mix * 0.95)   # slowly shrink step size
                print(f"[{time.time() - start_time:.1f}] LP improved to {best_score:.6f}")
            else:
                no_improve += 1
                mix = max(0.03, mix * 0.85)   # become more cautious
        best_seq = a.copy()

    # ------------------------------------------------------------------
    # 7. Symmetrisation (the problem is symmetric)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time() - start_time:.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final hill‑climb (phase 5)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (Gaussian perturb)")
        a, sc = hill_climb_gaussian(a, deadline, max_steps=1_200_000)
        if sc < best_score:
            best_score = sc
            print(f"[{time.time() - start_time:.1f}] hill‑climb final best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Return the best vector (as a plain Python list)
    # ------------------------------------------------------------------
    final_list = a.tolist()
    final_score = evaluate_sequence(final_list)
    print(f"\n[{time.time() - start_time:.1f}] finished – final reported score = {final_score:.6f}")
    return final_list
