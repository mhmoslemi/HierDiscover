import math
import numpy as np
from scipy.optimize import linprog

def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    # Implementation from Duchi et al.
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = -1
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def symmetric_product_family(n: int, alpha: float, eps: float = 0.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i) + eps
    w = denom ** (-alpha)
    a = w / w.sum()
    return a

def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a

def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a

def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape

        if time.time() >= deadline:

        k = int(np.argmax(conv))

        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]
        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")
    sc = compute_score(a)
    if sc < best_score:

        best_a = a.copy()
    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray,

def lp_refine(a: np.ndarray, M: float, deadline: float):
    n = a.shape[0]
    # Build convolution matrix A (2n-1) x n where A @ g = a * g (full conv)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g) → minimize -sum

    if not res.success:
        print(f"[{time.time():.1f}] LP failed: {res.message}")
        return None
    g = res.x
    return g

"""
Hybrid search for a non‑negative step‑function that minimises the evaluation
function provided by the competition harness.

The algorithm consists of several stages that respect the 1000 s wall‑clock
budget:

1. **Deterministic start** – a smoothed arcsine (symmetric product) shape.
   A fine grid over the exponent α is scanned to obtain the best
   analytically‑motivated seed.

2. **LP refinement** – repeatedly solve a linear programme that maximises
   the total mass of a new distribution g under the current self‑convolution
   bound  M = max(a*a).  The solution is mixed with the current iterate a
   and projected onto the probability simplex.  This is precisely the
   “linear programme” described in the problem statement and yields
   monotone improvements of the objective.

3. **Projected sub‑gradient descent** – a cheap, many‑iteration
   optimisation of the exact max‑convolution objective to polish the
   result.

4. **Final simulated‑annealing hill‑climb** (optional) and a symmetrisation
   step, because the problem is invariant under reversal.

All helper functions are top‑level, there are no closures or lambda
expressions, and the routine prints intermediate scores for transparency.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective (ignoring the normalisation performed by the harness)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def symmetric_product_family(n: int, alpha: float, eps: float = 0.0) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i) + eps)^(-α).
    The classic arcsine distribution corresponds to α=0.5, eps=0.
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i) + eps
    w = denom ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex blend of the uniform distribution and the exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Very cheap per iteration and thus can be run until the wall‑clock limit.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Very simple simulated‑annealing walk on the simplex.
    Accepts uphill moves with a temperature that decays linearly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

            if i != j:

                cand[i] -= delta

    return best_a, best_score


def lp_refine(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme
        maximise   Σ g_i
        subject to a * g ≤ M   (full convolution)
                   g_i ≥ 0 .
    Returns the optimiser g (or None on failure).
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build the Toeplitz‑like matrix that represents the convolution a * g.
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a

    c = -np.ones(n, dtype=np.float64)                # maximise Σ g_i  → minimise -Σ g_i
    b_ub = np.full(rows, M, dtype=np.float64)       # a * g ≤ M
    bounds = [(0, None)] * n

    # HiGHS (the default) is fast and reliable for the problem size.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        print(f"[{time.time():.1f}] LP failed: {res.message}")
        return None
    return res.x

        if g is None:
            break

        # The LP does **not** enforce a unit sum – project to the simplex.
        g = project_to_simplex(g)

        # Mixing weight (aggressive at the beginning, then taper).
        if lp_iter <= 5:
            t = 0.5
        elif lp_iter <= 12:
            t = 0.25
        else:
            t = 0.1

        a_new = (1 - t) * a + t * g
        a_new = project_to_simplex(a_new)
        sc_new = compute_score(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            print(f"[{time.time():.1f}] LP iter {lp_iter} improved to {best_score:.6f}")
        else:
            # try a smaller mixing factor before giving up on this iteration
            improved = False
            for tt in (t / 2, t / 4):
                a_tmp = (1 - tt) * a + tt * g
                a_tmp = project_to_simplex(a_tmp)
                sc_tmp = compute_score(a_tmp)
                if sc_tmp < best_score:
                    best_score = sc_tmp
                    best_a = a_tmp.copy()
                    a = a_tmp.copy()
                    improved = True
                    print(f"[{time.time():.1f}] LP iter {lp_iter} improved with t={tt:.3f}")
                    break
            if not improved:
                print(f"[{time.time():.1f}] LP iter {lp_iter} no improvement")
                if lp_iter > 10:                      # give up early if stagnating
                    break

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient optimisation (cheap, many iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_sub, sc_sub = subgrad_optimize(a, deadline,
                                         max_iters=2_000_000,
                                         step0=0.05)
        if sc_sub < best_score:
            best_score = sc_sub
            best_a = a_sub.copy()
            a = a_sub
            print(f"[{time.time():.1f}] sub‑gradient gave new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing hill‑climb (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_sa, sc_sa = hill_climb_sa(a, deadline, max_steps=500_000)
        if sc_sa < best_score:
            best_score = sc_sa
            best_a = a_sa.copy()
            a = a_sa
            print(f"[{time.time():.1f}] SA gave new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation (the problem is invariant under reversal)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        best_a = a_sym.copy()
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the provided wrapper
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
