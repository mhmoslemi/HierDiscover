def gradient_descent_smooth(a0, deadline, max_iter=10000, alpha_start=2.0, alpha_end=64.0):
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    m = 2*n - 1
    # pre-allocate arrays?
    t = 0
    # Adam parameters
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # schedule for alpha
    total_iters = max_iter
    # adapt when near deadline
    while time.time() < deadline and t < max_iter:
        t += 1
        progress = t / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** progress  # exponential schedule
        # compute conv
        conv = full_convolution(a)  # length m
        conv_max = conv.max()
        # compute softmax weights
        # to avoid overflow: subtract conv_max
        exp_term = np.exp(alpha * (conv - conv_max))
        p = exp_term / exp_term.sum()
        # compute gradient
        grad = compute_gradient_softmax(a, p)  # returns shape n
        # Adam update
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * (grad * grad)
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        # step size schedule: maybe eta = 0.5 / (alpha+1) / n?
        eta = 0.1 / (1 + alpha * 0.01) # placeholder
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a_new = project_to_simplex(a_new)
        # compute actual score
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        # optionally accept a if improvement else keep old?
        a = a_new
    return best_a, best_score

def compute_gradient_softmax(a, p):
    a_rev = a[::-1]
    # compute convolution of a_rev and p (full)
    # length needed: len(a_rev) + len(p) -1 = n + (2n-1) -1 = 3n-2
    # Use FFT-based convolution for speed.
    n = len(a)
    m = len(p)  # = 2n-1
    N = 1 << ((n + m -1 - 1).bit_length())
    A_rev = np.fft.rfft(a_rev, n=N)

    conv_full = np.fft.irfft(A_rev * P, n=N)[: n+m-1]  # may need slicing but ensure length
    # get slice
    start = n-1
    grad = 2.0 * conv_full[start: start + n]

def grad_softmax(a, p):
    n = len(a)
    m = len(p)  # 2n-1
    N = 1 << ((n + m - 1 - 1).bit_length())

"""
Propose a non‑negative step function minimizing `evaluate_sequence`.

Algorithm:
1. Generate a diverse set of initial shapes (uniform, sine, Hann,
   power‑law, geometric, beta, cosine‑power, random Dirichlet,
   piecewise‑constant blocks).
2. For each start run a smooth‑max gradient descent:
   - Objective uses a soft‑max approximation to `max(conv)`.
   - Gradient is computed analytically via FFT‑based convolutions.
   - Adam optimizer with projection onto the probability simplex.
   - The “temperature” α of the soft‑max is increased from a low to
     a high value, turning the smooth objective gradually into the true
     max‑objective.
3. Keep the best vector seen during the descent.
4. Finally apply a greedy stochastic hill‑climb (mass‑transfer moves)
   to polish the solution.
All stages obey a global wall‑clock budget (default 1000 s). The
returned list is a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the simplex (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # If all entries non‑positive, return the uniform distribution.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def gradient_softmax(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns ∇_a ( (1/α) log Σ exp(α conv) ).
    """
    n = a.shape[0]
    # convolution of reversed a with p
    a_rev = a[::-1]
    m = p.shape[0]           # = 2*n-1
    size = n + m - 1          # = 3*n-2
    N = 1 << ((size - 1).bit_length())
    A_rev = np.fft.rfft(a_rev, n=N)
    P = np.fft.rfft(p, n=N)
    conv_full = np.fft.irfft(A_rev * P, n=N)[:size]
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Exact score used by evaluate_sequence (sum a = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)) (assuming sum(a)=1)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Shape families
# ----------------------------------------------------------------------
def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    """a_i = (i+offset)^(-p), normalised to unit sum."""
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, r: float) -> np.ndarray:
    """a_i = r**i, normalised."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, beta: float) -> np.ndarray:
    """a_i = (1 - i/(n-1))**beta, normalised."""
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    a /= a.sum()
    return a


def cosine_power(n: int, expo: float) -> np.ndarray:
    """a_i = cos(π i / (2 n))**expo, normalised."""
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def sin_shape(n: int) -> np.ndarray:
    """Half‑wave sine shape, normalised."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.sin(theta)
    a /= a.sum()
    return a


def hann_shape(n: int) -> np.ndarray:
    """Raised‑cosine (Hann) window, normalised."""
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n, dtype=np.float64) / (n - 1)))
    a /= a.sum()
    return a


def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    """Coarse piecewise‑constant shape with K blocks, normalised."""
    blocks = rng.dirichlet(np.ones(K))
    blocks = np.sort(blocks)[::-1]  # decreasing blocks
    repeats = np.full(K, n // K, dtype=int)
    remainder = n - repeats.sum()
    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)[:n]
    a /= a.sum()
    return a

# ----------------------------------------------------------------------

    """
    Produce a modest diverse collection of feasible vectors.
    """
    rng = np.random.default_rng(123456)

# Smooth‑max gradient descent with Adam
# ----------------------------------------------------------------------
def smooth_max_descent(a0: np.ndarray, deadline: float,
                       max_iters: int = 5000,
                       alpha_start: float = 2.0,
                       alpha_end: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv) by annealing a soft‑max surrogate.
    Returns the best vector found and its exact score.
    """
    a = a0.copy()                     # always on the simplex
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    # Adam accumulator
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    t = 0
    while time.time() < deadline - 0.02 and t < max_iters:
        t += 1
        # linear schedule for the soft‑max temperature α
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iters)

        # convolution a*a
        conv = full_convolution_fft(a)
        conv_max = conv.max()

        # soft‑max weights (stable)
        exp_term = np.exp(alpha * (conv - conv_max))
        p = exp_term / exp_term.sum()

        # gradient of the smooth surrogate
        grad = gradient_softmax(a, p)

        # Adam update
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)

        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)

        # step size decays with α (larger α → more “sharp” landscape)
        eta = 0.15 / (alpha + 1.0)

        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        # exact objective
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        # Always keep the iterate feasible
        a = a_new

    return best_a, best_sc

# Greedy stochastic hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 300_000) -> np.ndarray:
    """Fine‑tune by moving a small fraction of mass between two positions."""
    a = a_start.copy()
    best_a = a.copy()

    n = a.shape[0]
    step_scale = 0.04
    it = 0
    no_improve = 0

        cand = a.copy()

        cand[j] += delta

            jitter = rng.standard_normal(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

"""
Propose a non‑negative step function minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Generate diverse seeds** – uniform, sine, Hann, several parametric
   families (power‑law, geometric, beta, cosine‑power) and random
   Dirichlet / block shapes.
2. **Smooth‑max Adam optimisation** – minimise a
   soft‑max surrogate of ``max(conv(a,a))``.
   * Soft‑max temperature α is annealed from a low to a high value,
     turning the smooth objective gradually into the true max‑objective.
   * The gradient of the surrogate is obtained analytically:
     ∇ₐ ( (1/α) log ∑ exp(α conv) ) = 2 · (a_rev ★ p) where
     p are the soft‑max weights and ★ is a linear convolution.
   * Adam updates are projected onto the probability simplex to keep the
     vector non‑negative and Σa = 1.
   * Several restarts are performed until the time budget is almost
     exhausted.
3. **Final hill‑climb polish** – tiny random mass‑transfer moves
   accepted only if they improve the exact objective.
4. Return the best sequence as a plain Python list; the final
   ``evaluate_sequence`` call reports the official score.

All stages respect the 1000 s wall‑clock limit (default). The code
uses only NumPy and the already‑imported ``evaluate_sequence``.

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a*a
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate
# ----------------------------------------------------------------------
def gradient_softmax(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log ∑exp(α·conv) w.r.t. a.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    m = p.shape[0]                     # = 2·n‑1
    size = n + m - 1                   # = 3·n‑2
    N = 1 << ((size - 1).bit_length())
    A_rev = np.fft.rfft(a_rev, n=N)
    P = np.fft.rfft(p, n=N)
    conv_full = np.fft.irfft(A_rev * P, n=N)[:size]
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (Σa = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)) (assuming sum(a)=1)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, beta: float) -> np.ndarray:
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    a /= a.sum()
    return a

    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.sin(theta)
    a /= a.sum()
    return a
