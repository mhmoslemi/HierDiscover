Pseudo-code for projection:

```
def project_to_simplex(v):
    # v: numpy array
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

The evaluation function is  

\[
J(a)=\frac{2\,n\,\max (a*a)}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the objective is homogeneous we may restrict ourselves to *probability vectors*
(\(\sum_i a_i = 1\)).  
Then  

\[
J(a)=2\,n\; \max_k (a*a)_k .
\]

The problem is non‑convex: the maximum of a convolution is a piece‑wise linear,
non‑smooth functional.  The earlier code uses an exponentiated‑gradient (EG)
sub‑gradient method.  Here we improve the optimisation in three ways

1. **Global exploration** – a denser sweep of parametric families (power‑law,
   exponential, beta) and many Dirichlet draws, keeping the best few candidates.
2. **Smooth continuation with Adam** – we replace the sub‑gradient by the
   gradient of a *soft‑max* (log‑sum‑exp) approximation of the maximum.  
   For a temperature  \(\tau\)

   \[
   f_\tau(a)=\frac{1}{\tau}\log\!\Bigl(\sum_k e^{\tau (a*a)_k}\Bigr)
   \]

   satisfies \(\displaystyle\lim_{\tau\to\infty}f_\tau(a)=\max_k (a*a)_k\) and is
   differentiable.  Its gradient is

   \[
   \nabla_i f_\tau(a)=2\sum_k p_k\,a_{k-i},
   \qquad
   p_k=\frac{e^{\tau (a*a)_k}}{\sum_j e^{\tau (a*a)_j}} .
   \]

   The gradient is obtained with two FFT‑based convolutions ( \(O(n\log n)\) ).
   We run Adam on this smooth objective while **continuously increasing**
   \(\tau\) (a standard continuation / “homotopy” strategy).  The Adam
   steps are projected onto the simplex after each update.
3. **Final EG polishing** – after the smooth phase we fall back to a few
   thousand EG iterations to clean up any remaining small spikes.

All steps respect the simplex constraint, stay within the 1000 s wall‑clock
budget, and print progress information.  The returned list of floats is a
valid candidate for `evaluate_sequence`.

```python
"""
Improved optimiser for ``evaluate_sequence``

* Global sweep over a few parametric families + many Dirichlet draws.
* Continuation optimisation:
    – smooth approximation of max‑convolution via log‑sum‑exp,
    – Adam updates with projection onto the probability simplex,
    – temperature τ is increased slowly (τ‑continuation).
* Short EG polishing at the end.

The algorithm works entirely with NumPy, uses only FFT‑based convolutions
(O(n log n) per iteration) and stays well inside the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic DSP utilities
# ----------------------------------------------------------------------


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of optimal length."""
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())           # next power of two ≥ n
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)
    return conv[:n]                               # discard zero‑padding tail


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (both arguments have the same length)."""
    return fft_convolve(a, a)


# ----------------------------------------------------------------------
# Objective on the simplex (∑a = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x≥0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008) – O(n log n).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the global sweep
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta: a_i ∝ (i+1)^{α‑1}·(n−i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration (baseline)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 5):
    """Return up to ``max_keep`` vectors with the smallest exact score."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s                     # simplex projection
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep (dense)
    for p in np.linspace(0.05, 3.0, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep (dense)
    for lam in np.linspace(0.00005, 0.08, 350):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep (medium)
    al_vals = np.linspace(0.2, 6.0, 30)
    be_vals = np.linspace(0.2, 6.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) random Dirichlet draws with several concentration values
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1500          # ≈ 7 500 draws total (still cheap)
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    # keep only the vectors (discard scores)
    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth continuation optimiser (Adam + temperature schedule)
# ----------------------------------------------------------------------


def smooth_adam_optimize(a0: np.ndarray,
                         deadline: float,
                         tau_start: float = 0.5,
                         tau_end: float = 800.0,
                         tau_mul: float = 1.015,
                         alpha0: float = 0.45,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimisation of the log‑sum‑exp approximation of max(a*a)
    while the temperature τ is increased (continuation).

    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_val = compute_score(a)

    tau = tau_start
    while time.time() < deadline:
        t += 1

        # ---- convolution a*a -------------------------------------------------
        conv = full_convolution(a)                 # length 2n‑1
        max_c = conv.max()

        # ---- soft‑max weights ------------------------------------------------
        # use (c - max_c) for numerical stability
        exp_terms = np.exp(tau * (conv - max_c))
        weight_sum = exp_terms.sum()
        if weight_sum == 0.0:                      # should never happen
            break
        p = exp_terms / weight_sum                 # size 2n‑1, sums to 1

        # ---- gradient  g_i = 2 * Σ_k p_k a_{k-i} ---------------------------
        a_rev = a[::-1]                            # reversed a
        rev_conv = fft_convolve(a_rev, p)          # length 3n‑2
        # slice indices n‑1 … n‑1+n‑1 (inclusive) give the needed values
        g = 2.0 * rev_conv[n - 1: n - 1 + n]

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = (alpha0 / (np.sqrt(v_hat) + eps)) * m_hat
        a = a - step
        a = project_to_simplex(a)

        # ---- keep best -------------------------------------------------------
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasional log
            if t % 200 == 0:
                print(f"[smooth‑adam] iter={t:,} τ={tau:.2f} score={cur_val:.6f}")

        # ---- raise temperature ------------------------------------------------
        tau = min(tau * tau_mul, tau_end)

        # safety: stop if τ has reached its maximum and there is no improvement
        if tau >= tau_end and time.time() > deadline - 2.0:
            break

    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (optional final sweep)
# ----------------------------------------------------------------------


def eg_polish(a0: np.ndarray, deadline: float,
              eta0: float = 0.7, tol: float = 1e-10) -> np.ndarray:
    """Classic EG scheme on the original max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * tol
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient (active set)
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # simplex projection via normalisation (EG keeps positivity)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 100 == 0:
                print(f"[eg‑polish] iter={it:,} score={cur_val:.6f}")

        # occasional jitter to escape plateaux
        if it % 800 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a

    """

    # -----------------------------------------------------------------

    fallback = globals().get("height_sequence_1")

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
```
