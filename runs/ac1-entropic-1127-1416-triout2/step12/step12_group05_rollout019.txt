"""
Propose a candidate step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

The strategy combines a broad search over Beta‑type families (which give
U‑shaped, edge‑heavy distributions that flatten the auto‑convolution),
follow‑up up‑sampling to a larger resolution and two stages of
gradient‑based refinement (a smooth‑max Adam optimiser and a projected
sub‑gradient descent).  The whole search respects the 1000 s wall‑clock
budget and returns the best vector found as a plain ``list[float]``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities (same as the reference implementation)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)^2).
    a is assumed non‑negative and normalised (sum>0).
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# ----------------------------------------------------------------------
# Analytic families used for the initial random search
# ----------------------------------------------------------------------


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Optimisers (Adam on a smooth‑max surrogate and a sub‑gradient method)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         beta_start: float = 10.0,
                         beta_end: float = 3000.0,
                         lr: float = 0.012,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate of max(conv)."""
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)        # log‑softmax parametrisation

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z) / np.exp(z).sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule for the smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z) / np.exp(z).sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # grad of the smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 2000 == 0:
            a_tmp = np.exp(z) / np.exp(z).sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = np.exp(z) / np.exp(z).sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 4_000_000,
                  temp_factor: float = 1e-3,
                  step_scale_init: float = 0.15) -> Tuple[np.ndarray, float]:
    """Simulated‑annealing hill‑climb on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # linear temperature schedule
    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine performs a fast exploration of Beta‑type families (including
    strongly edge‑heavy shapes), up‑samples the best candidate to a finer
    resolution (n=8192) and then refines it with Adam and sub‑gradient
    optimisation before a final short simulated‑annealing hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 0. Small‑scale (n_small) search over flexible families
    # ------------------------------------------------------------------
    n_small = 2048
    best_seq_small = None
    best_score_small = float("inf")

    # uniform baseline
    uniform = np.full(n_small, 1.0 / n_small, dtype=np.float64)
    sc = compute_score(uniform)
    best_seq_small = uniform
    best_score_small = sc
    print(f"[{time.time():.1f}] uniform score = {sc:.6f}")

    # exponential‑U family (coarse grid)
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline:
            break
        vec = exp_u_family(n_small, gamma)
        sc = compute_score(vec)
        if sc < best_score_small:
            best_seq_small = vec
            best_score_small = sc
            print(f"[{time.time():.1f}] exp‑U γ={gamma:.3f} → {sc:.6f}")

    # left‑heavy power family (coarse grid)
    for beta in np.linspace(0.0, 12.0, 31):
        if time.time() >= deadline:
            break
        vec = left_heavy_power_family(n_small, beta)
        sc = compute_score(vec)
        if sc < best_score_small:
            best_seq_small = vec
            best_score_small = sc
            print(f"[{time.time():.1f}] left‑β={beta:.3f} → {sc:.6f}")

    # detailed Beta‑type grid (including edge‑heavy parameters <1)
    p_grid = np.concatenate([np.linspace(0.05, 1.0, 20), np.linspace(1.0, 3.0, 10)])
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            if time.time() >= deadline:
                break
            vec = beta_family(n_small, p, q)
            sc = compute_score(vec)
            if sc < best_score_small:
                best_seq_small = vec
                best_score_small = sc
                print(f"[{time.time():.1f}] beta p={p:.3f}, q={q:.3f} → {sc:.6f}")

    # random dense search in the Beta family (focus on p,q<1.5)
    trials = 0
    max_trials = 3000
    while time.time() < deadline and trials < max_trials:
        trials += 1
        p = np.random.uniform(0.02, 1.5)
        q = np.random.uniform(0.02, 1.5)
        vec = beta_family(n_small, p, q)

        # occasional random right‑shift to break symmetry
        if np.random.rand() < 0.2:
            d = np.random.randint(0, n_small // 15 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
        vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score_small:
            best_seq_small = vec
            best_score_small = sc
            print(f"[{time.time():.1f}] random β‑trial #{trials} (p={p:.3f},q={q:.3f}) → {sc:.6f}")

    # ------------------------------------------------------------------
    # 1. Upsample the best small‑scale vector to the full resolution
    # ------------------------------------------------------------------
    n_large = 8192
    x_small = np.arange(n_small, dtype=np.float64)
    x_large = np.linspace(0, n_small - 1, n_large)
    upsampled = np.interp(x_large, x_small, best_seq_small)
    upsampled = normalize(upsampled)

    best_seq = upsampled
    best_score = compute_score(best_seq)
    print(f"[{time.time():.1f}] after up‑sampling score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam refinement on a smooth‑max surrogate (no enforced symmetry)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] Adam refinement start")
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=best_seq,
            deadline=deadline - 120.0,
            max_iters=400_000,
            beta_start=10.0,
            beta_end=2500.0,
            lr=0.012,
            sym_interval=0,
        )
        if sc_opt < best_score:
            best_seq = a_opt
            best_score = sc_opt
            print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent on the exact max‑objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] Sub‑gradient descent start")
        a_sub, sc_sub = subgrad_optimize(
            start_a=best_seq,
            deadline=deadline - 30.0,
            max_iters=1_200_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_seq = a_sub
            best_score = sc_sub
            print(f"[{time.time():.1f}] Sub‑grad improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final short simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=best_seq,
            deadline=deadline,
            max_steps=3_000_000,
            temp_factor=1e-3,
            step_scale_init=0.15,
        )
        if sc_final < best_score:
            best_seq = a_final
            best_score = sc_final
            print(f"[{time.time():.1f}] SA improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Official evaluation (includes all security checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq.tolist()
