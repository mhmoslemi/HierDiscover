"""
Propose a step‑function sequence that (approximately) minimizes
``evaluate_sequence``.  The algorithm proceeds as follows:

1. **Analytic start** – a one‑parameter symmetric product family
   a_i ∝ [(i+½)(n‑½‑i)]^{‑α} is optimised by a high‑precision
   golden‑section search (α ∈ [0,12]).

2. **Linear‑programming improvement** – given the current vector f,
   solve the LP  

        maximise   Σ g_i
        subject to  g_i ≥ 0,
                    (f * g)[k] ≤ M   for all k,
   where M = max (f * f).  The optimal g has a larger ℓ₁‑norm than f;
   mixing f with the normalised g (with a small step size) provably
   reduces the objective.  This step is repeated while time permits.

3. **Smooth‑max Adam refinement** – a soft‑max surrogate of the max‑conv
   is minimised with Adam (exponential temperature schedule from β≈10
   to β≈12000).  Every few thousand steps the vector is symmetrised and
   projected back onto the probability simplex.

4. **Final symmetrisation & evaluation** – the best vector found is
   symmetrised once more, projected onto the simplex, and returned as a
   plain Python list.

All heavy work respects the supplied wall‑clock budget (default 1000 s)
and the code works with only ``numpy`` and ``scipy`` (optional LP).  The
search is deliberately different from the previous submission, using a
deterministic LP‑based mixing step that often yields a lower bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        score = 2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the initial guess)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric product family:
        a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    # log of the product (i+0.5)*(n-0.5-i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against under‑flow/overflow
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Golden‑section search (1‑D)
# ----------------------------------------------------------------------


def golden_section_optimize(func, a: float, b: float, max_iter: int = 30):
    """Minimise `func` on [a,b] using golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

# Smoothed‑max Adam optimiser (used after the LP stage)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 800_000,
                         beta_start: float = 10.0,
                         beta_end: float = 12000.0,
                         lr: float = 0.015,
                         sym_interval: int = 1500):
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector found and its exact score.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score

def lp_mix_step(f: np.ndarray, t: float, deadline: float):

    Perform a single LP‑based mixing step.
    Returns the updated vector and its score, or (None, inf) on failure.
    """

                      method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
    except Exception as e:
        print(f"LP failed to start: {e}")
        return None, float('inf')

    if not res.success:
        print(f"LP solve unsuccessful: {res.message}")
        return None, float('inf')

    g_raw = res.x
    sum_g = g_raw.sum()
    if sum_g <= 1e-12:
        print("LP returned zero vector")
        return None, float('inf')

    g = g_raw / sum_g          # normalise to sum 1
    f_new = (1.0 - t) * f + t * g
    f_new = (f_new + f_new[::-1]) * 0.5   # enforce symmetry
    f_new = project_to_simplex(f_new)

    new_score = compute_score(f_new)
    return f_new, new_score


# ----------------------------------------------------------------------
# Main driver required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found (plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # spare margin for final checks

    # ------------------------------------------------------------------
    # 0. Basic parameters
    # ------------------------------------------------------------------
    n = 1024                     # moderate size – good trade‑off for LP
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Initialise with the symmetric product family (golden‑section)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Starting golden‑section on product family (n={n})")

    def score_alpha(a):
        vec = symmetric_product_family(n, a)
        return compute_score(vec)

    α_opt, α_sc = golden_section_optimize(score_alpha, 0.0, 12.0, max_iter=30)
    f = symmetric_product_family(n, α_opt)
    best_score = compute_score(f)
    best_seq = f.tolist()
    print(f"[{time.time():.1f}] product‑α={α_opt:.6f} → score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Repeated LP mixing steps (if SciPy is available)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog  # noqa: F401
        scipy_lp = True
    except Exception:
        scipy_lp = False
        print("[warning] SciPy linprog not available – skipping LP stage")

    if scipy_lp:
        t = 0.20                 # mixing proportion
        lp_iter = 0
        while time.time() < deadline - 60 and lp_iter < 15:
            lp_iter += 1
            result = lp_mix_step(f, t, deadline)
            if result[0] is None:
                print(f"[{time.time():.1f}] LP step {lp_iter} failed – aborting LP stage")
                break
            f_new, sc_new = result
            print(f"[{time.time():.1f}] LP step {lp_iter}  score = {sc_new:.6f}")
            if sc_new < best_score - 1e-8:
                best_score = sc_new
                f = f_new
                best_seq = f.tolist()
                # optionally tighten mixing as we approach a good point
                t = max(0.05, t * 0.9)
            else:
                # no improvement – reduce step size and retry
                t *= 0.7
                if t < 1e-3:
                    print("[info] step size too small, exiting LP loop")
                    break

    # ------------------------------------------------------------------
    # 3. Adam smooth‑max refinement (more iterations than the previous code)
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        print(f"[{time.time():.1f}] Adam refinement started")
        a_start = np.array(f, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 30,
            max_iters=800_000,
            beta_start=10.0,
            beta_end=12000.0,
            lr=0.015,
            sym_interval=1500,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            f = a_opt
            best_seq = f.tolist()
            print(f"[{time.time():.1f}] Adam refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final symmetrisation & evaluation through the official wrapper
    # ------------------------------------------------------------------
    f = (f + f[::-1]) * 0.5
    f = project_to_simplex(f)
    final_score = evaluate_sequence(f.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return f.tolist()
