```
def adam_optimize_logit(logits, deadline, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(logits)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = None
    best_val = float('inf')
    while time.time() < deadline:
        t += 1
        # compute probabilities
        max_logit = np.max(logits)  # for numerical stability
        exp_logits = np.exp(logits - max_logit)
        a = exp_logits / exp_logits.sum()
        # compute conv
        conv = full_convolution(a)
        max_c = conv.max()
        # maybe early exit if time nearing
        # active set
        eps = max_c * 1e-12
        active = np.where(conv >= max_c - eps)[0]
        # compute subgradient g (as above)
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            i_idx = np.arange(i_start, i_end+1)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]
        # compute avg = sum_i a_i * g_i
        avg = np.dot(a, g)
        # gradient wrt logits
        grad = a * (g - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate objective scaled
        score = 2 * n * max_c  # compute_score
        if score < best_val:
            best_val = score
            best_a = a.copy()
        # maybe adaptive lr schedule: if t%1000==0 and score improvement low, decrease lr
        if t % 5000 == 0:
            lr *= 0.5  # reduce step size gradually
        # optionally break if lr becomes too small
        if lr < 1e-8:
            break
    return best_a, best_val
```

"""
non‑negative step‑function optimiser for ``evaluate_sequence``

The optimiser works in three stages:

1. **Global exploration** – a wide sweep of several parametric families
   (power‑law, exponential, mixed power‑law·exponential, beta) together
   with many random Dirichlet draws.  The best ~15 candidates (according to
   the exact objective 2·n·max(conv)) are retained.

2. **Adam‑based mirror descent** – the probability vector is represented by
   logits (soft‑max) so the simplex constraint is satisfied automatically.
   A sub‑gradient of the max‑convolution is computed, turned into a gradient
   w.r.t. the logits, and an Adam optimiser is applied.  Each run is given a
   time slice; the best vector seen during the run is returned.  Multiple
   restarts (including the best candidate from stage 1 and random starts) are
   performed until the budget runs out.

3. **Final reporting** – the best vector found is evaluated through the
   official ``evaluate_sequence`` routine and returned as a plain list of
   floats.

The whole search respects the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective (sum of a is assumed to be 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_mixed(n: int, p: float, lam: float) -> np.ndarray:
    """Mixed power‑law / exponential: a_i ∝ (i+1)^{-p}·exp(-lam·i)."""
    a = (np.arange(1, n + 1, dtype=float) ** (-p)) * np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    """Logistic (sigmoid) shape centred at mu with scale sigma."""
    x = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp(-(x - mu) / sigma))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Global exploration – keep the best vectors
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 15):
    """Sweep families and many random Dirichlet draws, keep up to ``max_keep`` best."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s                     # project onto the simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep
    for p in np.linspace(0.05, 3.0, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(0.0001, 0.07, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) mixed family sweep (power‑law * exponential)
    for p in np.linspace(0.0, 2.0, 30):
        for lam in np.linspace(0.001, 0.05, 30):
            if time.time() >= deadline:
                break
            try_candidate(generate_mixed(n, p, lam))

    # 4) beta sweep
    al_vals = np.linspace(0.2, 5.0, 20)
    be_vals = np.linspace(0.2, 5.0, 20)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 5) logistic sweep (coarse)
    mus = np.linspace(0.25 * n, 0.75 * n, 10)
    sigmas = np.linspace(0.05 * n, 0.3 * n, 10)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                break
            try_candidate(generate_logistic(n, mu, sigma))

    # 6) massive random Dirichlet draws
    concs = [0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 2000   # ~12 000 draws total – cheap on modern hardware
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def adam_optimize(start_a: np.ndarray, deadline: float,
                  lr0: float = 0.08,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  tol_active: float = 1e-12,
                  jitter_iter: int = 4000,
                  lr_decay: float = 0.95,
                  lr_decay_every: int = 3000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent Adam optimiser.
    Returns the best probability vector seen (sum = 1) and its score.
    """
    n = start_a.shape[0]
    # initialise logits safely (avoid log(0))
    safe_a = np.maximum(start_a, np.finfo(float).tiny)
    logits = np.log(safe_a)
    # centre to avoid large values
    logits -= np.mean(logits)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    it = 0
    lr = lr0

    while time.time() < deadline:
        it += 1

        # ---------- softmax ----------
        max_logit = np.max(logits)
        exp_shift = np.exp(logits - max_logit)
        a = exp_shift / exp_shift.sum()

        # ---------- convolution ----------
        conv = full_convolution(a)
        max_c = conv.max()

        # ---------- active set ----------
        eps_active = max_c * tol_active
        active = np.where(conv >= max_c - eps_active)[0]

        # ---------- sub‑gradient of max(conv) ----------
        g = np.zeros_like(a)
        # loop over active indices (usually few)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # ---------- gradient wrt logits (mirror of softmax) ----------
        avg = np.dot(a, g)          # <a, g>
        grad = a * (g - avg)        # ∂f/∂logits

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        logits -= step

        # keep logits centred (softmax invariant)
        logits -= np.mean(logits)

        # ---------- keep best ----------
        cur_score = 2.0 * n * max_c
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---------- occasional jitter to escape plateaux ----------
        if it % jitter_iter == 0:
            rng = np.random.randn(n) * (lr0 * 0.001)
            logits += rng
            # tiny random scaling of learning rate
            lr *= 0.9

        # ---------- learning‑rate decay ----------
        if it % lr_decay_every == 0:
            lr *= lr_decay

        # stop if learning rate becomes excessively tiny
        if lr < 1e-7:
            break

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – 2048 offers a good trade‑off between quality and speed
    # -----------------------------------------------------------------
    n = 2048

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (previous best) – if the variable exists
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_score:
                best_score, best_seq = val, fallback_seq
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 300.0   # reserve ~5 min for polishing
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=15)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_score:
                best_score, best_seq = val, vec.tolist()
        print(f"[baseline] best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Intensive Adam optimisation – start from the current best and
    #    launch several random restarts while time permits.
    # -----------------------------------------------------------------
    opt_deadline = deadline - 30.0   # keep a few seconds for final reporting
    per_run = 25.0                   # seconds allocated to each Adam run

    # first polish the current best (if we have one)
    if best_seq is not None:
        a0 = np.array(best_seq, dtype=float)
        run_dead = min(opt_deadline, time.time() + per_run)
        a_opt, sc_opt = adam_optimize(a0, run_dead)
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[adam‑polish‑primary] score = {best_score:.6f}")

    # launch random restarts
    while time.time() < opt_deadline:
        # pick a fresh random start – Dirichlet with random concentration
        conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0])
        a_start = np.random.dirichlet(np.full(n, conc))
        run_dead = min(opt_deadline, time.time() + per_run)
        a_opt, sc_opt = adam_optimize(a_start, run_dead)
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[adam‑restart] new best = {best_score:.6f}")

        # adapt per‑run time if we are close to the deadline
        remaining = opt_deadline - time.time()
        if remaining < 5.0:
            break
        per_run = min(per_run, remaining * 0.5)

    # -----------------------------------------------------------------
    # 3. Fallback to uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 4. Final official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
