    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    n = len(a)

"""
propose_candidate – a fast multi‑phase optimiser for ``evaluate_sequence``

The optimiser consists of three stages:

1. **Broad parametric sweep** – A collection of simple analytic families
   (power‑law, geometric, mixed plateau, beta, inverse‑sqrt,
   stretched‑exponential, alternating, two‑block) is sampled over
   wide parameter ranges.  All vectors are normalised to Σa = 1, and
   the best one (lowest ``2·n·max(conv(a,a))``) is retained.

2. **Simulated‑annealing polish** – Starting from the best sweep
   candidate, a stochastic mass‑transfer move is repeatedly applied:
   a small amount of weight is moved from one index to another.
   Moves are accepted according to a temperature schedule, allowing
   temporary worsening to escape local minima.  The step scale is
   adapted on‑the‑fly and the vector is projected onto the simplex
   after each move.  The procedure runs until the remaining budget is
   exhausted.

3. **Fallback handling** – If the sweep or annealing fail to produce a
   vector, a uniform distribution (which yields a score of ``2.0``) is
   used as a safe default.

The final sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# Helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a*a using a real FFT (length 2·len(a)−1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        score = 2·n·max(conv(a,a))
    The objective is scale‑invariant, therefore we keep the vector normalised.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i}, 0 < q < 1."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1−x_i)^{β},  x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def inv_sqrt(n: int, c: float) -> np.ndarray:
    """a_i ∝ 1/√(i+1+c)."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64) + c)
    a /= a.sum()
    return a


def stretched_exponential(n: int, lam: float, beta: float) -> np.ndarray:
    """a_i ∝ exp(−lam·i^{β})."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * (i ** beta))
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def alternating(n: int, r: float) -> np.ndarray:
    """Two‑point alternating pattern, a_i = 1+r·(−1)^i."""
    coeff = 1.0 + r * ((-1) ** np.arange(n))
    coeff = np.clip(coeff, 0, None)
    if coeff.sum() == 0.0:
        return None
    coeff /= coeff.sum()
    return coeff


def block_shape(n: int, L_frac: float, ratio: float) -> np.ndarray:
    """
    First block (length L) gets weight ratio·1, remaining positions get weight 1.
    After normalisation the total sum is 1.
    """
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    w1 = ratio
    w2 = 1.0
    a = np.concatenate([np.full(L, w1, dtype=np.float64),
                        np.full(n - L, w2, dtype=np.float64)])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Broad parametric sweep
# ----------------------------------------------------------------------


def parametric_sweep(n: int, time_deadline: float) -> np.ndarray:
    """Explore many analytic families and return the best normalised vector."""
    best_vec = None
    best_val = float('inf')

    # ---- power‑law (coarse then fine) ----
    for p in np.linspace(0.10, 2.00, 191):
        if time.time() >= time_deadline:
            return best_vec
        a = power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # fine power‑law near the current best
    if best_vec is not None:
        for p in np.linspace(0.30, 1.20, 451):
            if time.time() >= time_deadline:
                return best_vec
            a = power_law(n, p)
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- geometric ratios (including very close to 1) ----
    for q in np.concatenate((
            np.linspace(0.90, 0.99, 91),
            np.linspace(0.990, 0.9999, 110))):
        if time.time() >= time_deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- mixed plateau + power‑law ----
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.30, 0.90, 31):
            if time.time() >= time_deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- beta shape ----
    alphas = np.linspace(-0.8, 3.0, 20)
    betas = np.linspace(-0.8, 3.0, 20)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= time_deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- inverse‑sqrt ----
    for c in np.logspace(-2, 2, 15):   # 0.01 … 100
        if time.time() >= time_deadline:
            return best_vec
        a = inv_sqrt(n, c)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- stretched exponential ----
    for lam in np.logspace(-3, -0.1, 12):   # 0.001 … 0.79
        for beta in np.linspace(0.5, 3.0, 14):
            if time.time() >= time_deadline:
                return best_vec
            a = stretched_exponential(n, lam, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- alternating pattern ----
    for r in np.linspace(0.0, 0.9, 10):
        if time.time() >= time_deadline:
            return best_vec
        a = alternating(n, r)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---- two‑block shape ----
    for L_frac in np.linspace(0.01, 0.30, 15):
        for ratio in np.logspace(0, 2, 10):   # 1 … 100
            if time.time() >= time_deadline:
                return best_vec
            a = block_shape(n, L_frac, ratio)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---- random convex combinations of basic shapes ----
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    power05 = power_law(n, 0.5)
    geom98 = geometric(n, 0.98)
    basic = [uniform, power05, geom98]

    for _ in range(3000):
        if time.time() >= time_deadline:
            break
        coeffs = np.random.rand(3)
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basic))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def simulated_annealing(start_vec: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    Stochastic mass‑transfer hill‑climb with an exponential cooling schedule.
    Accepts worsening moves with probability exp(−Δ/ T) to escape local minima.
    """
    a = start_vec.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    n = a.shape[0]

    # temperature schedule: start at 10 % of current score, finish at 1e‑4·score
    max_temp = best_score * 0.10
    min_temp = best_score * 1e-4
    start = time.time()
    step_scale = 0.15  # typical fraction of the source mass to move

    while time.time() < time_deadline:
        elapsed = time.time() - start
        # exponential decay of temperature
        frac = elapsed / (time_deadline - start + 1e-12)
        temp = max_temp * (min_temp / max_temp) ** frac

        i, j = np.random.randint(n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # propose moving a random fraction (up to step_scale) of mass from i to j
        delta = step_scale * a[i] * np.random.rand()
        delta = min(delta, a[i])          # keep non‑negative
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # simplex projection guarantees Σ=1 and non‑negativity
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        diff = cand_score - cur_score

        # Metropolis acceptance
        if diff < 0.0 or np.random.rand() < math.exp(-diff / (temp + 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                # after a good move become a little bolder
                step_scale = min(step_scale * 1.05, 0.6)

        else:
            # slowly shrink step size on rejected moves
            step_scale = max(step_scale * 0.995, 1e-5)

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # -----------------------------------------------------------------
    # 0. Optional fallback (previously discovered vector)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Parametric sweep (≈ 300 s)
    # -----------------------------------------------------------------
    sweep_deadline = deadline - 400.0   # leave generous time for polishing
    n = 2048
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Simulated‑annealing refinement (bulk of remaining budget)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 10.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_refined = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_refined)
        print(f"[annealing] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 3. Guard – uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
