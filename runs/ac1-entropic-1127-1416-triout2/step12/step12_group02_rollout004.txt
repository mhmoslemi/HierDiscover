The objective is  

\[
\operatorname{score}(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2}\;,
\qquad a_i\ge 0 .
\]

Because the score is invariant under a positive scaling of the vector we may
*normalise* every candidate to `∑a = 1`.  With this normalisation  

\[
\operatorname{score}(a)=2\,n\;\max_k (a*a)_k .
\]

The problem is highly non‑convex: the constraints  
\((a*a)_k\le M\) are quadratic and indefinite.  A practical way to make
progress is to **alternately**  

1. **solve a linear programme** that, for a fixed current vector `a`,
   finds a new vector `b` with the *largest possible* ℓ₁‑norm while keeping
   every convolution entry ≤ `M = max(a*a)`;
2. **mix** the current vector with `b` (convex combination) and repeat.

The linear programme is:

\[
\begin{aligned}
\max\;&\mathbf{1}^\top b \\
\text{s.t. }& A(a)\,b\;\le\;M\mathbf{1},\qquad b\ge 0,
\end{aligned}
\]

where the matrix `A(a)` encodes the convolution `a*b`.  This LP is a
standard dense problem of size `2n‑1` × `n` and can be solved with
`scipy.optimize.linprog` (HiGHS) in a few milliseconds for `n≈2000`.

After a modest number of LP steps (typically < 30) the vector is already
much better than the best hand‑crafted families.  To polish the result we
run a **smooth‑max Adam optimiser** on the exact objective (the same
code that was used in the previous baseline).  The Adam phase finishes
the search with a tiny final improvement.

The whole pipeline respects the 1000 s wall‑clock limit, prints intermediate
scores and finally returns the best sequence it has found.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a normalised vector (∑a = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for fall‑back initialisation)
# ----------------------------------------------------------------------


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate of max‑convolution
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 10.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.015,
                         sym_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector found and its exact (non‑smoothed) score.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)          # log‑soft‑max parametrisation

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linearly anneal the smooth‑max temperature β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # occasional symmetrisation (helps because the problem is nearly symmetric)
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        # full convolution via FFT
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a (see derivation in the original code)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a   # chain rule through soft‑max

        # Adam update on the logits `z`
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # keep the best ever seen
        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (core of the new idea)
# ----------------------------------------------------------------------


def build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A such that A @ b = a * b
    (full aperiodic convolution).  Shape: (2n‑1, n).
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        A[k, idx_j] = a[idx_i]
    return A


def lp_refine(a: np.ndarray, deadline: float,
              max_iters: int = 30,
              mix_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform up to `max_iters` linear‑programming refinements.
    Returns the best vector encountered and its exact score.
    """
    from scipy.optimize import linprog

    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline - 5.0:
            break

        # current maximal convolution value
        M = np.convolve(a, a).max()

        # build convolution matrix for the constraint a * b ≤ M
        A = build_convolution_matrix(a)
        ub = np.full(A.shape[0], M, dtype=np.float64)

        # LP: maximise Σ b_j  ⇔ minimise −Σ b_j
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs')
        if not res.success:
            print(f"[{time.time():.1f}] LP iteration {it} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            print(f"[{time.time():.1f}] LP iteration {it} gave zero mass")
            break

        # renormalise to sum = 1
        g = b / sum_b

        # convex mixing
        a_new = (1.0 - mix_factor) * a + mix_factor * g
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum > 0.0:
            a_new = a_new / a_new_sum

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[{time.time():.1f}] LP iteration {it} improved to {best_score:.6f}")

        a = a_new  # continue from the mixed vector

    return best_a, best_score

# Main driver required by the evaluation harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising evaluate_sequence().
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0.  Initialise from the previously discovered best (if available)
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` is provided by the environment when present
        init_vec = np.array([float(x) for x in height_sequence_1], dtype=np.float64)  # type: ignore
        if init_vec.ndim != 1:
            raise ValueError
        # optionally truncate to a manageable size (2048 works well)
        if init_vec.size > 2048:
            init_vec = init_vec[:2048]
        a = np.maximum(init_vec, 0.0)
        if a.sum() < 1e-9:
            raise ValueError
        a /= a.sum()
        print(f"[{time.time():.1f}] Initialised from previous best (n={a.size})")
    except Exception:
        # fallback: a simple decreasing power‑law shape
        n = 2048
        a = left_heavy_power_family(n, beta=2.0)
        print(f"[{time.time():.1f}] Fallback initialisation (n={n})")

    # ------------------------------------------------------------------
    # 1.  Linear‑programming refinement phase
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] Starting LP refinement (n={a.size})")
        a, lp_score = lp_refine(a, deadline, max_iters=30, mix_factor=0.5)
        print(f"[{time.time():.1f}] LP phase finished with score {lp_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Smooth‑max Adam polishing (uses the same code as the baseline)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] Starting Adam polishing")
        a, adam_score = adam_smooth_optimize(
            a,
            deadline,
            max_iters=400_000,
            beta_start=10.0,
            beta_end=8000.0,
            lr=0.015,
            sym_interval=2000,
        )
        print(f"[{time.time():.1f}] Adam polishing best score {adam_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final safety check and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)  # official wrapper (clamps, etc.)
    print(f"\n[{time.time():.1f}] Finished search – final reported score = {final_score:.6f}")

    return final_seq
