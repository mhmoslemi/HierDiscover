"""
Propose a step‑function sequence that (empirically) improves the
evaluation score 2·n·max(conv(a,a))/(sum(a)²).

Algorithm
---------
* Use a fairly large length (n = 8192) – the same as the reference.
* Randomly initialise many candidate vectors from a battery of analytic
  families (U‑shape, power‑law, beta, mixtures) and from small perturbations
  of the current best.
* For each candidate run a three‑stage local optimisation:
    1. Adam optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution.
    2. Projected sub‑gradient descent on the exact max‑convolution.
    3. A final simulated‑annealing hill‑climb with a higher initial temperature.
* The whole procedure is repeated (multiple restarts) until the time budget
  (≈1000 s) is exhausted.  The best sequence seen is returned.
* All operations keep the vector on the probability simplex (non‑negative,
  sum = 1), which makes the denominator in the objective constant and allows
  us to minimise only themax‑convolution term.
"""

import time
import math
import random
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, scoring, projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²).  Assumes a ≥ 0."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    # implementation from https://doi.org/10.1.1.522.4561
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / total
    return v


# ----------------------------------------------------------------------
# Analytic families (used for random initialisation)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    return normalize(w)


def generate_random_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a random analytic family and random parameters, optionally shift."""
    family = rng.integers(0, 5)
    if family == 0:          # product family
        alpha = rng.uniform(0.0, 15.0)
        vec = symmetric_product_family(n, alpha)
    elif family == 1:        # left‑heavy power
        beta = rng.uniform(0.0, 15.0)
        vec = left_heavy_power_family(n, beta)
    elif family == 2:        # exponential‑U
        gamma = rng.uniform(0.0, 8.0)
        vec = exp_u_family(n, gamma)
    elif family == 3:        # mixture uniform + U
        lam = rng.random()
        gamma = rng.uniform(0.0, 8.0)
        vec = mixed_uniform_u_family(n, lam, gamma)
    else:                    # beta family
        p = rng.uniform(0.1, 5.0)
        q = rng.uniform(0.1, 5.0)
        vec = beta_family(n, p, q)

    # random right‑shift (zero‑padding) to break symmetry
    if rng.random() < 0.35:
        d = rng.integers(0, n // 10 + 1)
        vec = np.concatenate([vec[d:], np.zeros(d, dtype=vec.dtype)])
        vec = normalize(vec)
    return vec


def perturb_sequence(v: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add small Gaussian noise and renormalise."""
    noise = rng.normal(scale=scale, size=v.shape)
    pert = v + noise
    return normalize(pert)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 5.0,
                         beta_end: float = 15000.0,
                         lr: float = 0.01,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns best (a, score) seen before the deadline.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)                # log‑softmax parameters
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule for smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = 0.5 * (a + a[::-1])
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max wrt a
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # back‑prop through soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 3000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_final = np.exp(z)
    a_final = a_final / a_final.sum()
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_200_000,
                     step0: float = 0.05) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient at the active index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 4_000_000,
                  temp_factor: float = 5e-3,
                  step_scale_init: float = 0.3) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * temp_factor          # higher initial temperature
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0        # a small safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0.  Optional fallback (previous best, if present)
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Working length
    # -----------------------------------------------------------
    n = 8192

    # -----------------------------------------------------------
    # 2. Quick scan to obtain a decent α for the product family
    # -----------------------------------------------------------
    best_alpha = None
    for alpha in np.linspace(0.0, 12.0, 25):
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_alpha = alpha
            print(f"[{time.time():.1f}] init scan α={alpha:.3f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Main optimisation loop – several random restarts
    # -----------------------------------------------------------
    restart = 0
    while time.time() < deadline - 30.0:
        restart += 1
        print(f"\n[{time.time():.1f}] *** Restart #{restart} ***")

        # --- initialise -------------------------------------------------
        if best_seq is not None and rng.random() < 0.3:
            # start from a perturbation of the current best
            init_vec = perturb_sequence(np.array(best_seq, dtype=np.float64), rng, scale=0.015)
        else:
            # random analytic family
            init_vec = generate_random_sequence(n, rng)

        # --- Adam phase -------------------------------------------------
        sub_deadline = deadline - 10.0
        a_adam, sc_adam = adam_smooth_optimize(
            start_a=init_vec,
            deadline=sub_deadline,
            max_iters=500_000,
            beta_start=5.0,
            beta_end=20000.0,
            lr=0.012,
            sym_interval=0
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.tolist()
            print(f"[{time.time():.1f}] after Adam restart #{restart} → {best_score:.6f}")

        # --- Sub‑gradient phase -----------------------------------------
        if time.time() < deadline - 5.0:
            sub_deadline = deadline - 5.0
            a_sub, sc_sub = subgrad_optimize(
                start_a=a_adam,
                deadline=sub_deadline,
                max_iters=1_500_000,
                step0=0.055,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[{time.time():.1f}] after sub‑grad restart #{restart} → {best_score:.6f}")

        # --- Simulated‑annealing phase ----------------------------------
        if time.time() < deadline - 2.0:
            sa_deadline = deadline - 2.0
            a_sa, sc_sa = hill_climb_sa(
                start_a=a_sub if time.time() < deadline - 5.0 else a_adam,
                deadline=sa_deadline,
                max_steps=6_000_000,
                temp_factor=1e-2,            # a bit hotter
                step_scale_init=0.35,
            )
            if sc_sa < best_score:
                best_score = sc_sa
                best_seq = a_sa.tolist()
                print(f"[{time.time():.1f}] after SA restart #{restart} → {best_score:.6f}")

        # quick break if we are very close to the target (optional)
        if best_score < 1.503:
            print(f"[{time.time():.1f}] target reached, exiting early.")
            break

    # -----------------------------------------------------------
    # 4. Final safety fallback (uniform) if nothing was found
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
