"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1. **Coarse global search** – sweep several parametric families
   (power‑law, exponential, beta, Gaussian, sine‑power,
   symmetric inverse‑power‑law, symmetric two‑exponential) and many
   random Dirichlet draws.  Keep the best few candidates.

2. **Fast exact sub‑gradient descent** – minimise the true objective
   ``2·n·max(conv(a,a))`` (sum(a)=1) using the sub‑gradient of the
   max‑convolution.  The sub‑gradient is obtained by a single
   FFT‑based cross‑correlation, giving an O(n log n) iteration.
   The step size follows a decreasing schedule and the iterate is
   projected onto the probability simplex after each step.
   The routine runs for the majority of the time budget,
   periodically applying a tiny random jitter to escape plateaus.

3. **Final stochastic polishing** – a short simulated‑annealing
   phase (pairwise mass moves) can sometimes escape shallow local
   minima and yields a final improvement.

All phases respect the 1000 s wall‑clock limit; the best sequence found
is returned as a plain ``list`` of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the exploratory phase)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i‑mid)²/(2σ²))."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_inverse_powerlaw_ends(n: int, p: float) -> np.ndarray:
    """Symmetric mass concentrated near the ends:
       a_i ∝ (i+1)^{-p} + (n-i)^{-p}.
    """
    i = np.arange(n, dtype=float) + 1.0
    rev_i = np.arange(n, dtype=float)[::-1] + 1.0
    a = i ** (-p) + rev_i ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_two_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric sum of two exponentials:
       a_i ∝ exp(-lam·i) + exp(-lam·(n‑1‑i)).
    """
    a = np.exp(-lam * np.arange(n, dtype=float))
    b = np.exp(-lam * np.arange(n - 1, -1, -1, dtype=float))
    a = a + b
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global baseline exploration
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 10,
    random_draws: int = 3000,
) -> list[np.ndarray]:
    """Explore many families and random Dirichlet draws.
    Returns a list of up to ``max_keep`` best vectors.
    """
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Power‑law sweep
    for p in np.linspace(0.1, 2.0, 101):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) Exponential sweep
    for lam in np.linspace(0.0001, 0.3, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta sweep
    al_vals = np.linspace(0.2, 4.0, 31)
    be_vals = np.linspace(0.2, 4.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.5 * n, 25):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # 5) Sine‑power sweep
    for p in np.linspace(0.1, 6.0, 31):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # 6) Symmetric inverse‑power‑law sweep
    for p in np.linspace(0.5, 3.0, 31):
        if time.time() >= deadline:
            break
        try_candidate(generate_inverse_powerlaw_ends(n, p))

    # 7) Symmetric two‑exponential sweep
    for lam in np.linspace(0.0001, 0.3, 31):
        if time.time() >= deadline:
            break
        try_candidate(generate_two_exponential(n, lam))

    # 8) Random Dirichlet draws with various concentrations
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_conc = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Fast exact sub‑gradient optimiser (FFT‑based)
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.4,
    eps_factor: float = 1e-9,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact objective
    2·n·max(conv(a,a)).  The sub‑gradient of the max‑convolution is
    computed with a single FFT‑based cross‑correlation.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        # convolution and active set
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * eps_factor
        active = np.where(conv >= max_val - eps)[0]

        # uniform weights on the active constraints (any convex combination works)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size

        # gradient = 2 * (a cross‑correlated with w)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 1 - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (m - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # step size schedule (decreasing)
        step = step0 / np.sqrt(t + 1.0)
        a = a - step * grad
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape flat regions
        if t % 4000 == 0 and t > 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t + 1.0))
            a = project_to_simplex(a + jitter)

        t += 1

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple Metropolis hill‑climb that moves a random amount of mass
    from one coordinate to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = 1e-3
    final_T = 1e-6
    cooling = 0.9995
    step_scale = 0.05
    min_step = 1e-5
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] == 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, min_step)

        T = max(final_T, T * cooling)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = project_to_simplex(a + jitter)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Choose a problem size (length of the step function)
    # -----------------------------------------------------------------
    n = 2048  # works well for the time budget; can be changed

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Fallback from a previously discovered construction (if any)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Coarse global exploration (≈ 150 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline - 300.0, start_time + 150.0)
    if time.time() < baseline_deadline:
        candidates = baseline_search(
            n, baseline_deadline, max_keep=12, random_draws=3000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Exact sub‑gradient optimisation (majority of the remaining time)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        # allocate about 60 % of the remaining budget to this phase
        remaining = deadline - time.time()
        subgrad_deadline = time.time() + remaining * 0.60

        # start from the current best and a few random restarts
        start_vectors = [
            np.array(best_seq, dtype=float),
            np.full(n, 1.0 / n),
            project_to_simplex(np.random.rand(n)),
            project_to_simplex(np.random.rand(n)),
        ]

        refined_best = None
        refined_val = float("inf")

        for idx, start_vec in enumerate(start_vectors):
            if time.time() >= subgrad_deadline:
                break
            # use a relatively large step size first, then a finer one
            mid_deadline = time.time() + (subgrad_deadline - time.time()) * 0.5
            a_mid = subgradient_optimize(start_vec, mid_deadline, step0=0.6)
            # second pass with a smaller step size for fine tuning
            a_ref = subgradient_optimize(a_mid, subgrad_deadline, step0=0.15)
            val = compute_score(a_ref)
            print(f"[subgrad {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_ref.copy()

        if refined_best is not None and refined_val < best_val:
            best_val, best_seq = refined_val, refined_best.tolist()
            print(f"[subgrad] best after refinement = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Simulated annealing polishing (last ~90 s)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        anneal_deadline = deadline - 5.0
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final reporting using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
