def simulated_annealing(a_start, deadline, init_temp=0.01, cooling=0.999):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    T = init_temp
    n = len(a)
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        delta = np.random.rand() * a[i] * 0.1   # random fraction of a_i
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0)
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            T = init_temp   # reset temp maybe
        else:
            # accept with probability exp(-(sc - best_sc)/T)
            if np.random.rand() < np.exp(-(sc - best_sc) / T):
                a = cand
        T *= cooling
    return best_a

"""
Improved search for a step‑function that minimises evaluate_sequence.

The algorithm consists of three stages:

1. **Broad sweep** – generate a diverse pool of candidates using
   parametric families (power‑law, geometric, beta‑shape, cosine‑power,
   block‑wise constant) together with many random Dirichlet draws.
   Every candidate is normalised to sum = 1 and the best few (by the
   exact objective 2·n·max(conv)) are retained.

2. **Annealed smoothed‑max optimisation** – starting from each of the
   retained candidates we minimise a smooth approximation of the
   max‑convolution:
        fβ(a) = (1/β)·log Σk exp(β·(a∗a)k) .
   The gradient of fβ is 2·(w∗a_rev) where w is the soft‑max
   distribution over the convolution entries.
   A continuation schedule (β = 5,10,20,40,80) is used; for each β a
   simple projected gradient descent with line‑search is performed.
   After every β the true objective 2·n·max(conv) is evaluated and the
   best vector is kept.

3. **Final stochastic hill‑climb** – a greedy mass‑transfer move between
   two random indices is repeatedly attempted, accepting only
   improvements (with a tiny probability of uphill moves to escape flat
   plateaus).  The step size is adapted on‑the‑fly.

All stages respect the overall wall‑clock budget (default 1000 s).  The
returned list is a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of possibly different‑length vectors using FFT."""
    n = len(x)
    m = len(y)
    L = n + m - 1
    n_fft = 1 << ((L - 1).bit_length())      # next power of two
    X = np.fft.rfft(x, n_fft)
    Y = np.fft.rfft(y, n_fft)
    conv = np.fft.irfft(X * Y, n_fft)[:L]
    return conv


def self_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    return convolve_full(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence for a normalised vector (Σa==1)."""
    n = a.shape[0]
    conv = self_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth max objective and its gradient
# ----------------------------------------------------------------------
def smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns (f, g) where
        f = (1/β)·log Σ exp(β·conv)
    is a smooth approximation of max(conv) and
        g = ∇_a f = 2·(w ∗ a_rev)   (a_rev = a[::-1])
    with w being the soft‑max weights over the convolution entries.
    """
    conv = self_convolution(a)               # length 2n‑1
    maxc = conv.max()
    # soft‑max weights (numerically stable)
    expb = np.exp(beta * (conv - maxc))
    sumexp = expb.sum()
    w = expb / sumexp                         # shape (2n‑1,)

    # gradient: 2 * (w convolved with reversed a)
    grev = convolve_full(w, a[::-1])          # length 3n‑2
    n = a.shape[0]
    g = 2.0 * grev[n - 1:n - 1 + n]           # slice to length n

    # smooth value (stable)
    f = maxc + np.log(sumexp) / beta
    return f, g


# ----------------------------------------------------------------------
# Parametric families (same as in the previous baseline)
# ----------------------------------------------------------------------
def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, beta: float) -> np.ndarray:
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    a /= a.sum()
    return a


def cosine_power(n: int, expo: float) -> np.ndarray:
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    blocks = rng.dirichlet(np.ones(K))
    blocks = np.sort(blocks)[::-1]                         # decreasing
    repeats = np.full(K, n // K, dtype=int)
    remainder = n - repeats.sum()
    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)[:n]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Annealed gradient descent on the smooth max‑convolution objective
# ----------------------------------------------------------------------
def annealed_gradient_descent(a0: np.ndarray, total_seconds: float) -> np.ndarray:
    """
    Continuation method: minimise smooth approximation of max(conv) for a
    schedule of β values.  Uses a simple projected gradient descent with
    a few line‑search candidates.
    """
    start = time.time()
    deadline = start + total_seconds
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    beta_schedule = [5.0, 10.0, 20.0, 40.0, 80.0]

    for beta in beta_schedule:
        # allocate a fraction of the remaining time to this β
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        sub_deadline = time.time() + remaining * 0.7

        while time.time() < sub_deadline:
            f, g = smooth_max_and_grad(a, beta)
            gn = np.linalg.norm(g) + 1e-14
            # base step inversely proportional to β and gradient norm
            base_step = 1.0 / (beta * gn)

            improved = False
            for factor in (1.0, 0.5, 0.25, 0.125):
                step = base_step * factor
                cand = a - step * g
                cand = project_to_simplex(cand)
                f_cand, _ = smooth_max_and_grad(cand, beta)
                if f_cand < f:
                    a = cand
                    improved = True
                    break
            if not improved:
                # small random jitter to escape a flat region
                jitter = np.random.randn(len(a)) * (0.02 / np.sqrt(beta))
                a = project_to_simplex(a + jitter)

        # after optimisation for this β, evaluate the true (non‑smooth) score
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()

    return best_a


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (greedy with occasional uphill moves)
# ----------------------------------------------------------------------
def final_hill_climb(a_start: np.ndarray, deadline: float) -> np.ndarray:
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    step_factor = 0.05
    no_improve = 0

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a random fraction (up to step_factor) of mass from i to j
        delta = step_factor * a[i] * np.random.rand()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_factor = min(step_factor * 1.07, 0.6)
            no_improve = 0
        else:
            # rarely accept a worse move (simulated‑annealing style)
            if np.random.rand() < 0.001:
                a = cand
            step_factor *= 0.985
            no_improve += 1

        if no_improve > 120_000:
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

    return best_a

            best_candidates.append((sc, vec))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if sc < best_candidates[-1][0]:
                best_candidates[-1] = (sc, vec)
                best_candidates.sort(key=lambda x: x[0])
        if sc < best_val:
            best_val, best_seq = sc, vec.tolist()

    phase1_deadline = start_time + 250.0  # spend ~250 s on the sweep

    # ----- uniform baseline -----
    consider(np.full(n, 1.0 / n))

    # ----- power‑law with offset -----
    for p in np.linspace(0.1, 2.0, 30):
        for off in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0]:
            if time.time() >= phase1_deadline:
                break
            if off >= n:
                continue
            consider(powerlaw_offset(n, p, off))

    # ----- geometric decay -----
    for r in np.linspace(0.4, 0.99, 35):
        if time.time() >= phase1_deadline:
            break
        consider(geometric(n, r))

    # ----- beta‑shape (polynomial tail) -----
    for beta in np.linspace(0.1, 30.0, 30):
        if time.time() >= phase1_deadline:
            break
        consider(beta_shape(n, beta))

    # ----- cosine‑power -----
    for e in np.linspace(0.2, 8.0, 25):
        if time.time() >= phase1_deadline:
            break
        consider(cosine_power(n, e))

    # ----- random block shapes (various numbers of blocks) -----
    for K in [10, 20, 40, 80, 160]:
        for _ in range(300):
            if time.time() >= phase1_deadline:
                break
            consider(block_shape(n, K, rng))

    # ----- random Dirichlet draws with various concentration parameters -----
    for alpha in [0.05, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(200):
            if time.time() >= phase1_deadline:
                break
            vec = rng.dirichlet(np.full(n, alpha))
            consider(vec)

    # ----- random permutations of the best few candidates (breaks structure) -----
    for _, vec in best_candidates[:top_k]:
        perm = rng.permutation(vec)
        consider(perm)

    # ------------------------------------------------------------------
    # 2. Annealed gradient descent on the smooth max‑convolution objective
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 60.0:
        # allocate roughly 60 % of the remaining budget to refinement
        refine_budget = remaining * 0.6
        # split roughly evenly among the retained candidates
        per_candidate = max(5.0, refine_budget / len(best_candidates))
        for _, vec in best_candidates:
            if time.time() >= deadline - 30.0:
                break
            a0 = np.array(vec, dtype=float)
            refined = annealed_gradient_descent(a0, per_candidate)
            sc = compute_score(refined)
            if sc < best_val:
                best_val, best_seq = sc, refined.tolist()

    # ------------------------------------------------------------------
    # 3. Final stochastic hill‑climb for fine‑tuning
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = final_hill_climb(a_start, hill_deadline)
        sc = compute_score(a_final)
        if sc < best_val:
            best_val, best_seq = sc, a_final.tolist()

    # ------------------------------------------------------------------
    # 4. Safeguard – fall back to a uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 5. Official evaluation (includes the security wrappers)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported (official) score = {final_score:.6f}")
    return best_seq
