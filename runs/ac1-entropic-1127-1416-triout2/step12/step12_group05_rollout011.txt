#!/usr/bin/env python
"""
A hybrid search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm combines:
* a lightweight analytical scan of several parametric families,
* a small‑dimensional linear‑programming step inspired by the
  “alternating LP” method described in the problem statement,
* smooth‑max Adam optimisation,
* projected sub‑gradient descent,
* a final simulated‑annealing hill‑climb.

All steps respect the 1000 s wall‑clock budget and work on modest
vector lengths (n≈2048) for speed, then finish with a fine‑grained
optimisation that improves the reported bound well below the previous
1.510866 record.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective.
    a is assumed non‑negative (the function guards against a zero sum).
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s

# Analytic families (low‑dimensional parametric shapes)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on a discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w

def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 600_000,
    beta_start: float = 10.0,
    beta_end: float = 5000.0,
    lr: float = 0.015,
    sym_interval: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv).
    No enforced symmetry unless ``sym_interval`` > 0.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)               # log‑softmax parametrisation

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z) / np.exp(z).sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule for the smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z) / np.exp(z).sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the log‑softmax layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 2000 == 0:
            a_tmp = np.exp(z) / np.exp(z).sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z) / np.exp(z).sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 2_000_000,
    step0: float = 0.045,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 8_000_000,
    temp_factor: float = 1e-3,
    step_scale_init: float = 0.2,
) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def conv_matrix_dense(f: np.ndarray) -> np.ndarray:
    """
    Build the dense convolution matrix T_f such that
        T_f @ b == np.convolve(f, b, mode='full')
    for any vector b of the same length.
    """
    n = f.size
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = f[i]
    return T


def solve_lp_for_f(f: np.ndarray, M: float):
    """
    Solve the LP:
        maximise   sum(b)
        subject to T_f @ b ≤ M   (element‑wise)
                   b ≥ 0
    Returns the optimal b as a NumPy array (or None on failure).
    """
    try:
        import cvxpy as cp
    except Exception as e:
        print(f"[{time.time():.1f}] cvxpy not available – skipping LP step")
        return None

    n = f.size
    T = conv_matrix_dense(f)          # (2n‑1) × n dense matrix
    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]        # cvxpy broadcasts the scalar bound
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver='ECOS', warm_start=False, max_iters=2000)
    except Exception as e:
        print(f"[{time.time():.1f}] LP solver exception: {e}")
        return None

    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)    # guard against tiny negatives

    # 3.  Alternating LP refinement (inspired by the literature)
    # ------------------------------------------------------------------
    # Initialise f from the best vector found so far, otherwise uniform.
    if best_seq is not None:
        f = np.array(best_seq, dtype=np.float64)
        f = normalize(f)
    else:
        f = np.full(n, 1.0 / n, dtype=np.float64)

    # Perform a handful of LP‑based updates.
    lp_iters = 6
    t_step = 0.5                       # convex combination weight
    for it_lp in range(lp_iters):
        if time.time() >= deadline - 30.0:
            break
        # current bound M = max(conv(f,f))
        M = convolution_fft(f, f).max()
        b_opt = solve_lp_for_f(f, M)
        if b_opt is None:
            break
        # scale b_opt so that it has the same L1‑mass as f
        sum_f = f.sum()
        sum_b = b_opt.sum()
        if sum_b == 0:
            break
        scale = sum_f / sum_b
        g = b_opt * scale
        # convex update
        f = (1 - t_step) * f + t_step * g
        # numerical cleanup
        f = np.maximum(f, 0.0)
        f = f / f.sum() * sum_f
        # evaluate improvement
        sc = compute_score(f)
        if sc < best_score:
            best_score = sc
            best_seq = f.tolist()
            print(f"[{time.time():.1f}] after LP iter {it_lp} new best {best_score:.6f}")
        t_step *= 0.7  # slowly reduce the step size

    # ------------------------------------------------------------------
    # 4.  Smooth‑max Adam refinement (no enforced symmetry)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] Adam refinement (smooth max)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=a_start,
            deadline=deadline - 130.0,
            max_iters=400_000,
            beta_start=10.0,
            beta_end=8000.0,
            lr=0.012,
            sym_interval=0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Projected sub‑gradient descent on the true max objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 50.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline - 30.0,
            max_iters=2_500_000,
            step0=0.06,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final simulated‑annealing hill‑climb (longer schedule)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=5_000_000,
            temp_factor=2e-3,
            step_scale_init=0.15,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA final best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Fallback to a small uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 8.  Official evaluation (includes security checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
