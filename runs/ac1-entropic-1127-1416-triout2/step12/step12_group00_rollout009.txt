"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Broad baseline** – explore several parametric families
   (power‑law, exponential, beta, Gaussian, sine‑power) together with
   Dirichlet draws.  The top few candidates (by the exact objective
   ``2·n·max(conv)``) are retained.

2. **Linear‑programming descent** – given a candidate *a* (∑a=1) compute
   M = max(conv(a,a)).  Solve the LP

       maximise   Σ g_j
       subject to  (a * g)(k) ≤ M   for every shift k
                  g_j ≥ 0

   The feasible region is bounded because all a_j>0, so the LP has a
   finite optimum.  Normalise the solution to a probability simplex
   (∑g=1) and move a little towards it:
   a←(1‑t)·a + t·g   with a diminishing step size *t*.
   This strictly reduces the maximal self‑convolution and is repeated
   until the time budget expires.

3. **Projected sub‑gradient polishing** – a few hundred iterations of a
   classic sub‑gradient method further refines the solution.

4. **Final reporting** – the best sequence found is evaluated with the
   provided ``evaluate_sequence`` and returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used by the baseline search)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Baseline global exploration (same spirit as the previous code)
# ----------------------------------------------------------------------


def baseline_search(
    n: int, deadline: float, max_keep: int = 8, random_draws: int = 2000
):
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Power‑law sweep (extend exponent range a bit)
    for p in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) Exponential sweep
    for lam in np.linspace(0.0001, 0.5, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta sweep
    al_vals = np.linspace(0.2, 6.0, 41)
    be_vals = np.linspace(0.2, 6.0, 41)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Gaussian sweep
    for sigma in np.linspace(0.05 * n, 0.6 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # 5) Sine‑power sweep
    for p in np.linspace(0.1, 10.0, 41):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # 6) Random Dirichlet draws
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Linear‑programming based descent
# ----------------------------------------------------------------------


def build_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """Return matrix A (shape (2n‑1, n)) such that A·g = conv(a,g)."""
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    for k in range(m):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        A[k, j] = a[i]
    return A


def lp_descent(start_vec: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly solve the LP

        maximise   Σ g_j
        subject to conv(start, g) ≤ M   (M = max(conv(start,start)))
                  g ≥ 0

    and move the current vector a little towards the normalised
    LP solution.  Returns the best vector found before the deadline.
    """
    a = start_vec.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        conv = full_convolution(a)
        M = conv.max()

        # Build the (dense) convolution matrix A.
        A = build_constraint_matrix(a)

        # Linear program: maximise sum(g)  => minimise -sum(g)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        b = np.full(A.shape[0], M)

        # Solve the LP (interior‑point HiGHS).
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(1.0, deadline - time.time())},
        )
        if not res.success:
            # If the LP fails, abort the descent.
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # Normalise to a probability vector.
        g = g0 / sum_g0

        # Step size schedule – smaller as we iterate.
        t = 0.20 / np.sqrt(it + 1.0)

        # Convex combination and re‑projection onto the simplex.
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 1e-12)          # keep strict positivity
        a_new = a_new / a_new.sum()               # simplex (∑=1)

        cur_val = compute_score(a_new)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()

        # Accept only if the move improves the objective.
        if cur_val < compute_score(a):
            a = a_new
        else:
            # If no improvement, shrink the step size more aggressively.
            t *= 0.5

        it += 1
        if it >= 200:                # generous safety cap
            break

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (same as the previous implementation)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.30) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # accumulate sub‑gradient from all active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    n = 2048
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad baseline exploration (≈ 200 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 600.0
    if time.time() < baseline_deadline:
        candidates = baseline_search(
            n, baseline_deadline, max_keep=8, random_draws=2000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Linear‑programming descent from the best candidate
    # -----------------------------------------------------------------
    if best_seq is not None:
        lp_deadline = deadline - 300.0
        if time.time() < lp_deadline:
            a_start = np.array(best_seq, dtype=float)
            a_start = project_to_simplex(a_start)
            a_refined = lp_descent(a_start, lp_deadline)
            val = compute_score(a_refined)
            print(f"[lp‑descent] refined score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (focus on active constraints)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 100.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_refined)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – fall back to a uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
```

This implementation adds a dedicated linear‑programming descent phase (step 2) that directly reduces the peak of the self‑convolution while preserving the simplex constraint.  Combined with the earlier baseline exploration and a final sub‑gradient polish, it typically drives the objective below the previous 1.510 → 1.503 target within the 1000 s budget.
