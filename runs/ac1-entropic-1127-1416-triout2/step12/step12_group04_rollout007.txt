def softmax(x):
    max_x = np.max(x)
    e = np.exp(x - max_x)
    return e / np.sum(e)

def objective_and_grad(x, beta):
    a = softmax(x)
    conv = convolution_fft(a, a)
    # compute logsumexp
    beta_conv = beta * conv
    max_bc = np.max(beta_conv)
    logZ = max_bc + np.log(np.sum(np.exp(beta_conv - max_bc)))
    f = (2.0 * len(a) / beta) * logZ
    # compute weights w = exp(beta_con - logZ)
    w = np.exp(beta_conv - logZ)  # already normalized
    # compute gradient w.r.t a
    term1 = np.correlate(w, a, mode='valid')
    term2 = np.correlate(w[::-1], a, mode='valid')[::-1]
    g_a = 2.0 * len(a) * (term1 + term2)   # (2n*sum)
    # gradient w.r.t x
    dot = np.dot(a, g_a)  # scalar
    g_x = a * (g_a - dot)
    return f, g_x
```

"""
Hybrid optimizer for the step‑function upper‑bound problem.

The pipeline:
1.  Scan a handful of analytic families (U‑shaped, heavy‑left,
    Gaussian, cosine windows, Kaiser) to obtain a decent seed.
2.  Refine the seed with a smooth‑max surrogate and L‑BFGS‑B.
    The surrogate uses a temperature β and the gradient is obtained
    analytically via FFT‑based correlation.
    β is increased (1 → … → 80) so the optimisation gradually
    approaches the original non‑smooth objective.
3.  Perform a few LP‑mixing steps (the classic state‑of‑the‑art
    refinement) to push the max‑convolution down further.
4.  Symmetrise the candidate (the objective is invariant under
    reversal) and finish with a cheap simulated‑annealing hill‑climb.
All phases respect the global wall‑clock budget (default 1000 s)
and the returned list is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (initial broad scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, window_type: str) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def kaiser_window_family(n: int, beta: float) -> np.ndarray:
    """Standard Kaiser window; beta ≥ 0."""
    from scipy.special import i0

    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = i0(beta * np.sqrt(1 - x ** 2))
    w /= i0(beta)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (FFT‑based)
# ----------------------------------------------------------------------


def correlation_valid_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Returns the valid part of the correlation Corr(w, a) which has
    length len(a) = n and corresponds to the sliding dot‑product
    w[i:i+n]·a for i = 0 … n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]           # = 2*n‑1
    N = next_pow2(m + n)     # zero‑pad length for FFT
    W = np.fft.rfft(w, n=N)
    A = np.fft.rfft(a, n=N)
    corr = np.fft.irfft(np.conj(W) * A, n=N)[: m + n - 1]
    # valid slice starts at index n‑1 and ends at index m‑1 (inclusive)
    return corr[n - 1 : m]


def smooth_max_obj_and_grad(x: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Surrogate of the original objective:
        f_beta(a) = (2 n / β) * log Σ_k exp(β·conv_k(a))

    a = softmax(x) guarantees a_i ≥ 0 and Σ a_i = 1.
    Returns (f_beta, ∂f/∂x) as required by scipy.optimize.minimize.
    """
    a = softmax(x)                      # probability vector, sum=1
    n = a.shape[0]

    # exact convolution (full length 2n‑1)
    conv = convolution_fft(a, a)

    # smooth max via log‑sum‑exp
    beta_conv = beta * conv
    # log‑sum‑exp in a numerically stable way
    lse = logsumexp(beta_conv)
    f = (2.0 * n / beta) * lse

    # weights w_k = exp(β·conv_k) / Σ exp(β·conv_l)
    w = np.exp(beta_conv - lse)          # shape (2n‑1)

    # gradient wrt a:
    # ∂f/∂a_i = 2 n Σ_k w_k (a_{k‑i} + a_{i‑k})
    term1 = correlation_valid_fft(w, a)                      # w[i:i+n]·a
    term2 = correlation_valid_fft(w[::-1], a)[::-1]          # reverse‑slice
    g_a = 2.0 * n * (term1 + term2)

    # chain rule through the softmax:
    # ∂a/∂x = diag(a) – a a^T
    # ∂f/∂x = a ⊙ (g_a – ⟨a, g_a⟩)
    dot = np.dot(a, g_a)
    g_x = a * (g_a - dot)

    return float(f), g_x.astype(np.float64)


# ----------------------------------------------------------------------
# LP refinement (state‑of‑the‑art mixing)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j  s.t.  (f * g)_k ≤ M,
    g ≥ 0, and return the convex mixture
        f_new = (1‑t)·f + t·g_norm,
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (size m×n) with A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)               # maximise Σ g → minimise –Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    return project_to_simplex(f_new)


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing
    ``evaluate_sequence``.  The routine respects the global wall‑clock
    budget and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. Optional warm‑start from a previously discovered sequence
    # --------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Choose problem size (larger n gives more freedom)
    # --------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    n = fallback_len if fallback_len >= 1024 else 4096
    print(f"[{time.time() - start_time:.1f}] using length n = {n}")

    # helper to test a candidate (normalised internally)
    def try_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best score = {sc:.6f}")

    # --------------------------------------------------------------
    # 2. Broad analytic scan
    # --------------------------------------------------------------
    try_candidate(np.full(n, 1.0 / n, dtype=np.float64))               # uniform baseline

    # Product family (U‑shaped)
    alphas = np.concatenate([np.linspace(0.0, 5.0, 31), np.linspace(5.1, 120.0, 30)])
    for alpha in alphas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(product_family(n, alpha))

    # Left‑heavy power family
    betas = np.linspace(0.0, 30.0, 31)
    for beta in betas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(left_heavy_power_family(n, beta))

    # Exponential‑U family
    gammas = np.linspace(0.0, 20.0, 21)
    for gamma in gammas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(exp_u_family(n, gamma))

    # Gaussian family
    gauss_gammas = np.logspace(-4, 2, 30)
    for gamma in gauss_gammas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(gaussian_family(n, gamma))

    # Cosine windows
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(window_family(n, win))

    # Kaiser window (extra family)
    for beta in np.linspace(0.0, 12.0, 13):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(kaiser_window_family(n, beta))

    # Random starts
    for _ in range(300):
        if time.time() >= deadline - 250.0:
            break
        if np.random.rand() < 0.5:
            try_candidate(random_monotone_start(n))
        else:
            try_candidate(random_u_start(n))

    # --------------------------------------------------------------
    # 3. Initialise smooth‑max optimisation from the best seed
    # --------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.asarray(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = compute_score(a)

    # Convert to soft‑max parameters (logits)
    x = np.log(a + 1e-12)

    # --------------------------------------------------------------
    # 4. Smooth‑max L‑BFGS refinement (annealing on β)
    # --------------------------------------------------------------
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    for beta in betas:
        if time.time() >= deadline - 100.0:
            break

        remaining = deadline - time.time()
        # allocate a modest number of L‑BFGS iterations; adapt if very little time left
        max_iter = max(30, int(remaining // 0.2))

        def fun(x_vec):
            f, g = smooth_max_obj_and_grad(x_vec, beta)
            return f, g

        print(f"[{time.time() - start_time:.1f}] L‑BFGS with β={beta:.1f} (max_iter={max_iter})")
        res = minimize(fun, x, method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter, 'disp': False})

        x = res.x
        a = softmax(x)                     # back to probability vector
        sc = compute_score(a)
        if sc < best_score:

            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] improved after β={beta:.1f} → {sc:.6f}")

    if time.time() < deadline - 50.0:

            else:

    # --------------------------------------------------------------
    # 6. Symmetrisation (the problem is symmetric under reversal)
    # --------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:

        print(f"[{time.time() - start_time:.1f}] symmetrisation improved best → {best_score:.6f}")

    # --------------------------------------------------------------

    if time.time() < deadline - 5.0:

        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_000_000)

    # --------------------------------------------------------------

    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
