    best_score = 2.0 * n * conv.max()
    # optionally keep momentum reset at lambda changes? Not needed.

"""
non‑negative step‑function optimiser for ``evaluate_sequence``

Approach
--------
* Optimise the *scale‑invariant* objective  
  f(a)=2·n·max_k (a*a)_k   with   a_i ≥ 0 , Σ a_i = 1.
* Use a smooth approximation of the max via the log‑sum‑exp
  (soft‑max) with an annealing schedule λ∈{2,5,10,20,50,100,200}.
* Optimise the *logits* z (softmax(z)=a) with Adam (mirrored‑gradient
  descent) on the smoothed objective.  This respects positivity and
  the simplex automatically.
* Run many short random restarts (Dirichlet starts) within the
  allotted wall‑clock budget and keep the best vector seen.
* A final long run starts from the current best to polish it.
* The final sequence is returned as a plain list of floats; the
  evaluation function is scale‑invariant so the sum‑to‑1 normalisation
  is safe.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution via real FFT (full mode)
# ----------------------------------------------------------------------
def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using a real FFT."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())            # next power of two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Numerics required for the optimisation
# ----------------------------------------------------------------------
def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def score_from_a(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) under the normalisation Σa=1.
    The sum is guaranteed to be 1, therefore the denominator of
    ``evaluate_sequence`` drops out.
    """
    n = a.shape[0]
    conv = convolve_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def optimise_from_start(init_a: np.ndarray, deadline: float,
                        max_iters: int = 5000) -> np.ndarray:
    """
    Gradient‑descent (Adam) on the log‑sum‑exp surrogate of max_k (a*a)_k.
    Returns the best vector (sum‑to‑1) seen before ``deadline``.
    """
    n = init_a.shape[0]
    eps = 1e-12
    # initialise logits
    z = np.log(init_a + eps)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    # λ‑schedule (smoothness parameter)
    lambdas = [2, 5, 10, 20, 50, 100, 200]
    lam_idx = 0
    lam = lambdas[lam_idx]

    # best‑ever
    best_a = softmax(z)
    best_score = score_from_a(best_a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # occasional λ increase (every 2000 iterations)
        if it % 2000 == 0 and lam_idx < len(lambdas) - 1:
            lam_idx += 1
            lam = lambdas[lam_idx]

        a = softmax(z)                     # current probability vector

        # --------- compute smoothed max and its gradient ----------
        conv = convolve_fft(a, a)          # length 2·n‑1
        max_conv = conv.max()

        # soft‑max weights over the lag‑indices
        # (shift by max_conv for numerical stability)
        w_raw = np.exp(lam * (conv - max_conv))
        w = w_raw / w_raw.sum()            # Σ w = 1

        # gradient of the smoothed objective w.r.t. a:
        #   g = 2 * Σ_k w_k * (M_k a) = 2 * (a_rev * w)_{n‑1 … n‑1+n‑1}
        conv_rev = convolve_fft(a[::-1], w)    # length 3·n‑2
        grad_a = 2.0 * conv_rev[n - 1: n - 1 + n]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # -------------- Adam update -----------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = np.random.randn(n) * (0.01 / np.sqrt(it))
            z += jitter

        # -------------------------------------------------
        # evaluate true (unsmoothed) objective and keep the best
        a_curr = softmax(z)
        cur_score = score_from_a(a_curr)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_curr.copy()
    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted time.  Returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Problem size – use the length of the fallback if it exists,
    # otherwise pick a convenient power‑of‑two.
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        n = len(fallback)
    else:
        n = 2048

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (previous best known)
    # -----------------------------------------------------------------
    if fallback is not None:
        try:
            cand = [float(x) for x in fallback]
            val = evaluate_sequence(cand)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # -----------------------------------------------------------------
    # Helper to evaluate a numpy vector (sum=1) with the official function
    # -----------------------------------------------------------------
    def true_score(vec_np):
        return evaluate_sequence(vec_np.tolist())

    # -----------------------------------------------------------------
    # 1. Random restarts (Dirichlet draws) while we have time.
    #    Each restart gets a short budget (≈8 s) to explore.
    # -----------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 15.0:          # reserve ~15 s for final polish
        remaining = deadline - time.time()
        run_time = min(8.0, remaining * 0.2)      # ≤8 s or 20 % of what is left
        sub_deadline = time.time() + run_time

        # random concentration for Dirichlet (spread of mass)
        conc = np.random.choice([0.2, 0.5, 1.0, 2.0, 4.0])
        init_a = np.random.dirichlet(np.full(n, conc))

        # run the Adam optimiser on this start
        refined_a = optimise_from_start(init_a, sub_deadline, max_iters=5000)

        # evaluate the true objective (sum is 1 → no scaling needed)
        val = score_from_a(refined_a)   # cheap – uses the same convolution
        if val < best_val:
            best_val = val
            best_seq = refined_a.tolist()
            print(f"[restart {restarts}] new best score = {val:.6f}")

        restarts += 1

    # -----------------------------------------------------------------
    # 2. Final long‑run polish from the current best vector
    # -----------------------------------------------------------------
    if best_seq is not None:
        a_start = np.array(best_seq, dtype=float)
    else:
        # fall back to a uniform distribution if everything failed
        a_start = np.full(n, 1.0 / n)

    final_deadline = deadline
    polished_a = optimise_from_start(a_start, final_deadline,
                                     max_iters=20000)

    final_score = score_from_a(polished_a)
    if final_score < best_val:
        best_val = final_score
        best_seq = polished_a.tolist()
        print(f"[final polish] improved score = {final_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Safety net – uniform distribution if we never found anything
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 4. Report using the official evaluation function
    # -----------------------------------------------------------------
    reported = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {reported:.6f}")
    return best_seq
