"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

The algorithm improves on the previous state‑of‑the‑art by
  • scanning a richer set of analytic families and doing a local
    ternary‑search on the most promising parameters,
  • a longer exponentiated‑gradient (mirror‑descent) phase,
  • a more aggressive LP‑refinement loop (up to 60 mixes),
  • simple convex combinations of the two best families,
  • final symmetrisation.
All phases respect a global wall‑clock budget (default 1000 s) and
only use the public ``evaluate_sequence`` routine.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalized vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (parameterised constructions)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone: w_i ∝ (i+0.5)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, wtype: str) -> np.ndarray:
    """Standard cosine windows."""
    i = np.arange(n, dtype=np.float64)
    if wtype == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif wtype == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif wtype == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def product_exp_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    """
    Combination: product‑shape multiplied by centre‑exponential.
    w_i ∝ ((i+0.5)*(n-0.5-i))^{-α} · exp(-γ·dist(i,centre))
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    prod = np.exp(-alpha * logprod)
    dist = np.minimum(i, n - 1 - i)
    expc = np.exp(-gamma * dist)
    w = prod * expc
    return w / w.sum()


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 400_000,
                eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑objective.
    Returns best vector encountered.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)

        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]                     # the self‑product gets weight 1

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time() - deadline + (deadline - start_time):.1f}] EG improved to {best_sc:.6f}")

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    One LP refinement step (state‑of‑the‑art mixing).
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0, then return the convex mixture
        f_new = (1−t)·f + t·g_norm,
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A: A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)           # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 15})
    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Helper for candidate evaluation / bookkeeping
# ----------------------------------------------------------------------


def try_candidate(vec: np.ndarray,
                  best_seq_ref: dict,
                  n: int,
                  start_time: float,
                  deadline: float):
    """Normalise *vec*, compute score, update best if improved."""
    a = np.asarray(vec, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return a / a.sum() if a.size else a
    a = a / s
    sc = compute_score(a)
    if sc < best_seq_ref["score"]:
        best_seq_ref["score"] = sc
        best_seq_ref["vec"] = a.copy()
        print(f"[{time.time() - start_time:.1f}] new best score = {sc:.6f}")
    return a


# ----------------------------------------------------------------------
# Fine‑grained ternary search for a single‑parameter family
# ----------------------------------------------------------------------


def ternary_search_family(func, lo: float, hi: float,
                          best_ref: dict,
                          name: str,
                          start_time: float,
                          deadline: float,
                          max_iters: int = 30):
    """
    Generic ternary search on a unimodal scalar function *func*.
    *best_ref* contains keys ``'alpha'`` and ``'score'`` to be updated.
    """
    for _ in range(max_iters):
        if time.time() >= deadline - 5.0:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        s1 = func(m1)
        s2 = func(m2)
        if s1 < s2:
            hi = m2
        else:
            lo = m1
    best_alpha = (lo + hi) / 2.0
    best_score = func(best_alpha)
    if best_score < best_ref.get("score", float("inf")):
        best_ref["alpha"] = best_alpha
        best_ref["score"] = best_score
        print(f"[{time.time() - start_time:.1f}] ternary {name} best α={best_alpha:.4f} score={best_score:.6f}")
    return best_alpha, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The function respects the global time budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    global start_time  # for convenience in helper prints
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 0.  Fallback (if a previous construction is available)
    # ------------------------------------------------------------------
    best = {"vec": None, "score": float("inf")}
    try:
        fb_vec = [float(x) for x in height_sequence_1]       # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}] fallback score = {fb:.6f}")
        if fb < best["score"]:
            best["score"] = fb
            best["vec"] = np.asarray(fb_vec, dtype=np.float64)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem size (length of the vector)
    # ------------------------------------------------------------------
    fallback_len = len(best["vec"]) if best["vec"] is not None else 0
    if fallback_len >= 1024:
        n = fallback_len
    else:
        n = 4096                     # modest size that still fits the LP step
    print(f"[{time.time() - start_time:.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # 2. Exhaustive scan of analytic families (coarse grid)
    # ------------------------------------------------------------------
    # uniform baseline
    try_candidate(np.full(n, 1.0 / n), best, n, start_time, deadline)

    # ---------- product family ----------
    prod_best = {"alpha": None, "score": float("inf")}
    for alpha in np.linspace(0.0, 8.0, 81):          # fine grid near the origin
        if time.time() >= deadline - 250.0:
            break
        vec = product_family(n, alpha)
        sc = compute_score(vec)
        if sc < prod_best["score"]:
            prod_best["alpha"] = alpha
            prod_best["score"] = sc
        try_candidate(vec, best, n, start_time, deadline)

    # ---------- left‑heavy power ----------
    left_best = {"beta": None, "score": float("inf")}
    for beta in np.linspace(0.0, 20.0, 81):
        if time.time() >= deadline - 250.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < left_best["score"]:
            left_best["beta"] = beta
            left_best["score"] = sc
        try_candidate(vec, best, n, start_time, deadline)

    # ---------- Gaussian ----------
    gauss_best = {"gamma": None, "score": float("inf")}
    for gamma in np.logspace(-4, 2, 45):
        if time.time() >= deadline - 250.0:
            break
        vec = gaussian_family(n, gamma)
        sc = compute_score(vec)
        if sc < gauss_best["score"]:
            gauss_best["gamma"] = gamma
            gauss_best["score"] = sc
        try_candidate(vec, best, n, start_time, deadline)

    # ---------- exponential‑U ----------
    for gamma in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(exp_u_family(n, gamma), best, n, start_time, deadline)

    # ---------- cosine windows ----------
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(window_family(n, win), best, n, start_time, deadline)

    # ------------------------------------------------------------------
    # 3. Fine‑grained ternary search around the best parameters found
    # ------------------------------------------------------------------
    # product family refinement
    if prod_best["alpha"] is not None:
        radius = 2.0
        lo = max(0.0, prod_best["alpha"] - radius)
        hi = prod_best["alpha"] + radius
        def prod_score(a):
            return compute_score(product_family(n, a))
        ternary_search_family(prod_score, lo, hi, prod_best, "product",
                              start_time, deadline, max_iters=30)
        try_candidate(product_family(n, prod_best["alpha"]), best, n,
                      start_time, deadline)

    # left‑heavy refinement
    if left_best["beta"] is not None:
        radius = 2.0
        lo = max(0.0, left_best["beta"] - radius)
        hi = left_best["beta"] + radius
        def left_score(b):
            return compute_score(left_heavy_power_family(n, b))
        ternary_search_family(left_score, lo, hi, left_best, "left‑heavy",
                              start_time, deadline, max_iters=30)
        try_candidate(left_heavy_power_family(n, left_best["beta"]), best, n,
                      start_time, deadline)

    # Gaussian refinement
    if gauss_best["gamma"] is not None:
        radius = gauss_best["gamma"] * 0.5
        lo = max(1e-6, gauss_best["gamma"] - radius)
        hi = gauss_best["gamma"] + radius
        def gauss_score(g):
            return compute_score(gaussian_family(n, g))
        ternary_search_family(gauss_score, lo, hi, gauss_best, "gaussian",
                              start_time, deadline, max_iters=30)
        try_candidate(gaussian_family(n, gauss_best["gamma"]), best, n,
                      start_time, deadline)

    # ------------------------------------------------------------------
    # 4. Simple convex combination of the two best families found
    # ------------------------------------------------------------------
    # we keep the two vectors with the lowest scores seen so far
    # (they are already stored inside *best*; we just extract a second one)
    second_vec = None
    second_score = float("inf")
    # gather a couple of promising vectors for mixing
    cand_pool = []
    cand_pool.append(product_family(n, prod_best["alpha"] if prod_best["alpha"] else 1.0))
    cand_pool.append(left_heavy_power_family(n, left_best["beta"] if left_best["beta"] else 1.0))
    cand_pool.append(gaussian_family(n, gauss_best["gamma"] if gauss_best["gamma"] else 1.0))
    for vec in cand_pool:
        sc = compute_score(vec / vec.sum())
        if sc < second_score and not np.allclose(vec / vec.sum(),
                                                  best["vec"], atol=1e-12):
            second_score = sc
            second_vec = vec / vec.sum()

    if second_vec is not None:
        for t in np.linspace(0.0, 1.0, 21):
            if time.time() >= deadline - 150.0:
                break
            mixed = t * best["vec"] + (1.0 - t) * second_vec
            mixed = project_to_simplex(mixed)
            try_candidate(mixed, best, n, start_time, deadline)

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient (mirror descent) optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0 and best["vec"] is not None:
        print(f"[{time.time() - start_time:.1f}] starting EG optimisation")
        eg_vec, eg_score = eg_optimize(
            best["vec"],
            deadline - 30.0,
            max_iters=600_000,
            eta0=0.7,
        )
        if eg_score < best["score"]:
            best["score"] = eg_score
            best["vec"] = eg_vec.copy()
            print(f"[{time.time() - start_time:.1f}] EG finished, new best = {eg_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Aggressive LP refinement loop (only for moderate n)
    # ------------------------------------------------------------------
    if n <= 4096 and time.time() < deadline - 80.0:
        print(f"[{time.time() - start_time:.1f}] entering LP refinement loop")
        mix_weight = 0.2
        lp_iters = 0
        while time.time() < deadline - 20.0 and lp_iters < 60:
            conv = convolution_fft(best["vec"], best["vec"])
            M = float(conv.max())
            new_vec = lp_refine_step(best["vec"], M, mix_weight=mix_weight)
            new_score = compute_score(new_vec)
            if new_score < best["score"] - 1e-8:
                best["score"] = new_score
                best["vec"] = new_vec
                print(f"[{time.time() - start_time:.1f}] LP step {lp_iters} improved to {new_score:.6f}")
            else:
                mix_weight = max(0.03, mix_weight * 0.85)
            lp_iters += 1

    # ------------------------------------------------------------------
    # 7. Symmetrisation (the problem is symmetric)
    # ------------------------------------------------------------------
    if best["vec"] is not None:
        sym = (best["vec"] + best["vec"][::-1]) * 0.5
        sym = project_to_simplex(sym)
        sym_score = compute_score(sym)
        if sym_score < best["score"]:
            best["score"] = sym_score
            best["vec"] = sym
            print(f"[{time.time() - start_time:.1f}] symmetrisation improved best = {sym_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation & return
    # ------------------------------------------------------------------
    final_seq = best["vec"].tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
