"""Search for low‑peak step-function using
- wide analytic families (Gaussian, Chebyshev, Kaiser, ...),
- smooth‑max exponentiated‑gradient (mirror descent) with soft‑max weighting,
- LP‑based refinement,
- final simulated‑annealing hill‑climb.
All vectors are kept symmetric and projected onto the probability simplex.
"""
import time, math
import numpy as np
from scipy.optimize import linprog
from scipy.signal import windows

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Based on algorithm of Michelot (1986)
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# -------------------------------------------------------------------------
# Analytic families (all produce symmetric vectors when appropriate)
# -------------------------------------------------------------------------

def product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()

def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()

def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()

def gaussian_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()

def window_family(n: int, win_type: str) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    if win_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif win_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif win_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()

def kaiser_family(n: int, beta: float) -> np.ndarray:
    w = windows.kaiser(n, beta)
    return w / w.sum()

def chebwin_family(n: int, att: float) -> np.ndarray:
    w = windows.chebwin(n, at=att)
    return w / w.sum()

def dpss_family(n: int, half_bw: float) -> np.ndarray:
    w = windows.dpss(n, half_bw, Kmax=1)[0]   # first DPSS eigenvector
    return w / w.sum()

def random_symmetric_start(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        r = np.random.rand(half) ** np.random.uniform(0.5, 5.0)
        full = np.concatenate([r, r[::-1]])
    else:
        r = np.random.rand(half) ** np.random.uniform(0.5, 5.0)
        mid = np.random.rand() ** np.random.uniform(0.5, 5.0)
        full = np.concatenate([r, [mid], r[::-1]])
    full = np.maximum(full, 0.0)
    full = full / full.sum()
    return full

# -------------------------------------------------------------------------
# Gradient utilities for the smooth‑max EG optimiser
# -------------------------------------------------------------------------

def weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given probability vector a (length n) and weight vector w (length m=2n-1),
    return ∇_a  Σ_k w_k (a*a)_k = 2 * cross‑correlation(a, w).
    """
    n = a.size
    m = w.size          # = 2n-1
    # convolution of a with reversed w
    conv = convolution_fft(a, w[::-1])            # length n+m-1
    # extract the segment conv[m-n : m] and reverse to align indices
    grad = 2.0 * conv[m - n : m][::-1]
    return grad

def eg_smooth_optimize(start_a: np.ndarray,
                       deadline: float,
                       max_iters: int = 300_000,
                       eta0: float = 0.6,
                       beta_start: float = 5.0,
                       beta_end: float = 25.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the sub‑gradient is a soft‑max
    combination of all convolution entries.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)                # length 2n-1
        beta = beta_start + (beta_end - beta_start) * t / max_iters
        # soft‑max weights (stable)
        w = np.exp(beta * (conv - conv.max()))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        grad = weighted_gradient(a, w)
        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * grad)
        a = a * factor

        # keep symmetry, then project onto simplex
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG‑smooth improved to {sc:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# LP‑based refinement (same as before, kept for continuity)
# -------------------------------------------------------------------------

def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    n = f.shape[0]
    m = 2 * n - 1
    # Build convolution matrix A (size m×n)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    c = -np.ones(n, dtype=np.float64)   # maximise Σ g → minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return f
    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f
    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new

# Stochastic hill‑climb (simulated‑annealing style)
# -------------------------------------------------------------------------

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        # enforce symmetry and simplex
        cand = (cand + cand[::-1]) * 0.5
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence`` within the given wall‑clock budget.
    The routine combines a broad analytic scan, a smooth‑max
    exponentiated‑gradient optimiser, LP refinements and a final
    simulated‑annealing hill‑climb.  Symmetry is enforced throughout.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0.  Optional fallback (if the harness supplies a pre‑computed vector)
    # -----------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1.  Choose problem size
    # -----------------------------------------------------------------
    # We start with a relatively large symmetric window; 4096 gives a good
    # trade‑off between expressiveness and speed.
    n = 4096
    print(f"[{time.time() - start_time:.1f}] using length n={n}")

    # -----------------------------------------------------------------
    # Helper to test a candidate (normalises internally)
    # -----------------------------------------------------------------
    def try_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            return
        a = a / a.sum()
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best score = {sc:.6f}")

    # -----------------------------------------------------------------
    # 2.  Analytic families (broad scan)
    # -----------------------------------------------------------------
    # Uniform baseline (symmetric automatically)
    try_candidate(np.full(n, 1.0 / n, dtype=np.float64))

    # Product family (U‑shaped)
    for alpha in np.concatenate([np.linspace(0.0, 8.0, 33),
                                 np.linspace(8.5, 60.0, 30)]):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(product_family(n, alpha))

    # Left‑heavy (monotone) – we keep only the symmetric part by mirroring
    for beta in np.linspace(0.0, 15.0, 31):
        if time.time() >= deadline - 250.0:
            break
        half = left_heavy_power_family(n // 2, beta)
        a_sym = np.concatenate([half, half[::-1]])
        try_candidate(a_sym)

    # Exponential “U” family
    for gamma in np.linspace(0.0, 12.0, 31):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(exp_u_family(n, gamma))

    # Gaussian family
    for gamma in np.logspace(-4, 2, 30):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(gaussian_family(n, gamma))

    # Standard cosine windows
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(window_family(n, win))

    # Kaiser windows
    for beta in [0.5, 2.0, 5.0, 10.0, 15.0, 20.0]:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(kaiser_family(n, beta))

    # Chebyshev windows
    for att in [20, 30, 40, 50, 60, 80, 100]:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(chebwin_family(n, att))

    # DPSS (Slepian) windows
    for bw in np.linspace(0.5, 4.0, 16):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(dpss_family(n, bw))

    # Random symmetric starts
    for _ in range(250):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(random_symmetric_start(n))

    # -----------------------------------------------------------------
    # 3.  Initialise optimiser from the current best
    # -----------------------------------------------------------------
    if best_seq is None:

"""Search for low‑peak step‑function using:
- wide analytic families (Gaussian, Chebyshev, Kaiser, DPSS, …),
- smooth‑max exponentiated‑gradient descent with soft‑max weighting,
- LP‑based refinement,
- final simulated‑annealing hill‑climb.
All vectors are kept symmetric and projected onto the probability simplex.
"""

import time, math
import numpy as np
from scipy.optimize import linprog
from scipy.signal import windows

# -------------------------------------------------------------------------
# Basic utilities
# -------------------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------------------
# Analytic families (all produce symmetric vectors when appropriate)
# -------------------------------------------------------------------------

def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑0.5‑i))^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone heavy‑left: w_i ∝ (i+0.5)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential: w_i ∝ exp(‑γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian: w_i ∝ exp(‑γ·(i‑mid)²)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, win_type: str) -> np.ndarray:
    """Standard cosine windows."""
    i = np.arange(n, dtype=np.float64)
    if win_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif win_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif win_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def kaiser_family(n: int, beta: float) -> np.ndarray:
    w = windows.kaiser(n, beta)
    return w / w.sum()


def chebwin_family(n: int, att: float) -> np.ndarray:
    w = windows.chebwin(n, at=att)
    return w / w.sum()


def dpss_family(n: int, half_bw: float) -> np.ndarray:
    w = windows.dpss(n, half_bw)  # first DPSS eigenvector
    return w / w.sum()


def random_symmetric_start(n: int) -> np.ndarray:
    """Random symmetric vector (positive, normalised)."""
    half = n // 2
    if n % 2 == 0:
        r = np.random.rand(half) ** np.random.uniform(0.5, 5.0)
        full = np.concatenate([r, r[::-1]])
    else:
        r = np.random.rand(half) ** np.random.uniform(0.5, 5.0)
        mid = np.random.rand() ** np.random.uniform(0.5, 5.0)
        full = np.concatenate([r, [mid], r[::-1]])
    full = np.maximum(full, 0.0)
    return full / full.sum()


# -------------------------------------------------------------------------
# Gradient utilities for the smooth‑max EG optimiser
# -------------------------------------------------------------------------

def weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a*a)_k with respect to a.
    Returns 2·cross‑correlation(a, w), a vector of length n.
    """
    n = a.size
    m = w.size                  # = 2n‑1
    conv = convolution_fft(a, w[::-1])   # length n+m‑1
    grad = 2.0 * conv[m - n : m][::-1]
    return grad


def eg_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.6,
    beta_start: float = 5.0,
    beta_end: float = 25.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the sub‑gradient is a soft‑max
    combination of all convolution entries.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)                     # length 2n‑1
        beta = beta_start + (beta_end - beta_start) * t / max_iters
        w = np.exp(beta * (conv - conv.max()))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        grad = weighted_gradient(a, w)
        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * grad)
        a = a * factor

        # enforce symmetry and simplex constraint
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG‑smooth improved to {sc:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# -------------------------------------------------------------------------
# LP‑based refinement (classic state‑of‑the‑art step)
# -------------------------------------------------------------------------

def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (m × n) so that A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# -------------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# -------------------------------------------------------------------------

def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
) -> tuple[np.ndarray, float]:
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                       # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # enforce symmetry + simplex
        cand = (cand + cand[::-1]) * 0.5
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    # -----------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.asarray(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    a = (a + a[::-1]) * 0.5          # enforce symmetry
    a = project_to_simplex(a)
    best_seq = a.copy()
    best_score = compute_score(a)

    # -----------------------------------------------------------------
    # 4. Smooth‑max EG optimisation (core descent)
    # -----------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print(f"[{time.time() - start_time:.1f}] starting smooth‑max EG optimisation")
        a_opt, sc_opt = eg_smooth_optimize(
            a,
            deadline - 180.0,
            max_iters=600_000,
            eta0=0.5,
            beta_start=4.0,
            beta_end=30.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] after EG smooth best = {best_score:.6f}")

        # One perturbed restart
        if time.time() < deadline - 180.0:
            perturb = a * np.exp(0.03 * np.random.randn(n))
            perturb = np.maximum(perturb, 0.0)
            perturb /= perturb.sum()
            a_opt2, sc_opt2 = eg_smooth_optimize(
                perturb,
                deadline - 120.0,
                max_iters=300_000,
                eta0=0.5,
                beta_start=4.0,
                beta_end=30.0,
            )
            if sc_opt2 < best_score:
                best_score = sc_opt2
                a = a_opt2
                print(f"[{time.time() - start_time:.1f}] after perturbed EG smooth best = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 5. LP‑based refinement loop
    # -----------------------------------------------------------------
    if time.time() < deadline - 100.0:
        print(f"[{time.time() - start_time:.1f}] starting LP refinement loop")
        mix_weight = 0.2
        lp_iters = 0
        while time.time() < deadline - 30.0 and lp_iters < 40:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc = compute_score(a_new)
            if sc < best_score - 1e-7:
                best_score = sc
                a = a_new
                print(f"[{time.time() - start_time:.1f}] LP step {lp_iters} improved to {best_score:.6f}")
            else:
                mix_weight = max(0.02, mix_weight * 0.85)
            lp_iters += 1

    # -----------------------------------------------------------------
    # 6. Final symmetry check (the problem is symmetric)
    # -----------------------------------------------------------------
    a = (a + a[::-1]) * 0.5
    a = project_to_simplex(a)
    sc_sym = compute_score(a)
    if sc_sym < best_score:
        best_score = sc_sym
        print(f"[{time.time() - start_time:.1f}] symmetrisation improved best = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 7. Stochastic hill‑climb (simulated annealing)
    # -----------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 8. Return final sequence
    # -----------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
