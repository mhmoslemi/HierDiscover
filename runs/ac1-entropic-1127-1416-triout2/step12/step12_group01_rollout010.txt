def optimize_with_smoothing(n, init_a, deadline):
    # allocate v = np.log(init_a + eps) # avoid zeros.
    # eps = 1e-12
    v = np.log(np.maximum(init_a, 1e-12))
    # Adam state
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    t = 0
    best_a = init_a.copy()
    best_score = compute_score(best_a)

    # schedule for mu
    mu0 = 4.0
    mu_final = 2000.0   # adjust based on time
    total_time = deadline - time.time()
    # We'll compute mu based on fraction done: mu = mu0 * (mu_final/mu0) ** (elapsed/total)
    while time.time() < deadline:
        t += 1
        elapsed = time.time() - start_time
        fraction = min(elapsed/total_time, 1.0)
        mu = mu0 * (mu_final / mu0) ** fraction

        a = np.exp(v)
        a = a / a.sum()
        conv = full_convolution(a)
        # compute smooth max loss
        conv_max = conv.max()
        # stable exp
        exp_terms = np.exp(mu * (conv - conv_max))
        Z = exp_terms.sum()
        w = exp_terms / Z   # probabilities
        # grad_a = 2 * sum_k w_k * a shifted; compute via convolution
        a_rev = a[::-1]
        # Use FFT: conv_w_a_rev = full_convolution(w, a_rev) # length = len(w)+len(a_rev)-1 = (2n-1)+n-1=3n-2
        conv_w_a_rev = full_convolution(w, a_rev)
        # take slice n-1 .. 2n-2 inclusive (size n)
        grad_a = 2.0 * conv_w_a_rev[(n-1):(2*n-1)]
        # gradient w.r.t v
        dot = np.dot(a, grad_a)
        grad_v = a * (grad_a - dot)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        lr = 0.1   # learning rate, may adjust
        m = beta1 * m + (1-beta1) * grad_v
        v_hat = beta2 * v_hat + (1-beta2) * (grad_v**2)
        m_hat = m / (1 - beta1**t)
        v_hat_corr = v_hat / (1 - beta2**t)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + 1e-8)

        # evaluate exact score every few iterations
        if t % 10 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[smooth_opt] iter {t}, score {cur_score:.6f}")

    return best_a

def fft_convolution(x, y):
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())

"""
non‑negative step‑function optimiser for ``evaluate_sequence``

Algorithm
---------
1. Global exploration (families + many Dirichlet draws) – keep the
   best few candidates (as probability vectors, i.e. sum = 1).

2. Smooth‑max gradient descent on the simplex.
   * Parameterise a probability vector as a soft‑max of free variables
     ``v``: a_i = exp(v_i)/∑_j exp(v_j).
   * Use a smooth approximation of the max in the objective:
         L(a) = 2·n·(1/μ)·log∑_k exp(μ·(a*a)_k)
     with a slowly growing temperature ``μ``.
   * The gradient w.r.t. ``v`` can be written in closed form using
     FFT‑based convolutions:
         g_i = a_i·( 4·n·∑_t a_t·w_{i+t} – ⟨a,g_a⟩ )
     where ``w`` is the soft‑max over the convolution entries.
   * Adam (adaptive moments) is applied to the free variables ``v``.
   * The schedule for ``μ`` starts low (smooth landscape) and
     exponentially grows to a large value, so the algorithm finally
     optimises the true max‑norm.

3. Optional short simulated‑annealing / EG polish of the best
   solution found.

All steps respect the 1000 s wall‑clock budget.  The final sequence
is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a power‑of‑two FFT."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (self‑convolution)."""
    return fft_convolve(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1):
       2·n·max( a*a ).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families for the global sweep
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_cosine(n: int, power: float = 1.0) -> np.ndarray:
    """a_i ∝ |cos(π·i/(n‑1) − π/2)|^power  – a smooth bump."""
    x = np.cos(np.pi * (np.arange(n, dtype=float) / (n - 1) - 0.5))
    a = np.abs(x) ** power
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration (baseline)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 12):
    """Return up to ``max_keep`` best probability vectors found by
    deterministic sweeps and random Dirichlet draws."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law sweep
    for p in np.linspace(0.1, 2.0, 240):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(0.0001, 0.07, 340):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta sweep (medium resolution)
    al_vals = np.linspace(0.2, 5.0, 30)
    be_vals = np.linspace(0.2, 5.0, 30)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # cosine bump family – a few powers
    for power in [0.5, 1.0, 2.0, 3.0]:
        if time.time() >= deadline:
            break
        try_candidate(generate_cosine(n, power))

    # massive Dirichlet draws
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1000
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser on the simplex
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float,
                    mu_start: float = 5.0,
                    mu_end: float = 1500.0,
                    lr: float = 0.02) -> np.ndarray:
    """
    Optimise 2·n·max(a*a) using a smooth log‑sum‑exp approximation.
    The probability vector a is parameterised as soft‑max(v) to
    automatically satisfy non‑negativity and sum‑to‑one constraints.
    Returns the best vector (as a NumPy array) seen before the deadline.
    """
    n = a0.shape[0]
    eps = 1e-14
    # initialise the free variables – avoid exact zeros
    v = np.log(np.maximum(a0, eps))

    # Adam state
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = a0.copy()
    best_val = compute_score(best_a)

    start = time.time()
    total_time = max(deadline - start, 1e-6)
    t = 0

    while time.time() < deadline:
        t += 1
        elapsed = time.time() - start
        frac = min(elapsed / total_time, 1.0)
        # geometric growth of μ from mu_start to mu_end
        mu = mu_start * (mu_end / mu_start) ** frac

        # current probability vector
        a = np.exp(v)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # self‑convolution
        conv = full_convolution(a)
        conv_max = conv.max()

        # soft‑max weights (stable)
        exp_terms = np.exp(mu * (conv - conv_max))
        Z = exp_terms.sum()
        w = exp_terms / Z  # shape (2·n‑1,)

        # gradient w.r.t. a:
        # grad_a_i = 4·n·∑_t a_t·w_{i+t}
        a_rev = a[::-1]
        conv_w_rev = fft_convolve(w, a_rev)  # length = 3·n‑2
        grad_a = 4.0 * n * conv_w_rev[(n - 1):(2 * n - 1)]

        # gradient w.r.t. v (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_v = a * (grad_a - dot)

        # Adam update on free variables v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat_corr = v_hat / (1.0 - beta2 ** t)

        v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps_adam)

        # occasional evaluation of the true objective
        if t % 20 == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[smooth_opt] iter {t}, exact score {cur_val:.6f}")

        # small random jitter to escape plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (1.0 / (mu + 1.0))
            v += jitter

    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (retained for a final quick pass)
# ----------------------------------------------------------------------


def eg_polish(a0: np.ndarray, deadline: float,
              eta0: float = 0.7, tol: float = 1e-10) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * tol
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if it % 500 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()
    return best_a

# Simple simulated‑annealing polish (optional)

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a, best_val = a.copy(), cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a, cur_val = cand, val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈15 % of total budget)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - (budget_s - 5.0) * 0.85
    if time.time() < baseline_deadline:
        cand_vectors = baseline_search(n, baseline_deadline, max_keep=12)
        for vec in cand_vectors:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smooth‑max Adam optimisation (major part of the budget)
    # -----------------------------------------------------------------
    smooth_deadline = deadline - (budget_s - 5.0) * 0.10   # leave ~10 % for final polish
    if best_seq is not None and time.time() < smooth_deadline:
        # Polish each of the retained baseline candidates
        to_polish = [np.array(best_seq, dtype=float)]
        # also add a few random restarts if time permits
        remaining = max(0, int((smooth_deadline - time.time()) // 30))
        for _ in range(remaining):
            conc = np.random.choice([0.2, 0.5, 1.0, 2.0, 4.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            to_polish.append(a_start)

        for idx, vec in enumerate(to_polish):
            if time.time() >= smooth_deadline:
                break
            print(f"[smooth] polishing candidate {idx + 1}/{len(to_polish)}")
            refined = smooth_optimize(vec, smooth_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth] new best score = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final quick EG polish (optional, uses remaining budget)
    # -----------------------------------------------------------------
    final_deadline = deadline
    if best_seq is not None and time.time() < final_deadline:
        a_start = np.array(best_seq, dtype=float)
        refined = eg_polish(a_start, final_deadline)
        val = compute_score(refined)
        if val < best_val:
            best_val, best_seq = val, refined.tolist()
            print(f"[eg‑final] improved score = {val:.6f}")

    # -----------------------------------------------------------------
    # 4. Optional simulated‑annealing fine‑tuning (short run)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 2.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        if val < best_val:
            best_val, best_seq = val, final.tolist()
            print(f"[anneal] final score = {val:.6f}")

    # -----------------------------------------------------------------
    # 5. Fallback to a uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 6. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
