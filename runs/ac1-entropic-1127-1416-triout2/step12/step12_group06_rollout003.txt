def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    n = 4096  # maybe 8192 if time
    # initial candidate from parametric sweep using broader families (including logistic)
    a0 = parametric_sweep(n, deadline - 100) # keep some time for annealing
    
    best = a0
    best_score = compute_score(best)

    # Simulated annealing
    cur = best.copy()
    cur_score = best_score

    # Annealing parameters
    init_temp = 1e-3 * best_score   # maybe relative to objective difference
    # Actually temperature to accept worse moves: set initial temp = 0.01 * best_score maybe
    temp = init_temp
    temp_decay = 0.9995   # per iteration
    step_scale = 0.05
    step_decay = 0.9999

    while time.time() < deadline:
        # pick random i,j
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1

        delta = step_scale * np.random.rand() * cur[i]
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure non-neg and reproject
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"[anneal] new best score {best_score:.6f} at {time.time() - start_time:.1f}s")
                # optionally increase step_scale after improvement
                step_scale = min(step_scale * 1.02, 0.5)
        else:
            # acceptance probability
            if temp > 0:
                prob = math.exp(-(cand_score - cur_score) / temp)
                if np.random.rand() < prob:
                    cur = cand
                    cur_score = cand_score

        # decay
        temp *= temp_decay
        step_scale *= step_decay
        # avoid step_scale too small
        if step_scale < 1e-6:
            step_scale = 1e-6

    # final hill climb from best
    hill_deadline = deadline
    a_final = hill_climb(best, hill_deadline, max_iter=500_000)
    final_score = compute_score(a_final)
    if final_score < best_score:
        best = a_final
        best_score = final_score

    # output
    final_seq = best.tolist()
    print(f"\nSearch finished. Final score = {evaluate_sequence(final_seq):.6f}")
    return final_seq

def full_convolution(a):
    n = a.shape[0]
    m = 2*n - 1
    n_fft = 1 << ((m-1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv
```

def logistic(n, center_frac, scale):
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    a = 1.0 / (1.0 + np.exp((i - center) / scale))
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a

Optimiser for the self‑convolution upper bound.

Algorithm:
    1. Broad parametric sweep over several families (power‑law, geometric,
       mixed plateau‑power, beta, logistic, random Dirichlet).  The best
       vector from this stage is kept as the starting point.
    2. A short projected sub‑gradient polishing pass (to quickly reduce
       the max‑convolution from the sweep).
    3. Simulated‑annealing style stochastic mass‑transfer search:
       repeatedly move a small amount of mass from one coordinate to another,
       accepting improvements and occasional deteriorations according to a
       decaying temperature schedule.  The current best vector is projected
       onto the probability simplex after each move.
    4. Final deterministic hill‑climb (greedy mass transfers) for fine
       polishing.

All vectors are normalised to have sum = 1, which makes the objective
scale‑invariant.  The implementation works with any length; a length of
4096 provides a good trade‑off between solution quality and runtime
within the 1000 s budget.

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())        # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i} (0 < q < 1)."""
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Plateau of length L followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{alpha} (1-x_i)^{beta},   x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


def logistic_shape(n: int, center_frac: float, scale_frac: float) -> np.ndarray:
    """Decreasing logistic shape: a_i ∝ 1/(1+exp((i-center)/scale))."""
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    scale = max(scale_frac * n, 1e-6)          # avoid division by zero
    a = 1.0 / (1.0 + np.exp((i - center) / scale))
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


def random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw a random point from Dirichlet(conc,…,conc)."""
    if concentration <= 0:
        return None
    y = np.random.gamma(concentration, 1.0, size=n)
    if y.sum() == 0.0:
        return None
    return y / y.sum()

def subgradient_polish(a0: np.ndarray, time_deadline: float,

        for k in active:

def anneal_search(a_start: np.ndarray, time_deadline: float) -> np.ndarray:
