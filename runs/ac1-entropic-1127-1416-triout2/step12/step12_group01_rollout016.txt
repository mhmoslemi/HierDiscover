def cross_corr_linear(p, a):
    n = a.shape[0]
    m = p.shape[0]  # 2n-1
    L = 1 << ((n + m - 1 - 1).bit_length())
    Fa = np.fft.fft(a, L)
    Fp = np.fft.fft(p, L)
    corr = np.fft.ifft(Fa * np.conj(Fp)).real
    # correlation k means sum_i a[i] * p[i + k] with zero padding beyond boundaries.
    # This mapping corresponds to indices 0..n-1 we need.
    return corr[:n]

        beta = min(beta_end, beta_start * (t/1000.0))   # increase temperature (beta) gradually

        # compute softmax

        # Adam update
        lr = 0.05 / (1 + 0.001 * t)  # maybe decreasing

        a = np.maximum(a, 0)

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
1. Initialise a feasible distribution (fallback, uniform or a random Dirichlet).
2. Normalise it to the convenient scale Σa = √(2·n); with this normalisation
   the objective coincides with the maximum entry of the auto‑convolution.
3. Repeatedly solve a *linear programme*:
        maximise   Σ b_j
        s.t.       (f * b)_k ≤ M   for all k,
                   b_j ≥ 0,
   where f is the current distribution and M = max(f*f).  The LP finds a
   distribution b that spreads the mass while respecting the same worst‑case
   convolution bound.
   The obtained b is renormalised to √(2·n) and mixed with f:
        f ← (1–τ)·f + τ·b .
   This step is a Frank‑Wolfe–type update that provably never increases M.
4. After the LP phase a short exponentiated‑gradient (EG) polishing run
   (sub‑gradient on the true max‑convolution) refines the solution.
5. A final brief simulated‑annealing pass tries to escape any remaining
   local plateaux.
6. The best sequence (scaled back to Σa = 1) is returned.  All heavy work
   respects the 1000 s wall‑clock budget.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (linear‑time via FFT)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())        # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (Σa = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------


def lp_improve(f: np.ndarray, deadline: float, tau0: float = 0.4) -> np.ndarray:
    """
    Perform a sequence of LP‑based updates to reduce max(conv(f,f)).
    ``f`` must have Σf = sqrt(2·n).  The routine returns the best vector seen.
    """
    n = f.shape[0]
    sqrt_two_n = np.sqrt(2.0 * n)

    # Ensure the normalisation is exact (avoids drift)
    f = f * (sqrt_two_n / f.sum())

    best_f = f.copy()
    best_M = full_convolution(f).max()
    tau = tau0
    it = 0

    while time.time() < deadline:
        it += 1

        # Build the convolution matrix A where (A @ b)[k] = (f * b)[k]
        col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
        row = np.concatenate([np.array([f[0]], dtype=f.dtype),
                               np.zeros(n - 1, dtype=f.dtype)])
        A = toeplitz(col, row)                # shape (2·n‑1, n)

        M = best_M                             # current bound

        # Linear program: maximise Σ b_j   ⇔  minimise –Σ b_j
        c = -np.ones(n, dtype=np.float64)      # minimise –sum(b)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Solve using HiGHS (fast for dense problems of this size)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            # Very unlikely, but bail out safely
            print(f"[LP] iteration {it}: linprog failed – {res.message}")
            break

        b = res.x
        if b.sum() == 0.0:
            break

        # Scale b back to the required norm Σb = √(2·n)
        scale = sqrt_two_n / b.sum()
        g = b * scale

        # Frank‑Wolfe mixing step
        f_new = (1.0 - tau) * f + tau * g
        # Re‑normalise (numerical drift)
        f_new = f_new * (sqrt_two_n / f_new.sum())

        M_new = full_convolution(f_new).max()
        improved = M_new < best_M - 1e-12

        if improved:
            best_M = M_new
            best_f = f_new.copy()
            f = f_new
            # keep τ reasonably large while we are making progress
            tau = min(0.9, tau * 1.02)
        else:
            # No progress – shrink the step‑size
            tau *= 0.7
            if tau < 1e-4:
                # step too small, stop the LP phase
                break

        # occasional progress report
        if it % 20 == 0 or not improved:
            print(f"[LP] iter {it:4d}  M = {best_M:.8f}  τ = {tau:.5f}")

    return best_f


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (sub‑gradient on the true max‑conv)
# ----------------------------------------------------------------------


def eg_polish(a0: np.ndarray, deadline: float,
              eta0: float = 0.7, tol: float = 1e-10) -> np.ndarray:
    """
    Multiplicative EG descent on  max_k (a*a)_k  over the simplex.
    Returns the best vector observed before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * tol
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: g_i = 2·∑_{k∈active} a_{k-i}
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            jdx = k - idx
            g[idx] += 2.0 * a[jdx]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # projection onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # small random jitter every few hundred steps
        if it % 500 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (optional)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb with a slowly cooling temperature."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a

    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # Problem size – 2048 balances quality and speed
    n = 2048
    sqrt_two_n = np.sqrt(2.0 * n)

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Initialise from fallback (if present) or a simple uniform start
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            init = np.array([float(x) for x in fallback], dtype=float)
            init = np.maximum(init, 0.0)
            if init.sum() == 0:
                init = np.full(n, 1.0 / n)
            else:
                init = init / init.sum()
            val = compute_score(init)
            best_seq = init.tolist()
            best_val = val
            print(f"[fallback] score = {val:.6f}")
        except Exception:
            pass

    if best_seq is None:
        # simple uniform start (guaranteed feasible)
        init = np.full(n, 1.0 / n, dtype=float)
        best_seq = init.tolist()
        best_val = compute_score(init)
        print(f"[uniform start] score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 1. LP improvement phase (≈ 40 % of the total time)
    # -----------------------------------------------------------------
    lp_deadline = deadline - (budget_s * 0.55)  # leave ~45 % for polishing
    if time.time() < lp_deadline:
        a = np.array(best_seq, dtype=float)
        a_lp = lp_improve(a * (sqrt_two_n / a.sum()), lp_deadline)
        # bring back to Σa = 1 for the downstream phases
        a_lp_norm = a_lp / a_lp.sum()
        val = compute_score(a_lp_norm)
        print(f"[LP phase] best score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_lp_norm.tolist()

    # -----------------------------------------------------------------
    # 2. Exponentiated‑gradient polishing (the majority of the budget)
    # -----------------------------------------------------------------
    eg_deadline = deadline - (budget_s * 0.10)   # reserve ~10 % for annealing
    if time.time() < eg_deadline:
        a0 = np.array(best_seq, dtype=float)
        refined = eg_polish(a0, eg_deadline)
        val = compute_score(refined)
        print(f"[EG polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (short run)
    # -----------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, deadline)
        val = compute_score(final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Safety fallback – uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
