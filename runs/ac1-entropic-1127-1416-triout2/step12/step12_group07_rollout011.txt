def adam_optimize(a0, deadline, initial_lr=0.1):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        # compute gradient w.r.t smooth max with tau
        tau = max(1e-6, 0.01 / np.sqrt(t))  # decreasing
        g = smooth_gradient(a, tau)
        # adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = -initial_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a + step)
        # optionally record best score
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a
```

"""
Propose a non‑negative step‑function (list of heights) that minimises the
objective used by ``evaluate_sequence``.

Algorithm
---------
1. **Broad random / parametric sweep** – generate many candidate vectors
   using simple families (power‑law, geometric, beta, cosine‑power,
   triangular, exponential) and random Dirichlet / block shapes.
   Keep the best few (lowest exact score).

2. **Smooth projected Adam optimisation** – starting from the current best
   candidate, minimise a smooth approximation of the max‑convolution:
   fτ(a)=2·n·τ·log(∑exp(conv(a,a)/τ)).  The gradient is
   g=4·n·(a∗p) where p is the soft‑max over the convolution entries.
   We perform Adam updates with Euclidean projection onto the probability
   simplex (Σa=1, a≥0).  The temperature τ is annealed (τ∝1/√t) so the
   optimisation gradually focuses on the true max.

3. **Fine‑grained stochastic hill‑climb** – a greedy mass‑transfer
   neighbourhood search that accepts only improvements, with an adaptive
   step size and occasional random jitter to escape plateaux.

All stages respect the overall wall‑clock budget (default 1000 s).  The
returned list sums to one, which makes the score scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    m = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power of two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    return np.fft.irfft(Z, n=n_fft)[:m]


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length = 2·len(a)‑1)."""
    return conv_fft(a, a)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a normalised vector
    (Σa == 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    """a_i = (i+offset)^(-p)   (offset>0 to avoid division by zero)."""
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, r: float) -> np.ndarray:
    """a_i = r^i  (geometric decay)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, beta: float) -> np.ndarray:
    """a_i = (1 - i/(n-1))^beta   (monotonically decreasing)."""
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    a /= a.sum()
    return a


def cosine_power(n: int, expo: float) -> np.ndarray:
    """a_i = cos(π i / (2 n))^expo  (decreasing cosine shape)."""
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def triangle_shape(n: int, peak_frac: float) -> np.ndarray:
    """Linear decrease to zero after L = peak_frac·n entries."""
    L = max(1, int(peak_frac * n))
    a = np.maximum(0.0, L - np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def exponential_decay(n: int, scale: float) -> np.ndarray:
    """a_i = exp(-scale·i)."""
    a = np.exp(-scale * np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    """
    Coarse piecewise‑constant shape: generate K positive values (Dirichlet),
    sort decreasingly and repeat each value to fill the n entries.
    """
    blocks = rng.dirichlet(np.ones(K))
    blocks = np.sort(blocks)[::-1]                     # decreasing
    repeats = np.full(K, n // K, dtype=int)
    remainder = n - repeats.sum()
    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)[:n]
    a /= a.sum()
    return a


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the simplex (no monotonicity)."""
    a = rng.dirichlet(np.ones(n))
    return a

# ----------------------------------------------------------------------
# Smooth gradient (log‑sum‑exp) utilities
# ----------------------------------------------------------------------
def smooth_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        fτ(a) = 2·n·τ·log Σ_i exp(conv_i(a)/τ)
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)            # length 2n‑1
    # stable soft‑max over conv / τ
    max_c = conv.max()
    exp_args = (conv - max_c) / tau
    w = np.exp(exp_args)
    p = w / w.sum()                           # weights, length 2n‑1
    # g = 4·n·(a * p)  (truncated to first n entries)
    conv_ap = conv_fft(a, p)[:n]
    return 4.0 * n * conv_ap


# ----------------------------------------------------------------------
# Projected Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Adam optimisation with simplex projection; returns best vector seen."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    # initial temperature (a fraction of the initial spread)
    init_tau = max(1e-6, (full_convolution_fft(a).max() - full_convolution_fft(a).min()) * 0.05)

    while time.time() < deadline:
        t += 1
        # anneal temperature
        tau = max(1e-9, init_tau / np.sqrt(t))
        g = smooth_gradient(a, tau)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule
        lr = 0.2 / np.sqrt(t)

        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a + step
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

        # early termination if stagnating for a while
        if t % 500 == 0:
            # if no improvement in last 500 iters, stop early
            if sc >= best_sc * (1.0 + 1e-4):
                break

    return best_a


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float, max_iter: int = 800_000) -> np.ndarray:
    """Tiny random mass‑transfer moves, greedy acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]               # move a fraction of mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        if no_improve > 150_000:
            # shake to escape flat plateaus
            jitter = rng.normal(scale=0.001, size=n)
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the outer harness
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback (if a previously good construction is available)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global parametric & random sweep
    # -----------------------------------------------------------------
    n = 2048
    sweep_deadline = min(deadline - 500.0, start_time + 150.0)  # ~150 s for sweep
    if time.time() < sweep_deadline:
        top_candidates = sweep_candidates_extended(n, sweep_deadline, top_k=12,
                                                  rng=np.random.default_rng(seed + 1))
        # keep the best among them
        best_sc, best_vec = top_candidates[0]
        print(f"[global sweep] best score = {best_sc:.6f}")
        if best_sc < best_val:
            best_val, best_seq = best_sc, best_vec.tolist()

    # -----------------------------------------------------------------
    # 2. Smooth Adam optimisation (main refinement)
    # -----------------------------------------------------------------
    adam_deadline = min(deadline - 150.0, time.time() + 600.0)  # give up to 600 s
    if best_seq is not None and time.time() < adam_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_opt = adam_optimize(a0, adam_deadline)
        val = compute_score(a_opt)
        print(f"[adam optimise] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – fall back to a uniform vector if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        uniform = [1.0 / n] * n
        best_seq = uniform
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Official evaluation (includes the security wrappers)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported (official) score = {final_score:.6f}")
    return best_seq
