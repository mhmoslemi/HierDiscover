"""
Minimalist optimiser for the step‑function objective.

The strategy:

1. Choose a relatively large problem size (2^15 = 32768).  The bound
   2·n·max(conv) / (∑a)² improves with n as the continuous optimum
   (value ≈ 1) becomes reachable.

2. Initialise the distribution with a short scan over the symmetric‑product
   family a_i ∝ ((i+½)(n‑½‑i))^{-α}.  The scanning uses only a few
   values of α (0.30 … 0.70) and the exact objective (no surrogate).

3. Refine the initial guess with three cheap optimisation phases,
   all respecting the simplex constraint:

   * **Exponentiated‑gradient (mirror descent)** on the true max‑convolution.
     The sub‑gradient is assembled analytically; the step‑size decays as
     η_t = η₀ / √t.  This phase drives down the worst convolution entry.

   * **Projected sub‑gradient descent** – a few thousand cheap steps with
     decreasing step size, again based on the active entry.

   * **Simple simulated‑annealing hill‑climb** to escape any remaining
     local plateau.

4. The best vector found is returned as a plain Python list.
   All phases stop as soon as the global deadline (budget‑2 s) is reached.

The code uses only NumPy and the built‑in ``time``/``math`` modules,
contains no closures or lambdas, and prints progress messages for
debugging.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution a*b using real FFT.
    Returned length = len(a) + len(b) – 1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the evaluator.
    The vector a is assumed to be non‑negative and normalised to sum = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of [Wang & Carreira‑Perpiñán 2013].
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for a quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    For α=0.5 this is the discrete arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a

# Optimisers
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 300_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Step size η_t = η₀ / √t.
    Returns the best vector seen and its score.
    """
    n = start_a.size
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]          # correct gradient (diagonal included)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)            # multiplicative update
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        if t % 5_000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG iter {t:>6d} – score {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Very cheap per iteration; we can afford many steps.
    """
    n = start_a.size
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2_000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad iter {t:>7d} – score {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style random walk on the simplex.
    Occasionally accepts uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 1e-4          # initial temperature
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

            a = a_eg
            print(f"[{time.time():.1f}] EG improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6.  Projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() + subg_time < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient phase (~{subg_time:.1f}s)")
        a_sub, sc_sub = subgrad_optimize(a,
                                         time.time() + subg_time,
                                         max_iters=1_500_000,
                                         step0=0.07)
        if sc_sub < best_val:
            best_val = sc_sub
            a = a_sub
            print(f"[{time.time():.1f}] Sub‑gradient improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7.  Optional symmetrisation (helps for some shapes but not mandatory).
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_val:
        best_val = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] Symmetrisation improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8.  Final simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    if time.time() + sa_time < deadline:
        print(f"[{time.time():.1f}] SA hill‑climb (~{sa_time:.1f}s)")
        a_sa, sc_sa = hill_climb_sa(a,
                                    time.time() + sa_time,
                                    max_steps=2_000_000)
        if sc_sa < best_val:
            best_val = sc_sa
            a = a_sa
            print(f"[{time.time():.1f}] SA improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final evaluation via the provided evaluator (for the harness).
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    print(f"Length of the returned sequence: {len(final_seq)}")
    return final_seq
