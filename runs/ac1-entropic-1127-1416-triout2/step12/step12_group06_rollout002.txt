"""
Optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Broad parametric sweep** – explore many analytic families
   (power‑law, geometric, beta, sinusoidal, plateau+power, random
   sparse, mixtures of betas, triangular).  Each candidate is
   normalised to a probability simplex (Σa = 1) and the best vector
   is kept.

2. **Exponentiated‑gradient (mirror descent)** – a fast
   non‑negative optimisation on the simplex.  The sub‑gradient of the
   current maximal convolution entry is computed, an exponential
   update is performed and the vector is re‑normalised.  The step‑size
   decays as 1/√t.

3. **Simulated annealing** – global random perturbations with a
   Metropolis acceptance rule.  The perturbation size (σ) is
   geometrically cooled, allowing a coarse search early and fine
   tuning later.

4. **Stochastic hill‑climb** – a simple greedy move that swaps a
   fraction of mass between two coordinates, with adaptive step‑size
   and occasional jitter to escape plateaus.

All phases respect the overall 1000 s budget, leaving a small safety
margin.  If no candidate is found a uniform vector is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Families of candidate vectors
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    """a_i ∝ q^{i} (0 < q < 1)."""
    if not (0.0 < q < 1.0):
        return None
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    """Plateau of constant value followed by a power‑law tail."""
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i ∝ x_i^{alpha} (1-x_i)^{beta},
    where x_i = (i+0.5)/n.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


def sin_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^{p}."""
    x = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    a = x ** p
    a /= a.sum()
    return a


def triangular_shape(n: int, p: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+1)^{p}·(n-i)^{p}, a symmetric hill."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** p
    a /= a.sum()
    return a


def random_sparse(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k uniformly random support points with equal weight."""
    if k <= 0 or k > n:
        return None
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a


def mix_two_beta(n: int, a1: float, b1: float, a2: float, b2: float,
                 w: float, rng: np.random.Generator) -> np.ndarray:
    """Mixture w·Beta(a1,b1) + (1‑w)·Beta(a2,b2)."""
    p = beta_shape(n, a1, b1)
    q = beta_shape(n, a2, b2)
    if p is None or q is None:
        return None
    a = w * p + (1.0 - w) * q
    a = project_to_simplex(a)
    return a

        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---------------- mixed plateau + power ----------
    for L_frac in np.linspace(0.02, 0.40, 20):
        for p in np.linspace(0.20, 1.20, 31):
            if time.time() >= deadline:
                return best_vec
            a = mixed_plateau_power(n, L_frac, p)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---------------- beta shapes -------------------
    alphas = np.linspace(-0.99, 4.0, 30)   # allow more negative exponents
    betas = np.linspace(-0.99, 4.0, 30)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # ---------------- sinusoidal shapes ---------------
    for p in np.linspace(0.5, 5.0, 46):
        if time.time() >= deadline:
            return best_vec
        a = sin_shape(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---------------- triangular shapes ----------------
    for p in np.linspace(0.1, 3.0, 30):
        if time.time() >= deadline:
            return best_vec
        a = triangular_shape(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # ---------------- random sparse vectors ----------------
    for k in range(12, 80, 4):   # vary sparsity
        for _ in range(120):
            if time.time() >= deadline:
                return best_vec
            a = random_sparse(n, k, rng)
            val = compute_score(a)
            if val < best_val:

    # ---------------- mixtures of two betas -----------------
    for w in np.linspace(0.2, 0.8, 13):

# ----------------------------------------------------------------------
# Phase 2 – exponentiated‑gradient (mirror descent) on the simplex
# ----------------------------------------------------------------------
def exp_gradient_descent(a0: np.ndarray, deadline: float,
                         eta0: float = 0.8) -> np.ndarray:
    """Mirror descent with exponential updates, projected onto the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        eps = 1e-12
        active = np.where(conv >= M - eps)[0]

        # sub‑gradient of each active entry: 2·a shifted
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        eta = eta0 / math.sqrt(t)

        # exponential update with clipping to avoid overflow
        update = -eta * g
        update = np.clip(update, -50.0, 50.0)
        a = a * np.exp(update)
        a = np.maximum(a, 1e-30)           # keep numerically positive
        a /= a.sum()                       # projection onto simplex (Euclidean)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Phase 3 – simulated annealing (Gaussian perturbations)
# ----------------------------------------------------------------------
def simulated_annealing(a0: np.ndarray, deadline: float,
                        sigma0: float = 0.08,
                        sigma_min: float = 1e-5,
                        cooling: float = 0.9994) -> np.ndarray:
    """Metropolis‑type annealing on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    sigma = sigma0

    while time.time() < deadline:
        # propose a Gaussian perturbation and project back to simplex
        cand = a + sigma * np.random.randn(len(a))
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        delta = cand_score - cur_score
        # acceptance probability – temperature linked to sigma
        if delta < 0 or np.random.rand() < math.exp(-delta / (sigma * 10.0 + 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        sigma = max(sigma_min, sigma * cooling)

    return best_a


# ----------------------------------------------------------------------
# Phase 4 – stochastic hill‑climb (aggressive local search)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    """Greedy stochastic moves with adaptive step‑size."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]            # move a fraction of source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)       # keep non‑negative
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            # escape possible flat region
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the harness
    deadline = start + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # 0. Fallback – previous construction if available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Broad parametric sweep (≈ 250 s)
    # ------------------------------------------------------------------
    sweep_deadline = start + min(250.0, budget_s * 0.25)
    n = 2048
    if time.time() < sweep_deadline:
        a_initial = parametric_sweep(n, sweep_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # ------------------------------------------------------------------
    # 2. Exponentiated‑gradient polishing (≈ 250 s)
    # ------------------------------------------------------------------
    grad_deadline = start + min(500.0, budget_s * 0.5)
    if best_seq is not None and time.time() < grad_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_exp = exp_gradient_descent(a0, grad_deadline, eta0=0.8)
        val = compute_score(a_exp)
        print(f"[exp‑grad] best score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_exp.tolist()

    # ------------------------------------------------------------------
    # 3. Simulated annealing (≈ 250 s)
    # ------------------------------------------------------------------
    anneal_deadline = start + min(750.0, budget_s * 0.75)
    if best_seq is not None and time.time() < anneal_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_sa = simulated_annealing(a0, anneal_deadline,
                                   sigma0=0.08, sigma_min=1e-5, cooling=0.9994)
        val = compute_score(a_sa)
        print(f"[sim‑anneal] best score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_sa.tolist()

    # ------------------------------------------------------------------
    # 4. Final hill‑climb polishing (remainder of time)
    # ------------------------------------------------------------------
    hill_deadline = deadline
    if best_seq is not None and time.time() < hill_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_hc = hill_climb(a0, hill_deadline, max_iter=800_000)
        val = compute_score(a_hc)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_hc.tolist()

    # ------------------------------------------------------------------
    # Safeguard – uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
