def adam_smooth_max(a0, deadline, lr=0.1, tau_start=0.2, tau_end=1e-8, tau_decay=0.99, decay_every=100):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = compute_score(a)
    tau = tau_start
    while time.time() < deadline:
        t += 1
        # compute conv
        conv = full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()
        # compute gradient
        rev_w = w[::-1]
        size = 1 << ((a.shape[0] + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        corr = np.fft.irfft(A * B, n=size)[: a.shape[0] + rev_w.shape[0] - 1]
        idx = (len(conv)-1) - np.arange(a.shape[0])  # conv length = 2n -1
        grad = 2.0 * corr[idx]
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)
        # optionally adjust tau
        if t % decay_every == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)
        # track best
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # occasional jitter
        if t % 2000 == 0:
            a = a + np.random.randn(a.shape[0]) * (lr * 0.001 / np.sqrt(t+1))
            a = project_to_simplex(a)
    return best_a

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

The optimiser combines three ideas that improve on the previous baseline:

1. **Wider baseline exploration** – we sweep many more parametric families,
   especially very “U‑shaped” Beta distributions (α,β≪1) and strong power‑law
   decays (p up to 6).  The best few candidates are kept.

2. **Adam‑based smooth‑max descent** – a soft‑max approximation of the
   max‑convolution is minimised with the Adam optimiser.  After each Adam
   update the vector is projected onto the probability simplex, staying
   feasible for ``evaluate_sequence``.  The temperature τ is annealed from
   0.5 down to 1e‑9, giving an ever‑more accurate approximation of the true
   max.

3. **Polishing & stochastic hill‑climb** – a short projected sub‑gradient
   phase followed by a simulated‑annealing phase finishes the search.

All phases respect a 1000 s wall‑clock budget.  The best sequence found is
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        # If everything is non‑positive, return the uniform distribution.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(n, 0.0)
    return a / s


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(n, 0.0)
    return a / s


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape a_i ∝ (i+1)^{α-1}·(n-i)^{β-1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(n, 0.0)
    return a / s


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i‑mid)²/(2σ²))."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(n, 0.0)
    return a / s


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(n, 0.0)
    return a / s


# ----------------------------------------------------------------------
# Extended baseline exploration
# ----------------------------------------------------------------------


def baseline_search_extended(
    n: int, deadline: float, max_keep: int = 10, random_draws: int = 3000
):
    """
    Generate many candidates from a wide variety of families and Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as NumPy arrays).
    """
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)  # guarantees sum=1
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Power‑law sweep (very steep decays as well)
    for p in np.linspace(0.05, 6.0, 121):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) Exponential sweep (larger λ range)
    for lam in np.linspace(0.0005, 0.6, 301):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta sweep – focus on U‑shapes (α,β << 1)
    al_vals = np.concatenate(
        (np.linspace(0.02, 0.2, 10), np.linspace(0.3, 2.0, 18))
    )
    be_vals = al_vals.copy()
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Gaussian sweep (wide sigmas)
    for sigma in np.linspace(0.1 * n, 0.7 * n, 30):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # 5) Sine‑power sweep (very spiky)
    for p in np.linspace(0.1, 10.0, 51):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # 6) Random Dirichlet draws with many concentration levels
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    per_conc = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_max(
    a0: np.ndarray,
    deadline: float,
    lr: float = 0.12,
    tau_start: float = 0.5,
    tau_end: float = 1e-9,
    tau_decay: float = 0.97,
    decay_every: int = 80,
) -> np.ndarray:
    """
    Minimise a smooth approximation of ``max(conv)`` using Adam.
    The temperature τ is annealed to make the surrogate converge to the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = compute_score(a)

    tau = tau_start
    conv_len = 2 * n - 1

    while time.time() < deadline:
        t += 1

        # --- smooth‑max weights ---
        conv = full_convolution(a)               # length 2n‑1
        shift = conv.max()
        # use shift for numerical stability
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()            # soft‑max distribution, length 2n‑1

        # --- gradient (2·a ⋆ w) ---
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        corr_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]
        idx = (conv_len - 1) - np.arange(n)       # pick the part needed for the gradient
        grad = 2.0 * corr_full[idx]

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)                # keep feasibility

        # --- temperature schedule ---
        if t % decay_every == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # keep the best ever seen
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (lr * 0.004 / np.sqrt(t + 1.0))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (same as before, but longer)
# ----------------------------------------------------------------------


def subgradient_polish(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.30,
) -> np.ndarray:
    """Classic projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of gradients from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (final stochastic hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    init_temp: float = 1e-3,
    final_temp: float = 1e-8,
    cooling_factor: float = 0.9993,
    step_scale_init: float = 0.05,
    step_scale_min: float = 1e-6,
) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two indices and transfer a small amount
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a + jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam smooth‑max optimisation from promising starts
    # -----------------------------------------------------------------
    # collect a few promising starts
    start_vectors = []
    if best_seq is not None:
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        start_vectors.append(best_vec)
    # uniform vector
    start_vectors.append(np.full(n, 1.0 / n))

        start_vectors.append(project_to_simplex(np.random.rand(n)))

    # time budgeting for the Adam phase
    adam_deadline = min(deadline - 180.0, time.time() + 500.0)
    for idx, init_vec in enumerate(start_vectors):
        if time.time() >= adam_deadline:
            break
        local_deadline = adam_deadline - (time.time() - start_time) - 1.0
        if local_deadline <= time.time():
            break
        refined = adam_smooth_max(init_vec, local_deadline)
        val = compute_score(refined)
        print(f"[adam {idx}] refined score = {val:.6f}")
        if val < best_val:

    # -----------------------------------------------------------------
    polish_deadline = min(deadline - 60.0, time.time() + 120.0)
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_refined)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:

    # 4. Simulated annealing (final stochastic hill‑climb)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")

    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
