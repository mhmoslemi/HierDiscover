"""
propose_candidate
=================

A multi‑phase heuristic for the “minimal max‑self‑convolution” problem.
The routine

* starts from several analytically‑motivated families (Beta‑shapes,
  exponential, mixed, random), scanning a fine grid of the “shape”
  exponent α (both edge‑heavy and centre‑heavy).

* refines the best start with a **quadratic‑L2** optimiser that minimises
  ‖a∗a‖₂² (the sum of squares of the convolution).  Reducing the L₂‑norm
  tends to flatten the whole convolution and therefore lowers its peak.

* continues with the previously successful **exponentiated‑gradient**
  (mirror descent), a **smooth‑max Adam** optimiser and a long projected
  **sub‑gradient** descent.

* finishes with a small simulated‑annealing hill‑climb.

All phases respect the global 1000 s wall‑clock budget and the
best‑found sequence (a plain Python list) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution by real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the evaluator (up to the invariant scaling).
    The vector `a` is assumed non‑negative and normalised to sum = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# Analytic families (starting points)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    α>0 → edges heavy, α<0 → centre heavy (Beta(p,p) with p=1‑α).
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law a_i ∝ (i+½)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Unimodal exponential shape a_i ∝ exp(-γ·dist_to_center)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Mixture of uniform and exponential."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sorted powers of uniform draws)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑type start."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

def l2_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 400_000,
                step0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Minimise ‖a∗a‖₂² (sum of squares of the self‑convolution) by
    projected gradient descent.  The gradient is
        ∇‖c‖₂² = 4·conv(c, a_rev)
    where c = a∗a and a_rev is a reversed copy of a.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        c = convolution_fft(a, a)           # length 2n‑1
        rev = a[::-1]
        grad_full = convolution_fft(c, rev)  # length 3n‑2
        grad = 4.0 * grad_full[len(a) - 1 : len(a) - 1 + len(a)]

        step = step0 / math.sqrt(t)
        a = a - step * grad
        a = project_to_simplex(a)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] L2 opt improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 800_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) on the true max‑convolution."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed max w.r.t. a (same as in the literature)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        # chain rule through softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                 # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (a list of heights) that
    minimises the supplied `evaluate_sequence` objective.
    The routine respects the supplied time budget and returns a plain
    Python list ready for evaluation.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load a fallback candidate if the harness supplies one.
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose the problem size.
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # Prefer a fairly long vector (more degrees of freedom) but not
    # unreasonably large for the time budget.
    if fallback_len >= 4096:
        n = fallback_len
    else:
        n = 8192
    print(f"[{time.time():.1f}] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Quick analytic scan over several families.
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        # a) Symmetric beta‑family, both edge‑heavy (α>0) and centre‑heavy (α<0)
        alphas = np.concatenate((
            np.linspace(-0.8, -0.1, 15),   # centre‑heavy
            np.linspace(0.0, 1.2, 25)     # edge‑heavy (includes α=0 → uniform)
        ))
        for alpha in alphas:
            if time.time() >= deadline - 5.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (α={alpha:.3f}) ⇒ {best_score:.6f}")

        # b) Monotone power law (left‑heavy)
        betas = np.linspace(0.2, 2.5, 18)
        for beta in betas:
            if time.time() >= deadline - 5.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (β={beta:.3f}) ⇒ {best_score:.6f}")

        # c) Exponential “U‑shape”
        gammas = np.linspace(0.0, 12.0, 14)
        for gamma in gammas:
            if time.time() >= deadline - 5.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (γ={gamma:.3f}) ⇒ {best_score:.6f}")

        # d) Random starts (few)
        for _ in range(4):
            if time.time() >= deadline - 5.0:
                break
            vec = random_monotone_start(n) if np.random.rand() < 0.5 else random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) ⇒ {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Normalise the working vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)
    print(f"[{time.time():.1f}] after initial scan best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Split the remaining budget.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # Allocate heuristically (the numbers are just proportions, the actual
    # time taken depends on the cost of each phase).
    l2_time   = 0.12 * remaining
    eg_time   = 0.20 * remaining
    adam_time = 0.20 * remaining
    subg_time = 0.30 * remaining
    sa_time   = remaining - (l2_time + eg_time + adam_time + subg_time)

    # ------------------------------------------------------------------
    # 5.  L2‑norm flattening phase.
    # ------------------------------------------------------------------
    if time.time() + l2_time < deadline:
        print(f"[{time.time():.1f}] L2‑flattening phase (≈{l2_time:.1f}s)")
        a_l2, sc_l2 = l2_optimize(a, time.time() + l2_time,
                                 max_iters=600_000,
                                 step0=1.0)
        if sc_l2 < best_score:
            best_score = sc_l2
            a = a_l2
            print(f"[{time.time():.1f}] L2 phase improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Exponentiated‑gradient (mirror descent).
    # ------------------------------------------------------------------
    if time.time() + eg_time < deadline:
        print(f"[{time.time():.1f}] EG phase (≈{eg_time:.1f}s)")
        a_eg, sc_eg = eg_optimize(a, time.time() + eg_time,
                                 max_iters=800_000,
                                 eta0=0.6)
        if sc_eg < best_score:
            best_score = sc_eg
            a = a_eg
            print(f"[{time.time():.1f}] EG improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Adam optimisation of the smooth‑max surrogate.
    # ------------------------------------------------------------------
    if time.time() + adam_time < deadline:
        print(f"[{time.time():.1f}] Adam phase (≈{adam_time:.1f}s)")
        a_adam, sc_adam = adam_smooth_optimize(a,
                                                time.time() + adam_time,
                                                max_iters=400_000,
                                                beta_start=4.0,
                                                beta_end=5000.0,
                                                lr=0.018)
        if sc_adam < best_score:
            best_score = sc_adam
            a = a_adam
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Long projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() + subg_time < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient phase (≈{subg_time:.1f}s)")
        a_sub, sc_sub = subgrad_optimize(a,
                                         time.time() + subg_time,
                                         max_iters=5_000_000,
                                         step0=0.07)
        if sc_sub < best_score:
            best_score = sc_sub
            a = a_sub
            print(f"[{time.time():.1f}] Sub‑gradient improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Symmetrisation (optional).
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] Symmetrisation improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10.  Final simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    if time.time() + sa_time < deadline:
        print(f"[{time.time():.1f}] Hill‑climb (SA) phase (≈{sa_time:.1f}s)")
        a_sa, sc_sa = hill_climb_sa(a,
                                    time.time() + sa_time,
                                    max_steps=2_000_000)
        if sc_sa < best_score:
            best_score = sc_sa
            a = a_sa
            print(f"[{time.time():.1f}] Hill‑climb improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 11.  Final evaluation through the harness.
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
