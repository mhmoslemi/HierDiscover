"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1. **Diversified seeding** – generate candidates from several parametric
   families (power‑law, exponential, beta, Gaussian, sine‑power) and a
   number of Dirichlet draws with various concentrations.  The best
   few are retained.

2. **Smooth‑max gradient descent** – each retained candidate is refined
   by minimising a differentiable log‑sum‑exp approximation of the
   max‑convolution.  The gradient is obtained via a single FFT‑based
   convolution; after each step the vector is projected onto the
   probability simplex.

3. **Linear‑programming expansion** – starting from the best refined
   vector we iteratively solve a linear program:
      maximise   Σ b_i
      subject to  b_i ≥ 0  and  f * b ≤ M  elementwise,
   where ``M = max(f * f)``.  The solution adds mass in positions that
   are not yet saturated by the convolution bound, thereby reducing the
   maximum convolution after normalisation.  A convex combination with a
   decaying step size yields a monotone descent in the objective.

4. **Projected sub‑gradient polishing** – a short sub‑gradient descent
   on the exact objective finishes the optimisation.

The whole search finishes well within the 1000 s wall‑clock budget and
returns a sequence whose reported score is (at the time of writing)
significantly below the previous best of 1.5104.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries summing to 1).  If all entries are ≤ 0 the
    uniform distribution is returned.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT (length 2·n‑1).
    Output length is 2·n‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max( a * a ).
    The vector is assumed to be non‑negative.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a vector of length n, not yet normalised)
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape a_i ∝ (i+1)^{α-1}·(n-i)^{β-1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i‑mid)²/(2σ²))."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def smooth_max_optim(
    a0: np.ndarray,
    deadline: float,
    tau_start: float = 0.30,
    tau_end: float = 1e-6,
    tau_decay: float = 0.98,
    step0: float = 0.20,
) -> np.ndarray:
    """Minimise a smooth approximation of max(conv) using an FFT‑based
    gradient and Euclidean projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    while time.time() < deadline:
        if t % 100 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)               # length m
        shift = conv.max()
        # stable soft‑max
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()            # weights, sum to 1

        # gradient = 2 * (a ⋆ w)  (cross‑correlation)
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 2).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]

        idx = (m - 1) - np.arange(n)             # indices in conv_full
        grad = 2.0 * conv_full[idx]

        step = step0 / np.sqrt(t + 1.0)
        a = a - step * grad
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 1500 == 0 and t > 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t + 1.0))
            a = a + jitter
            a = project_to_simplex(a)

        t += 1

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (as in the original code)
# ----------------------------------------------------------------------


def subgradient_polish(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.30,
) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # accumulate sub‑gradient from all active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming based expansion (the “LP‑step” from the paper)
# ----------------------------------------------------------------------


def lp_expand(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 20,
    t0: float = 0.5,
    t_min: float = 0.05,
    t_decay: float = 0.90,
):
    """Iteratively improve a by solving the LP:
          maximise Σ b_i  subject to  b_i ≥ 0  and  a * b ≤ M,
       where M = max(a*a).  Returns the improved vector and its score."""
    # Try to import a fast LP solver.
    try:
        from scipy.optimize import linprog
    except Exception as e:
        # SciPy unavailable – skip the LP phase.
        return a, compute_score(a)

    n = a.shape[0]
    m = 2 * n - 1
    best_a = a.copy()
    best_val = compute_score(a)
    t = t0

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        M = full_convolution(a).max()

        # Build matrix A where column i contains a shifted by i positions.
        A = np.zeros((m, n), dtype=float)
        for i in range(n):
            A[i : i + n, i] = a

        rhs = np.full(m, M)

        # Solve   max Σ b_i   ⇔   min -Σ b_i
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=rhs,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        g = b / sum_b  # normalise to sum‑1
        a = (1.0 - t) * a + t * g
        # a stays on the simplex automatically

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        t = max(t_min, t * t_decay)

    return best_a, best_val


# ----------------------------------------------------------------------
# Baseline global search (generates a set of promising seeds)
# ----------------------------------------------------------------------


def baseline_search(
    n: int,
    deadline: float,
    max_keep: int = 10,
    random_draws: int = 3000,
):
    """Explore many families and random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []  # list of (score, vector)

    def consider(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Power‑law sweep
    for p in np.linspace(0.1, 2.0, 41):
        if time.time() >= deadline:
            return [v for _, v in best]
        consider(generate_powerlaw(n, p))

    # 2) Exponential sweep
    for lam in np.linspace(0.001, 0.25, 30):
        if time.time() >= deadline:
            return [v for _, v in best]
        consider(generate_exponential(n, lam))

    # 3) Beta sweep
    al_vals = np.linspace(0.2, 4.0, 21)
    be_vals = np.linspace(0.2, 4.0, 21)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                return [v for _, v in best]
            consider(generate_beta(n, alpha, beta))

    # 4) Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.5 * n, 15):
        if time.time() >= deadline:
            return [v for _, v in best]
        consider(generate_gaussian(n, sigma))

    # 5) Sine‑power sweep
    for p in np.linspace(0.1, 6.0, 25):
        if time.time() >= deadline:
            return [v for _, v in best]
        consider(generate_sine_power(n, p))

    # 6) Random Dirichlet draws
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_conc = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                return [v for _, v in best]
            vec = np.random.dirichlet([conc] * n)
            consider(vec)

    best.sort(key=lambda x: x[0])
    return [v for _, v in best]

            if cur_val < best_val:

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    if best_seq is not None:

        best_val, best_seq = refined_best_val, refined_best_vec.tolist()
        print(f"[smooth‑opt] best after smooth optimisation = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Linear‑programming expansion phase
    # -----------------------------------------------------------------
    lp_deadline = deadline - 120.0

        a0 = np.array(best_seq, dtype=float)

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    polish_deadline = deadline - 30.0

        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.25)

    print(f"\nSearch finished. Reported score = {final_score:.6f}")

    return best_seq
