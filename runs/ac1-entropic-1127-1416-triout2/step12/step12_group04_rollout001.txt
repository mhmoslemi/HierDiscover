    surrogate of `max(conv)` using the log‑sum‑exp trick.
    The gradient is obtained by a single FFT‑based correlation.
    After each update we project onto the probability simplex and
    enforce symmetry (a ← ½(a + a[::-1])) which is natural for the
    problem.
3.  A final exponentiated‑gradient (mirror‑descent) pass on the genuine
    `max(conv)` objective to polish the result.

All phases respect the global wall‑clock budget (default 1000 s) and
return the best sequence found (as plain Python floats).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # only for the fallback import – not used below

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (real inputs only)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (Σa = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (initial scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped (ends heavy) family."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone left‑heavy power law."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, window_type: str) -> np.ndarray:
    """Classic cosine windows."""
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing (left‑heavy) start."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (continuous surrogate of max)
# ----------------------------------------------------------------------


def smooth_optimize(start_a: np.ndarray,
                    start_time: float,
                    deadline: float,
                    max_iters: int = 200_000,
                    step0: float = 0.20,
                    alpha_start: float = 8.0,
                    alpha_end: float = 120.0) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth surrogate of max(conv) :

        f(a) = (1/α)·log Σ_k exp(α·conv_k)

    using projected gradient descent on the simplex.
    After each step we enforce symmetry a ← (a + a[::-1])/2.
    Returns the best vector encountered and its exact max‑objective value.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # full convolution a*a
        conv = convolution_fft(a, a)

        # smooth‑max parameter (increase gradually)
        alpha = alpha_start + (alpha_end - alpha_start) * min(t / max_iters, 1.0)

        # softmax weights (numerically stable)
        conv_max = conv.max()
        w = np.exp(alpha * (conv - conv_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # gradient of smooth‑max: 2·correlation(a, w)
        w_rev = w[::-1]
        grad_full = convolution_fft(a, w_rev)          # length 3n‑2
        slice_part = grad_full[n - 1 : 2 * n - 1]      # length n
        grad = 2.0 * slice_part[::-1]                  # ⟹ ∇f

        # step size (diminishing)
        step = step0 / math.sqrt(t)

        # normalise gradient to keep steps comparable across dimensions
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            a = a - (step / gnorm) * grad

        # enforce non‑negativity, sum‑to‑one and symmetry
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        # occasional reporting
        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:.1f}] smooth_opt improved to {sc:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the true max‑objective (mirror descent)
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                start_time: float,
                deadline: float,
                max_iters: int = 200_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) directly on max(conv).
    Uses the sub‑gradient of the active convolution entry.
    Returns the best vector seen.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (see comment in original code)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]  # diagonal term weight 1

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        # stay on the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # enforce symmetry (helps a lot)
        a = (a + a[::-1]) * 0.5

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:.1f}] EG improved to {sc:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  The routine respects a wall‑clock budget (default 1000 s)
    and returns a plain Python list suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from the harness (if present)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]       # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a problem size
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # Use a fairly large vector unless we already have a long and good fallback
    n = fallback_len if fallback_len >= 1024 else 8192
    print(f"[{time.time() - start_time:.1f}] problem size n = {n}")

    # ------------------------------------------------------------------
    # Helper that evaluates a candidate (normalised internally)
    # ------------------------------------------------------------------
    def try_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Fast analytic scan (much lighter than the previous exhaustive sweep)
    # ------------------------------------------------------------------
    try_candidate(np.full(n, 1.0 / n, dtype=np.float64))                # uniform

    # product family – coarse sweep (fast)
    for alpha in np.linspace(0.0, 12.0, 13):
        if time.time() >= deadline - 200.0:
            break
        try_candidate(product_family(n, alpha))

    # left‑heavy power family
    for beta in np.linspace(0.0, 12.0, 13):
        if time.time() >= deadline - 200.0:
            break
        try_candidate(left_heavy_power_family(n, beta))

    # exponential‑U family
    for gamma in np.linspace(0.0, 12.0, 13):
        if time.time() >= deadline - 200.0:
            break
        try_candidate(exp_u_family(n, gamma))

    # Gaussian family
    for gamma in np.logspace(-4, 1.5, 12):
        if time.time() >= deadline - 200.0:
            break
        try_candidate(gaussian_family(n, gamma))

    # cosine windows
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 200.0:
            break
        try_candidate(window_family(n, win))

    # a few random monotone starts
    for _ in range(30):
        if time.time() >= deadline - 200.0:
            break
        try_candidate(random_monotone_start(n))

    # ------------------------------------------------------------------
    # 3. Initialise the working vector (fallback to uniform if needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        a_cur = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a_cur = np.asarray(best_seq, dtype=np.float64)
    a_cur = np.maximum(a_cur, 0.0)
    a_cur = a_cur / a_cur.sum()
    a_cur = (a_cur + a_cur[::-1]) * 0.5          # enforce symmetry from the start
    best_seq = a_cur.copy()
    best_score = compute_score(a_cur)

    # ------------------------------------------------------------------
    # 4. Smooth‑max gradient descent (major workhorse)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print(f"[{time.time() - start_time:.1f}] starting smooth‑max descent")
        a_opt, sc_opt = smooth_optimize(
            a_cur,
            start_time,
            deadline - 180.0,
            max_iters=250_000,
            step0=0.25,
            alpha_start=8.0,
            alpha_end=130.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a_cur = a_opt
            print(f"[{time.time() - start_time:.1f}] after smooth‑max best = {best_score:.6f}")

        # try a second run from a perturbed start (helps escape local minima)
        if time.time() < deadline - 180.0:
            pert = a_cur * np.exp(0.03 * np.random.randn(n))
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            a_opt2, sc_opt2 = smooth_optimize(
                pert,
                start_time,
                deadline - 120.0,
                max_iters=150_000,
                step0=0.20,
                alpha_start=10.0,
                alpha_end=120.0,
            )
            if sc_opt2 < best_score:
                best_score = sc_opt2
                a_cur = a_opt2
                print(f"[{time.time() - start_time:.1f}] after perturbed smooth‑max best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final mirror‑descent (exponentiated gradient) on the true max
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print(f"[{time.time() - start_time:.1f}] final EG refinement")
        a_opt, sc_opt = eg_optimize(
            a_cur,
            start_time,
            deadline - 30.0,
            max_iters=180_000,
            eta0=0.55,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a_cur = a_opt
            print(f"[{time.time() - start_time:.1f}] after final EG best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation test (the optimum is symmetric)
    # ------------------------------------------------------------------
    a_sym = (a_cur + a_cur[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a_cur = a_sym
        print(f"[{time.time() - start_time:.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = a_cur.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
