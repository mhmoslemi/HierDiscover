"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Iterative linear‑programming (LP) refinement** –  
   Starting from a uniform vector `a` (∑a=1) we repeatedly solve  

        max   ∑ g_i
        s.t.  g_i ≥ 0,
              (a * g)_k ≤ M   ∀k               (1)

   where `M = max(a * a)` and `*` denotes linear convolution.
   The LP (1) is a sparse problem with (2n‑1) linear inequalities and n
   variables.  The optimal `g` is an extreme point that pushes mass as
   far to the left as possible while keeping the autocorrelation bound.
   After solving, `g` is normalised to sum = 1 and mixed with the
   current iterate:

        a ← (1‑t)·a + t·g        (t≈0.2, slowly decreased)

   This convex combination never raises the bound `M`; in practice
   `M` strictly decreases after a few iterations, yielding a flatter
   convolution and a lower objective value.

2. **Stochastic hill‑climbing polishing** –  
   The LP phase stops when the time budget is low.  A final
   greedy hill‑climb (mass transfer between two random positions,
   always accepting an improvement) is applied to squeeze out the
   last fraction of the bound.

Both phases respect the simplex constraint (`∑a = 1, a_i ≥ 0`) by
explicit projection (Euclidean) after each update.  All heavy linear
algebra uses FFT‑based convolution and a sparse LP matrix, keeping the
runtime well inside the 1000 s budget.

The returned list is a plain Python ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a*a (linear, not circular)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT, length = 2*len(a)-1."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A                                    # element‑wise square
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Exact objective identical to ``evaluate_sequence`` for Σa = 1
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Solve the LP: maximise sum(g) st. conv(a,g) ≤ M
# ----------------------------------------------------------------------
def solve_lp_max_sum(a: np.ndarray, M: float,
                     rows: np.ndarray, cols: np.ndarray) -> np.ndarray | None:
    """
    Returns the optimal g (as a 1‑D NumPy array) or None on failure.
    The constraint matrix is built from the pre‑computed ``rows`` and
    ``cols`` (same pattern for any a) – only the data values (f_i) change.
    """
    n = a.shape[0]
    # each column i appears in exactly n rows with coefficient a[i]
    data = np.repeat(a, n)               # length = n*n
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Use HiGHS interior‑point solver (fast for medium‑size LPs)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (fine polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """Greedy hill‑climb: move a fraction of mass from i to j, accept improvements."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]   # transfer fraction of mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        if no_improve > 120_000:
            # random shake to escape flat region
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The method first runs a few LP‑based
    refinement steps and then a stochastic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0        # safety margin

    # --- 0. Simple fallback (if a previously good sequence exists) ---
    best_seq = None
    best_val = float('inf')
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. LP‑based iterative refinement
    # ------------------------------------------------------------------
    n = 2048                                 # vector length; larger values give tighter lower bound
    a = np.full(n, 1.0 / n)                  # start from the uniform distribution
    best_score = compute_score(a)
    best_a = a.copy()
    print(f"[init] uniform score = {best_score:.6f}")

    # Pre‑compute the sparse matrix pattern (rows, cols) – independent of a
    i_idx = np.arange(n, dtype=np.int32)
    rows = (i_idx[:, None] + i_idx[None, :]).ravel()        # shape (n*n,)
    cols = np.repeat(i_idx, n)                             # shape (n*n,)

    iteration = 0
    t = 0.25                     # initial mixing weight
    while time.time() < deadline and iteration < 30:
        conv = full_convolution(a)
        M = conv.max()
        # Solve LP (1)
        g = solve_lp_max_sum(a, M, rows, cols)
        if g is None:
            print("[LP] solver failed, terminating LP phase")
            break
        sum_g = g.sum()
        if sum_g <= 0:
            print("[LP] zero‑sum solution, terminating")
            break

        # Normalise g to a probability vector
        g_norm = g / sum_g

        # Convex combination – slowly reduce the mixing weight
        a = (1.0 - t) * a + t * g_norm
        # (a stays non‑negative and sums to 1)

        cur_score = compute_score(a)
        print(f"[LP iter {iteration:02d}] M={M:.6e}, sum(g)={sum_g:.3f}, t={t:.3f}, score={cur_score:.6f}")

        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()

        # gradually shrink the step size to stabilise the process
        t = max(t * 0.92, 0.02)
        iteration += 1

        # Early stop if improvement stalls
        if iteration > 5 and best_score - cur_score < 1e-6:
            break

    # ------------------------------------------------------------------
    # 2. Final stochastic hill‑climb polishing
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        a_polished = hill_climb(best_a, hill_deadline)
        final_score = compute_score(a_polished)
        print(f"[hill‑climb] polished score = {final_score:.6f}")
        if final_score < best_score:
            best_score, best_a = final_score, a_polished

    # ------------------------------------------------------------------
    # 3. Safeguard – fall back to uniform if nothing found
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 4. Official evaluation (includes the security wrappers)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Official score = {final_val:.6f}")
    return final_seq
