   v = np.zeros_like(a0)
   t = 0
   best_a=a0.copy()
   best_val=compute_score(a0)
   while time.time()<deadline:
        t+=1
        # compute conv & weights as earlier
        conv=full_convolution(a)
        shift=conv.max()
        exp_vals=np.exp((conv-shift)/tau) # need tau schedule
        w=exp_vals/exp_vals.sum()
        # gradient:
        rev_w=w[::-1]
        # using FFT for cross-correlation.
        size = 1 << (( (a.shape[0] + rev_w.shape[0] -1) -1).bit_length()) # compute next pow2
        A=np.fft.rfft(a,size=size)
        B=np.fft.rfft(rev_w,size=size)
        conv_full=np.fft.irfft(A*B,size=size)[:a.shape[0]+rev_w.shape[0]-1]
        idx=a.shape[0]-1 - np.arange(a.shape[0])
        grad=2.0*conv_full[idx]
        # Adam update
        m=beta1*m+(1-beta1)*grad
        v=beta2*v+(1-beta2)*(grad**2)
        m_hat=m/(1-beta1**t)
        v_hat=v/(1-beta2**t)
        a = a - lr*m_hat/(np.sqrt(v_hat)+epsilon)
        a = np.maximum(a,0)
        a = a/np.sum(a)  # renormalize sum to 1 (project onto simplex)
        cur_val=compute_score(a)
        if cur_val<best_val:
           best_val=cur_val
           best_a=a.copy()
        # adjust tau schedule (e.g., multiply by decay each 100 steps)
        if t%200==0:

"""
Improved non‑negative step‑function optimiser for ``evaluate_sequence``.

The optimiser works in four stages:

1. **Extensive baseline exploration** – sweep several parametric families
   (power‑law, exponential, beta, Gaussian, sine‑power, power‑exponential)
   on a dense grid and also draw many random Dirichlet vectors.
   The best ~10 candidates are kept.

2. **Adam‑based smooth‑max minimisation** – from each retained candidate we
   minimise a smooth approximation of ``max(conv(a,a))`` using an Adam
   optimiser.  The temperature of the soft‑max is annealed automatically.
   The optimisation runs for a few hundred seconds, constantly keeping the
   best vector seen.

3. **Projected sub‑gradient polishing** – a classic sub‑gradient step on the
   exact ``max(conv)`` objective (the active set of maximal indices) refines
   the result further.

4. **Final simulated‑annealing escape** – a short Metropolis hill‑climb
   removes any remaining shallow local minima.

All phases obey the 1000 s wall‑clock budget.  The returned list of floats
constitutes a feasible sequence (non‑negative, sum>0) and is evaluated by
``evaluate_sequence`` by the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a)).
    The scaling invariance of ``evaluate_sequence`` makes this equivalent.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i‑mid)²/(2σ²))."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_power_exponential(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-lam·i)."""
    a = (np.arange(1, n + 1, dtype=float) ** (-p)) * np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_triangular(n: int, width_frac: float) -> np.ndarray:
    """
    Symmetric triangular shape.
    ``width_frac`` is the fraction of the full support occupied by the triangle.
    """
    if width_frac <= 0.0:
        return np.full(n, 0.0)
    width = max(2, int(np.round(width_frac * n)))
    # generate a linear ramp up then down
    ramp = np.arange(1, width // 2 + 1, dtype=float)
    if width % 2 == 0:
        tri = np.concatenate([ramp, ramp[::-1]])
    else:
        tri = np.concatenate([ramp, ramp[-2::-1]])  # peak repeated once
    a = np.zeros(n, dtype=float)
    start = (n - len(tri)) // 2
    a[start:start + len(tri)] = tri
    total = a.sum()
    if total == 0:
        return np.full(n, 0.0)
    return a / total


def generate_smoothed_random(n: int, scale: float = 0.02) -> np.ndarray:
    """
    Random Gaussian‑filtered noise, clipped to non‑negative and normalised.
    ``scale`` controls the width of the smoothing kernel.
    """
    rnd = np.random.randn(n)
    # simple moving‑average kernel approximated by a Gaussian
    x = np.arange(-3 * n, 3 * n + 1, dtype=float)
    kernel = np.exp(-0.5 * (x / (scale * n)) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(rnd, kernel, mode="same")
    smooth = np.maximum(smooth, 0.0)
    total = smooth.sum()
    if total == 0:
        return np.full(n, 0.0)
    return smooth / total


# ----------------------------------------------------------------------
# Baseline exploration (expanded)
# ----------------------------------------------------------------------


def add_to_best(best, vec, max_keep):
    """Insert ``vec`` into the sorted ``best`` list if it improves the score."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    vec = project_to_simplex(vec)
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=lambda x: x[0])


def extended_baseline_search(
    n: int, deadline: float, max_keep: int = 12, random_draws: int = 5000
):
    """Generate a diverse pool of candidates from many families."""
    best = []  # list of (score, vector)

    # 1) Power‑law sweep
    for p in np.linspace(0.05, 8.0, 161):
        if time.time() >= deadline:
            break
        add_to_best(best, generate_powerlaw(n, p), max_keep)

    # 2) Exponential sweep
    for lam in np.linspace(0.00001, 0.5, 250):
        if time.time() >= deadline:
            break
        add_to_best(best, generate_exponential(n, lam), max_keep)

    # 3) Beta sweep (random grid, dense enough)
    al_vals = np.linspace(0.2, 8.0, 41)
    be_vals = np.linspace(0.2, 8.0, 41)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            add_to_best(best, generate_beta(n, alpha, beta), max_keep)

    # 4) Gaussian sweep
    for sigma in np.linspace(0.05 * n, 0.45 * n, 31):
        if time.time() >= deadline:
            break
        add_to_best(best, generate_gaussian(n, sigma), max_keep)

    # 5) Sine‑power sweep
    for p in np.linspace(0.1, 12.0, 61):
        if time.time() >= deadline:
            break
        add_to_best(best, generate_sine_power(n, p), max_keep)

    # 6) Power‑exponential random sampling
    for _ in range(800):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.0, 8.0)
        lam = np.random.uniform(0.0, 0.5)
        add_to_best(best, generate_power_exponential(n, p, lam), max_keep)

    # 7) Triangular shapes
    for width in np.linspace(0.02, 0.6, 30):
        if time.time() >= deadline:
            break
        add_to_best(best, generate_triangular(n, width), max_keep)

    # 8) Smoothed random draws
    for _ in range(800):
        if time.time() >= deadline:
            break
        scale = np.random.uniform(0.01, 0.05)
        add_to_best(best, generate_smoothed_random(n, scale), max_keep)

    # 9) Dirichlet draws (various concentrations)
    concs = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0]
    per_conc = max(1, random_draws // len(concs))
    for conc in concs:
        for _ in range(per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            add_to_best(best, a, max_keep)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max approximation
# ----------------------------------------------------------------------


def adam_smooth_max(a0: np.ndarray, deadline: float,
                    lr: float = 0.20,
                    tau_start: float = 0.30,
                    tau_end: float = 1e-6,
                    tau_decay: float = 0.94) -> np.ndarray:
    """
    Adam optimiser for the smooth approximation of max(conv).

    The temperature ``tau`` is multiplied by ``tau_decay`` every 200 steps.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    best_a = a.copy()
    best_val = compute_score(a)

    # Adam state
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t = 0
    tau = tau_start
    while time.time() < deadline:
        t += 1

        # anneal temperature every 200 iterations
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # ----- smooth‑max computation ----------
        conv = full_convolution(a)                     # length m
        shift = conv.max()
        # stable soft‑max
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()

        # ----- gradient of soft‑max ----------
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 1 - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]

        idx = (m - 1) - np.arange(n)                  # indices for each coordinate
        grad = 2.0 * conv_full[idx]

        # ----- Adam step ----------
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex (ensures non‑negativity and sum=1)
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional random jitter to escape plateaus
        if t % 1500 == 0:
            jitter = np.random.randn(n) * (lr * 0.003 / np.sqrt(t + 1.0))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # treat all entries within 1e-12 relative tolerance as active
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # accumulate sub‑gradient from all active positions
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # jitter every few thousand steps to avoid stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (final stochastic escape)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 2e-3,
                        final_temp: float = 5e-7,
                        cooling_factor: float = 0.9997,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            # slowly increase the step size when we accept a move
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        # occasional jitter to keep the walk lively
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

# Main entry point expected by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.

    The function respects the 1000 second wall‑clock limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    n = 2048
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Fallback (if a previously discovered construction exists)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 200 s)
    # -----------------------------------------------------------------
    baseline_deadline = min(deadline - 600.0, start_time + 200.0)
    if time.time() < baseline_deadline:
        candidates = extended_baseline_search(
            n, baseline_deadline, max_keep=12, random_draws=5000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam smooth‑max optimisation from the top candidates
    # -----------------------------------------------------------------
    if best_seq is not None:
        grad_deadline = deadline - 200.0  # leave time for polishing & annealing
        # keep a few diverse start points
        start_vectors = []
        # current best (projected again to be safe)
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        start_vectors.append(best_vec)

        # a uniform vector (sometimes a good basin)
        start_vectors.append(np.full(n, 1.0 / n))

        # a few random points
        for _ in range(3):
            start_vectors.append(project_to_simplex(np.random.rand(n)))

        refined_best = best_vec.copy()
        refined_val = best_val

        for idx, start_vec in enumerate(start_vectors):
            if time.time() >= grad_deadline:
                break
            # give each start a slice of the remaining time
            local_deadline = grad_deadline - (time.time() - start_time) - 0.5
            if local_deadline <= time.time():
                break
            a_ref = adam_smooth_max(start_vec, local_deadline)
            val = compute_score(a_ref)
            print(f"[adam‑opt {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_ref.copy()

        best_val, best_seq = refined_val, refined_best.tolist()
        print(f"[adam‑opt] best after Adam optimisation = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (focus on true max)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_refined)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing (final stochastic hill‑climb)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fallback to a uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final evaluation with the official function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
