```
   n = len(a)
   best = a.copy()
   best_score = compute_score(a)
   for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        M = conv.max()
        # Build Toeplitz matrix
        col = np.r_[a, np.zeros(n-1)]
        row = np.r_[a[0], np.zeros(n-1)]

        # Objective -sum(b)
        c = -np.ones(n)
        # constraints: A @ b <= M
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), method='highs')
        if not res.success:
            print(f"LP iteration {it} failed")
            break
        b = res.x
        sum_b = b.sum()

            break
        g = b / sum_b

        a = (1 - t) * a + t * g
```

```
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iteration {i} improved to {best_score:.6f}")
        t *= 0.9  # decay step
    return best_a, best_score
```

The program searches for a non‑negative step‑function (a list of heights) that minimises  

```
    2·n·max(conv(a,a)) / (sum(a)²)
```

under the simplex constraint `sum(a)=1`.  
The search consists of the following stages

1. **Initialisation** – try a few analytic families (arcsine, exponential,
   left‑heavy…) and keep the best start.
2. **Linear‑programming polishing** – for the current vector `a` we solve  

   ```
   max   Σ b_i
   s.t.  conv(a,b) ≤ M   (M = max(conv(a,a)))
         b_i ≥ 0
   ```

   The solution `b` is normalised to a probability vector `g` and blended
   with `a` (`a ← (1‑t)a + t g`).  Repeating this step reduces the worst
   convolution value dramatically.
3. **Exponentiated‑gradient (mirror descent)** – sub‑gradient descent on the
   exact max‑convolution objective with a diminishing stepsize.
4. **Adam on a smooth‑max surrogate** – minimise  
   `f_β(a) = (1/β)·log Σ exp(β·(a*a)_k)` while annealing `β` up to 5000.
5. **Projected sub‑gradient descent** – cheap iterations that keep the
   iterate on the probability simplex.
6. **Optional symmetrisation** – the problem is invariant under reversal,
   averaging the vector with its mirror can improve the bound.
7. **Final simulated‑annealing hill‑climb** – a short stochastic local‑search
   to pick up any remaining tiny improvements.

All phases respect the global wall‑clock budget (default 1000 s).  The
implementation uses only NumPy, SciPy (`linprog`, `toeplitz`) and no unsafe
constructs.

```python
"""
search for a non‑negative step‑function that minimises the provided
evaluate_sequence() objective.  The algorithm combines an analytic start,
a linear‑programming polishing step, exponentiated‑gradient,
Adam on a smooth‑max surrogate, cheap projected sub‑gradient, optional
symmetrisation and a final simulated‑annealing hill‑climb.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the evaluator (up to the global scaling that
    cancels out).  The vector `a` is assumed to be non‑negative and
    normalised to sum = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    For α=0.5 this is the discrete arcsine distribution.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 1_200_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror‑descent) on the true max‑convolution."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry (k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if t % 10_000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate f_β."""
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a (via convolution with w)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 5_000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 8_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 20_000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Very simple simulated‑annealing style random walk on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming polishing phase
# ----------------------------------------------------------------------


def lp_polish(a: np.ndarray,
              deadline: float,
              max_iters: int = 7,
              start_t: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP
        max Σ b_i   s.t.  conv(a,b) ≤ M,  b ≥ 0
    where M = max(conv(a,a)).
    The solution b is normalised to a probability vector g and blended:
        a ← (1‑t)·a + t·g .
    This reduces the worst convolution entry.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    t = start_t

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        M = conv.max()

        # Build the Toeplitz matrix representing convolution with a.
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = toeplitz(col, row)[:, :n]          # (2n‑1) × n

        # Linear program: maximise Σ b_i  ⇔ minimise −Σ b_i
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.1f}] LP failed (iter {it}) – {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # No improvement possible.
            break

        g = b / sum_b                     # normalise to a probability vector
        a = (1.0 - t) * a + t * g
        a = project_to_simplex(a)         # numerical safety

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iteration {it} improved to {best_score:.6f}")

        # gradually shrink the blending factor
        t = max(t * 0.85, 0.02)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Main entry point.  Returns a plain Python list of non‑negative heights.
    The total wall‑clock time is limited by `budget_s` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # small safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  Load a fallback candidate if the harness supplies one.
    # --------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Choose problem size.
    # --------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # aim for a reasonably large n, but keep memory usage modest
    n_target = 2048 if fallback_len < 2048 else fallback_len
    print(f"[{time.time():.1f}] using sequence length n = {n_target}")

    # --------------------------------------------------------------
    # 2.  Quick scan over analytic families to obtain a good start.
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        alphas = np.linspace(0.25, 1.5, 26)
        for alpha in alphas:
            if time.time() >= deadline - 5.0:
                break
            vec = symmetric_product_family(n_target, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (α={alpha:.3f}) ⇒ {best_score:.6f}")

    if best_seq is None:
        a = np.full(n_target, 1.0 / n_target, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = a / a.sum()                     # enforce simplex
    best_seq = a.tolist()
    best_score = compute_score(a)
    print(f"[{time.time():.1f}] after scan best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Allocate time slices for the remaining phases.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # Rough split; will be adjusted on the fly.
    lp_time   = max(remaining * 0.15, 10.0)   # some seconds reserved for LP polishing
    eg_time   = max(remaining * 0.20, 30.0)
    adam_time = max(remaining * 0.25, 30.0)
    subg_time = max(remaining * 0.30, 30.0)
    sa_time   = max(remaining - (lp_time + eg_time + adam_time + subg_time), 10.0)

    # --------------------------------------------------------------
    # 4.  Linear‑programming polishing.
    # --------------------------------------------------------------
    if time.time() + lp_time < deadline:
        print(f"[{time.time():.1f}] LP polishing phase (≈{lp_time:.1f}s)")
        a_lp, sc_lp = lp_polish(a, time.time() + lp_time,
                                max_iters=7, start_t=0.5)
        if sc_lp < best_score:
            best_score = sc_lp
            a = a_lp
            print(f"[{time.time():.1f}] LP polishing improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Exponentiated‑gradient (mirror descent).
    # --------------------------------------------------------------
    if time.time() + eg_time < deadline:
        print(f"[{time.time():.1f}] EG phase (≈{eg_time:.1f}s)")
        a_eg, sc_eg = eg_optimize(a, time.time() + eg_time,
                                 max_iters=1_200_000,
                                 eta0=0.6)
        if sc_eg < best_score:
            best_score = sc_eg
            a = a_eg
            print(f"[{time.time():.1f}] EG improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # 6.  Adam optimisation of the smooth‑max surrogate.
    # --------------------------------------------------------------
    if time.time() + adam_time < deadline:
        print(f"[{time.time():.1f}] Adam phase (≈{adam_time:.1f}s)")
        a_adam, sc_adam = adam_smooth_optimize(a,
                                                time.time() + adam_time,
                                                max_iters=600_000,
                                                beta_start=4.0,
                                                beta_end=5000.0,
                                                lr=0.018)
        if sc_adam < best_score:
            best_score = sc_adam
            a = a_adam
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # 7.  Projected sub‑gradient descent.
    # --------------------------------------------------------------
    if time.time() + subg_time < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient phase (≈{subg_time:.1f}s)")
        a_sub, sc_sub = subgrad_optimize(a,
                                         time.time() + subg_time,
                                         max_iters=8_000_000,
                                         step0=0.07)
        if sc_sub < best_score:
            best_score = sc_sub
            a = a_sub
            print(f"[{time.time():.1f}] Sub‑gradient improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # 8.  Optional symmetrisation.
    # --------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] Symmetrisation improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # 9.  Final simulated‑annealing hill‑climb.
    # --------------------------------------------------------------
    if time.time() + sa_time < deadline:
        print(f"[{time.time():.1f}] Hill‑climb (SA) phase (≈{sa_time:.1f}s)")
        a_sa, sc_sa = hill_climb_sa(a,
                                    time.time() + sa_time,
                                    max_steps=2_000_000)
        if sc_sa < best_score:
            best_score = sc_sa
            a = a_sa
            print(f"[{time.time():.1f}] Hill‑climb improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # 10.  Final evaluation (using the provided `evaluate_sequence`).
    # --------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
```
