"""
Propose a step‑function (non‑negative sequence) that minimizes the
``evaluate_sequence`` objective.

The search strategy combines several ideas that were not used together
in the original implementation:

1️⃣  Scan several analytic families (product, power‑law, exponential‑U,
    mixed, asymmetric, product+U) on a *larger* support size
    ``n = 16384`` (still cheap with FFT).

2️⃣  Refine the most promising families with high‑precision one‑dimensional
    golden‑section and with SciPy’s ``Nelder‑Mead`` for the two‑parameter
    families.  A small random‑parameter sweep is also performed to avoid
    missing narrow minima.

3️⃣  Perform three stages of smooth‑max optimisation (Adam) with
    increasingly aggressive temperature schedules and occasional
    symmetrisation.  The Adam updates work on the log‑soft‑max (entropy)
    parametrisation, guaranteeing feasibility.

4️⃣  Run a projected sub‑gradient descent on the *true* max‑convolution
    objective (the exact sub‑gradient is cheap to compute).

5️⃣  Finish with a simulated‑annealing hill‑climb that makes random
    mass‑transfer moves on the simplex.

All phases respect the global 1000 s wall‑clock budget; the best
sequence found is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for the given vector a (no normalisation)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max()) / (float(a.sum()) ** 2)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ [(i+½)(n‑½‑i)]^{‑α}.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow for huge α
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}  (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def product_expU_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    """
    Product family multiplied by an exponential‑U factor:
        a_i ∝ [(i+½)(n‑½‑i)]^{‑α}·exp(‑γ·dist(i)).
    """
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod - gamma * dist)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def asymmetric_product_family(n: int, beta_left: float, beta_right: float) -> np.ndarray:
    """Two‑parameter product family:
        a_i ∝ (i+½)^{‑β₁}·(n‑½‑i)^{‑β₂}.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta_left * np.log(i + 0.5) - beta_right * np.log(n - 0.5 - i))
    w /= w.sum()
    return w

# Random initialisers (fallback)
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse start: ~5 % non‑zero entries, uniform mass on them."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sort a powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 10.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.015,
                         sym_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    start_a
        Probability vector (sum = 1, non‑negative).
    deadline
        Unix timestamp after which the optimisation must stop.
    max_iters
        Upper bound on the number of Adam iterations.
    beta_start, beta_end
        Annealing schedule for the smooth‑max temperature.
    lr
        Base learning rate.
    sym_interval
        Every ``sym_interval`` steps the iterate is symmetrised.
    """
    n = start_a.shape[0]
    eps = 1e-14
    z = np.log(start_a + eps)                # log‑softmax parameters

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------------- temperature -----------------
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---------------- current iterate -------------
        a = softmax(z)

        # optional symmetrisation
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        # --------------- smooth‑max gradient ----------
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max over the whole convolution (numerically stable)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # weights that approximate the arg‑max

        # gradient of the smoothed max w.r.t. a:
        # ∂/∂a_i  Σ_j w_j (a * a)_j = 2 Σ_j w_j a_{j-i}
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]   # same size as a

        # convert to gradient w.r.t. logits z (entropy mirror map)
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ------------------- Adam update --------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------------------- bookkeeping ---------------
        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *exact* max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.075) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent for the exact max‑convolution objective.
    The step‑size schedule is η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at shift k:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (simple but effective)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a slowly decreasing temperature.
    Moves are either swaps or small mass transfers.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# One‑dimensional golden‑section search (high‑precision refinement)
# ----------------------------------------------------------------------


def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 25) -> tuple[float, float]:
    """Minimise ``func`` on [a,b] using golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a, fd, b = d, fc, d
            c = b - invphi * (b - a)
            fc = func(c)
        else:
            b, fc, a = c, fd, c
            d = a + invphi * (b - a)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Main driver – the function that will be called by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously found construction (if available)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Set working length (a power of two keeps FFT cheap)
    # ------------------------------------------------------------------
    n = 16384   # a larger support gives a finer discretisation

    # ------------------------------------------------------------------
    # 2. Coarse scan of analytic families (still cheap)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse scan (n={n})")

    # Uniform baseline
    vec = np.full(n, 1.0 / n, dtype=np.float64)
    sc = compute_score(vec)
    if sc < best_score:
        best_score = sc
        best_seq = vec.tolist()
        print(f"[{time.time():.1f}] new best (uniform) = {best_score:.6f}")

    # --------------------------------------------------------------
    # Helper to test and possibly update the global best
    # --------------------------------------------------------------
    def maybe_update(v: np.ndarray, label: str):
        nonlocal best_seq, best_score
        sc = compute_score(v)
        if sc < best_score:
            best_score = sc
            best_seq = v.tolist()
            print(f"[{time.time():.1f}] new best ({label}) = {best_score:.6f}")
        return sc

    # ----- symmetric product family (scan α) -----
    best_alpha = None
    best_alpha_sc = float('inf')
    for alpha in np.linspace(0.0, 20.0, 101):         # 0 … 20 step 0.2
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = maybe_update(vec, f"sym‑α={alpha:.3f}")
        if sc < best_alpha_sc:
            best_alpha_sc = sc
            best_alpha = alpha

    # ----- left‑heavy power family (scan β) -----
    best_beta = None
    best_beta_sc = float('inf')
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = maybe_update(vec, f"left‑β={beta:.3f}")
        if sc < best_beta_sc:
            best_beta_sc = sc
            best_beta = beta

    # ----- exponential‑U family (scan γ) -----
    best_gamma = None
    best_gamma_sc = float('inf')
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = maybe_update(vec, f"exp‑γ={gamma:.3f}")
        if sc < best_gamma_sc:
            best_gamma_sc = sc
            best_gamma = gamma

    # ----- mixed uniform + U (λ,γ) -----
    best_mix = None
    best_mix_sc = float('inf')
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = maybe_update(vec, f"mix λ={lam:.2f},γ={gamma:.2f}")
            if sc < best_mix_sc:
                best_mix_sc = sc
                best_mix = (lam, gamma)

    # ----- asymmetric product (β₁,β₂) -----
    best_asym = None
    best_asym_sc = float('inf')
    beta_grid = np.linspace(0.0, 12.0, 13)
    for b1 in beta_grid:
        for b2 in beta_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = asymmetric_product_family(n, b1, b2)
            sc = maybe_update(vec, f"asym β1={b1:.2f},β2={b2:.2f}")
            if sc < best_asym_sc:
                best_asym_sc = sc
                best_asym = (b1, b2)

    # ----- product + exponential‑U (α,γ) -----
    best_pe = None
    best_pe_sc = float('inf')
    a_grid = np.linspace(0.0, 20.0, 21)
    g_grid = np.linspace(0.0, 6.0, 13)
    for a in a_grid:
        for g in g_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = product_expU_family(n, a, g)
            sc = maybe_update(vec, f"prod‑exp α={a:.2f},γ={g:.2f}")
            if sc < best_pe_sc:
                best_pe_sc = sc
                best_pe = (a, g)

    # ----- random sampling of (α,γ) for product+U -----
    for _ in range(300):
        if time.time() >= deadline - 30.0:
            break
        a = np.random.uniform(0.0, 20.0)
        g = np.random.uniform(0.0, 6.0)
        vec = product_expU_family(n, a, g)
        maybe_update(vec, f"rand prod‑exp α={a:.3f},γ={g:.3f}")

    # ------------------------------------------------------------------
    # 3. Local refinement of promising families (SciPy if available)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import minimize, minimize_scalar
        scipy_available = True
    except Exception:
        scipy_available = False

    if scipy_available and time.time() < deadline - 20.0:
        # ---- 1‑D golden‑section refinement for product family ----
        if best_alpha is not None:
            lo = max(0.0, best_alpha - 1.0)
            hi = min(20.0, best_alpha + 1.0)

            def score_alpha(a):
                return compute_score(symmetric_product_family(n, a))

            α_opt, α_sc = golden_section_optimize(score_alpha, lo, hi, max_iter=30)
            vec = symmetric_product_family(n, α_opt)
            maybe_update(vec, f"refined product‑α={α_opt:.6f}")

        # ---- 2‑D Nelder‑Mead for mixture (λ,γ) ----
        if best_mix is not None:
            lam0, gam0 = best_mix

            def score_mix(p):
                lam = np.clip(p[0], 0.0, 1.0)
                gam = np.clip(p[1], 0.0, 6.0)
                return compute_score(mixed_uniform_u_family(n, lam, gam))

            res = minimize(score_mix,
                           x0=np.array([lam0, gam0]),
                           method='Nelder‑Mead',
                           options={'maxiter': 250, 'xatol': 1e-5, 'fatol': 1e-5})
            lam_opt, gam_opt = np.clip(res.x, [0.0, 0.0], [1.0, 6.0])
            vec = mixed_uniform_u_family(n, lam_opt, gam_opt)
            maybe_update(vec, f"refined mix λ={lam_opt:.6f},γ={gam_opt:.6f}")

        # ---- 2‑D Nelder‑Mead for product+expU family ----
        if best_pe is not None:
            a0, g0 = best_pe

            def score_pe(p):
                a = np.clip(p[0], 0.0, 20.0)
                g = np.clip(p[1], 0.0, 6.0)
                return compute_score(product_expU_family(n, a, g))

            res = minimize(score_pe,
                           x0=np.array([a0, g0]),
                           method='Nelder‑Mead',
                           options={'maxiter': 350, 'xatol': 1e-5, 'fatol': 1e-5})
            a_opt, g_opt = np.clip(res.x, [0.0, 0.0], [20.0, 6.0])
            vec = product_expU_family(n, a_opt, g_opt)
            maybe_update(vec, f"refined prod‑exp α={a_opt:.6f},γ={g_opt:.6f}")

        # ---- 2‑D Nelder‑Mead for asymmetric product ----
        if best_asym is not None:
            b1_0, b2_0 = best_asym

            def score_asym(p):
                b1 = np.clip(p[0], 0.0, 12.0)
                b2 = np.clip(p[1], 0.0, 12.0)
                return compute_score(asymmetric_product_family(n, b1, b2))

            res = minimize(score_asym,
                           x0=np.array([b1_0, b2_0]),
                           method='Nelder‑Mead',
                           options={'maxiter': 300, 'xatol': 1e-5, 'fatol': 1e-5})
            b1_opt, b2_opt = np.clip(res.x, [0.0, 0.0], [12.0, 12.0])
            vec = asymmetric_product_family(n, b1_opt, b2_opt)
            maybe_update(vec, f"refined asym β1={b1_opt:.6f},β2={b2_opt:.6f}")

    # ------------------------------------------------------------------
    # 4. First Adam (smooth‑max) refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] first Adam refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 150.0,
            max_iters=800_000,           # more iterations than before
            beta_start=10.0,
            beta_end=10000.0,
            lr=0.015,
            sym_interval=1500,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after first Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 50.0,
            max_iters=1_800_000,
            step0=0.09,                   # a slightly larger initial step
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Second Adam pass (higher β, less symmetrisation)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] second Adam pass")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = adam_smooth_optimize(
            a_start,
            deadline - 30.0,
            max_iters=400_000,
            beta_start=50.0,
            beta_end=15000.0,
            lr=0.012,
            sym_interval=1200,
        )
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.tolist()
            print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Explicit symmetrisation test (often gives a tiny boost)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_500_000,          # a bit fewer steps than earlier
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Fallback to a safe uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # Final evaluation via the official wrapper (includes checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
