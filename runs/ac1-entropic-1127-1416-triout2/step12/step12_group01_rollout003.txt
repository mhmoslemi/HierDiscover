def smooth_eg(a0, deadline, eta0=0.4, gamma_start=0.5, gamma_end=20.0, gamma_mul=1.02, gamma_update=200, tol=1e-12):
    a = a0.copy()
    n = a.shape[0]
    best_val = compute_score(a)
    best_a = a.copy()
    it = 0
    gamma = gamma_start
    while time.time() < deadline:
        it += 1
        # compute convolution
        conv = full_convolution(a)
        # compute softmax weights
        maxc = conv.max()
        # scaling to avoid overflow
        exp_term = np.exp(gamma * (conv - maxc))
        w = exp_term / exp_term.sum()
        # gradient via correlation
        # use np.correlate for efficiency
        grad = 2.0 * np.correlate(a, w, mode='valid')
        # update eta
        eta = eta0 / np.sqrt(it)
        # EG update
        a = a * np.exp(-eta * grad)
        # renorm
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum
        # evaluate score
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # update gamma periodically
        if it % gamma_update == 0 and gamma < gamma_end:
            gamma = min(gamma_end, gamma * gamma_mul)
        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = np.random.randn(n) * 0.001
            a = a * np.exp(jitter)
            a = a / a.sum()
    return best_a
```

def project_simplex(v):
    # returns projection of v onto simplex sum=1, v>=0
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
1. **Quick global sweep** – a lightweight scan of three parametric families
   (power‑law, exponential, beta) plus a few Dirichlet draws.  The top‑k
   candidates (by the exact 2·n·max(conv)) are kept.

2. **Smooth exponentiated‑gradient descent** – minimise a smooth surrogate of
   the max‑convolution:  
      φγ(a)= (1/γ)·log∑k exp(γ·(a∗a)k).  
   The gradient of φγ is
      ∇φγ = 2·corr(a, w)   ,  w_k ∝ exp(γ·(a∗a)_k),
   i.e. a correlation between the current sequence and the soft‑max
   weighting of its self‑convolution.  An EG update (multiplicative
   mirror‑descent on the simplex) is applied with a decaying step‑size.
   A continuation on γ (slowly increasing from a smooth to a sharp regime)
   helps avoid poor local minima.

3. **Multiple random restarts** – as long as time permits a fresh random
   Dirichlet start is polished with the smooth EG routine.  The best vector
   seen over all runs is kept.

4. **Light simulated‑annealing polish** – a short Metropolis walk refines
   the final candidate.

All steps respect the 1000 s wall‑clock budget; the algorithm is
deterministic apart from the random seeds (fixed for reproducibility) and
produces a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def _next_pow2(x):
    """Return the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a):
    """Linear convolution a∗a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution_two(x, y):
    """Linear convolution x∗y using FFT."""
    m = len(x) + len(y) - 1
    n_fft = _next_pow2(m)
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def compute_score(a):
    """Exact objective for a (assumed normalised to sum = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def project_simplex(v):
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    Returns a new array with the same shape as v.
    """
    if v.ndim != 1:
        raise ValueError("project_simplex expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for the quick baseline sweep)
# ----------------------------------------------------------------------


def generate_powerlaw(n, p):
    """a_i ∝ (i+1)^{‑p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n, lam):
    """a_i ∝ exp(‑lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n, alpha, beta):
    """Discrete beta shape."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Quick baseline sweep (keeps a few promising candidates)
# ----------------------------------------------------------------------


def quick_baseline(n, deadline, max_keep=5):
    """
    Very light exploration of three families + random Dirichlet draws.
    Returns a list of up to ``max_keep`` best vectors (numpy arrays).
    """
    best = []   # list of (score, vector)

    def _try(vec):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec /= s
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # power‑law sweep
    for p in np.linspace(0.1, 3.0, 80):
        if time.time() >= deadline:
            break
        _try(generate_powerlaw(n, p))

    # exponential sweep
    for lam in np.linspace(0.0001, 0.08, 100):
        if time.time() >= deadline:
            break
        _try(generate_exponential(n, lam))

    # beta sweep (coarse)
    al_vals = np.linspace(0.2, 8.0, 15)
    be_vals = np.linspace(0.2, 8.0, 15)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            _try(generate_beta(n, alpha, beta))

    # random Dirichlet draws (very few, just to inject diversity)
    concs = [0.2, 0.5, 1.0, 2.0, 5.0]
    for _ in range(200):
        if time.time() >= deadline:
            break
        conc = np.random.choice(concs)
        a = np.random.dirichlet(np.full(n, conc))
        _try(a)

    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Smooth exponentiated‑gradient optimiser
# ----------------------------------------------------------------------


def smooth_eg_optimize(a0, deadline,
                       eta0=0.45,
                       gamma_start=0.5,
                       gamma_end=25.0,
                       gamma_mul=1.02,
                       gamma_update=150):
    """
    Minimise φγ(a) = (1/γ)·log∑ exp(γ·(a∗a)_k) with EG on the simplex.
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    gamma = gamma_start

    while time.time() < deadline:
        it += 1

        # 1) current convolution
        conv = full_convolution(a)                     # length 2·n‑1
        maxc = conv.max()

        # 2) soft‑max weights (numerically stable)
        exp_term = np.exp(gamma * (conv - maxc))
        w = exp_term / exp_term.sum()                 # still length 2·n‑1, sums to 1

        # 3) gradient = 2·corr(a, w)  (valid mode gives length n)
        grad = 2.0 * np.correlate(a, w, mode='valid')

        # 4) EG update with decreasing step size
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # 5) projection back onto the simplex
        a = project_simplex(a)

        # 6) keep the best objective (the true max‑convolution score)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # 7) slowly tighten the soft‑max (anneal γ)
        if it % gamma_update == 0 and gamma < gamma_end:
            gamma = min(gamma_end, gamma * gamma_mul)

        # occasional random jitter to shake off plateaus
        if it % 800 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = project_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing (optional final pass)
# ----------------------------------------------------------------------


def simulated_annealing(a_start, deadline,
                        init_temp=5e-4,
                        final_temp=1e-7,
                        cooling=0.9998,
                        step_scale_init=0.02,
                        step_scale_min=1e-7):
    """Metropolis hill‑climb with a temperature schedule."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1

        # propose moving a tiny amount from i to j
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling)

        # global jitter every few thousand steps
        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = project_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence`` within the allotted wall‑clock time.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # problem size – a modest increase over the previous 2048
    n = 4096

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously known solution
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Quick global baseline (very cheap)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 250.0   # reserve the bulk for polishing
    if time.time() < baseline_deadline:
        cand_vectors = quick_baseline(n, baseline_deadline, max_keep=5)
        for vec in cand_vectors:
            sc = compute_score(vec)
            if sc < best_val:
                best_val, best_seq = sc, vec.tolist()
        print(f"[baseline] best score (raw) = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smooth EG polishing – many restarts while time permits
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        # polish the current best first
        a0 = np.array(best_seq, dtype=float)
        refined = smooth_eg_optimize(a0, polish_deadline)
        val = compute_score(refined)
        print(f"[smooth‑EG‑primary] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, refined.tolist()

        # extra random restarts
        while time.time() < polish_deadline:
            # random Dirichlet start with a randomly chosen concentration
            conc = np.random.choice([0.2, 0.5, 1.0, 2.0, 5.0])
            a_start = np.random.dirichlet(np.full(n, conc))
            refined = smooth_eg_optimize(a_start, polish_deadline)
            val = compute_score(refined)
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"[smooth‑EG‑restart] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Final simulated‑annealing polish (short)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(final)
        print(f"[anneal] final raw score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, final.tolist()

    # -----------------------------------------------------------------
    # 4. Fallback to a uniform distribution if nothing succeeded
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
