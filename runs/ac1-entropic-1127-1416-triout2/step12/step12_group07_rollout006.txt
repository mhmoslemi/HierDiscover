"""
A three‑stage optimizer for the step‑function problem.

1. Parameter sweep – explores simple analytic families and many random
   Dirichlet vectors, keeping the top‐k candidates (lowest 2·n·max(conv)).
2. Smoothed‑max gradient descent – minimises a smooth approximation of
   the max‑convolution using a soft‑max (temperature β).  The gradient is
   obtained analytically via one extra FFT convolution per iteration.
   β is increased geometrically, step sizes are reduced with the iteration
   count and each update is Euclidean‑projected onto the probability simplex.
3. Greedy hill‑climb – final fine‑tuning by moving a small fraction of mass
   between two random positions, accepting only improvements.

All operations respect the global wall‑clock budget (default 1000 s) and
the returned list consists of non‑negative numbers that sum to one
(scale‑invariant for ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:n]
    return conv


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)−1) via FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` for Σa = 1:
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

# Candidate families
# ----------------------------------------------------------------------
def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, beta: float) -> np.ndarray:
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    a /= a.sum()
    return a


def cosine_power(n: int, expo: float) -> np.ndarray:
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    blocks = rng.dirichlet(np.ones(K))
    blocks = np.sort(blocks)[::-1]                     # decreasing blocks
    repeats = np.full(K, n // K, dtype=int)
    remainder = n - repeats.sum()
    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)[:n]
    a /= a.sum()
    return a


def random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Sample a Dirichlet(α,…,α) vector (α can be <1 for sparser draws)."""
    a = rng.gamma(shape=alpha, scale=1.0, size=n)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Global parametric + stochastic sweep
# ----------------------------------------------------------------------
def sweep_candidates(n: int, deadline: float, top_k: int = 12) -> list[tuple[float, np.ndarray]]:
    """Explore analytic families and many random Dirichlet vectors.
    Returns the best (score, vector) pairs (sorted ascending)."""
    best = []  # list of (score, vector)

    def consider(vec: np.ndarray):
        if vec is None or np.isnan(vec).any():
            return
        sc = compute_score(vec)
        if len(best) < top_k:
            best.append((sc, vec))
            best.sort(key=lambda x: x[0])
        else:
            if sc < best[-1][0]:
                best[-1] = (sc, vec)
                best.sort(key=lambda x: x[0])

    rng = np.random.default_rng(987654321)

    # ----- power‑law with offset -----
    for p in np.linspace(0.35, 1.20, 44):
        if time.time() >= deadline:
            break
        for off in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]:
            if off >= n:
                continue
            consider(powerlaw_offset(n, p, off))

    # ----- geometric decay -----
    for r in np.linspace(0.55, 0.995, 88):
        if time.time() >= deadline:
            break
        consider(geometric(n, r))

    # ----- beta‑shape -----
    for beta in np.linspace(0.5, 20.0, 38):
        if time.time() >= deadline:
            break
        consider(beta_shape(n, beta))

    # ----- cosine‑power -----
    for e in np.linspace(0.3, 6.0, 30):
        if time.time() >= deadline:
            break
        consider(cosine_power(n, e))

    # ----- plateau + power‑law (smooth) -----
    for Lfrac in np.linspace(0.05, 0.30, 6):
        if time.time() >= deadline:
            break
        L = max(1, int(Lfrac * n))
        for p in np.linspace(0.4, 0.9, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head_val = tail[0] if tail.size else 1.0
            head = np.full(L, head_val)
            vec = np.concatenate([head, tail])
            vec /= vec.sum()
            consider(vec)

    # ----- random block shapes -----
    K = 30
    for _ in range(2000):
        if time.time() >= deadline:
            break
        consider(block_shape(n, K, rng))

    # ----- random Dirichlet vectors (various concentrations) -----
    for alpha in [0.3, 0.8, 1.0, 2.0, 5.0]:
        for _ in range(500):
            if time.time() >= deadline:
                break
            consider(random_dirichlet(n, rng, alpha=alpha))

    return best


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent
# ----------------------------------------------------------------------
def smooth_max_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected gradient descent on a soft‑max approximation of max(conv).
    The temperature β is increased geometrically; step size decays with
    √iteration count.  Returns the best vector seen."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    # schedule: (β, max_iters_at_this_β)
    schedule = [(5, 2000), (10, 2000), (20, 2000), (40, 2000),
                (80, 2000), (160, 2000), (320, 2000), (640, 2000)]

    base_lr = 0.15  # base learning‑rate factor
    iter_count = 0

    for beta, max_it in schedule:
        for _ in range(max_it):
            if time.time() >= deadline:
                return best_a
            # convolution a*a
            conv = full_convolution_fft(a)
            max_val = conv.max()
            # soft‑max weights (stable)
            exp_vals = np.exp(beta * (conv - max_val))
            w = exp_vals / exp_vals.sum()
            # gradient of the smoothed objective
            grad = 2.0 * fft_convolve(w, a[::-1])[n - 1:n - 1 + n]
            # step size inversely proportional to gradient norm
            grad_norm = np.linalg.norm(grad)
            if grad_norm == 0.0:
                step = 0.0
            else:
                step = base_lr / (np.sqrt(iter_count + 1) * grad_norm)
            # gradient step + projection
            a = a - step * grad
            a = project_to_simplex(a)

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc, best_a = cur_sc, a.copy()
            iter_count += 1

    return best_a


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float, max_iter: int = 600_000) -> np.ndarray:
    """Tiny random mass‑transfer moves with greedy acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        if no_improve > 120_000:
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1

    return best_a

    # -----------------------------------------------------------------

        if best_sc < best_val:

    # -----------------------------------------------------------------
    # 2. Smoothed‑max gradient descent (main optimisation)
    # -----------------------------------------------------------------
    descent_deadline = deadline - 150.0

    # 3. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0

        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)

    # -----------------------------------------------------------------

        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Official evaluation (includes security checks)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported (official) score = {final_score:.6f}")
    return best_seq
