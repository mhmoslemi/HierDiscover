    for _ in range(100):

4.   Finish with a projected sub‑gradient descent on the true max‑objective
     (the exact ``max(conv)``) for a few hundred thousand steps.
5.   Return the best sequence found (as a plain list of floats).  The official
     ``evaluate_sequence`` wrapper is called once at the end to obtain the
     reported score.

All helper functions are top‑level, no closures or lambda expressions are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two that is ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used inside the optimiser:
        2 * n * max(a * a)   (a is assumed to sum to 1)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of a vector onto the probability simplex. """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """ Numerically stable soft‑max (returns a probability vector). """
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    Beta‑type “U” family:
        a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    log_w = -alpha * logprod
    log_w -= np.max(log_w)               # stabilise the exponentiation
    w = np.exp(log_w)
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """ Monotone decreasing power law: a_i ∝ (i+½)^{‑β}. """
    i = np.arange(n, dtype=np.float64)
    log_w = -beta * np.log(i + 0.5)
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """ Pure exponential U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i)). """
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """ Convex mixture of uniform and the exponential‑U shape. """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def asymmetric_product_family(n: int, beta_left: float, beta_right: float) -> np.ndarray:
    """
    Two‑parameter Beta family:
        a_i ∝ (i+½)^{‑β_left}·(n‑½‑i)^{‑β_right}.
    """
    i = np.arange(n, dtype=np.float64)
    log_w = -beta_left * np.log(i + 0.5) - beta_right * np.log(n - 0.5 - i)
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w /= w.sum()
    return w


def product_expU_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    """ Product family multiplied by an exponential‑U factor. """
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    log_w = -alpha * logprod - gamma * dist
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Random initialisers (fallback)
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse start: ~5 % non‑zero entries, uniform mass on them."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sort a powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 10.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.015,
                         sym_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate of the max‑convolution.
    The vector is kept on the probability simplex via a soft‑max parametrisation.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)                 # initialise logits

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric annealing)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights over the convolution entries
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth‑max wrt a (see the paper for derivation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # chain rule through the soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam updates on the logits
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improvement to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent for the exact max‑convolution objective.
    The iterate is always projected onto the simplex.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improvement to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to keep the top‑K candidates without using lambdas
# ----------------------------------------------------------------------


def insert_top(top_list, score, vec, max_len=3):
    """
    Insert (score,vec) into a length‑limited sorted list (ascending by score).
    """
    inserted = False
    for i, (s, _) in enumerate(top_list):
        if score < s:
            top_list.insert(i, (score, vec))
            inserted = True
            break
    if not inserted:
        top_list.append((score, vec))
    if len(top_list) > max_len:
        top_list.pop()


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (plain Python list) found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")
    top_three = []          # list of (score, vector) – kept sorted

    # ------------------------------------------------------------------
    # 0. basic uniform baseline
    # ------------------------------------------------------------------
    n = 8192                # power of two – fast FFTs
    uniform_vec = np.full(n, 1.0 / n, dtype=np.float64)
    sc = compute_score(uniform_vec)
    insert_top(top_three, sc, uniform_vec)
    if sc < best_score:
        best_score = sc
        best_seq = uniform_vec.tolist()
        print(f"[{time.time():.1f}] baseline uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. systematic scans of analytic families (extended ranges)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] scanning analytic families (n={n})")

    # product / Beta family – α ∈ [0,30]
    for alpha in np.linspace(0.0, 30.0, 61):
        if time.time() >= deadline - 5.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        insert_top(top_three, sc, vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (prod α={alpha:.2f}) = {best_score:.6f}")

    # exponential U family – γ ∈ [0,12]
    for gamma in np.linspace(0.0, 12.0, 31):
        if time.time() >= deadline - 5.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        insert_top(top_three, sc, vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (expU γ={gamma:.2f}) = {best_score:.6f}")

    # product + exponential‑U – α ∈ [0,30], γ ∈ [0,12] (coarse grid)
    for a in np.linspace(0.0, 30.0, 13):
        for g in np.linspace(0.0, 12.0, 7):
            if time.time() >= deadline - 5.0:
                break
            vec = product_expU_family(n, a, g)
            sc = compute_score(vec)
            insert_top(top_three, sc, vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (prod‑exp α={a:.2f},γ={g:.2f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. random sampling of asymmetric Beta shapes (adds diversity)
    # ------------------------------------------------------------------
    for _ in range(200):
        if time.time() >= deadline - 5.0:
            break
        beta_l = np.random.uniform(0.0, 30.0)
        beta_r = np.random.uniform(0.0, 30.0)
        vec = asymmetric_product_family(n, beta_l, beta_r)
        sc = compute_score(vec)
        insert_top(top_three, sc, vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (asym βL={beta_l:.2f},βR={beta_r:.2f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smooth‑max Adam refinement on the top candidates
    # ------------------------------------------------------------------
    for _, vec in top_three:
        if time.time() >= deadline - 15.0:
            break
        # allocate roughly half of the remaining budget to Adam
        remaining = deadline - time.time()
        # empirical cost ≈ 0.004 s per iteration (two FFTs)
        max_iters = max(5000, int(0.4 * remaining / 0.004))
        print(f"[{time.time():.1f}] Adam start (max_iters={max_iters})")
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=vec,
            deadline=deadline - 5.0,
            max_iters=max_iters,
            beta_start=5.0,
            beta_end=20000.0,
            lr=0.02,
            sym_interval=1000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final exact sub‑gradient descent (uses the true max‑objective)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_iters = int(0.5 * remaining / 0.003)  # ~0.003 s per iteration
        max_iters = max(2000, max_iters)
        print(f"[{time.time():.1f}] final sub‑gradient descent (iters={max_iters})")
        a_sub, sc_sub = subgrad_optimize(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline - 1.0,
            max_iters=max_iters,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] sub‑gradient improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. safety net – if nothing succeeded fall back to uniform
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = compute_score(np.array(best_seq, dtype=np.float64))

    # final evaluation through the official wrapper (includes security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
