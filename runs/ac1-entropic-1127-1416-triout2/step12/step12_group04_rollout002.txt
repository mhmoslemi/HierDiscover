"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

using a multi‑phase search that differs from the previous baseline:

1. Wide analytic scan (including a Kaiser window, which was missing before).
2. Smooth‑max gradient descent (exponentiated‑gradient on a β‑softened max).
   The β‑parameter is gradually increased, allowing the optimizer to first
   move globally and then focus on the true peak.
3. A few LP‑based refinement steps (as in the state‑of‑the‑art mixing).
4. Final symmetric‑isation and a short stochastic hill‑climb.

All phases respect the global 1000 s wall‑clock budget and keep every
candidate on the probability simplex (∑a = 1, a_i ≥ 0).  The returned
list is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.special import i0

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution via real FFT.
    Both a and b are real 1‑D arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped (ends heavy) family: w_i ∝ ((i+0.5)*(n‑0.5‑i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone, heavy left side: w_i ∝ (i+0.5)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential: w_i ∝ exp(-γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian: w_i ∝ exp(-γ·(i‑mid)²)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, window_type: str) -> np.ndarray:
    """Standard cosine windows (Hann, Hamming, Blackman)."""
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – useful for low side‑lobes."""
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    r = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1 - r * r)) / i0(beta)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (heavy left side)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped (product‑like) start."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


def random_sparse_start(n: int, k: int = 200) -> np.ndarray:
    """Sparse vector with k random non‑zero entries."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(len(idx))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    if a.sum() == 0:
        a[0] = 1.0
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth‑max gradient (exponentiated‑gradient) optimisation
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the β‑softened max objective:
        f(a) = (2·n/β)·log(∑_k exp(β·conv_k(a,a)))
    Returns ∇f (shape n).  The factor 2·n is already included.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)               # length 2n‑1
    # stable soft‑max on β·conv
    z = beta * conv
    z_max = np.max(z)
    expz = np.exp(z - z_max)
    sum_exp = expz.sum()
    w = expz / sum_exp                         # weights, sum to 1

    # first term: (a_rev * w)[n‑1 : n‑1+n]
    conv_rev = convolution_fft(a[::-1], w)    # length 3n‑2
    grad1 = conv_rev[n - 1:n - 1 + n]

    # diagonal term: w[2·i]
    diag = w[2 * np.arange(n)]

    grad = 2.0 * n * (grad1 + a * diag)
    return grad


def smooth_expgd(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
    beta0: float = 5.0,
    beta_max: float = 200.0,
    beta_step: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth max objective.
    β is increased exponentially every `beta_step` iterations.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    beta = beta0
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        g = smoothed_gradient(a, beta)
        eta = eta0 / math.sqrt(t)

        # multiplicative update, keep numerical stability
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / s

        # occasional evaluation
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] smooth‑EG improved to {best_score:.6f}")

        # increase β
        if beta_step and t % beta_step == 0 and beta < beta_max:
            beta = min(beta_max, beta * 2.0)

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (the “state‑of‑the‑art” iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0, then return the convex mixture
        f_new = (1−t)·f + t·g_norm,
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A such that A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)      # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)

    return f_new


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The function respects the global time budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose problem size
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # Prefer a reasonably large n but keep FFT cheap
    n = fallback_len if fallback_len >= 1024 else 4096
    print(f"[{time.time() - start_time:.1f}] using length n={n}")

    # ------------------------------------------------------------------
    # Helper: test a candidate (normalised internally)
    # ------------------------------------------------------------------
    def try_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Analytic scan (families + random starts)
    # ------------------------------------------------------------------
    # uniform baseline
    try_candidate(np.full(n, 1.0 / n, dtype=np.float64))

    # Product family – wide sweep
    alphas = np.concatenate([np.linspace(0.0, 5.0, 41), np.linspace(5.1, 150.0, 30)])
    for alpha in alphas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(product_family(n, alpha))

    # Left‑heavy power family
    betas = np.linspace(0.0, 30.0, 31)
    for beta in betas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(left_heavy_power_family(n, beta))

    # Exponential‑U (distance) family
    gammas = np.linspace(0.0, 20.0, 21)
    for gamma in gammas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(exp_u_family(n, gamma))

    # Gaussian family
    gauss_gammas = np.logspace(-4, 2, 30)
    for gamma in gauss_gammas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(gaussian_family(n, gamma))

    # Cosine windows
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(window_family(n, win))

    # Kaiser windows – new family not explored previously
    for beta in np.linspace(0.5, 12.0, 12):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(kaiser_window(n, beta))

    # Random monotone / U‑shape starts
    for _ in range(200):
        if time.time() >= deadline - 250.0:
            break
        if np.random.rand() < 0.5:
            try_candidate(random_monotone_start(n))
        else:
            try_candidate(random_u_start(n))

    # Random sparse starts
    for k in (50, 100, 200):
        for _ in range(30):
            if time.time() >= deadline - 250.0:
                break
            try_candidate(random_sparse_start(n, k))

    # ------------------------------------------------------------------
    # 3. Normalise the best candidate (fallback to uniform)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.asarray(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 4. Smooth‑max exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print(f"[{time.time() - start_time:.1f}] starting smooth‑EG optimisation")
        a_opt, sc_opt = smooth_expgd(
            a,
            deadline - 180.0,
            max_iters=500_000,
            eta0=0.8,
            beta0=5.0,
            beta_max=200.0,
            beta_step=8000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] after smooth‑EG best = {best_score:.6f}")

        # One perturb‑restart to escape possible shallow basins
        if time.time() < deadline - 180.0:
            pert = a * np.exp(0.02 * np.random.randn(n))
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            a_opt2, sc_opt2 = smooth_expgd(
                pert,
                deadline - 120.0,
                max_iters=300_000,
                eta0=0.8,
                beta0=5.0,
                beta_max=200.0,
                beta_step=8000,
            )
            if sc_opt2 < best_score:
                best_score = sc_opt2
                a = a_opt2
                print(f"[{time.time() - start_time:.1f}] after perturbed smooth‑EG best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. LP‑based refinement loop (multiple mixes)
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print(f"[{time.time() - start_time:.1f}] starting LP refinement loop")
        mix_weight = 0.2
        it = 0
        while time.time() < deadline - 20.0 and it < 30:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc = compute_score(a_new)
            if sc < best_score - 1e-7:
                best_score = sc
                a = a_new
                print(f"[{time.time() - start_time:.1f}] LP step {it} improved to {best_score:.6f}")
            else:
                mix_weight = max(0.05, mix_weight * 0.85)
            it += 1

    # ------------------------------------------------------------------
    # 6. Symmetrisation test (the problem is symmetric)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time() - start_time:.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
