def full_convolution_fft(a, b):
    m = len(a) + len(b) -1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

"""
search for a non‑negative step‑function that minimises the evaluation
function 2·n·max(conv(a,a)) (the sum of the coefficients is 1).
The algorithm consists of three stages:

1. **Broad parametric sweep** – a few analytical families are sampled,
   in particular the *discrete arcsine* law whose self‑convolution is
   (asymptotically) uniform.  All candidates are normalised to a simplex
   and the best few (by the exact objective) are retained.

2. **Smooth max‑gradient descent** – we minimise a differentiable
   surrogate of the max using a soft‑max (log‑sum‑exp) with a
   temperature that is raised from 1 to ~40 (homotopy).  Gradients are
   computed by FFT‑based convolutions and an Adam optimiser with a
   simplex projection is applied.  The best candidate from step 1 is used
   as the starting point.

3. **Greedy hill‑climb** – a final stochastic refinement that moves a
   small fraction of mass between two random positions and keeps only
   improvements.

The whole search respects a 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * b using a real FFT.
    Length = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((m - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a normalised vector
    (Σa == 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant; we therefore keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families of candidates
# ----------------------------------------------------------------------
def powerlaw_offset(n: int, p: float, offset: float) -> np.ndarray:
    """a_i = (i+offset)^(-p)   (offset>0 to avoid division by zero)."""
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a /= a.sum()
    return a


def geometric(n: int, r: float) -> np.ndarray:
    """a_i = r^i  (geometric decay)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def beta_shape(n: int, beta: float) -> np.ndarray:
    """a_i = (1 - i/(n-1))^beta   (monotonically decreasing)."""
    if beta < 0:
        return None
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** beta
    a /= a.sum()
    return a


def cosine_power(n: int, expo: float) -> np.ndarray:
    """a_i = cos(π i / (2 n))^expo  (decreasing cosine shape)."""
    theta = np.pi * np.arange(n, dtype=np.float64) / (2.0 * n)
    a = np.cos(theta) ** expo
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def symmetric_powerlaw(n: int, p: float, eps: float = 0.5) -> np.ndarray:
    """a_i ∝ ((i+eps)*(n-1-i+eps))**(-p), symmetric around the centre."""
    i = np.arange(n, dtype=np.float64)
    d = (i + eps) * (n - 1 - i + eps)
    a = d ** (-p)
    a /= a.sum()
    return a


def arcsine_shape(n: int, eps: float = 0.5) -> np.ndarray:
    """
    Discrete arcsine law: a_i ∝ 1/√((i+eps)*(n-1-i+eps)).
    Its self‑convolution is (asymptotically) uniform, giving a very low
    max‑convolution value.
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a /= a.sum()
    return a


def block_shape(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
    """
    Coarse piecewise‑constant shape: generate K positive values,
    sort them decreasingly, repeat each value to fill the n entries.
    Normalise to sum = 1.
    """
    blocks = rng.dirichlet(np.ones(K))
    blocks = np.sort(blocks)[::-1]                     # decreasing
    repeats = np.full(K, n // K, dtype=int)
    remainder = n - repeats.sum()
    repeats[:remainder] += 1
    a = np.repeat(blocks, repeats)[:n]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Global parametric sweep (keeps top‑K candidates)
# ----------------------------------------------------------------------
def sweep_candidates(n: int, deadline: float, top_k: int = 12) -> list[tuple[float, np.ndarray]]:
    """
    Explore a collection of analytic families and a set of random block
    shapes; return the best ``top_k`` candidates sorted by the exact
    objective (ascending).
    """
    best = []   # list of (score, vector)

    def consider(vec: np.ndarray):
        if vec is None or np.isnan(vec).any():
            return
        sc = compute_score(vec)
        if len(best) < top_k:
            best.append((sc, vec))
            best.sort(key=lambda x: x[0])
        else:
            if sc < best[-1][0]:
                best[-1] = (sc, vec)
                best.sort(key=lambda x: x[0])

    rng = np.random.default_rng(123456)

    # -------- power‑law with offset --------
    for p in np.linspace(0.35, 1.40, 45):
        if time.time() >= deadline:
            break
        for off in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]:
            if off >= n:
                continue
            vec = powerlaw_offset(n, p, off)
            consider(vec)

    # -------- geometric decay --------
    for r in np.linspace(0.55, 0.9995, 90):
        if time.time() >= deadline:
            break
        vec = geometric(n, r)
        consider(vec)

    # -------- beta‑shape --------
    for beta in np.linspace(0.5, 20.0, 40):
        if time.time() >= deadline:
            break
        vec = beta_shape(n, beta)
        consider(vec)

    # -------- cosine‑power --------
    for e in np.linspace(0.3, 7.0, 35):
        if time.time() >= deadline:
            break
        vec = cosine_power(n, e)
        consider(vec)

    # -------- symmetric power‑law --------
    for p in np.linspace(0.2, 1.0, 17):
        if time.time() >= deadline:
            break
        vec = symmetric_powerlaw(n, p, eps=0.5)
        consider(vec)

    # -------- discrete arcsine (theoretically strong) --------
    if time.time() < deadline:
        vec = arcsine_shape(n, eps=0.5)
        consider(vec)

    # -------- random block shapes (coarse search) --------
    K = max(10, n // 256)       # about 16‑20 blocks for n=4096
    trials = 2500
    for _ in range(trials):
        if time.time() >= deadline:
            break
        vec = block_shape(n, K, rng)
        consider(vec)

    return best


# ----------------------------------------------------------------------
# Smooth surrogate max‑gradient descent (Adam + simplex projection)
# ----------------------------------------------------------------------
def smooth_gradient_descent(a0: np.ndarray,
                            deadline: float,
                            lr: float = 0.05,
                            lam_start: float = 1.0,
                            lam_end: float = 40.0) -> np.ndarray:
    """
    Minimises a smooth approximation of   max(conv(a,a))
    using the log‑sum‑exp surrogate.
    ``a0`` is assumed to lie on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    t = 0
    start_opt = time.time()
    total_opt = deadline - start_opt

    while time.time() < deadline:
        t += 1
        # 1) current convolution a*a
        conv = full_convolution_fft(a, a)            # length 2n‑1
        max_val = conv.max()

        # 2) temperature schedule (homotopy)
        elapsed = time.time() - start_opt
        fraction = min(elapsed / total_opt, 1.0)
        lam = lam_start * (lam_end / lam_start) ** fraction

        # 3) soft‑max weights w_i ∝ exp(lam * (conv_i - max_val))
        #    subtraction of max_val keeps numbers stable
        exp_scaled = np.exp(lam * (conv - max_val))
        w = exp_scaled / exp_scaled.sum()            # ∑ w_i = 1

        # 4) gradient of the surrogate = 2 * (w * a) convolution
        grad = 2.0 * full_convolution_fft(w, a)[:n]

        # 5) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # 6) move and project back onto the simplex
        a = project_to_simplex(a - step)

        # 7) keep the best point seen so far
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc, best_a = cur_sc, a.copy()

        # occasional random jitter to help escape flat plateaus
        if t % 500 == 0:
            jitter = np.random.randn(n) * (0.003 / np.sqrt(t + 1))
            a = project_to_simplex(a + jitter)

    return best_a


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (fine polishing)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float, max_iter: int = 600_000) -> np.ndarray:
    """Tiny random mass‑transfer moves, greedy acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]               # move a fraction of mass from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        if no_improve > 100_000:
            jitter = np.random.randn(n) * 0.001
            a = project_to_simplex(best_a + jitter)
            no_improve = 0

        it += 1

    return best_a

        a0 = np.array(best_seq, dtype=float)
        a_opt = smooth_gradient_descent(a0, grad_deadline,
                                        lr=0.05, lam_start=1.0, lam_end=40.0)
        val = compute_score(a_opt)
        print(f"[smooth‑grad] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb (fine‑tuning)
    # -----------------------------------------------------------------
    hill_deadline = deadline - 3.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final raw score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 4. Safeguard – fallback to a uniform vector if everything failed
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5. Official evaluation (includes security wrappers)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported (official) score = {final_score:.6f}")
    return best_seq
