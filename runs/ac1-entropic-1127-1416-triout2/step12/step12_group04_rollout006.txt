def propose_candidate(seed=12345, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine length n (choose power of two maybe 4096)
    n = 4096
    print(f"Using vector length n={n}")

    best_vec = None
    best_score = float('inf')

    # Helper to evaluate and update best
    def update_best(vec):
        nonlocal best_vec, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            return
        a /= s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
            print(f"[{time.time()-start:.1f}s] new best: {sc:.6f}")

    # 0. Fallback from global var height_sequence_1 if available
    try:
        fallback_vec = [float(x) for x in height_sequence_1]  # type: ignore
        update_best(fallback_vec)
    except Exception:
        pass

    # 1. Uniform baseline
    update_best(np.full(n, 1.0/n))

    # 2. Product family (dense scan)
    alphas = np.linspace(0.0, 6.0, 121)  # step ~0.05
    for alpha in alphas:
        if time.time() >= deadline - 30.0:
            break
        w = product_family(n, alpha)
        update_best(w)

    # 3. Mixed uniform+product
    lambdas = np.linspace(0.0, 0.4, 9)  # 0 to 0.4
    for lam in lambdas:
        for alpha in alphas:
            if time.time() >= deadline - 30.0:
                break
            prod = product_family(n, alpha)
            mix = lam * np.full(n, 1.0/n) + (1.0 - lam) * prod
            # Already sums to 1 because both sum to 1
            update_best(mix)

    # 4. Left heavy power family
    betas = np.linspace(0.0, 10.0, 81)
    for beta in betas:
        if time.time() >= deadline - 30.0:
            break
        w = left_heavy_power_family(n, beta)
        update_best(w)

    # 5. Exponential U family
    gammas = np.linspace(0.0, 10.0, 81)
    for gamma in gammas:
        if time.time() >= deadline - 30.0:
            break
        w = exp_u_family(n, gamma)
        update_best(w)

    # 6. Gaussian family
    gauss_gammas = np.logspace(-4, 2, 80)
    for gamma in gauss_gammas:
        if time.time() >= deadline - 30.0:
            break
        w = gaussian_family(n, gamma)
        update_best(w)

    # 7. Random starts
    for _ in range(200):
        if time.time() >= deadline - 30.0:
            break
        if np.random.rand() < 0.5:
            w = random_monotone_start(n)
        else:
            w = random_u_start(n)
        update_best(w)

    # 8. EG optimization on best vector
    if best_vec is not None and time.time() < deadline - 10.0:
        print(f"Starting EG from best score={best_score:.6f}")
        a_opt, sc_opt = eg_optimize(best_vec, deadline, max_iters=300_000, eta0=0.7)
        if sc_opt < best_score:
            best_score = sc_opt
            best_vec = a_opt
            print(f"After EG best={best_score:.6f}")

    # 9. Hill climb SA if time left
    if best_vec is not None and time.time() < deadline - 1.0:
        print("Starting final SA hill climb")
        a_opt, sc_opt = hill_climb_sa(best_vec, deadline, max_steps=1_500_000)
        if sc_opt < best_score:
            best_score = sc_opt
            best_vec = a_opt
            print(f"After SA best={best_score:.6f}")

    # 10. Return result as list float
    final_seq = best_vec.tolist() if best_vec is not None else [1.0]
    final_score = evaluate_sequence(final_seq)
    print(f"Finished – final report score={final_score:.6f}")
    return final_seq

"""
Search for a non‑negative step‑function that minimises the objective

        2·n·max(convolve(a, a)) / (sum(a)²)

implemented as the exact score 2·n·max(conv(a,a)) for a normalised
probability vector (∑a = 1).

Algorithm
---------
1.  Choose a single length n (power of two, 4096 by default).  
    Enumerate a wide variety of analytic families (product‑Beta,
    left‑heavy power, exponential‑U, Gaussian) and simple mixtures of
    each with the uniform distribution.  Every candidate is normalised
    and evaluated with an FFT‑based exact score; the best candidate is
    kept.

2.  Starting from that best candidate, run exponentiated‑gradient (mirror
    descent) on the true max‑objective until the global time budget is
    exhausted.  The sub‑gradient for the active convolution entry is
    computed analytically and the iterates stay on the simplex via the
    usual multiplicative update.

3.  If time permits, a short stochastic hill‑climb (simulated‑annealing
    style) refines the result.

4.  The final vector is returned as a plain Python list ready for
    ``evaluate_sequence``.  All helper functions are defined at the
    module level – no closures are used.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # kept for possible extensions; not required here


# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (parameterised shapes)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta‑type family: w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone heavy‑left family: w_i ∝ (i+0.5)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential decay from the centre: w_i ∝ e^{-γ·dist(i,centre)}."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)        # distance to the nearer end
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian: w_i ∝ e^{-γ·(i‑mid)²}."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (heavy left side)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped start (exponential‑U family with random γ)."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the true max‑objective
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 2_000_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) for the objective
    f(a) = 2·n·max(conv(a,a)).  Returns the best vector encountered.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]          # off‑diagonal part
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]                # diagonal contribution has weight 1

        # multiplicative (exponentiated) update
        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional bookkeeping
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    # final check
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (simple stochastic refinement)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The function respects the global time budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0        # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size (power of two, reasonably large but still fast)
    # ------------------------------------------------------------------
    n = 4096
    print(f"[{time.time() - start_time:.1f}s] Using vector length n={n}")

    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper: evaluate a raw (unnormalised) vector and keep the best
    # ------------------------------------------------------------------
    def try_vec(vec: np.ndarray):
        nonlocal best_vec, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
            print(f"[{time.time() - start_time:.1f}s] New best score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        try_vec(np.array(fb_vec, dtype=np.float64))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Uniform baseline
    # ------------------------------------------------------------------
    try_vec(np.full(n, 1.0 / n, dtype=np.float64))

    # ------------------------------------------------------------------
    # 2. Product‑Beta family (dense sweep)
    # ------------------------------------------------------------------
    alphas = np.linspace(0.0, 6.0, 121)          # step ≈0.05
    for alpha in alphas:
        if time.time() >= deadline - 30.0:
            break
        try_vec(product_family(n, alpha))

    # ------------------------------------------------------------------
    # 3. Mixture of product family with uniform (adds a smoothing component)
    # ------------------------------------------------------------------
    lambdas = np.linspace(0.0, 0.4, 9)           # 0,0.05,…,0.4
    for lam in lambdas:
        for alpha in alphas:
            if time.time() >= deadline - 30.0:
                break
            prod = product_family(n, alpha)
            mix = lam * np.full(n, 1.0 / n) + (1.0 - lam) * prod
            # Both components already sum to 1, so mix also sums to 1
            try_vec(mix)

    # ------------------------------------------------------------------
    # 4. Left‑heavy power family
    # ------------------------------------------------------------------
    betas = np.linspace(0.0, 10.0, 81)
    for beta in betas:
        if time.time() >= deadline - 30.0:
            break
        try_vec(left_heavy_power_family(n, beta))

    # ------------------------------------------------------------------
    # 5. Exponential‑U family
    # ------------------------------------------------------------------
    gammas = np.linspace(0.0, 10.0, 81)
    for gamma in gammas:
        if time.time() >= deadline - 30.0:
            break
        try_vec(exp_u_family(n, gamma))

    # ------------------------------------------------------------------
    # 6. Gaussian family
    # ------------------------------------------------------------------
    gauss_gammas = np.logspace(-4, 2, 80)
    for gamma in gauss_gammas:
        if time.time() >= deadline - 30.0:
            break
        try_vec(gaussian_family(n, gamma))

    # ------------------------------------------------------------------
    # 7. Random starts (both monotone and U‑shaped)
    # ------------------------------------------------------------------
    for _ in range(200):
        if time.time() >= deadline - 30.0:
            break
        if np.random.rand() < 0.5:
            try_vec(random_monotone_start(n))
        else:
            try_vec(random_u_start(n))

    # ------------------------------------------------------------------
    # 8. Mirror‑descent (exponentiated‑gradient) refinement
    # ------------------------------------------------------------------
    if best_vec is not None and time.time() < deadline - 10.0:

        if sc_opt < best_score:

            best_vec = a_opt

        # One more short perturbed run to try escaping tiny basins

            a_opt2, sc_opt2 = eg_optimize(perturb, deadline,

            if sc_opt2 < best_score:

                best_vec = a_opt2
                print(f"[{time.time() - start_time:.1f}s] Perturbed EG improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Symmetrisation (the convolution objective is symmetric)
    # ------------------------------------------------------------------
    if best_vec is not None:
        a_sym = (best_vec + best_vec[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_vec = a_sym
            print(f"[{time.time() - start_time:.1f}s] Symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Final stochastic hill‑climb (if time permits)

            print(f"[{time.time() - start_time:.1f}s] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 11. Return final sequence as a plain Python list
    # ------------------------------------------------------------------
    if best_vec is None:
        # fallback – never should happen
        final_seq = [1.0]
    else:
        final_seq = best_vec.tolist()

    # One final call to the provided evaluation function (its result is printed

    final_eval = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}s] Finished – final reported score = {final_eval:.6f}")
    return final_seq
