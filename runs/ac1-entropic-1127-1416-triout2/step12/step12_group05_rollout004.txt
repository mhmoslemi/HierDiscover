propose_candidate
-----------------
Searches for a non‑negative step‑function (probability vector) that

    2·n·max(conv(a,a)) / (sum(a)**2)

The search proceeds in several stages within the given wall‑clock budget:

   (product‑power, left‑heavy power, exponential‑U, uniform+U mixture,
   and the 2‑parameter Beta family).  Random right‑shifts are also tested.

6. **Greedy hill‑climb** that transfers small amounts of mass between
   entries and accepts any improvement.

All stages respect a common 1000 s wall‑clock limit; the best sequence

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution via real FFT (fast, O(N log N)).
    Both inputs are 1‑D real arrays.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)**2).
    a is assumed non‑negative (the optimiser keeps it on the simplex).
    """
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    # Implementation of the algorithm from
    # Wang & Carreira‑Perpiñán, “Projection onto the probability simplex”.
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Beta‑type density on the discrete grid.
    p,q>0.  When p,q<1 the shape is U‑shaped; p,q>1 gives a bell.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    # Guard against overflow/underflow
    w = np.where(np.isfinite(w), w, 0.0)
    return normalize(w)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 5.0,
                         beta_end: float = 20_000.0,
                         lr: float = 0.012,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    The temperature ``beta`` grows geometrically from ``beta_start`` to
    ``beta_end``.  No forced symmetry unless ``sym_interval``>0.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # Initialise via a log‑softmax to guarantee positivity
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z)
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a = a / a.sum()

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t a (see derivation in the
        # original script).  We convolve a reversed with w.
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 3000 == 0:
            a_tmp = np.exp(z)
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = np.exp(z)
    a_fin = a_fin / a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_greedy(start_a: np.ndarray,
                      deadline: float,
                      max_steps: int = 10_000_000,
                      init_scale: float = 0.15) -> Tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb that moves a small fraction of mass from the
    largest entry to a random other entry whenever this reduces the
    objective.  ``init_scale`` controls the initial transfer fraction.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    scale = init_scale

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # pick donor (large) and receiver (small)
        i = np.argmax(cur_a)                # strongest donor
        j = np.random.randint(n)            # random receiver
        if i == j:
            continue

        delta = scale * cur_a[i]            # amount to move
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            # slowly increase step size when successful
            scale = min(scale * 1.03, 0.5)
        else:
            # shrink step size when not successful
            scale = max(scale * 0.97, 1e-4)

        # occasional logging
        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] greedy step {step}, best {best_score:.6f}")

    return best_a, best_score


def golden_section_optimize(func, lo: float, hi: float, max_iter: int = 12) -> Tuple[float, float]:
    """Minimises ``func`` on [lo, hi] using golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = func(d)
        else:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

# Parameter‑space local refinement helpers

def hill_climb_alpha(alpha0: float, n: int, deadline: float) -> float:
    """Simple hill‑climb in the one‑dimensional product‑family parameter."""
    best = alpha0
    best_score = compute_score(symmetric_product_family(n, best))
    step = 0.8
    while time.time() < deadline and step > 1e-4:
        improved = False
        for sign in (-1, 1):
            cand = max(0.0, best + sign * step)
            sc = compute_score(symmetric_product_family(n, cand))
            if sc < best_score:
                best_score = sc
                best = cand
                improved = True
                print(f"[{time.time():.1f}] alpha refined to {best:.6f} score {best_score:.6f}")
                break
        if not improved:
            step *= 0.5
    return best


def hill_climb_beta(p0: float, q0: float, n: int, deadline: float) -> Tuple[float, float]:
    """Simple hill‑climb in the two‑dimensional Beta‑family parameters."""
    best_p, best_q = p0, q0
    best_score = compute_score(beta_family(n, best_p, best_q))
    step = 0.4
    while time.time() < deadline and step > 1e-3:
        improved = False
        for dp in (-step, step):
            for dq in (-step, step):
                cand_p = max(0.01, best_p + dp)
                cand_q = max(0.01, best_q + dq)
                sc = compute_score(beta_family(n, cand_p, cand_q))
                if sc < best_score:
                    best_score = sc
                    best_p, best_q = cand_p, cand_q
                    improved = True
                    print(f"[{time.time():.1f}] Beta refined to p={best_p:.6f}, q={best_q:.6f} score {best_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
    return best_p, best_q

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` under a fixed wall‑clock budget.
    Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Optional fallback using a previously known sequence
    # -----------------------------------------------------------
    try:
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = np.array([float(x) for x in height_sequence_1], dtype=np.float64)  # type: ignore
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Choose a working length
    # -----------------------------------------------------------
    # Using a larger n gives a finer approximation of the continuous problem.
    n = 16384

    # -----------------------------------------------------------
    # 2. Coarse scan over analytic families (with a few random shifts)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] Starting coarse scan (n={n})")

    # product family
    best_alpha = None
    for alpha in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best (product α={alpha:.4f}) = {best_score:.6f}")

        # random right‑shifts
        for _ in range(2):
            d = np.random.randint(0, n // 30 + 1)
            shifted = np.concatenate([vec[d:], np.zeros(d)])
            shifted = normalize(shifted)
            sc2 = compute_score(shifted)
            if sc2 < best_score:
                best_score = sc2
                best_seq = shifted
                print(f"[{time.time():.1f}] shifted product α={alpha:.4f} d={d} → {best_score:.6f}")

    # left‑heavy power family
    for beta in np.linspace(0.0, 15.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] new best (left‑heavy β={beta:.4f}) = {best_score:.6f}")

    # exponential‑U family
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] new best (exp‑U γ={gamma:.4f}) = {best_score:.6f}")

    # mixture uniform + exponential‑U
    for lam in np.linspace(0.0, 1.0, 6):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

    # Beta family (grid)
    for p in np.linspace(0.2, 2.0, 10):
        for q in np.linspace(0.2, 2.0, 10):
            if time.time() >= deadline - 30.0:
                break
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec
                print(f"[{time.time():.1f}] new best (Beta p={p:.3f},q={q:.3f}) = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Random dense search over families (≈3000 evaluations)
    # -----------------------------------------------------------
    trials = 0
    max_trials = 3000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(5)
        if choice == 0:        # product family
            alpha = np.random.uniform(0.0, 15.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 1:      # beta family (full)
            p = np.random.uniform(0.05, 2.5)
            q = np.random.uniform(0.05, 2.5)
            vec = beta_family(n, p, q)
        elif choice == 2:      # left‑heavy power
            beta = np.random.uniform(0.0, 15.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 3:      # exponential‑U
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_u_family(n, gamma)
        else:                  # mixture uniform + product
            lam = np.random.uniform(0.0, 1.0)
            alpha = np.random.uniform(0.0, 15.0)
            vec = lam * np.full(n, 1.0 / n) + (1.0 - lam) * symmetric_product_family(n, alpha)
            vec = normalize(vec)

        # occasional random shift
        if np.random.rand() < 0.25:
            d = np.random.randint(0, n // 30 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] random trial #{trials} improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Fine local refinement of the best product‑α and Beta parameters
    # -----------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] local hill‑climb on product‑α (start {best_alpha:.4f})")
        refined_alpha = hill_climb_alpha(best_alpha, n, deadline - 10.0)
        vec = symmetric_product_family(n, refined_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            best_alpha = refined_alpha
            print(f"[{time.time():.1f}] refined product‑α={refined_alpha:.6f} → {best_score:.6f}")

    # If we have stored a beta candidate, refine it
    # (we keep the last beta parameters that gave an improvement)
    # Simple heuristic: reuse the parameters from the best beta grid scan
    # The grid scan stored the last (p,q) that improved the score.
    # We'll retrieve them from the last iteration if possible.
    # For simplicity we redo a short hill‑climb from a typical start.
    if time.time() < deadline - 20.0:
        # start from a balanced U‑shape (p=q≈0.5)
        start_p, start_q = 0.5, 0.5
        refined_p, refined_q = hill_climb_beta(start_p, start_q, n, deadline - 10.0)
        vec = beta_family(n, refined_p, refined_q)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec
            print(f"[{time.time():.1f}] refined Beta p={refined_p:.6f}, q={refined_q:.6f} → {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Adam (smooth‑max) refinement – no enforced symmetry
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Adam refinement (smooth‑max, no symmetry)")
        a_start = best_seq.astype(np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            start_a=a_start,
            deadline=deadline - 150.0,
            max_iters=800_000,
            beta_start=5.0,
            beta_end=20_000.0,
            lr=0.012,
            sym_interval=0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Projected sub‑gradient descent on the true max‑objective
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            start_a=best_seq.astype(np.float64),
            deadline=deadline - 60.0,
            max_iters=2_500_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub
            print(f"[{time.time():.1f}] after sub‑gradient best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Final greedy hill‑climb (mass‑transfer moves)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final greedy hill‑climb")
        a_final, sc_final = hill_climb_greedy(
            start_a=best_seq.astype(np.float64),
            deadline=deadline,
            max_steps=10_000_000,
            init_scale=0.15,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final
            print(f"[{time.time():.1f}] greedy improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 8. Fallback to a safe uniform vector if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list
    return best_seq.tolist()
