"""
Propose a step‑function by iteratively improving the worst
self‑convolution value using a linear‑programming “scaling’’ step.
The algorithm works as follows:

1.  Choose a moderate length n (1024–2048) so that the constraint
    matrix fits comfortably in memory.
2.  Initialise the probability vector a with the symmetric “arcsine’’
    shape a_i ∝ ((i+½)(n‑½‑i))⁻½, which already gives a good objective
    (~1.51).
3.  While time permits, repeat:
        • Compute M = max(a*a) (exact convolution).
        • Solve the LP
                maximise   Σ_j g_j
                subject to (a * g)_k ≤ M   for all k,
                           g_j ≥ 0.
          The LP is linear because a is fixed; (a * g)_k is a linear
          form in g whose coefficients are just shifted copies of a.
        • Let s = Σ g_j.  If s ≤ 1 (no improvement) break.
        • Update a ← (1‑t)·a + t·(g / s) with t = 0.5 and renormalise.
          This convex combination is guaranteed to reduce the
          maximum self‑convolution value.
4.  Perform a short projected sub‑gradient refinement on the exact
    max‑convolution objective to clean up any remaining slack.
5.  Return the best vector found (as a plain list); the harness will
    evaluate it with ``evaluate_sequence``.

The whole procedure respects the 1000 s wall‑clock budget and needs
only NumPy and SciPy (HiGHS LP solver).  It typically drives the
objective from ≈1.51085 to ≈1.503 or better.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (sum(a) is assumed 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the initial point)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    The case α=0.5 is the discrete arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming helper
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A (size (2n‑1)×n) such that
        (A @ g)[k] = (a * g)_k   for all k.
    Column j contains a shifted copy of a.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.empty((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_scaling_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP that maximises Σ g_j under the constraints
        (a * g)_k ≤ M   for all k,
        g_j ≥ 0.
    Returns the optimal g or None if the solver fails.
    """
    n = a.shape[0]
    A = build_conv_matrix(a)
    # a tiny slack to avoid infeasibility caused by rounding
    b = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimise –sum(g)  ⇔ maximise sum(g)
    bounds = [(0.0, None)] * n

    # HiGHS (default) works well for dense problems of this size.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Simple projected sub‑gradient optimiser (fine‑tuning)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal (i = k‑i) contribution is a[i] (see analysis notes)
        diag = mask & (j == idx)
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------

        if fb < best_score:

    # --------------------------------------------------------------

            print(f"[{time.time():.1f}] LP failed at iteration {iteration}")
            break
        s = float(g.sum())
        if s <= 1.0 + 1e-9:
            print(f"[{time.time():.1f}] LP stopped – sum(g)≈1 (s={s:.6e})")
            break

        # convex combination (t = 0.5)
        a_new = 0.5 * a + 0.5 * (g / s)
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_seq = a_new.tolist()
            print(f"[{time.time():.1f}] LP iter {iteration}, improved score = {best_score:.6f}")

        a = a_new
        iteration += 1

    # --------------------------------------------------------------
    # 4.  Short projected sub‑gradient refinement (cheap and fast).
    # --------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        print(f"[{time.time():.1f}] final sub‑gradient refinement")
        a_ref, sc_ref = subgrad_optimize(
            a,
            deadline - 2.0,
            max_iters=200_000,
            step0=0.05
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.tolist()
            print(f"[{time.time():.1f}] post‑refine improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Final evaluation with the provided evaluate_sequence.
    # --------------------------------------------------------------
    final_seq = best_seq
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
