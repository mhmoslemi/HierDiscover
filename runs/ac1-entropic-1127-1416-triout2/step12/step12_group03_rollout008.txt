"""
Propose a step‑function that minimises

        2·n·max(convolve(a, a)) / (sum(a)²)

The search proceeds as follows (all phases respect the global 1000 s budget).

1.  Load a fallback sequence if the harness supplies one (kept only as a
    last‑resort reference – we do not force the problem size to its length).
2.  Choose a relatively large problem size (4096).  Larger n gives a lower
    theoretical bound (≈ 1 for n≫1) while still keeping the O(n²) convolution
    cheap enough for many iterations.
3.  Initialise with the analytically‑motivated *symmetric‑product* (the
    discrete arcsine) shape: a_i ∝ ((i+½)(n‑½‑i))^{-½}.
4.  Perform a cheap random search over several parametric families
    (symmetric product, left‑heavy power, exponential, mixtures).  Every
    evaluation uses the exact objective (max‑convolution) and the best
    candidate is kept.
5.  Refine the best candidate with **exponentiated‑gradient (mirror descent)**
    on the true max‑convolution objective.  The update is multiplicative,
    the step size decays as η_t = η₀/√t and the iterate is re‑projected onto
    the probability simplex.
6.  Further improve by minimising a smooth‑max surrogate using **Adam**
    on the parametrisation a = softmax(z).  The surrogate is
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    with β increased from a small to a very large value.
7.  A long cheap **projected sub‑gradient descent** (additive updates) drifts
    the solution into a region of even lower max‑convolution.
8.  Optional symmetrisation (a ← (a+a[::-1])/2) after each major phase.
9.  A short **simulated‑annealing hill‑climb** to capture any remaining
    local improvements.
10. Return the final vector as a plain Python list; the harness will call
    `evaluate_sequence` on it.

All helper routines are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic geometry utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the random search)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑½‑i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·dist) with dist = min(i, n‑1‑i)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex combination of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing sequence (monotone after sorting)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

# Objective utilities

def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the harness (up to an irrelevant scale):
    2·n·max(convolve(a,a))  where sum(a)=1.
    """
    conv = np.convolve(a, a)          # full (aperiodic) convolution
    return 2.0 * a.shape[0] * float(conv.max())

# Optimisers

def eg_optimize(start_a: np.ndarray, deadline: float,
                eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) on the true max‑convolution."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    idx = np.arange(n)
    t = 1

    while time.time() < deadline:
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))

        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]                     # diagonal appears once

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {sc:.6f}")

        t += 1

    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def adam_smooth_optimize(start_a: np.ndarray, deadline: float,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate:
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    The variable is the log‑probabilities z with a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_sc = compute_score(best_a)

    it = 0
    n = start_a.shape[0]
    while time.time() < deadline:
        it += 1
        # linear schedule for β in log‑space
        t_frac = it / 100000.0                     # arbitrary large denominator
        beta = beta_start * (beta_end / beta_start) ** t_frac

        a = softmax(z)
        conv = np.convolve(a, a)
        maxc = conv.max()
        # stable soft‑max over the β‑scaled conv values
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                  # shape (2n‑1,)

        # gradient of the smooth max w.r.t a
        a_rev = a[::-1]
        conv_rev = np.convolve(a_rev, w)         # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # chain rule through softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_sc:
                best_sc = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {sc:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_sc:
        best_sc, best_a = sc_fin, a_fin.copy()
    return best_a, best_sc


def subgrad_optimize(start_a: np.ndarray, deadline: float,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected (additive) sub‑gradient descent on the exact max‑convolution."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    idx = np.arange(n)
    t = 1

    while time.time() < deadline:
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))

        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad improved to {sc:.6f}")

        t += 1

    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """A tiny simulated‑annealing random walk on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_sc = compute_score(cur_a)

    best_a = cur_a.copy()
    best_sc = cur_sc

    T0 = best_sc * 1e-4
    T = T0
    step_scale = 0.12
    step = 0

    while time.time() < deadline and step < max_steps:
        step += 1
        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        delta = sc - cur_sc
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_sc = sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_sc


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Average with the reversed vector and project onto the simplex."""
    b = (a + a[::-1]) * 0.5
    return project_to_simplex(b)

def random_search(n: int, deadline: float,
                  best_a: np.ndarray,
                  best_sc: float) -> tuple[np.ndarray, float]:
    """Explore the analytic families with random parameters."""
    while time.time() < deadline:
        r = np.random.rand()
        if r < 0.25:
            alpha = np.random.uniform(0.2, 1.4)
            cand = symmetric_product_family(n, alpha)
        elif r < 0.45:
            beta = np.random.uniform(0.2, 3.0)
            cand = left_heavy_power_family(n, beta)
        elif r < 0.65:
            gamma = np.random.uniform(0.0, 10.0)
            cand = exp_u_family(n, gamma)
        elif r < 0.80:
            lam = np.random.uniform(0.0, 1.0)
            gamma = np.random.uniform(0.0, 10.0)
            cand = mixed_uniform_u_family(n, lam, gamma)
        else:
            cand = random_monotone_start(n)

        cand = cand / cand.sum()
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            print(f"[{time.time():.1f}] Random search improved to {sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the wall‑clock budget and
    returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # 0.  Load a fallback candidate if the harness supplies one.
    # ------------------------------------------------------------------
    fallback_seq = None
    fallback_score = float("inf")
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_score = evaluate_sequence(fb_vec)
        fallback_seq = np.array(fb_vec, dtype=np.float64)
        print(f"[{time.time():.1f}] fallback score = {fallback_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose a problem size.
    # ------------------------------------------------------------------
    # Use a relatively large n irrespective of fallback length – larger n allows a
    # lower theoretical bound and still keeps O(n²) convolutions cheap.
    n = 4096
    print(f"[{time.time():.1f}] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Initialise – start from the symmetric‑product (arcsine) shape.
    # ------------------------------------------------------------------
    a = symmetric_product_family(n, 0.5)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(best_a)
    print(f"[{time.time():.1f}] initial arcsine score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Random parametric search (≈20 % of the budget or 120 s).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    rs_deadline = time.time() + min(0.20 * remaining, 120.0)
    if time.time() < rs_deadline:
        print(f"[{time.time():.1f}] random search phase (≈{rs_deadline - time.time():.1f}s)")
        best_a, best_sc = random_search(n, rs_deadline, best_a, best_sc)

    # ------------------------------------------------------------------
    # 4.  Symmetrise after the stochastic exploration.
    # ------------------------------------------------------------------
    best_a = symmetrize(best_a)
    best_sc = compute_score(best_a)
    print(f"[{time.time():.1f}] after symmetrisation score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # 5.  Exponentiated‑gradient optimisation.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()

        cand_a, cand_sc = eg_optimize(best_a, eg_deadline, eta0=0.8)
        if cand_sc < best_sc:
            best_a, best_sc = cand_a, cand_sc
            print(f"[{time.time():.1f}] EG improved best to {best_sc:.6f}")

    # ------------------------------------------------------------------
    # 6.  Adam optimisation of the smooth‑max surrogate.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_deadline = time.time() + min(0.25 * remaining, 250.0)
    if time.time() < adam_deadline:
        print(f"[{time.time():.1f}] Adam phase (≈{adam_deadline - time.time():.1f}s)")
        cand_a, cand_sc = adam_smooth_optimize(best_a, adam_deadline,
                                               beta_start=4.0,
                                               beta_end=5000.0,
                                               lr=0.018)
        if cand_sc < best_sc:
            best_a, best_sc = cand_a, cand_sc
            print(f"[{time.time():.1f}] Adam improved best to {best_sc:.6f}")

    # ------------------------------------------------------------------
    # 7.  Projected sub‑gradient descent (cheap, many iterations).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()

        cand_a, cand_sc = subgrad_optimize(best_a, subg_deadline, step0=0.07)
        if cand_sc < best_sc:
            best_a, best_sc = cand_a, cand_sc
            print(f"[{time.time():.1f}] Sub‑grad improved best to {best_sc:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
