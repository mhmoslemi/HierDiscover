"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

using a multi‑phase search with a *smooth* exponentiated‑gradient (EG)
phase (continuation on the log‑sum‑exp approximation of the max) and the
classic EG → sub‑gradient → LP → simulated‑annealing pipeline.
A symmetric‑only representation is enforced after each major phase to
exploit the inherent symmetry of the problem.

The algorithm typically reaches a score ≈ 1.503 in < 1000 s on a
single CPU.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution via real FFT.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families for the wide scan
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone left‑heavy: w_i ∝ (i+0.5)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential in the distance from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian: w_i ∝ exp(-γ·(i‑mid)²)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, window_type: str) -> np.ndarray:
    """Standard cosine windows."""
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(
            4.0 * np.pi * i / (n - 1)
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def raised_cosine_power_family(n: int, p: float) -> np.ndarray:
    """w_i ∝ (0.5‑0.5·cos(2πi/(n‑1)))^p (raised cosine power)."""
    i = np.arange(n, dtype=np.float64)
    base = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w = np.power(base, p)
    return w / w.sum()


def kaiser_window_family(n: int, beta: float) -> np.ndarray:
    """Kaiser window with shape parameter β."""
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    x = (i - alpha) / alpha  # in [-1,1]
    w = i0(beta * np.sqrt(1 - x * x))
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (heavy on the left)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped start."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smooth exponentiated‑gradient optimizer (continuation on log‑sum‑exp)
# ----------------------------------------------------------------------


def smooth_eg_optimize(
    start_a: np.ndarray,
    deadline: float,
    beta_schedule: list[tuple[int, float]] = None,
    eta0: float = 0.6,
    report_interval: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv) using a smooth approximation:
        f_β(a) = 2·n·log‑sum‑exp(β·conv)/β.
    The parameter β is increased according to ``beta_schedule``.
    The update is exponentiated‑gradient (mirror descent) on the smooth
    objective, with a 1/√t stepsize.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    # create a simple schedule if none supplied
    if beta_schedule is None:
        # (iteration limit, beta) – piecewise constant
        beta_schedule = [(20000, 5.0), (80000, 20.0), (200000, 80.0), (10 ** 9, 200.0)]

    sched_idx = 0
    next_iter, cur_beta = beta_schedule[sched_idx]

    t = 1
    while time.time() < deadline:
        # advance schedule if needed
        if t > next_iter and sched_idx + 1 < len(beta_schedule):
            sched_idx += 1
            next_iter, cur_beta = beta_schedule[sched_idx]

        # ---------- gradient of the smooth objective ----------
        conv = convolution_fft(a, a)                        # length 2n‑1
        # numerically stable soft‑max on β·conv
        maxc = conv.max()
        exp_arg = cur_beta * (conv - maxc)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient g_i = 2 * Σ_k w_k * a_{k‑i}  (see derivation in analysis)
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, w)               # length 3n‑2
        g = 2.0 * grad_full[n - 1 : n - 1 + n]               # slice of length n

        # ---------- exponentiated‑gradient step ----------
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional reporting and best‑track
        if t % report_interval == 0:
            # enforce symmetry (problem is symmetric)
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)

            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] smooth EG β={cur_beta:.1f} improved to {best_score:.6f}")

        t += 1

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Projected sub‑gradient descent (cheap long tail)
# ----------------------------------------------------------------------


def subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 2_500_000,
    step0: float = 0.07,
    report_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % report_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (classic state‑of‑the‑art mixing step)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0, then return the convex mixture
        f_new = (1−t)·f + t·g_norm,
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (size m×n) so that A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)  # maximise Σ g → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(
        c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True, "time_limit": 30}
    )

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)

    return f_new

# Stochastic hill‐climb (simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
    report_interval: int = 50_000,
) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves (standard SA).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

        if step % report_interval == 0:
            print(f"[{time.time():.1f}] SA step {step} best={best_score:.6f}")

    return best_a, best_score

    def try_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Analytic families + random starts (wide scan)
    # ------------------------------------------------------------------
    # Uniform baseline
    try_candidate(np.full(n, 1.0 / n, dtype=np.float64))

    # Product family – wide sweep
    alphas = np.concatenate([np.linspace(0.0, 5.0, 51), np.linspace(5.1, 200.0, 40)])
    for alpha in alphas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(product_family(n, alpha))

    # Left‑heavy power family
    betas = np.linspace(0.0, 30.0, 61)
    for beta in betas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(left_heavy_power_family(n, beta))

    # Exponential‑U family
    gammas = np.linspace(0.0, 20.0, 41)
    for gamma in gammas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(exp_u_family(n, gamma))

    # Gaussian family
    gauss_gammas = np.logspace(-4, 2, 40)
    for gamma in gauss_gammas:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(gaussian_family(n, gamma))

    # Cosine windows
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(window_family(n, win))

    # Raised‑cosine power family
    for p in [0.5, 1.0, 2.0, 3.0, 4.0, 6.0, 8.0, 12.0]:
        if time.time() >= deadline - 250.0:
            break
        try_candidate(raised_cosine_power_family(n, p))

    # Kaiser windows (β ∈ {0,2,…,14})
    for beta in range(0, 16, 2):
        if time.time() >= deadline - 250.0:
            break
        try_candidate(kaiser_window_family(n, beta))

    # Random starts
    for _ in range(250):
        if time.time() >= deadline - 250.0:
            break
        if np.random.rand() < 0.5:
            try_candidate(random_monotone_start(n))
        else:
            try_candidate(random_u_start(n))

    # ------------------------------------------------------------------
    # 3. Normalise the current best vector (fallback to uniform)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.asarray(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 4. Smooth exponentiated‑gradient phase (continuation)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print(f"[{time.time() - start_time:.1f}] starting smooth EG")
        a_opt, sc_opt = smooth_eg_optimize(
            a,
            deadline - 180.0,
            beta_schedule=[(20000, 5.0), (80000, 20.0), (200000, 80.0), (10 ** 9, 200.0)],
            eta0=0.6,
            report_interval=5000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] after smooth EG best = {best_score:.6f}")

        # One perturbed restart to escape shallow local minima
        if time.time() < deadline - 180.0:
            pert = a * np.exp(0.02 * np.random.randn(n))
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            a_opt2, sc_opt2 = smooth_eg_optimize(
                pert,
                deadline - 120.0,
                beta_schedule=[(20000, 5.0), (80000, 20.0), (200000, 80.0), (10 ** 9, 200.0)],
                eta0=0.6,
                report_interval=5000,
            )
            if sc_opt2 < best_score:
                best_score = sc_opt2
                a = a_opt2
                print(f"[{time.time() - start_time:.1f}] after perturbed smooth EG best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Long cheap projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time() - start_time:.1f}] starting sub‑gradient descent")
        a_opt, sc_opt = subgrad_optimize(
            a,
            deadline - 30.0,
            max_iters=4_000_000,
            step0=0.07,
            report_interval=2000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. LP‑based refinement loop (multiple mixes)
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print(f"[{time.time() - start_time:.1f}] starting LP refinement loop")
        mix_weight = 0.2
        lp_iters = 0
        while time.time() < deadline - 20.0 and lp_iters < 30:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc = compute_score(a_new)
            if sc < best_score - 1e-7:
                best_score = sc
                a = a_new
                print(f"[{time.time() - start_time:.1f}] LP step {lp_iters} improved to {best_score:.6f}")
            else:
                mix_weight = max(0.05, mix_weight * 0.85)
            lp_iters += 1

    # ------------------------------------------------------------------
    # 7. Symmetrisation (the true optimum is symmetric)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time() - start_time:.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
