"""
Propose a non‑negative step‑function that minimizes

    2·n·max(convolve(a,a)) / (sum(a)²)

using a multi‑phase strategy:

1.  Very wide exploration of analytic families and random starts.
2.  Exponentiated‑gradient (mirror descent) from the best candidate.
3.  **Soft‑max‑smoothed gradient descent** – a smooth approximation of the
    “max‑convolution” objective that gives useful gradients even when the
    true max is attained at a single index.
4.  A few linear‑programming mixing steps (the classic state‑of‑the‑art
    refinement).
5.  Simulated‑annealing hill‑climb for a final polish.

All phases respect the global 1000 s wall‑clock limit.
The returned list is ready for ``evaluate_sequence`` (which will be imported
by the harness).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_weights(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns probabilities)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone, heavy left side: w_i ∝ (i+0.5)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential distance: w_i ∝ exp(-γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def gaussian_family(n: int, gamma: float) -> np.ndarray:
    """Discrete Gaussian."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.exp(-gamma * ((i - mid) ** 2))
    return w / w.sum()


def window_family(n: int, window_type: str) -> np.ndarray:
    """Standard cosine windows."""
    i = np.arange(n, dtype=np.float64)
    if window_type == "hann":
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif window_type == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (heavy left)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shaped start."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


def random_dirichlet_start(n: int, conc: float = 0.5) -> np.ndarray:
    """Dirichlet draw – often very sparse."""
    a = np.random.gamma(conc, 1.0, size=n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimisation
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 300_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution
    objective.  Returns the best vector encountered.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]  # diagonal term gets weight 1

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Soft‑max‑smoothed gradient descent
# ----------------------------------------------------------------------


def softmax_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 250_000,
                     step0: float = 0.06,
                     beta0: float = 0.5,
                     beta_mult: float = 1.001,
                     beta_max: float = 2000.0) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of max(conv):
        F(a) = log-sum-exp(β·conv(a)) / β

    The gradient can be computed efficiently via an extra convolution.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    beta = float(beta0)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)                # length 2n‑1
        # soft‑max weights (stable)
        max_c = np.max(conv)
        w = np.exp(beta * (conv - max_c))
        w /= w.sum()                                # shape (2n‑1,)

        # gradient of weighted sum: g = Σ_k w_k ∇ conv_k
        # using the convolution trick described in the analysis
        a_rev = a[::-1]
        tmp = convolution_fft(w, a_rev)            # length 3n‑2
        start = n - 1                               # index of first full overlap
        g = 2.0 * tmp[start:start + n]              # double count all pairs
        diag_idx = 2 * np.arange(n)                 # positions where i=j
        g -= w[diag_idx] * a                        # correct the diagonal

        # projected gradient step
        step = step0 / math.sqrt(t)
        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # slowly increase the sharpness of the soft‑max
        if t % 2000 == 0:
            beta = min(beta * beta_mult, beta_max)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Soft‑max GD improved to {best_score:.6f} (β={beta:.2f})")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (the classic state‑of‑the‑art iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0, then return the convex mixture
        f_new = (1−t)·f + t·g_norm,
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A such that A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)      # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # Use the HiGHS LP solver (fast and reliable)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)

    return f_new


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function respects the global wall‑clock budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback supplied by the harness (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time() - start_time:.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose a problem size (the length of the step function)
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    n = fallback_len if fallback_len >= 1024 else 2048
    print(f"[{time.time() - start_time:.1f}] using length n={n}")

    # ------------------------------------------------------------------
    # Helper to test a candidate vector (normalised internally)
    # ------------------------------------------------------------------
    def try_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        a = np.asarray(vec, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}] new best score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Analytic families + random starts (very wide scan)
    # ------------------------------------------------------------------
    try_candidate(np.full(n, 1.0 / n, dtype=np.float64))

    # Product family (U‑shaped)
    alphas = np.concatenate([np.linspace(0.0, 5.0, 51),
                             np.linspace(5.1, 200.0, 40)])
    for alpha in alphas:
        if time.time() >= deadline - 300.0:
            break
        try_candidate(product_family(n, alpha))

    # Left‑heavy power family
    betas = np.linspace(0.0, 30.0, 61)
    for beta in betas:
        if time.time() >= deadline - 300.0:
            break
        try_candidate(left_heavy_power_family(n, beta))

    # Exponential‑U family
    gammas = np.linspace(0.0, 20.0, 41)
    for gamma in gammas:
        if time.time() >= deadline - 300.0:
            break
        try_candidate(exp_u_family(n, gamma))

    # Gaussian family
    gauss_gammas = np.logspace(-4, 2, 40)
    for gamma in gauss_gammas:
        if time.time() >= deadline - 300.0:
            break
        try_candidate(gaussian_family(n, gamma))

    # Cosine windows
    for win in ("hann", "hamming", "blackman"):
        if time.time() >= deadline - 300.0:
            break
        try_candidate(window_family(n, win))

    # Random monotone / U‑shapes / Dirichlet starts
    for _ in range(300):
        if time.time() >= deadline - 300.0:
            break
        r = np.random.rand()
        if r < 0.4:
            try_candidate(random_monotone_start(n))
        elif r < 0.8:
            try_candidate(random_u_start(n))
        else:
            try_candidate(random_dirichlet_start(n, conc=np.random.uniform(0.1, 2.0)))

    # ------------------------------------------------------------------
    # 3.  Normalise the current best (fallback to uniform if needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.asarray(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 4.  Exponentiated‑gradient (mirror descent) phase
    # ------------------------------------------------------------------
    if time.time() < deadline - 250.0:
        print(f"[{time.time() - start_time:.1f}] starting EG optimisation")
        a_opt, sc_opt = eg_optimize(
            a,
            deadline - 180.0,
            max_iters=400_000,
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] after EG best = {best_score:.6f}")

        # Perturbed restart of EG
        if time.time() < deadline - 180.0:
            perturb = a * np.exp(0.02 * np.random.randn(n))
            perturb = np.maximum(perturb, 0.0)
            perturb /= perturb.sum()
            a_opt2, sc_opt2 = eg_optimize(
                perturb,
                deadline - 120.0,
                max_iters=250_000,
                eta0=0.6,
            )
            if sc_opt2 < best_score:
                best_score = sc_opt2
                a = a_opt2
                print(f"[{time.time() - start_time:.1f}] after perturbed EG best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Soft‑max‑smoothed gradient descent (smooth approximation of max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time() - start_time:.1f}] starting soft‑max GD")
        # Run a few independent restarts – each gets a slice of the remaining budget
        remaining = deadline - time.time()
        # allocate roughly equal time for up to 3 restarts
        for restart in range(3):
            if time.time() >= deadline - 30.0:
                break
            # small random perturbation of the current best
            pert = a * np.exp(0.03 * np.random.randn(n))
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            # time left for this restart
            sub_deadline = min(deadline, time.time() + remaining / (3 - restart))
            a_opt_s, sc_opt_s = softmax_optimize(
                pert,
                sub_deadline,
                max_iters=200_000,
                step0=0.06,
                beta0=0.5,
                beta_mult=1.002,
                beta_max=2000.0,
            )
            if sc_opt_s < best_score:
                best_score = sc_opt_s
                a = a_opt_s
                print(f"[{time.time() - start_time:.1f}] after soft‑max restart {restart} best = {best_score:.6f}")
            remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 6.  LP‑based refinement loop (multiple mixes)
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print(f"[{time.time() - start_time:.1f}] starting LP refinement loop")
        mix_weight = 0.2
        lp_iters = 0
        while time.time() < deadline - 20.0 and lp_iters < 30:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc = compute_score(a_new)
            if sc < best_score - 1e-9:
                best_score = sc
                a = a_new
                print(f"[{time.time() - start_time:.1f}] LP step {lp_iters} improved to {best_score:.6f}")
            else:
                mix_weight = max(0.05, mix_weight * 0.85)
            lp_iters += 1

    # ------------------------------------------------------------------
    # 7.  Symmetrisation (the problem is symmetric)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time() - start_time:.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time() - start_time:.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final evaluation (using the provided evaluate_sequence) and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time() - start_time:.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
