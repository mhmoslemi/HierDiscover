def var_optimize(start_a, deadline, max_iters=200_000, step0=0.2, report_every=2000):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    mu = 1.0 / (2*n - 1)
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # conv = convolution_fft(a, a)
        conv = convolution_fft(a, a)
        diff = conv - mu
        # compute gradient: g = 4 * (diff cross-correlate a)
        a_rev = a[::-1]
        # conv_full = convolution_fft(diff, a_rev) => shape 3n-2
        conv_full = convolution_fft(diff, a_rev)
        g = 4.0 * conv_full[n-1:n-1+n]  # slice of length n
        # step size
        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)
        if t % report_every == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] var_opt improved to {best_score:.6f}")
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

"""
Improved step‑function search.

The algorithm proceeds as follows:

1. Load a fallback candidate (if provided) and keep it only if it is
   longer than the length we intend to use.
2. Choose a problem size `n`.  We aim for a relatively large `n`
   (4096–8192) – larger sequences give a better theoretical lower bound.
3. Initialise the distribution with a symmetric “product’’ family
   `a_i ∝ ((i+½)(n‑½‑i))^{-α}` (continuous arcsine shape).  A quick
   scan over a few `α` values yields a good starting point.
4. **Variance descent** – we minimise the L2‑variance of the self‑
   convolution,
        L(a)=∑_k ( (a*a)_k – μ )² ,   μ = 1/(2n‑1) .
   The exact gradient is
        g = 4·( (a*a) * a_rev )[n‑1:n‑1+n] – 4·μ·1,
   where `*` is a linear convolution and `a_rev` is the reversed vector.
   A projected gradient step (step size ∝ 1/√t) and Euclidean projection
   onto the probability simplex are performed many times (until the
   allotted time slice expires).  This flattening of the convolution
   quickly reduces the maximal entry.
5. **Projected sub‑gradient descent on the true max‑convolution** – a
   cheap refinement that directly attacks the worst convolution entry.
6. **Optional simulated‑annealing hill‑climb** for the last few seconds.
7. Return the best sequence found (as a plain Python list) and let the
   harness evaluate it with `evaluate_sequence`.

All phases respect the global wall‑clock budget (default 1000 s) and
use only NumPy (and optionally SciPy for a tiny LP fallback).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # used only for a tiny fallback, not essential


# ----------------------------------------------------------------------
# Basic utilities

def next_pow2(x: int) -> int:
    """Return the smallest power of two that is ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution using real FFT (rfft/irfft).
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the evaluator (up to the global scaling that
    cancels out).  The vector `a` is assumed to be non‑negative and
    normalised to sum = 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)(n‑½‑i))^{-α}.
    For α=0.5 this is the discrete arcsine distribution.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a

def var_optimize(start_a: np.ndarray,
                 deadline: float,
                 max_iters: int = 200_000,
                 step0: float = 0.45,
                 report_every: int = 4000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the variance of the self‑convolution:
        L(a)=∑_k ( (a*a)_k – μ )² ,   μ = 1/(2n‑1).

    The exact gradient is
        g = 4·( (a*a) * a_rev )[n‑1 : n‑1+n] – 4·μ·1 .
    """
    n = start_a.shape[0]
    a = start_a.copy()
    mu = 1.0 / (2 * n - 1)

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)                  # a*a
        # convolution of conv with reversed a
        a_rev = a[::-1]
        conv_rev = convolution_fft(conv, a_rev)       # length 3n‑2
        # gradient (slice of length n)
        g = 4.0 * (conv_rev[n - 1:n - 1 + n] - mu)

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % report_every == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] var_opt improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.07,
                     report_every: int = 2000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Very cheap per iteration, allowing many steps.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]          # diagonal term appears only once

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % report_every == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000,
                  report_every: int = 20_000) -> tuple[np.ndarray, float]:
    """
    Very simple simulated‑annealing style random walk on the simplex.
    Occasionally accepts uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

        if step % report_every == 0:
            print(f"[{time.time():.1f}] SA step {step}/{max_steps}, best {best_score:.6f}")

    return best_a, best_score
