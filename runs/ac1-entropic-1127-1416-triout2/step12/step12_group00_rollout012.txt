"""
search for optimal non‑negative step heights that minimise evaluate_sequence.

Algorithm
---------
1. Broad generation of candidate probability vectors from many parametric
   families (power‑law, exponential, beta, U‑shape, sine‑power) and a large
   pool of random Dirichlet draws.  The best few are kept.

2. For each kept vector we run a projected Adam optimiser that minimises a
   smooth approximation of  max(conv)  (τ·log‑sum‑exp).  The temperature τ is
   annealed exponentially to the final value 1e‑7.  After each Adam step the
   vector is projected onto the probability simplex.

3. A short sub‑gradient polishing phase (as in the previous code) refines
   the best vector.

4. A stochastic hill‑climb (simulated annealing) gives a final escape from
   shallow local minima.

All phases respect the 1000 s wall‑clock budget; the best sequence found
is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # Return uniform distribution if v is all non‑positive.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # Should not happen, but fallback to uniform.
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient g = 2 * (a ⋆ w_rev) where w_rev is w reversed.
    The result has length len(a) and corresponds to the derivative of
    sum_k w_k * conv_k with respect to a (see smooth‑max derivation).
    """
    n = a.shape[0]
    m = len(w)
    rev_w = w[::-1]
    size = 1 << ((n + m - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(rev_w, n=size)
    conv_full = np.fft.irfft(A * B, n=size)[: n + m - 1]
    idx = (m - 1) - np.arange(n)          # indices that correspond to a_i
    grad = 2.0 * conv_full[idx]
    return grad


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def smooth_max(conv: np.ndarray, tau: float) -> (float, np.ndarray):
    """
    Return the smooth approximation S = τ·log(∑ exp(conv / τ))
    and the associated soft‑max weights w (sum to one).
    """
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}.  p can be negative to obtain an increasing shape."""
    exponents = -(p)
    a = np.arange(1, n + 1, dtype=float) ** exponents
    total = a.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i).  lam can be negative for growing exponentials."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i‑mid)²/(2σ²))."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_ushape(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution a_i ∝ ((i+1)*(n-i))^{p} (p>0 puts mass near the edges)."""
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n + 1 - i)) ** p
    total = a.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total

def explore_candidates(n: int, deadline: float, keep: int = 10) -> list[np.ndarray]:
    """Generate many vectors from the families above and from Dirichlet draws.
    Keep the ``keep`` best according to compute_score."""
    best = []   # list of (score, vector)

    def consider(v: np.ndarray):
        if not np.isfinite(v).all():
            return
        v = np.maximum(v, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        if len(best) < keep:
            best.append((sc, v.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, v.copy())
            best.sort(key=lambda x: x[0])

    # 1) Uniform (baseline)
    consider(np.full(n, 1.0 / n))

    # 2) Power‑law sweep (both decreasing and increasing)
    for p in np.linspace(0.0, 4.0, 41):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))
        consider(generate_powerlaw(n, -p))

    # 3) Exponential sweep
    for lam in np.linspace(-0.3, 0.3, 61):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # 4) Beta family
    al_vals = np.linspace(0.1, 5.0, 21)
    be_vals = np.linspace(0.1, 5.0, 21)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, alpha, beta))

    # 5) Gaussian widths
    for sigma in np.linspace(0.05 * n, 0.5 * n, 15):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # 6) Sine‑power shapes
    for p in np.linspace(0.1, 8.0, 20):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # 7) U‑shape (edge‑heavy) families
    for p in np.linspace(0.1, 5.0, 20):
        if time.time() >= deadline:
            break
        consider(generate_ushape(n, p))

    # 8) Random Dirichlet draws with varied concentration
    concentrations = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_conc = max(1, (keep * 200) // len(concentrations))
    for conc in concentrations:
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            consider(a)

    # Return just the vectors, sorted by score ascending.
    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]

def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 0.5,
                  tau_end: float = 1e-7,
                  tau_decay: float = 0.90,
                  lr: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Projected Adam minimising τ·log‑sum‑exp(conv) with an exponential τ‑schedule."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    # Adam state
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0

    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        # anneal τ every few iterations
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        S, w = smooth_max(conv, tau)          # smooth‑max value and weights

        grad = conv_with_weights(a, w)         # ∂/∂a (∑ w_k conv_k)

        # Adam update on a (we minimise S, which is proportional to the objective)
        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (lr * 0.002 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a

def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # accumulate sub‑gradient from all active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # propose moving some mass from i to j
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

    return 2.0 * conv_full[idx]


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def smooth_max(conv: np.ndarray, tau: float) -> (float, np.ndarray):
    """
    Return the smooth approximation S = τ·log(∑ exp(conv/τ))
    together with the soft‑max weights w (∑ w = 1).
    """
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    Z = exp_vals.sum()
    w = exp_vals / Z
    S = shift + tau * np.log(Z)
    return S, w


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (p may be negative to obtain an increasing shape)."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) (lam may be negative)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α−1}·(n−i)^{β−1}."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(−(i−mid)²/(2σ²))."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


def generate_ushape(n: int, p: float) -> np.ndarray:
    """U‑shaped: a_i ∝ ((i+1)*(n−i))^{p}, p>0 concentrates mass near the edges."""
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n + 1 - i)) ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global exploration – keep a few promising vectors
# ----------------------------------------------------------------------


def explore_candidates(n: int, deadline: float, keep: int = 12) -> list[np.ndarray]:
    """Generate many vectors, keep the ``keep`` best (according to compute_score)."""
    best = []                     # list of (score, vector)

    def consider(v: np.ndarray):
        if not np.isfinite(v).all():
            return
        v = np.maximum(v, 0.0)
        v = project_to_simplex(v)
        sc = compute_score(v)
        if len(best) < keep:
            best.append((sc, v.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, v.copy())
            best.sort(key=lambda x: x[0])

    # 1) Uniform (benchmark)
    consider(np.full(n, 1.0 / n))

    # 2) Power‑law (both decreasing and increasing)
    for p in np.linspace(0.0, 4.0, 41):
        if time.time() >= deadline:
            break
        consider(generate_powerlaw(n, p))
        consider(generate_powerlaw(n, -p))

    # 3) Exponential sweep
    for lam in np.linspace(-0.3, 0.3, 61):
        if time.time() >= deadline:
            break
        consider(generate_exponential(n, lam))

    # 4) Beta family
    al_vals = np.linspace(0.1, 5.0, 21)
    be_vals = np.linspace(0.1, 5.0, 21)
    for a in al_vals:
        for b in be_vals:
            if time.time() >= deadline:
                break
            consider(generate_beta(n, a, b))

    # 5) Gaussian widths
    for sigma in np.linspace(0.05 * n, 0.5 * n, 15):
        if time.time() >= deadline:
            break
        consider(generate_gaussian(n, sigma))

    # 6) Sine‑power shapes
    for p in np.linspace(0.1, 8.0, 20):
        if time.time() >= deadline:
            break
        consider(generate_sine_power(n, p))

    # 7) U‑shape (edge‑heavy) families
    for p in np.linspace(0.1, 5.0, 20):
        if time.time() >= deadline:
            break
        consider(generate_ushape(n, p))

    # 8) Random Dirichlet draws with many concentrations
    concentrations = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws = max(1, (keep * 200) // len(concentrations))
    for conc in concentrations:
        for _ in range(draws):
            if time.time() >= deadline:
                break
            consider(np.random.dirichlet([conc] * n))

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  tau_start: float = 0.5,
                  tau_end: float = 1e-7,
                  tau_decay: float = 0.90,
                  lr: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Projected Adam minimising τ·log‑sum‑exp(conv) with exponential τ‑schedule."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_val = compute_score(a)

    while time.time() < deadline:
        if t % 200 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        conv = full_convolution(a)
        _, w = smooth_max(conv, tau)          # w are the soft‑max weights
        grad = conv_with_weights(a, w)

        t += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad ** 2)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 1200 == 0:
            jitter = np.random.randn(n) * (lr * 0.002 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient polishing (as before, but a bit longer if time permits)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb (keeps an upper bound)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.07,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.03, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.97, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    n = 2048
    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0. Quick fallback if a known construction is available
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Broad exploration (≈ 150 s)
    # -----------------------------------------------------------------
    explore_deadline = deadline - 600.0
    if time.time() < explore_deadline:
        print("[explore] generating candidates …")
        cand_vecs = explore_candidates(n, explore_deadline, keep=12)
        for vec in cand_vecs:
            val = compute_score(vec)
            if val < best_score:
                best_score, best_seq = val, vec.tolist()
        print(f"[explore] best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Adam optimisation from the best few candidates (≈ 300 s)
    # -----------------------------------------------------------------
    if best_seq is not None:
        grad_deadline = min(deadline - 250.0,
                            start + 150.0 + (deadline - start) * 0.30)
        start_points = []

        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        start_points.append(best_vec)

        start_points.append(np.full(n, 1.0 / n))
        for _ in range(4):
            start_points.append(project_to_simplex(np.random.rand(n)))

        # add a few top candidates from the exploration (if they exist)
        for vec in cand_vecs[:3]:
            start_points.append(vec)

        refined_best = best_vec.copy()
        refined_val = best_score

        for idx, a0 in enumerate(start_points):
            if time.time() >= grad_deadline:
                break
            local_deadline = grad_deadline - (time.time() - start) - 0.5
            if local_deadline <= time.time():
                break
            print(f"[adam {idx}] start")
            a_opt = adam_optimize(a0, local_deadline)
            val = compute_score(a_opt)
            print(f"[adam {idx}] refined score = {val:.6f}")
            if val < refined_val:
                refined_val, refined_best = val, a_opt.copy()

        best_score, best_seq = refined_val, refined_best.tolist()
        print(f"[adam] best after Adam = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (≈ 150 s)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        print("[polish] start")
        a_polished = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_polished)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing (final refinement, ≈ 80 s)
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        print("[anneal] start")
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_score:
            best_score, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – uniform vector if nothing has been found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
