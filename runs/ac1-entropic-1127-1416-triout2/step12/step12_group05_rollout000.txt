"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

under the 1000 s wall‑clock budget.

Algorithm (single‑CPU, deterministic)
-------------------------------------------------
1. Choose a relatively large length `n = 8192` (more points give a lower
   theoretical bound, while FFT‑based convolutions stay fast).
2. Scan a handful of analytic families (product‑power, left‑heavy power,
   exponential‑U, mixed uniform‑U, beta) with random parameters.
   Keep the top‑k (k≤5) candidates.
3. Refine each of the top candidates with an Adam optimiser that works on
   the **smooth‑max** surrogate of the infinity‑norm.  Symmetry
   `a[i]=a[n‑1‑i]` is enforced each iteration (averaging + simplex projection).
   The best Adam result is kept.
4. Starting from that best vector, run a projected sub‑gradient descent on
   the exact max‑objective.  The sub‑gradient is obtained via FFT, the
   step‑size decays like `step0/√t`, and symmetry is re‑imposed after each
   update.
5. Finish with a simulated‑annealing hill‑climb on the simplex (again with
   symmetry).  The temperature starts proportional to the current score
   and cools linearly.
6. Return the best vector found (as a plain ``list[float]``).  The final
   official score is obtained with the provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution using real FFT (fast O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    conv = convolution_fft(a, a)
    mx = float(conv.max())
    return 2.0 * n * mx / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry v[i] = v[n-1-i]."""
    return (v + v[::-1]) * 0.5


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip negative values, then renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v /= s
    return v


# ----------------------------------------------------------------------
# Analytic families (all return a *sum‑normalised* vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}   (symmetric, U‑shaped for α>0)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return normalize(w)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}   (heavy on the left, monotone)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return normalize(w)


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return normalize(w)


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    return normalize(w)


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    return normalize(w)


def triangular_family(n: int, height: float) -> np.ndarray:
    """Symmetric triangular bump: linearly rises to centre then falls."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.maximum(0.0, height - np.abs(i - centre))
    return normalize(w)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         beta_start: float = 5.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.015,
                         sym_interval: int = 1) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Symmetry is enforced every ``sym_interval`` iterations (default every iter).
    """
    n = start_a.shape[0]
    eps = 1e-12
    # Initialise log‑softmax variables (positivity guaranteed)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = normalize(np.exp(z))
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # Temperature schedule for the smooth‑max
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # Current a (softmax) and enforce symmetry
        a = np.exp(z)
        a = a / a.sum()
        if sym_interval and (it % sym_interval == 0):
            a = symmetrize(a)
            a = project_to_simplex(a)
            # keep z consistent with the symmetrised a
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()

        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # Back‑prop through the soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 2000 == 0:
            a_tmp = normalize(np.exp(z))
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = normalize(np.exp(z))
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.075) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    Symmetry is enforced each iteration.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # indicator of active indices (within tiny tolerance)
        active = np.where(conv >= maxc - 1e-12)[0]

        # build weight vector that is 1 on every active index, else 0
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(1, active.size)   # average over the active set

        # sub‑gradient of max(conv) = sum_k w_k * grad conv_k
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        g = 2.0 * conv_rev[n - 1:n - 1 + n]

        # diminishing step size
        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)
        a = symmetrize(a)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000,
                  temp_factor: float = 1e-3,
                  step_scale_init: float = 0.2) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex with enforced symmetry.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # higher initial temperature (proportional to current score)
    T0 = best_score * temp_factor
    T = T0
    step_scale = step_scale_init

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                       # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # enforce symmetry and simplex constraints
        cand = symmetrize(cand)
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        delta_score = cand_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] SA step {step:,} – best = {best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given ``budget_s`` (seconds).
    Returns the best sequence found as a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin of a few seconds to guarantee we can return
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # ----------------------------------------------------------------------
    # Optional fallback to a previously known good solution
    # ----------------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be defined in the global namespace of the runner
        fb = evaluate_sequence([float(x) for x in height_sequence_1])  # type: ignore
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = [float(x) for x in height_sequence_1]  # type: ignore
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # 1. Choose working length (larger n gives a finer approximation)
    # ----------------------------------------------------------------------
    n = 8192
    print(f"[{time.time():.1f}] starting search with n = {n}")

    # ----------------------------------------------------------------------
    # 2. Coarse scan of analytic families
    # ----------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse scan of families")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    def consider(vec: np.ndarray, hint: str = "") -> None:
        nonlocal best_score, best_seq, top_candidates
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new global best {hint} = {best_score:.6f}")
        # keep a small pool of good candidates for later refinement
        if len(top_candidates) < 5 or sc < top_candidates[-1][0]:
            top_candidates.append((sc, vec.copy()))
            top_candidates.sort(key=lambda x: x[0])
            if len(top_candidates) > 5:
                top_candidates.pop()

    # product family (symmetric)
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        consider(vec, f"prod‑α={alpha:.2f}")

    # left‑heavy power family
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        consider(vec, f"left‑β={beta:.2f}")

    # exponential‑U family
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        consider(vec, f"exp‑γ={gamma:.2f}")

    # mixed uniform + U shape
    for lam in np.linspace(0.0, 1.0, 6):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            consider(vec, f"mix λ={lam:.2f},γ={gamma:.2f}")

    # beta family
    for _ in range(200):
        if time.time() >= deadline - 30.0:
            break
        p = np.random.uniform(0.1, 4.0)
        q = np.random.uniform(0.1, 4.0)
        vec = beta_family(n, p, q)
        consider(vec, f"beta p={p:.2f},q={q:.2f}")

    # ----------------------------------------------------------------------
    # 3. Random dense search over families (≈2000 evaluations)
    # ----------------------------------------------------------------------
    trials = 0
    max_trials = 2000
    while time.time() < deadline - 30.0 and trials < max_trials:
        trials += 1
        choice = np.random.randint(5)
        if choice == 0:  # product family
            alpha = np.random.uniform(0.0, 12.0)
            vec = symmetric_product_family(n, alpha)
        elif choice == 1:  # beta family
            p = np.random.uniform(0.1, 4.0)
            q = np.random.uniform(0.1, 4.0)
            vec = beta_family(n, p, q)
        elif choice == 2:  # left‑heavy power
            beta = np.random.uniform(0.0, 12.0)
            vec = left_heavy_power_family(n, beta)
        elif choice == 3:  # exponential‑U
            gamma = np.random.uniform(0.0, 6.0)
            vec = exp_u_family(n, gamma)
        else:  # mixed uniform + product
            lam = np.random.uniform(0.0, 1.0)
            alpha = np.random.uniform(0.0, 12.0)
            vec = lam * np.full(n, 1.0 / n) + (1.0 - lam) * symmetric_product_family(n, alpha)
            vec = normalize(vec)

        # occasional random right‑shift (zero‑padding)
        if np.random.rand() < 0.25:
            d = np.random.randint(0, n // 20 + 1)
            vec = np.concatenate([vec[d:], np.zeros(d)])
            vec = normalize(vec)

        consider(vec, f"rand‑trial #{trials}")

    # ----------------------------------------------------------------------
    # 4. Refine the best few candidates with Adam (smooth‑max)
    # ----------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] Adam refinement of top candidates")
        for idx, (sc, cand) in enumerate(top_candidates):
            if time.time() >= deadline - 120.0:
                break
            print(f"  candidate {idx+1} (score {sc:.6f})")
            a_opt, sc_opt = adam_smooth_optimize(
                start_a=cand,
                deadline=deadline - 100.0,
                max_iters=400_000,
                beta_start=5.0,
                beta_end=5000.0,
                lr=0.012,
                sym_interval=1,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.tolist()
                print(f"[{time.time():.1f}] Adam improved global best → {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5. Projected sub‑gradient descent on the true max‑objective
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] sub‑gradient descent refinement")
        a_sub, sc_sub = subgrad_optimize(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline - 60.0,
            max_iters=1_200_000,
            step0=0.075,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] sub‑grad improved global best → {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline,
            max_steps=5_000_000,
            temp_factor=1e-3,
            step_scale_init=0.2,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved global best → {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 7. Safety fallback (uniform) – should never be needed
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ----------------------------------------------------------------------
    # 8. Final official evaluation (includes security checks)
    # ----------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
