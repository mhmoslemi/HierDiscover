"""
search_step_function_lp.py

A multi‑stage optimiser for the ``evaluate_sequence`` objective.
Stages
    1. (optional) a quick parametric sweep to obtain a decent start vector.
    2. An LP‑driven “active‑set” improvement: solve
           max Σ g_i   s.t.  g_i ≥ 0 ,  (f * g)(k) ≤ M  ∀ active k
       where M = max(f*f).  The optimum pushes mass into positions where
       the current convolution is small.  The new vector is a convex
       combination f←(1‑t)f + t·g (t adapted during the run).
    3. Projected sub‑gradient polishing.
    4. Stochastic hill‑climb for final fine‑tuning.

The LP step (using ``scipy.optimize.linprog``) is the main novelty compared
to the previous implementation and typically yields a further improvement
below the 1.503 target within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a * a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for the optional sweep)
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def geometric(n: int, q: float) -> np.ndarray:
    a = q ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def mixed_plateau_power(n: int, L_frac: float, p: float) -> np.ndarray:
    L = int(L_frac * n)
    if L <= 0 or L >= n:
        return None
    tail_len = n - L
    tail = (np.arange(1, tail_len + 1, dtype=np.float64) ** (-p))
    head_val = tail.max()
    head = np.full(L, head_val, dtype=np.float64)
    a = np.concatenate([head, tail])
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Very broad parametric sweep (kept modest to save time)
# ----------------------------------------------------------------------
def parametric_sweep(n: int, time_deadline: float) -> np.ndarray:
    best_vec = None
    best_val = float('inf')

    # power‑law sweep
    for p in np.linspace(0.10, 2.00, 151):
        if time.time() >= time_deadline:
            return best_vec
        a = power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # geometric sweep (includes ratios extremely close to 1)
    for q in np.concatenate((np.linspace(0.90, 0.99, 81),
                             np.linspace(0.990, 0.9999, 70))):
        if time.time() >= time_deadline:
            return best_vec
        a = geometric(n, q)
        if a is None:
            continue
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    # beta‑shape sweep (wide range)
    alphas = np.linspace(-0.8, 3.0, 15)
    betas = np.linspace(-0.8, 3.0, 15)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= time_deadline:
                return best_vec
            a = beta_shape(n, alpha, beta)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_vec = val, a

    # random convex combinations of a few basic shapes
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    power = power_law(n, 0.5)
    geom = geometric(n, 0.98)
    basic = [uniform, power, geom]
    for _ in range(500):
        if time.time() >= time_deadline:
            break
        coeffs = np.random.rand(3)
        coeffs /= coeffs.sum()
        a = sum(c * b for c, b in zip(coeffs, basic))
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_vec = val, a

    return best_vec


# ----------------------------------------------------------------------
# Sub‑gradient polishing (projected)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.15) -> np.ndarray:
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of each active entry: 2 * a shifted
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / math.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Aggressive stochastic hill‑climb
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_iter: int = 800_000) -> np.ndarray:
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.02
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.6)
            no_improve = 0
        else:
            step_scale *= 0.995
            no_improve += 1

        it += 1
        if no_improve > 150_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming based active‑set improvement
# ----------------------------------------------------------------------
def lp_active_improvement(f: np.ndarray, step: float,
                          time_deadline: float) -> np.ndarray:
    """
    One iteration of the LP–based improvement:
    maximise Σ g_i subject to g≥0 and (f*g)(k) ≤ M for all k
    where k ranges over the currently active constraints (those
    attaining the maximal convolution value M).
    Returns the convex combination (1‑step)·f + step·g_normalised.
    """
    n = f.shape[0]
    conv = full_convolution(f)
    M = conv.max()

    # active indices (allow a tiny absolute tolerance)
    eps = max(1e-12, M * 1e-12)
    active = np.where(conv >= M - eps)[0]

    # Build the constraint matrix only for active rows
    rows = []
    for k in active:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1)
        row = np.zeros(n, dtype=np.float64)
        row[idx] = f[k - idx]
        rows.append(row)

    if not rows:                 # fallback – should never happen
        return f

    A = np.vstack(rows)
    b = np.full(A.shape[0], M)

    # Linear programme: maximise sum(g)  <=> minimise -sum(g)
    try:
        from scipy.optimize import linprog
        c = -np.ones(n, dtype=np.float64)          # minimise negative sum
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
    except Exception as e:
        print("[LP] linprog import/solve failed:", e)
        return f

    if not res.success:
        # In rare cases the active set may be infeasible; fall back to f
        print("[LP] linprog not successful:", res.message)
        return f

    g = res.x
    s = g.sum()
    if s <= 0:
        return f
    g = g / s                     # project back onto the simplex (sum = 1)

    # convex blend
    f_new = (1.0 - step) * f + step * g
    f_new = project_to_simplex(f_new)
    return f_new

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  The routine follows the stages described in the
    module docstring and respects the given time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # -----------------------------------------------------------------
    # 0. Optional fallback (global variable may contain a good start)
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Quick parametric sweep (≈ 150 s)
    # -----------------------------------------------------------------
    n = 2048
    sweep_deadline = deadline - 300.0
    if time.time() < sweep_deadline:
        a0 = parametric_sweep(n, sweep_deadline)
        if a0 is not None:
            val = compute_score(a0)
            print(f"[parametric sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a0.tolist()

    # -----------------------------------------------------------------
    # 2. If we still have nothing, initialise a random simple vector
    # -----------------------------------------------------------------
    if best_seq is None:
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        best_seq = a_rand.tolist()
        best_val = compute_score(np.array(best_seq))
        print(f"[rand init] score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. LP‑based active‑set improvement
    # -----------------------------------------------------------------
    lp_deadline = deadline - 120.0
    if time.time() < lp_deadline:
        f = np.array(best_seq, dtype=np.float64)
        best_f = f.copy()
        best_score = compute_score(f)
        step = 0.30                     # initial convex‑combination weight
        it = 0
        while time.time() < lp_deadline and it < 40:
            f_new = lp_active_improvement(f, step, lp_deadline)
            score = compute_score(f_new)
            if score < best_score:
                best_score = score
                best_f = f_new.copy()
                f = f_new
                step = min(step * 1.1, 0.6)   # become bolder after improvement
                print(f"[LP] iter {it:02d} improved score = {score:.6f}")
            else:
                # no improvement – shrink step and try again
                step *= 0.85
                f = f_new                     # still move a bit
            it += 1
        best_seq = best_f.tolist()
        best_val = compute_score(best_f)
        print(f"[LP] final best score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Sub‑gradient polishing (bulk of remaining time)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 40.0
    if time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=np.float64)
        a_polished = subgradient_polish(a0, polish_deadline, step0=0.12)
        val = compute_score(a_polished)
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 5. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=np.float64)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # 6. Safeguard – fall back to uniform if still empty
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 7. Final official evaluation
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
