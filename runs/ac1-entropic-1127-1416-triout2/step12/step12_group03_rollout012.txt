"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

under the simplex constraint.  The driver
* picks a reasonably large problem size (default 8192),
* builds a good analytic start (the discrete arcsine family) and
  scans a few simple parametric families,
* refines the candidate with a cascade of optimisation phases:
    – exponentiated‑gradient (mirror‑descent) on the exact max‑convolution,
    – a custom “max‑diff” descent that simultaneously lowers the highest
      convolution entry and raises the lowest one,
    – Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate with a
      β‑schedule up to 2·10⁴,
    – cheap projected sub‑gradient steps,
    – a short simulated‑annealing hill‑climb,
* symmetrises the final vector (the problem is invariant under reversal) and
* returns the best sequence found before the wall‑clock deadline (≈1000 s).

All heavy work is done with FFT‑based convolution (O(n log n) per iteration)
and simple vectorised NumPy operations; the code respects the time budget
and avoids closures or lambda‑expressions as required by the execution rules.
"""

# ------------------------------------------------------------
# Imports and low‑level utilities
# ------------------------------------------------------------
import time
import math
import numpy as np

# ------------------------------------------------------------
# Convolution helpers
# ------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the harness when the vector is normalised
    to sum = 1:   2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic families (used for quick seeding)
# ------------------------------------------------------------
def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑½‑i))^{‑α}  (α=0.5 gives the discrete arcsine law)."""
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 600_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true max‑convolution objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        if k % 2 == 0:
            mid = k // 2
            if mid < n:
                g[mid] = a[mid]          # correct the diagonal term

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def maxdiff_optimize(start_a: np.ndarray,
                    deadline: float,
                    step0: float = 0.5,
                    max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Custom descent that reduces the current maximum of the convolution
    while raising its minimum.  For each iteration we compute the
    gradients of the maximal and minimal convolution entries and move
    a in the direction *grad_min – grad_max*.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    idx = np.arange(n)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        l = int(np.argmin(conv))

        # direction = grad_l – grad_k
        d = np.zeros_like(a)

        # contribution from the max index (to be subtracted)
        j_k = k - idx
        mask_k = (j_k >= 0) & (j_k < n)
        d[mask_k] -= a[j_k[mask_k]]
        if k % 2 == 0:
            mid = k // 2
            if mid < n:
                d[mid] -= a[mid]          # extra diagonal term

        # contribution from the min index (to be added)
        j_l = l - idx
        mask_l = (j_l >= 0) & (j_l < n)
        d[mask_l] += a[j_l[mask_l]]
        if l % 2 == 0:
            mid = l // 2
            if mid < n:
                d[mid] += a[mid]          # extra diagonal term

        step = step0 / math.sqrt(t)
        a = a + step * d
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] max‑diff improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 4.0,
                         beta_end: float = 20000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    with a geometric schedule for β.
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = np.exp(z) / np.exp(z).sum()
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (log‑linear)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = np.exp(z)
        a /= a.sum()

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights (stable)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth‑max w.r.t. a (via convolution)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through the softmax parametrisation a = softmax(z)
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = np.exp(z)
            a_tmp /= a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = np.exp(z)
    a_fin /= a_fin.sum()
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    idx = np.arange(n)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        if k % 2 == 0:
            mid = k // 2
            if mid < n:
                g[mid] = a[mid]   # diagonal correction (as in the original code)

        step = step0 / math.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        best_score = sc

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Very light simulated‑annealing walk on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided ``evaluate_sequence`` function.  The routine respects the
    wall‑clock budget, uses a large problem size (default 8192), starts
    from an analytically‑motivated arcsine shape, explores a few random
    seeds, and then refines the best candidate with a chain of
    optimisation phases described above.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback (ignore if it would limit the problem size)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose problem size – aim for a fairly large n.
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    target_n = 8192
    n = target_n if fallback_len < target_n else fallback_len
    print(f"[{time.time():.1f}] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Quick analytic sweep over a few families (incl. arcsine)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        alphas = np.linspace(0.3, 1.0, 15)
        for alpha in alphas:
            if time.time() >= deadline - 5.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (α={alpha:.3f}) ⇒ {best_score:.6f}")

        # a few random seeds (monotone and exponential families)
        for _ in range(5):
            if time.time() >= deadline - 5.0:
                break
            vec = random_monotone_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (random monotone) ⇒ {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Normalise the working vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)
    print(f"[{time.time():.1f}] after initial scan best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Allocate time slices for the optimisation phases.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # generous split – can be adjusted on the fly
    eg_frac     = 0.15
    maxdiff_frac= 0.25
    adam_frac   = 0.25
    subg_frac   = 0.20
    sa_frac     = max(0.0, 1.0 - (eg_frac + maxdiff_frac + adam_frac + subg_frac))

    eg_deadline      = time.time() + eg_frac * remaining
    maxdiff_deadline = time.time() + maxdiff_frac * remaining
    adam_deadline    = time.time() + adam_frac * remaining
    subg_deadline    = time.time() + subg_frac * remaining
    sa_deadline      = time.time() + sa_frac * remaining

    # ------------------------------------------------------------------
    # 5.  Exponentiated‑gradient phase.
    # ------------------------------------------------------------------
    if time.time() < eg_deadline:
        print(f"[{time.time():.1f}] EG phase (≈{eg_deadline - time.time():.1f}s)")
        a_eg, sc_eg = eg_optimize(a, eg_deadline,
                                 max_iters=800_000,
                                 eta0=0.6)
        if sc_eg < best_score:
            best_score = sc_eg
            a = a_eg
            print(f"[{time.time():.1f}] EG improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Max‑diff descent (push max down, min up).
    # ------------------------------------------------------------------
    if time.time() < maxdiff_deadline:
        print(f"[{time.time():.1f}] Max‑diff phase (≈{maxdiff_deadline - time.time():.1f}s)")
        a_md, sc_md = maxdiff_optimize(a, maxdiff_deadline,
                                       step0=0.5,
                                       max_iters=1_500_000)
        if sc_md < best_score:
            best_score = sc_md
            a = a_md
            print(f"[{time.time():.1f}] Max‑diff improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Adam optimisation of the smooth‑max surrogate.
    # ------------------------------------------------------------------
    if time.time() < adam_deadline:
        print(f"[{time.time():.1f}] Adam phase (≈{adam_deadline - time.time():.1f}s)")
        a_adam, sc_adam = adam_smooth_optimize(a,
                                                adam_deadline,
                                                max_iters=500_000,
                                                beta_start=4.0,
                                                beta_end=20000.0,
                                                lr=0.018)
        if sc_adam < best_score:
            best_score = sc_adam
            a = a_adam
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Projected sub‑gradient descent (cheap many steps).
    # ------------------------------------------------------------------
    if time.time() < subg_deadline:
        print(f"[{time.time():.1f}] Sub‑gradient phase (≈{subg_deadline - time.time():.1f}s)")
        a_sub, sc_sub = subgrad_optimize(a,
                                         subg_deadline,
                                         max_iters=5_000_000,
                                         step0=0.07)
        if sc_sub < best_score:
            best_score = sc_sub
            a = a_sub
            print(f"[{time.time():.1f}] Sub‑gradient improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Optional symmetrisation (the problem is invariant under reversal).
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] Symmetrisation improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10.  Final simulated‑annealing hill‑climb (short).
    # ------------------------------------------------------------------
    if time.time() < sa_deadline:
        print(f"[{time.time():.1f}] SA hill‑climb (≈{sa_deadline - time.time():.1f}s)")
        a_sa, sc_sa = hill_climb_sa(a,
                                    sa_deadline,
                                    max_steps=2_000_000)
        if sc_sa < best_score:
            best_score = sc_sa
            a = a_sa
            print(f"[{time.time():.1f}] SA improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 11.  Final evaluation through the provided harness.
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
