"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

The search proceeds in several stages:

1. **Coarse analytic families** – a handful of low‑dimensional families
   (product, asymmetric product, beta‑distribution, mixtures…) are
   evaluated on a modest support size (n=16384).  The best parameters are
   recorded.

2. **Local refinement** – the promising families are refined with
   SciPy’s derivative‑free optimisers (golden‑section for 1‑D, Nelder‑Mead
   for 2‑D).  The refined vectors replace the previous best.

3. **Smooth‑max Adam optimisation** – the current best vector is used as
   a warm‑start for a Adam optimiser applied to the log‑sum‑exp surrogate
   of the max‑convolution.  The temperature β is annealed from 10 to a
   large final value (≈2·10⁴); the iterate is periodically symmetrised
   and projected onto the probability simplex.

4. **Projected sub‑gradient descent** – a short exact max‑sub‑gradient
   descent (∼10⁵ steps) gives a final polishing.

All stages respect the 1000 s wall‑clock budget.  The final vector is
returned as a plain list of Python floats (sum≈1, all entries ≥0)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score_normed(a: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum=1:
        score = 2·n·max(a*a) .
    """
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    a = a / s
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a sum‑1 vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape w_i ∝ exp(‑γ·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def product_expU_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    """Product family multiplied by an exponential‑U factor."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod - gamma * dist)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def asymmetric_product_family(n: int, beta_left: float, beta_right: float) -> np.ndarray:
    """Two‑parameter product: w_i ∝ (i+½)^{‑β₁}(n‑½‑i)^{‑β₂}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta_left * np.log(i + 0.5) -
               beta_right * np.log(n - 0.5 - i))
    w /= w.sum()
    return w


def mixed_product_uniform_family(n: int, lam: float, alpha: float) -> np.ndarray:
    """Mixture of the symmetric product family and the uniform vector."""
    prod = symmetric_product_family(n, alpha)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    w = lam * prod + (1.0 - lam) * uniform
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Beta‑distribution density on the grid {0,…,n‑1}:
        w_i ∝ x_i^{p‑1} (1‑x_i)^{q‑1},  x_i = i/(n‑1).
    """
    eps = 1e-12
    x = np.arange(n, dtype=np.float64) / (n - 1)
    x = np.clip(x, eps, 1.0 - eps)
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    w = np.exp(logw)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w

def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 20) -> tuple[float, float]:
    """1‑D golden‑section search on a closed interval."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - invphi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + invphi * (b - a)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 10.0,
                         beta_end: float = 20000.0,
                         lr: float = 0.012,
                         sym_interval: int = 1500) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # work in log‑space to keep positivity and simplex constraint
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score_normed(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # occasional symmetrisation (helps maintain the natural symmetry)
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        # smooth max of the convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth max w.r.t. a
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # chain‑rule through the soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # keep track of the best iterate
        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score_normed(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score_normed(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 120_000,
                     step0: float = 0.045) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the exact max‑convolution objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score_normed(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the k‑th convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score_normed(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score_normed(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

    """

    deadline = start_time + budget_s - 2.0   # a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback from a previous run (height_sequence_1 may exist)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. choose a working length (a power of two for fast FFT)
    # ------------------------------------------------------------------
    n = 16384          # 2¹⁴ – a good compromise between resolution and speed

    # ------------------------------------------------------------------
    # 2. baseline uniform
    # ------------------------------------------------------------------
    vec = np.full(n, 1.0 / n, dtype=np.float64)
    sc = compute_score_normed(vec)
    if sc < best_score:
        best_score = sc
        best_seq = vec.tolist()
        print(f"[{time.time():.1f}] baseline uniform = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. scan a collection of low‑dimensional families
    # --------------------------------------------------------------
    print(f"[{time.time():.1f}] scanning analytic families (n={n})")
    family_records = {}

    # ---- symmetric product family -----------------------------------
    best_alpha = None
    best_alpha_sc = float("inf")
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score_normed(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.3f}) = {best_score:.6f}")
        if sc < best_alpha_sc:
            best_alpha_sc = sc
            best_alpha = alpha
    family_records["sym_product"] = (best_alpha, best_alpha_sc)

    # ---- left‑heavy power family ------------------------------------
    best_beta = None
    best_beta_sc = float("inf")
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score_normed(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")
        if sc < best_beta_sc:
            best_beta_sc = sc
            best_beta = beta
    family_records["left_power"] = (best_beta, best_beta_sc)

    # ---- exponential‑U family ---------------------------------------
    best_gamma = None
    best_gamma_sc = float("inf")
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score_normed(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")
        if sc < best_gamma_sc:
            best_gamma_sc = sc
            best_gamma = gamma
    family_records["exp_U"] = (best_gamma, best_gamma_sc)

    # ---- mixed uniform + exponential‑U ------------------------------
    best_mix = None
    best_mix_sc = float("inf")
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score_normed(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")
            if sc < best_mix_sc:
                best_mix_sc = sc
                best_mix = (lam, gamma)
    family_records["mix_U"] = (best_mix, best_mix_sc)

    # ---- mixed product + uniform ------------------------------------
    best_mix_prod = None
    best_mix_prod_sc = float("inf")
    for lam in np.linspace(0.0, 1.0, 11):
        for alpha in np.linspace(0.0, 12.0, 61):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_product_uniform_family(n, lam, alpha)
            sc = compute_score_normed(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix‑prod λ={lam:.2f},α={alpha:.2f}) = {best_score:.6f}")
            if sc < best_mix_prod_sc:
                best_mix_prod_sc = sc
                best_mix_prod = (lam, alpha)
    family_records["mix_prod"] = (best_mix_prod, best_mix_prod_sc)

    # ---- asymmetric product (two parameters) -----------------------
    best_asym = None
    best_asym_sc = float("inf")
    beta_grid = np.linspace(0.0, 12.0, 13)
    for b1 in beta_grid:
        for b2 in beta_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = asymmetric_product_family(n, b1, b2)
            sc = compute_score_normed(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (asym β₁={b1:.2f},β₂={b2:.2f}) = {best_score:.6f}")
            if sc < best_asym_sc:
                best_asym_sc = sc
                best_asym = (b1, b2)
    family_records["asym_product"] = (best_asym, best_asym_sc)

    # ---- product × exponential‑U ------------------------------------
    best_pe = None
    best_pe_sc = float("inf")
    a_grid = np.linspace(0.0, 12.0, 13)
    g_grid = np.linspace(0.0, 6.0, 7)
    for a in a_grid:
        for g in g_grid:
            if time.time() >= deadline - 30.0:
                break
            vec = product_expU_family(n, a, g)
            sc = compute_score_normed(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (prod‑exp α={a:.2f},γ={g:.2f}) = {best_score:.6f}")
            if sc < best_pe_sc:
                best_pe_sc = sc
                best_pe = (a, g)
    family_records["prod_expU"] = (best_pe, best_pe_sc)

    # ---- symmetric beta family (p = q) -------------------------------
    best_beta_sym = None
    best_beta_sym_sc = float("inf")
    for p in np.linspace(0.2, 1.5, 27):   # dense near 1 to explore arcsine‑like shapes
        if time.time() >= deadline - 30.0:
            break
        vec = beta_family(n, p, p)
        sc = compute_score_normed(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (beta p={p:.3f}) = {best_score:.6f}")
        if sc < best_beta_sym_sc:
            best_beta_sym_sc = sc
            best_beta_sym = p
    family_records["beta_sym"] = (best_beta_sym, best_beta_sym_sc)

    # ------------------------------------------------------------------
    # 4. Local refinement of promising families (SciPy if present)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import minimize, minimize_scalar
        scipy_ok = True
    except Exception:
        scipy_ok = False

    if scipy_ok:
        # --- symmetric product (1‑D) ---------------------------------
        if family_records.get("sym_product")[0] is not None:
            a0 = family_records["sym_product"][0]
            lo, hi = max(0.0, a0 - 0.6), min(12.0, a0 + 0.6)

            def score_alpha(a):
                return compute_score_normed(symmetric_product_family(n, a))

            a_opt, a_sc = golden_section_optimize(score_alpha, lo, hi, max_iter=25)
            vec = symmetric_product_family(n, a_opt)
            sc = compute_score_normed(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined sym‑α={a_opt:.6f} → {best_score:.6f}")

        # --- mixed product + uniform (2‑D) --------------------------
        if family_records.get("mix_prod")[0] is not None:
            lam0, alp0 = family_records["mix_prod"][0]

            def score_mix(p):
                lam = np.clip(p[0], 0.0, 1.0)
                alp = np.clip(p[1], 0.0, 12.0)
                return compute_score_normed(mixed_product_uniform_family(n, lam, alp))

            res = minimize(score_mix,
                           x0=np.array([lam0, alp0]),
                           method='Nelder-Mead',
                           options={'maxiter': 250, 'xatol': 1e-5, 'fatol': 1e-5})
            lam_opt, alp_opt = np.clip(res.x, [0.0, 0.0], [1.0, 12.0])
            vec = mixed_product_uniform_family(n, lam_opt, alp_opt)
            sc = compute_score_normed(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined mix‑prod λ={lam_opt:.6f},α={alp_opt:.6f} → {best_score:.6f}")

        # --- product × exponential‑U (2‑D) -------------------------
        if family_records.get("prod_expU")[0] is not None:
            a0, g0 = family_records["prod_expU"][0]

            def score_pe(p):
                a = np.clip(p[0], 0.0, 12.0)
                g = np.clip(p[1], 0.0, 6.0)
                return compute_score_normed(product_expU_family(n, a, g))

            res = minimize(score_pe,
                           x0=np.array([a0, g0]),
                           method='Nelder-Mead',
                           options={'maxiter': 300, 'xatol': 1e-5, 'fatol': 1e-5})
            a_opt, g_opt = np.clip(res.x, [0.0, 0.0], [12.0, 6.0])
            vec = product_expU_family(n, a_opt, g_opt)
            sc = compute_score_normed(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined prod‑exp α={a_opt:.6f},γ={g_opt:.6f} → {best_score:.6f}")

        # --- asymmetric product (2‑D) --------------------------------
        if family_records.get("asym_product")[0] is not None:
            b1_0, b2_0 = family_records["asym_product"][0]

            def score_asym(p):
                b1 = np.clip(p[0], 0.0, 12.0)
                b2 = np.clip(p[1], 0.0, 12.0)
                return compute_score_normed(asymmetric_product_family(n, b1, b2))

            res = minimize(score_asym,
                           x0=np.array([b1_0, b2_0]),
                           method='Nelder-Mead',
                           options={'maxiter': 250, 'xatol': 1e-5, 'fatol': 1e-5})
            b1_opt, b2_opt = np.clip(res.x, 0.0, 12.0)
            vec = asymmetric_product_family(n, b1_opt, b2_opt)
            sc = compute_score_normed(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined asym β₁={b1_opt:.6f},β₂={b2_opt:.6f} → {best_score:.6f}")

        # --- symmetric beta (1‑D) ------------------------------------
        if family_records.get("beta_sym")[0] is not None:
            p0 = family_records["beta_sym"][0]
            lo, hi = max(0.01, p0 - 0.4), min(2.0, p0 + 0.4)

            def score_beta(p):
                return compute_score_normed(beta_family(n, p, p))

            p_opt, p_sc = golden_section_optimize(score_beta, lo, hi, max_iter=25)
            vec = beta_family(n, p_opt, p_opt)
            sc = compute_score_normed(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] refined beta p={p_opt:.6f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. First Adam (smooth‑max) refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] first Adam (smooth max) refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 150.0,
            max_iters=300_000,
            beta_start=10.0,
            beta_end=15000.0,
            lr=0.012,
            sym_interval=1800,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after first Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = subgrad_optimize(
            a_start,
            deadline - 20.0,
            max_iters=120_000,
            step0=0.045,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Optional symmetrisation test
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score_normed(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation via the official wrapper
    # ------------------------------------------------------------------
    if best_seq is None:
        # a tiny safe fallback (uniform 256‑vector)
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
