"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  The routine combines:

* a quick scan of several analytic families (product, exponential‑U,
  mixtures, …) on a moderate support size,
* a low‑dimensional “block” projected sub‑gradient descent (the full
  vector is forced to be symmetric and piecewise‑constant, which gives
  many cheap iterations and a finer convergence),
* a final Adam optimisation on the exact smooth‑max surrogate,
* an optional short simulated‑annealing hill‑climb.

The total wall‑clock budget is respected – the search stops
automatically when the deadline is reached and returns the best vector
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, scoring, projections)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # implementation from https://arxiv.org/abs/1309.1541
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for a quick coarse scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric product family a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def product_expU_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    """Product family multiplied by an exponential‑U factor."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod - gamma * dist)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def asymmetric_product_family(n: int, beta_left: float, beta_right: float) -> np.ndarray:
    """Two‑parameter product a_i ∝ (i+½)^{‑β₁}(n‑½‑i)^{‑β₂}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta_left * np.log(i + 0.5) - beta_right * np.log(n - 0.5 - i))
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate (as in the reference code)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 20.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.012,
                         sym_interval: int = 1500) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max approximation of the max‑convolution."""
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal smooth‑max temperature
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                 # soft‑max weights

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)    # gradient of smooth‑max
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a          # chain rule for log‑param

        # Adam update on z
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 5000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score

# Block‑wise projected sub‑gradient descent (new component)
# ----------------------------------------------------------------------


def block_subgrad_optimize(b_start: np.ndarray,
                           n: int,
                           block_sz: int,
                           deadline: float,
                           max_iters: int = 5_000_000,
                           step0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent where the full vector is forced to be symmetric
    and piecewise‑constant (blocks of equal height).
    """
    half = n // 2
    B = half // block_sz

    # pre‑compute block index for each position (full length n)
    left_block_idx = np.repeat(np.arange(B), block_sz)
    right_block_idx = left_block_idx[::-1]
    block_idx_arr = np.concatenate([left_block_idx, right_block_idx])

    def blocks_to_a(b):
        """Expand block vector b (size B) to the full symmetric array."""
        left = np.repeat(b, block_sz)
        right = np.repeat(b[::-1], block_sz)
        return np.concatenate([left, right])

    # initialise
    b = b_start.copy()
    a = blocks_to_a(b)
    a = project_to_simplex(a)        # ensure a is a probability vector
    # re‑average to keep exact block structure after projection
    left = a[:half]
    b = left.reshape(B, block_sz).mean(axis=1)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and locate the active index
        conv = convolution_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of max_{k} (a*a)_k  :  g_i = 2 a_{k-i} (if index in range)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # collapse the gradient onto blocks (sum of components belonging to each block)
        G = np.bincount(block_idx_arr, weights=g, minlength=B)

        # diminishing step
        step = step0 / math.sqrt(it)
        b -= step * G
        b = np.maximum(b, 0.0)                 # keep non‑negative

        # rebuild full vector, project onto simplex, then re‑average blocks
        a = blocks_to_a(b)
        a = project_to_simplex(a)
        left = a[:half]
        b = left.reshape(B, block_sz).mean(axis=1)

        # occasional progress report
        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] block sub‑grad it {it}  new best {best_score:.6f}")

    return best_a, best_score

# Simple simulated‑annealing hill‑climb (kept from reference)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Tiny‑scale random walk with a slowly decreasing temperature."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
