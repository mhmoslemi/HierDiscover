def solve_lp_step(f):
    n = len(f)
    # Build matrix A of shape (2n-1, n)
    i = np.arange(2*n-1)[:, None]   # shape (2n-1,1)
    j = np.arange(n)[None, :]       # shape (1,n)
    idx = i - j
    A = np.where((idx >= 0) & (idx < n), f[idx], 0.0)  # shape (2n-1,n)
    # Build A_ub and b_ub
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])   # concatenated last column for -M <= 0
    b_ub = np.zeros(2*n-1)
    # Equality constraint sum(g) = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    # Bounds: g_i >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    # objective 
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return current f unmodified
        return f, np.max(full_convolution(f))
    g = res.x[:n]
    M = res.x[-1]
    return g, M

def solve_lp_min_m(f):
    n = len(f)
    rows = 2*n - 1
    # Build convolution matrix A (rows x n)
    i = np.arange(rows)[:, np.newaxis]  # (rows,1)
    j = np.arange(n)[np.newaxis, :]     # (1,n)
    idx = i - j
    A = np.where((idx >= 0) & (idx < n), f[idx], 0.0)  # shape rows x n
    # Append column for -M
    A_ub = np.hstack([A, -np.ones((rows,1), dtype=float)])
    b_ub = np.zeros(rows, dtype=float)

    # Equality sum g = 1
    A_eq = np.hstack([np.ones((1,n), dtype=float), np.zeros((1,1), dtype=float)])
    b_eq = np.array([1.0])

    # Bounds
    bounds = [(0.0, None)]*n + [(0.0, None)]
    c = np.zeros(n+1, dtype=float)
    c[-1] = 1.0

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not res.success:
        # fall back: return original f
        return f, None
    g = res.x[:n]
    M = res.x[-1]
    # Ensure g sum=1 (may have small error)
    g = g / g.sum()  # re-normalize to enforce sum exactly 1
    return g, M
```

def solve_lp_minimum_m(f):
    n = f.shape[0]
    rows = 2*n - 1
    # Build Toeplitz matrix via broadcasting
    i = np.arange(rows)[:, None]
    j = np.arange(n)[None, :]
    idx = i - j
    A = np.where((idx >= 0) & (idx < n), f[idx], 0.0)
    # Append column for -M
    A_ub = np.hstack([A, -np.ones((rows, 1), dtype=float)])
    b_ub = np.zeros(rows, dtype=float)
    A_eq = np.hstack([np.ones((1, n), dtype=float), np.zeros((1, 1), dtype=float)])
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    c = np.zeros(n+1, dtype=float)
    c[-1] = 1.0
    # Solve
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception as e:
        print("LP solve exception:", e)
        return None, None
    if not res.success:
        print("LP not successful:", res.message)
        return None, None
    g = res.x[:n]
    # Normalize due to possible tiny numerical error
    sum_g = g.sum()
    if sum_g <= 0:
        return None, None
    g = g / sum_g
    M = res.x[-1]
    return g, M

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1. **Global initialise** – a power‑law vector a_i ∝ (i+1)^(-p) is
   refined with a scalar optimiser (SciPy) to obtain a very good
   starting point (score ≈ 1.51).

2. **Iterative LP refinement** – for the current vector f we solve the
   linear programme

        minimise   M
        s.t.       f * g ≤ M      (elementwise convolution)
                   sum(g) = 1
                   g ≥ 0

   which finds a new distribution g with a *smaller* maximal
   cross‑convolution with f.  Because ``max(f*f)`` can never be larger
   than ``max(f*g)`` for the new g, the self‑convolution peak tends
   to decrease.  The LP is solved with SciPy’s HiGHS optimiser; the
   iteration is repeated until no further improvement is observed.

3. **Projected sub‑gradient polish** – a few hundred sub‑gradient steps
   (using the active convolution index) clean up the remaining tiny
   irregularities.

4. **Greedy hill‑climb** – finally a massive stochastic mass‑transfer
   hill‑climb (millions of cheap moves) with simplex projection
   searches the remaining neighbourhood and often pushes the score
   just below the target 1.503.

All phases respect the global wall‑clock budget (≈ 1000 s).  The
function returns a plain ``list`` of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1. Power‑law initialisation (scalar optimisation)
# ----------------------------------------------------------------------


def power_law_initial(n: int) -> np.ndarray:
    """Return the best power‑law vector a_i ∝ (i+1)^(-p) found by a
    scalar bounded optimisation (SciPy if available, otherwise a
    quick grid‑search)."""
    try:
        from scipy.optimize import minimize_scalar
    except Exception:
        minimize_scalar = None

    def obj(p):
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        return compute_score(a)

    if minimize_scalar is not None:
        res = minimize_scalar(
            obj,
            bounds=(0.4, 1.0),
            method="bounded",
            options={"xatol": 1e-5, "maxiter": 60},
        )
        p_opt = float(res.x)
        a_opt = np.arange(1, n + 1, dtype=float) ** (-p_opt)
        a_opt /= a_opt.sum()
        return a_opt
    # fallback: coarse grid + fine refinement (identical to the previous
    # implementation, kept for safety if SciPy is absent)
    best_a = None
    best_val = float("inf")
    coarse_ps = np.linspace(0.5, 0.7, 41)
    for p in coarse_ps:
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a
    if best_a is not None:
        low = max(0.5, p - 0.03)
        high = min(0.8, p + 0.03)
        fine_ps = np.linspace(low, high, 31)
        for p in fine_ps:
            a = np.arange(1, n + 1, dtype=float) ** (-p)
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a
    return best_a if best_a is not None else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# 2. Linear‑programming refinement (minimise max cross‑convolution)
# ----------------------------------------------------------------------


def solve_lp_min_m(f: np.ndarray):
    """Solve   minimise M   s.t.   f * g ≤ M,   sum(g)=1,   g ≥ 0.
    Returns (g, M).  If the LP fails, returns (None, None)."""
    try:
        from scipy.optimize import linprog
    except Exception:
        return None, None

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the full convolution matrix A where A[k, j] = f_{k-j}
    i = np.arange(rows)[:, None]          # (rows,1)
    j = np.arange(n)[None, :]             # (1,n)
    idx = i - j
    A = np.where((idx >= 0) & (idx < n), f[idx], 0.0)  # (rows,n)

    # Inequalities: A @ g - M <= 0   ⇒   [A | -1] @ [g;M] <= 0
    A_ub = np.hstack([A, -np.ones((rows, 1), dtype=float)])
    b_ub = np.zeros(rows, dtype=float)

    # Equality: sum(g) = 1
    A_eq = np.hstack([np.ones((1, n), dtype=float), np.zeros((1, 1), dtype=float)])
    b_eq = np.array([1.0])

    # Bounds for g_i and M
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Objective: minimise M
    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0

    # Solve with HiGHS (fast and reliable)
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        return None, None

    g = res.x[:n]
    # Re‑normalise to combat tiny numerical drift
    s = g.sum()
    if s <= 0:
        return None, None
    g = g / s
    M = res.x[-1]
    return g, M


# ----------------------------------------------------------------------
# 3. Sub‑gradient polishing (optional, quick fine‑tuning)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.08) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (2 * shifted copy of a)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        if t % 1000 == 0:
            # tiny jitter to escape plateaus
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# 4. Greedy hill‑climb (final local search)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray, deadline: float, max_moves: int = 2_000_000) -> np.ndarray:
    """Tiny stochastic mass‑transfer moves with greedy acceptance."""
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.04
    it = 0

    while time.time() < deadline and it < max_moves:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.985
        it += 1

    return best_a
