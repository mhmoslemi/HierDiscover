"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

Algorithm
---------

0. Load a fallback candidate if the harness provides ``height_sequence_1``.
1. Quick scan of several analytic families (U‑shapes, exponential,
   power‑law, mixed) on the current length.  The scan is dense
   (many parameter values) and keeps the best.
2. Refine the best starter with three complementary local‑optimisers,
   each respecting the simplex:
     * Adam on a smooth‑max surrogate (log‑sum‑exp of the convolution)
       – β is driven up to a very large value (≈ 5000) to approach the true max.
     * Projected sub‑gradient descent on the exact max‑objective.
     * An LP‑based “active‑set” refinement: maximise Σ g_j under
       (f * g)_k ≤ M (M = max f*f) and mix the resulting g into f.
       This step is repeated many times (≈ 25) with a modest mixing weight.
3. Symmetrise the vector (f←(f+f[::-1])/2) if it improves the score.
4. Final stochastic hill‑climb (simulated‑annealing style) that
   performs many small mass‑transfer moves; the temperature is cooled
   linearly to near‑zero.
5. If time permits a few extra random restarts (perturb the current best
   and repeat steps 2‑4) are launched.
6. The best sequence found is returned as a plain Python list.

All phases constantly check the global wall‑clock deadline, so the
function always finishes before the 1000 s budget expires.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a) .
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shape ≈ ((i+½)(n‑½‑i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing ≈ (i+½)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay from the two ends."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex combination of uniform and the exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing sequence (sorted powers of uniform draws)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max optimiser (Adam on the log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ exp(β·(a*a)_k)
    with Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))          # stabilised
        w = exp_w / exp_w.sum()

        # gradient wrt a of the smooth‑max
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 800 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheap per iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 800_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑objective.
    Uses step size step0 / sqrt(t) and Euclidean projection onto the simplex.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of (a*a)_k  (see derivation in the original script)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal correction (i=j) where factor should be 1 instead of 2
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (state‑of‑the‑art step)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.15) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M   for all k,   g ≥ 0,
    then return the convex mixture
        f_new = (1‑t)·f + t·g_norm
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A so that A @ g = f * g (full convolution)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)           # maximise Σ g → minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # High‑s level solver – small time limit to keep the overall budget safe
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    Temperature is linearly cooled to almost zero.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Small deterministic local fine‑tuning (tiny mass swaps)
# ----------------------------------------------------------------------


def tiny_local_search(a: np.ndarray,
                      deadline: float,
                      trials: int = 300_000,
                      eps: float = 1e-4) -> np.ndarray:
    """
    Very cheap deterministic perturbations: move a tiny amount ε from one
    coordinate to another if it reduces the exact objective.
    """
    n = a.size
    best = a.copy()
    best_sc = compute_score(best)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(n, size=2)
        if i == j or best[i] < eps:
            continue
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps
        # renormalisation is not needed: sum unchanged and both stay ≥0
        sc = compute_score(cand)
        if sc + 1e-12 < best_sc:
            best = cand
            best_sc = sc
    return best


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function respects the global time budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0        # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Determine problem size (use length of fallback or default 2048)
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # ------------------------------------------------------------------
    # 2. Quick analytic / random scan (dense)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] quick scan (n={n})")
        # Parameter grids (more fine‑grained than the previous run)
        alphas = np.linspace(0.0, 30.0, 61)      # 0 … 30
        betas = np.linspace(0.0, 30.0, 61)
        gammas = np.linspace(0.0, 12.0, 61)
        lams = np.linspace(0.0, 1.0, 21)

        for alpha in alphas:
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best sym‑α={alpha:.2f} → {best_score:.6f}")

        for beta in betas:
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best left‑β={beta:.2f} → {best_score:.6f}")

        for gamma in gammas:
            if time.time() >= deadline - 30.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best exp‑γ={gamma:.2f} → {best_score:.6f}")

        for lam in lams:
            if time.time() >= deadline - 30.0:
                break
            # reuse one of the gammas from above (mid‑range)
            gamma = 4.0
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best mix‑λ={lam:.2f} → {best_score:.6f}")

        # Random draws (both monotone and U‑shape)
        for _ in range(200):
            if time.time() >= deadline - 30.0:
                break
            if np.random.rand() < 0.5:
                vec = random_monotone_start(n)
            else:
                vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best random → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise the working vector (∑ = 1)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 3. Adam‑based smooth‑max refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 180.0:
        print(f"[{time.time():.1f}] Adam refinement start")
        a, sc = adam_smooth_optimize(
            a,
            deadline - 60.0,
            max_iters=500_000,
            beta_start=4.0,
            beta_end=5000.0,
            lr=0.018,
        )
        if sc < best_score:
            best_score = sc
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent (still cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] sub‑gradient descent start")
        a, sc = subgrad_optimize(
            a,
            deadline - 30.0,
            max_iters=1_200_000,
            step0=0.07,
        )
        if sc < best_score:
            best_score = sc
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Extended LP‑based refinement (more iterations than before)
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] entering LP refinement loop")
        mix_weight = 0.15
        lp_iters = 0
        while time.time() < deadline - 20.0 and lp_iters < 25:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc_new = compute_score(a_new)
            if sc_new < best_score - 1e-7:
                a = a_new
                best_score = sc_new
                print(f"[{time.time():.1f}] LP iter {lp_iters} improved to {best_score:.6f}")
                # keep mixing weight moderate
                mix_weight = min(0.2, mix_weight * 1.05)
            else:
                # become more conservative
                mix_weight = max(0.02, mix_weight * 0.7)
            lp_iters += 1

    # ------------------------------------------------------------------
    # 6. Symmetrisation test
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        a = a_sym
        best_score = sc_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final stochastic hill‑climb (simulated‑annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) start")
        a, sc = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc < best_score:
            best_score = sc
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Tiny deterministic local search (very cheap, final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] tiny local search start")
        a = tiny_local_search(a, deadline, trials=400_000, eps=1e-5)
        best_score = compute_score(a)
        print(f"[{time.time():.1f}] after tiny search score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Optional extra random‑restart polishing (if still time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # perturb current best and give it a short sub‑grad + hill‑climb burst
        pert = a + np.random.normal(scale=0.02, size=n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()
        sub_dead = min(time.time() + 12.0, deadline - 5.0)
        pert, sc_sub = subgrad_optimize(pert, sub_dead, max_iters=200_000, step0=0.05)
        if time.time() >= deadline - 5.0:
            break
        sa_dead = min(time.time() + 12.0, deadline - 3.0)
        pert, sc_sa = hill_climb_sa(pert, sa_dead, max_steps=500_000)
        sc_pert = compute_score(pert)
        if sc_pert < best_score:
            a = pert
            best_score = sc_pert
            print(f"[{time.time():.1f}] extra restart improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
