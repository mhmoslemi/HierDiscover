"""
Search for a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The algorithm works in several stages:

1.  Pick a length `n` (larger when time permits – up to 8192).
2.  Generate a diverse pool of candidate vectors:
    * a uniform vector,
    * a fine grid of the analytic families already used,
    * many random Dirichlet draws with several concentration parameters.
3.  Keep the best `K` candidates (according to the exact objective
    `2·n·max(conv)`, i.e. ``compute_score``) and refine each of them
    with a long L‑BFGS‑B run on a smooth log‑sum‑exp surrogate.
    The β‑schedule is pushed up to 20000 and the iteration budget per β
    is increased.
4.  Take the current best vector and run a long exponentiated‑gradient
    (mirror sub‑gradient) descent on the true max‑convolution objective.
5.  Symmetrise the result (the optimum is often symmetric) and keep the
    better half.
6.  Finish with a low‑temperature simulated‑annealing hill‑climb that
    swaps entries or moves a tiny fraction of mass between random
    positions.
7.  Return the best sequence found as a plain Python list of floats.

All stages respect a global wall‑clock deadline (default 1000 s).  The
implementation is self‑contained, uses only NumPy (and SciPy when
available) and works on a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS‑B)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt                     # type: ignore
    _HAS_SCIPY = True
except Exception:                                     # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (aperiodic) convolution using a real FFT.
    Returns an array of length ``len(a)+len(b)-1``.
    """
    m = a.size + b.size - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation for a *probability* vector (i.e. ``sum(a)=1``):
        score = 2·n·max(a*a)
    This is identical to ``evaluate_sequence`` for such vectors.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities (Euclidean projection & soft‑max)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑shape: a_i ∝ (i/(n‑1))^α·(1‑i/(n‑1))^α."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, concentration: float) -> np.ndarray:
    """Draw a vector from a symmetric Dirichlet(α) distribution."""
    if concentration <= 0:
        # fall back to a uniform draw
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        g = np.random.gamma(shape=concentration, scale=1.0, size=n)
        a = g / g.sum()
    return a


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse probability vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]  # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate – log‑sum‑exp of the convolution
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Computes the smooth surrogate
        f_β(a) = (1/β)·log∑_k exp[β·(a*a)_k]
    and its gradient w.r.t. the log‑soft‑max parameters ``z``.
    """
    a = softmax(z)                     # a ∈ simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()

    exp_terms = np.exp(beta * (conv - maxc))   # stable
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # gradient w.r.t. a
    w = exp_terms / sum_exp                      # shape (2n‑1,)
    a_rev = a[::-1]
    # (a_rev * w) convolution gives the needed Jacobian component
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1: 2 * n - 1]   # slice to size n

    # chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray,
                  deadline: float,
                  beta_list=(200.0, 800.0, 2000.0, 8000.0, 20000.0),
                  max_iter_per_beta=1500) -> tuple[np.ndarray, float]:
    """
    Refine ``start_a`` using L‑BFGS‑B on the smoothed log‑sum‑exp surrogate.
    Returns the best vector encountered and its exact ``compute_score`` value.
    """
    if not _HAS_SCIPY:                      # pragma: no cover
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)                # initialise log‑soft‑max

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break

        # adapt iteration budget to remaining time (very coarse)
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 12))

        def func(z_vec):
            f, g = _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])
            return f, g

        try:
            opt_res = _opt.fmin_l_bfgs_b(
                func,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:               # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved → {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(start_a: np.ndarray,
                           deadline: float,
                           step0: float = 0.6,
                           check_interval: int = 1000) -> tuple[np.ndarray, float]:
    """
    Exponentiated (multiplicative) sub‑gradient descent on
        F(a) = 2·n·max(a*a)
    Keeps the iterate on the simplex by renormalisation.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)          # multiplicative update

        # re‑project onto the simplex (entropy projection)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} → {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000,
                  init_temp_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
        - with prob 0.5 swap two entries,
        - otherwise transfer a fraction of mass from one index to another.
    The temperature decays linearly; the scale of the move adapts.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * max(0.0, 1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns the final sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Choose a length (larger when we have enough budget)
    # --------------------------------------------------------------
    if budget_s >= 950:
        n = 8192
    elif budget_s >= 800:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] Using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  Optional fallback to a known construction
    # --------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Build a pool of cheap candidates
    # --------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # analytic families – a finer grid than the previous run
    for alpha in np.linspace(0.0, 10.0, 121):
        if time.time() >= deadline - 40.0:
            break
        candidates.append(symmetric_product_family(n, alpha))
    for beta in np.linspace(0.0, 10.0, 121):
        if time.time() >= deadline - 40.0:
            break
        candidates.append(left_heavy_power_family(n, beta))
    for gamma in np.linspace(0.0, 9.0, 101):
        if time.time() >= deadline - 40.0:
            break
        candidates.append(exponential_decay_family(n, gamma))
    for alpha in np.linspace(0.0, 10.0, 101):
        if time.time() >= deadline - 40.0:
            break
        candidates.append(beta_distribution_family(n, alpha))

    # random Dirichlet draws with several concentrations
    conc_params = [0.1, 0.3, 0.7, 1.0, 2.0, 5.0]
    for conc in conc_params:
        for _ in range(12):               # 12 draws each
            if time.time() >= deadline - 40.0:
                break
            candidates.append(random_dirichlet_start(n, conc))

    # a few sparse and monotone starts (already cheap)
    for _ in range(20):
        if time.time() >= deadline - 40.0:
            break
        candidates.append(random_sparse_start(n))
        candidates.append(random_monotone_start(n))

    # --------------------------------------------------------------
    # 2.  Keep the K best cheap candidates
    # --------------------------------------------------------------
    K = 12
    scored = []
    for vec in candidates:
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new cheap best = {best_score:.6f}")
        scored.append((sc, vec))
    scored.sort(key=lambda x: x[0])
    top_vectors = [v for (_, v) in scored[:K]]

    # --------------------------------------------------------------
    # 3.  L‑BFGS refinement on the surrogate (longer runs)
    # --------------------------------------------------------------
    if time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Starting L‑BFGS refinement on top {K} candidates")
        for idx, vec in enumerate(top_vectors, 1):
            if time.time() >= deadline - 250.0:
                break
            a_ref, sc_ref = l_bfgs_refine(
                vec,
                deadline - 200.0,
                beta_list=(200.0, 800.0, 2000.0, 8000.0, 20000.0),
                max_iter_per_beta=1500,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.tolist()
                print(f"[{time.time():.1f}] L‑BFGS candidate {idx}/{K} improved → {best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Mirror sub‑gradient descent (a long run)
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0 and best_seq is not None:
        print(f"[{time.time():.1f}] Mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            step0=0.6,
            check_interval=1000,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Symmetrisation (many optimal solutions are symmetric)
    # --------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6.  Final tiny simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7.  Safeguard – if everything failed return a modest uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final official evaluation (the external harness will also call it)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
