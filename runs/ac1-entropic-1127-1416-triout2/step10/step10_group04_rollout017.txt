def subgrad_multi_optimize(init_a, time_deadline, max_iter=300_000, step0=0.2, topK=3):
    a = init_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        # find topK indices
        # we want unique indices; may include duplicates.
        if topK >= len(conv):
            idx = np.arange(len(conv))
        else:
            idx = np.argpartition(conv, -topK)[-topK:]
        w = np.zeros_like(conv)
        w[idx] = 1.0
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n-1:n-1+n]
        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
        # occasional jitter
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a = a + jitter
            a = project_to_simplex(a)
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

def generate_beta_distribution(n, alpha, beta):
    """
    Discrete approximation to Beta(alpha,beta) on [0,1].
    Returns nonnegative vector normalised to sum 1.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    # numeric stability: if alpha<1 produce large near zero, but x>0.
    return project_to_simplex(a)

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Main ideas
---------
* Build a diverse set of analytic starters (uniform, central‑binomial,
  power‑law, geometric, beta‑distribution, random Dirichlet).
* Pick the best of these by the exact objective.
* Refine the best vector with a *multi‑peak* sub‑gradient descent
  (instead of the single‑peak version used before).  The gradient is
  computed for the K largest convolution entries, which drives down the
  overall peak faster.
* Perform several short random‑restart sub‑gradient bursts to escape
  local minima.
* End with a cheap stochastic hill‑climb.
* All phases respect the wall‑clock deadline (default 1000 s) and keep
  the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT.
    Returns length len(a)+len(b)-1 array.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, normalised to the simplex."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised to the simplex."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exactly if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete approximation of a Beta(alpha,beta) density on {0,…,n‑1}.
    The points are placed at the mid‑points of the intervals.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Multi‑peak sub‑gradient optimiser (hard max)
# ----------------------------------------------------------------------


def subgrad_multi_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.3,
    topK: int = 3,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that reduces the K largest
    convolution entries simultaneously.
    """
    a = init_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # full auto‑convolution
        conv = convolution_fft(a, a)

        # indices of the K largest values (including ties)
        if topK >= conv.size:
            idx = np.arange(conv.size)
        else:
            idx = np.argpartition(conv, -topK)[-topK:]

        # weight vector w: 1 on selected positions, 0 elsewhere
        w = np.zeros_like(conv)
        w[idx] = 1.0

        # gradient of the weighted sum = 2·rev(a) * w
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]        # restriction to length n

        # simple diminishing step size
        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape flats
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimiser – Adam (optional)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed max‑convolution objective."""
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal temperature β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed objective
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps: int = 200_000) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
