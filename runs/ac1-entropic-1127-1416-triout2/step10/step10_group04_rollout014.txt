"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Improvements over the previous implementation
------------------------------------------------
* Exhaustive coarse search over several analytic families (uniform,
  central‑binomial, power‑law, geometric, Beta‑shaped, mixed power‑law/
  geometric).  The best primitive candidate is taken forward.
* Univariate golden‑section refinement of the power‑law exponent and the
  geometric ratio – cheap but often yields a few‑per‑mille gains.
* Longer, higher‑temperature Adam optimisation of the log‑sum‑exp
  smoothed max‑convolution objective (β_end enlarged to 500) with a
  dynamic iteration budget.
* A last polishing step using projected sub‑gradient descent and a
  final stochastic hill‑climb, all respecting the wall‑clock deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns length len(a)+len(b)-1 array.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, normalised to sum 1 (monotone decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exactly sum 1 if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑shaped density:
        a_i ∝ (i+1)^{α‑1} · (n‑i)^{β‑1}, normalised to sum 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1..n
    j = np.arange(n, 0, -1, dtype=np.float64)        # j = n..1
    # work in log‑space to avoid overflow
    log_a = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(j)
    a = np.exp(log_a)
    return project_to_simplex(a)


def generate_powerlaw_geometric(n: int, p: float, r: float) -> np.ndarray:
    """
    Mixed shape: a_i ∝ (i+1)^{-p}·r^{i}, normalised to sum 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p) * (r ** np.arange(n, dtype=np.float64))
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Uses a log‑sum‑exp surrogate with annealed temperature.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    phi = (1.0 + 5**0.5) / 2.0  # Golden ratio (unused here, kept for reference)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal temperature (smoothness) β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights (log‑sum‑exp trick)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed objective:
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)  # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 200_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Determine problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore
        base = np.array(height_sequence_1, dtype=np.float64)
        # sanitise the supplied vector
        base = np.maximum(base, 0.0)
        base = np.minimum(base, 1000.0)
        base = project_to_simplex(base)
        init_candidates = [base]
        print(f"Using provided height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        init_candidates = []
        print(f"No external sequence provided – using length {n}.")

    # ------------------------------------------------------------------
    # 1. Populate a wide set of analytic starters.
    # ------------------------------------------------------------------
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))        # uniform
    init_candidates.append(generate_central_binomial(n))                # central‑binomial

    # coarse power‑law sweep
    for p in np.linspace(0.2, 2.0, 30):
        init_candidates.append(generate_powerlaw(n, float(p)))

    # coarse geometric sweep
    for r in np.linspace(0.9, 0.9999, 30):
        init_candidates.append(generate_geometric(n, float(r)))

    # random Beta‑shaped families
    for _ in range(100):
        alpha = 10 ** np.random.uniform(-1, 1)   # roughly 0.1 … 10
        beta  = 10 ** np.random.uniform(-1, 1)
        init_candidates.append(generate_beta_shape(n, alpha, beta))

    # random mixtures of power‑law and geometric
    for _ in range(100):
        p = np.random.uniform(0.1, 2.0)
        r = np.random.uniform(0.85, 0.9999)
        init_candidates.append(generate_powerlaw_geometric(n, p, r))

    # ------------------------------------------------------------------
    # 2. Pick the best primitive candidate.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        if time.time() >= deadline:
            break
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        if idx % 20 == 0:
            print(f"Starter {idx:03d}: score = {val:.6f}")

    print(f"Best starter after coarse sweep: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Fine‑tune a pure power‑law exponent via golden‑section.
    # ------------------------------------------------------------------
    def score_powerlaw(p):
        return compute_score(generate_powerlaw(n, p))

    phi = (1.0 + 5**0.5) / 2.0
    p_low, p_high = 0.05, 2.5
    for _ in range(15):
        if time.time() >= deadline:
            break
        p1 = p_high - (p_high - p_low) / phi
        p2 = p_low + (p_high - p_low) / phi
        s1 = score_powerlaw(p1)
        s2 = score_powerlaw(p2)
        if s1 < s2:
            p_high = p2
            if s1 < best_val:
                best_val = s1
                best_seq = generate_powerlaw(n, p1)
        else:
            p_low = p1
            if s2 < best_val:
                best_val = s2
                best_seq = generate_powerlaw(n, p2)
    print(f"After power‑law fine‑tune: best_score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Fine‑tune a pure geometric ratio via golden‑section.
    # ------------------------------------------------------------------
    def score_geometric(r):
        return compute_score(generate_geometric(n, r))

    r_low, r_high = 0.80, 0.9999
    for _ in range(15):
        if time.time() >= deadline:
            break
        r1 = r_high - (r_high - r_low) / phi
        r2 = r_low + (r_high - r_low) / phi
        s1 = score_geometric(r1)
        s2 = score_geometric(r2)
        if s1 < s2:
            r_high = r2
            if s1 < best_val:
                best_val = s1
                best_seq = generate_geometric(n, r1)
        else:
            r_low = r1
            if s2 < best_val:
                best_val = s2
                best_seq = generate_geometric(n, r2)
    print(f"After geometric fine‑tune: best_score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Smooth Adam optimisation – allocate the bulk of the remaining time.
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        remaining = deadline - time.time()
        # Estimate ~3000 iterations per second (empirical)
        max_iters = int(min(1_200_000, remaining * 3000))
        print(f"Running smooth Adam optimisation for up to {max_iters} iters.")
        a_smooth, smooth_score = smooth_gradient_optimize(
            best_seq,
            time_deadline=deadline - 20,
            max_iters=max_iters,
            beta_start=5.0,
            beta_end=500.0,
            step0=0.9,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.copy()
            print(f"Smooth optimisation improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Projected sub‑gradient polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        remaining = deadline - time.time()
        max_iter = int(min(600_000, remaining * 2500))
        print(f"Running sub‑gradient polishing for up to {max_iter} iters.")
        a_sub, sub_score = subgrad_optimize(
            best_seq,
            time_deadline=deadline - 5,
            max_iter=max_iter,
            step0=0.07,
        )
        if sub_score < best_val:
            best_val = sub_score
            best_seq = a_sub.copy()
            print(f"Sub‑gradient polishing improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final stochastic hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        steps = int(min(500_000, (deadline - time.time()) * 2000))
        print(f"Final hill‑climb for up to {steps} steps.")
        polished_seq, hill_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val = hill_score
            best_seq = np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final reporting using the provided evaluate_sequence().
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain list of floats (already on the simplex)
    return best_seq.tolist()
