Given a sequence a of length n (list of floats). evaluate_sequence does:

- Checks input type and that each element is numeric, not NaN, not infinite.
- Then protect against negative numbers: sequence = [max(0, x) for x in sequence]. So negative entries become zero in the modified sequence.
- Clip values > 1000 to 1000.
- n = len(sequence)
- b_sequence = np.convolve(sequence, sequence) (full linear convolution length 2n-1)
- max_b = max(b_sequence)
- sum_a = np.sum(sequence)
- If sum_a < 0.01, return inf.
- Return float(2 * n * max_b / (sum_a**2))

assistantfinal**Solution Overview**

The evaluator is  

\[
\text{score}(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge0 .
\]

The score is *scale‑invariant*: after normalising the vector to a probability
distribution (\(\sum a_i = 1\)) the score becomes  

\[
\text{score}=2\,n\;\max_k (a*a)_k .
\]

Hence the optimisation problem is:  

*find a non‑negative probability vector whose auto‑convolution is as flat as
possible*.

The previous implementation minimised a smooth surrogate of the
\(L_\infty\)‑norm (log‑sum‑exp) and then performed a sub‑gradient walk.
That gets very close to the optimum, but a little more work can still lower
the peak of the convolution.

The new search adds a **variance‑flattening phase**.  
Instead of directly minimising the maximum, we minimise the quadratic
dispersion of the convolution

\[
V(a)=\sum_k\bigl((a*a)_k-\bar c\bigr)^2 ,
\qquad\bar c=\frac{1}{2n-1}
\]

because a flatter convolution automatically reduces its peak.
The gradient of \(V\) is cheap to compute via a single FFT‑based
correlation:

\[
\nabla V(a)=4\;\bigl((a*a)-\bar c\bigr)\star a ,
\]

where \(\star\) denotes the (valid) correlation.  
A projected gradient descent (with a simple step‑size schedule) is used,
followed by the original soft‑max GD, a short sub‑gradient refinement and a
final stochastic hill‑climb.

All phases respect the 1000 s wall‑clock limit and the best sequence found
is returned.

```python
"""
Advanced step‑function optimiser.

Phases
------
1 – diverse seeds, keep the best.
2 – soft‑max (log‑sum‑exp) gradient descent on the max‑convolution.
3 – variance‑flattening gradient descent (minimise L2‑variance of the
    convolution, i.e. flatten the whole profile).
4 – sub‑gradient descent on the true max‑objective.
5 – stochastic hill‑climb polishing.

All updates are projected onto the probability simplex, so the
objective 2·n·max(conv) is directly comparable to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n  # points in (0,1)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Correlation helper used by the variance‑flattening phase
# ----------------------------------------------------------------------


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid correlation (length = len(a)):
        (e ⋆ a)[i] = Σ_m e[i+m] * a[m].
    Implemented via one FFT‑convolution with the reversed vector.
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    return conv[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Phase 1 – soft‑max gradient descent (same idea as the previous code)
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step_coef: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth surrogate of the max‑norm:
        Lβ(a)=log Σ exp(β·conv(a,a))
    using projected gradient descent with an annealed temperature β.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    it = 0
    start = time.time()
    while time.time() < deadline:
        # exact score for bookkeeping
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # annealed temperature
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max weights
        shifted = conv - max_conv               # max(conv) → 0
        exp_vals = np.exp(beta * shifted)      # all ≤1, stable
        w = exp_vals / exp_vals.sum()

        # gradient of the surrogate
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        # step size (decays with √iteration and with β)
        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        # projected update
        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        # occasional progress report
        if it % 5000 == 0:
            print(
                f"[soft‑max GD] it {it:6d} | time {elapsed:6.1f}s | "
                f"best {best_score:.6f} | β {beta:.1f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – variance‑flattening gradient descent
# ----------------------------------------------------------------------


def variance_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Minimise the variance of the auto‑convolution
        V(a)=∑_k (conv(k)-avg)²,
    where avg = 1/(2n‑1).  The gradient is
        ∇V = 4·((conv-avg) ⋆ a) .
    A simple projected gradient descent is employed.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    avg = 1.0 / (2 * n - 1)

    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # error from the flat target
        e = conv - avg

        # gradient of the variance
        grad = 4.0 * correlation_valid(e, cur_a)

        it += 1
        step = step0 / np.sqrt(it)

        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        # very occasional reporting
        if it % 2000 == 0:
            print(
                f"[var‑GD] it {it:6d} | time {time.time():6.1f}s | "
                f"best {best_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – sub‑gradient refinement (identical to the previous programme)
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – greedy hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Stochastic moves that are kept only when they improve the exact score."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()

        # two elementary move types
        if np.random.rand() < 0.5:                # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand /= cand.sum()                     # numerical drift correction
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 0 – optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"Fallback score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1 – choose a moderate length (2048 works well; 4096 if we have time)
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 850.0:
        n = 4096
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2 – generate a diverse set of seeds and keep the best one
    # ------------------------------------------------------------------
    seeds = []

    # deterministic shapes
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    # Beta‑pdf families (α,β)
    for alpha, beta in [(0.2, 0.2), (0.5, 0.5), (1.0, 1.0),
                        (2.0, 2.0), (5.0, 5.0), (0.5, 2.0), (2.0, 0.5)]:
        seeds.append(generate_beta_pdf(n, alpha, beta))

    # random Dirichlet (various concentrations)
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # very sparse spikes – often a good starting point
    for k in (30, 50, 80):
        seeds.append(random_spike(n, k=k))

    # include fallback if it matches the current length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float) / np.sum(best_seq))

    # evaluate seeds (exact objective) and keep the best
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)          # guard against tiny drift
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[seed] improvement → score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3 – start optimisation from the best seed
    # ------------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    # ------------------------------------------------------------------
    # 4 – phase allocation based on remaining time
    # ------------------------------------------------------------------
    t0 = time.time()
    phase1_deadline = t0 + total_time * 0.38
    phase2_deadline = phase1_deadline + total_time * 0.27
    phase3_deadline = phase2_deadline + total_time * 0.27
    # the rest is for hill‑climb

    # ------------------------------------------------------------------
    # Phase 1: soft‑max gradient descent
    # ------------------------------------------------------------------
    print("\n[Phase 1] soft‑max gradient descent")
    cur_a, cur_score = softmax_gradient_descent(
        cur_a,
        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=2000.0,
        step_coef=0.001,
    )
    if cur_score < best_val:
        best_val = cur_score
        print(f"[after Phase 1] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2: variance‑flattening descent
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        print("\n[Phase 2] variance‑flattening gradient descent")
        cur_a, cur_score = variance_gradient_descent(
            cur_a,
            cur_score,
            phase2_deadline,
            step0=0.05,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 2] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3: sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        print("\n[Phase 3] sub‑gradient refinement")
        cur_a, cur_score = subgradient_refine(
            cur_a,
            cur_score,
            phase3_deadline,
            max_iter=600_000,
            step0=0.07,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after Phase 3] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4: hill‑climb polishing (uses the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n[Phase 4] hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=2_000_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after hill‑climb] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
```
