"""
propose_candidate
=================

A multi‑scale optimisation pipeline that tries to beat the previous
1.511180 bound.  The idea is:

1. **Coarse search** on a modest size (≈ 8 k) using many analytic families,
   Dirichlet draws and a few random sparse starts.
2. **Smooth Adam optimisation** on the log‑sum‑exp surrogate of the
   max‑convolution.  β is increased from a small value to a very large
   one (≈ 5·10⁴) while the Adam step runs for many iterations.
3. **Upscaling** – the best vector is interpolated to a larger size
   (≈ 32 k) and refined again with the same high‑β Adam.
4. **Projected gradient descent** on the true max‑convolution using an
   explicit sub‑gradient and Euclidean projection onto the probability
   simplex.
5. **Symmetrisation** and a tiny simulated‑annealing hill‑climb as a final
   polish.

All stages respect the global time budget (default 1000 s) and keep the
overall best vector seen.  The returned list can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using real FFT (fast, no Python loops).
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by the competition (scale‑invariant).
    The vector a is assumed to be non‑negative and Σ a = 1.
    Returns 2·n·max(a⋆a) where n = len(a) and “⋆” denotes aperiodic convolution.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – needed after gradient steps
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx=1} (Euclidean)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for cheap initial scanning)
# ----------------------------------------------------------------------


def uniform_family(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) ."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def triangular_family(n: int) -> np.ndarray:
    """Symmetric triangular shape a_i ∝ (i+1)*(n-i)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) * (n - i)
    a = w / w.sum()
    return a


def cosine_window_family(n: int) -> np.ndarray:
    """a_i ∝ 1 - cos(2π i/(n‑1)) (raised‑cosine)."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    a = w / w.sum()
    return a


def sin_power_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin^{p} (π (i+0.5)/n)."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** p
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Draw a symmetric Dirichlet(α) vector, α = conc."""
    vec = np.random.gamma(conc, 1.0, size=n)
    return vec / vec.sum()


def random_sparse_start(n: int, sparsity: float = 0.05) -> np.ndarray:
    """Very sparse probability vector (≈ sparsity fraction non‑zeros)."""
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed optimisation (log‑sum‑exp surrogate) via Adam
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 20_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·(a⋆a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β grows exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)               # initialise in log‑softmax space
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule β exponentially
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # smoothed weights over convolution shifts
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))        # stabilise
        w = exp_w / exp_w.sum()                     # soft‑max over shifts

        # gradient of f_β w.r.t. a (see paper notes)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
        n = a.shape[0]
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]   # pick the relevant slice

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}]   Adam iter {it:7d} – new best = {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def max_convolution_subgradient(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Returns the sub‑gradient of max_k (a⋆a)_k at the current a,
    together with the active shift index k.
    The sub‑gradient g_i = 2·a_{k-i} for all i where the index is valid,
    otherwise g_i = 0.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    k = int(np.argmax(conv))

    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    return g, k


def projected_gradient_descent(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.5,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple projected (Euclidean) gradient descent on the true objective
    2·n·max(a⋆a).  Step size η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        g, _ = max_convolution_subgradient(a)
        eta = step0 / np.sqrt(t)
        a = a - eta * g
        a = project_to_simplex(a)

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   PGD step {t:7d} – new best = {best_score:.6f}")

    # final evaluation
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Small simulated‑annealing hill‑climb (tiny perturbations)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 500_000,
    init_temp_factor: float = 5e-4,
) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature simulated‑annealing walk on the simplex.
    Uses swaps and small mass transfers.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.04

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Upscaling / interpolation helper
# ----------------------------------------------------------------------


def upscale_vector(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to length new_n and renormalise."""
    if a.shape[0] == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, a.shape[0])
    new_x = np.linspace(0.0, 1.0, new_n)
    b = np.interp(new_x, old_x, a)
    return project_to_simplex(b)

    except Exception:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        print(f"[{time.time():.1f}] Upscaling to fine dimension n = {n_fine}")
        a_fine = upscale_vector(np.array(best_seq, dtype=np.float64), n_fine)

        a_ref, sc_ref = adam_smooth_optimize(
            a_fine,
            deadline - 120.0,
            max_iters=240_000,
            beta_start=3000.0,
            beta_end=50_000.0,
            lr=0.010,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.tolist()
            print(f"[{time.time():.1f}] after fine‑Adam = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 4. Projected gradient descent on the true max‑objective (fine dim)
        # ------------------------------------------------------------------
        if time.time() < deadline - 70.0:
            print(f"[{time.time():.1f}] Starting projected gradient descent")
            a_pgd, sc_pgd = projected_gradient_descent(
                np.array(best_seq, dtype=np.float64),
                deadline - 45.0,
                step0=0.4,
                check_interval=1500,
            )
            if sc_pgd < best_score:
                best_score = sc_pgd
                best_seq = a_pgd.tolist()
                print(f"[{time.time():.1f}] after PGD best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation (most good solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        maybe_update(a_sym, "symmetrised")

    # ------------------------------------------------------------------
    # 6. Final tiny simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Safeguard – if nothing was found fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final evaluation (the outer harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    print(f"Length of returned sequence = {len(best_seq)}")
    return best_seq
