def conv_matrix(a):
    n = len(a)
    m = 2*n-1
    # using np.zeros
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(m):
        # j ranges such that i-j in [0,n-1] => j ∈ [i-(n-1), i]
        j_start = max(0, i - (n - 1))
        j_end = min(i, n-1)
        # set entries
        # for j from j_start to j_end inclusive:
        # A[i,j] = a[i - j]
        # we can vectorize using slice assignments:
        A[i, j_start:j_end+1] = a[i - np.arange(j_start, j_end+1)]
    return A

def hill_climb(sequence, time_deadline, max_steps=200_000):
    a = np.array(sequence,dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.1
    for step in range(max_steps):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a,a)
        k = int(np.argmax(conv))
        # compute gradient contributions
        idx = np.arange(n)
        j = k - idx
        mask = (j >=0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # find indices
        i_max = np.argmax(g)
        i_min = np.argmin(g)
        if i_max != i_min:
            delta = step_scale * a[i_max]
            cand = a.copy()
            cand[i_max] -= delta
            cand[i_min] += delta
            cand = project_to_simplex(cand)
        else:
            # fallback random move
            i, j = np.random.randint(n, size=2)
            delta = step_scale * a[i]
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            # perhaps accept if small improvement? else reduce step
            step_scale *= 0.97
    return best_a.tolist(), best_score

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
* Build a palette of analytic starters (uniform, central‑binomial,
  power‑law, geometric) and a handful of random Dirichlet vectors.
* Keep the best starter (by the exact objective 2·n·max(conv)).
* Run a **long smooth Adam** phase with a log‑sum‑exp surrogate.
  We anneal the temperature from β≈5 up to β≈2000 and allow many
  millions of cheap FFT‑based iterations.
* Polish the result with a **projected sub‑gradient** optimiser
  (the true max‑convolution objective) using an adaptive step‑size.
* Finish with a **guided hill‑climb** that moves mass from the
  most responsible coordinate (according to the sub‑gradient) to the
  least responsible one; the move size is adapted on‑the‑fly.
* All phases respect the wall‑clock deadline (default 1000 s) and
  keep the best sequence ever seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the probability simplex (∑a=1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        ∑v = 1, v ≥ 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exactly sum 1 if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimiser – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 800_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Uses a log‑sum‑exp surrogate with an annealed temperature β.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal temperature (smoothness) β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # full auto‑convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights (log‑sum‑exp trick)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed objective:
        #   ∂/∂a  = 2·(rev(a) * w)   (restricted to length n)
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)

        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 800_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to escape flat spots
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Guided hill‑climb
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Hill‑climb that moves a little mass from the coordinate that most
    contributes to the current peak towards the one that contributes
    the least.  The move size adapts based on recent success.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        # compute the current peak and the associated gradient contributions
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))

        cand = a.copy()
        if i_max != i_min and a[i_max] > 0:
            delta = step_scale * a[i_max]
            cand[i_max] -= delta
            cand[i_min] += delta
        else:
            # fallback random move
            i, j_ = np.random.randint(n, size=2)
            delta = step_scale * a[i]
            cand[i] -= delta
            cand[j_] += delta

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97
            # occasionally accept a neutral move to keep exploring
            if np.random.rand() < 0.001:
                a = cand

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds spare for the harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 0. Determine problem size (length of the sequence)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore
        init_candidates = [np.array(height_sequence_1, dtype=np.float64)]
        print(f"Using provided height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        init_candidates = []
        print(f"No external sequence provided – using length {n}.")

    # --------------------------------------------------------------
    # 1. Analytic starters + a few random Dirichlet vectors
    # --------------------------------------------------------------
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))        # uniform
    init_candidates.append(generate_central_binomial(n))                # central‑binomial
    # power‑law exponents
    for p in np.linspace(0.3, 1.0, 9):
        init_candidates.append(generate_powerlaw(n, float(p)))
    # geometric ratios
    for r in [0.970, 0.980, 0.985, 0.990, 0.992, 0.995]:
        init_candidates.append(generate_geometric(n, float(r)))
    # a few Dirichlet draws (diverse random starts)
    for _ in range(5):
        init_candidates.append(np.random.dirichlet(np.ones(n)))

    # evaluate starters, keep the best
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"Best starter score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Short multi‑start smooth Adam runs (quick local exploration)
    # --------------------------------------------------------------
    # give each of the top 3 starters a quick polishing run
    remaining = deadline - time.time()
    quick_deadline = time.time() + max(0.0, remaining * 0.15)
    # select up to three best starters by score
    starter_scores = [(compute_score(c), c) for c in init_candidates]
    starter_scores.sort(key=lambda x: x[0])
    top_three = [c for _, c in starter_scores[:3]]

    for i, cand in enumerate(top_three):
        if time.time() >= quick_deadline:
            break
        # limited number of iterations – enough to move away from the very
        # local basin but cheap enough to try several seeds
        max_iters = 150_000
        cand_opt, cand_score = smooth_gradient_optimize(
            cand,
            time_deadline=quick_deadline,
            max_iters=max_iters,
            beta_start=5.0,
            beta_end=800.0,
            step0=0.9,
        )
        print(f"Quick smooth run {i}: score = {cand_score:.6f}")
        if cand_score < best_val:
            best_val = cand_score
            best_seq = cand_opt.copy()
            print(f" --> new best after quick run {i}")

    # --------------------------------------------------------------
    # 3. Long smooth Adam optimisation (major phase)
    # --------------------------------------------------------------
    smooth_deadline = deadline - 200.0  # reserve time for polishing steps
    if time.time() < smooth_deadline:
        remaining = smooth_deadline - time.time()
        # estimate about 2500 iterations per second on typical hardware
        max_iters = int(min(2_500_000, remaining * 2500))
        print(f"Long smooth Adam optimisation – up to {max_iters} iterations.")
        best_seq, smooth_score = smooth_gradient_optimize(
            best_seq,
            time_deadline=smooth_deadline,
            max_iters=max_iters,
            beta_start=5.0,
            beta_end=2000.0,
            step0=0.8,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            print(f"Smooth phase improved best score to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Projected sub‑gradient polishing
    # --------------------------------------------------------------
    subgrad_deadline = deadline - 100.0
    if time.time() < subgrad_deadline:
        remaining = subgrad_deadline - time.time()
        max_iter = int(min(1_200_000, remaining * 3000))
        print(f"Sub‑gradient optimisation – up to {max_iter} iterations.")
        best_seq, sub_score = subgrad_optimize(
            best_seq,
            time_deadline=subgrad_deadline,
            max_iter=max_iter,
            step0=0.06,
        )
        if sub_score < best_val:
            best_val = sub_score
            print(f"Sub‑gradient phase improved best score to {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final guided hill‑climb
    # --------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 400_000
        print(f"Final hill‑climb – up to {steps} steps.")
        polished_seq, hill_score = hill_climb(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val = hill_score
            best_seq = np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Fallback in the extremely unlikely event that nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # --------------------------------------------------------------
    # 7. Final reporting using the provided evaluate_sequence()
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain list of floats (already on the simplex)
    return best_seq.tolist()
