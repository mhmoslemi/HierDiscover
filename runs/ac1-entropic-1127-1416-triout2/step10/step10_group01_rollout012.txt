def solve_lp_for_g(f, M):
    n = f.shape[0]
    m = 2*n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    c = -np.ones(n)
    bounds = [(0.0, None)]*n
    b = np.full(m, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    g_norm = g / sum_g
    return g_norm
```

def compute_subgradient_multiple(a, conv, eps=1e-12):
    maxc = conv.max()
    # choose indices where conv >= maxc - tol
    tol = max(maxc * 1e-12, 1e-12)
    ks = np.where(conv >= maxc - tol)[0]
    n = a.shape[0]
    g_sum = np.zeros_like(a)
    for k in ks:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        # contributions: 2*a[j] for i != j, a[j] for i=j
        # We can vectorize: for mask positions where j != idx => 2*a[j], else a[idx]
        g = np.zeros_like(a)
        # non-diagonal
        nd = mask & (j != idx)
        g[nd] = 2.0 * a[j[nd]]
        # diagonal
        dg = mask & (j == idx)
        g[dg] = a[dg]
        g_sum += g
    g_avg = g_sum / len(ks) if len(ks) > 0 else np.zeros_like(a)
    return g_avg

"""
Improved search for a non‑negative step‑function that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

The algorithm proceeds in several timed stages:

0️⃣  Load a fallback if supplied by the harness.
1️⃣  Choose a problem size (2048 entries – a good compromise between
    quality and speed of the linear‑programming refinement).
2️⃣  Quick scan of a few analytic families (U‑shapes, powers, exponentials,
    random draws) to obtain a decent starter.
3️⃣  Several short Adam runs on the smooth‑max surrogate, each starting
    from a different promising candidate.
4️⃣  A long Adam optimisation with a very large smoothing parameter
    (β up to 2000) to approach the true max‑objective.
5️⃣  Projected sub‑gradient descent that averages the sub‑gradients of
    all (or almost all) maximising shifts; a back‑tracking line search
    guarantees monotone improvement.
6️⃣  LP‑based refinement (the classic “direction‑finding” step) with a
    cheap line‑search over the mixing coefficient.
7️⃣  Optional symmetrisation.
8️⃣  Final stochastic hill‑climb (simulated‑annealing style) for the
    remaining budget.

All phases respect a global wall‑clock budget (default 1000 s) and
periodically report the current best score.  The returned list is ready
for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT (fast, O(n log n))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shape: w_i ∝ ((i+½)(n‑½‑i))^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 0.5 - i)
    w = np.power(prod, -alpha, where=prod > 0, out=np.ones_like(prod))
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing power law: w_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, -beta, where=i >= 0, out=np.ones_like(i))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay from the edges toward the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex combination of uniform and exp_u shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing sequence (sorted power of uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)

# Smooth‑max (Adam) optimisation
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 4.0,
                         beta_end: float = 800.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate

          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    a is represented as softmax(z); we optimise z.
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

# Projected sub‑gradient descent (averaged sub‑gradient + line search)
# ----------------------------------------------------------------------


def averaged_subgradient(a: np.ndarray, conv: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    """
    Return the average of sub‑gradients for all shifts whose convolution
    value is within `tol` of the maximum.
    """
    n = a.shape[0]
    maxc = conv.max()
    # tolerance can be absolute or relative; using absolute fraction of maxc
    eps = max(tol, maxc * 1e-12)
    ks = np.where(conv >= maxc - eps)[0]

    if ks.size == 0:
        ks = np.array([int(np.argmax(conv))])

    g_acc = np.zeros_like(a)
    for k in ks:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        # non‑diagonal contributions
        nd = mask & (j != idx)
        g_acc[nd] = 2.0 * a[j[nd]]
        # diagonal contributions where i == j
        dg = mask & (j == idx)
        g_acc[dg] = a[dg]
    return g_acc / ks.size


def subgrad_descent(start_a: np.ndarray,
                    deadline: float,
                    max_iters: int = 2_000_000,
                    step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected (averaged) sub‑gradient descent with a simple back‑tracking
    line search.  The iterate is always kept on the simplex.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        g = averaged_subgradient(a, conv)

        # basic step size schedule  α / sqrt(t)
        step = step0 / np.sqrt(t)

        # back‑tracking: try to reduce the objective, otherwise shrink step
        for _ in range(3):
            a_new = project_to_simplex(a - step * g)
            sc = compute_score(a_new)
            if sc <= best_score:
                a = a_new
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved to {sc:.6f}")
                break
            step *= 0.5

        # periodic reporting (every few thousand iterations)
        if t % 5000 == 0:
            # evaluate current point without extra reduction attempts
            sc_cur = compute_score(a)
            if sc_cur < best_score:
                best_score = sc_cur
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved (periodic) to {sc_cur:.6f}")

    # final evaluation
    sc_fin = compute_score(a)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a.copy()
    return best_a, best_score

# Linear‑programming refinement (direction finding)

def solve_lp_for_direction(f: np.ndarray, M: float, time_limit: float = 30.0):
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M  (for all k),
    g ≥ 0.
    Returns the normalised direction g_norm (sums to 1) or None on failure.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (size m × n) where (A @ g)_k = (f * g)_k
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': time_limit})

    if not res.success:
        return None

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return None
    return g / sum_g


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float) -> np.ndarray:
    """
    Perform one LP‑based refinement step and return the mixed vector.
    The mixing weight is kept fixed; a separate line‑search is applied
    in the driver.
    """
    g_norm = solve_lp_for_direction(f, M)
    if g_norm is None:
        return f
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    return project_to_simplex(f_new)

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing walk on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature scaled to the current score
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=1_800_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 9.  Official evaluation and return
    # --------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
