"""
Step‑function minimiser (improved)

The routine builds a probability vector a (non‑negative, sum = 1) that
minimises  2·n·max(a*a)  (the value returned by evaluate_sequence).
It combines several ideas that were not used in the previous baseline:

*   A small number of L‑BFGS optimiser runs on a smooth log‑sum‑exp
    approximation of the max‑convolution.  The smoothness parameter β is
    increased successively (10 → 100 → 1000 → 5000) so that the optimiser
    first sees a very smooth landscape and finally refines the exact
    maximum.

*   After the smooth phase a cheap sub‑gradient descent (multiplicative
    KL‑mirror update) is performed for a few ten‑thousands of iterations.

*   Finally a targeted hill‑climb moves mass from the index that contributes
    most to the current maximal convolution entry to the index that
    contributes the least, accepting only improving moves.

All operations keep the vector on the probability simplex (by using a
soft‑max parametrisation for the L‑BFGS phase and an explicit projection
otherwise).  The algorithm respects the 1000 s wall‑clock budget and
returns the best sequence found.
"""

import time

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = x - np.max(x)
    e = np.exp(shift)
    return e / np.sum(e)

def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already satisfies sum = 1:
        2·n·max( a * a )
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta distribution on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()

class TimeBudgetExceeded(Exception):
    """Raised when the global time budget is exhausted."""


def _smooth_obj_factory(beta: float):
    """Factory returning a closure (value,grad) for a given beta."""
    def obj(x):
        # ----- a from soft‑max -----
        a = softmax(x)

        # ----- convolution a*a -----
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # ----- smooth max (log‑sum‑exp) -----
        diff = conv - maxc                # ≤ 0
        exp_term = np.exp(beta * diff)    # stable because diff ≤ 0
        w = exp_term / exp_term.sum()     # weights, sum to 1

        # ----- gradient w.r.t a (still on the simplex) -----
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)[len(a) - 1:len(a) - 1 + len(a)]

        # ----- chain rule to obtain gradient w.r.t x -----
        dot_aw = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot_aw)

        # ----- smooth objective value -----
        f = maxc + (1.0 / beta) * np.log(exp_term.sum())

        return f, grad_x
    return obj


def optimize_lbfgs(start_a: np.ndarray, beta: float,
                   maxiter: int, deadline: float):
    """
    Runs L‑BFGS on the smooth approximation with parameter `beta`.
    Returns (optimised a, exact score).
    """
    if not _SCIPY_OK:
        raise RuntimeError("SciPy not available – cannot run L‑BFGS stage.")

    eps = 1e-15
    x0 = np.log(start_a + eps)

    # container for the most recent iterate (used if we run out of time)
    last_x = [x0.copy()]

    def callback(xk):
        last_x[0] = xk.copy()
        if time.time() > deadline:
            raise TimeBudgetExceeded()

    obj = _smooth_obj_factory(beta)

    try:
        res = _opt.minimize(obj, x0, method='L-BFGS-B', jac=True,
                            options={'maxiter': maxiter, 'disp': False},
                            callback=callback)
        x_opt = res.x
    except TimeBudgetExceeded:
        x_opt = last_x[0]

    a_opt = softmax(x_opt)
    sc_opt = compute_score(a_opt)
    return a_opt, sc_opt


def max_descent_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 20000,
                         eta0: float = 0.4):
    """
    Sub‑gradient (multiplicative KL‑mirror) descent on the exact max.
    Returns (best a, best exact score).
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.size

    best_a = a.copy()
    best_sc = compute_score(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it + 1)

        a = np.maximum(a * np.exp(-eta * g), 1e-15)
        a /= a.sum()

        if (it % 200) == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        it += 1

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def targeted_hill_climb(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 50000) -> tuple[np.ndarray, float]:
    """
    Greedy improvement: repeatedly move a fraction of the mass from the index
    that contributes most to the current maximal convolution entry to the index
    that contributes the least.  Only improving moves are kept.
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.08  # relative amount of mass moved each attempt
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)

        contrib = best_a[idx] * best_a[k_max - idx]
        i_max = idx[np.argmax(contrib)]
        i_min = idx[np.argmin(contrib)]

        if i_max == i_min:
            # no useful move – pick a random partner
            i_min = np.random.randint(n)

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[i_min] += delta
        cand = project_to_simplex(cand)  # renormalise safely

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            # reward successful moves by allowing a slightly larger step
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # punish unsuccessful moves – shrink the step
            step_scale *= 0.95

        steps += 1

    return best_a, best_sc

                a_opt, sc_opt = optimize_lbfgs(
                    np.array(best_seq, dtype=float),
                    beta=beta,
                    maxiter=30,
                    deadline=deadline - 5.0,
                )
                if sc_opt < best_score:
                    best_score, best_seq = sc_opt, a_opt.tolist()
                    print(f"[{time.time():.1f}] after L‑BFGS β={beta:.0f} best = {best_score:.6f}")
            except Exception:
                # any failure (including time‑budget) just abort the L‑BFGS stage
                print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} aborted")
                break

    # -------------------------------------------------
    # 3️⃣  Exact max‑descent (sub‑gradient) for a few ten‑thousands iters
    # -------------------------------------------------
    if time.time() < deadline - 20.0:
        a_opt, sc_opt = max_descent_optimize(
            np.array(best_seq, dtype=float),
            deadline - 5.0,
            max_iters=25000,
            eta0=0.5,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after exact max‑descent best = {best_score:.6f}")

    # -------------------------------------------------
    # 4️⃣  Targeted hill‑climb polishing
    # -------------------------------------------------
    if time.time() < deadline - 2.0:
        a_opt, sc_opt = targeted_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=100000,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # 5️⃣  Safeguard – fall back to a tiny uniform vector if everything failed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # 6️⃣  Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
