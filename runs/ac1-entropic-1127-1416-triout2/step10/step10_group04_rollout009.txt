```
def targeted_hill_climb(a, time_deadline, max_moves=500000):
    n = len(a)
    best = a.copy()
    best_score = compute_score(best)
    # temperature schedule
    T0 = 1e-3
    T_end = 1e-6
    moves = 0
    while time.time() < time_deadline and moves < max_moves:
        # pick i in active region (i where a_i > 0)
        i = np.random.randint(n)
        if best[i] == 0:
            continue
        # pick j near far from i? we just random
        j = np.random.randint(n)
        if i == j:
            continue
        # propose moving fraction alpha of mass from i to j
        alpha = 0.01 * (0.5 ** (moves // 10000)) # adapt gradually shrink
        delta = min(alpha * best[i], best[i]) # ensure non-negative
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # project to simplex (ensures positivity)
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        delta_score = cand_score - best_score
        # temperature dependent acceptance
        T = T0 * (T_end / T0) ** (moves / max_moves)
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / (T + 1e-12)):
            best = cand
            best_score = cand_score
        moves += 1
    return best, best_score
```

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
* Build a diverse set of analytic initialisations (uniform, central‑binomial,
  power‑law, geometric and a mixed power‑law / geometric family).  Keep the
  best according to the exact objective.
* Refine the best vector with a long, annealed log‑sum‑exp (smooth) Adam
  optimiser.  The temperature is driven to a very high value (β≈1000) so
  that the smooth surrogate is almost the true max‑convolution.
* Polish with a projected sub‑gradient method (the exact max‑convolution
  gradient) using a small step‑size schedule.
* Run a second smooth Adam pass for fine‑tuning.
* Finish with a simulated‑annealing style hill‑climb that moves tiny
  fractions of probability mass between positions while allowing occasional
  uphill moves.  The whole routine respects the 1000 s wall‑clock limit and
  keeps the best solution seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the probability simplex
    (sum(a) == 1, a_i ≥ 0).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, normalised to sum 1 (monotone decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exactly sum 1 if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_mixed_power_geo(n: int, p: float, r: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^(-p)·r^i, normalised to sum 1.
    This interpolates between pure power‑law (r≈1) and geometric decay.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (idx ** (-p)) * (r ** (idx - 1))
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimiser – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Uses a log‑sum‑exp surrogate with an annealed temperature β.
    The function stops when the wall‑clock deadline is reached.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal β geometrically
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # Soft‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # Gradient of the smooth surrogate:
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional evaluation
        if it % 1000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Uses the gradient of the active convolution entry.
    """
    a = init_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient for the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional tiny jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style hill‑climb
# ----------------------------------------------------------------------


def hill_climb_anneal(
    start_a: np.ndarray,
    time_deadline: float,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic moves (transfer of a fraction of mass) with a
    decreasing temperature that occasionally accepts uphill moves.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # temperature schedule (linear in log‑space)
    T0 = 5e-3
    T_end = 5e-7

    move = 0
    while move < max_moves and time.time() < time_deadline:
        # pick source and destination
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # fraction of mass to move (decreases slowly)
        eps = 0.02 * (0.99 ** (move // 5000))
        delta = eps * a[i]
        if delta == 0.0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        delta_score = cand_score - best_score

        T = T0 * (T_end / T0) ** (move / max_moves)
        accept = False
        if delta_score < 0:
            accept = True
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-delta_score / (T + 1e-12)):
                accept = True

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        move += 1

    return best_a, best_score
