def local_search(a, conv, best_score, deadline):
    n = len(a)
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_score = best_score
    best_a = cur_a.copy()
    best_score = cur_score
    # hyperparameters
    temp_start = best_score * 0.01
    temp_end = 1e-6
    step_scale = 0.1 # for transfers
    accept_count = 0
    total_iter = 0

    while time.time() < deadline:
        total_iter += 1
        # temperature schedule linearly decreasing based on elapsed fraction
        elapsed = time.time() - start_time
        fraction = elapsed / (deadline - start_time)
        temp = temp_start * (1 - fraction) + temp_end

        # choose operation
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.choice(n, size=2, replace=False)
            if cur_a[i] == cur_a[j]:
                continue
            delta_i = cur_a[j] - cur_a[i]
            delta_j = -delta_i  # cur_a[i] - cur_a[j]
            new_conv = cur_conv.copy()
            # add slice updates
            new_conv[i:i+n] += 2 * delta_i * cur_a
            new_conv[j:j+n] += 2 * delta_j * cur_a
            new_conv[2*i] += delta_i * delta_i
            new_conv[2*j] += delta_j * delta_j
            idx = i + j
            new_conv[idx] += 2 * delta_i * delta_j

            new_score = 2 * n * new_conv.max()
            # acceptance
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                if np.random.rand() < np.exp((cur_score - new_score) / max(temp, 1e-12)):
                    accept = True
            if accept:
                # update
                cur_conv = new_conv
                cur_score = new_score
                # swap in a
                cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = cur_a.copy()
        else:
            # transfer mass
            src = np.random.randint(n)
            if cur_a[src] < 1e-8:
                continue
            dst = np.random.randint(n)
            if src == dst:
                continue
            # delta as fraction
            delta = cur_a[src] * step_scale * np.random.rand()
            if delta <= 0:
                continue
            delta_src = -delta
            delta_dst = delta
            new_conv = cur_conv.copy()
            new_conv[src:src+n] += 2 * delta_src * cur_a
            new_conv[dst:dst+n] += 2 * delta_dst * cur_a
            new_conv[2*src] += delta_src * delta_src
            new_conv[2*dst] += delta_dst * delta_dst
            new_conv[src+dst] += 2 * delta_src * delta_dst

            new_score = 2 * n * new_conv.max()
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                if np.random.rand() < np.exp((cur_score - new_score) / max(temp, 1e-12)):
                    accept = True
            if accept:
                cur_conv = new_conv
                cur_score = new_score
                cur_a[src] += delta_src
                cur_a[dst] += delta_dst
                # renorm occasionally
                if total_iter % 5000 == 0:

                if cur_score < best_score:
                    best_score = cur_score

"""
Improved search for a non‑negative step‑function that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages

1. **Seeding** – a handful of hand‑crafted shapes (sine‑squared,
   Beta‑family) together with a fallback sequence (if available) and a few
   random Dirichlet draws. The best seed (according to the exact objective)
   is retained.

2. **Smooth surrogate optimisation** – projected gradient descent on the
   “soft‑max” surrogate
        Lβ(a) = log Σ exp(β·conv(a,a)),
   with an aggressive annealing schedule for the temperature β (up to 5000)
   and Adam‑style adaptive steps.  The iterate with the lowest true score
   is kept.

3. **Local discrete refinement** – a fast incremental hill‑climb / simulated
   annealing that works directly on the convolution.  Two elementary moves
   are used:
        • swapping two entries,
        • moving a small amount of mass from one position to another.
   The convolution is updated in O(n) using slice‑additions, so millions of
   trials can be performed within the remaining time budget.

The whole search respects the 1000 s wall‑clock limit and finally returns a
plain ``list[float]`` of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for n ≳ 500)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin^2 shape a_i = sin^2(pi·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    p>0 clusters mass in the centre, p<0 pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 2 – soft‑max gradient descent (with Adam‑style steps)
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    step0: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Minimise the surrogate Lβ(a)=log Σ exp(β·conv(a,a)) via projected
    gradient descent with an Adam‑style adaptive step.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    # Adam state
    m = np.zeros_like(cur_a)
    v = np.zeros_like(cur_a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    start = time.time()

    while time.time() < deadline:
        # exact objective for bookkeeping -------------------------------------------------
        conv = convolution_fft(cur_a, cur_a)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
        # temperature schedule -------------------------------------------------------------
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        beta = beta_start + frac * (beta_end - beta_start)

        # soft‑max weights -----------------------------------------------------------------
        max_conv = conv.max()
        shifted = conv - max_conv                     # max becomes 0
        exp_vals = np.exp(beta * shifted)            # stable, ≤1
        w = exp_vals / exp_vals.sum()                # soft‑max distribution

        # gradient of surrogate ------------------------------------------------------------
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        # Adam update ----------------------------------------------------------------------
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        nxt = cur_a - step * m_hat
        nxt = project_to_simplex(nxt)

        cur_a = nxt

        # occasional progress report
        if it % 5000 == 0:
            print(f"[SGD] it {it:6d} | time {elapsed:6.1f}s | best {best_score:.6f} | β {beta:.1f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – fast incremental local search (swap / mass‑transfer)
# ----------------------------------------------------------------------


def local_search(
    a: np.ndarray,
    cur_score: float,
    deadline: float,
    init_temp: float = 0.02,
    step_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing / hill‑climb that works directly on the convolution.
    Two moves are used:
        * swapping two entries,
        * moving a small amount of mass from one index to another.
    The convolution is updated in O(n) by slice additions, allowing millions
    of trials under the time budget.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_conv = convolution_fft(cur_a, cur_a)          # full convolution, length 2n‑1
    best_a = cur_a.copy()
    best_score = cur_score

    start = time.time()
    it = 0
    accept_cnt = 0

    while time.time() < deadline:
        it += 1
        # linear temperature schedule
        fraction = (time.time() - start) / (deadline - start)
        temp = max(init_temp * (1.0 - fraction), 1e-9)

        # choose move type
        if np.random.rand() < 0.5:
            # ----- swap two positions ------------------------------------------------
            i, j = np.random.choice(n, size=2, replace=False)
            if cur_a[i] == cur_a[j]:
                continue
            delta_i = cur_a[j] - cur_a[i]          # change that will be added to position i
            delta_j = -delta_i                      # opposite change for j

            new_conv = cur_conv.copy()
            # add the linear part 2 * a * d
            new_conv[i:i + n] += 2.0 * delta_i * cur_a
            new_conv[j:j + n] += 2.0 * delta_j * cur_a
            # add the quadratic d*d part
            new_conv[2 * i] += delta_i * delta_i
            new_conv[2 * j] += delta_j * delta_j
            new_conv[i + j] += 2.0 * delta_i * delta_j

            new_score = 2.0 * n * float(new_conv.max())

            # acceptance test (simulated annealing)
            if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / temp):
                # perform swap
                cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
                cur_conv = new_conv
                cur_score = new_score
                accept_cnt += 1
                if new_score < best_score:
                    best_score = new_score
                    best_a = cur_a.copy()

        else:
            # ----- transfer a small amount of mass ------------------------------------
            src = np.random.randint(n)
            if cur_a[src] < 1e-12:
                continue
            dst = np.random.randint(n)
            if src == dst:
                continue

            # choose a fraction of the source mass
            delta = cur_a[src] * step_frac * np.random.rand()
            if delta <= 0.0:
                continue

            delta_src = -delta
            delta_dst = delta

            new_conv = cur_conv.copy()
            new_conv[src:src + n] += 2.0 * delta_src * cur_a
            new_conv[dst:dst + n] += 2.0 * delta_dst * cur_a
            new_conv[2 * src] += delta_src * delta_src
            new_conv[2 * dst] += delta_dst * delta_dst
            new_conv[src + dst] += 2.0 * delta_src * delta_dst

            new_score = 2.0 * n * float(new_conv.max())

            if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / temp):
                # perform transfer
                cur_a[src] += delta_src
                cur_a[dst] += delta_dst
                cur_conv = new_conv
                cur_score = new_score
                accept_cnt += 1
                if new_score < best_score:
                    best_score = new_score
                    best_a = cur_a.copy()

        # occasional re‑normalisation (numerical drift) and full recomputation
        if it % 10000 == 0:
            cur_a /= cur_a.sum()
            cur_conv = convolution_fft(cur_a, cur_a)
            cur_score = 2.0 * n * float(cur_conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            # adapt step size mildly
            acc_ratio = accept_cnt / 10000.0
            if acc_ratio > 0.3:
                step_frac = min(step_frac * 1.1, 0.1)
            elif acc_ratio < 0.1:
                step_frac = max(step_frac * 0.7, 0.005)
            accept_cnt = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – optional fine‑grained hill‑climb (same idea as before)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Very small stochastic hill‑climb that accepts only improving moves."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                      # pure swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # mass transfer
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand /= cand.sum()
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96
        steps += 1
    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the outer harness
    deadline = start_time + budget_s - 0.9
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 0. (Optional) use a previously discovered sequence as fallback
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"Fallback score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Choose a moderate length – 2048 works well; 4096 if time permits
    # --------------------------------------------------------------
    n = 2048
    if total_time > 850.0:
        n = 4096
    print(f"Using sequence length = {n}")

    # --------------------------------------------------------------
    # 2. Build a diverse pool of seeds and keep the best one
    # --------------------------------------------------------------
    seeds = []

    # deterministic shapes
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    # random Dirichlet draws (different concentrations)
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # include fallback if its length matches
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float))

    # evaluate seeds (exact objective) and keep the best so far
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)                     # guard against tiny drift
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"Initial seed improvement: score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Prepare the current best vector for further optimisation
    # --------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a /= cur_a.sum()
    cur_score = compute_score(cur_a)
    print(f"Starting score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # 4. Soft‑max gradient descent (major part of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + total_time * 0.55
    print("\n[Phase 1] Soft‑max gradient descent (Adam‑style)")
    cur_a, cur_score = softmax_gradient_descent(
        cur_a,
        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=5000.0,
        step0=0.001,
    )
    if cur_score < best_val:
        best_val = cur_score
        best_seq = cur_a.tolist()
        print(f"After GD improvement: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Incremental local search (swap / transfer moves)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n[Phase 2] Incremental local search (simulated annealing)")
        phase2_deadline = deadline - 30.0
        cur_a, cur_score = local_search(cur_a, cur_score, phase2_deadline)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
            print(f"After local search: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final quick hill‑climb polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:

        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=800_000)

            best_seq = cur_a.tolist()
            print(f"After polishing: best score = {best_val:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
