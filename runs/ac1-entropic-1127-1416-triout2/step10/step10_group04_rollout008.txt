"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Main ideas
----------
* Assemble a reasonably large pool of analytic seed families (uniform,
  central‑binomial, power‑law, symmetric beta, sine‑power, cosine‑power)
  and perform a quick grid search over their shape parameters.
* Take the best seed and run a continuation of smooth Adam optimisation
  on the log‑sum‑exp surrogate of the max‑convolution.  The temperature
  (β) is annealed from a modest value to a very large one (≈2000) so
  the surrogate approaches the true max.
* Refine the result with a projected sub‑gradient descent on the exact
  max‑convolution objective.
* Perform a few random‑restart sub‑gradient bursts to escape local
  minima.
* Finish with a cheap stochastic hill‑climb that transfers a small
  amount of mass between random positions (including occasional swaps)
  while staying on the simplex.
* All phases respect a wall‑clock deadline (default 1000 s) and keep
  the best sequence seen.  The final sequence is returned as a plain
  Python list of floats (already summing to 1).

The algorithm is deliberately different from the previous baseline
and explores a richer family of initial shapes together with a
continuation on the smooth surrogate – this has a good chance of
pushing the score below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector ``a`` (sum(a)==1):
        score = 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, normalised to sum 1 (monotone decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_symmetric_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^(-p)  (U‑shaped for p>0)."""
    i = np.arange(n, dtype=np.float64) + 1
    a = (i * (n - i + 1)) ** (-p)
    return project_to_simplex(a)


def generate_beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^(α-1)·(n-i)^(β-1).  General Beta(α,β) shape on a discrete grid.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return project_to_simplex(a)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin^p(π·(i+0.5)/n).  Vanishes at the edges, peak in the centre."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.sin(np.pi * x) ** p
    return project_to_simplex(a)


def generate_cosine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (1‑cos(π·i/(n‑1)))^p.  Similar to a raised‑cosine window."""
    if n == 1:
        return np.array([1.0])
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = (1.0 - np.cos(np.pi * x)) ** p
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam with large temperature schedule
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 0.5,
    beta_end: float = 2000.0,
    step0: float = 0.7,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed surrogate
        f_β(a) = (1/β)·log( Σ_i exp(β·conv_i) )
    with β annealed from ``beta_start`` to ``beta_end``.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # temperature schedule (exponential in iteration fraction)
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()

        # soft‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed objective
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)      # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)

        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact score check
        if it % 1000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    The sub‑gradient is taken at the currently active convolution index.
    """
    a = init_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # build sub‑gradient for the active index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape flat spots
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny mass moves)
# ----------------------------------------------------------------------


def hill_climb(
    start_seq,
    time_deadline,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local moves (swap or fractional transfer)
    with simplex projection after each move.
    """
    a = np.array(start_seq, dtype=np.float64)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    steps_done = 0
    while steps_done < max_steps and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # accelerate when improving
        else:
            step_scale *= 0.96                         # shrink when no improvement

        steps_done += 1

    return best_a, best_score
