"""
Advanced step‑function optimizer.

The search proceeds in three stages:

1. **Diversified seeding** – a large pool of hand‑crafted shapes
   (beta‑family, mixtures, spikes) and random Dirichlet draws.
   The best seed (according to the exact objective)
   initializes the optimisation.

2. **Adam on a smooth max‑surrogate** – we minimise
   `Lβ(a)=log Σ exp(β·conv(a,a))` with an Adam optimiser.
   The temperature β is annealed from a modest value to a very large
   one (≈ 10 000), so the surrogate progressively approaches the true
   max‑norm.  After each step the iterate is projected onto the
   probability simplex.

3. **Targeted sub‑gradient refinement + stochastic hill‑climb** – a
   classic sub‑gradient descent on the true max‑objective, followed by
   a cheap stochastic hill‑climbing that transfers a tiny fraction of
   mass between indices.  Both phases keep the best vector seen.

All work respects the 1000 s wall‑clock budget; intermediate progress
(and any improvement over the previous best 1.511131) is logged.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (O(n log n)).
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (`np.sum(a)==1`):
        score = 2·n·max(conv(a,a))
    The implementation uses FFT for speed.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin² shape a_i = sin²(pi·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p concentrates mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w = np.where(w > 0, w, 0.0)
    a = w / w.sum()
    return a


def generate_mixture_beta(n: int, p1: float, p2: float, w_mix: float) -> np.ndarray:
    """
    Convex combination of two beta‑family shapes.
    """
    a1 = generate_beta_family(n, p1)
    a2 = generate_beta_family(n, p2)
    a = w_mix * a1 + (1.0 - w_mix) * a2
    a /= a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: pick `k` positions, assign random positives, normalise."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def adam_surrogate(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 10000.0,
    lr: float = 0.005,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate Lβ.
    Returns the best vector found and its exact score.
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = start_score

    it = 0
    start = time.time()
    while time.time() < deadline:
        # convolution and max for the current iterate (for logging)
        conv = convolution_fft(a, a)
        max_conv = conv.max()

        # update the exact best if needed (cheap, no extra conv)
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max weights (stable)
        shifted = conv - max_conv                 # max becomes 0
        exp_vals = np.exp(beta * shifted)        # ≤ 1
        w = exp_vals / exp_vals.sum()            # distribution over 2n‑1 entries

        # gradient of the surrogate
        grad = 2.0 * beta * np.correlate(w, a, mode="valid")

        # Adam moment updates
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected update
        a = a - step
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(
                f"[Adam] it {it:6d} | time {elapsed:6.1f}s | β {beta:6.1f} | best {best_score:.6f}"
            )
    # final check
    conv = convolution_fft(a, a)
    cur_score = 2.0 * n * float(conv.max())
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score


def subgradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    step0: float = 0.18,
    jitter_freq: int = 800,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the true max‑objective.
    Uses the analytically known sub‑gradient at the maximal convolution index.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = start_score
    cur_a = a.copy()
    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / math.sqrt(t)
        cur_a = cur_a - step * g
        cur_a = project_to_simplex(cur_a)

        # occasional jitter to escape plateaus
        if t % jitter_freq == 0:
            cur_a += np.random.randn(n) * (step0 * 0.004 / math.sqrt(t))
            cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
    init_step_scale: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climb that exchanges or transfers mass
    between two coordinates.  Only moves that strictly improve the exact
    objective are kept (plus an adaptive step scale).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = init_step_scale
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        # either swap two entries or shift a fraction of mass
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        # renormalise to combat floating‑point drift
        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.95

        steps += 1
    return best_a, best_score

    except Exception:

    # --------------------------------------------------------------

    n = 2048

    seed_vectors = []
