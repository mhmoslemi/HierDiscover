```
def exponential_gradient_optimize(start_a, time_deadline, max_iters=500000, beta_start=5.0, beta_end=2000.0, eta0=1.0):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        # compute beta
        beta = beta_start * (beta_end / beta_start) ** ((t - 1) / (max_iters - 1))
        # compute conv and softmax weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # compute softmax weights
        exps = np.exp(beta * (conv - maxc))
        w = exps / exps.sum()
        # gradient of smoothed loss
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)  # length 3n-2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]
        # exponentiated gradient update
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        # renormalize to simplex
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum
        # update best
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # occasional print
        if t % 50000 == 0:
            print(f"[EG] iter {t}, cur score {score:.6f}, best {best_score:.6f}, beta={beta:.2f}")
    return best_a, best_score
```

assistantfinal**Solution Overview**

This implementation searches for a non‑negative step‑function `a` (a probability vector,
i.e. `sum(a)=1`) that minimises  

\[
\text{score}(a)=2\,n\;\max_k (a*a)_k\;,
\]

the same quantity used by ``evaluate_sequence``.
The main ideas are:

1. **Diverse analytic seeds** – uniform, central‑binomial, power‑law, geometric
   distributions and the previously supplied `height_sequence_1`.  
   The best of them is taken as the starting point.

2. **Exponentiated‑gradient optimisation on a smooth surrogate** –  
   minimise the *log‑sum‑exp* approximation of the max‑convolution  

   \[
   f_\beta(a)=\frac{1}{\beta}\log\!\Big(\sum_k e^{\beta\, (a*a)_k}\Big)
   \]

   using an exponentiated‑gradient (mirror‑descent) update that works natively
   on the simplex (no Euclidean projection is required).  
   The temperature `β` is annealed from a small value to a very large one
   (up to ≈ 5000) so the surrogate becomes tighter and the optimiser
   concentrates on the true worst lag.

3. **Projected sub‑gradient polishing** – after the smooth phase we
   directly minimise the original non‑smooth objective with a classic
   sub‑gradient method (the same update used in the reference code but
   with a slightly different step schedule).

4. **Random restarts** – a few short EG runs from Gaussian‑perturbed copies of
   the current best vector help to escape shallow basins.

5. **Simulated‑annealing hill‑climb** – a cheap stochastic move (transfer of a
   small amount of mass between two random positions) is accepted either
   if it improves the score or with a Metropolis probability that slowly
   cools down.  This final phase can still find tiny improvements that the
   deterministic gradients miss.

All phases are time‑controlled; the algorithm stops before the global time
budget (`budget_s`) and returns the best vector seen so far.

```python
"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1.  Build several analytic starters (uniform, central‑binomial,
    power‑law, geometric) and keep the best.
2.  Smooth optimisation via a log‑sum‑exp surrogate.
    The surrogate is minimised with an exponentiated‑gradient (mirror‑descent)
    update that stays on the probability simplex.
    Temperature β is annealed from a small to a very large value,
    turning the smooth loss into the true max‑convolution.
3.  Polishing with a projected sub‑gradient descent on the exact objective.
4.  A few random restarts (Gaussian perturbation + a short EG run) to
    explore new basins.
5.  Final stochastic hill‑climb with a simulated‑annealing acceptance rule.
    The best vector found at any moment is kept.

The whole search respects a wall‑clock deadline (default 1000 s) and
always returns a feasible sequence (non‑negative, sum = 1).  The code is
self‑contained, uses only NumPy and the provided ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum == 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, normalised to sum 1 (monotone decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exactly sum 1 if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth optimisation – exponentiated‑gradient on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the soft‑max weights  w_k = exp(beta*(conv_k - max(conv))) / Z .
    Numerical stability is ensured by subtracting the max before exponentiation.
    """
    maxc = conv.max()
    # Clip exponent to avoid under/overflow
    exps = np.exp(np.clip(beta * (conv - maxc), -700, 0.0))
    Z = exps.sum()
    if Z == 0.0:
        # all under‑flowed – fallback to a Kronecker at the argmax
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return exps / Z


def exponential_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 800_000,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the smoothed objective.
    The iterate always stays on the probability simplex.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # pre‑compute schedule factors
    log_beta_ratio = np.log(beta_end / beta_start) if beta_end != beta_start else 0.0

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # exponential schedule for β
        t_frac = (it - 1) / (max_iters - 1)
        beta = beta_start * np.exp(log_beta_ratio * t_frac)

        # convolution and soft‑max weights
        conv = convolution_fft(a, a)
        w = softmax_weights(conv, beta)

        # gradient of the smoothed loss: g = 2 * rev(a) * w (restricted to n entries)
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # exponentiated‑gradient update
        eta = eta0 / np.sqrt(it)
        # Clip gradient to avoid overflow in the exponent
        grad_clip = np.clip(grad, -50.0, 50.0)
        a = a * np.exp(-eta * grad_clip)

        # Re‑normalise to simplex
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # Keep the best vector seen so far
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional progress report
        if it % 100_000 == 0:
            print(
                f"[EG] iter {it}/{max_iters}, beta={beta:.1f}, "
                f"cur_score={cur_score:.6f}, best_score={best_score:.6f}"
            )

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (same idea as reference)
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.06,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))  # active lag

        # Sub‑gradient of conv_k w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional small random jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------


def simulated_annealing(
    start_seq,
    time_deadline: float,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb with Metropolis acceptance.
    Moves consist of transferring a small fraction of mass from one index
    to another.  Temperature decays linearly.
    """
    a = np.array(start_seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    # Temperature parameters
    T0 = 5e-3   # initial scale (in absolute score units)
    T_final = 1e-5

    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        # Linear temperature schedule
        frac = step / max_steps
        T = T0 * (1 - frac) + T_final * frac

        cand = best_a.copy()
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # Transfer up to 1 % of the donor's mass
        delta = (np.random.rand() * 0.01) * cand[i]
        cand[i] -= delta
        cand[j] += delta

        # Numerical safety – enforce non‑negativity and renormalise
        cand = np.clip(cand, 0.0, None)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / cand_sum

        cur_score = compute_score(cand)
        delta_score = cur_score - best_score

        # Metropolis acceptance
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / (T + 1e-12)):
            best_a = cand
            if cur_score < best_score:
                best_score = cur_score
                # occasional progress output
                if step % 50000 == 0:
                    print(f"[SA] step {step}, new best {best_score:.6f}")

    return best_a.tolist(), best_score
```
