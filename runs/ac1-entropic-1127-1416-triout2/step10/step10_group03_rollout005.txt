def adam_lp_descent(a0, deadline, p_start=2.0, p_end=80.0, lr=0.01):
    n=len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        # conv shape 2n-1
        # compute p based on elapsed ratio
        fraction = (time.time() - start) / (deadline - start)
        p = p_start + fraction * (p_end - p_start)
        # if p < 2, set to 2
        if p < 2: p = 2
        # compute weights
        conv_eps = np.clip(conv, 1e-15, None)
        w = conv_eps ** (p - 1)
        # gradient
        grad = 2.0 * p * np.correlate(w, a, mode='valid')
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step
        # small noise injection
        if t % 1000 == 0:
            noise = np.random.randn(n) * (lr * 0.1 / np.sqrt(t))
            a += noise
        # projection onto simplex
        a = project_to_simplex(a)
        # evaluate
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # occasional print
        if t % 5000 == 0:
            print(f"[Adam] it {t} | p {p:.1f} | best {best_score:.6f}")
    return best_a, best_score

def subgradient_refine(a, start_score, deadline, max_iter=400_000, step0=0.07):
    n = len(a)
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))
        # Build subgradient
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        if mask.any():
            g[mask] = 2.0 * cur_a[j[mask]]
        # step size
        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)
        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
        # occasional jitter to escape flat regions
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)
    return best_a, best_score

def hill_climb(a, deadline, max_steps=1_000_000):
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.05
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        # propose a random perturbation
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # mass transfer
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0:

                cand[i] -= delta
                cand[j] += delta

        cand /= cand.sum()
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96
        steps += 1
        if steps % 200000 == 0:

"""
Proposes a non‑negative step‑function (list of heights) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser works in three stages (all respecting the 1000 s wall‑clock
budget):

1. **Seeding** – a diverse collection of hand‑crafted shapes (sin²,
   symmetric beta‑polynomials, Gaussian, sparse spikes) together with
   random Dirichlet draws.  The best seed (exact objective) is kept.

2. **Annealed Lp‑Adam descent** – a smooth surrogate of the L∞‑norm,
   Lp(a)= ( Σ conv(a,a)^p )^{1/p}, is minimised by projected Adam
   optimisation.  The exponent p is annealed from 2 to a large value
   (≈100) so that the surrogate gradually focuses on the peak of the
   auto‑convolution.  After each step the vector is projected onto the
   probability simplex (non‑negative, sum = 1).

3. **Refinement** – a classic sub‑gradient descent on the true max‑objective
   followed by a stochastic hill‑climb that accepts only improving moves.

All phases keep the best sequence seen so far and finally return it as a
plain Python list.  The routine adapts the sequence length (2048 or 4096)
according to the available time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(pi·i/(n+1)), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian centred in the middle, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions, assign random weights,
    then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a

# Optimisation phases

def adam_lp_descent(
    init_a: np.ndarray,
    start_global: float,
    deadline: float,
    p_start: float = 2.0,
    p_end: float = 100.0,
    lr: float = 0.1,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smooth surrogate
        Lp(a) = ( Σ conv(a,a)^p )^{1/p}
    with p annealed linearly from `p_start` to `p_end`.
    Returns the best vector seen and its exact score.
    """
    a = init_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    inner_start = time.time()

    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)          # length 2n‑1
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # anneal exponent p
        elapsed = time.time() - inner_start
        total = deadline - inner_start
        frac = min(elapsed / total, 1.0)
        p = p_start + frac * (p_end - p_start)
        if p < 2.0:
            p = 2.0

        # gradient of the surrogate (ignoring the outer 1/p factor)
        conv_eps = np.clip(conv, 1e-15, None)
        w = conv_eps ** (p - 1.0)                 # shape (2n‑1)
        grad = 2.0 * p * np.correlate(w, a, mode="valid")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size decays with √iteration and with p
        step_scale = lr / (p * np.sqrt(t))
        step = step_scale * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # occasional tiny random jitter to help escape plateaus
        if t % 2000 == 0:
            a += np.random.randn(n) * (lr * 0.01 / np.sqrt(t))

        a = project_to_simplex(a)

        if t % 5000 == 0:
            print(f"[Adam] it {t:6d} | p {p:6.1f} | best {best_score:.6f}")

    return best_a, best_score


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the true max‑objective.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        if mask.any():
            g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # periodic jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts only improving moves.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.05
    steps = 0

    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand /= cand.sum()               # renormalise (numerical drift)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1
        if steps % 200_000 == 0:
            print(f"[Hill] step {steps:,} best {best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # choose a length that fits the budget
    n = 4096 if total_time > 800.0 else 2048
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 0. Collect a diverse set of seeds
    # ------------------------------------------------------------------
    seeds = []

    seeds.append(generate_deterministic(n))
    for p in np.linspace(-3.0, 3.0, 13):
        seeds.append(generate_beta_family(n, p))

    for sigma in (n / 8.0, n / 6.0, n / 5.0, n / 4.0):
        seeds.append(generate_gaussian(n, sigma))

    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    for k in (30, 60, 100):
        seeds.append(random_spike(n, k=k))

    # optional fallback from a previously discovered construction
    try:
        fallback = [float(x) for x in height_sequence_1]    # type: ignore
        if len(fallback) == n:
            seeds.append(np.array(fallback, dtype=float) / np.sum(fallback))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Pick the best hand‑crafted seed (exact objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for s in seeds:
        a = project_to_simplex(np.asarray(s, dtype=float))
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"Initial seed improvement: score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Annealed Lp‑Adam descent (major share of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + total_time * 0.55
    print("\n[Phase 1] Annealed Lp‑Adam descent")
    cur_a, cur_score = adam_lp_descent(
        best_seq,
        start_time,
        phase1_deadline,
        p_start=2.0,
        p_end=120.0,
        lr=0.12,
    )
    if cur_score < best_val:
        best_val = cur_score
        best_seq = cur_a.copy()
    print(f"After Adam: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        phase2_deadline = deadline - 30.0
        print("\n[Phase 2] Sub‑gradient refinement")
        cur_a, cur_score = subgradient_refine(
            best_seq, best_val, phase2_deadline,
            max_iter=600_000,
            step0=0.08,
        )
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.copy()
            print(f"After sub‑grad: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final greedy hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 3] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(best_seq, deadline, max_steps=1_500_000)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.copy()
            print(f"After polishing: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the result (the official `evaluate_sequence` will be
    #    called by the harness, we call it once for a final sanity check)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")
    return final_seq
