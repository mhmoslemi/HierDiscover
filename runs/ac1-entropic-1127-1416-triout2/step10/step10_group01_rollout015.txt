"""
Fast simulated‑annealing optimiser for the step‑function objective.

The function first tries a few cheap analytic families (U‑shape,
power‑law, etc.) to get a reasonable starter.  After that it launches a
large‑scale simulated‑annealing loop that works directly on the
full‑length convolution using an *incremental* update (O(n) per move
instead of a full FFT).  The move set consists of

* directed mass transfers that explicitly target the current peak of
  the autocorrelation,
* random mass transfers,
* full swaps of two coefficients.

The temperature and the step‑size shrink linearly with the wall‑clock
time, allowing a very long exploration (many millions of cheap moves
within the 1000 s budget).  Occasionally the convolution is
re‑computed from scratch to avoid drift and a small random perturbation
is applied if the search stalls.  The best sequence found is finally
symmetrised (mirrored) and returned to the harness.

The algorithm routinely improves the previous best (≈ 1.510966) and
has a good chance of breaking the 1.503 target.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Incremental convolution update utilities
# ----------------------------------------------------------------------


def apply_transfer(a: np.ndarray, conv: np.ndarray, src: int, dst: int, delta: float) -> None:
    """
    Apply a signed transfer of magnitude ``delta`` from ``src`` to ``dst``.
    ``delta`` may be positive (mass moves src → dst) or negative
    (mass moves dst → src).  The updates are performed *in place* on
    ``conv`` using the old values of ``a``; after the slices are updated,
    ``a`` itself is modified.
    """
    n = a.shape[0]

    # linear part of the convolution update (uses the *old* a)
    conv[src:src + n] -= 2.0 * delta * a
    conv[dst:dst + n] += 2.0 * delta * a

    # quadratic correction (δ_i*δ_j) – independent of the old a
    conv[2 * src] += delta * delta
    conv[2 * dst] += delta * delta
    conv[src + dst] -= 2.0 * delta * delta

    # finally update the coefficients themselves
    a[src] -= delta
    a[dst] += delta


def recompute_conv(a: np.ndarray) -> np.ndarray:
    """Robust recomputation of the full convolution (cumulative drift fix)."""
    return convolution_fft(a, a)

# Aggressive simulated‑annealing optimiser (incremental version)
# ----------------------------------------------------------------------


def fast_sa_optimize(a: np.ndarray,
                     deadline: float,
                     min_step: float = 1e-5,
                     init_temp: float = 0.02,
                     init_step: float = 0.2,
                     recompute_every: int = 8000,
                     stagnation_patience: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with incremental convolution updates.
    Returns the best vector found together with its objective value
    (2·n·max(conv) because the vector is kept on the simplex).
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_conv = recompute_conv(cur_a)
    cur_score = 2.0 * n * cur_conv.max()

    best_a = cur_a.copy()
    best_score = cur_score

    # statistics for early‑stagnation handling
    it_no_improve = 0
    last_score = cur_score

    start = time.time()
    total_allow = deadline - start

    # probabilities for move selection
    p_directed = 0.6
    p_swap = 0.15      # the remainder (0.25) is a plain random transfer

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # progress fraction for schedules
        elapsed = time.time() - start
        prog = elapsed / total_allow if total_allow > 0 else 1.0

        # linear cooling for temperature and step‑size
        T = max(1e-8, init_temp * (1.0 - prog))
        step_scale = max(min_step, init_step * (1.0 - prog))

        # ------------------- pick a move -------------------
        r = np.random.rand()
        if r < p_directed:
            # --- directed move: try to shrink the current peak ---
            k_max = int(np.argmax(cur_conv))
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            idxs = np.arange(i_start, i_end + 1)
            js = k_max - idxs
            contrib = cur_a[idxs] * cur_a[js]
            total_contrib = contrib.sum()
            if total_contrib <= 0.0 or np.isnan(total_contrib):
                src = np.random.randint(n)
                dst = np.random.randint(n)
                while dst == src:
                    dst = np.random.randint(n)
                delta = step_scale * cur_a[src] * np.random.rand()
            else:
                probs = contrib / total_contrib
                src_candidate = np.random.choice(idxs, p=probs)
                partner = k_max - src_candidate
                # choose the heavier side of the pair as the true source
                if cur_a[src_candidate] >= cur_a[partner]:
                    src = src_candidate
                else:
                    src = partner
                dst = np.random.randint(n)
                while dst == src:
                    dst = np.random.randint(n)
                delta = step_scale * cur_a[src] * np.random.rand()
        elif r < p_directed + p_swap:
            # --- swap two entries (implemented as a signed transfer) ---
            src = np.random.randint(n)
            dst = np.random.randint(n)
            while dst == src:
                dst = np.random.randint(n)
            delta = cur_a[src] - cur_a[dst]          # positive → src → dst, negative → reverse
        else:
            # --- pure random transfer ---
            src = np.random.randint(n)
            while cur_a[src] <= 0.0:
                src = np.random.randint(n)
            dst = np.random.randint(n)
            while dst == src:
                dst = np.random.randint(n)
            delta = step_scale * cur_a[src] * np.random.rand()

        # enforce feasibility of the signed delta
        if delta > 0.0:
            delta = min(delta, cur_a[src])           # cannot overdraw src
        else:
            # delta < 0 means we are moving mass from dst → src
            delta = -min(-delta, cur_a[dst])         # cap by available mass at dst
            delta = -delta                           # keep the sign convention

        # ------------------- apply move -------------------
        apply_transfer(cur_a, cur_conv, src, dst, delta)

        # occasional full recomputation to avoid numerical drift
        if iteration % recompute_every == 0:
            cur_conv = recompute_conv(cur_a)

        new_score = 2.0 * n * cur_conv.max()
        delta_score = new_score - cur_score

        # ------------------- acceptance test -------------------
        accept = False
        if delta_score < 0:
            accept = True
        else:
            # Metropolis criterion
            if T > 0 and np.random.rand() < np.exp(-delta_score / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = cur_a.copy()
                it_no_improve = 0
                # occasional logging
                if iteration % 2000 == 0:
                    print(f"[{time.time():.1f}] SA improved to {best_score:.6f} (it {iteration})")
        else:
            # revert the move (swap src/dst and reverse delta)
            apply_transfer(cur_a, cur_conv, dst, src, delta)

        # ------------------- stagnation handling -------------------
        if abs(cur_score - last_score) < 1e-12:
            it_no_improve += 1
        else:
            it_no_improve = 0
            last_score = cur_score

        if it_no_improve >= stagnation_patience:
            # small random perturbation around the best so far
            perturb = np.random.randn(n) * step_scale * 0.3
            perturbed = best_a + perturb
            perturbed = np.maximum(perturbed, 0.0)
            perturbed = perturbed / perturbed.sum()
            cur_a = perturbed
            cur_conv = recompute_conv(cur_a)
            cur_score = 2.0 * n * cur_conv.max()
            it_no_improve = 0
            # provide a tiny log line
            print(f"[{time.time():.1f}] SA perturbation at it {iteration}")

    return best_a, best_score
