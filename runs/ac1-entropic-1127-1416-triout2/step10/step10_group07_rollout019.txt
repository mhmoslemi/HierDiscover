"""
Propose a step‑function sequence that (approximately) minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The search runs inside the given time budget (≈1000 s).  The strategy is
a mixture of cheap analytic families, random Dirichlet draws and
multiple stages of continuous optimisation:

1.  Initialise a pool of candidates (uniform, power‑law, product‑family,
    exponential, random Dirichlet, sparse random patterns).  Keep the
    best one.

2.  Refine the best candidate with a smooth surrogate of the L∞‑norm:
    fβ(a) = (1/β)·log Σk exp(β·(a*a)k).  Adam is used on the
    soft‑max parametrisation a = softmax(z).  β is annealed from a
    small to a very large value (≥ 5·10⁴) so that the surrogate converges
    to the true maximum.

3.  Perform entropy‑mirror (multiplicative) sub‑gradient descent directly
    on the true max‑convolution.  The step‑size decays as 1/√t.

4.  A short simulated‑annealing hill‑climb is applied to escape shallow
    local minima.

All intermediate scores are printed for monitoring.  The final list is
returned as plain Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast, no Python loops)."""
    m = a.size + b.size - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families of probability vectors
# ----------------------------------------------------------------------


def uniform_family(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) , monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def two_cluster_start(n: int, sep: int = None) -> np.ndarray:
    """
    Very sparse distribution with two equally‑weighted clusters.
    The clusters are placed far apart (default ~ opposite ends).
    """
    if sep is None:
        sep = n // 2
    idx = np.random.choice(n, size=2, replace=False)
    # force a large separation if possible
    while abs(idx[0] - idx[1]) < sep:
        idx = np.random.choice(n, size=2, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 0.5
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Draw from a symmetric Dirichlet(α) (α = conc)."""
    vec = np.random.gamma(conc, 1.0, size=n)
    return vec / vec.sum()


def random_sparse_start(n: int, sparsity: float = None) -> np.ndarray:
    """Sparse uniform vector (≈ 5 % non‑zeros)."""
    if sparsity is None:
        sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed optimisation (log‑sum‑exp surrogate) via Adam
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 100_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise   fβ(a) = (1/β)·log Σk exp(β·(a*a)k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β is exponentially increased from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # ---- smoothed weights over shifts ----
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))   # stabilised
        w = exp_w / exp_w.sum()                # softmax over shift indices

        # ---- gradient of fβ w.r.t. a ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        # extract the part that aligns with a (centre of length‑2n+1 conv)
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # ---- back‑prop through soft‑max ----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasional exact evaluation ----
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(
                    f"[{time.time():.1f}]   Adam iter {it:7d} – new best = {best_score:.6f}"
                )

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy‑mirror) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 1.0,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror (multiplicative) sub‑gradient descent for
    2·n·max(a*a).  Step size η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)               # multiplicative mirror update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time():.1f}]   Mirror step {t:7d} – new best = {best_score:.6f}"
                )

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Simple low‑temperature simulated‑annealing walk on the simplex.
    Swaps or mass‑transfer moves are attempted.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The algorithm respects the given time budget
    (≈ 1000 s) and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # -------------------------------------------------------------
    # Choose length (bigger n gives a tighter theoretical bound)
    # -------------------------------------------------------------
    if budget_s >= 850:
        n = 65536
    elif budget_s >= 600:
        n = 32768
    elif budget_s >= 350:
        n = 16384
    else:
        n = 8192
    print(f"[{time.time():.1f}] Using length n = {n}")

    # -------------------------------------------------------------
    # Global best bookkeeping
    # -------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    def maybe_update(vec: np.ndarray, tag: str = "") -> None:
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<20} = {best_score:.6f}")

    # -------------------------------------------------------------
    # Baseline candidates
    # -------------------------------------------------------------
    maybe_update(uniform_family(n), "uniform")
    maybe_update(symmetric_product_family(n, 0.5), "sym‑α=0.5")
    maybe_update(left_heavy_power_family(n, 1.0), "pow‑β=1.0")
    maybe_update(exponential_decay_family(n, 1e-3), "exp‑γ=1e-3")

    # -------------------------------------------------------------
    # Scan analytic families on a coarse grid
    # -------------------------------------------------------------
    for alpha in np.linspace(0.2, 2.0, 19):
        if time.time() >= deadline - 30.0:
            break
        maybe_update(symmetric_product_family(n, alpha), f"sym‑α={alpha:.2f}")

    for beta in np.linspace(0.5, 4.0, 15):
        if time.time() >= deadline - 30.0:
            break
        maybe_update(left_heavy_power_family(n, beta), f"pow‑β={beta:.2f}")

    for gamma in np.linspace(0.0, 4.0, 17):
        if time.time() >= deadline - 30.0:
            break
        maybe_update(exponential_decay_family(n, gamma), f"exp‑γ={gamma:.2f}")

    # -------------------------------------------------------------
    # Random Dirichlet draws (wide concentration range)
    # -------------------------------------------------------------
    conc_vals = np.logspace(-3, 1, 20)   # 0.001 … 10
    for conc in conc_vals:
        for _ in range(4):
            if time.time() >= deadline - 30.0:
                break
            maybe_update(random_dirichlet_start(n, conc), f"dir‑α={conc:.3f}")

    # -------------------------------------------------------------
    # Random sparse starts (few non‑zeros)
    # -------------------------------------------------------------
    for _ in range(6):
        if time.time() >= deadline - 30.0:
            break
        maybe_update(random_sparse_start(n), "sparse")

    # -------------------------------------------------------------
    # Two‑cluster seeds
    # -------------------------------------------------------------
    for _ in range(4):
        if time.time() >= deadline - 30.0:
            break
        maybe_update(two_cluster_start(n), "two‑cluster")

    # -------------------------------------------------------------
    # 1st optimisation phase – smooth‑max Adam (many restarts)
    # -------------------------------------------------------------
    while time.time() < deadline - 250.0:
        start_vec = random_sparse_start(n)
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 200.0,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=100_000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}]   new best after Adam = {best_score:.6f}")

    # -------------------------------------------------------------
    # 2nd optimisation – high‑β Adam starting from the current best
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}]   refinement – high‑β Adam")
        a_ref, sc_ref = adam_smooth_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 90.0,
            max_iters=120_000,
            beta_start=100_000.0,
            beta_end=300_000.0,
            lr=0.012,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.tolist()
            print(f"[{time.time():.1f}]   after high‑β Adam = {best_score:.6f}")

    # -------------------------------------------------------------
    # Mirror sub‑gradient descent on the true max‑objective
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}]   start mirror sub‑gradient")
        a_sub, sc_sub = mirror_subgrad_optimize(

            print(f"[{time.time():.1f}]   after mirror sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------------------
    # Symmetrisation (often improves the max‑convolution)
    # -------------------------------------------------------------
    if best_seq is not None:

    # -------------------------------------------------------------
    # Final tiny simulated‑annealing hill‑climb

    if best_seq is not None and time.time() < deadline - 5.0:

        a_final, sc_final = hill_climb_sa(

            print(f"[{time.time():.1f}]   SA improved best = {best_score:.6f}")

    # -------------------------------------------------------------
    # Safety fallback (very unlikely to be used)
    # -------------------------------------------------------------
    if best_seq is None:

        best_score = compute_score(np.array(best_seq))

    # -------------------------------------------------------------
    final_score = compute_score(np.array(best_seq))
    print(
        f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}"
    )
    return best_seq
