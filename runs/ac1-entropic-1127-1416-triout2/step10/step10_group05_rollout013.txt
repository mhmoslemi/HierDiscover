"""
Candidate generator for the “step‑function” challenge.

Strategy
--------
* Choose the longest admissible vector length (up to 32768) – a larger
  n gives more freedom to flatten the self‑convolution.
* Initialise a small pool of promising shapes:
  – uniform,
  – a dense grid of exponential / power‑law families,
  – a 2‑parameter discrete Beta family,
  – a few random Dirichlet draws.
* Keep the best few (default 5) after this cheap scan.
* For each seed run three optimisation phases, always respecting the
  global deadline (1000 s):
  1. **Smooth max‑descent** – minimise a log‑sum‑exp surrogate of the
     maximal convolution entry using a KL‑mirror (exponentiated‑gradient)
     update.  The smoothing parameter β grows from ≈0.5 to several
     thousand, allowing a smooth start and a tight final approximation.
  2. **Exact max‑descent** – sub‑gradient descent on the true ℓ∞‑objective
     (the entry with maximal convolution value).  The multiplicative KL
     update is kept, but the iteration budget is larger than in the
     reference implementation.
  3. **Targeted hill‑climb** – greedy mass transfer from the index that
     contributes most to the current worst convolution entry to a random
     destination.  The step size adapts automatically; the loop runs
     until time runs out (hundreds of thousands of successful moves).
* The best sequence found across all seeds is returned.

All updates keep the vector inside the probability simplex
(`project_to_simplex`), so the evaluation is scale‑invariant and matches
the behaviour of `evaluate_sequence`.  The routine prints progress and
the best score so far, and finally returns a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector already
    normalised to sum = 1.  Returns 2·n·max(a⋆a) where n = len(a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) with lam≥0, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam == 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} with p≥0, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p == 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ((i+0.5)/n)^{a-1}·(1-(i+0.5)/n)^{b-1}
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


# ----------------------------------------------------------------------
# Grid searches for the simple families
# ----------------------------------------------------------------------


def scan_exponential_and_power(n: int, deadline: float):
    """Exhaustive coarse scan of exponential and power families."""
    best_a, best_sc = None, float("inf")
    # exponential (lam from 0 to ~0.01 – dense near 0)
    lam_grid = np.concatenate((np.array([0.0]),
                               np.logspace(-6, -1, 250)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    # power‑law (p from 0 to 6)
    p_grid = np.linspace(0.0, 6.0, 250)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def scan_beta_family(n: int, deadline: float):
    """Modest 2‑parameter Beta family scan (α,β)."""
    best_a, best_sc = None, float("inf")
    a_grid = np.concatenate((np.logspace(-0.8, -0.1, 12),
                             np.logspace(-0.1, 0.2, 12),
                             np.linspace(0.3, 3.0, 12)))
    b_grid = a_grid.copy()
    for a in a_grid:
        for b in b_grid:
            if time.time() >= deadline:
                return best_a, best_sc
            a_vec = beta_shape(n, a, b)
            sc = compute_score(a_vec)
            if sc < best_sc:
                best_sc, best_a = sc, a_vec.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 800_000,
                        beta_start: float = 0.5,
                        beta_end: float = 8000.0,
                        eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent on a log‑sum‑exp surrogate of max(conv).
    β grows exponentially from `beta_start` to `beta_end`.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable soft‑max weights
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of smooth max
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * grad), 1e-15)
        a /= a.sum()

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def max_descent_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 1_200_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true ℓ∞‑objective (max convolution entry)
    using the same KL‑mirror update as in the smooth phase.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient of the active entry
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * grad), 1e-15)
        a /= a.sum()

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def targeted_hill_climb(start_seq: np.ndarray,
                        deadline: float,
                        max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Greedy move of a small fraction of the mass from the index that
    contributes most to the current worst convolution entry.  The step
    size adapts automatically; only improving moves are accepted.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.10
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        # contributions to the active index
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k_max - idx]

        i_max = idx[np.argmax(contrib)]

        # pick a random destination (avoid trivial move)
        j = np.random.randint(n)
        if j == i_max:
            continue

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        # the move preserves the sum, only negativity needs fixing
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()          # re‑normalise to stay on simplex

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_sc

    scan_deadline = deadline - 250.0
    a_scan, sc_scan = scan_exponential_and_power(n, scan_deadline)
    if a_scan is not None and sc_scan < best_score:
        best_score, best_seq = sc_scan, a_scan.tolist()
        print(f"[{time.time():.1f}] after exp/power scan = {best_score:.6f}")

    a_beta, sc_beta = scan_beta_family(n, scan_deadline)
    if a_beta is not None and sc_beta < best_score:
        best_score, best_seq = sc_beta, a_beta.tolist()
        print(f"[{time.time():.1f}] after beta scan = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Random Dirichlet starts (light diversification)
    # -------------------------------------------------
    for _ in range(6):
        if time.time() >= scan_deadline:
            break
        v = np.random.rand(n)
        v /= v.sum()
        sc = compute_score(v)
        if sc < best_score:
            best_score, best_seq = sc, v.tolist()
            print(f"[{time.time():.1f}] random start improved to {best_score:.6f}")

    # -------------------------------------------------
    # 3. Assemble a short list of promising seeds
    # -------------------------------------------------
    # evaluate the pool again and keep the top 5 vectors
    pool = []
    # uniform baseline
    pool.append(np.full(n, 1.0 / n))
    # include the best from the scans (if any)
    if best_seq is not None:
        pool.append(np.array(best_seq, dtype=float))
    # a few extra random starts
    for _ in range(3):
        v = np.random.rand(n)
        v /= v.sum()
        pool.append(v)

    # keep the five with smallest exact scores
    scored = [(compute_score(v), v) for v in pool]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:5]]
    print(f"[{time.time():.1f}] selected {len(top_seeds)} seeds for refinement")

    # -------------------------------------------------
    # 4. Optimisation phases for each seed
    # -------------------------------------------------
    for seed_id, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline - 30.0:
            break
        print(f"[{time.time():.1f}] seed {seed_id} – start optimisation")
        # Phase A: smooth max descent
        a_smooth, sc_smooth = exp_smooth_optimize(
            seed_vec,
            deadline - 20.0,
            max_iters=800_000,
            beta_end=8000.0,
            eta0=0.6,
        )
        # Phase B: exact max descent (longer budget)
        a_max, sc_max = max_descent_optimize(
            a_smooth,
            deadline - 10.0,
            max_iters=1_200_000,
            eta0=0.5,
        )
        # Phase C: targeted hill‑climb polishing
        a_final, sc_final = targeted_hill_climb(
            a_max,
            deadline,
            max_steps=2_000_000,
        )
        final_score = evaluate_sequence(a_final.tolist())
        print(f"[{time.time():.1f}] seed {seed_id} – final score = {final_score:.6f}")
        if final_score < best_score:
            best_score, best_seq = final_score, a_final.tolist()
            print(f"[{time.time():.1f}] *** new global best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Safeguard: tiny uniform vector if everything failed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # 6. Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
