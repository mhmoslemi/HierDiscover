"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s total)
--------------------------------
1. Choose a length n (larger n gives a lower theoretical bound).  
   For a full budget we use n = 32768, otherwise a smaller power‑of‑2.
2. Scan a few low‑dimensional analytic families that are known to give
   “U‑shaped’’ distributions (mass near the ends, low in the middle):
        * symmetric product   a_i ∝ [(i+½)(n‑½‑i)]^(‑α)
        * symmetric sum of power laws
          a_i ∝ (i+½)^(‑β) + (n‑½‑i)^(‑β)
        * symmetric sum of exponentials
          a_i ∝ exp(‑γ·i) + exp(‑γ·(n‑1‑i))
   For each family we do a quick grid search, then a 1‑D golden‑section
   refinement of the free parameter.  The best vector obtained is kept.
3. A short local optimisation phase refines the best candidate with an
   *entropy‑mirror sub‑gradient method* that averages the gradient over
   all (near‑)maximal convolution shifts (instead of just a single argmax).
   The step size η_t = η₀ / √t, η₀ is set to 1.0.
4. Symmetrise the result (a ← (a+a[::-1])/2) and project onto the simplex.
5. A tiny stochastic hill‑climb finishes the run – random mass transfers
   are accepted only if they improve the exact objective.
6. The final sequence (plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast, no Python loops)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective when ``sum(a) == 1``:
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (all vectors are normalised to sum = 1)
# ----------------------------------------------------------------------


def sym_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α)   (U‑shaped, symmetric)."""
    i = np.arange(n, dtype=np.float64)
    logw = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logw)
    a = w / w.sum()
    return a


def sym_sum_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) + (n‑½‑i)^(‑β)   (symmetric sum of two power laws)."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-beta * np.log(i + 0.5))
    right = np.exp(-beta * np.log(n - 0.5 - i))
    w = left + right
    a = w / w.sum()
    return a


def sym_sum_exp_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(‑γ·i) + exp(‑γ·(n‑1‑i))   (symmetric sum of exponentials)."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = np.exp(-gamma * (n - 1 - i))
    w = left + right
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# 1‑D golden‑section search (parameter optimisation)
# ----------------------------------------------------------------------


def golden_section_search(
    func,
    lo: float,
    hi: float,
    tol: float = 1e-4,
    max_iter: int = 30,
    deadline: float = 1e20,
):
    """
    Minimise ``func(x)`` on [lo,hi] using golden‑section.
    Returns (x_opt, f_opt).  Stops early if the global deadline is passed.
    """
    phi = (np.sqrt(5) - 1) / 2  # ≈0.618
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    fc = func(c)
    fd = func(d)

    it = 0
    while it < max_iter and time.time() < deadline:
        if fc < fd:
            hi, fd = d, fc
            d = c
            c = hi - phi * (hi - lo)
            fc = func(c)
        else:
            lo, fc = c, fd
            c = d
            d = lo + phi * (hi - lo)
            fd = func(d)
        if abs(hi - lo) < tol:
            break
        it += 1

    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Refined entropy‑mirror sub‑gradient descent (averaged active set)
# ----------------------------------------------------------------------


def refined_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 1.0,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Mirror (multiplicative) sub‑gradient descent on the exact max‑objective.
    The gradient is averaged over all convolution indices that are
    within numerical tolerance of the current maximum.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # treat all shifts that are essentially maximal
        tol = maxc * 1e-12 + 1e-15
        active = np.where(conv >= maxc - tol)[0]
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)

        # sub‑gradient for the averaged active set
        grad = 2.0 * convolution_fft(a[::-1], w)

        eta = step0 / np.sqrt(t)          # classic 1/√t schedule
        a = a * np.exp(-eta * grad)       # multiplicative update

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] refined sub‑grad step {t:7d} – new best = {best_score:.6f}")

    # final check
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climb (low‑temperature simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 500_000,
    init_temp_factor: float = 5e-5,
) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature SA walk on the simplex.
    Small random mass transfers are kept only if they improve the exact score.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * init_temp_factor
    T = T0
    step_scale = 0.08

    for k in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # move a small fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise (simple because only two entries changed)
        s = cand.sum()
        if s == 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / s

        cand_score = compute_score(cand)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - k / max_steps)

    return best_a, best_score

# Main driver

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose length
    # ------------------------------------------------------------------
    if budget_s >= 950:
        n = 32768
    elif budget_s >= 700:
        n = 16384
    elif budget_s >= 500:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] Using length n = {n}")

    # ------------------------------------------------------------------
    # Helper to keep the globally best vector
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    def maybe_update(vec: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<20} = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Very cheap baseline candidates
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    maybe_update(uniform, "uniform")

    # random Dirichlet draws (a few, just to diversify)
    for conc in (0.05, 0.2, 1.0, 5.0):
        if time.time() >= deadline - 30.0:
            break
        rand = np.random.gamma(conc, 1.0, size=n)
        rand = rand / rand.sum()
        maybe_update(rand, f"dir‑α={conc}")

    # ------------------------------------------------------------------
    # 2. Grid search over analytic families
    # ------------------------------------------------------------------
    # Helper that returns score for a given parameter and family
    def score_for_param(family_func, param):
        vec = family_func(n, param)
        return compute_score(vec)

    # ---- symmetric product family ------------------------------------
    if time.time() < deadline - 5.0:
        alphas = np.linspace(0.0, 6.0, 61)
        for a in alphas:
            if time.time() >= deadline - 5.0:
                break
            vec = sym_product_family(n, a)
            maybe_update(vec, f"prod‑α={a:.2f}")

        # refine around best α using golden‑section
        # locate the best α from the coarse grid
        best_alpha = min(alphas, key=lambda x: score_for_param(sym_product_family, x))
        lo, hi = max(0.0, best_alpha - 1.0), min(6.0, best_alpha + 1.0)
        opt_alpha, _ = golden_section_search(
            lambda x: score_for_param(sym_product_family, x),
            lo,
            hi,
            tol=1e-5,
            deadline=deadline - 2.0,
        )
        vec = sym_product_family(n, opt_alpha)
        maybe_update(vec, f"prod‑α*={opt_alpha:.5f}")

    # ---- symmetric sum of power laws --------------------------------
    if time.time() < deadline - 5.0:
        betas = np.linspace(0.0, 6.0, 61)
        for b in betas:
            if time.time() >= deadline - 5.0:
                break
            vec = sym_sum_power_family(n, b)
            maybe_update(vec, f"sum‑β={b:.2f}")

        best_beta = min(betas, key=lambda x: score_for_param(sym_sum_power_family, x))
        lo, hi = max(0.0, best_beta - 1.0), min(6.0, best_beta + 1.0)
        opt_beta, _ = golden_section_search(
            lambda x: score_for_param(sym_sum_power_family, x),
            lo,
            hi,
            tol=1e-5,
            deadline=deadline - 2.0,
        )
        vec = sym_sum_power_family(n, opt_beta)
        maybe_update(vec, f"sum‑β*={opt_beta:.5f}")

    # ---- symmetric sum of exponentials -------------------------------
    if time.time() < deadline - 5.0:
        gammas = np.linspace(0.0, 5.0, 51)
        for g in gammas:
            if time.time() >= deadline - 5.0:
                break
            vec = sym_sum_exp_family(n, g)
            maybe_update(vec, f"exp‑γ={g:.2f}")

        best_gamma = min(gammas, key=lambda x: score_for_param(sym_sum_exp_family, x))
        lo, hi = max(0.0, best_gamma - 0.8), min(5.0, best_gamma + 0.8)
        opt_gamma, _ = golden_section_search(
            lambda x: score_for_param(sym_sum_exp_family, x),
            lo,
            hi,
            tol=1e-5,
            deadline=deadline - 2.0,
        )
        vec = sym_sum_exp_family(n, opt_gamma)
        maybe_update(vec, f"exp‑γ*={opt_gamma:.5f}")

    # ------------------------------------------------------------------
    # 3. Refined sub‑gradient optimisation on the best candidate so far
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] starting refined sub‑gradient descent")
        a_sub, sc_sub = refined_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            step0=1.0,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after refined sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional symmetrisation (most optimal solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        maybe_update(a_sym, "symmetrised")

    # ------------------------------------------------------------------
    # 5. Tiny hill‑climb (low‑temperature SA) to escape shallow pits
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Fallback in the unlikely case that nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 7. Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return best_seq
