"""
Propose a step‐function that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where a is a non‑negative vector (the “heights”).  The algorithm is
a three‑stage pipeline:

1.  Quick sweep of a few analytic families (product, power, beta‑type,
    sigmoid, exponential‑U) to obtain a decent seed.
2.  A low‑dimensional evolutionary search: a is parametrised as
    a = softmax( B·θ ) (B contains the first m DCT basis functions).
    A simple (μ,λ)‑ES with a constantly adapted mutation step explores the
    space of smooth shapes while keeping the vector on the simplex.
3.  The best candidate from stage 2 is refined with a smoothed‑max
    Adam optimiser (the same routine used in the reference code) and
    a final symmetrisation.  All evaluations use a fast FFT convolution
    and the scaling invariance of the objective, so only normalised
    vectors (∑a = 1) are needed.

The whole search respects the 1000 s wall‑clock budget and returns a
plain list of floats that can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Fast convolution utilities (used by the objective)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (both inputs 1‑D)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simple simplex projection (used by Adam optimisation)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for a quick coarse sweep)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density ∝ (i+½)^{p‑1} (n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    # avoid infinities for extreme p/q
    w = np.where(np.isfinite(w), w, 0.0)
    return w / w.sum()


def sigmoid_family(n: int, kappa: float) -> np.ndarray:
    """Monotone sigmoid a_i ∝ 1/(1+exp(k·(i‑mid)))."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = 1.0 / (1.0 + np.exp(kappa * (i - mid)))
    return w / w.sum()


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (used for final refinement)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 5.0,
                         beta_end: float = 1200.0,
                         lr: float = 0.018,
                         sym_interval: int = 4000) -> Tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max (conv) via Adam on the
    soft‑max parametrisation a = softmax(z).  Periodically the iterate
    is symmetrised and re‑projected onto the simplex.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smoothed max (log‑sum‑exp)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a (see derivation in reference)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 1000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Evolutionary search in a low‑dimensional DCT basis
# ----------------------------------------------------------------------


def evolve_in_dct_basis(n: int,
                        dim: int,
                        deadline: float,
                        init_best: np.ndarray = None) -> Tuple[np.ndarray, float]:
    """
    (μ,λ)‑ES where a = softmax( B·θ ), B contains the first ``dim``
    DCT‑II basis functions on a grid of length ``n``.  The population
    size is modest (≈40) so that many generations fit into the allotted
    time.  The routine returns the best normalised vector found.
    """
    # Pre‑compute the basis matrix B (n × dim)
    i = np.arange(n, dtype=np.float64)
    k = np.arange(dim, dtype=np.float64)
    B = np.cos(np.pi * np.outer(i + 0.5, k) / n)  # DCT‑II, no scaling needed

    # Evolutionary parameters
    lam = max(30, 4 + int(3 * np.log(dim)))   # population size
    mu = lam // 2
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights /= np.sum(weights)

    # Initialise the search distribution
    theta_mean = np.zeros(dim)
    sigma = 0.5

    best_vec = None
    best_score = float("inf")

    # If an external starting vector is supplied, initialise the mean
    # to the corresponding log‑probabilities (soft‑max inverse).
    if init_best is not None:
        # avoid log(0)
        eps = 1e-12
        theta_mean = np.linalg.lstsq(B, np.log(init_best + eps), rcond=None)[0]

    while time.time() < deadline - 5.0:
        # Sample λ candidates
        thetas = theta_mean + sigma * np.random.randn(lam, dim)   # (lam, dim)

        # Compute the logits for all candidates at once: (n, lam)
        logA = B @ thetas.T
        # Numeric stabilisation per column
        logA -= np.max(logA, axis=0, keepdims=True)
        A = np.exp(logA)
        # Normalise each column to sum‑to‑one
        sums = np.sum(A, axis=0, keepdims=True)
        A /= sums

        # Evaluate the population
        scores = np.empty(lam)
        for idx in range(lam):
            scores[idx] = compute_score(A[:, idx])

        # Keep global best
        min_idx = int(np.argmin(scores))
        if scores[min_idx] < best_score:
            best_score = float(scores[min_idx])
            best_vec = A[:, min_idx].copy()
            print(f"[{time.time():.1f}] ES new global best {best_score:.6f}")

        # Select the μ best individuals and recombine
        best_mu_idxs = np.argsort(scores)[:mu]
        selected = thetas[best_mu_idxs]                 # (μ, dim)
        theta_mean = np.dot(weights, selected)          # weighted mean

        # Simple step‑size adaptation (1/5‑rule like)
        # If a reasonable fraction of the offspring improved over the
        # previous generation's best, increase σ, otherwise decrease.
        improve_frac = np.mean(scores < best_score + 1e-12)
        if improve_frac > 0.2:
            sigma *= 1.02
        else:
            sigma *= 0.98
        sigma = max(0.02, min(sigma, 2.0))

    if best_vec is None:
        # fallback – uniform distribution
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = compute_score(best_vec)

    return best_vec, best_score

        best_score = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # Final reporting (using the real evaluate_sequence for consistency)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
