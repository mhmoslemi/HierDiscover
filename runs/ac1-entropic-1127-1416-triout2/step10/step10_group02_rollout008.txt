def full_convolve_fft(x,y):
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n-1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:n]
    return conv

def gradient_descent(a, end_time):

    total = end_time - time_start
    # hyperparameters
    alpha_start = 1.0
    alpha_end = 200.0
    step_start = 0.5
    step_end = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    v = np.zeros_like(a)
    best_a = a.copy()

    t = 0
    while time.time() < end_time:
        t += 1
        # compute conv
        conv = full_convolution(a)  # full conv length 2n-1
        max_conv = conv.max()
        # compute smoothing weight
        # time fraction
        elapsed = (time.time() - time_start) / total
        alpha = alpha_start + elapsed * (alpha_end - alpha_start)
        # stable softmax
        exp_arg = alpha * (conv - max_conv)
        # clip exp_arg to avoid overflow: cap at maybe 700? but small.
        w = np.exp(exp_arg)

            w = np.zeros_like(conv)

        # but continue
    return best_a

        # compute convolution of a with itself
        conv = full_convolution(a)
        max_c = conv.max()
        # smoothing schedule
        elapsed = (time.time() - start) / total
        alpha = 1.0 + 199.0 * elapsed  # from 1 to 200
        # compute softmax weights
        exp_arg = alpha * (conv - max_c)
        # Clip extreme values to avoid overflow
        # exp on very negative yields zero; fine.
        # Possibly underflow to zero, but then sum may be zero.
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback: weight only max index
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum
        # gradient
        g_full = full_convolve_fft(w, a)  # length 3n-2
        g = 2.0 * g_full[n-1:2*n-1]

        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step size schedule: start higher, end lower

        if val < best_val:

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Highlights
----------
* Uses a *symmetrised power‑law* family as a cheap global start.
* Refines the shape with an **Adam‑style projected gradient descent** on a
  *soft‑max* approximation of the max‑convolution (the true objective is
  2·n·max(a∗a) with sum(a)=1).  The soft‑max parameter α is ramped from
  1 to ~200, turning the smooth surrogate into the true max.
* Finishes with a cheap stochastic hill‑climb that moves small amounts of
  mass between coordinates.
* All steps respect a global 1000 s wall‑clock budget and constantly keep
  the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two arbitrary vectors using FFT.
    """
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a normalised vector.
    (The function works for any a; it normalises internally.)
    """
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-9:
        return float('inf')
    return float(2.0 * n * max_c / (sum_a ** 2))


# ----------------------------------------------------------------------
# Families for fast global search
# ----------------------------------------------------------------------


def sym_power_law(n: int, p: float) -> np.ndarray:
    """Symmetrised decreasing power‑law: high mass at both ends, low in the middle."""
    base = np.arange(1, n + 1, dtype=float) ** (-p)
    sym = (base + base[::-1]) / 2.0
    sym /= sym.sum()
    return sym


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised Beta(α+1,β+1) density on {0,…,n‑1}."""
    x = np.linspace(0, 1, n)
    vals = (x ** a) * ((1 - x) ** b)
    # avoid exact zeros – they would be frozen forever after projection
    vals = np.maximum(vals, 1e-12)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a smoothed max‑convolution
# ----------------------------------------------------------------------


def gradient_descent_adam(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Refines ``a0`` by minimising  max(a*a) using a smooth soft‑max surrogate.
    The algorithm keeps ``a`` on the simplex (sum=1, a_i ≥ 0) and employs
    an Adam update rule with a linearly increasing soft‑max temperature.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    start = time.time()
    total = max(deadline - start, 1e-6)

    t = 0
    while time.time() < deadline:
        t += 1
        # --------------------------------------------------------------
        # 1. current convolution and a (soft‑max) weight vector
        # --------------------------------------------------------------
        conv = full_convolution(a)                     # length 2n‑1
        max_c = conv.max()
        # linear schedule for the temperature α (1 → 200)
        elapsed = (time.time() - start) / total
        alpha = 1.0 + 199.0 * elapsed

        # numerically stable soft‑max centred at the maximum value
        exp_arg = alpha * (conv - max_c)               # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback: keep only the truly maximal entry
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum                                     # shape (2n‑1,)

        # --------------------------------------------------------------
        # 2. gradient of the soft‑max w.r.t. a
        #    g = 2 * (w ⋆ a)_{valid}   (valid = length‑n slice)
        # --------------------------------------------------------------
        g_full = full_convolve_fft(w, a)               # length 3n‑2
        g = 2.0 * g_full[n - 1:2 * n - 1]               # shape (n,)

        # --------------------------------------------------------------
        # 3. Adam update + projection onto simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # step size slowly decays from 0.5 → 0.01
        step = 0.5 * (1 - elapsed) + 0.01 * elapsed
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # keep the iterate feasible
        a = project_to_simplex(a)

        # --------------------------------------------------------------
        # 4. bookkeeping
        # --------------------------------------------------------------
        curr_val = compute_score(a)
        if curr_val < best_val:
            best_val = curr_val
            best_a = a.copy()
            print(f"[gd] iter {t:6d} | val {curr_val:.6f} | α {alpha:6.1f} | step {step:.5f}")

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 500_000) -> np.ndarray:
    """
    Tiny stochastic moves (move a fraction of mass from i to j) with greedy
    acceptance.  Operates on the simplex and keeps the best solution found.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05          # initial fraction of mass moved
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_no_improve:
        i, j = np.random.randint(n, size=2)
        if i == j:

        if no_improve > 200_000:

            step_scale = max(step_scale * 0.5, 1e-4)
            no_improve = 0

    return best_a
