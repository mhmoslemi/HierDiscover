    # also some left-heavy

        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc; best_seq = vec
    # also uniform for baseline
    vec = np.full(n, 1.0/n)
    sc = compute_score(vec)
    if sc < best_score:
        best_score = sc; best_seq = vec

    # Ensure sum 1
    best_seq = project_to_simplex(best_seq)

    f = best_seq.copy()
    # LP iterative refinement
    max_lp_iters = 30
    lp_start_time = time.time()
    for it in range(max_lp_iters):
        if time.time() >= deadline - 30:
            break
        # compute current max conv
        conv_f = convolution_fft(f, f)
        M = conv_f.max()
        # solve LP for g
        g = cp.Variable(n, nonneg=True)
        conv_fg = cp.conv(f, g)  # length 2n-1
        constraints = [conv_fg <= M + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-9, reltol=1e-9)
        except Exception as e:
            print(f"LP error at iteration {it}: {e}")
            break
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            print(f"LP not optimal: {prob.status}")
            break
        g_val = np.maximum(g.value, 0.0)  # ensure nonneg
        sum_g = g_val.sum()
        if sum_g <= 1e-12:
            print(f"LP gave zero sum at iteration {it}")
            break
        g_norm = g_val / sum_g
        # mixing step
        t = 0.2  # or adapt
        f_new = (1 - t) * f + t * g_norm
        # symmetrize
        f_new = (f_new + f_new[::-1]) * 0.5
        # project onto simplex
        f_new = project_to_simplex(f_new)
        # evaluate score
        sc_new = compute_score(f_new)
        print(f"[{time.time():.1f}] LP iter {it+1}, score {sc_new:.6f}")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = f_new.copy()
        f = f_new
        # optional decrease t: maybe reduce if improvement small

    # after LP stage, optional small Adam refinement (fast)

"""
LP‑guided convex‑combination refinement for the max‑auto‑convolution problem.
The method builds an initial step‐function from simple analytic families,
then iteratively solves a linear programme
    max   1ᵀg
    s.t.  f * g ≤ M   (componentwise)
    g ≥ 0
where M = max(f*f).  The obtained g points toward a direction that reduces
the peak autocorrelation.  We move a fraction towards a normalised g,
project onto the simplex, enforce symmetry and repeat.
A short Adam smoothing pass finishes the search.
"""

import time
import numpy as np
import cvxpy as cp
from typing import List

def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w = w / w.sum()
    return w

def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w = w / w.sum()
    return w

def softmax(z: np.ndarray) -> np.ndarray:
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 100_000,
                         beta_start: float = 20.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.015,
                         sym_interval: int = 2000) -> tuple[np.ndarray, float]:
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 5000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

LP‑guided refinement for the max‑auto‑convolution problem.

The algorithm proceeds as follows:
1. Pick a moderate length (2048) and initialise the step‑function from a
   few cheap analytic families (symmetric product, left‑heavy power and
   the uniform baseline).  The best of these candidates is taken as f.

2. Repeatedly solve a linear programme
        max   1ᵀg
        s.t.  f * g ≤ M   (component‑wise)      (M = max(f*f))
              g ≥ 0

   without raising the peak of the convolution.  A convex combination
   ‖f←(1‑t)f + t·g_norm‖ is taken, the result is symmetrised and projected
   onto the simplex.  The loop stops after a few iterations without
   improvement or when the time budget is exhausted.

3. A short Adam optimisation on the smooth‑max approximation of the
   objective (log‑sum‑exp) is run to fine‑tune the final vector.
