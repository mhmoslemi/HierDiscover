def build_convolution_matrix(f):
    n = len(f)
    m = 2*n-1
    rows = []
    cols = []
    data = []
    for j in range(n):
        # rows indices affected: j..j+n-1
        rows_j = np.arange(j, j+n, dtype=int)
        cols_j = np.full(n, j, dtype=int)
        rows.extend(rows_j.tolist())
        cols.extend(cols_j.tolist())
        data.extend(f.tolist())
    A = sparse.csr_matrix((data, (rows, cols)), shape=(m, n))
    return A

"""
Propose a low‑dimensional step‑function with a very low ∥f∗f∥∞.

The search consists of three stages

1. **Seeding** – a handful of analytically‑motivated shapes (sin²,
   symmetric beta‑family, sparse spikes) and a few random Dirichlet draws.
   The best seed (exact objective) is kept.

2. **LP‑based refinement** – an iterative scheme taken from the
   “alternating LP” idea described in the statement.  For the current
   distribution f we build the convolution matrix A (so A·g = f∗g) and
   solve a linear programme that maximises  Σg_j  under the linear
   constraints  A·g ≤ M  (M = max(f∗f)) and  g ≥ 0.  The optimiser returns a
   vector g that spreads mass where f is small.  After normalising g we
   mix it with f using a decaying step size.  This process repeatedly
   reduces the bound M.

3. **Fine‑tuning** – a short soft‑max gradient descent (smooth surrogate
   of the max‑norm) followed by a stochastic hill‑climb, exactly as in
   the previous state‑of‑the‑art implementation.

All phases respect the 1000 s wall‑clock budget; the current best vector
is returned as a plain ``list[float]``.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for the sizes we need)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin² shape a_i = sin²(pi·i/(n+1)), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    p>0 clusters mass in the centre, p<0 pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign random
    positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a

# Phase‑2: LP based refinement
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Return the (2n‑1)×n matrix A such that A @ g = f ∗ g (full convolution).
    """
    n = f.shape[0]
    rows = np.arange(n)[:, None] + np.arange(n)      # shape (n,n)
    rows = rows.ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    m = 2 * n - 1
    return sp.csr_matrix((data, (rows, cols)), shape=(m, n))


def lp_step(f: np.ndarray, M: float, mix: float) -> np.ndarray | None:
    """
    One LP‑based improvement step.
    Solve   max Σ g_j   s.t.  f∗g ≤ M,  g ≥ 0.
    Return the mixed vector (1‑mix)·f + mix·ĝ, where ĝ is g normalised to sum = 1.
    If the LP fails, return None.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)

    # tiny numeric slack – it prevents infeasibility caused by round‑off
    b_ub = np.full(2 * n - 1, M + 1e-12)
    c = -np.ones(n)                     # maximise Σ g_j  →  minimise −Σ g_j
    bounds = [(0.0, None)] * n

    # solve with HiGHS (fast and robust)
    res = opt.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    g_norm = g / s                       # now Σ g_norm = 1

    # convex combination with current f
    new_f = (1.0 - mix) * f + mix * g_norm
    new_f = project_to_simplex(new_f)
    return new_f


def lp_refine(a: np.ndarray, cur_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑based mixing.
    The mixing coefficient starts at 0.5 and decays like 1/√iteration.
    """
    cur = a.copy()
    best = cur.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline:
        M = convolution_fft(cur, cur).max()
        mix = 0.5 / np.sqrt(it + 1.0)          # decreasing step size
        nxt = lp_step(cur, M, mix)
        if nxt is None:
            break
        cur = nxt
        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
            print(f"[LP] iter {it:4d} | best {best_score:.6f}")
        it += 1
        # a very cheap stop‑condition – if improvement stalls we quit
        if it > 30 and it % 10 == 0 and best_score - cur_score < 1e-7:
            break
    return best, best_score


# ----------------------------------------------------------------------
# Phase‑1: soft‑max gradient descent (copied from the reference code)
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step_coef: float = 0.0015,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth surrogate
        Lβ(a)=log Σ exp(β·conv(a,a)).
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    start = time.time()
    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()                     # soft‑max

        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 3000 == 0:
            print(f"[GD] it {it:6d} | time {elapsed:6.1f}s | best {best_score:.6f} | β {beta:.1f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: hill‑climb polishing (as before)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Stochastic moves that are kept only when they improve the exact score."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                     # pure swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # small transfer
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand /= cand.sum()                               # renormalise
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin for the surrounding harness
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------------
    # Choose a moderate length – 1024 gives a good trade‑off.
    # --------------------------------------------------------------
    n = 1024
    print(f"Sequence length = {n}")

    # --------------------------------------------------------------
    # 1. Seed generation and selection
    # --------------------------------------------------------------
    seeds = []

    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc))

    for k in (30, 50, 80):
        seeds.append(random_spike(n, k))

    # optional fallback (might be a better handcrafted solution)
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        if len(fallback) == n:
            seeds.append(np.array(fallback, dtype=float) / np.sum(fallback))
    except Exception:
        pass

    best_seq = None
    best_val = float("inf")
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)
        val = compute_score(s)
        if val < best_val:
            best_val = val
            best_seq = s.tolist()
            print(f"Initial seed improvement: score = {best_val:.6f}")

    cur_a = np.array(best_seq, dtype=float)
    cur_a = project_to_simplex(cur_a)
    cur_score = compute_score(cur_a)

    # --------------------------------------------------------------
    # 2. LP‑based refinement (takes a good chunk of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n[Phase LP] Starting LP‑based refinement")
        lp_deadline = deadline - 25.0
        cur_a, cur_score = lp_refine(cur_a, cur_score, lp_deadline)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
            print(f"After LP best: {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Soft‑max gradient descent (short but still useful)
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        print("\n[Phase GD] Soft‑max gradient descent")
        gd_deadline = min(deadline - 5.0, time.time() + (deadline - time.time()) * 0.45)
        cur_a, cur_score = softmax_gradient_descent(
            cur_a, cur_score, gd_deadline, beta_start=5.0, beta_end=2000.0, step_coef=0.0015
        )
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
            print(f"After GD best: {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final greedy hill‑climb polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase Hill] Polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=1_200_000)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
            print(f"After polishing best: {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Return the best vector found (the harness will call evaluate_sequence)
    # --------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
