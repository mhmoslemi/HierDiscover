"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

Algorithm
---------
0.  Load the fallback candidate if present.
1.  Choose a problem size (default 2048).  
2.  Build a modest library of analytically‑motivated shapes, including a
    *square‑root of the uniform convolution* (FFT based) and a fine
    grid of the symmetric product family (the “U‑shaped” arcsine‑like
    distributions).  The best of these low‑dimensional candidates is kept.
3.  Refine the best candidate with a smooth‑max Adam optimisation
    (β is increased up to 2000, the surrogate approaches the true max).
4.  Perform a final stochastic hill‑climb (simulated‑annealing style) on
    the full simplex.
5.  Return the best sequence found before the wall‑clock deadline.

All heavy work (FFT‑based convolutions) runs in O(n log n) and the
search respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution using a real FFT.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families used for the fast start
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family a_i ∝ ((i+½)(n‑½‑i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing (left‑heavy) sequence."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Exponential‑U shape: a_i ∝ exp(-γ·dist(i, centre))."""
    gamma = np.random.uniform(0.0, 6.0)
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def sqrt_uniform_candidate(n: int) -> np.ndarray:
    """
    Construct a vector a such that a * a ≈ uniform on length 2n‑1,
    by taking the (complex) square‑root of the FFT of the uniform
    convolution and projecting back to the non‑negative simplex.
    """
    L = 2 * n - 1
    u = np.full(L, 1.0 / L, dtype=np.float64)      # uniform convolution target
    N = next_pow2(L)                               # FFT length
    u_pad = np.zeros(N, dtype=np.float64)
    u_pad[:L] = u
    U = np.fft.fft(u_pad)                          # complex spectrum
    A = np.sqrt(U)                                 # principal complex square‑root
    a_full = np.fft.ifft(A).real                   # keep the real part
    a = a_full[:n]                                 # first n entries
    a = np.maximum(a, 0.0)                         # enforce non‑negativity
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (works on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 4.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.015) -> tuple[np.ndarray, float]:

    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    # initialise logits from the current probabilities
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (exponential increase)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smoothed‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derivation from the surrogate)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through the soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional progress report
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a small amount
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling schedule
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The function respects a wall‑clock budget (default 1000 s) and returns
    a plain list of floats ready for evaluation.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Slight safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")
    best_name = None

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if supplied)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]    # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_seq = fb_vec
            best_score = fb
            best_name = "fallback"
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Determine problem size
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048
    print(f"[{time.time():.1f}] problem size n = {n}")

    # ------------------------------------------------------------------
    # 2. Quick low‑dimensional scan
    # ------------------------------------------------------------------
    candidates = []

    # uniform baseline
    candidates.append(("uniform", np.full(n, 1.0 / n, dtype=np.float64)))

    # fine grid of the symmetric product family (U‑shaped / arcsine‑like)
    for alpha in np.arange(0.2, 2.01, 0.1):
        candidates.append((f"sym_alpha_{alpha:.2f}", symmetric_product_family(n, alpha)))

    # a few random monotone starts
    for _ in range(4):
        candidates.append(("rand_monotone", random_monotone_start(n)))

    # a few random exponential‑U starts
    for _ in range(4):
        candidates.append(("rand_u", random_u_start(n)))

    # the square‑root‑uniform construction
    candidates.append(("sqrt_uniform", sqrt_uniform_candidate(n)))

    # evaluate all candidates (stop early if time runs out)
    for name, vec in candidates:
        if time.time() >= deadline:
            break
        a = np.maximum(vec, 0.0)
        if a.sum() == 0.0:
            continue
        a = a / a.sum()
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            best_name = name
            print(f"[{time.time():.1f}] new best {name} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine the current best with smooth‑max Adam
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)

    # ensure exact simplex normalisation
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    if time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting Adam refinement (seed = {best_name})")
        a_opt, sc_opt = adam_smooth_optimize(
            a,
            deadline - 60.0,
            max_iters=400_000,
            beta_start=4.0,
            beta_end=2000.0,
            lr=0.015,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=1_200_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return result
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
