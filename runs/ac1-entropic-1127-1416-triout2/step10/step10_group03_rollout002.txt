"""
Propose a step‑function (non‑negative vector) that minimizes

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm uses three stages and a larger search space (up to n=4096)
to push the bound below the previous 1.511131 value.

1. **Seeding** – a diverse collection of deterministic shapes
   (sin², Beta‑family, exponential, mirrored versions) and random
   Dirichlet / sparse spike draws.  The best seed (exact objective)
   is kept.

2. **Smooth‑max gradient descent** – minimise a differentiable
   surrogate
        Lβ(a)=log Σ exp(β·conv(a,a))
   with an ADAM‑style projected optimiser on the probability simplex.
   The temperature β is annealed from 5 to 4000.  The best iterate
   (according to the true objective) is retained.

3. **Sub‑gradient refinement** – classic projected sub‑gradient descent
   on the true max‑objective, followed by a simulated‑annealing hill
   climber that makes small mass‑transfer moves and accepts occasional
   uphill steps.

All work respects the 1000 s wall‑clock budget and prints progress.
The returned value is a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast and accurate)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector with sum = 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin² shape a_i = sin²(pi·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    p>0 clusters mass in centre, p<0 pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum 1."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector with k non‑zero entries, normalised."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max ADAM optimiser
# ----------------------------------------------------------------------


def adam_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 4000.0,
    lr0: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate Lβ using ADAM on the simplex.
    Returns the best iterate w.r.t. the true objective.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    # ADAM moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    start = time.time()

    while time.time() < deadline:
        # exact objective for bookkeeping
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # temperature schedule
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        beta = beta_start + frac * (beta_end - beta_start)

        # soft‑max weights
        shifted = conv - max_conv          # stability
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient of surrogate
        grad = 2.0 * beta * np.correlate(w, cur, mode="valid")

        # ADAM step
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        cur = cur - step
        cur = project_to_simplex(cur)

    return best, best_score

def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur, cur)
        k = int(np.argmax(conv))

        # sub‑gradient for max entry (only the active offset)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]

        step = step0 / np.sqrt(t)
        cur -= step * g
        cur = project_to_simplex(cur)

        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur += jitter
            cur = project_to_simplex(cur)

    return best, best_score

# Phase 4 – simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sim_anneal(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
    init_temp: float = 0.02,
    final_temp: float = 0.0002,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that occasionally accepts worsening moves.
    Moves comprise swapping two entries or transferring a fraction of mass.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = compute_score(best)

    start = time.time()
    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # temperature linearly decays
        T = init_temp + (final_temp - init_temp) * (step / max_steps)

        cand = cur.copy()
        # two types of elementary moves
        if np.random.rand() < 0.5:               # swap two masses
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = (np.random.rand() * 0.1 + 0.05) * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand /= cand.sum()          # re‑normalise numerical drift
        cand_score = compute_score(cand)

        # accept if better, otherwise with Boltzmann probability
        if cand_score < cur_score if (cur_score := compute_score(cur)) else True:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        else:
            if np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                cur = cand
                cur_score = cand_score

        # occasional progress print
        if step % 200_000 == 0:
            print(f"[HillClimb] step {step:,} | best {best_score:.6f}")

    return best, best_score

        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=4000.0,
        lr0=1e-3,
    )
    if cur_score < best_val:
        best_val = cur_score
    print(f"After ADAM: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Sub‑gradient refinement (classic)
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print("\n[Phase 2] Sub‑gradient refinement")
        phase2_deadline = deadline - 45.0
        cur_a, cur_score = subgradient_refine(
            cur_a, cur_score, phase2_deadline, max_iter=600_000, step0=0.07
        )
        if cur_score < best_val:
            best_val = cur_score
        print(f"After sub‑grad: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final hill‑climb with simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 3] Simulated‑annealing hill‑climb")
        cur_a, cur_score = hill_climb_sim_anneal(cur_a, deadline, max_steps=1_500_000)
        if cur_score < best_val:
            best_val = cur_score
        print(f"After hill‑climb: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Return the result (evaluate_sequence will apply its own clipping)
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nSearch finished. Final reported score = {final_val:.6f}")

    return final_seq
