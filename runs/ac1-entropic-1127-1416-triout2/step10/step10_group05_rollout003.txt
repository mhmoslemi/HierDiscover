"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence.  The algorithm works in several stages:

1. Choose a sequence length `n` (larger when time permits).
2. Scan a few low‑dimensional parametric families (exponential,
   power‑law, Beta) and a handful of random Dirichlet vectors.
   Keep the best few candidates (on the exact objective).
3. Refine each candidate with a series of L‑BFGS runs on a smooth
   log‑sum‑exp approximation of the maximal convolution.  The
   optimisation variable is the log‑soft‑max (v‑vector) so that the
   non‑negativity and simplex constraints are satisfied automatically.
   A schedule of increasing smoothing parameters β makes the
   approximation approach the true max.
4. Perform a final targeted hill‑climb that repeatedly moves a small
   packet of mass from the index that contributes most to the current
   maximal convolution entry to the index with the smallest value.
5. Return the best sequence found.
"""

import time
import numpy as np
import scipy.optimize as opt


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using a real FFT (zero‑padded).
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence for a vector already on
    the probability simplex (sum = 1). Returns 2·n·max(a*a) where the
    convolution is full (length 2·n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Parametric families (used in the cheap grid scans)
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam <= 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p <= 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta distribution on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()

# Grid searches (very cheap compared with the later optimisation)
# ----------------------------------------------------------------------


def scan_exponential_and_power(n: int, deadline: float):
    """Exhaustive scan of exponential and power‑law families."""
    best_a, best_sc = None, float("inf")

    # exponential decay – include lam=0 (uniform) and a dense log‑space
    lam_grid = np.concatenate(
        (np.array([0.0]), np.logspace(-5, -1, 250))
    )
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    # power‑law – sweep p from 0 (uniform) up to 8
    p_grid = np.linspace(0.0, 8.0, 300)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    return best_a, best_sc


def scan_beta_family(n: int, deadline: float):
    """Moderate scan of a 2‑parameter Beta family."""
    best_a, best_sc = None, float("inf")
    # concentrate around values that give edge‑ vs centre‑emphasis
    a_grid = np.concatenate((
        np.logspace(-0.7, -0.1, 12),
        np.logspace(-0.1, 0.3, 12),
        np.linspace(0.5, 3.0, 12)
    ))
    b_grid = a_grid.copy()
    for a_val in a_grid:
        for b_val in b_grid:
            if time.time() >= deadline:
                return best_a, best_sc
            a_vec = beta_shape(n, a_val, b_val)
            sc = compute_score(a_vec)
            if sc < best_sc:
                best_sc, best_a = sc, a_vec.copy()
    return best_a, best_sc

class SmoothMaxObjective:
    """
    Callable object for scipy.optimize.minimize.
    Holds the length n and smoothing parameter β.
    Returns the smooth objective (2·n·softmax(conv)) and its gradient
    with respect to the log‑softmax parameters v (so that a = softmax(v)).
    """

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, v: np.ndarray):
        # a = softmax(v)
        maxv = np.max(v)
        expv = np.exp(v - maxv)
        a = expv / expv.sum()

        # convolution a*a
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # log‑sum‑exp (smooth max)
        expw = np.exp(self.beta * (conv - maxc))
        sumw = expw.sum()
        smooth = maxc + np.log(sumw) / self.beta
        obj = 2.0 * self.n * smooth

        # gradient w.r.t a of the smooth max
        w = expw / sumw
        # gradient of smooth max wrt a is 2 * (a_rev * w) taken at the proper slice
        rev = a[::-1]
        g_full = convolution_fft(rev, w)               # length 3·n‑2
        grad_a = 2.0 * g_full[self.n - 1:self.n - 1 + self.n]

        # transform gradient to the v‑space (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_v = a * (grad_a - dot)

        return float(obj), grad_v.astype(float)


# ----------------------------------------------------------------------
# L‑BFGS based refinement
# ----------------------------------------------------------------------


def l_bfgs_optimize(start_a: np.ndarray,
                    deadline: float,
                    beta_schedule: list[float],
                    maxiter_per_phase: int = 200) -> tuple[np.ndarray, float]:
    """
    Refine a candidate vector by a sequence of L‑BFGS runs.
    The smoothing parameter β is increased according to beta_schedule.
    Returns the refined vector and its exact score.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break

        obj = SmoothMaxObjective(n, beta)

        # initialise logits from the current a (avoid log(0))
        v0 = np.log(a + 1e-14)

        # approximate remaining iterations we can afford
        time_left = deadline - time.time()
        maxiter = maxiter_per_phase
        if time_left < 20.0:
            maxiter = max(5, int(time_left * 5))

        try:
            res = opt.minimize(obj,
                               v0,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': maxiter,
                                        'disp': False,
                                        'gtol': 1e-12})
        except Exception as e:
            print(f"[{time.time():.1f}] L‑BFGS raised: {e}")
            break

        if not res.success:
            # not a failure, just continue – we may still have a usable point
            print(f"[{time.time():.1f}] L‑BFGS warning: {res.message}")

        v_opt = res.x
        maxv = np.max(v_opt)
        expv = np.exp(v_opt - maxv)
        a = expv / expv.sum()

        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.1f} improved best score to {best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple targeted hill‑climb (more systematic than the previous random version)
# ----------------------------------------------------------------------


def targeted_hill_climb(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Greedy walk: repeatedly remove a fraction of mass from the index that
    contributes most to the current maximal convolution entry and add it to
    the currently smallest entry.  Moves are accepted only when they reduce
    the exact objective.
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_frac = 0.08  # initial fraction of the chosen source mass to move

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        # indices that appear in the active convolution term
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k_max - idx]

        i_src = idx[np.argmax(contrib)]           # biggest contributor
        # choose destination: the smallest entry that is not the source
        j_dst = np.argmin(best_a + (np.arange(n) == i_src) * 1e9)

        if best_a[i_src] <= 0.0:
            break

        delta = step_frac * best_a[i_src]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_src] -= delta
        cand[j_dst] += delta
        # re‑normalize to avoid tiny numerical drift
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_frac = min(step_frac * 1.06, 0.5)          # become more aggressive
        else:
            step_frac *= 0.94                                 # back‑off

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """Search for a step‑function that minimizes evaluate_sequence."""
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 3.0

    remaining = deadline - start_time
    # pick a length that roughly matches the remaining budget
    if remaining > 800:
        n = 65536
    elif remaining > 500:
        n = 32768
    elif remaining > 250:
        n = 16384
    else:
        n = 8192
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  optional fallback from a previously discovered vector
    # ------------------------------------------------------------------
    fallback = globals().get("height_sequence_1", None)
    if fallback is not None:
        try:
            fallback = [float(x) for x in fallback]
            fb = evaluate_sequence(fallback)
            print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
            if fb < best_score:
                best_score, best_seq = fb, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  cheap scans of parametric families
    # ------------------------------------------------------------------
    # Give each scan a modest budget; we leave ~30 % of the time for refinement
    scan_deadline = deadline - 200.0
    if time.time() < scan_deadline:
        a_exp, sc_exp = scan_exponential_and_power(n, scan_deadline)
        if a_exp is not None and sc_exp < best_score:
            best_score, best_seq = sc_exp, a_exp.tolist()
            print(f"[{time.time():.1f}] after exp/power scan = {best_score:.6f}")

    if time.time() < scan_deadline:
        a_beta, sc_beta = scan_beta_family(n, scan_deadline)
        if a_beta is not None and sc_beta < best_score:
            best_score, best_seq = sc_beta, a_beta.tolist()
            print(f"[{time.time():.1f}] after beta scan = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  random Dirichlet starts (cheap diversification)
    # ------------------------------------------------------------------
    random_candidates = []
    while time.time() < scan_deadline and len(random_candidates) < 8:
        # draw a Dirichlet vector with a random concentration parameter
        alpha = 10 ** np.random.uniform(-2, 1)   # values roughly in [0.01, 10]
        gammas = np.random.gamma(alpha, 1.0, size=n)
        a_rand = gammas / gammas.sum()
        sc = compute_score(a_rand)
        random_candidates.append((sc, a_rand.copy()))
        if sc < best_score:
            best_score, best_seq = sc, a_rand.tolist()
            print(f"[{time.time():.1f}] random Dirichlet improved to {best_score:.6f}")

    # keep the top‑k seeds for the heavy refinement stage
    all_candidates = []
    if best_seq is not None:
        all_candidates.append((best_score, np.array(best_seq, dtype=float)))
    for sc, vec in random_candidates:
        all_candidates.append((sc, vec))
    # sort by score (ascending) and keep the best few
    all_candidates.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in all_candidates[:5]]

    # ------------------------------------------------------------------
    # 3.  L‑BFGS refinement on the smooth max objective
    # ------------------------------------------------------------------
    beta_schedule = [20.0, 80.0, 300.0, 1200.0, 5000.0]  # increasing smoothness
    for idx, cand in enumerate(top_candidates):
        if time.time() >= deadline - 120.0:
            break
        print(f"[{time.time():.1f}] L‑BFGS refinement start for candidate {idx+1}")
        refined, sc_ref = l_bfgs_optimize(cand,
                                          deadline - 100.0,
                                          beta_schedule,
                                          maxiter_per_phase=250)
        if sc_ref < best_score:
            best_score, best_seq = sc_ref, refined.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS candidate {idx+1} best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Targeted hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] final hill‑climb polishing")
        a_final, sc_final = targeted_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=300_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Safety net – tiny uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 6.  Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
