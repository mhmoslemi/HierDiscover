def solve_minmax_lp(f, deadline):
    n = len(f)
    # Build A_g matrix
    total = n*n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    data = np.empty(total, dtype=np.float64)
    offset = 0
    for i in range(n):
        rows[offset:offset+n] = np.arange(i, i+n)
        cols[offset:offset+n] = i
        data[offset:offset+n] = f
        offset += n
    A_g = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    # column for -t
    neg_t = sp.csr_matrix((-np.ones((2*n-1, 1)), (np.arange(2*n-1), np.zeros(2*n-1))), shape=(2*n-1,1))
    A_ub = sp.hstack([A_g, neg_t], format='csr')
    b_ub = np.zeros(2*n-1)
    # equality: sum(g)=1
    A_eq = sp.csr_matrix(np.concatenate([np.ones((1,n)), np.zeros((1,1))], axis=1))
    b_eq = np.array([1.0])
    c = np.concatenate([np.zeros(n), np.array([1.0])])
    bounds = [(0, None)]*n + [(0, None)]

    # call linprog
    try:
        res = _opt.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='highs', options={'time_limit': max(0, deadline - time.time())})
    except Exception:
        return None, None

    if not res.success:
        return None, None
    x = res.x
    g = x[:n]
    t = x[-1]
    return g, t

"""

Algorithm (≈1000 s budget)
---------------------------
1. Choose a modest length `n` (2048 or 4096) – large enough to get a good
   bound but small enough that the heavy linear‑programming (LP) step
   stays fast.
2. Scan a few analytic families (U‑shaped, power‑law, exponential,
   beta) on a fine grid and keep the best dense vector (sum = 1).
3. Refine the best vector with L‑BFGS‑B on a log‑sum‑exp surrogate of the
   max‑convolution (β schedule 200 → 800 → 2000 → 5000 → 10000).
4. **LP refinement** – for the current vector `f` solve

        minimise   t
        subject to  f * g ≤ t   (all shifts)
                    ∑ g = 1
                    g ≥ 0

   (the constraints are linear, we use `scipy.optimize.linprog` with the
   HIGHs solver).  The optimiser returns a new vector `g` with a lower
   bound on the max‑convolution.  If the self‑convolution of `g` is better
   than the best known value we replace `f` by `g`.  A few such LP
   iterations (normally 2‑4) give a substantial drop.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (aperiodic) convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        score = 2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑shape: a_i ∝ (i/(n‑1))^α·(1‑i/(n‑1))^α."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    a = w / w.sum()
    return a

def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where
        f   = (1/β)·log∑exp[β·(a*a)_k]   (log‑sum‑exp surrogate)
        grad = ∂f/∂z   (z are the log‑soft‑max parameters)
    """
    a = softmax(z)                     # a ∈ simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # gradient w.r.t. a
    w = exp_terms / sum_exp            # shape (2n‑1,)
    # convolution of a_rev with w gives the needed Jacobian
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:2 * n - 1]

    # chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray, deadline: float,
                  beta_list=(200.0, 800.0, 2000.0, 5000.0, 10000.0),
                  max_iter_per_beta=800) -> tuple[np.ndarray, float]:
    """
    Refine `start_a` using L‑BFGS‑B on the smooth surrogate.
    Returns the best vector (as numpy array) and its exact score.
    """
    if not _HAS_SCIPY:          # pragma: no cover
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)                     # initialise parameters

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 8))  # generous budget

        def func(z_vec):
            f, g = _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])
            return f, g

        try:
            opt_res = _opt.fmin_l_bfgs_b(
                func,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved score = {best_score:.6f}")

    return best_a, best_score

def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 0.8,
                           check_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Exponentiated (entropy‑mirror) sub‑gradient descent on
        F(a) = 2·n·max(a*a)
    The iterate stays on the simplex via a normalisation step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)            # multiplicative update

        # re‑project onto the simplex (entropy projection)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 1_200_000,
                  init_temp_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a very low temperature."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:        # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                             # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (minimise max (f * g))
# ----------------------------------------------------------------------


def solve_lp_minmax(f: np.ndarray, deadline: float):
    """
    Solve the LP:
        minimise   t
        subject to  f * g ≤ t          (all shifts)
                    ∑ g = 1
                    g ≥ 0
    Returns (g, t) where g is a probability vector and t the optimal bound.
    If the solver fails or runs out of time, (None, None) is returned.
    """
    if not _HAS_SCIPY:
        return None, None

    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    # Fill the matrix column‑wise: column i receives the whole vector f,
    # shifted down by i rows.
    offset = 0
    base = np.arange(n, dtype=np.int32)
    for i in range(n):
        rows[offset:offset + n] = base + i          # i,…,i+n‑1
        cols[offset:offset + n] = i
        data[offset:offset + n] = f
        offset += n

    A_g = _sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    # add the -t column (one per inequality)
    neg_t = _sp.csr_matrix((-np.ones((2 * n - 1, 1))), shape=(2 * n - 1, 1))
    A_ub = _sp.hstack([A_g, neg_t], format='csr')
    b_ub = np.zeros(2 * n - 1)

    # Equality: ∑ g = 1
    A_eq = np.concatenate([np.ones((1, n)), np.zeros((1, 1))], axis=1)
    b_eq = np.array([1.0])

    # Objective: minimise t
    c = np.concatenate([np.zeros(n), np.array([1.0])])
    bounds = [(0, None)] * n + [(0, None)]

    # Give the LP a small time budget (the outer deadline handles the rest)
    remaining = max(0.0, deadline - time.time())
    opts = {"time_limit": remaining}

    try:
        res = _opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                           A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='highs',
                           options=opts)
    except Exception:            # pragma: no cover
        return None, None

    if not res.success:
        return None, None

    x = res.x
    g = x[:n]
    t = x[-1]
    # Numerical safety – enforce the simplex exactly
    g = project_to_simplex(g)
    return g, float(t)

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    # --------------------------------------------------------------
    # Length selection (larger n → potentially better bound)
    # --------------------------------------------------------------
    n = 4096 if budget_s >= 850 else 2048
    print(f"[{time.time():.1f}] Using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  Optional known construction
    # --------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Scan analytic families (coarse but cheap)
    # --------------------------------------------------------------
    # finer grids than the previous run
    grids = {
        "sym_product": np.linspace(0.0, 8.0, 141),
        "left_power": np.linspace(0.0, 8.0, 141),
        "exp_decay": np.linspace(0.0, 7.0, 111),
        "beta": np.linspace(0.0, 8.0, 111),
    }

    for alpha in grids["sym_product"]:
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

    for beta in grids["left_power"]:
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

    for gamma in grids["exp_decay"]:
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

    for alpha in grids["beta"]:
        if time.time() >= deadline - 30.0:
            break
        vec = beta_distribution_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (beta‑α={alpha:.2f}) = {best_score:.6f}")

    # Random monotone draws
    for _ in range(200):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. L‑BFGS‑B refinement on the log‑sum‑exp surrogate
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] Starting L‑BFGS refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_refined, sc_refined = l_bfgs_refine(
            a_start,
            deadline - 150.0,
            beta_list=(200.0, 800.0, 2000.0, 5000.0, 10000.0),
            max_iter_per_beta=1200,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Linear‑programming refinement (few iterations)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] Starting LP refinement")
        lp_iters = 0
        while time.time() < deadline - 40.0 and lp_iters < 4:
            f_vec = np.array(best_seq, dtype=np.float64)
            g_vec, t_val = solve_lp_minmax(f_vec, deadline - 5.0)
            if g_vec is None:
                break
            lp_iters += 1
            sc_g = compute_score(g_vec)
            print(f"[{time.time():.1f}] LP iter {lp_iters}: t={t_val:.6f}  self‑score={sc_g:.6f}")

            if sc_g + 1e-9 < best_score:
                best_score = sc_g
                best_seq = g_vec.tolist()
                print(f"[{time.time():.1f}] LP produced improvement → {best_score:.6f}")
                continue

            # try a simple convex combination if the pure LP result did not improve
            mix = 0.5
            mixed = (1 - mix) * f_vec + mix * g_vec
            mixed = project_to_simplex(mixed)
            sc_mix = compute_score(mixed)
            print(f"[{time.time():.1f}]   mixed score = {sc_mix:.6f}")
            if sc_mix + 1e-9 < best_score:
                best_score = sc_mix
                best_seq = mixed.tolist()
                print(f"[{time.time():.1f}] mixed vector improved best → {best_score:.6f}")

        print(f"[{time.time():.1f}] LP refinement finished after {lp_iters} iterations")

    # --------------------------------------------------------------
    # 4. Mirror sub‑gradient descent on the true max‑objective
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] Starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 15.0,
            step0=0.9,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Symmetrisation (most optimal solutions are symmetric)
    # --------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final tiny simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Fallback in the unlikely case we have nothing
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final official evaluation (the harness will also call this)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
