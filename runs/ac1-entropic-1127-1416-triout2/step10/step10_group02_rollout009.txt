"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
We treat the problem as minimising  f(a)=2·n·max(a∗a)  with the constraint
∑ a_i = 1, a_i ≥ 0.  A smooth surrogate of the max‑operator (soft‑max)
gives a differentiable objective whose gradient can be written as a
correlation of the current vector with a weight vector derived from the
soft‑max.  Using this gradient we perform projected gradient descent with a
decaying step‑size, gradually increasing the soft‑max temperature so that
the optimisation first explores a smooth landscape and finally sharpens
around the true max‑value.

The search consists of three phases:

1.  A quick global scan of power‑law shapes (the previous best construction)
    and a few random Dirichlet starts.
2.  Several short runs of the smooth‑gradient descent described above
    (different random starts, temperature schedule, tiny step‑size).
3.  A finishing sub‑gradient polish (active‑set) followed by a greedy
    hill‑climb as in the earlier implementation.

All updates are projected onto the probability simplex to keep the sum
normalised and the variables non‑negative.  Time is carefully budgeted so
the function always returns a candidate before the 1000‑second limit.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Numerical utilities
# ------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # zero vector -> uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# 1. Power‑law sweep (global cheap search)
# ------------------------------------------------------------

def best_power_law_vector(n: int, deadline: float) -> np.ndarray:
    """Search p∈[0.5,0.8] for the best (i+1)^‑p shape."""
    best_a = None
    best_val = float("inf")
    best_p = 0.5

    # coarse grid
    for p in np.linspace(0.5, 0.8, 61):
        if time.time() >= deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a, best_p = val, a, p

    # fine grid around the best p
    if best_a is not None:
        low = max(0.5, best_p - 0.04)
        high = min(0.9, best_p + 0.04)
        for p in np.linspace(low, high, 41):
            if time.time() >= deadline:
                break
            a = np.arange(1, n + 1, dtype=float) ** (-p)
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a, best_p = val, a, p

    return best_a


# ------------------------------------------------------------
# 2. Smooth‑max gradient descent
# ------------------------------------------------------------

def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution entries (numerically stable)."""
    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def smooth_descent(a0: np.ndarray, deadline: float,
                   beta_start: float = 2.0,
                   beta_end: float = 30.0,
                   beta_steps: int = 6,
                   step0: float = 0.001,
                   max_iter_per_beta: int = 3000) -> (np.ndarray, float):
    """
    Projected gradient descent on the smooth surrogate of the objective.
    The surrogate is 2·n·softmax_beta(conv(a,a)).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    betas = np.linspace(beta_start, beta_end, beta_steps)
    t = 0  # iteration counter for step‑size decay

    for beta in betas:
        it = 0
        while time.time() < deadline and it < max_iter_per_beta:
            it += 1
            t += 1

            conv = full_convolution(a)
            w = softmax_weights(conv, beta)

            # correlation r = (a_rev * w) and extraction of the n relevant entries
            r_full = np.convolve(a[::-1], w)                 # length = n + (2n-1) -1 = 3n-2
            start = (len(conv) - 1) // 2                     # = n-1
            r = r_full[start:start + n]                     # size n

            # gradient of the surrogate: 4·n·r
            grad = 4.0 * n * r

            # diminishing step‑size
            step = step0 / np.sqrt(t)

            a = a - step * grad
            a = project_to_simplex(a)

            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()

        if time.time() >= deadline:
            break

    return best_a, best_val


# ------------------------------------------------------------
# 3. Sub‑gradient polish (active‑set version)
# ------------------------------------------------------------

def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12,
                       tol: float = 1e-8) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses *all* indices within `tol` of the current maximum to form a
    convex combination of sub‑gradients (a small “active set”).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        mx = conv.max()
        # active set: all indices whose value is within `tol` of the maximum
        active = np.where(conv >= mx - tol)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)])

        # build weight vector on the convolution domain
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # correlation to obtain the averaged sub‑gradient
        r_full = np.convolve(a[::-1], w)
        start = (len(conv) - 1) // 2
        r = r_full[start:start + n]                # size n
        grad = 4.0 * n * r                          # same scaling as smooth version

        step = step0 / np.sqrt(t)
        a = a - step * grad
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # occasional jitter to avoid stagnation
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ------------------------------------------------------------
# 4. Simple greedy hill‑climb (final polishing)
# ------------------------------------------------------------

def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 500_000) -> np.ndarray:
    """
    Tiny stochastic mass‑transfer moves with greedy acceptance.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_no_improve:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
    return best_a


# ------------------------------------------------------------
# 5. Helper – a random point on the simplex
# ------------------------------------------------------------

def random_simplex_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet(1))."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / x.sum()


def perturbed_vector(base: np.ndarray, scale: float = 0.1) -> np.ndarray:
    """Add Gaussian noise (relative to the mean) and re‑project."""
    noise = np.random.randn(base.size) * scale * base.mean()
    cand = base + noise
    cand = np.maximum(cand, 0.0)
    return project_to_simplex(cand)

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    n = 2048                     # length of the step function

    best_seq = None
    best_val = float("inf")

    def consider(candidate: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_val
        val = compute_score(candidate)
        if val < best_val:
            best_val, best_seq = val, candidate.copy()
            print(f"[{tag}] new best = {val:.6f}")

    # -----------------------------------------------------------------
    # 0. Fallback (if a previous construction is available)
    # -----------------------------------------------------------------
    try:
        fallback = globals().get("height_sequence_1")
        if fallback is not None:
            fallback = np.array([float(x) for x in fallback])
            fallback = project_to_simplex(fallback)          # ensure normalised
            consider(fallback, "fallback")
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Uniform baseline (very cheap)
    # -----------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    consider(uniform, "uniform")

    # -----------------------------------------------------------------
    # 2. Power‑law sweep – cheap global search
    # -----------------------------------------------------------------
    pll_deadline = deadline - 300.0          # reserve time for deeper phases
    if time.time() < pll_deadline:
        a_pow = best_power_law_vector(n, pll_deadline)
        if a_pow is not None:
            consider(a_pow, "power‑law")

    # -----------------------------------------------------------------
    # 3. Random starts + smooth‑max descent
    # -----------------------------------------------------------------
    # allocate roughly half of the remaining budget to this phase
    while time.time() < deadline - 250:
        # each run gets at most ~25 s (or whatever is left)
        sub_deadline = min(time.time() + 25.0, deadline - 250)
        if best_seq is not None:
            # sometimes start from a perturbation of the current best,
            # otherwise from a fresh Dirichlet point
            if np.random.rand() < 0.6:
                a0 = perturbed_vector(best_seq, scale=0.2)
            else:
                a0 = random_simplex_vector(n)
        else:
            a0 = random_simplex_vector(n)

        a_opt, val_opt = smooth_descent(
            a0,
            sub_deadline,
            beta_start=2.0,
            beta_end=30.0,
            beta_steps=6,
            step0=0.001,
            max_iter_per_beta=3000,
        )
        consider(a_opt, "smooth‑descent")
        # early exit if we run out of time
        if time.time() >= deadline - 250:
            break

    # -----------------------------------------------------------------
    # 4. Sub‑gradient polishing (active‑set)
    # -----------------------------------------------------------------
    sub_deadline = deadline - 120.0
    if best_seq is not None and time.time() < sub_deadline:
        a0 = best_seq.copy()
        a_ref = subgradient_polish(a0, sub_deadline, step0=0.12, tol=1e-8)
        consider(a_ref, "sub‑grad")

    # -----------------------------------------------------------------
    # 5. Final greedy hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = best_seq.copy()
        a_final = hill_climb(a_start, hill_deadline)
        consider(a_final, "hill‑climb")

    # -----------------------------------------------------------------
    # Safeguard – ensure we always return something
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = compute_score(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official optimise‑function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq.tolist()
