def next_pow2(x):
    return 1 << (x-1).bit_length()
def convolution_fft(a,b):
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A*B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv
def compute_score(a):
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v):
    # Euclidean projection onto simplex {x >=0, sum=1}
    v = np.asarray(v, dtype=np.float64)
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
```

The algorithm follows a different route from the reference code:

1. Initialise a probability vector `a` (non‑negative, sum = 1) from a
   family of analytic shapes (the beta‑type “U‑shape” works well).

    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (exact for our lengths)."""
    m = a.size + b.size - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective assuming Σa = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for diverse initialisations)
# ----------------------------------------------------------------------


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density ∝ (i+½)^{p‑1}·(n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    return w / w.sum()


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w = np.where(np.isfinite(w), w, 0.0)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse uniform‑mass vector (≈5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         beta_start: float = 5.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.015,
                         sym_interval: int = 2000) -> (np.ndarray, float):
    """Minimise the smooth‑max approximation using Adam on the soft‑max parametrisation."""
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(np.maximum(start_a, eps))

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(np.maximum(a, eps))

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a (see note in original source)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 2000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------


def _build_conv_matrix(vec: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix C such that C @ g = vec * g (full convolution).
    For each column j the vector ``vec`` is placed starting at row j.
    """
    n = vec.shape[0]
    m = 2 * n - 1
    C = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = vec
    return C


def _lp_step(f: np.ndarray, eps: float = 1e-8) -> (np.ndarray, float):
    """
    One LP improvement: maximise total mass under the current max‑convolution bound.
    Returns the new probability vector (symmetrised, projected) and its score.
    """
    n = f.shape[0]
    conv_f = convolution_fft(f, f)
    M = conv_f.max()

    # keep a strictly positive version of f for the constraints (avoids unbounded columns)
    f_pos = np.maximum(f, eps)
    f_pos = f_pos / f_pos.sum()

    A = _build_conv_matrix(f_pos)                     # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)                # maximise Σg  ↔ minimise -Σg

    # Solve the LP (highs is the most robust interior‑point method in SciPy)
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'disp': False})
    if not res.success:
        raise RuntimeError(f"LP failed: status {res.status}")

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1.0 + 1e-12:        # no useful increase – stop iterating
        raise StopIteration

    # Normalise to a probability vector, enforce symmetry and project back onto the simplex.
    g = g / sum_g
    g = (g + g[::-1]) * 0.5
    g = project_to_simplex(g)

    return g, compute_score(g)


# ----------------------------------------------------------------------
# Main driver required by the challenge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative vector) minimising
    ``evaluate_sequence``.  The routine mixes a linear‑programming
    refinement loop with a short Adam polish and respects the 1000 s wall‑clock.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ---------------------------------------------------------
    # 1. Choose a length (2048 works well for the LP size)
    # ---------------------------------------------------------
    n = 2048

    # ---------------------------------------------------------
    # 2. Diverse initialisations – keep the best one found
    # ---------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # Helper to safely evaluate a candidate
    def try_candidate(v: np.ndarray):
        nonlocal best_vec, best_score
        sc = compute_score(v)
        if sc < best_score:
            best_score = sc
            best_vec = v.copy()
            print(f"[{time.time():.1f}] new global best = {best_score:.6f}")

    # Initialise with a few analytic families and a random sparse start
    init_candidates = [
        beta_family(n, 0.5, 0.5),                     # classic U‑shape (arcsine‑like)
        symmetric_product_family(n, 2.0),
        left_heavy_power_family(n, 1.5),
        random_sparse_start(n),
    ]
    for init in init_candidates:
        init = np.maximum(init, 1e-12)
        init = project_to_simplex(init)
        try_candidate(init)

    # ---------------------------------------------------------
    # 3. LP refinement loop (multiple random restarts)
    # ---------------------------------------------------------
    if linprog is None:
        print("SciPy linprog not available – falling back to the previous heuristic.")
    else:
        while time.time() < deadline - 120.0:   # keep ~2 min for final polishing
            # pick a random start among analytic families
            choice = np.random.choice(['beta', 'prod', 'power', 'sparse'])
            if choice == 'beta':
                p = np.random.uniform(0.1, 3.0)
                q = np.random.uniform(0.1, 3.0)
                f = beta_family(n, p, q)
            elif choice == 'prod':
                a = np.random.uniform(0.0, 12.0)
                f = symmetric_product_family(n, a)
            elif choice == 'power':
                b = np.random.uniform(0.0, 12.0)
                f = left_heavy_power_family(n, b)
            else:
                f = random_sparse_start(n)

            f = np.maximum(f, 1e-12)
            f = project_to_simplex(f)
            try_candidate(f)

            # perform a short upward‑mass LP chain (max 12 iterations)
            for _ in range(12):
                if time.time() >= deadline - 120.0:
                    break
                try:
                    f, sc = _lp_step(f)
                except StopIteration:
                    break
                except Exception as e:            # e.g. unbounded – abort this chain
                    print(f"[{time.time():.1f}] LP error: {e}")
                    break
                try_candidate(f)

                # stop early if we already passed the target bound
                if best_score < 1.503:
                    break

            if best_score < 1.503:
                break

    # ---------------------------------------------------------
    # 4. Final Adam polishing (if time permits)
    # ---------------------------------------------------------
    if best_vec is not None and time.time() < deadline - 5.0:
        try:
            a_opt, sc_opt = adam_smooth_optimize(
                best_vec,
                deadline,
                max_iters=180_000,
                beta_start=5.0,
                beta_end=2500.0,
                lr=0.014,
                sym_interval=3000,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_vec = a_opt.copy()
                print(f"[{time.time():.1f}] after Adam polish best = {best_score:.6f}")
        except Exception as e:                                            # pragma: no cover
            print("Adam polishing failed:", e)

    # ---------------------------------------------------------
    # 5. Safeguard – ensure we always return a valid sequence
    # ---------------------------------------------------------
    if best_vec is None:
        # a trivial uniform fallback
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    final_seq = best_vec.tolist()
    # final sanity check with the supplied evaluator (clamping, etc.)
    final_val = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_val:.6f}")
    return final_seq
