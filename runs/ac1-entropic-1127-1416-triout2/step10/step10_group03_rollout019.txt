"""
Improved search for a non‑negative step‑function that minimises
        evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of four phases:

1. **Rich seeding** – a large collection of hand‑crafted families
   (β‑shapes, exponentials, power‑laws, tiny edge‑spikes) and random
   Dirichlet draws.  Every candidate is projected onto the probability
   simplex (∑a = 1) and the exact objective (2·n·max‑conv) is evaluated.
   The best seed is kept.

2. **Adam‑accelerated soft‑max gradient descent** – we minimise the
   smooth surrogate
        Lβ(a)=log Σ exp(β·conv(a,a))
   whose gradient is  ∇Lβ = 2β·correlate(softmaxβ(conv),a).
   The temperature β is annealed from a small value to a large one.
   An Adam optimiser (with projection onto the simplex) performs the
   updates.  The best iterate (according to the true objective) is
   retained.

3. **Simulated‑annealing polishing** – stochastic mass‑transfer moves
   are attempted for the remaining time budget; a move is kept if it
   improves the exact score or with a probability that decreases
   according to a linear temperature schedule.  This helps to escape
   shallow local minima.

4. **Return** – the final vector (still normalised to sum 1) is handed
   back as a plain ``list[float]``.  The harness will evaluate it with
   the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def generate_deterministic(n: int) -> np.ndarray:
    """Sin^2 shape a_i = sin^2(pi·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p concentrates mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    # work in log‑space to avoid overflow for large |p|
    logw = p * (np.log(i + 1.0) + np.log(n - i))
    w = np.exp(logw)
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)"""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def generate_powerlaw(n: int, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-q}  (q>0 gives decreasing tail)"""
    i = np.arange(n, dtype=float) + 1.0
    a = i ** (-q)
    a /= a.sum()
    return a


def generate_piecewise_edge(n: int, k: int, ratio: float) -> np.ndarray:
    """
    Very heavy mass on the first and last `k` positions,
    the remaining (n‑2k) positions share the residual uniformly.
    """
    if k <= 0 or 2 * k >= n:
        # fallback to uniform
        return np.full(n, 1.0 / n, dtype=float)

    mid_len = n - 2 * k
    w_mid = 1.0 / (2 * k * ratio + mid_len)
    w_edge = ratio * w_mid
    a = np.full(n, w_mid, dtype=float)
    a[:k] = w_edge
    a[-k:] = w_edge
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: k distinct positions with random positive weights."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a

def adam_softmax_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    lr: float = 0.003,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises the smooth surrogate
    Lβ(a)=log Σ exp(β·conv(a,a)).
    The temperature β is linearly annealed.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    # Adam moments
    m = np.zeros_like(cur_a)
    v = np.zeros_like(cur_a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    start = time.time()
    while time.time() < deadline:
        t += 1
        # exact objective for bookkeeping
        conv = convolution_fft(cur_a, cur_a)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # ----- temperature schedule -----
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # ----- soft‑max weights -----
        maxc = conv.max()
        shifted = conv - maxc               # for numerical stability
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # ----- gradient of the surrogate -----
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        # ----- Adam update (projected) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        nxt = cur_a - step
        cur_a = project_to_simplex(nxt)

        # occasional progress report
        if t % 5000 == 0:
            print(
                f"[Adam] it {t:6d} | time {elapsed:6.1f}s | best {best_score:.6f} | β {beta:.1f}"
            )

    return best_a, best_score

# Phase‑3: Simulated‑annealing polishing

def simulated_annealing(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    init_temp: float = 1e-3,
    final_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a random fraction of mass from one index to another.
    Accepts worse moves with probability exp(-Δ/T).  Temperature decays linearly.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        temp = init_temp * (1.0 - fraction) + final_temp * fraction

        # propose a move
        i, j = np.random.randint(n, size=2)
        if cur_a[i] == 0.0 or i == j:
            continue

        # move a random fraction (up to 30 %) of the mass from i to j
        frac = np.random.rand() * 0.3
        delta = frac * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # projection is not needed – the move preserves sum and non‑negativity
        cand_score = compute_score(cand)
        delta_score = cand_score - cur_score

        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / max(temp, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()

    return best_a, best_score

    # 1. Choose a moderate length (2048 works well)
    # --------------------------------------------------------------
    n = 2048
    if total_time > 850.0:
        n = 4096
    print(f"Using sequence length = {n}")

    # --------------------------------------------------------------
    # 2. Generate a large variety of seeds and keep the best one
    # --------------------------------------------------------------
    seeds = []

    # deterministic baselines
    seeds.append(generate_deterministic(n))

    # β‑family with a fine sweep (negative favours edges)
    for p in np.linspace(-4.5, -0.2, 23):
        seeds.append(generate_beta_family(n, p))

    # exponential family (edge‑heavy)
    for lam in np.linspace(0.0005, 0.025, 30):
        seeds.append(generate_exponential(n, lam))

    # power‑law decreasing (edge‑heavy)
    for q in np.linspace(0.5, 3.0, 20):
        seeds.append(generate_powerlaw(n, q))

    # piecewise edge – try a few edge‑width / ratio combos
    for k in (1, 2, 5, 10, 20):
        for r in (4.0, 8.0, 12.0):
            seeds.append(generate_piecewise_edge(n, k, r))

    # random Dirichlet draws (various concentrations)
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(5):
            seeds.append(random_dirichlet(n, conc))

    # very sparse spikes – often a good starting point
    for k in (30, 50, 80, 120):
        seeds.append(random_spike(n, k=k))

    # if we already have a fallback of the right length, add it
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.asarray(best_seq, dtype=float) / np.sum(best_seq))

    # ------------------------------------------------------------------
    # Evaluate all seeds (exact objective) and remember the best
    # ------------------------------------------------------------------
    for idx, s in enumerate(seeds):
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)          # guard against tiny drift
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[Seed {idx:03d}] new best score = {best_val:.6f}")

    cur_a = np.array(best_seq, dtype=float)
    cur_a = project_to_simplex(cur_a)   # ensure exact simplex
    cur_score = compute_score(cur_a)

    # --------------------------------------------------------------
    # 3. Adam soft‑max gradient descent (major part of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + total_time * 0.60
    print("\n[Phase 1] Adam soft‑max gradient descent")
    cur_a, cur_score = adam_softmax_descent(
        cur_a,
        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=2500.0,
        lr=0.003,
    )
    if cur_score < best_val:
        best_val = cur_score
        best_seq = cur_a.tolist()
    print(f"After Adam: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing polishing (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n[Phase 2] Simulated‑annealing polishing")
        cur_a, cur_score = simulated_annealing(
            cur_a,
            cur_score,
            deadline,
            init_temp=1e-3,
            final_temp=1e-7,
        )
        if cur_score < best_val:
            best_val = cur_score
            best_seq = cur_a.tolist()
        print(f"After SA: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final conversion and official evaluation
    # --------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
