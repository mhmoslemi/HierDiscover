    n = len(a)

    best_sc = compute_score(a)

        k_max = int(conv.argmax())

def simulated_annealing(start_seq, deadline, max_iters=500_000):
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)
    # temperature schedule: start at T0 = best_sc * 0.1 maybe
    T = best_sc * 0.1
    T_factor = 0.9995 # exponential decay per iteration
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # pick source and destination
        i = np.random.randint(n)
        j = (i + np.random.randint(1, n)) % n  # ensure different
        # propose moving a fraction f of mass from i to j
        f = np.random.rand() * 0.1  # up to 10% of source
        delta = f * a[i]
        if delta == 0:

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        dE = sc - best_sc
        if dE < 0 or np.exp(-dE / T) > np.random.rand():
            a = cand
            if sc < best_sc:
                best_sc, best_a = sc, cand.copy()
        T *= T_factor
    return best_a, best_sc

    a /= a.sum()

    v = np.zeros_like(z)
    best_a = a.copy()

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute conv
        conv = convolution_fft(a, a)
        k_max = conv.argmax()

        i_end = min(n - 1, k_max)

        # grad w.r.t z

        # Adam updates

        a = np.maximum(a, 0.0)
        a_sum = a.sum()

    sc = compute_score(a)

"""
A fast, multi‑phase optimiser for the step‑function “max‑convolution” problem.

The algorithm:
1. Choose a problem size n (larger is better if time permits).
2. Exhaustively scan a dense exponential family – this one‑parameter family already
   beats the uniform distribution and usually lands close to the optimum.
3. Refine the best candidate with an Adam‑style mirror descent on the exact
   max‑convolution (KL‑mirror on the softmax representation).  This uses the true
   sub‑gradient (the active convolution entry) and converges much faster than
   simple exponentiated‑gradient.
4. Run a targeted hill‑climb that repeatedly removes mass from the index that
   contributes most to the current maximal convolution entry and redistributes it
   to the index with the smallest contribution.  The move is accepted only if
   it improves the exact objective; the step size is adapted automatically.
5. Return the best sequence seen.

All operations keep the vector on the probability simplex (non‑negative,
sum = 1), which makes the evaluation invariant under scaling and matches the
behaviour of `evaluate_sequence`.  The routine respects the 1000 s wall‑clock
budget and prints progress information useful for debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector that already
    satisfies the simplex constraint (sum = 1).  It returns
        2·n·max(a*a)
    where the convolution is full (length 2·n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam == 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p == 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta distribution on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


# ----------------------------------------------------------------------
# Scanning helpers
# ----------------------------------------------------------------------


def scan_exponential(n: int, deadline: float,
                     points: int = 2500) -> tuple[np.ndarray, float]:
    """Dense scan of the exponential family."""
    best_a, best_sc = None, float("inf")
    # lam = c / n, we search c in [0.1, 10] (covers the useful region)
    c_grid = np.linspace(0.1, 10.0, points)
    for c in c_grid:
        if time.time() >= deadline:
            break
        lam = c / n
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def scan_power(n: int, deadline: float,
               points: int = 500) -> tuple[np.ndarray, float]:
    """Simple scan of the power‑law family."""
    best_a, best_sc = None, float("inf")
    p_grid = np.linspace(0.0, 8.0, points)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def scan_beta(n: int, deadline: float,
              points: int = 200) -> tuple[np.ndarray, float]:
    """Very modest scan of a 2‑parameter Beta family."""
    best_a, best_sc = None, float("inf")
    a_grid = np.logspace(-0.7, 0.5, points)
    b_grid = a_grid.copy()
    for a in a_grid:
        for b in b_grid:
            if time.time() >= deadline:
                return best_a, best_sc
            vec = beta_shape(n, a, b)
            sc = compute_score(vec)
            if sc < best_sc:
                best_sc, best_a = sc, vec.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Adam‑style mirror descent on the exact max‑convolution
# ----------------------------------------------------------------------


def adam_max_descent(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 800_000,
                     lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (softmax) Adam optimisation using the true max convolution.
    The parameters (lr, β1, β2) are tuned for stability on large n.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    # work in the log‑softmax space
    z = np.log(a + 1e-30)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and sub‑gradient of the active entry
        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        # gradient w.r.t. softmax parameters
        avg = np.dot(a, g)
        grad_z = a * (g - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # back to probability simplex
        a = np.exp(z - np.max(z))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Aggressive hill‑climb (deterministic destination)
# ----------------------------------------------------------------------


def aggressive_hill_climb(start_seq: np.ndarray,
                          deadline: float,
                          max_steps: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Greedy walk that always extracts a fixed fraction of mass from the index
    that contributes most to the current maximal convolution entry and adds it
    to the index that contributes the least (both measured on the same k).
    The step size is adapted: it grows when moves succeed, shrinks otherwise.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_frac = 0.10
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)

        # contribution of each index to conv[k_max]
        contrib = a[idx] * a[k_max - idx]

        i_src = idx[np.argmax(contrib)]                     # most harmful
        i_dst = idx[np.argmin(contrib)]                     # most helpful

        # avoid a no‑op
        if i_src == i_dst:
            i_dst = (i_dst + 1) % n

        delta = step_frac * a[i_src]
        if delta <= 0.0:
            break

        cand = a.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
            step_frac = min(step_frac * 1.07, 0.5)   # be bold after success
        else:
            step_frac *= 0.96                       # become more cautious
            a = cand                                # still move (allows escape)

        steps += 1

    return best_a, best_sc
