"""
Propose a non‑negative step‑function (list of floats) that minimises
`evaluate_sequence`.  The algorithm combines:

1.  A broad sweep of analytic families (U‑shaped, power‑law, exponential,
    beta, sine‑inverse) with a coarse grid, followed by a one‑dimensional
    Brent optimisation of the most promising parameters.
2.  A handful of random draws (Dirichlet, sparse, monotone) for extra
    diversity.
3.  L‑BFGS‑B optimisation of a smooth log‑sum‑exp surrogate of the max‑
    convolution, using a schedule of β = [200, 500, 1000, 2000].
4.  Mirror (exponentiated) sub‑gradient descent on the true max‑objective,
    with occasional random perturbations to escape shallow plateaus.
5.  Symmetrisation (many optimal solutions are symmetric) and a final
    lightweight simulated‑annealing hill‑climb.

All stages respect the global time budget (default 1000 s) and keep only
the best sequence seen so far.  The returned list is ready for the
external `evaluate_sequence` call.
"""

# FFT utilities – linear convolution via real FFT
# ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * b using a real FFT."""
    m = a.size + b.size - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        score = 2·n·max(a * a)
    This matches `evaluate_sequence` up to the normalisation.
    """
    n = a.shape[0]
    conv = _convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ---------------------------------------------------------
# Simplex utilities
# ---------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# Analytic families (all return a probability vector with sum=1)
# ---------------------------------------------------------
def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑shaped density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    a = w / w.sum()
    return a


def sine_inverse_family(n: int, alpha: float) -> np.ndarray:
    """
    U‑shaped based on the sine function:
        a_i ∝ sin(π·(i+0.5)/(n+1))^(‑α)
    The ends (where sine is small) receive larger weight.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    s = np.sin(np.pi * i / (n + 1))
    w = np.power(s + 1e-12, -alpha)
    a = w / w.sum()
    return a

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse probability vector (≈5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))

    a = np.zeros(n, dtype=np.float64)

    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = a / a.sum()
    return a

    Small concentration → spikier vectors, large → almost uniform.
    """
    alpha = np.full(n, concentration, dtype=np.float64)
    a = np.random.dirichlet(alpha)
    return a


# ---------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) utilities for L‑BFGS‑B
# ---------------------------------------------------------
def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where
        f   = (1/β)·log∑_k exp[β·(a*a)_k]
        grad = ∂f/∂z   (z are the log‑soft‑max parameters)
    """
    eps = 1e-12
    a = softmax(z + eps)                     # a ∈ simplex
    conv = _convolution_fft(a, a)            # length 2n‑1
    maxc = conv.max()
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # Gradient w.r.t. a (see derivation in the previous state‑of‑the‑art)
    w = exp_terms / sum_exp                  # shape (2n‑1,)
    a_rev = a[::-1]
    conv_rev = _convolution_fft(a_rev, w)    # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1: 2 * n - 1]

    # Chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray,
                  deadline: float,
                  beta_list=(200.0, 500.0, 1000.0, 2000.0),
                  max_iter_per_beta=1200) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector using L‑BFGS‑B on the smoothed surrogate.
    Returns the best vector found and its exact score.
    """
    if not _HAS_SCIPY:          # pragma: no cover
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)    # initialise log‑softmax parameters

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break

        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 10))

        def func(z_vec):
            return _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])

        try:
            opt_res = _opt.fmin_l_bfgs_b(
                func,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved score = {best_score:.6f}")

    return best_a, best_score


# ---------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max‑objective
# ---------------------------------------------------------
def mirror_subgrad_optimize(start_a: np.ndarray,
                            deadline: float,
                            step0: float = 0.9,
                            perturb_scale: float = 0.02,
                            max_no_improve: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on F(a)=2·n·max(a*a).
    Periodic random perturbations break stagnation.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    last_improve = 0

    while time.time() < deadline:
        t += 1
        conv = _convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # Renormalise → simplex (entropy projection)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # Check for improvement every few thousand iterations
        if t % 5000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                last_improve = t
                print(f"[{time.time():.1f}] mirror iter {t} new best = {best_score:.6f}")

        # If we have not improved for a while, inject a small perturbation
        if t - last_improve > max_no_improve:
            perturb = 1.0 + perturb_scale * np.random.randn(n)
            perturb = np.maximum(perturb, 0.0)
            a = a * perturb
            a = project_to_simplex(a)
            last_improve = t
            # (no immediate score check – continue optimisation)

    # final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ---------------------------------------------------------
# Tiny simulated‑annealing hill‑climb (final polishing)
# ---------------------------------------------------------
def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000,
                  init_temp_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature SA that swaps or transfers a fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# High‑level driver
# ---------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Main entry point used by the harness.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs more)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 16384          # a good trade‑off for the 1000 s budget
    else:
        n = 8192
    print(f"[{time.time():.1f}] Using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously known construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Helper to keep the top‑K candidates (for later refinement)
    # ------------------------------------------------------------------
    TOP_K = 8
    top_candidates = []      # list of (score, ndarray)

    def consider(vec: np.ndarray):
        nonlocal best_seq, best_score, top_candidates
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new global best = {best_score:.6f}")
        # maintain top‑K list
        if len(top_candidates) < TOP_K:
            top_candidates.append((sc, vec.copy()))
        else:
            # replace worst if better
            max_idx = max(range(len(top_candidates)), key=lambda i: top_candidates[i][0])
            if sc < top_candidates[max_idx][0]:
                top_candidates[max_idx] = (sc, vec.copy())

    # ------------------------------------------------------------------
    # 1. Scan analytic families on a coarse grid
    # ------------------------------------------------------------------
    # U‑shaped (product) family
    for alpha in np.linspace(0.0, 12.0, 121):
        if time.time() >= deadline - 30.0:
            break
        consider(symmetric_product_family(n, alpha))

    # Left‑heavy power law
    for beta in np.linspace(0.0, 12.0, 121):
        if time.time() >= deadline - 30.0:
            break
        consider(left_heavy_power_family(n, beta))

    # Exponential decay
    for gamma in np.linspace(0.0, 10.0, 101):
        if time.time() >= deadline - 30.0:
            break
        consider(exponential_decay_family(n, gamma))

    # Symmetric beta distribution
    for alpha in np.linspace(0.0, 12.0, 101):
        if time.time() >= deadline - 30.0:
            break
        consider(beta_distribution_family(n, alpha))

    # Sine‑inverse U‑shape
    for alpha in np.linspace(0.0, 8.0, 81):
        if time.time() >= deadline - 30.0:
            break
        consider(sine_inverse_family(n, alpha))

    # ------------------------------------------------------------------
    # 2. Refine the promising parameters of the two best families
    #    (local 1‑D optimisation via Brent if SciPy is available)
    # ------------------------------------------------------------------
    if _HAS_SCIPY and time.time() < deadline - 10.0:
        # Find the alpha that gave the best symmetric_product_family
        best_alpha = min(
            [(np.linspace(0.0, 12.0, 121)[i], top_candidates[i][0])
             for i in range(len(top_candidates)) if top_candidates[i][1].size == n],
            key=lambda x: x[1],
            default=(None, None),
        )[0]
        if best_alpha is not None:
            res = _opt.minimize_scalar(
                lambda a: compute_score(symmetric_product_family(n, a)),
                bounds=(max(0.0, best_alpha - 2.0), min(12.0, best_alpha + 2.0)),
                method='bounded',
                options={'xatol': 1e-3, 'maxiter': 30},
            )
            if res.success:
                consider(symmetric_product_family(n, res.x))

        # Same for left_heavy_power_family
        best_beta = min(
            [(np.linspace(0.0, 12.0, 121)[i], top_candidates[i][0])
             for i in range(len(top_candidates)) if top_candidates[i][1].size == n],
            key=lambda x: x[1],
            default=(None, None),
        )[0]
        if best_beta is not None:
            res = _opt.minimize_scalar(
                lambda b: compute_score(left_heavy_power_family(n, b)),
                bounds=(max(0.0, best_beta - 2.0), min(12.0, best_beta + 2.0)),
                method='bounded',
                options={'xatol': 1e-3, 'maxiter': 30},
            )
            if res.success:
                consider(left_heavy_power_family(n, res.x))

    # ------------------------------------------------------------------
    # 3. Random starts (Dirichlet with several concentrations + sparse)
    # ------------------------------------------------------------------
    concentrations = [0.1, 0.3, 0.6, 1.0, 2.0]
    for conc in concentrations:
        for _ in range(30):
            if time.time() >= deadline - 30.0:
                break
            consider(random_dirichlet_start(n, conc))

    # Additional sparse/monotone draws
    for _ in range(100):
        if time.time() >= deadline - 30.0:
            break
        consider(random_sparse_start(n))
    for _ in range(100):
        if time.time() >= deadline - 30.0:
            break
        consider(random_monotone_start(n))

    # ------------------------------------------------------------------
    # 4. L‑BFGS refinement of the top‑K candidates
    # ------------------------------------------------------------------
    if time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Starting L‑BFGS refinement on {len(top_candidates)} candidates")
        for sc, vec in top_candidates:
            a_start = vec.astype(np.float64).copy()
            a_ref, sc_ref = l_bfgs_refine(
                a_start,
                deadline - 200.0,
                beta_list=(200.0, 500.0, 1000.0, 2000.0),
                max_iter_per_beta=1200,
            )
            consider(a_ref)

    # ------------------------------------------------------------------
    # 5. Mirror sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] Mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            step0=0.9,
        )
        consider(a_sub)

    # ------------------------------------------------------------------
    # 6. Symmetrisation (many optima are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        consider(a_sym)

    # ------------------------------------------------------------------
    # 7. Final tiny SA hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
        )
        consider(a_final)

    # ------------------------------------------------------------------
    # 8. Safeguard: if nothing succeeded, fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 9. Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
