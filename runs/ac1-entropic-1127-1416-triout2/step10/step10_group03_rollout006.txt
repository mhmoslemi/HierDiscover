"""
Advanced search for a step‑function that minimizes

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Key ideas
---------
* Work with **symmetric** sequences – the optimal solution is invariant under
  reversal, which halves the search space.
* Use a **log‑sum‑exp surrogate** Lβ(a)=log∑exp(β·conv(a,a)) and minimise it
  with **Adam** (adaptive moments) while the temperature β is linearly
  increased from a small to a very large value.  For β→∞ the surrogate
  converges to the true max‑norm.
* The iterates are projected onto the probability simplex (Euclidean
  projection) after each update, guaranteeing non‑negativity and a fixed
  sum of 1.  Symmetrisation is performed after every projection.
* A short **stochastic hill‑climb** finishes the optimisation, using tiny
  mass‑transfer moves that are kept only when they improve the exact objective.
* The whole procedure respects the 1000 s wall‑clock budget, automatically
  choosing a longer sequence (4096) when enough time is available.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for many calls)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a probability vector (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x | x_i ≥ 0, Σx_i = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry: v = (v + v[::-1]) / 2."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Seed generators (all produce non‑negative vectors, later symmetrised)
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """a_i = sin^2(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a = symmetrize(a)
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    p>0 concentrates mass in the centre, p<0 pushes it towards the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = symmetrize(w)
    a = a / a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc) and symmetrise."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a = symmetrize(a)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse symmetric vector: choose `k` distinct positions (and their mirrors)
    and assign random positive weights, then normalise.
    """
    a = np.zeros(n, dtype=float)
    # pick positions only from the first half, mirror them automatically
    half = (n + 1) // 2
    idx = np.random.choice(half, size=min(k, half), replace=False)
    w = np.random.rand(len(idx))
    a[idx] = w
    # mirror
    a[n - 1 - idx] = w
    if n % 2 == 1:
        a[n // 2] = np.random.rand()  # centre may get its own weight
    a /= a.sum()
    return a

def adam_softmax_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    start_phase_time: float,
    beta_start: float = 2.0,
    beta_end: float = 8000.0,
    lr: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate  Lβ(a)=log∑exp(β·conv(a,a)) using Adam.
    β is increased linearly from `beta_start` to `beta_end` across the phase.
    Each iteration projects onto the probability simplex and enforces symmetry.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    m = np.zeros_like(cur)          # first moment
    v = np.zeros_like(cur)          # second moment
    it = 0

    while time.time() < deadline:
        # ---- exact objective for bookkeeping ----
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # ---- temperature schedule ----
        fraction = (time.time() - start_phase_time) / (deadline - start_phase_time)
        beta = beta_start + fraction * (beta_end - beta_start)

        # ---- soft‑max weights ----
        shifted = conv - max_conv                # stability shift
        exp_vals = np.exp(beta * shifted)       # all ≤ 1
        w = exp_vals / exp_vals.sum()           # soft‑max distribution

        # ---- gradient of the surrogate ----
        grad = 2.0 * beta * np.correlate(w, cur, mode="valid")

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step_factor = lr / (np.sqrt(it) + 1.0)      # extra decay for stability
        step = step_factor * m_hat / (np.sqrt(v_hat) + eps)

        nxt = cur - step
        nxt = project_to_simplex(nxt)
        nxt = symmetrize(nxt)                        # keep symmetry
        cur = nxt

        # occasional jitter to shake out of plateaus
        if it % 2000 == 0:
            cur += np.random.randn(n) * (lr * 0.002 / np.sqrt(it))
            cur = project_to_simplex(cur)
            cur = symmetrize(cur)

    return best, best_score


def hill_climb(
    a: np.ndarray,
    deadline: float,
    step_scale_init: float = 0.06,
    max_steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that keeps a move only when it strictly improves the
    exact objective.  Moves consist of transferring a small fraction of mass
    from one index to another (or swapping two entries).  Symmetry is restored
    after each move.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = step_scale_init
    steps = 0

    while steps < max_steps and time.time() < deadline:
        cand = best.copy()

        # decide move type
        if np.random.rand() < 0.5:
            # swap two entries (useful when the optimum is very sparse)
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small chunk from i to j
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        # enforce constraints
        cand = symmetrize(cand)
        cand = project_to_simplex(cand)   # correct any tiny drift

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)   # become bolder after success
        else:
            step_scale *= 0.96                         # become more cautious

        steps += 1

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the surrounding harness
    deadline = start_time + budget_s - 0.5
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # Determine a reasonable sequence length.
    # --------------------------------------------------------------
    # 4096 works well if we have at least ~800 s total; otherwise fall back to 2048.
    if total_time >= 800.0:
        n = 4096
    else:
        n = 2048
    print(f"[Info] Using sequence length n = {n}")

    # --------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence.
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        # height_sequence_1 may be defined in the global scope by the harness
        fallback = [float(x) for x in globals()["height_sequence_1"]]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[Info] Fallback score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Generate a diverse pool of seeds (all symmetrised and normalised).
    # --------------------------------------------------------------
    seeds = []

    # deterministic sine‑squared shape
    seeds.append(generate_deterministic(n))

    # beta‑family with a fine grid of exponents
    for p in np.linspace(-3.0, 3.0, 13):
        seeds.append(generate_beta_family(n, p))

    # Dirichlet draws with several concentrations
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # sparse spikes – often good for reducing the maximum of the auto‑convolution
    for k in (30, 60, 90):
        seeds.append(random_spike(n, k=k))

    # if the fallback matches the current length, use it as a seed too
    if best_seq is not None and len(best_seq) == n:
        fallback_arr = np.array(best_seq, dtype=float)
        fallback_arr = fallback_arr / fallback_arr.sum()
        seeds.append(fallback_arr)

    # --------------------------------------------------------------
    # 2. Evaluate all seeds and keep the best one.
    # --------------------------------------------------------------
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)          # guard against tiny drifts
        s = symmetrize(s)
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[Seed] improved score = {best_val:.6f}")

    # Prepare the current best vector for further optimisation
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)
    print(f"[Info] Starting optimisation with score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # 3. Main optimisation – Adam on the log‑sum‑exp surrogate.
    #    Use ~70 % of the budget for this phase.
    # --------------------------------------------------------------
    phase1_deadline = start_time + total_time * 0.70
    phase1_start = time.time()
    cur_a, cur_score = adam_softmax_descent(
        cur_a,
        cur_score,
        phase1_deadline,
        phase1_start,
        beta_start=2.0,
        beta_end=8000.0,
        lr=0.07,
    )
    if cur_score < best_val:
        best_val = cur_score
    print(f"[Phase 1] End of Adam descent – best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final stochastic hill‑climb polishing (remaining time).
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Phase 2] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, step_scale_init=0.06)
        if cur_score < best_val:
            best_val = cur_score
            print(f"[Phase 2] polishing improved score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Return the result (the official harness will call evaluate_sequence).
    # --------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\n[Result] final reported score = {final_val:.6f}")

    return final_seq
