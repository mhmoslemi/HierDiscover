Improved search for the max‑auto‑convolution problem.

The algorithm proceeds in several stages within the 1000 s wall‑clock limit:

1. **Coarse analytic sweep** – many parametric families (U‑shape, power‑law,
   beta‑type, sinusoidal, Gaussian, mixed) are sampled on a modest size
   (n = 1024) and the best candidate is kept.

2. **Smooth‑max Adam refinement** – the best candidate is further optimised
   using a soft‑max approximation of max‑convolution.  The temperature β is
   increased exponentially up to 1500 and Adam with a modest learning‑rate
   is run for up to 250 k iterations.

3. **Linear‑programming “LP‑boost”** – an iteration of the classical LP
   step (maximise ∑g_j under the constraint f∗g ≤ M) is performed several
   times.  The LP is solved with SciPy’s HiGHS driver; the new vector is
   blended with the current one (t = 0.5) and projected onto the simplex.
   This step often pushes the solution past the barrier encountered by pure,
   gradient‑based methods.

4. **Second Adam pass** – a finer Adam optimisation with a larger β‑range
   (up to 3000) consolidates the improvement.

5. **Final simulated‑annealing hill‑climb** – a long random walk on the
   simplex (swap / mass‑transfer moves) with a decaying temperature cleans
   up any remaining irregularities.

All loops respect the remaining time budget; the routine returns a plain
Python list of non‑negative heights.  The objective is invariant under
positive scaling, so the vectors are kept normalised to unit L¹‑norm.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1).
    Returns 2·n·max(a∗a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    return lam * uniform + (1.0 - lam) * u_shape


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density ∝ (i+½)^{p‑1}·(n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    # avoid NaNs when p,q<1 at the boundaries
    w = np.where(np.isfinite(w), w, 0.0)
    w[w < 0] = 0.0
    return w / w.sum()


def sin_power_family(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+½)/n), symmetric, zero at the ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** power
    # tiny epsilon to avoid exact zeros which would kill the simplex projection
    w += 1e-12
    return w / w.sum()


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w += 1e-12
    return w / w.sum()


# ----------------------------------------------------------------------
# Random initialisers (used for diversification)
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse uniform‑mass vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Decreasing vector obtained by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 5.0,
                         beta_end: float = 1500.0,
                         lr: float = 0.018,
                         sym_interval: int = 3000) -> Tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution
    f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    via Adam on the soft‑max parametrisation a = softmax(z).
    A symmetrisation step is performed every `sym_interval` iterations.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise in log‑softmax space

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights for the smooth max gradient
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t a (derived from smooth max)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional explicit scoring
        if it % 2000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection / symmetrisation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (lightweight)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> Tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
      – 50 % swap two entries,
      – 50 % transfer a fraction of mass.
    Temperature decays linearly; occasional uphill moves are allowed.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear temperature decay
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based “boost” step (maximise L1 under convolution bound)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    import scipy.sparse as sp
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def lp_boost(a: np.ndarray,
             max_steps: int,
             deadline: float) -> Tuple[np.ndarray, float]:
    """
    Repeatedly solve:
        maximise   ∑ g_i
        subject to f∗g ≤ M   (all convolution indices)
                  g_i ≥ 0
    where f is the current vector and M = max(f∗f).
    The resulting g is blended with f (t = 0.5) and projected onto the simplex.
    """
    if not _SCIPY_AVAILABLE:
        return a, compute_score(a)

    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        M = conv.max()

        # Build sparse constraint matrix: each row k contains coefficients a_i
        # for g_{k-i} when the index is valid.
        rows, cols, data = [], [], []
        for k in range(2 * n - 1):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # i runs over all indices where a_i participates in (f∗g)_k
            i_vals = np.arange(i_start, i_end + 1, dtype=np.int32)
            j_vals = k - i_vals
            rows.extend([k] * len(i_vals))
            cols.extend(j_vals.tolist())
            data.extend(a[i_vals].tolist())
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if not res.success:
            # fallback – stop the LP loop
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break

        # Keep the total L¹‑mass constant (blend with current a)
        g_scaled = g * (a.sum() / sum_g)
        t = 0.5
        a = (1.0 - t) * a + t * g_scaled
        a = (a + a[::-1]) * 0.5               # enforce symmetry (often beneficial)
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
            print(f"[{time.time():.1f}] LP boost step {step+1} improved to {best_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence minimising
    ``evaluate_sequence``.  The routine respects a wall‑clock budget and
    returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # choose a moderate length – the objective is scale‑invariant,
    # smaller n speeds up FFT/LP dramatically.
    n = 1024

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to update the incumbent
    # ------------------------------------------------------------------
    def consider(vec: np.ndarray, tag: str) -> None:
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] New best ({tag}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 0. Optional fallback to the previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = compute_score(np.array(fb_vec))
        if fb_score < best_score:
            best_score = fb_score
            best_seq = fb_vec
            print(f"[{time.time():.1f}] Fallback (previous) = {best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families (coarse scan)
    # ------------------------------------------------------------------
    families = []

    # symmetric product (α)
    families.append(('sym-prod',
                     lambda n_: [symmetric_product_family(n_, a) for a in np.linspace(0.5, 12.0, 24)]))

    # left‑heavy power (β)
    families.append(('left‑heavy',
                     lambda n_: [left_heavy_power_family(n_, b) for b in np.linspace(0.2, 12.0, 24)]))

    # exponential U‑shape (γ)
    families.append(('exp‑U',
                     lambda n_: [exp_u_family(n_, g) for g in np.linspace(0.0, 6.0, 20)]))

    # beta‑type (p,q)
    def beta_grid(n_):
        out = []
        for p in np.linspace(0.1, 3.0, 9):
            for q in np.linspace(0.1, 3.0, 9):
                out.append(beta_family(n_, p, q))
        return out
    families.append(('beta', beta_grid))

    # mixed uniform‑U
    def mix_grid(n_):
        out = []
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            for g in np.linspace(0.0, 6.0, 12):
                out.append(mixed_uniform_u_family(n_, lam, g))
        return out
    families.append(('mix‑U', mix_grid))

    # sin‑power
    families.append(('sin‑pow',
                     lambda n_: [sin_power_family(n_, p) for p in np.linspace(0.5, 12.0, 20)]))

    # Gaussian
    families.append(('gauss',
                     lambda n_: [gaussian_family(n_, s) for s in np.linspace(0.1, n_/2.0, 15)]))

    # random starts (few to diversify)
    families.append(('rand‑sparse', lambda n_: [random_sparse_start(n_) for _ in range(6)]))
    families.append(('rand‑mono',   lambda n_: [random_monotone_start(n_) for _ in range(6)]))
    families.append(('rand‑U',      lambda n_: [random_u_start(n_) for _ in range(6)]))

    for tag, generator in families:
        if time.time() >= deadline - 30.0:
            break
        for vec in generator(n):
            if time.time() >= deadline - 30.0:
                break
            vec = np.asarray(vec, dtype=np.float64)
            vec = project_to_simplex(vec)      # safety
            consider(vec, tag)

    # ------------------------------------------------------------------
    # 2. First Adam (smooth‑max) refinement on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] First Adam refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 120.0,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=1500.0,
            lr=0.018,
            sym_interval=3000,
        )
        consider(a_opt, "adam1")

    # ------------------------------------------------------------------
    # 3. LP boost (maximise L1 while keeping convolution ≤ current bound)
    # ------------------------------------------------------------------
    if best_seq is not None and _SCIPY_AVAILABLE and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] LP boost phase")
        a_lp, sc_lp = lp_boost(np.array(best_seq, dtype=np.float64),
                               max_steps=7,
                               deadline=deadline - 90.0)
        consider(a_lp, "lp‑boost")

    # ------------------------------------------------------------------
    # 4. Second, finer Adam pass (higher β)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] Second Adam refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = adam_smooth_optimize(
            a_start,
            deadline - 40.0,
            max_iters=200_000,
            beta_start=50.0,
            beta_end=3000.0,
            lr=0.015,
            sym_interval=2000,
        )
        consider(a_opt2, "adam2")

    # ------------------------------------------------------------------
    # 5. Explicit symmetrisation test (often helps)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        consider(a_sym, "symmetrised")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
        )
        consider(a_final, "SA")

    # ------------------------------------------------------------------
    # 7. Safeguard fallback (very simple uniform vector)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = compute_score(np.array(best_seq))

    # Final sanity evaluation via the provided `evaluate_sequence`
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] Search finished – final reported score = {final_score:.6f}")

    return best_seq
