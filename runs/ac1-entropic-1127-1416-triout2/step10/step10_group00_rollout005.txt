"""
Proposes a step‑function (non‑negative sequence) that minimises the
objective used in the challenge

    score = 2·n·max(convolve(a, a)) / (sum(a)**2)

The algorithm works inside the given time budget:

1. **Exploratory sampling** – several analytic families (U‑shape,
   power‑law, beta‑type, mixtures) are sampled at a few different
   lengths (1024, 2048, 4096, 8192).  The exact objective is evaluated
   with an FFT‑based convolution and the best candidate is kept.

2. **Smooth‑max Adam refinement** – the best candidate is refined three
   times using Adam on a smooth approximation of the max‑convolution.
   The optimisation is performed in the soft‑max (log‑probability)
   parametrisation, enforces the simplex constraint and periodically
   symmetrises the vector.

3. **Final simulated‑annealing hill‑climb** – a long random walk on the
   simplex is performed if time permits.

The routine respects the wall‑clock limit and returns a plain
``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector that sums to 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    return lam * uniform + (1.0 - lam) * u_shape


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density ∝ (i+½)^{p‑1} (n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    # guard against infinities when p or q < 1
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full(n, 1.0 / n)
    return w / w.sum()

# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse uniform‑mass vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Decreasing vector obtained by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 1200.0,
    lr: float = 0.018,
    sym_interval: int = 4000,
) -> Tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation of max‑convolution:
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    Optimisation is performed on the soft‑max parametrisation
    a = softmax(z).  The iterate is periodically symmetrised
    (a ← (a + a[::-1]) / 2) to encourage symmetric solutions.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    # Adam moments
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derived from smooth max)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation for logging
        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 3_000_000,
) -> Tuple[np.ndarray, float]:
    """
    Random walk on the simplex:
        – 50 % chance to swap two entries,
        – 50 % chance to transfer a fraction of mass.
    A slowly decreasing temperature permits occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # start temperature is a tiny fraction of the current score
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0.  Optional fallback from previous run
    # -----------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1.  Random sampling of analytic families at several lengths
    # -----------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 120.0:
            break
        print(f"[{time.time():.1f}] sampling families, n = {n}")

        # keep a small record of the best beta‑family parameters
        best_beta_params = (0.5, 0.5)

        for _ in range(250):
            if time.time() >= deadline - 30.0:
                break

            fam = np.random.choice(["sym", "left", "exp", "mix", "beta"])
            if fam == "sym":
                alpha = np.random.uniform(0.0, 14.0)
                vec = symmetric_product_family(n, alpha)
            elif fam == "left":
                beta = np.random.uniform(0.0, 14.0)
                vec = left_heavy_power_family(n, beta)
            elif fam == "exp":
                gamma = np.random.uniform(0.0, 10.0)
                vec = exp_u_family(n, gamma)
            elif fam == "mix":
                lam = np.random.uniform(0.0, 1.0)
                gamma = np.random.uniform(0.0, 10.0)
                vec = mixed_uniform_u_family(n, lam, gamma)
            else:  # beta family
                p = np.random.uniform(0.05, 5.0)
                q = np.random.uniform(0.05, 5.0)
                vec = beta_family(n, p, q)
                best_beta_params = (p, q)

            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best ({fam}) n={n} = {best_score:.6f}")

        # a few purely random starts (monotone or U‑shape)
        for _ in range(40):
            if time.time() >= deadline - 30.0:
                break
            if np.random.rand() < 0.5:
                vec = random_monotone_start(n)
            else:
                vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) n={n} = {best_score:.6f}")

        # --- fine grid around the best beta parameters (if any) ---
        if best_seq is not None and n == len(best_seq):
            p0, q0 = best_beta_params
            p_vals = np.linspace(max(0.01, p0 - 0.4), p0 + 0.4, 9)
            q_vals = np.linspace(max(0.01, q0 - 0.4), q0 + 0.4, 9)
            for p in p_vals:
                for q in q_vals:
                    if time.time() >= deadline - 20.0:
                        break
                    vec = beta_family(n, p, q)
                    sc = compute_score(vec)
                    if sc < best_score:
                        best_score = sc
                        best_seq = vec.tolist()
                        print(f"[{time.time():.1f}] fine beta (p={p:.2f}, q={q:.2f}) = {best_score:.6f}")

    # -----------------------------------------------------------
    # 2.  Adam‑based smooth‑max refinement (three stages)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        print(f"[{time.time():.1f}] starting Adam refinement")
        a_curr = np.array(best_seq, dtype=np.float64)

        # Stage 1 – moderate smoothing
        a_curr, sc = adam_smooth_optimize(
            a_curr,
            deadline - 70.0,
            max_iters=350_000,
            beta_start=5.0,
            beta_end=1500.0,
            lr=0.018,
            sym_interval=4000,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a_curr.tolist()
            print(f"[{time.time():.1f}] after stage‑1 Adam best = {best_score:.6f}")

        # Stage 2 – finer smoothing, higher β
        if time.time() < deadline - 40.0:
            a_curr, sc = adam_smooth_optimize(
                a_curr,
                deadline - 30.0,
                max_iters=250_000,
                beta_start=50.0,
                beta_end=3000.0,
                lr=0.012,
                sym_interval=2000,
            )
            if sc < best_score:
                best_score = sc
                best_seq = a_curr.tolist()
                print(f"[{time.time():.1f}] after stage‑2 Adam best = {best_score:.6f}")

        # Stage 3 – very high β, very small step size
        if time.time() < deadline - 15.0:
            a_curr, sc = adam_smooth_optimize(
                a_curr,
                deadline - 10.0,
                max_iters=150_000,
                beta_start=200.0,
                beta_end=6000.0,
                lr=0.006,
                sym_interval=1000,
            )
            if sc < best_score:
                best_score = sc
                best_seq = a_curr.tolist()
                print(f"[{time.time():.1f}] after stage‑3 Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3.  Explicit symmetrisation (cheap)
    # -----------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4.  Final simulated‑annealing hill‑climb (if time permits)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64), deadline, max_steps=2_000_000
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5.  Safeguard fallback (should never trigger)
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print("[fallback] using uniform small vector")

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
