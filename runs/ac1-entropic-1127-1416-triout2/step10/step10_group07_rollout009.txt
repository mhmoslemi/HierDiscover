assistantfinal```python
"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

Algorithm (≈ 1000 s)
--------------------
0. Choose a length n (32768 for the full‑time budget, otherwise less).
1. Scan several cheap analytic families (uniform, symmetric‑product,
   left‑heavy power, exponential, Gaussian) and a few Dirichlet draws.
   Keep the best (lowest true objective, i.e. 2·n·max(a*a) for sum(a)=1).
2. Run many short Adam optimisations on the *log‑sum‑exp* surrogate of the
   max‑convolution.  The surrogate is f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k).
   β is increased exponentially from 5 to 5 000.  The vector is
   parametrised by soft‑max logits so the simplex constraint is automatic.
3. Refine the current best vector with a high‑β Adam run
   (β up to 50 000) and a very small learning rate.
4. Perform entropy‑mirror sub‑gradient descent directly on the true
   max‑convolution objective (step size η_t = 0.9/√t).
5. Symmetrise the vector (a←(a+a[::-1])/2) and re‑project onto the simplex.
6. A final low‑temperature simulated‑annealing hill‑climb (tiny SA) is
   performed to escape any remaining shallow minima.
7. Return the best plain Python list found before the time deadline.
All intermediate scores are printed for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (fast, no Python loops)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a) = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection and soft‑max
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for cheap initialisation)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    # avoid log(0) at the very ends – add 0.5 inside the product
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) , monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def gaussian_family(n: int, sigma_factor: float) -> np.ndarray:
    """a_i ∝ exp(-½·((i‑center)/σ)²) , centred, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Draw a symmetric Dirichlet(α) vector (α = conc) and normalise to sum = 1."""
    vec = np.random.gamma(conc, 1.0, size=n)
    vec = vec / vec.sum()
    return vec


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse uniform probability vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)  # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed optimisation (log‑sum‑exp) via Adam on the soft‑max layer
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 5_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β is increased exponentially from beta_start to beta_end.
    Returns the best vector seen (as a plain numpy array) and its exact score.
    """
    eps = 1e-12
    # initialise logits from the start vector (log‑softmax)
    z = np.log(start_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- β schedule (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # current a
        a = softmax(z)

        # ----- smoothed weights w_k ∝ exp(β·conv_k) -----
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))   # stabilised
        w = exp_w / exp_w.sum()                # soft‑max over shifts

        # ----- gradient of f_β w.r.t. a -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # ----- back‑prop through soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}]   Adam iter {it:7d} – new best = {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.9,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror (multiplicative) sub‑gradient descent for
    2·n·max(a*a).  Step size η_t = step0 / sqrt(t).
    Returns best vector seen and its exact score.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)                     # multiplicative update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   Mirror step {t:7d} – new best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (tiny simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_000_000,
    init_temp_factor: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature simulated‑annealing walk on the simplex.
    - with prob 0.5 swap two entries
    - otherwise transfer a fraction of mass from one index to another
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                        # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()           # simplex projection
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` while respecting the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer harness
    deadline = start_time + budget_s - 4.0

    # -------------------------------------------------------------
    # Choose a length.  Larger n gives a lower theoretical bound.
    # -------------------------------------------------------------
    if budget_s >= 950:
        n = 32768
    elif budget_s >= 800:
        n = 16384
    else:
        n = 8192
    print(f"[{time.time():.1f}] Using length n = {n}")

    # -------------------------------------------------------------
    # Global best bookkeeping
    # -------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    def maybe_update(vec: np.ndarray, tag: str = "") -> None:
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<20} = {best_score:.6f}")

    # -------------------------------------------------------------
    # 0. Fallback – previously known construction if it exists
    # -------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = fb_vec
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Cheap analytic families & random Dirichlet draws
    # -------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    maybe_update(uniform, "uniform")

    # symmetric product family (scan a modest grid of α)
    for alpha in np.linspace(0.1, 5.0, 25):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        maybe_update(vec, f"symα={alpha:.2f}")

    # left‑heavy power law
    for beta in np.linspace(0.1, 4.0, 25):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        maybe_update(vec, f"powβ={beta:.2f}")

    # exponential decay family
    for gamma in np.linspace(0.0, 5.0, 25):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, gamma)
        maybe_update(vec, f"expγ={gamma:.2f}")

    # Gaussian‑like centred families
    for sigma_factor in [0.06, 0.09, 0.12, 0.15]:
        if time.time() >= deadline - 30.0:
            break
        vec = gaussian_family(n, sigma_factor)
        maybe_update(vec, f"gaussσ={sigma_factor:.2f}")

    # symmetric Dirichlet draws with a wide range of concentrations
    for conc in np.logspace(-2, 1, 15):
        for _ in range(2):
            if time.time() >= deadline - 30.0:
                break
            vec = random_dirichlet_start(n, conc)
            maybe_update(vec, f"dirα={conc:.3f}")

    # -------------------------------------------------------------
    # 2. Adam smoothing optimisation (many restarts)
    # -------------------------------------------------------------
    while time.time() < deadline - 260.0:
        start_vec = random_sparse_start(n)
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 240.0,
            max_iters=180_000,
            beta_start=5.0,
            beta_end=5_000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}]   after Adam‑smooth = {best_score:.6f}")

    # -------------------------------------------------------------
    # 3. High‑β refinement (tight surrogate)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 130.0:
        print(f"[{time.time():.1f}] high‑β refinement")
        a_ref, sc_ref = adam_smooth_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 110.0,
            max_iters=130_000,
            beta_start=5_000.0,
            beta_end=50_000.0,
            lr=0.008,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.tolist()
            print(f"[{time.time():.1f}]   after high‑β = {best_score:.6f}")

    # -------------------------------------------------------------
    # 4. Mirror sub‑gradient descent on the exact max‑objective
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting mirror descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 50.0,
            step0=0.9,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}]   after mirror = {best_score:.6f}")

    # -------------------------------------------------------------
    # 5. Symmetrisation (many optimal solutions are symmetric)
    # -------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        maybe_update(a_sym, "symmetrised")

    # -------------------------------------------------------------
    # 6. Final low‑temperature hill‑climb (tiny SA)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_200_000,
            init_temp_factor=1e-5,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}]   SA improved best = {best_score:.6f}")

    # -------------------------------------------------------------
    # Safeguard – fall back to a modest uniform vector if everything failed
    # -------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final official evaluation
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
