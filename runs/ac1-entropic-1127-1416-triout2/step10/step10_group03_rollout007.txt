"""
Propose a non‑negative step‑function that (approximately) minimizes

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a three‑phase search:

1. **Seeding** – a diverse collection of deterministic windows
   (sin², Hann, Blackman, cosine, Beta‑family) and random Dirichlet /
   sparse spikes.  The best seed (by the exact objective) is kept.

2. **Adam‑based soft‑max gradient descent** – we minimise the smooth
   surrogate Lβ(a)=log ∑ exp(β·conv(a,a)).  The gradient is obtained
   via an FFT‑based correlation (O(n log n)).  β is annealed from a
   small value up to several thousand while an Adam optimiser with
   Euclidean projection onto the probability simplex updates the
   vector.  The optimiser runs for most of the allotted time.

3. **Final polishing** – stochastic hill‑climbing moves (mass swaps
   and transfers) accepted only when they improve the true objective.

All steps respect the 1000 s wall‑clock budget.  The function returns
a plain ``list[float]`` suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (fast for large n).
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t]   (0 ≤ i < len(a)).
    Implemented as the “valid” part of conv(w, a reversed) using FFT.
    """
    a_rev = a[::-1]
    full = convolution_fft(w, a_rev)
    start = len(a) - 1
    return full[start:start + len(a)]


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum ≈ 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

# Seed generators
# ----------------------------------------------------------------------


def generate_sin2(n: int) -> np.ndarray:
    """a_i = sin²(π·i/(n+1)), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial weight w_i ∝ ((i+1)(n-i))^p."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_hann(n: int) -> np.ndarray:
    """Standard Hann window, normalised."""
    i = np.arange(n, dtype=float)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    a /= a.sum()
    return a


def generate_blackman(n: int) -> np.ndarray:
    """Blackman window, normalised."""
    i = np.arange(n, dtype=float)
    a = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_cosine(n: int) -> np.ndarray:
    """Cosine window (raised‑cosine), normalised."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 0.5) / n)  # half‑period cosine
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration parameter `conc`."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: k random positions with random weights."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


def resample_sequence(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length."""
    if seq.size == target_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.size)
    x_new = np.linspace(0.0, 1.0, target_len)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    if new.sum() == 0:
        new = np.full(target_len, 1.0 / target_len)
    else:
        new /= new.sum()
    return new

# Optimisation phases
# ----------------------------------------------------------------------


def adam_softmax_descent(
    a: np.ndarray,
    deadline: float,
    beta_start: float = 0.5,
    beta_end: float = 3000.0,
    lr: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smooth surrogate
    Lβ(a)=log Σ exp(β·conv(a,a)).
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = compute_score(best)

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    start = time.time()

    while True:
        now = time.time()
        if now >= deadline:
            break

        # exact objective (for bookkeeping)
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # annealed β
        elapsed = now - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max weights
        shifted = conv - max_conv          # max becomes 0
        exp_vals = np.exp(beta * shifted)  # all ≤ 1, safe
        w = exp_vals / exp_vals.sum()

        # gradient via FFT‑based correlation
        grad = 2.0 * beta * correlation_valid(w, cur)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cur = cur - step
        cur = project_to_simplex(cur)

        # occasional progress report
        if t % 25000 == 0:
            print(
                f"[Adam] it {t:,} | time {now - start:.1f}s | best {best_score:.6f} | β {beta:.1f}"
            )

    return best, best_score


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climb that keeps only improving moves."""
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)

    step_scale = 0.07
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best.copy()
        # two simple move families
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise numeric drift
        cand /= cand.sum()

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.08, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best, best_score

# Main entry point

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety buffer for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # Choose sequence length.  Longer vectors give a slightly lower constant.
    # ------------------------------------------------------------------
    if total_time > 850:
        n = 4096
    elif total_time > 600:
        n = 3072
    else:
        n = 2048
    print(f"Using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 0. Fallback (if a previous construction is provided)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fb_seq = np.asarray(fallback, dtype=float)
            fb_seq = resample_sequence(fb_seq, n)
            fb_val = compute_score(fb_seq)
            print(f"Fallback seed score = {fb_val:.6f}")
            best_seq = fb_seq
            best_val = fb_val
        except Exception as e:
            print("Fallback unavailable:", e)

    # ------------------------------------------------------------------
    # 1. Generate a diversified set of seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic windows
    seeds.append(generate_sin2(n))
    seeds.append(generate_hann(n))
    seeds.append(generate_blackman(n))
    seeds.append(generate_cosine(n))

    # beta‑family (both centre‑ and edge‑heavy)
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    # random Dirichlet draws
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc))

    # sparse spikes (often a good start for low autocorrelation)
    for k in (30, 60, 100):
        seeds.append(random_spike(n, k=k))

    # incorporate fallback if we have one of exactly the correct length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(best_seq.copy())

    # ------------------------------------------------------------------
    # 2. Evaluate seeds (exact objective) and keep the best one
    # ------------------------------------------------------------------
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)          # guard against tiny drift
        val = compute_score(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
            print(f"Seed improvement: score = {best_val:.6f}")

    cur_a = best_seq.copy()
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    # ------------------------------------------------------------------
    # 3. Main Adam‑based soft‑max gradient descent (major part of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + total_time * 0.60
    print("\n[Phase 1] Adam soft‑max descent")
    cur_a, cur_score = adam_softmax_descent(
        cur_a,
        phase1_deadline,
        beta_start=0.5,
        beta_end=3000.0,
        lr=0.04,
    )
    if cur_score < best_val:
        best_val = cur_score
        best_seq = cur_a.copy()
        print(f"After Phase 1 best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional short random‑restart local searches (if time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        remaining = deadline - time.time()
        # spend at most 15 s on each restart
        sub_deadline = time.time() + min(15.0, remaining / 2.0)
        # perturb current best
        sigma = 0.02
        pert = cur_a + sigma * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert = project_to_simplex(pert)

        pert, pert_score = adam_softmax_descent(
            pert,
            sub_deadline,
            beta_start=0.5,
            beta_end=2500.0,
            lr=0.04,
        )
        if pert_score < best_val:
            best_val = pert_score
            cur_a = pert
            print(f"Restart improvement: score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final hill‑climb polishing (uses the exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 2] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=2_000_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"After hill‑climb best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Return result (list of floats)
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")
    return final_seq
