The function evaluate_sequence first projects sequence: clip negative to zero, clip >1000 to 1000, then compute length n, compute convolution b = np.convolve(sequence, sequence) (full linear convolution). Compute max_b = max(b). Compute sum_a = sum of sequence. If sum_a < 0.01 => invalid. Returns 2 * n * max_b / (sum_a**2).

def expgrad_optimize(start_a, deadline, max_iters=2_000_000, step0=0.2):
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        j = k - np.arange(n)
        mask = (j >= 0) & (j < n)
        # gradient
        g = np.zeros_like(a)
        # gradient coefficient 2*a[j] for valid entries
        g[mask] = 2.0 * a[j[mask]]
        eta = step0 / np.sqrt(t)
        # exponentiated update
        a = a * np.exp(-eta * g)
        # renormalize to simplex
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum
        # periodically evaluate
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] expgrad improved to {best_score:.6f}")
    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score
```

"""
Improved search for a non‑negative step‑function that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

The algorithm proceeds in several time‑bounded phases:

0. Load the fallback candidate (global ``height_sequence_1``) if it exists.
1. Quick scan of several analytic families (including Beta‑, Dirichlet‑ and
   exponential shapes) plus random draws; keep the best.
2. Adam optimisation on a smooth approximation of the max (log‑sum‑exp)
   with a very large final smoothing parameter (β≈3000).
3. Exponentiated‑gradient (mirror‑descent) on the *exact* max‑objective.
4. LP‑based refinement (the classic “g‑step” from the reference) with an
   adaptive mixing weight.
5. Optional symmetrisation.
6. Final stochastic hill‑climb (simulated‑annealing) for the remaining budget.

All steps respect a global wall‑clock budget (default 1000 s) and return a
plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast for many calls)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective assuming ∑a = 1 (i.e. 2·n·max(a*a))."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (used by Adam optimiser)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families for the quick scan
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ 1 / ((i+½)(n‑½‑i))^α  – a smooth “U‑shape”."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i ∝ (i+½)^‑β – left‑heavy power law."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential: w_i ∝ exp(-γ·dist(i,ends))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def beta_family(n: int, a: float, b: float) -> np.ndarray:
    """Discretised Beta(α,β) density on {0,…,n‑1}."""
    x = np.linspace(0.0, 1.0, n)
    # offset by a tiny epsilon to avoid 0**negative.
    eps = 1e-12
    w = np.power(x + eps, a - 1.0) * np.power(1.0 - x + eps, b - 1.0)
    # numerical guard against overflow/underflow
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / w.sum()
    return w


def dirichlet_family(n: int, concentration: float) -> np.ndarray:
    """Random Dirichlet draw with all α equal (controls flatness)."""
    alphas = np.full(n, concentration, dtype=np.float64)
    a = np.random.dirichlet(alphas)
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Monotone decreasing random start (sorted power‑law)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U start."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 4.0,
                         beta_end: float = 3000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise  f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # log‑sum‑exp weights (numerically stable)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a : 2·(w * a_rev)  (central part)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def expgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the (exact) max‑conv term.
    Update rule: a_i ← a_i·exp(-η·g_i) and renormalise,
    where g_i = 2·a_{k_max-i} (or 0 if out of range).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)

        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # learning rate schedule (diminishing)
        eta = step0 / np.sqrt(t)

        # mirror‑descent step (exponential update)
        a = a * np.exp(-eta * g)

        # re‑normalize onto the simplex (maintains positivity)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Exp‑Grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (state‑of‑the‑art iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve max Σ g_j  s.t. (f * g)_k ≤ M  ∀k,   g ≥ 0,
    then mix f←(1−t)f + t·g_norm and project onto the simplex.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # build convolution matrix A (A @ g = f * g)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g → minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # solve with HiGHS (fast) – occasional time limit safeguard
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 15})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_000_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a small amount
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
