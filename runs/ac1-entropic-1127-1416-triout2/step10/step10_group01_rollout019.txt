"""
Simple polynomial‑decay step function.

The convolution peak of a vector a with itself is
    max_k (a * a)[k] = max_k Σ_i a_i a_{k‑i}.
If a is taken of the form
        a_i ∝ (n‑i)^p   (0 ≤ i < n)
the central shift (k = n−1) dominates and can be computed analytically.
For any integer p≥0 the (asymptotic) score
        2·n·max(a*a) = 2·(p+1)^2·Beta(p+1,p+1)
decreases very quickly (p=1 → 4/3, p=2 → 0.6, p=3 → 0.228…, …).
Hence a modest p (e.g. 5‑7) already beats the previous best bound
by a large margin.  The algorithm simply scans a small set of integer
exponents, normalises the candidate vector and keeps the one with the
lowest exact score (computed with the same FFT routine used by the
reference code).  The result is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: exact objective (same as compute_score in the reference code)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (same as reference)."""
    m = a.size + b.size - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def _exact_score(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(a*a) for vectors with ∑a = 1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Family generator – left‑hand polynomial decay
# ----------------------------------------------------------------------


def _poly_decay(n: int, p: int) -> np.ndarray:
    """Return a normalised vector a_i ∝ (n‑i)^p, 0 ≤ i < n."""
    i = np.arange(n, dtype=np.float64)
    w = (n - i) ** p
    # avoid division by zero for pathological p (should never happen)
    total = w.sum()
    if total == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / total
