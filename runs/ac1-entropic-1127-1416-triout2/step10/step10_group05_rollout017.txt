def hill_climb_incremental(a, deadline, max_moves=None):
    n = a.shape[0]
    conv = convolution_fft(a, a)  # size 2n-1
    best_a = a.copy()
    best_score = compute_score(a)  # = 2*n*conv.max()
    best_max = conv.max()
    step_scale = 0.1
    moves = 0
    last_refresh = 0
    refresh_interval = max(1000, n//10)  # something
    while True:
        if time.time() >= deadline:
            break
        if max_moves is not None and moves >= max_moves:
            break

        # identify max conv index
        k_max = int(conv.argmax())
        # compute contributions to max share
        i_start = max(0, k_max - (n - 1))
        i_end = min(n-1, k_max)
        idx = np.arange(i_start, i_end+1)
        # contributions using a
        contrib = a[idx] * a[k_max - idx]
        i_max = idx[np.argmax(contrib)]
        # pick destination j not equal i_max
        j = np.random.randint(n)
        if j == i_max:
            continue

        delta = step_scale * a[i_max]
        if delta <= 1e-12:
            # cannot move anything; break?
            # maybe shrink step_scale
            step_scale *= 0.5
            continue

        # store old values for conv update slice
        # use a_copy = a.copy()? we need a for linear part as original; we have not modified a yet.
        # 
        # Perform linear updates
        # note: Using original a, okay.
        conv[i_max:i_max + n] -= 2.0 * delta * a
        conv[j:j + n] += 2.0 * delta * a

        # quadratic corrections
        idx2 = 2 * i_max
        conv[idx2] += delta * delta
        idx2 = 2 * j
        conv[idx2] += delta * delta
        idx_cross = i_max + j
        conv[idx_cross] -= 2.0 * delta * delta

        # update a
        a[i_max] -= delta
        a[j] += delta
        # Ensure numerical nonnegativity (clip)
        if a[i_max] < 0:
            a[i_max] = 0.0

        # compute new max maybe updated; to avoid recompute whole max each iteration, could track but simpler recompute
        current_max = conv.max()
        # compute new score only if current_max less than best_max; we can compute task-specific score.
        if current_max < best_max:
            best_max = current_max
            best_score = 2.0 * n * best_max
            best_a = a.copy()
            # maybe increase step_scale a bit
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            # optionally decrease step scale
            step_scale *= 0.96

        moves += 1
        # periodic refresh
        if moves - last_refresh >= refresh_interval:
            conv = convolution_fft(a, a)
            # no update to best here
            last_refresh = moves

    return best_a, best_score

"""
Improved step‑function optimizer.

The search proceeds in four stages:

1. **Parameterised families** – exhaustive / random search over several
   1‑ and 2‑parameter families (exponential, power‑law, stretched‑exponential,
   mixed exponential×power, Beta, random Dirichlet).  The best vector is kept.

2. **Fast greedy hill‑climb on the exact max‑convolution** – the convolution
   is maintained incrementally so that each move costs O(n) instead of an FFT.
   A packet of mass is taken from the index that contributes most to the
   current maximal convolution entry and moved to the smallest entry.
   Only improving moves are kept; the step size adapts and the convolution is
   re‑computed from scratch periodically to eliminate numerical drift.

3. **Optional short smooth‑max descent** – if time permits a few
   exponentiated‑gradient iterations are performed (same code as the original
   routine) to polish the result.

4. **Final evaluation** – the best sequence found is returned as a plain
   Python list.  The whole routine respects the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector that already
    satisfies sum = 1:  2 · n · max(a*a)  where the convolution is full.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Families of candidate vectors

def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), λ≥0."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), p≥0."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def stretched_exponential_shape(n: int, lam: float, p: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i^p), with λ≥0, p>0."""
    i = np.arange(n, dtype=float)
    if lam <= 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * (i ** p))
    return w / w.sum()


def exp_power_mix_shape(n: int, lam: float, p: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)·(i+1)^(-p)."""
    i = np.arange(n, dtype=float)
    if lam <= 0.0:
        exp_part = np.ones_like(i)
    else:
        exp_part = np.exp(-lam * i)
    if p <= 0.0:
        pow_part = np.ones_like(i)
    else:
        pow_part = (i + 1.0) ** (-p)
    w = exp_part * pow_part
    return w / w.sum()

def scan_exponential_and_power(n: int, deadline: float):
    """Grid scan over exponential and power‑law families."""
    best_a, best_sc = None, float("inf")
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-6, -2, 120)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    p_grid = np.linspace(0.0, 6.0, 180)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def scan_beta_family(n: int, deadline: float):
    """Modest scan of the discrete Beta family."""
    best_a, best_sc = None, float("inf")
    a_grid = np.concatenate((
        np.logspace(-0.7, -0.1, 14),
        np.logspace(-0.1, 0.3, 14),
        np.linspace(0.5, 3.0, 12),

            if sc < best_sc:
                best_sc, best_a = sc, a_vec.copy()
    return best_a, best_sc


def scan_stretched_exponential(n: int, deadline: float):
    """Random‐sampled search over λ and p for a stretched exponential."""
    best_a, best_sc = None, float("inf")
    # coarse grid + random refinements
    lam_vals = np.logspace(-7, -2, 30)
    p_vals = np.linspace(0.5, 4.0, 30)

        for p in p_vals:

            a = stretched_exponential_shape(n, lam, p)

    # a few extra random points for extra coverage
    for _ in range(300):

        lam = 10 ** np.random.uniform(-7, -2)
        p = np.random.uniform(0.5, 4.0)
        a = stretched_exponential_shape(n, lam, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def scan_exp_power_mix(n: int, deadline: float):
    """Random search over the mixed exp·power family."""
    best_a, best_sc = None, float("inf")
    for _ in range(400):

def random_dirichlet_starts(n: int, count: int, deadline: float):
    """Generate a few random Dirichlet (uniform) vectors."""
    best_a, best_sc = None, float("inf")
    for _ in range(count):
        if time.time() >= deadline:
            break
        a = np.random.rand(n)
        a /= a.sum()
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc

# Incremental hill‑climb (fast exact max‑descent)
# ----------------------------------------------------------------------


def hill_climb_incremental(start_a: np.ndarray, deadline: float,
                           refresh_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy walk that moves mass from the index that contributes most
    to the current maximal convolution entry towards the smallest entry.
    The convolution vector is kept incrementally; a full recompute is
    performed every `refresh_interval` moves to avoid drift.
    """
    a = start_a.astype(float).copy()
    n = a.size
    conv = convolution_fft(a, a)          # full convolution, length 2n‑1
    best_a = a.copy()
    best_max = conv.max()
    best_sc = 2.0 * n * best_max
    step_scale = 0.12
    moves = 0
    last_refresh = 0

    while True:
        if time.time() >= deadline:
            break

        # current maximal index
        k_max = int(conv.argmax())
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k_max - idx]
        i_max = idx[np.argmax(contrib)]

        # destination = smallest entry (helps keep sum constant)
        j = np.argmin(a)
        if j == i_max:
            # pick second smallest to avoid a no‑op
            order = np.argsort(a)
            j = order[1]

        delta = step_scale * a[i_max]
        if delta <= 1e-14:
            # magnitude too small – shrink step scale and continue
            step_scale *= 0.5
            continue

        # Incremental update of convolution (linear part)
        factor = 2.0 * delta
        conv[i_max:i_max + n] -= factor * a
        conv[j:j + n] += factor * a

        # Quadratic correction
        conv[2 * i_max] += delta * delta
        conv[2 * j] += delta * delta
        conv[i_max + j] -= 2.0 * delta * delta

        # Update the vector
        a[i_max] -= delta
        a[j] += delta
        # Numerical safety
        if a[i_max] < 0.0:
            a[i_max] = 0.0

        # Check improvement
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_sc = 2.0 * n * best_max
            best_a = a.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        moves += 1
        # occasional full recompute to correct drift
        if moves - last_refresh >= refresh_interval:
            conv = convolution_fft(a, a)
            last_refresh = moves

    return best_a, best_sc
