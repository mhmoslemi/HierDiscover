#!/usr/bin/env python3
"""
Propose a step‑function (non‑negative vector) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of the following stages (all run within the
1000 s wall‑clock budget):

1. **Seed generation** – a diverse collection of hand‑crafted shapes
   (sin², β‑polynomials, linear ramps, sparse spikes) together with
   random Dirichlet draws.  The best seed (exact objective) is kept.

2. **Adam on a smooth max‑surrogate** – we minimise the log‑sum‑exp
   surrogate of the convolution max:

        Lβ(a) = log Σ exp(β·conv(a,a))

   The variables are the log‑softmax parameters `u` (so `a = softmax(u)`
   automatically satisfies the simplex constraints).  Adam with an
   annealed learning rate and a linearly increasing temperature β
   performs fast descent.  The best iterate (according to the true
   objective) is retained.

3. **Random‑restart Adam** – while time permits we start from many
   random seeds and run a short Adam optimisation; any improvement
   updates the global best.

4. **Projected sub‑gradient descent** – a classic refinement on the
   true max‑objective (identical to the previous state‑of‑the‑art).

5. **Simulated‑annealing hill climb** – stochastic mass transfers are
   accepted if they improve the score or with a temperature‑controlled
   probability otherwise, allowing the method to escape shallow local
   minima.

All phases respect the time limit, print occasional status messages
and finally return the best sequence found (as a plain list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large n)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(u: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(u)
    e = np.exp(u - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    p>0 clusters mass in the centre, p<0 pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_linear(n: int) -> np.ndarray:
    """Linear ramp 1,2,…,n normalised."""
    a = np.arange(1, n + 1, dtype=float)
    a /= a.sum()
    return a


def generate_reverse_linear(n: int) -> np.ndarray:
    """Reverse linear ramp n,n‑1,…,1 normalised."""
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: place k random spikes and normalise."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def adam_surrogate_descent(
    a0: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    lr0: float = 0.08,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam on the log‑sum‑exp surrogate of max(conv).
    Parameters are the usual Adam hyper‑parameters; the temperature β
    grows linearly from beta_start to beta_end over the allotted time.
    Returns the best a (as a probability vector) and its exact objective.
    """
    n = a0.shape[0]
    # initialise log‑softmax parameters
    u = np.log(a0 + 1e-32)          # avoid log(0)
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = a0.copy()
    best_score = start_score

    t = 0
    start = time.time()
    while time.time() < deadline:
        # current probability vector
        a = softmax(u)

        # surrogate temperature (linear schedule)
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # full convolution and soft‑max weights
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        shifted = conv - max_conv                # for numerical stability
        exp_vals = np.exp(beta * shifted)        # every entry ≤ 1
        w = exp_vals / exp_vals.sum()            # soft‑max distribution over 2n‑1 points

        # gradient wrt a of Lβ = log Σ exp(β conv)
        grad_a = 2.0 * beta * np.correlate(w, a, mode="valid")   # shape (n,)

        # map gradient to the u‑space (simplex‑aware)
        mu = np.dot(grad_a, a)                     # ⟨grad_a , a⟩
        grad_u = a * (grad_a - mu)                 # ∇_u Lβ

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (decays like 1/√t, scaled by lr0)
        lr = lr0 / np.sqrt(t)

        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact‑objective evaluation
        if t % 2000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # small log for tracking
                print(
                    f"[Adam] it {t:6d} | time {elapsed:6.1f}s | β {beta:7.1f} | best {best_score:.6f}"
                )
    # final check
    final_score = compute_score(best_a)
    if final_score < best_score:
        best_score = final_score
        best_a = best_a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient refinement (as in the previous state‑of‑the‑art)
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))                     # worst convolution index

        # sub‑gradient: g_i = 2*a_{k-i} for feasible i
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = np.maximum(cur_a, 0.0)
        cur_a /= cur_a.sum()                         # projection onto simplex

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = np.maximum(cur_a, 0.0)
            cur_a /= cur_a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb_simulated_annealing(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
    init_temp: float = 1e-3,
    decay: float = 0.99995,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts worsening moves with a
    temperature‑controlled probability (simulated annealing).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    cur_a = best_a.copy()
    cur_score = best_score

    # temperature schedule
    temp = init_temp
    step_scale = 0.08
    steps = 0

    while steps < max_steps and time.time() < deadline:
        cand = cur_a.copy()

        # a random move: either swap two entries or transfer a small mass
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]               # pure swap
        else:
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                # guard against numerical negativity
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise (floating‑point drift)
        cand /= cand.sum()
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        accept = False
        if delta < 0.0:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_a = cand
            cur_score = cand_score
            # keep track of the global best
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # enlarge step_scale modestly when we improve
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            # shrink step_scale on rejection
            step_scale *= 0.96

        # temperature decay
        temp *= decay
        steps += 1

        # occasional progress report
        if steps % 100_000 == 0:
            elapsed = time.time() - (deadline - (deadline - time.time()))
            print(
                f"[Hill] step {steps:7d} | best {best_score:.6f} | temp {temp:.2e}"
            )

    return best_a, best_score
