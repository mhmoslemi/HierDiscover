def lp_iterative_refine(f_start, deadline, max_iters=30, t_start=0.5, t_end=0.05):
    f = f_start.astype(np.float64).copy()
    best_f = f.copy()
    best_score = compute_score(f)
    n = f.shape[0]
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute current conv
        conv_f_f = convolution_fft(f, f)
        M = conv_f_f.max()
        # build constraint matrix
        A = build_conv_matrix_sparse(f)  # shape (2n-1, n)
        b_ub = np.full(2*n - 1, M, dtype=np.float64)
        # solve LP
        c = -np.ones(n)  # maximize sum(b)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs', options={'presolve': True, 'time_limit': max(1.0, (deadline - time.time()))})
        if not res.success:
            # fallback break
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b / sum_b
        # symmetrize g
        g = (g + g[::-1]) / 2.0
        g = project_to_simplex(g)
        # compute step size
        t = t_start * ((t_end / t_start) ** (it / (max_iters - 1))) if max_iters>1 else t_start
        f = (1 - t) * f + t * g
        # ensure symmetry and simplex
        f = (f + f[::-1]) / 2.0
        f = project_to_simplex(f)
        sc = compute_score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
            # print improvement
            print(f"[{time.time():.1f}] LP iter {it} improved to {best_score:.6f}")
    return best_f, best_score

"""
Propose a candidate sequence using a linear‑program based
convex‐combination iteration followed by a smooth‑max Adam
refinement and a final simplex‑projected local search.
The LP step solves
   max  Σ b_j   s.t. b ≥ 0, (f ∗ b)_i ≤ M,
where M = max (f ∗ f).  The new point g = b / Σ b is mixed
with the current f by a decreasing step‐size.  Repeating
the LP iteration quickly drives down the peak of the auto‑
convolution.  After a few LP passes we run the Adam optimiser
from the original solution (soft‑max parametrisation) to
smooth the max and finish with a short simulated‑annealing
hill‑climb.  The routine respects a 1000 s wall‑clock budget
and returns a plain list of non‑negative heights.
"""

import time
import numpy as np
from typing import List

# try importing scipy, else define a placeholder that raises.
try:
    from scipy.optimize import linprog
    import scipy.sparse as sp
except Exception as e:
    linprog = None
    sp = None
    print("Warning: SciPy not available, LP stage will be skipped.")


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast for n≈few‑thousand)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalized."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, normalized."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Decreasing vector obtained by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


def build_conv_matrix_sparse(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n convolution matrix T such that
        conv_f_b = T @ b
    Using a sparse representation constructed from the diagonals of f.
    """
    n = f.shape[0]
    m = 2 * n - 1
    offsets = [-l for l in range(n)]               # -0, -1, … , -(n‑1)
    data = [np.full(n - l, f[l], dtype=np.float64) for l in range(n)]
    # diags treats each array as the diagonal of length equal to its size.
    T = sp.diags(data, offsets, shape=(m, n), format='csr')
    return T


def lp_iterative_refine(start_f: np.ndarray,
                        deadline: float,
                        max_iters: int = 30,
                        t_start: float = 0.5,
                        t_end: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the linear programme
        max Σ b_j  s.t. b ≥ 0, (f * b) ≤ M,
    where M = max(f * f).  The solution b is renormalised to unit sum,
    symmetrised and mixed with the current f using a decaying step size.
    Returns the best vector encountered and its exact score.
    """
    if linprog is None:
        # SciPy not available – just return the start vector.
        return start_f, compute_score(start_f)

    f = start_f.astype(np.float64).copy()
    best_f = f.copy()
    best_sc = compute_score(f)
    n = f.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv_ff = convolution_fft(f, f)
        M = conv_ff.max()

        # Build the (2n‑1)×n convolution matrix for the constraints (f * b) ≤ M.
        A = build_conv_matrix_sparse(f)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear programme: maximise Σ b_j  → minimise -Σ b_j
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        # We give the solver a tiny time budget to avoid overrunning.
        time_left = deadline - time.time()
        # “highs” accepts a ‘time_limit’ option (seconds).  If unavailable we fall back.
        options = {"presolve": True}
        if time_left > 0.0:
            options["time_limit"] = max(0.5, time_left)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options=options)

        if not res.success:
            # LP failed (perhaps time out).  Abort further LP passes.
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        g = b / sum_b
        # enforce symmetry and simplex
        g = (g + g[::-1]) * 0.5
        g = project_to_simplex(g)

        # decay step size geometrically
        if max_iters > 1:
            t = t_start * ((t_end / t_start) ** (it / (max_iters - 1)))
        else:
            t = t_start

        f = (1.0 - t) * f + t * g
        f = (f + f[::-1]) * 0.5
        f = project_to_simplex(f)

        sc = compute_score(f)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
            print(f"[{time.time():.1f}] LP iter {it:02d} improved to {best_sc:.6f}")

    return best_f, best_sc


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.018,
                         sym_interval: int = 4000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth max

"""
Propose a candidate step‑function using a linear‑program based refinement,
followed by a smooth‑max Adam optimiser and a brief simulated‑annealing
hill‑climb.  The LP step solves

    max   Σ b_j   s.t.   b ≥ 0 ,   (f ∗ b)_i ≤ M

where M = max(f ∗ f).  The normalised solution g = b / Σb is mixed with the
current f by a geometrically decaying step size.  After a few LP passes
the vector is fine‑tuned with Adam on a smooth approximation of the
max‑convolution, symmetrised, and finally polished by a cheap SA walk.
The routine respects a 1000 s wall‑clock budget and returns a plain
Python list of non‑negative heights.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (fast for a few thousand)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (row‑vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# Analytic families (used for random starts)

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Decreasing vector obtained by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()

def build_conv_matrix_sparse(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n convolution matrix T such that conv_f_b = T @ b.
    The matrix is built from the diagonals of f (each diagonal contains
    the same coefficient f[l]).
    """
    n = f.shape[0]
    m = 2 * n - 1
    # offsets -0, -1, … , -(n‑1)
    offsets = [-l for l in range(n)]
    # each diagonal consists of n‑l copies of f[l]
    data = [np.full(n - l, f[l], dtype=np.float64) for l in range(n)]
    T = sp.diags(data, offsets, shape=(m, n), format='csr')
    return T


def lp_iterative_refine(start_f: np.ndarray,
                        deadline: float,
                        max_iters: int = 30,
                        t_start: float = 0.5,
                        t_end: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP

        max Σ b_j
        s.t.  b ≥ 0,
              (f ∗ b)_i ≤ M ,

    where M = max(f ∗ f).  The optimal b is normalised,
    symmetrised and mixed with the current f using a decaying step size.
    Returns the best vector seen and its exact score.
    """
    if linprog is None:                     # pragma: no cover
        # If SciPy is missing we simply return the starting point.
        return start_f, compute_score(start_f)

    f = start_f.astype(np.float64).copy()
    best_f = f.copy()
    best_sc = compute_score(f)
    n = f.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv_ff = convolution_fft(f, f)
        M = conv_ff.max()

        # Constraint matrix for (f ∗ b) ≤ M
        A = build_conv_matrix_sparse(f)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear program: maximise Σ b_j  → minimise -Σ b_j
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        # Give the solver a modest time allowance.
        time_left = deadline - time.time()
        options = {"presolve": True}
        if time_left > 0.0:
            options["time_limit"] = max(0.5, time_left)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options=options)

        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        g = b / sum_b
        # enforce symmetry and simplex constraints
        g = (g + g[::-1]) * 0.5
        g = project_to_simplex(g)

        # geometric decay of the mixing coefficient
        if max_iters > 1:
            t = t_start * ((t_end / t_start) ** (it / (max_iters - 1)))
        else:
            t = t_start

        f = (1.0 - t) * f + t * g
        f = (f + f[::-1]) * 0.5
        f = project_to_simplex(f)

        sc = compute_score(f)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
            print(f"[{time.time():.1f}] LP iter {it:02d} improved to {best_sc:.6f}")

    return best_f, best_sc

            # ----- c) Short simulated‑annealing hill‑climb -----
            if time.time() < deadline - 5.0:
                sa_opt, sc_sa = hill_climb_sa(
                    np.array(best_seq, dtype=np.float64),
                    deadline - 2.0,
                    max_steps=1_500_000,
                )
                if sc_sa < best_score:
                    best_score = sc_sa
                    best_seq = sa_opt.tolist()
                    print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Final symmetrisation & evaluation
    # --------------------------------------------------------------
    if best_seq is not None:
        arr = np.array(best_seq, dtype=np.float64)
        arr = (arr + arr[::-1]) * 0.5
        arr = project_to_simplex(arr)
        best_seq = arr.tolist()
        final_score = evaluate_sequence(best_seq)
        print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    else:
        # safety fallback (uniform vector)
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        final_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] uniform fallback score = {final_score:.6f}")

    return best_seq
